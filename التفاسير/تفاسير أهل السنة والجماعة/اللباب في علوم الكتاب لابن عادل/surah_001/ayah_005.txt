قوله تعالى : { إياك نعبد وإياك نستعين }<br />
إِيَّاكَ : كلمة ضمير خُصَّت بالإضافةِ إلى المُضْمَر ،   ويُسْتَعْمل مقدماً على الفعل .   وإيَّاكَ أَسْأَلُ ؛ ولا يُسْتَعملُ مؤخراً إلاّ منفصلاً ؛ فيُقالُ ،   ما عنيتُ إِلاَّ إِيَّاك .  <br />
وهو مفعولٌ مُقَدَّمٌ على  "  نعبد  "  قُدِّم للاختصاصِ ،   وهوَ واجِبُ الانفصالِ .  <br />
واخْتلَفُوا فيه : هَلْ هو مِنْ قَبِيل الأسماءِ الظاهرة أو المضمرة ؟ فالجمهورُ : على أنه مُضْمَرٌ .  <br />
وقال الزَّجَّاجُ{[271]} رحمه الله تعالى : هو اسمٌ ظاهر .  <br />
والقائِلُون بأَنَّهُ ضميرٌ اخْتَلَفُوا فيه على أربَعةِ أقوالٍ :<br />
أحدُها : أنه كلمةُ ضَمِيرٍ .  <br />
والثاني : عَلَى أَنَّ  "  إِيَّا  "  وَحْدَهُ ضَمِيرٌ ،   وما بَعْدَهُ اسمٌ مُضَافٌ إليه يبيّن ما يُرادُ به [ من تكلّم وغيبة وخطابِ ] .  <br />
وثَالِثُها : أَنَّ  "  إِيَّا  "  وحده ضميرٌ ،   وما بعده حُرُوفٌ تبين{[272]} ما يُرادُ به [ من تكلم وغيبة وخطاب ]{[273]} .  <br />
ورابعُها : أَنَّ  "  إيَّا  "  عمادٌ وما بعده هو الضميرُ ،   وشذّت إضافته إلى الظاهِرِ في قولِهِم :  "  إذا بلغ الرَّجُلَ السِّتِّينَ ،   فإياه وإِيَّا الشَّواب  "  بِإِضَافَةِ  "  إِيَّا  "  إلى  "  الشواب  "  ،   وهذا يُؤَيِّدُ قَوْلَ مَنْ جَعَلَ الكَافَ ،   والهاء ،   والياء في محلّ جر ،   إِذا قُلْتَ :  "  إِيَّاكَ إِيَّاه إِيَّايَ{[274]} "  وقد أَبْعَدَ بعضُ النَّحوِيِّينَ ،   فجعل له اشْتِقاقَاً ،   ثُمَّ قال : هَلْ هو مشتقٌّ من  "  أَوَّ  "  ؛ كقول الشاعر في ذلك : [ الطويل ]<br />
62- فَأَوِّ لِذِكْرَاهَا إِذَا ما ذَكَرْتُهَا- *** - .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .  {[275]}<br />
أَوْ منْ  "  آيَة  "  ؛ كقوله [ الرجز ]<br />
63- لَمْ يُبْقِ هَذَا الدَّهْرُ مِنْ آيَائِهِ-{[276]} *** -<br />
وهل وَزْنُه :  "  إفْعَل ،   أو فَعِيل ،   أو فَعُول  "  ثم صَيرَّه التصريفُ إلى صِيغةِ  "  إِيَّا  "  ؟ وهذا الذي ذكره لا يُجدِي فائدةً ،   مع أنَّ التصريفَ والاشتقاق لاَ يَدْخُلاَن في المتوغِّلِ في البناءِ وفيه لُغاتٌ : أَشْهرُها : كَسْرُ الهمزةِ ،   وتَشْدِيدُ اليَاءِ ،   ومنها ،   فَتْحُ الهمزةِ وإبدالُها هاء مع تشديدِ الياءِ وتَخْفِيفهَا ؛ قال الشَّاعر : [ الطويل ]<br />
64- فَهَيَّاكَ وَالأَمْرَ الَّذِي الَّذِي إِنْ تَرَاحَبَتْ- *** -مَوَارِدُهُ ضَاقَتْ عَلِيْكَ مَصَادِرُهْ{[277]}<br />
وقال بعضُهم :  "  إِيَّاكَ  "  بالتَخْفِيفِ مرغوبٌ عنه ؛ لأنه يَصِيرُ :  "  شَمْسَك نعبد  "  ؛ فإِنَّ إِيَاةَ الشمسِ : ضَوْؤُها - بكسر الهَمزةِ ،   وقد تُفْتَحُ .   وقيل : هي لها بمنزلةِ الهَالةِ للقمر ،   فإذا حذفت التاءَ ،   مَدَدْتَ ؛ قال : [ الطويل ]<br />
65- سَقَتْهُ إِيَاءُ الشَّمْسِ إِلاَّ لِثَاتِهِ- *** -أُسِفَّ فَلَمْ تَكْدِمْ عَلَيْهِ بإثْمِدِ{[278]}<br />
وقد قٌرِئ{[279]} ببعضِهَا شَاذًّا .  <br />
وللضَّمائِرِ تَقسيمٌ مُتَّسع لا يحتمله هذا الكتاب ،   وإنما يأتي في غُضُونِه ما يليقُ به .  <br />
و  "  نَعْبُدُ  "  فعلٌ مضارعٌ مرفوع ؛ لتجردِه من الناصبِ والجازمِ ،   وقِيل : لوقوعِه موقعَ الاسمِ ،   وهذا رأْيُ البصريين .  <br />
ومعنى المضارع المشابه ،   يعني : أنه أشبه{[280]} الاسمَ في حركاتِهِ ،   وسكناتِه ،   وعدَدَ حُرُوفِهِ ،   أَلاَ تَرَى أَنَّ  "  ضَارِباً  "  يُشْبه  "  يَضْرِب  "  فيما ذكرت ،   وأنه يشيع ويختصُّ في الأزمانِ كما يشيعُ الاسمُ ،   ويختص في الأَشْخاصِ ،   وفَاعِلُهُ مستترٌ وُجُوباً لما مَرّ في الاستعاذة .  <br />
والعبادَةُ : غايةُ التذللِ ،   ولا يستحقُّها إلا مَنْ له غايةُ الإِفْضَالِ ،   وهو الباري - تعالى - وهو أبلغ من العُبُوديةِ ؛ لأن العُبُودِيَِّة إظهار التذلُّلِ ،   ويُقالُ : طريقٌ مُعَبَّدٌ ،   أَيْ : مُذَلَّلٌ بالوطْء فيه .   وقال طَرَفَة{[281]} في ذلك : [ الطويل ]<br />
66- تُبَارِي عِتَاقاً نَاجِيَاتٍ وَأُتْبِعَتْ- *** -وَظِيفاً وظيفا فَوْقَ مَوْرٍ مُعَبَّدِ{[282]}<br />
ومنه : العَبْدُ ؛ لِذلَّتِهِ ،   وبَعيرٌ معبَّدٌ : أَيْ مُذلَّلٌ بِالقَطِرَان .  <br />
وقيل : العبادةُ التَّجَرُّدُ ،   ويُقالُ : عَبَدْتُ اللهَ - بالتخفيف فقط - وعَبَّدْتُ الرجلَ - بالتشديدِ فقط ،   أَيْ : ذللتُه ،   واتخذتُه عبداً .  <br />
وفي قوله تعالى :  "  إِيَّاكَ نَعْبُد  "  التفاتٌ مِنَ الغَيْبَةِ إلى الخِطَابِ ،   إِذْ لو جرى الكلامُ على أصله ،   لَقِيل : الحمد لله ،   ثم قيل : إِيَّاهُ نَعبدُ ،   والالتفاتُ : نوعٌ مِن البلاغَةِ .  <br />
قال ابنُ الخَطِيب - رحمه الله - : والفائدةُ في هذا الالتفاتِ وجوه :<br />
أحدُها : أن المصلِّي كان أَجْنَبِيًّا عند الشروعِ في الصَّلاةِ ،   فلا جَرَمَ أَثْنَى على الله - تعالى - بألفاظ الغيبة ،   إلى قوله :  "  يَوْمِ الدِّينِ  "  ،   ثم إنه تعالى كأنه قال له : حَمَدْتَنِي وأَقْرَرْتَ بكونِي إلهاً ،   ربًّا ،   رحماناً ،   رحيماً ،   مالكاً ليوم الدين ،   فَنعْمَ العبدُ أنت ،   فرفعنا الحجابَ ،   وأبدلنا البُعدَ بالقُرْبِ ،   فتكلّم بالمخاطبة وقل : إياك نعبد .  <br />
الثاني : أنّ أحسنَ السؤالِ ما وقع على سبيلِ المُشَافَهَةِ ،   [ والسبب فيه أن الردَّ مِنَ الكريمِ إذا سُئِل ]{[283]} على سبيل المشَافهة والمخاطبة بَعِيدٌ .  <br />
ومن الالتفَاتِ - إلاّ كونه عَكْسَ هذا - قولُه تبارك وتعالى :{ حَتَّى إِذَا كُنتُمْ فِي الْفُلْكِ وَجَرَيْنَ بِهِم } [ يونس : 22 ] ولم يَقُلْ :  "  بكم  "  ؛ وقد التفتَ امرؤ القَيْسِ ثَلاثَ التفاتاتٍ في قوله : [ المتقارب ] .  <br />
67- تَطَاوَلَ لَيْلُكَ بَالأَثْمُدِ- *** -وَنَامَ الخَلِيُّ وَلَمْ تَرْقُدِ<br />
وَبَاتَ وبَاتَتْ لَهُ لَيْلَةٌ- *** -كَلَيْلَةِ ذِي العَائِرِ الأَرْمَدِ<br />
وَذَلِكَ مِنْ نَبَأٍ جَاءنِي- *** -وَخُبِّرْتُهُ عَن أَبِي الأَسْوَدِ{[284]}<br />
وقد خَطَّأَ بعضُهم الزمخشريَّ - رحمه الله تعالى - في جَعْلِه هذا ثَلاثةَ التفاتَاتٍ ،   وقال : بل هما التفاتان :<br />
أحدُهما : خُروجٌ مِنَ الخطابِ المفتتح به في قولِهِ :  "  لَيْلُك  "  ،   إلى الغَيْبَةِ في قوله :  "  وبَاتَتْ له لَيْلَةٌ  "  .  <br />
والثاني : الخروجُ من هذه الغيبةِ إلى التكلُّم ،   في قولِه :  "  مِنْ نَبَأ جَاءَنِي وخُبِّرْتُهُ  "  .  <br />
والجوابُ : أَنَّ قولَه أَوّلاً :  "  تَطَاوَلَ لَيْلُك  "  فيه التفاتٌ ؛ لأنه كان أصل الكلامِ أَنْ يقولَ :  "  تطاول لَيْلِي  "  ؛ لأنه هو المقصودُ ،   فالتفتَ مِنْ مقامِ التكلُّمِ إلى مقامِ الخِطَابِ ،   ومن مقامِ الخِطَابِ إلى الغيبةِ ،   ثُمَّ مِنَ الغيبةِ إلى التكلُّمِ الذي هو الأصل .  <br />
وقُرِئ شاذًّا{[285]} :  "  إِيَّاكَ يُعْبَدُ  "  على بنائِهِ للمفعول الغائبِ ؛ ووجهُهَا على إِشْكالِهَا : أن فيها استعارةً والتفاتاً :<br />
أما الاستعارةُ : [ فإنه اسْتُعِير ]{[286]} فيها ضميرُ النصبِ لضمير الرفْعِ ،   والأصل : أنت تُعْبَد ،   وهو شائع ؛ كقولِهم :  "  عَسَاكَ ،   وعَسَاهُ ،   وعَسَانِي  "  في أحدِ الأقوالِ ؛ وقول الآخر : [ الرجز ]<br />
68- يَا ابْنَ الزُّبَيْرِ طَالَمَا عَصَيْكَا- *** -وَطَالَمَا عَنَّيْتَنَا إِلَيْكَا{[287]}<br />
فالكافُ في  "  عَصَيْكَا  "  نائبةٌ عن التاءِ ،   والأصل :  "  عَصَيْتَ  "  .  <br />
وأما الالتفاتُ : فكان من حَقِّ هذا القارىء أَنْ يَقْرأَ :  "  إِيَّاكَ تُعْبَدُ  "  بالخطابِ ،   ولكنه التفت من الخطاب في  "  إِيَّاكَ  "  إلى الغَيْبَةِ في  "  يُعْبَدُ  "  إلاّ أن هذا الالتفاتَ غَرِيبٌ ؛ لكونِهِ في جُمْلةٍ واحدةٍ ،   بخلاف الالتفاتِ المتقدّمِ ؛ ونظيرُ هذا الالتفات قولُه : [ الطويل ]<br />
69- أَأَنْتَ الهِلاَلِيُّ الَّذِي كُنْتَ مَرَّةً- *** -سَمِعْنَا بِهِ وَالأَرْحَبِيُّ المُغَلَّبُ{[288]}<br />
فقال :  "  بِهِ  "  بعد قوله :  "  أَنْتَ  "  و  "  كُنْتَ  "  .  <br />
و  "  إيَّاكَ  "  واجبُ التقديم على عامله ؛ لأَنَّ القاعدَةَ أَنَّ المفعولَ به إذا كان ضميراً - لو تأخر عم عامله - وجب اتصالُهَ وَوَجَبَ تَقْدِيمُهُ .  <br />
وتَحَرَّزُوا بقولِهِم :  "  لو تأَخَّرَ عنه وجب اتصالُه  "  ،   من نحو :  "  الدرهم إياه أعطيتك  "  لأنك لو أخرتَ الضميرَ هنا فقلتَ :  "  الدِّرْهَمُ أَعْطَيْتُكَ إِيَّاهُ  "  لم يَلْزم الاتصالُ ،   لما سيأتي بل يجوزُ :  "  أعطيتكَهُ  "  .  <br />
فصل في معنى العبادة<br />
قال ابنُ الخَطِيب{[289]} رحمه الله تعالى : العبادةُ عبارةٌ عن الفعل الذي يؤتى به لغرضِ تَعْظِيم الغَيْرِ ،   من قولهم : طريقٌ مُعبَّدٌ ،   أَيْ : مذلّلٌ ،   فقوله : إِيَّاكَ نَعْبُدُ ،   معناه : لا أعبد أحداً سواك ،   ويدلّ على هذا الحصر وجوه :<br />
فَذَكَر من جملتها : تسميَة الله ،   والرب ،   والرحمن ،   والرحيم ،   ومالك يوم الدين ،   وكونَهُ قادراً بأن يُمْسِكَ السّماءَ بلا إعانة ،   وأَرْضاً بلا دِعَامة ،   ويُسَيِّرُ الشمسَ والقمر ،   ويسكن القُطْبَيْن ،   ويخرجُ من السَّماءِ تارة النَّارَ ؛ وهو البرق ،   وتارة الهواءَ ؛ وهو الريح ،   وتارة الماء ؛ وهو المطر .  <br />
وأما في الأرضِ فتارةً يُخْرج الماء من الحَجَرِ ؛ وتارةً يخْرج الحجرَ من الماء ؛ وهو الجمد ،   ثم جعل في الأرض أجساماً مُقيمةً لا تسافر ؛ [ وهي الجبال ]{[290]} ،   وأجساماً مسافرة لا تقيم ؛ وهي الأنهار ،   وخسف بقارون فجعل الأَرضَ فَوْقَهُ ،   ودفع محمداً - عليه الصلاة والسلام - إلى قَاب قَوْسَيْن ،   وجعل الماء ناراً لى قوم فرعون ؛ لقوله :<br />
{ أُغْرِقُواْ فَأُدْخِلُواْ نَاراً } [ نوح : 25 ] ،   وجعل النارَ بَرْداً وسلاماً على إِبْرَاهِيمَ - عليه السلام - ورفع مُوسَى - عليه السلام - فوق الطُّورِ ،   وغرق الدنيا من التّنُّورِ ،   وجعل البحر يبساً لموسى - عليه الصلاة والسلام - فهذا من أداة الحَصْرِ .  <br />
والكلام في  "  إِيَّاكَ نَسْتَعِينُ  "  كالكَلاَمِ في  "  إِيَّاكَ نَعْبُدُ  "  .  <br />
والواو : عاطِفَةٌ ،   وهي من المشتركةِ في الإعرابِ والمعنَى ،   ولا تقتضى تَرْتِيباً على قول الجمهور ،   خلافاً لطائفةٍ من الكُوفيينَ ولها أحكام تختص بها تأتي إن شاء الله تعالى .  <br />
وأصل  "  نَسْتَعِين  "  :  "  نَسْتَعْوِنُ  "  ؛ مَثْلُ :  "  نَسْتَخْرِجُ  "  في الصحيح ؛ لأنه من العَوْنِ ،   فاسْتُثْقِلَتِ الكسرةُ على الوَاوِ ،   فنقلت إلى السَّاكن قَبْلَها ،   فسكنت الواوُ بعد النَّقْلِ وانكَسَر ما قبلها ؛ فَقُلِبَتْ ياءً .  <br />
وهذه قاَعِدَةٌ مطّردَةٌ ؛ نَحْوُ :  "  مِيزَان ،   ومِيِقَات  "  ،   وهما من : الوَزْنِ ،   والوَقْتِ .  <br />
والسِّينُ فيه معناها : الطلبُ ،   أَيْ نَطْلُبُ منك العونَ على العِبَادَةِ ،   وهو أحد المعاني التي ل  "  استفعل  "  وله مَعََانٍ أُخَرٌ :<br />
الاتخاذُ : نحو :  "  اسْتَعْبِدْهُ  "  أي : اتخذْهُ عبداً .  <br />
والتحولُ ؛ نحو :  "  اسْتَحْجَرَ الطِّينُ  "  أَيْ : صار حجراً ،   ومنه قوله :  "  إِنَّ الْبُغَاثَ بأَرْضِنَا يَسْتَنْسِرُ  "  أي : تتحولُ إلى صفة النُّسور .  <br />
ووجُودُ الشَّيْءِ بمعنى ما صِيغَ منه ؛ نحو :  "  اسْتَعْظَمَهُ  "  أَيْ : وجده عظيماً .  <br />
وعَدُّ الشَّيْء كذلك ،   وإِن لم يكُنْ ؛ نحو :  "  اسْتَحْسَنَهُ  "  .  <br />
ومطاوعةُ  "  أَفْعَل  "  ؛ نحو :  "  أَشْلاَه فَاسْتَشْلَى  "  .  <br />
وموافقتُه له أيضاً ؛ نحو :  "  أَبَلَّ الْمَرِيضُ وَاسْتَبَلَّ  "  .  <br />
وموافقةُ  "  تَفَعَّلَ  "  ؛ نحو :  "  اسْتَكْبَرَ  "  بمعنى  "  تكبر  "  .  <br />
وموافقةُ  "  افْتَعَلَ  "  ؛ نحو :  "  اسْتَعْصَمَ  "  بمعنى  "  اعْتَصَمَ  "  .  <br />
والإِغْنَاءُ{[291]} عن المجرد ؛ نحو :  "  اسْتَكَفّ  "  و  "  اسْتَحْيَا  "  ،   لم يتلفظ لهما بمجردِ استغناء بهما عنه .  <br />
والإِغْنَاءُ بهما عن  "  فَعَلَ  "  أي المجرد الملفوظ به نحو :  "  اسْتَرْجَعَ  "  و  "  استعان  "  ،   أَيْ : رجع وحَلَق عانته .  <br />
وقُرئ{[292]} :  "  نِسْتَعينُ  "  بكسرْ حرف المضارعة ؛ وهي لُغَةٌ مطردةٌ في حروف المُضَارعة .  <br />
وذلك بشرط ألا يكن حرفُ المضَارعة ياءً ؛ لثقل ذلك ،   على أنَّ بعضَهُم قال :  " ييجَلُ "  ،   مضارع  "  وجَلَ  "  ،   وكأنه قصد إلى تَخْفِيفِ الواو إلى الياء ،   فكسر ما قبلها لتنقلب ؛ وقد قُرِئ{[293]} :  "  فإنَهم ييلَمُون  "  [ النساء : 104 ] ،   وهي هادمةٌ لهذا الاستثناءِ ،   وسيأتي تحقيقُ ذلك في موضعه إِنْ شاء الله تعالى .  <br />
وأن يكون المُضَارعُ من ماضٍ مكسورِ العَيْن ؛ نحو :  "  تِعْلَمُ  "  من  "  عَلِمَ  "  ،   أو في أوله همزةُ وصلٍ ،   نحو  "  نِسْتَعِينُ  "  من  "  اسْتَعَانَ  "  ،   أو تاءُ مُطَاوَعةٍ ؛ نحو :  "  نِتَعَلَّمُ  "  من  "  تَعَلَّمَ  "  ،   فلا يجوزُ في  "  يضْربُ  "  و  "  يقتلُ  "  كسر حرف المُضَارعة ؛ لعدم الشُّروط المذكورة .  <br />
والاستعانَةُ : طلبُ العَوْنِ : وهو المُظَاهرة والنصرة ،   وقدم العِبَادَةَ على الاسْتِعَانَةِ ؛ لأنها وصلةٌ لطلب الحاجة .  <br />
وقال ابنُ الخَطِيبِ : كأنه يقولُ : شَرَعْتُ في العبادَةِ ،   فأستعين بك في إتمامها ،   فلا تمنعني من إتمامها بالمَوْتِ ،   ولا بالمرضِ ،   ولا بقلب الدَّواعي وتَغَيُّرِهَا .  <br />
وقال البَغَوِيُّ - رحمه الله تعالى - فإن قِيل : لم قدم ذِكْرَ العِبَادةِ على الاستعانَةِ ،   والاستعانةُ لا تكون إلاَّ قبل العبادة ؟<br />
قلنا : هذا يلزمُ من جَعَلَ الاستعانَة قبلَ الفعل ،   ونحن نَجْعلُ التوفِيقَ ،   والاستعانَة مع الفعل ،   فلا فرق بين التقديم والتأخير .  <br />
وقيل : الاستعانَةُ نوعُ تعبُّدٍ ،   فكأنه ذكر جملة العبادَةِ أوّلاً ،   ثم ذكر ما هو من تفاصيلها وأطلق كُلاًّ من فِعْلَيْ العبادَةِ والاستعانَةِ فلم يذكر لهما مفعولاً ؛ ليتناول كلَّ معبود به ،   وكلَّ مُسْتَعان [ عليه ]{[294]} ،   أَوْ يكون المرادُ وقوعَ الفعلِ من غير نظر إلى مفعول ؛ نحو :<br />
{ كُلُواْ وَاشْرَبُواْ }[ البقرة : 60 ] أي أوقعوا هذين الفِعْلَيْنِ .  <br />
فصل في نظم الآية<br />
قال ابنُ الخَطِيْبِ{[295]} - رحمه الله تعالى - : قال تعالى :  "  إِيَّاكَ نَعْبُدُ  "  فقدَّمَ قولَه :  "  إيَّاك  "  على قوله :  "  نعبد  "  ولم يقل :  "  نعبدك  "  لوجوه :<br />
أحدُها : أنه - تبارك وتَعَالَى - قدّم ذِكْرَ نَفْسِهِ ؛ لينبه{[296]} العَابِدَ على أن المعبودَ هو اللهُ - تعالى - فلا يتكاسَلُ في التعظيم .  <br />
وثانيها : أَنَّه إِنْ ثقلت عليك العبادات والطاعات وصعبت ،   فَاذْكُرْ أوَّلاً قولَه :  "  إياك  "  ؛ لتذكرني ،   وتحضر في قلبك معرفتي ،   فإذا ذكرتَ جَلاَلي وعظمتي ،   وعلمتَ أني مولاك ،   وأنك عبدي ؛ سهلت عليك تلك العبادة .  <br />
وثالثها : أن القديمَ الواجبَ لذاتِه متقدمٌ في الوجودِ على المحدث الممكن لذاته ،   فوجب أن يكون ذكره متقدماً على جميع الأذكار .  <br />
فصل في نون  "  نعبد  " <br />
قال ابنُ الخَطِيب - رحمه الله تعالى - : لقائلٍ أَنْ يقولَ : النُّون في قوله تعالى :  "  نعبد  "  إما أن تكون نونَ الجمع ،   أو نونَ العظمةِ ،   والأول باطِلٌ ،   لأن الشخصَ الواحدَ لا يكون جَمْعاً ،   والثاني باطل أيضاً ؛ لأن عند أداء العبوديةِ ،   اللاَّئق بالإنسان أن يذكر نفسه بالعَجْز والذّلة لا بالعَظَمَةِ .  <br />
واعلم أنه يمكن الجوابُ عنه مِنْ وُجُوه :<br />
أحدها : أنّ المرادَ مِنْ هذه النونِ نونُ الجَمْعِ ،   وهو تنبيه على أنَّ الأَوْلَى بالإنسان ،   أَنْ يؤدي الصَّلاة بالجماعة .  <br />
الثاني : أنَّ الرجلَ إِنْ كان يُصَلِّي في جماعة ،   فقوله :  "  نعبد  "  ،   المُرَاد منه ذلك الجمعُ ،   وإن كان يصلّي وحده كان المراد أني أعبدك ،   والملائكةُ معي .  <br />
الثالث : أنَّ المُؤْمِنين إخوةٌ ،   فلو قال :  "  إياك أعبدُ  "  كان قد ذكر عبادَةَ نفسِه ،   ولم يذكر عبادَةَ غَيْرِه ،   أما إذا قال :  "  إياك نعبدُ  "  كان قد ذكر عبادَة نفسِه ،   وعبادة جميع المؤمنين شرقاً وغرباً .
