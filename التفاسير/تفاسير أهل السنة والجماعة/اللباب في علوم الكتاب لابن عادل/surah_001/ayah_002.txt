{ الْحَمْدُ للَّهِ رَبِّ الْعَالَمِينَ }<br />
قولُه تعالى : { الْحَمْدُ لِلَّهِ } .  <br />
الحمدُ{[153]} : الثناءُ على الجَمِيل سواءٌ كانت نِعْمةً مْبْتدأة إلَى أَحَدٍ أَمْ لاَ .  <br />
يُقال : حَمَدْتُ الرجلَ على ما أَنْعَمَ به ،   وحمدتُه على شَجَاعته ،   ويكون باللسانِ وَحْدَهُ ،   دون عمل الجَوَارِح ،   إذْ لا يُقالُ : حمدت زيداً أيْ : عملت له بيدي عملاً حسناً ،   بخلاف الشكر ؛ فإنه لا يكونُ إلاّ على نعمةٍ مُبْتَدأةٍ إلى الغير .  <br />
يُقال : شَكَرْتُه على ما أعطاني ،   ولا يُقالُ : شكرتُه على شَجَاعَتِه ،   ويكون بالقلبِ ،   واللِّسَانِ ،   والجَوَارح ؛ قال الله تعالى :<br />
{ اعْمَلُواْ آلَ دَاوُودَ شُكْراً }<br />
[ سبأ : 13 ] وقال الشاعرُ : [ الطويل ] .  <br />
37- أَفَادَتْكُمُ النَّعْمَاءُ مِنِّي ثَلاثَةً- *** -يَدِي وَلِسَانِي وَالضَّمِيرَ الْمُحجَّبَا{[154]}<br />
فيكونُ بين الحَمْدِ والشُّكْرِ عُمُومٌ وخُصُوصٌ من وجه .  <br />
وقيل : الحمدُ هو الشكر ؛ بدليلِ قولِهم :  "  الحمدُ لِلَّهِ شُكْراً  "  .  <br />
وقيل : بينهما عُمومٌ وخصوص مُطْلق .  <br />
والحمدُ أَعَمُّ مِنَ الشُّكْرِ .  <br />
وقيلَ : الحمدُ : الثناءُ عليه تعالى [ بأوصافِه ،   والشكرُ : الثناءُ عليهِ بِأَفْعَاله ]{[155]} فالحامِدُ قِسْمَانِ : شاكِرٌ ومُثْنٍ بالصفاتِ الجَمِيلة .  <br />
وقيل : الحمدُ مَقْلُوبٌ من المَدْحِ ،   وليس بِسَدِيدٍ - وإِنْ كان منقولاً عن ثَعْلَب ؛ لأنَّ المقلوبَ{[156]} أقلُّ اسْتِعْمالاً من المقلوب منه ،   وهذان مُسْتَوِيَان في الاستعمالِ ،   فليس ادعاءُ قلبِ أَحَدِهَما مِنَ الآخر أَوْلَى من العَكْس ،   فكانا مادّتين مُسْتَقِلَّتَيْن .  <br />
وأيضاَ فإنه يمتنعُ إطلاقُ المدْحِ حيثُ يَجُوزُ إطلاقُ الحَمْد ،   فإنه يُقالُ : حمدتُ الله - تعالى - ولا يقال : مَدَحْتُه ،   ولو كان مَقْلُوباً لما امتنع ذلك .  <br />
ولقائلٍ : أَنْ يَقُولَ : منع من ذلك مانِعٌ ،   وهو عَدَمُ الإِذْنِ في ذلك .  <br />
وقال الرَّاغِبُ{[157]} :  "  الحَمْدُ لله  "  : الثناءُ بالفَضِيلَةِ ،   وهو أخصُّ من المدحِ ،   وأَعَمُّ من الشُّكْرِ ،   فإنَّ المدْحَ يقال فيما يكونُ من الإنسانِ باختيارِه ،   وما يكونُ منه بغَيْرِ اختيار ،   فقد يُمْدَح الإنسان بطول قَامَتِهِ ،   وصَبَاحة وجهه ،   كما يمدح ببذل ماله وشجاعته وعلمه ،   والحمدُ يكونُ في الثَّاني دُونَ الأوّلِ .  <br />
قال ابنُ الخَطِيبِ - رحمه الله تعالى - : الفَرْقُ بين الحَمْدِ والمَدْحِ من وجوه :<br />
أحدها : أن المَدْحَ قد يحصلُ لِلْحَيِّ ،   ولغيرِ الحَيِّ? أَلاَ تَرَى أَنَّ من رَأَى لُؤْلُؤَةً في غايةِ الحُسْنِ ،   فإنه يَمْدَحُها ؟ فثبت أَنَّ المدحَ أَعمُّ من الحمدِ .  <br />
الثَّاني : أن المدحَ قد يكونُ قَبْلَ الإِحْسَانِ ،   وقد يكونُ بعدَه ،   أما الحمدُ فإنه لا يكونُ إلاَّ بعد الإحسان .  <br />
الثالث : أنَّا المدحَ قَدْ يكونُ مَنْهِياً عنه ؛ قال عليه الصلاةُ والسلامُ :  "  احْثُوا التُّرَابَ في وُجُوهِ المَدَّاحِينَ{[158]}  "  أما الحمدُ فإنه مأمورٌ به مُطْلَقاً ؛ قال - عليه الصلاة والسلام - :  "  مَنْ لَمْ يَحْمَدِ النَّاسَ لَمْ يَحْمَدِ اللهَ  " {[159]} .  <br />
الرابعُ : أنَّ المدحَ عبارةٌ عنِ القولِ الدَّالُّ على كونه مُخْتَصاً بنوع من أنواع الفَضَائل .  <br />
وأمّا الحمدُ فهو القولُ الدّالُّ على كونه مُخْتَصًّا بِفَضِيلة مُعَيَّنَةٍ ،   وهي فضيلةُ الإنعامِ والإحسان ،   فثبت أنَّ المدحَ أعمُّ من الحمدِ .  <br />
وأَمَّا الفرقُ بين الحمدِ والشُّكْرِ ،   فهو أَنَّ الحمدَ يَعُمُّ إذا وَصَلَ ذلك الإنْعَامُ إليك أَوْ إلَى غَيْرِك ،   وأما الشُّكْرُ ،   فهو مُخْتَصٌّ بالإنعامِ الواصلِ إليك .  <br />
وقال الرَّاغِبُ - رحمه الله - : والشكرُ لا يُقالُ إلاَّ في مُقَابلة نعمة ،   فكلُّ شُكْرٍ حَمْدٌ ،   وليس كُلُّ حمدٍ شُكْراً ،   وكل حمد مَدْحٌ ،   وليس كُلُّ مَدْحٍ حَمداً .  <br />
ويقال : فُلانٌ مَحْمُودٌ إذَا حُمِد ،   ومُحَمَّدٌ وُجِدَ مَحْمُوداً ،   ومحمد كثرت خصالُه المحمودَةُ .  <br />
واحمدُ أَيْ : أَنَّه يَفُوقُ غَيْرَه في الحَمْدِ .  <br />
والألفُ : واللام في  "  الحَمْد  "  قِيل : للاستغراقِ .  <br />
وقيل : لتعريفِ الجِنْس ،   واختاره الزَّمَخْشَرِيُّ ؛ وقال الشاعر : [ الطويل ]<br />

وقيل : للعَهْدِ ،   ومنع الزمخشريُّ كونَها للاستغراقِ ،   ولم يُبَيّنْ وجهةَ ذلك ،   ويشبه أن يُقالَ : إنَّ المطلوبَ من العبدِ إنشاء الحَمْدِ ،   لا الإخبار به ،   وحينئذٍ يَسْتَحيلُ كونها للاستغراقِ ،   إذْ لا يمكنُ العَبْد أن ينشىءَ جميعَ المَحَامِدِ منه ومن غيرِه ،   بخلاف كونها للجِنْسِ .  <br />

وحكى ابنُ الأَعْرَابِيِّ{[161]} جَمْعَهُ على  "  أَفْعُل  "  ؛ وأنشد : [ الطويل ]<br />
39- وَأَبْيَضَ مَحْمُودِ الثَّنَاءِ خَصَصْتُهُ- *** -بأَفْضَلِ أَقْوَالِي وَأَفْضَلِ أَحْمُدِي{[162]}<br />
وقرأ الجُمْهُورُ{[163]} :  "  الحَمْدُ لِلَّه  "  برفْعِ الدّال وكسرِ لاَمِ الجَرِّ ،   ورفعُهُ على الابتداءِ ،   والخبرُ الجارُّ والمجرورُ بعده يَتَعَلَّقُ بمحذوفٍ وهو الخَبَرُ في الحقيقة .  <br />
ثم ذلك المحذوفُ إنْ شئتَ قَدَّرْتَهُ [ اسْماً ،   وهو المُخْتارُ ،   وإنْ شِئْتَ قَدَّرْتَهُ ]{[164]} فِعْلاً أَي : الحمدُ مُسْتَقِرٌّ لله ،   واسْتَقَرَّ لِلَّه .  <br />
والدليلُ على اختيارِ القَوْلِ الأَوَّلِ : أَنَّ ذَلك يَتَعَيّنُ في بَعْضِ الصورِ ،   فلا أَقَلَّ مِنْ ترجيحِه في غَيْرِها ،   وذلك أنّك إذا قُلْتَ :  "  خرجتُ فإِذَا في الدَّار زَيْدٌ  "  و  "  أمَّا في الدارِ فَزَيْدٌ  "  يتعيّنُ في هاتَيْن الصُّورَتَيْنِ [ أن يقدر بالاسم ]{[165]}  ؛ لأَنَّ  "  إذا  "  الفُجائية وأَمَّا التَّفصِيلِيَّةُ لا يَلِيهِمَا إلاَّ المبتدأُ .   وقد عُورِضَ هذا اللَّفظُ بأنه يَتَعيَّنُ تقدير الفِعْلِ في بعض الصُّورِ ،   وهو ما إِذا وَقَعَ الجَارُّ والمجرورُ صِلَةً لموصولٍ ،   نحو : الَّذِي في الدارِ فليكن رَاجِحاً في غيره ؟<br />
والجوابُ : أَنَّ ما رجحنا به من باب المبتدإِ ،   أَو الخبر ،   وليس أَجْنَبِيًّا ،   فكان اعتباره أَوْلَى ،   بخلاف وقوعه صِلةً ،   [ والأول غيرُ أَجْنَبِيٍّ ]{[166]} .  <br />
ولا بُدَّ مِنْ ذِكْرِ قاعدةٍ -ها هنا- لعُمُومِ فائدتها ،   وهي أَنَّ الجار والمجرور والظرف إذا وَقَعا صلةً أو صِفَةً ،   أو حالاً ،   أو خبراً تعلقا بمحذوفٍ ،   وذلك المحذوف لا يجوز ظهوره إذا كان كَوْناً مُطلقاً : فأمّا قول الشاعر : [ الطويل ]<br />
40 - لَكَ الْعِزُّ إِنْ مَوْلاَكَ عَزَّ ،   وَإِنْ يَهُنْ- *** -فَأَنْتَ لَدَى بُحْبُوحَةِ الْهُونِ كَائِنُ{[167]}<br />
وأما قولُه تبارك وتَعَالى : { فَلَمَّا رَءَاهُ مُسْتَقِرّاً عِندَهُ } [ النمل : 40 ] فلم يقصدْ جعل الظَّرفِ كائناً فلذلك ذكر المتعلّقَ به ،   ثم ذلك المحذوفُ يجوزُ تقديرُه باسمٍ أَوْ فعلٍ إلاّ في الصلَةِ ،   فإنه يتعيّنُ أن يكونَ فِعْلاً{[168]} .   واختلفُوا : أَيُّ التقديرَيْنِ أَوْلَى فيما عدا الصور المستثناة ؟<br />
فقومٌ رجّحُوا تقديرَ الفِعْلِ ،   [ وقومٌ رجَّحُوا تقدير الاسمِ ]{[169]} ،   وقد تقدمَ دليلُ الفريقين .  <br />
وقُرِئَ{[170]} شَاذًّا بنصب الدالِ من  "  الحَمْد  "  ،   وفيه وجهان :<br />
أظهرُهُما : أنه منصوبٌ على المصدريَّةِ ،   ثم حُذِف العاملُ ،   ونابَ المصدرُ مَنَابه ؛ كقولِهِم في الأخبار :  "  حمداً ،   وشكراً لا كُفْراً  "  والتقدير :  "  أَحمد الله حمداً  "  ،   فهو مصدرٌ نَابَ عن جملةٍ خبريَّةٍ .  <br />
وقال الطَّبَرِيُّ{[171]} - رحمه الله تعالى - :  "  إنَّ في ضمنِهِ أَمْرَ عبادِه أَنْ يُثْنُوا به عليه ،   فكأَنَّه قال :  "  قولوا : الحَمْد للهِ  "  وعلى هذا يَجِيءُ قُولُوا : إِيَّاكَ  "  .  <br />
فعلى هذه العبارة يكونُ من المصادِر النائبَةِ عن الطَّلبِ لا الخبرِ ،   وهو محتملٌ للوجْهَيْنِ ،   ولكنْ كونُهُ خَبَرِيًّا أَوْلَى من كونه طَلَبياً ،   ولا يجوزُ إظهار الناصب ،   لئلاَّ يجمعَ بين البدلِ والمُبْدَلِ مِنْه .  <br />
والثاني : أنه منصوبٌ على المَفْعُولِ بهِ ،   أَي : اقْرَءُوا الحَمْدَ ،   أَو اتْلُوا الحَمْدَ ؛ كقولهم :  "  اللَّهُمَّ ضَبُعاً وَذِئْباً  "  ،   أَي : اجْمَعْ ضَبُعاً ،   والأوّلُ أَحْسَنُ ؛ للدَّلالَةِ اللفظيةِ .  <br />
وقراءَةُ الرفْعِ أمكنُ ،   وأَبْلَغُ مِنْ قراءَةِ النَّصب ،   لأنَّ الرفعَ في باب المَصَادِر التي أَصْلُها النِّيَابَةُ عَنْ أَفْعَالِها يدل على الثُّبُوتِ والاستقرَارِ ،   بخلافِ النَّصْبِ ،   فإنه يدلُّ على التجددِ والحُدوثِ ،   ولذلك قال العلماء - رحمهم الله - : إن جوابَ إِبْرَاهِيمَ - عليه الصلاة والسّلام - في قوله تَعَالَى حكايةً عنه :<br />
{ قَالَ سَلاَمٌ }[ هود : 69 ] أَحْسَنُ من قولِ الملائكة :<br />
{ قَالُواْ سَلاَماً }[ هود : 69 ] امتثالاً لقولِه تعالى :<br />
{ فَحَيُّواْ بِأَحْسَنَ مِنْهَآ }[ لنساء : 86 ] .  <br />
و  "  لله  "  على قراءةِ النصبِ{[172]} يتعلّقُ بمحذوفٍ لا بالمصدرِ ،   لأنَّها للبيانِ ،   تقديرهُ : أَعْنِي لله ،   كقولِهم :  "  سُقْياً له ورَعياً لك  "  تقديرُه :  "  أَعْنِي له ولك  "  ،   ويدلُّ على أنَّ اللام تتعلّقُ في هذا النوع بمحذوف لا بنفس المصدر ،   أنَّهم لم يُعْمِلُوا المصدر المتعدِّي في المجرور باللام ،   فينصبوه به فيقُولُوا : سُقْياً زيداً ،   ولا رَعْياً عمراً ،   فدلَّ على أنه ليس مَعْمولاً للمصدرِ ،   ولذلك غَلِطَ من جعل قولَه تَعَالَى :<br />
{ وَالَّذِينَ كَفَرُواْ فَتَعْساً لَّهُمْ }[ محمد : 8 ] ،   مِنْ بَابِ الاشْتِغَالِ ؛ لأنَّ  "  لَهُمْ  "  لا يتعلّقُ ب  "  تَعْساً  "  كما مَرَّ .  <br />
ويحتملُ أَنْ يُقالُ : إن اللام في  "  سُقياً لك  "  ونحوِهِ مقويةٌ لتعدِيَةِ العامل ؛ لكونِهِ فَرْعاً فيكون عاملاً فيما بعده .  <br />

الأصل : وَيْلٌ لأُمِّهَا ،   فحذفَ اللاَّمَ الأُولَى ،   واستثقَلَ ضَمَّةَ الهمزةِ بعد الكَسْرَةِ ،   فنقلَها إلى اللام بعد سَلْبِ حَرَكَتِها ،   وحذَفَ الهَمْزَةَ ،   ثم أَتْبَعَ اللاَّمَ المِيمَ ،   فصار اللفظ :  "  وَيْلِمِّهَا  "  .  <br />
ومِنْهم مَنْ لا يُتْبِعُ ،   فيقول :  "  وَيْلُمِّهَا  "  بِضَمِّ اللاَّمِ ،   قال : [ البسيط ]<br />
43 - وَيْلُمِّهَا خُلَّةً قَدْ سِيْطَ مِنْ دَمِهَا- *** -فَجْعٌ وَوَلْعٌ وَإِخْلافٌ وتَبْديلُ{[176]}<br />
ويحتملُ أَنْ تَكُونَ هذه القراءةُ مِنْ رَفْعٍ ،   وأَنْ تَكُونَ مِنْ نَصْبٍ ،   لأنَّ الإعرابَ مُقَدَّرٌ مَنَعَ من ظُهُورِهِ حَرَكَةُ الإتباعِ .  <br />
وقرئ أيضاً{[177]} :  "  لُلَّهِ  "  بضم لاَمِ الجَرِّ ،   قَالُو : وهي إتباعٌ لحركةِ الدَّالِ وفضّلها الزمخشريُّ على قراءة كَسْرِ الدَّالِ ،   مُعَلِّلاً لذلك بِأَنَّ إتباعَ حركَةِ الإعرابِ أَحْسَنُ مِنَ العَكْسِ ،   وهي لغةُ بَعْضِ  "  قَيْس  "  ،   يُتْبِعُون الثانِي نحو :  "  مُنْحَدُر ومُقُبِلِينَ  "  بضم الدَّال والقاف لأجل الميم ،   وعليه قرئ{[178]} :{ مُرُدفين } [ الأنفال : 9 ] بِضَمِّ الراءِ ،   إِتْباعاً لِلْمِيمِ .  <br />
فهذه أَرْبَعُ قِرَاءَاتٍ في  "  الحَمْدُ لِلَّهِ  "  .  <br />
ومعنى لام الجَرِّ - هنا - الاستحقاقُ أَيْ : الحمدُ مستحقٌّ لله - تعالى - ولها معانٍ أخر نَذْكُرُها وهي :<br />
المُلْكُ : المالُ لِزَيْدٍ .   والاستحقاقُ : الجُل لِلْفَرَسِ .   والتَّمْلِيكُ : نحو : وهبتُ لَكَ وَشِبْهُهُ نحو :{ جَعَلَ لَكُم مِّنْ أَنفُسِكُمْ أَزْوَاجاً }[ الشورى : 11 ] لتسكنوا إليها .  <br />
والنسب : نحو : لِزَيْدٍ عَمٌّ .  <br />
والتعليلُ : نحو :{ لِتَحْكُمَ بَيْنَ النَّاسِ }[ النساء : 105 ] ،  <br />
والتبليغُ : نحو قلتُ لك .  <br />
وللتعجُّبِ في القَسَمِ خاصَّةً ؛ كقوله : [ البسيط ]<br />
44- لِلَّهِ يَبْقَى عَلَى الأَيَّامِ ذُو حِيَدٍ- *** -بِمُشْمَخِرٍّ بِهِ الظَّيَّانُ وَالآسُ{[179]}<br />
والتَّبْيِينُ نحو قولِه تَعَالَى :{ هَيْتَ لَكَ }[ يوسف : 23 ] .  <br />
والصيرورةُ : نحو قولِهِ تَعَالَى :{ لِيَكُونَ لَهُمْ عَدُوّاً وَحَزَناً }[ القصص : 8 ] .  <br />
والظرفية إِمَّا بِمَعْنَى  "  فِي  "  : كقوله تعالى :{ وَنَضَعُ الْمَوَازِينَ الْقِسْطَ لِيَوْمِ الْقِيَامَةِ }[ الأنبياء : 47 ] ،   أَوْ بِمَعْنَى  "  عِنْدَ  "  : كقولِهم :  "  كَتَبْتُهُ لِخَمْسٍ  "  ،   أيْ : عِنْدَ خَمْسٍ ،   أَوْ بِمَعْنَى  "  بَعْدَ  "  : كقوله تعالى : { أَقِمِ الصَّلاَةَ لِدُلُوكِ الشَّمْسِ } [ الإسراء : 78 ] أيْ : بَعْدَ دُلُوكها .  <br />
والانتهاءُ : كقوله تعالى : { كُلٌّ يَجْرِي لأَِجَلٍ مُّسَمًّى } [ الرعد : 2 ] .  <br />
والاستعلاءُ : نحو قوله تعالى : { وَيَخِرُّونَ لِلأَذْقَانِ }[ الإسراء : 109 ] .  <br />
وقد تُزَادُ باطّرادِ في معمول الفعلِ مُقدَّماً عليه ؛ كقولِه تَعَالى :{ إِن كُنتُمْ لِلرُّؤْيَا تَعْبُرُونَ } [ يوسف : 43 ] ،   [ وإذا ] كان العامِلُ فرعاً ،   نحو قوله تعالى :{ فَعَّالٌ لِّمَا يُرِيدُ } هود : 107 ] .  <br />
وَبِغَيْرِ اطرادٍ ؛ نحو قوله في ذلك البيت : [ الوافر ]<br />
45- فَلَمَّا أَنْ تَوَاقَفْنَا قَلِيلاً- *** -أَنَخْنَا لِلْكَلاَكِلِ فَارْتَمَيْنَا{[180]}<br />
وأما قولُه تَعَالَى :{ عَسَى أَن يَكُونَ رَدِفَ لَكُم }[ النمل : 72 ] ،   فقِيل : على التَّضْمِينِ ،   وقِيلَ : هي زَائِدَةٌ .  <br />
ومن الناسِ مَنْ قال : تقديرُ الكَلام : قُولُوا : الحمد لله .  <br />
قال ابنُ الخَطِيب : - رحمه الله تعالى - : وهذا عندي ضعيفٌ ؛ لأنْ الإضمارَ إنما يُصَار إليه ليصحّ الكلامُ ،   وهذا الإضمار يُوجِبُ فسادَ الكَلام ،   والدليل عليه : أن قوله - تعالى -  "  الحَمْدُ لِلَّهِ  "  إخبارٌ عن كونِ الحَمْدِ حقًّا [ لله تعالى ] وملَكاً له ،   وهذا كَلاَمٌ تام في نفسه ،   فلا حاجةَ إلى الإضمار .  <br />
وأيضاً فإن قولَه :  "  الحمد لله  "  يدلُّ على كونِهِ مُسْتَحقاً للحمدِ بحسب ذاته ،   وبحسبِ أَفْعالِه ،   سواءٌ حَمَدُوه أَوْ لَمْ يَحْمِدُوه .  <br />
قال ابنُ الخَطِيب{[181]} : - رحمه الله تعالى - :  "  الحَمْدُ لِلَّه ثمانيةُ أَحْرُفٍ ،   وأبوابُ الجَنَّةِ ثمانية [ أبواب ]{[182]} ،   فمن قال :  "  الحمد لله  "  بصفاءِ قَلْبِهِ استحقَّ أَبْوابَ الجَنَّةِ الثمانية  "  والله أعلم .  <br />
فَصْلٌ<br />
تمسّكَ الجَبْرِيَّةُ والقدريَّةُ بقوله تعالى :  "  الحمدُ لِلَّهِ  "  أما الجبريةُ فقد تمسَّكوا به من وجوه :<br />
الأولُ : أَنَّ كُلَّ مَنْ كَان فِعْلهُ أشْرَفَ وأَكْمَل ،   وكانت النعمةُ الصادِرَةُ عنه أَعْلَى وأفضل ،   كان استحقاقُه للحمدِ أكثرَ ،   ولا شك أنَّ أَشْرَفَ{[183]} المخلُوقَاتِ هو الإيمانُ ،   فلو كان الإيمانُ فِعْلاً للعبد ،   لكان استحقاقُ العبدِ للحمدِ أَوْلَى وأجلَّ مِنِ اسْتِحْقاقِ الله له ،   ولما لم يكنْ كذلك ،   علمنا أنَّ الإيمانَ حَصَلَ بخلقِ الله - تعالى - لا بِخَلْقِ العَبْدِ .  <br />
الثاني : أجمعتِ الأمّةُ على قولِهم :  "  الحمدُ لِلَّهِ على نعمةِ الإيمانِ  "  ،   فلو كان العَبْدُ فاعلاً للإيمانِ لَكَان قولُهم :  "  الحمد للَّه على نعمة الإيمان  "  ؛ باطلاً ،   فإنَّ حمد الفاعِل على ما لا يَكُون فِعْلاً له باطلٌ قَبِيحٌ ؛ لقوله تعالى :{ وَّيُحِبُّونَ أَن يُحْمَدُواْ بِمَا لَمْ يَفْعَلُواْ }<br />
[ آل عمران : 188 ] .  <br />
الثالثُ : أَن قوله تعالى :  "  الحمدُ للهِ  "  يدلُّ ظاهِرُهُ على أنَّ كُلَّ الحمدِ لله ،   وانه لَيْسَ لِغَيْر الله - تعالى - حَمْدٌ أَصْلاً ،   وإنما يكونُ كلُّ الحمدِ لله تعالى إذا كان كُلُّ النِّعمِ من اللهِ تعالى ،   والإيمانُ أَفْضَلُ النعم ،   فوجب أَنْ يكونَ الإيمانُ من الله تعالى .  <br />
الرابعُ : أَن قولَهُ{[184]} :  "  الحَمد لله  "  مَدْحٌ مِنْهُ لِنَفْسِهِ{[185]} ،   ومدحُ النَّفْسِ قَبيحٌ فيما بين الخَلْقِ ،   فلما بدأ كتابَهُ بمدحِ النفْسِ ،   دَلَّ ذلك على أَنَّ حالَهُ بخلافِ حَالِ الخلقِ ،   وأَنَّه يَحْسُن منه ما يقبحُ من الخَلْقِ ،   وذلك يدلُّ على أنه - تبارك وتعالى - مقدَّسٌ عن أن تُقَاس أفعالُه على أَفعالِ العِبَادِ .  <br />
الخامسُ : عند المعتزلةِ أفعالُه - تعالى - يجبُ أَنْ تكونَ حَسَنةً ،   ويجبُ أَنْ تكونَ لها صفةٌ زائِدَةٌ على الحُسْنِ ،   وإلا كان عبثاً ،   وذلك في حقه تعالى محالٌ ،   والزائدةُ على الحُسْنِ إمَّا [ أن تَكُونَ ]{[186]} واجِبةً ،   وإما أن تكونَ من باب التَّفَضُّلِ .  <br />
أما الواجبُ فهو مثلُ إِيصالِ الثواب ،   والعوض إلى المُكَلَّفِين .  <br />
وأما الذي يكونُ من باب التفضل ،   فهو مثلُ أنَّه يزيد على قَدْرِ الواجِبِ على سبيلِ الإحْسَانِ .  <br />
فنقولُ : هذا يَقْدّحُ في كونه - تعالى - مستحقاً للحمد ،   ويُبْطِلُ صحَّةَ قولنا : الحمدُ لله .  <br />
وتقريرهُ أن نقولَ : أما أداةُ الواجِباتِ ،   فإنه لا يفيد استحقاقَ الحَمْدِ ،   ألا ترى أنَّ مَنْ كان له على غيره دَيْنُ دِينارٍ ،   فأدّاه ،   فإنه لا يَسْتَحِقُّ الحمدَ ،   فلو أوجبنا على الله تعالى فعلاً ،   لكان ذلك الفعلُ مخلصاً [ له ]{[187]} عن الذَّمِّ ،   ولا يُوجِبُ استحقاقه للحمد .  <br />
وأما فِعْلُ التفضُّلِ فعند الخصم أنه يستفيد بذلكَ مزِيدَ حَمْدٍ ولو لم يصدرْ عنه ذلك الفعلُ ،   لما حَصَل له الحمدُ ،   فإذا كان كذلك كان ناقصاً لذاته مستكملاً بغيره ،   وذلك يمنع مِنْ كونه - تعالى - مُسْتحقاً للحمدِ والمدح .  <br />
السَّادسُ : قولُه : الحمدُ لله يدلُّ على أنه - تعالى - محمودٌ ،   فنقولُ : استحقاقُه للحمد والمدحِ إما أن يكونَ أَمْراً ثابتاً لذاته ،   أو ليس تابثا لذاته ،   فإنْ كان الأوّل ،   امتنَع ثُبوتُه لغيره ،   فامتنع - أيضاً - أنْ يكون شَيْءٌ من الأفعالِ موجباً له استحقاق الذم ؛ لأن ما ثبت لِذَاتِهِ امتنع ارتفاعه ،   فوجب أَلاَّ يجبَ للعباد عليه شيءٌ مِنَ الأعْواضِ والثَّوَابِ ،   وذلك يَهْدِمُ أصولَ المعتزلة .  <br />
وأمّا القسم الثَّاني - وهو أن يكون استحقاق الحمد لله ليس ثابتاً لذاتِهِ - فنقول : فيلزم منه أن يكونَ ناقِصاً لذاته مُسْتكملاً بغيره ،   وذلك على الله - تعالى - محالٌ .  <br />
أما قول المعتزلة : إنَّ قَوْلَهُ :  "  الحَمْدُ لِلَّهِ  "  لا يتم إلاَّ على قولِنَا ؛ لأن المستحقَّ للحمدِ على الإطْلاقِ هو الذي لا قَبِيحَ في فِعْلِهِ ،   ولا جَوْرَ في قَضِيَّتِهِ ،   ولا ظُلمَ في أحكامِهِ ،   وعندنا أنَّ الله - تعالى - كذلك ؛ فكان مُسْتَحِقًّا لأعظمِ المَحَامِدِ والمدائح .  <br />
أمّا على مذهب الجَبْرِ لا قَبِيحَ إلا وهو فِعْلُه ،   ولا جَوْرَ إلا وهو حُكْمُه ،   ولاَ عَبَثَ إلا وهو صُنْعُه ؛ لأنه يخلقُ الكُفْرَ في الكافر ،   ثم يعذبُه عليه ،   ويؤلم الحيواناتِ مِنْ غَيْرِ أن يُعَوِّضَهَا ،   فكيف يُعْقلُ على هذا التقدِيرِ كونُه مُسْتحقاً للحمد ؟<br />
وأيضاً ذلك الحمد الذي يستحقه الله - تعالى - بسبب الإلهيَّة ؛ إِمَّا أن يستحِقَّهُ على العبدِ ،   أَوْ عَلَى نفسه ،   فإن كان الأول وجب كونُ العبدِ قادراً على الفِعْلِ ؛ وذلك يُبْطِلُ القول بالجَبْرِ .  <br />
وإنْ كان الثاني كان معناه أن الله تعالى يجب عليه أن يحمد نَفْسَهُ ؛ وذلكَ بَاطِلٌ ،   قالوا : فثبت أَنَّ القولَ بالحَمْدِ لا يصحُّ إلا على قولنا .  <br />
فَصْل هل وجوب الشكر يثبت بالعقل أو الشرع ؟<br />
اختلفوا في أَنَّ وُجُوبَ الشُّكْرِ ثابِتٌ بالعَقْلِ أَوْ بالسَّمْعِ .  <br />
مِنَ الناس مَنْ قال : إنه ثابِتٌ بالسَّمْعِ ،   لقوله تبارك وتعالى :{ وَمَا كُنَّا مُعَذِّبِينَ حَتَّى نَبْعَثَ رَسُولاً }[ الإسراء : 15 ] ،   ولقوله تبارك وتعالى :{ رُّسُلاً مُّبَشِّرِينَ وَمُنذِرِينَ لِئَلاَّ يَكُونَ لِلنَّاسِ عَلَى اللَّهِ حُجَّةٌ بَعْدَ الرُّسُلِ }[ النساء : 165 ] .  <br />
ومِنْهم مَنْ قال : إنه ثَابِتٌ{[188]} قبلَ مَجيء الشرع ،   وبعد مجيئه على الإطلاقِ ؛ والدليلُ عليه قولُه تبارك وتعالى :  "  الحَمدُ لِلَّهِ  "  وبيانه من وجوه :<br />
الأولُ : أَن قولَه تعالى :  " الحمدُ لله "  يدلُّ على أن هذا الحمدَ حَقُّهُ ،   وملكُه على الإطْلاَقِ ،   وذلك يدل على أنّ ثبوت هذا الاستحقاقِ كان قبل مَجِيء الشرْع .  <br />
الثاني : أنه تعالى قال : { الْحَمْدُ لِلَّهِ رَبِّ الْعَالَمِينَ } [ الفاتحة : 2 ]  ؛ وقد ثَبَتَ في [ أصُول ]{[189]} الفِقْهِ أَنَّ تَرْتِيبَ الحُكْمِ على الوَصْفِ المناسِبِ ،   يَدُلُّ على كونِ الحُكْمِ مُعَلَّلاً بذلك الوصف ،   فها هنا أثبتَ الحَمْدَ لنفسه ،   ووصف نَفْسَهُ بكونِه رَبَّ العالَمِينَ رَحْماناً رَحِيماً بِهِم ،   مالكاً لعاقبةِ أمْرِهم في القيامَةِ ،   فهذا يدلُّ على أن استحقاقَ الحمدِ ثابتٌ - لله تعالى - في كل الأوقات ،   سواءٌ كان قَبْلَ مَجِيء النَّبي ،   أو بعده .  <br />
فصل<br />
قال ابنُ الخَطيب - رحمه الله تعالى - : تَحميدُ اللهِ - تعالى - ليس عبارةً عن قَوْلِنا : الحمدُ لِلَّهِ ؛ لأن قوْلنا :  "  الحمد للَّه  "  إخبارٌ عن حُصُولِ الحَمْد ،   والإخبارُ عَن الشيءِ مغايرٌ للخَبَرِ عنه ،   فنقولُ : حَمْدُ المنعم عبارةٌ عن كُلِّ فِعْلٍ يُشْعِرُ بتعظيم المنعم بسبب كَوْنِهِ مُنْعماً ،   وذلك الفعل : إما أن يكونَ فِعْلَ القلبِ ،   أو فعل اللّسَانِ ،   أوْ فِعل الجوارح .  <br />
أمَّا فعلُ القلبِ : فهو أنْ يَعْتَقِدَ فيه كونَهُ مَوْصُوفاً بصفات الكمال والإجْلاَل .  <br />
وأما فعل اللِّسان فهو أنْ يذكر ألفاظاً دالَّةً على كونه مَوْصُوفاً بصفات الكمال [ والإجلال ]{[190]} .  <br />
[ وأما فعل الجوارح ؛ فهو أنْ يأتي بأفعالٍ دالّةٍ على كَوْنِ المنعمِ مَوصُوفاً بصفات الكمال والإجلال ]{[191]} .  <br />
واعلم أن أهل العلم - رحمهم الله - افترقوا في هذا المقام فِرقاً كثيرةً :<br />
فمنهم مَنْ قال : إنه لا يجوزُ عقلاً أنُ يأمرَ الله عَبِيدَه بأن يَحْمَدُوه ،   واحتجوا عليه بوجوه :<br />
الأولُ : أن ذلك التحميدَ ،   إما أن يكونَ بناءً على إنْعَام وَصَل إليهم ،   أَوْ لا بِنَاءَ عليه ،   فالأول باطِلٌ ؛ لأن هذا يقتضي أنه - تعالى - طلبَ منهم على إنعامِهِ جَزَاء ومُكافَأةً ،   وذلك يقدحُ في كَمَال الكرم ،   فإنّ الكريم إذا أنعم لم يُطالِبْ بالمُكَافأة .  <br />
وأما الثاني : فهو إِتْعَابٌ لِلْغَيْرِ ابتِداءً ،   وذلك يُوجِبُ الظُّلْمَ .  <br />
الثاني : قالُوا : إِنَّ الاشتغالَ بهذا الحمدِ مُتْعِبٌ للحامدِ ،   وغيرُ نَافِعٍ للمحمُودِ ،   لأنه كامِلٌ لذاتِهِ ،   والكامل [ لذاته ] يستحيل أن يستكملَ بِغَيْره ،   فثبت أنَّ الاشتغال بهذا التحميدِ عَبَثٌ وضَرَرٌ ،   فوجب ألا يكونَ مَشروعاً .  <br />
الثالثُ : أنَّ مَعْنَى الإِيجَابِ : أنه لَوْ لم يفعلْ لاسْتَحَقَّ العذابَ ،   فإيجابُ حَمْدِ الله تعالى معناه : أنه لو لم تشتغلْ بهذا الحمدِ ،   لَعَاقَبْتُكَ ،   وهذا الحمدُ لا نفعَ لَهُ في حَقِّ اللهِ تبارك وتعالى ،   فكان معناه أن هذا الفعلَ لا فَائِدَة فِيه لأحدٍ ،   ولو تركته [ لعاقبتك ]{[192]} أَبَدَ الآبادِ ،   وهذا لا يَلِيقُ بالحَلِيم الكريم .  <br />
والفريقُ الثاني : قَالُوا : الاشتغالُ بِحَمْدِ الله - تعالى - سُوءُ أَدَبٍ من وجوه :<br />
الأولُ : أنه يَجْرِي مَجْرَى مقابلَةِ إحْسَانِ اللهِ بذلك الشُّكر القَلِيلِ .  <br />
والثاني : أنَّ الاشتغالَ بالشُّكْرِ لا يتأتى إلاَّ مع استحضارِ تلك النِّعَمِ في القلْبِ ،   واشتغالُ القلبِ بالنعم يمنعه من الاسْتِغْرَاق في مَعْرِفَة المُنْعِمِ .  <br />
والثالثُ : أنَّ الثناءَ على الله - تعالى - عند وُجْدَانِ النِّعمةِ يدلُّ على أنه إنَّما أَثْنَى عليه ؛ لأجْلِ الفوز بتلك النعم ،   وهذا الرَّجُلُ في الحَقِيقَةِ مَعْبُوده ،   ومَطْلُوبُه إنما هو تلك النِّعَمِ ،   وحظُّ النَّفسِ ،   وذلك مقامٌ نَازِلٌ .  <br />
وهذانِ مَرْدُودانِ بما تَقَدَّم وبأنَّ أفعالَهُ وأقوالَه وأسماءَهُ لا مدخل للعَقْلِ فيها ،   فقد سَمَّى رُوحَه مَاكِراً بقوله تعالى :{ وَمَكَرَ اللَّهُ وَاللَّهُ خَيْرُ الْمَاكِرِينَ }[ آل عمران : 54 ] ،   ومتكبراً وغيرَ ذلك ممَّا تقدّم في أسمائه من قوله تعالى :{ اللَّهُ يَسْتَهْزِىءُ بِهِمْ }[ البقرة : 15 ] وغَيْرِه .  <br />
فإن قِيل : إنما ورد ذلك مِنْ حيثُ المُقَابلة ،   قُلْنَا : نُسلِّمُ ،   ولكنه قد سمى نفسه به ،   ونحن لا يجوزُ لنا تسميتُهُ به .  <br />
وأما مِنْ حَيْثُ ورودُه في الشرع ،   فقال اللهُ تعالى :{ فَاذْكُرُونِي أَذْكُرْكُمْ وَاشْكُرُواْ لِي وَلاَ تَكْفُرُونِ }[ البقرة : 152 ] .  <br />
قولُه تَعَالَى : { رَبِّ الْعَالَمِينَ } .  <br />
الرَّبُّ : لُغَةً : السيدُ ،   والمَالِكُ ،   والثَّابِتُ والمَعْبُودُ ؛ ومنه قولُه : [ الطويل ]<br />
46- أرَبٌّ يَبُولُ الثُّعْلُبَانُ بِرَأْسِهِ- *** -لَقَدْ هَانَ مَنْ بَالَتْ عَلَيْهِ الثَّعَالِبُ{[193]}<br />
والمُصْلِحُ ،   وزاد بعضُهم أنه بمعنى : الصّاحبِ ؛ وأنشد القائل : [ الكامل ]<br />
47- قَدْ نَالَهُ رَبُّ الكِلاَبِ بِكَفِّهِ- *** -بِيضٌ رِهَابٌ ريشُهُنَّ مُقزَّعُ{[194]}<br />
والظاهِرُ أَنَّهُ - هنا - بمعنى المَالِك ،   فليس هو معنى زائداً .  <br />
وقيل : يكون بمعنى الخَالِقِ .  <br />
واختُلِفَ فيه : هل هو في الأصلِ وَصْفٌ أو مصدر ؟<br />
فمنهم من قال : [ هو وَصْفٌ أي صِفَة مشبهة بمعنى  "  مُرَبٍّ  "  ]{[195]} ،   ثم اختلف هَؤلاءِ في وزنه .  <br />
[ فمنهم من قال ]{[196]} : هو على  "  فَعِل  "  كقولك :  "  نَمَّ - يَنُمَّ - فهو نَمٌّ  "  من النّمام ،   بمعنى غَمّاز .  <br />
وقيل : وزنه  "  فَاعِل  "  ،   وأصلُه :  "  رَابٌّ  "  ،   ثم حُذِفت الألفُ ؛ لكثرةِ الاستعمالِ ؛ لقولِهم : رَجَلٌ بَارٌّ وَبَرٌّ .  <br />
ولقائلٍ أنْ يقولَ : لا نسلم أن  "  بَرّ  "  مأخوذ من  "  بَارّ  "  بل هما صِفتان مُسْتقلتَانِ ،   فَلاَ يَنْبَغِي أنْ يُدّعَى أنّ  "  ربًّا  "  أصله  "  رابٌّ  "  .  <br />
ومنهم مَنْ : قال إنه مَصْدرٌ  "  رَبَّهُ - يَرُبُّهُ - رَبًّا  "  أي : مَلَكَهُ .  <br />
قال :  "  لأنْ يَرُبَّنِي رَجَلٌ مِنْ قُرَيْشٍ أَحَبُّ إِلَيَّ من أن يَرُبَّنِي رَجَلٌ من هَوَازِنَ  "  .  <br />
فهو مصدر في معنى الفاعل نحو :  "  رجل عَدْل وصَوْم  "  .  <br />
ولا يُطْلقُ على غَيْر الباري - تعالى- إلاّ بقيد إضافةٍ ،   نحو قوله تعالى : { ارْجِعْ إِلَى رَبِّكَ }[ يوسف : 50 ] ،   ويقولون :  "  هو رَبُّ الدَّارِ ،   ورَبُّ البَعِير "  ،   وقد قالته الجاهليةُ لِلْمَلِكِ من الناس مِنْ غَيْرِ قَيْدٍ ؛ قال الحَارِثُ بْنُ حِلِّزَةَ{[197]} : [ الخفيف ]<br />
48- وَهُوَ الرَّبُّ وَالشَّهِيدُ عَلَى يَوْ- *** -مِ الحِيَارَيْنِ وَالبَلاءُ بَلاَءُ{[198]}<br />
وهذا مِنْ كُفْرِهِمْ .  <br />
وقرأ الجمهورُ :  "  رَبِّ  "  مجروراً على النعتِ  "  لله  "  ،   أو البَدَلِ منه .  <br />
وقرِئ مَنْصُوباً{[199]} ،   وفيه ثلاثةُ أَوْجُهٍ :<br />
إِمَّا بِمَا دَلَّ عليه الحمدُ ،   تقدِيرُه :  "  أحمد{[200]} ربَّ العالمين  "  .  <br />
أو على القطع من التبعية ،   أو على النِّدَاءِ وهذا أضعفُهَا ،   لأنه يُؤَدِّي إلى الفَصْلِ بين الصفة والموصوف .  <br />
وقُرِئَ{[201]} مَرْفُوعاً على القَطْعِ من التبعية ،   فيكونُ خبراً لمبتدإٍ مَحْذُوفٍ ،   أيْ :  "  هُوَ رَبُّ  "  وإذْ قد عرض ذِكْرُ القَطْعِ في التَبعيَّةِ ،   فلنستطردْ ذِكْرَهُ ،   لِعُمومِ فَائِدَتِه فنقول :<br />
اعلمْ أنَّ الموصوفَ إذا كان معلوماً بدون صفته ،   وكان الوصفُ مَدْحاً ،   أو ذماً ،   أو ترحُّماً - جاز في الوَصْفِ الإتباعُ والقطعُ .  <br />
والقطعُ : إما على النصْبِ بإضمار فعل لائقٍ ،   وإمَّا على الرّفعِ على خَبَرٍ لمبتدإٍ مَحْذُوفٍ ،   ولا يجوزُ إظهارُ هذا الناصِبِ ،   ولا هذا المبتدإِ ،   نحو قولِهم :  "  الحَمْدُ لله أَهْلَ الحَمْدِ  "  رُوِيَ بنصبِ  "  أَهْل "  ورفعِه ،   أيْ : أعني أَهْلَ ،   أو هو أَهلُ الحمدِ .  <br />
وإِذا تكررتِ النُّعُوتُ ،   والحالةُ هذه ،   كُنْتَ مُخَيَّراً بين ثلاثة أوجه :<br />
إما إتباعُ الجَميعِ ،   أو قَطْعُ الجَمِيع ،   أوْ قَطْعُ البَعْضِ ،   وإتباعُ البَعْضِ .  <br />
إلاّ أنك إذا أَتْبَعْتَ البعضَ ،   وقطعتَ البعضَ وجب أَنْ تَبْدَأ بالإتباعِ ،   ثُمَّ تأتي بالقَطْعِ من غير عَكْسٍ ،   نحو :  "  مررتُ بزيدٍ الفَاضِلِ الكَرِيمُ  "  ؛ لِئَلاَّ يلزمَ الفصلُ بين الصفَةِ والموصُوفِ بالجملةِ المَقْطُوعَةِ .  <br />
و  " العَالَمِينَ "  خَفْضٌ بالإضافَةِ ،   عَلاَمةُ خفضِه الياءُ ؛ لجريانه مَجْرى جمع المذكرِ السَّالِمِ ،   وهم اسْمُ جَمْعٍ ؛ لأنَّ واحِدَهُ مِنْ غَيْرِ لفظه ،   ولا يَجوزُ أن يكونَ جمعاً ل  " عَالَمٍ "  ؛ لأن الصحيح في  " عَالَمِ "  أن يُطْلَق على كل موجودٍ سوى الباري تَعَالى ،   لاشتقاقه من العَلاَمةِ ،   بمعنى أنه دالّ على صانِعهِ .  <br />
و  " عَالَمُون "  بصيغة الجمع لا يُطْلق إلا على العُقلاءِ دون غيرهم ،   فاستحال أن يكون  " عالمون  "  جمع  "  عَالَم  "  ؛ لأنَّ الجمعَ لا يكونُ أخَصَّ مِنَ المُفْرَدِ .  <br />
وهذا نظيرُ ما فعله سِيْبَويَه{[202]} - رحمه الله تعالى - في أنَّ  "  أعراباً  "  لَيْسَ جَمْعاً ل  "  عَرَب  "  ؛ لأنَّ  "  عرباً  "  يُطْلَقُ على البَدَوِيِّ دُونَ القروِيّ .  <br />
فإن قِيلَ : لِمَ لاَ يَجُوزُ أَنِ يكونَ  "  عَالَمُونَ  "  جَمْعاً ل  "  عَالَم  "  مُرَاداً به العاقل دُونَ غَيْره ،   فيزولَ المحذُورُ المذكور ؟<br />
وأُجِيبَ عنه : بأنه لَوْ جاز ذلك ،   لَجَازَ أَنْ يُقالَ :  "  شَيْئُون  "  جَمْعُ  "  شَيْءٍ  "  مُرَاداً به العاقل دون غيره ،   فدل عَدَمُ جَوَازِه على عدم ادّعاءِ ذلك .  <br />
وفي الجواب نَظَرٌ ،   إذْ لِقائل أنْ يقول :  "  شيئون  "  منع منه مانِعٌ آخرُ ،   وهو كونهُ لَيْسَ صِفَةً ولا علماً ،   فلا يلزَمُ مِنْ مَنْعِ ذلك منعُ  "  عَالَمِين  "  مراداً به العاقل .  <br />
ويُؤَيِّدُ هذا ما نَقَلَ الراغِبُ عَنِ ابنِ عَبَّاسٍ - رضى الله تعالى عنهما - أنَّ  "  عَالَمِين "  إنما جمع هذا الجمع ؛ لأن المراد به الملائكةُ والجنُّ والإنْسُ .  <br />
وقال الراغِبُ{[203]} أيضاً :  "  إنَّ العَالَم في الأصل اسم لما يُعْلَمُ به كالطَّابَعِ اسم لما يُطْبَعُ  "  وجُعِلَ بناؤُه على هذه الصيغَةِ ،   لكونه كالآلةِ ،   فالعالَمُ آلة في الدلالةِ على صَانعه .  <br />
وقال الرَّاغِبُ :  "  وأما جمعُه جَمْعَ السَّلامَةِ ،   فلكون الناس في جُمْلَتِهم ،   والإنسانُ إذا شَارَك غيرَهُ في اللّفظِ غَلَبَ حُكْمُه  "  ،   فظاهر هذا أَنَّ  "  العَالَمِين  "  يطلق على العُقَلاء وغَيْرِهم ،   وهو مُخالِفٌ لما تقدّم من اختصاصِهِ بالعقلاء ،   كما زعم بعضُهم ،   وكلام الراغِبِ هو الأصَحُّ الظّاهرُ .  <br />
فصل في وجوه تربية الله لعبده<br />
قال ابنُ الخَطِيبِ{[204]} - رحمه الله تعالى - :  "  وجوه تَرْبِيَةِ الله لِلْعَبْدِ كثيرةٌ غيرُ مُتَنَاهِيَةٍ ،   ونحن نذكر منها أمثلة :<br />
الأولُ : لما وَقَعَتِ النّطفةُ مِنْ صُلْبِ الأَبِ إلى رَحِمِ الأُمِّ ،   فَرَبَّاهَا حتى صارت عَلَقَةً أَولاً ،   ثم مُضْغَةً ثانيةً ،   ثم تولّدت منه أعضاء مُختلفةٌ ،   مثلُ العِظَامِ ،   والغَضَارِيفِ ،   والرّبَاطَاتِ ،   والأَوْتَارِ ،   والأوردَةِ ،   والشرايِين ،   ثم اتصل البعضُ بالبعضِ ،   ثُمَّ حَصَلَ في كُلِّ واحِدٍ منها نَوْعٌ خَاصٌّ من أنواع القُوَى ،   فحصلت القوّةُ الباصرة في العَيْنِ? والسَّامِعَةُ في الأُذُنِ ،   والنَّاطِقَةُ في اللِّسانِ ،   فسُبْحَانَ مَنْ بَصَّرَ بِشَحْمٍ ،   وأَسْمَعَ بِعَظْمٍ ،   وأَنْطَقَ بِلَحْمٍ  "  ! .  <br />
والثَّاني : أن الحَبَّةَ الواحِدَةَ إذا وقعت في الأرض ،   فإذا وصلت نَدَاوةُ الأرضِ إليها ،   انتفَخَتْ ولا تنشق من شيء من الجوانِبِ إلاّ مِنْ أَعْلاَها وأسفلها ،   مَعَ أنَّ الانتفاخَ حاصلٌ من جميع الجوانب .  <br />
أما الشق الأعلى ،   فيخرجُ منه الجزءُ الصاعِدُ ،   فبعد صعودِهِ يحصُلُ له سَاقٌ ،   ثم ينفصِلُ من ذلك الساقِ أَغْصَانٌ كثيرةٌ ،   ثم يظهر على تلك الأغصانِ الأَنْوَارُ أوَّلاً ،   ثُمَّ الثِّمَار ثانياً ،   ويحصل لتلك الثمار أجزاء مختلفة بالكَثَافَةِ ،   واللطافة ،   وهي القُشُور ،   واللّبوبُ ،   ثم الأدهان .  <br />
وأما الجُزْءُ الغائِصُ من الشجرة ،   فإن تلك العروق تنتهي إلى أطرافها ،   وتلك الأطراف تكون في اللطافة كأنها مياه منعقدة ،   ومع غايةِ لُطْفِها ،   فإنها تغوص في الأرض الصّلبة اليابسة ،   وأودع فيها قُوًى جاذبةً تجذِبُ اللّطيفةَ من الطين إلى نفسها ،   والحكمةُ في كُلّ هذه التدبِيرَاتِ تحصيلُ ما يحتاج العبد إليه من الغِذَاءِ ،   والإدام ،   والفواكه ،   والأشربةِ ،   والأدْوِيَةِ ؛ كما قال تعالى :{ أَنَّا صَبَبْنَا الْمَآءَ صَبّاً ثُمَّ شَقَقْنَا الأَرْضَ شَقّاً }[ عبس : 25 ،   26 ] .  <br />
فَصْلٌ<br />
اختلفوا في { الْعَالَمِينَ } .  <br />
قال ابن عباس رضي الله عنهما :  "  هُمُ الجنُّ والإِنْسُ{[205]} ؛ لأنهم المكلّفون بالخِطَابِ  "  ؛ قال الله تعالى :{ لِيَكُونَ لِلْعَالَمِينَ نَذِيراً }[ الفرقان : 1 ] .  <br />
وقال قتادةُ{[206]} : والحَسَنُ ،   ومُجَاهِدٌ - رضي الله تعالى عنهم - :  "  جميعُ المخلوقِينَ "  ؛ قال تبارك وتعالى :{ قَالَ فِرْعَوْنُ وَمَا رَبُّ الْعَالَمِينَ قَالَ رَبُّ السَّمَاوَاتِ وَالأَرْضِ وَمَا بَيْنَهُمَآ }[ الشعراء : 23 ،   24 ] .  <br />
واشْتِقَاقُه من العَلمِ والعَلاَمَةِ ،   سُمُّوا بذلك ؛ لِظُهورِ أَثَرِ الصنعة فيهم .  <br />
قال أَبُو عُبَيد{[207]} - رحمه الله تعالى - : هم أرْبَعُ أُمَمٍ : الملائكةُ ،   والإنسُ ،   والجِنُّ ،   والشَّيَاطِينُ ،   مُشْتَقٌّ من العلم ،   ولا يُقَال للبهائِمِ ؛ لأنها لا تَعْقِلُ .  <br />
واخْتَلَفُوا في مبلغِهِم .  <br />
قال سَعِيدُ بنُ المُسَيَّب{[208]}- رضي الله عنه - :  "  لِلَّه ألْفُ عَالَمٍ : سِتُّمائةٍ في البَحْرِ ،   وأربعمائة في البرِ{[209]}  "  .  <br />
وقال مُقَاتِلُ بْنُ حَيَّان{[210]} - رضي الله عنه - :  "  ثَمانُونَ أَلْفاً ،   أَرْبَعُون ألفا في البَحْرِ ،   وأربعونَ ألفاً في البَرِّ  "  .  <br />
وقال وَهْبٌ{[211]} - رضي الله عنه - :  "  لله ثَمَانِيَةَ عَشرَ ألفَ عَالَمٍ ،   الدّنيا عَالَمٌ منها ،   وما العمران في الخَرَابِ إلا كفُسْطاطٍ في صَحْرَاء{[212]} "  .  <br />
وقال كَعْبُ الأحبارِ - رضي الله عنه - :  "  لا يُحْصي عَدَدَ العَالَمِين إلاّ اللهُ - عز وجل -  "  ؛ قال تعالى :{ وَمَا يَعْلَمُ جُنُودَ رَبِّكَ إِلاَّ هُوَ }[ المدثر : 31 ] .
