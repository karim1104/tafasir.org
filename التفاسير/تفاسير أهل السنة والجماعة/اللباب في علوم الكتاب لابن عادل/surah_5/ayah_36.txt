قوله تعالى : { إِنَّ الَّذِينَ كَفَرُواْ لَوْ أَنَّ لَهُمْ [ مَّا فِي الأَرْضِ جَمِيعاً وَمِثْلَهُ مَعَهُ لِيَفْتَدُواْ بِهِ مِنْ عَذَابِ يَوْمِ الْقِيَامَةِ مَا تُقُبِّلَ مِنْهُمْ وَلَهُمْ عَذَابٌ أَلِيمٌ }*{ يُرِيدُونَ أَن يَخْرُجُواْ مِنَ النَّارِ وَمَا هُم بِخَارِجِينَ مِنْهَا وَلَهُمْ عَذَابٌ مُّقِيمٌ }{[11658]}<br />
قد تقدم الكلام على  "  أنَّ  "  الواقعة بعد  "  لو  "  على أنَّ فيها مَذْهَبيْن .  <br />
و  "  لَهُمْ  "  خبر ل  "  أنَّ  "  ،   و { مَا فِي الأرْض } اسمها ،   و  "  جميعاً  " {[11659]} توكيد لهُ ،   أو حال منه و  "  مِثلَه  "  في نَصْبِه وجهان :<br />
أحدهما : عطف على اسم  "  أن  "  وهو  "  مَا  "  الموصولة .  <br />
والثاني : أنه منصُوب على المَعِيَّة ،   وهو رأي الزَّمخْشَرِي ،   وسيأتي ما يرد على ذلك والجوابُ عنه .  <br />
و  "  معهُ  "  ظرف واقع موقع الحال .  <br />
[  "  واللام  "  ]{[11660]} متعلِّقة بالاستِقْرَار الذي تعلَّق به الخبر ،   وهو  "  لَهُمْ  "  .  <br />
و  "  به  "  و  "  مِنْ عذاب  "  متعلِّقان بالافْتِدَاء ،   والضَّمير في  "  بِهِ  "  عائدٌ على  "  مَا  "  الموصولة ،   وجيء بالضَّمِير مُفْرَداً وإن تقدَّمه شَيْئَان وهما { مَا فِي الأرْض } و  "  مِثْلَهُ  "  ،   إما لتلازُمهمَا فهما في حُكْم شيء واحد ؛ وإما لأنَّه حذف من الثَّانِي لدلالة ما في الأوَّل عليه ،   كقوله رحمة الله عليه : [ الطويل ]<br />
 .   .   .   *** فَإنِّي وَقَيَّارٌ بِهَا لَغَرِيبُ{[11661]}<br />
أي : لو أنَّ لهم ما في الأرض لِيَفْتَدُوا به ،   ومثله معه ليفْتَدُوا به وإما لإجراء الضمير مُجْرَى اسم الإشارة ،   كقوله : [ الرجز ]<br />
 .   .   .   *** كَأنَّهُ فِي الجِلْدِ{[11662]}<br />
وقال بعضهم : ليفْتَدُوا بذلك المَال .  <br />
وقد تقدم في  "  البقرة  "  .  <br />
و  "  عذاب  "  بمعنى : تَعْذِيب بإضافته إلى  "  يَوْم  "  خرج  "  يَوْم  "  عن الظرفية ،   و  "  مَا  "  نَافِية وهي جواب  "  لَوْ  "  ،   وجاء على الأكْثَر من كونِ الجواب النَّفي بغير  "  لام  "  ،   والجملة الامْتِنَاعية في محل رفع خبراً ل  "  إن  "  ،   وجعل الزَّمَخْشَرِيُّ توحيد الضَّمير في  "  بِهِ  "  لمَدْرك آخر ،   وهو أنَّ  "  الوَاو  "  في  "  ومِثْلَهُ  "  [ واو  "  مع  "  قال بعد أن ذكر الوجهين المتقدمين : ويجُوزُ أن تكُونَ الواوُ في  "  ومِثله  "  ]{[11663]} بمعنى  "  مَعَ  "  فيتوحد المَرْجُوع إليه .  <br />
فإن قُلْتَ : فبم يُنْصب المَفْعُول معه ؟ .  <br />
قلت : بما تسْتَدْعِيه  "  لَوْ  "  من الفعل ؛ لأن التَّقدير : لو ثبت أنَّ لَهُمْ ما في الأرض ،   يعني : أنَّ حكم ما قبل المفعُول معه في الخَبَرِ والحَالِ ،   وعود الضَّمِير حكمه لو لم يكن بعده مفعول معه ،   تقول :  "  كُنْتُ وَزَيْدَاً كالأخِ  "  قال الشاعر : [ الطويل ]<br />
فَكَانَ وَإيَّاهَا كَحَرَّانَ لَمْ يُفِقْ *** عَنِ المَاءِ إذْ لاَقَاهُ حَتَّى تَقَدَّدَا{[11664]}<br />
فقال :  "  كَحَرَّان  "  بالإفْرَاد ولم يقُلْ{[11665]} :  "  كحرَّانَيْن  "  ،   وتقول :  "  جَاءَ زَيْدٌ وهنْداً ضاحِكاً في داره  "  .  <br />
وقد اختار الأخْفَشُ أن يُعطى حُكْم المُتَعَاطفين ،   يعني : فَيطَابق الخبر ،   والحَال ،   والضمير له ولما بَعْده ،   فتقول :  "  كُنْتُ وَزيْداً كالأخوين  "  .  <br />
قال بعضهم : والصَّحِيح جوازه على قِلِّة .  <br />
وقد رد أبُو حيَّان على الزمخشري ،   وطوَّل معه .  <br />
قال شهاب الدِّين{[11666]} : ولا بد من نَقْل نصِّه ؛ قال{[11667]} : وقول الزمَخْشَرِي ويجُوزُ أن تكون  "  الواو  "  بمعنى  "  مع  "  ليس بِشَيْء ؛ لأنَّه يصير التقدير : مع مثله معه ،   أي : مع مِثْل ما في الأرْضِ [ مع ما في الأرض ]{[11668]} إن جعلت الضَّمِير في  "  مَعَه  "  عائِداً على  "  مَا  "  يكون معه حَالاً من  "  مِثْله  "  .  <br />
وإذا كان مَا في الأرض مع مثله كان مثله معه ضرورة ،   فلا فائدة في ذكر  "  معه  "  لملازمة معيّة كل منهما للآخر .  <br />
وإن جعلت{[11669]} الضمير عائداً على  "  مثله  "  ،   أي : مع مثله مع ذلك المثل ،   فيكون المعنى مع مِثْلَيْنِ ،   فالتَّعْبِير عن هذا المَعْنَى بتلك العِبَارة عَيِيٌّ ؛ إذ الكلام المُنْتَظِم أن يكون التَّرْكيب إذا أُريد ذلك المعنى مع مِثْلَيْهِ .  <br />
وقول الزَّمَخْشَرِي :  "  فإن قُلْت  "  إلى آخِر الجواب [ هذا السؤال ]{[11670]} لا يرد ؛ لأنَّا قد بَيَّنَّا فسادَ أن يكون  "  الوَاو  "  واو  "  مَعَ  "  ،   وعلى تقدِير وُرودِه فهذا بناء منه على أنَّ [  "  أن  "  ]{[11671]} إذا جاءت بعد  "  لَوْ  "  كانت في مَحَلِّ رفع بالفاعليّة ،   فيكون التقدير على هذا لو ثَبَتَ كينونة ما في الأرْضِ مع مثله لهم لِيَفْتَدُوا به ،   فيكون الضَّمِير عائِداً على  "  مَا  "  فقط .  <br />
وهذا الذي ذكره هو تَفْريعٌ منه على مذهَب المُبَرِّد{[11672]} في أنَّ  "  أن  "  بعد  "  لَوْ  "  في محل رفع على الفاعليَّة ،   وهو مذهب مرجُوحٌ ،   ومذهب سيبويه{[11673]} : أن  "  أنَّ  "  بعد  "  لَوْ  "  في محلِّ رفع مُبْتَدأ .  <br />
والذي يظهر من كلام الزَّمخْشَرِي هنا وفي تصانيفه أنَّهُ ما وَقَفَ على مذهب سيبويْه في هذه المسألة .  <br />
وعلى المفرع على مذهَب المُبَرِّد لا يجوز أن تكون  "  الوَاوُ  "  بمعنى  "  مَعَ  "  والعامِلَ فيها  "  ثَبَتَ  "  المقدَّرة لما تقدم من وجود لفظة  "  مَعَهُ  "  ،   وعلى تقديره سُقُوطها لا يصحُّ ؛ لأن  "  ثَبَتَ  "  ليس رَافِعاً ل  "  مَا  "  العائد عليها الضمير وإنما هو رَافِعٌ مصدراً مُنْسَبكاً من  "  أن  "  وما بعدها ،   وهُو كونُ ؛ إذ التقدير لو ثَبَتَ كون ما في الأرض جَمِيعاً لهم ومِثْله معه لِيَفْتدُوا به ،   والضمير عَائِد على [ ما ] دُون الكوْنِ ،   فالرَّافِع الفاعِل غير النَّاصب للمفعُول معه ،   إذ لو كان إيَّاه للزم من ذلك وجود الثُّبُوت مُصَاحباً للمثل [ والمعنى على كينونة ما في الأرض مُصاحباً للمثل ،   لا على ثُبُوت ذلك مُصَاحِباً للمثل ،   ]{[11674]} وهذا فيه غُمُوض .  <br />
وبيانُهُ : أنَّك إذا قلت :  "  يُعْجِبُني قيام زَيْد وعَمْراً  "  ،   جعلت  "  عمراً  "  مفعُولاً معه ،   والعامِلُ فيه  "  يُعْجُبني  "  [ لزم ]{[11675]} من ذلك أن  "  عَمْراً  "  لم يَقُمْ ،   وأعجبك القِيَامُ وعمرو .  <br />
وإن جَعَلْتَ{[11676]} العامل فيه القِيَام : كان عمرو قَائِماً ،   وكان الإعْجَاب قد تعلَّق بالقيام مصاحباً لقيام عَمْرو .  <br />
فإن قلت : هلاّ كان  "  ومِثْلَهُ مَعَهُ  "  مفعولاً معه ،   والعامِلُ فيه هو العَامِلُ في  "  لَهُمْ  "  ؛ إذ المَعْنَى عليه ؟ .  <br />
قلت : لا يَصِحُّ ذلك لِمَا ذكرْنَاه من وجود  "  مَعَهُ  "  في الجُمْلَة ،   وعلى تقديرِ سُقُوطِهَا لا يصحُّ ؛ لأنَّهُم نَصُّوا على أنَّ قولك :  "  هَذَا لَكَ وأبَاك  "  ممنوع في الاختيار .  <br />
قال سيبويْه{[11677]} : وأما  "  هَذَا لَكَ وأبَاك  "  فَقَبيحٌ ؛ لأنَّه لم يذكر فِعْلاً ولا حَرْفاً فيه معنى فعل حتى يَصِير كأنَّه قد تكلَّم بالفعل ،   فأفْصَح سيبويه بأن اسْم الإشَارَة وحرف الجر المتضمن{[11678]} [ المعنى الاستقرار لا يعملان في المفعول معه وقد أجاز بعض النحويين في حرف الجر والظرف أن يعملا ]{[11679]} نحو  "  هذا لك وأباك  "  .  <br />
فقوله :  "  وأبَاكَ  "  يكون مفعُولاً مَعَهُ ،   والعَامِلُ الاستِقْرَار في  "  لَكَ  "  .   انتهى .   ومع هذا الاعتراض الذي ذكره ،   فقد يَظْهر عنه جوابٌ ،   وهو أنَّا نقول : نختار أن يكون الضَّمِير في قوله :  "  مَعَهُ  "  عائداً على  "  مِثْله  "  ويَصِيرُ المعنى : مع مِثْلَين ،   وهو أبْلَغُ من أن يكون مع مِثْل واحد .  <br />
وقوله :  "  تَرْكِيبٌ عَيِيٌّ  "  فَهْم قَاصِرُ ،   ولا بُدَّ من جُمْلَةٍ محذُوفَة قَبْل قوله : { ما تُقبِّلَ مِنْهُم } تقديره : وَبذَلُوه ،   أو وافْتَدُوا به ،   ليصِحَّ التَّرْتيب المذكُور ؛ إذ لا يترتَّب على اسْتِقْرَار ما فِي الأرْض جَمِيعاً ومثله معه لهم عدم التَّقبُّل ،   إنما يترتَّب عدم التَّقَبُّل على البَذلِ والافْتِدَاء والعامَّة على  "  تُقُبِّل  "  مبنياً [ للمفعول حذف فاعله لعظمته وللعلم به .  <br />
وقرأ يزيد بن قطيب{[11680]} :  "  ما تقبَّل  "  مبنياً للفاعل ]{[11681]} وهو ضميرُ البَارِي تبارك وتعالى .  <br />
قوله [ تعالى ]  "  ولَهُمْ عَذابٌ  "  مبتدأ وخبرُهُ مُقَدَّمٌ عليه ،   و  "  ألِيمٌ  "  صفته بمعنى : مُؤلِمٌ ،   وهذه الجُمْلَة أجَازُوا فيها ثلاثة أوجه :<br />
أحدها : أن تكون حالاً ،   وفيه ضَعْفٌ من حيث المعنى .  <br />
المعنى الثاني : أن تكون في مَحَلِّ رفع عَطْفاً على خَبَرِ  "  أن  "  أخبر عن الذين كفروا بخبرين لو استقرَّ لَهُمْ جَمِيعُ ما في الأرضِ مع مثله فَبَذلُوه ،   لم يُتَقَبَّلْ مِنْهُم وأنَّ لهم عَذَاباً أليماً .  <br />
الثالث : أن تكون مَعْطُوفة على الجُمْلَة من قوله : { إن الذين كفرُوا } ،   وعلى هذا فلا مَحَلَّ لها ؛ لِعَطْفها على ما لا مَحَلَّ له .
