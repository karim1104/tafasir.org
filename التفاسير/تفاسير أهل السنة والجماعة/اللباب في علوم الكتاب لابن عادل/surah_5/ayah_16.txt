قوله تعالى :  "  يَهْدِي  "  فيه خَمْسَة أوجه :<br />
أظهرها : أنَّه في محلِّ رفْع ؛ لأنَّهُ صفة ثَانِيَةٌ{[7]} ل  "  كِتَاب  "  ،   وصفَهُ بالمفرد ثم بالْجُمْلَة ،   وهو الأصْلُ .  <br />
الثاني : أن يكون صِفَةً أيضاً لكن ل  "  نُورٌ  "  ،   وصَفَهُ بالمُفْرَد ثم بالجُمْلَة ،   ذكره أبو البقاء{[8]} وفيه نظر ،   إذ القَاعِدَةُ أنَّه إذا اجْتَمَعَتِ التَّوابع قُدِّم النَّعْت على عَطْف النَّسَق ،   تقول جاء زَيْدٌ العَاقِل وعَمْرو ،   ولا تقُولُ : جاء زيْدٌ وعمرو العَاقِل ؛ لأنَّ فيه إلباساً أيضاً .  <br />
الثالث : أن يكون حالاً من  "  كِتَاب  "  ،   لأن النَّكرة لما تخصَّصَت بالوَصْف قَرُبت من المَعْرِفة .  <br />
وقياس{[9]} قول أبي البقاء أنَّه يجوز أن يكون [ حالاً من  "  نُورٌ  "  ،   كما جاز أن يكُون ]{[10]} صِفَة لَهُ{[11]} .  <br />
الرابع : أنَّهُ حال من  "  رسُولُنا  "  بدلاً من الجُمْلَة الواقِعَة حالاً له ،   وهي قوله :  "  يُبَيِّنُ  "  .  <br />
الخامس : أنَّهُ حالٌ من الضَّمير في  "  يُبَيِّن  "  ذكرهما أبُو البقاء{[12]} ،   ولا يَخْفَى ما فيهما من الفَصْلِ ؛ ولأن فيه ما يُشْبِهُ تَهْيِئَة{[13]} العَامِل للعَمَلِ ،   وقَطْعه عنه ،   والضَّمِير في  "  بِهِ  "  يعُودُ على من جعل  "  يَهْدي  "  حالاً منه ،   أو صِفَة لَه .  <br />
قال أبو البقاء{[14]} : فَلِذَلِكَ أفْرِدَ ،   أي : إن الضمير في  "  بِهِ  "  أُتِي به مُفْرداً ،   وقد تقدَّمه شيئان وهُمَا :  "  نُورٌ  "  و  "  كِتَابٌ  "  ،   ولكن لما قصد بالجُمْلَة من قوله :  "  يَهْدي  "  الحال ،   أو الوصف من أحدهما ،   أُفْرِد الضَّمير .  <br />
وقيل : الضمير في  "  بِهِ  "  يعود على الرَّسُول [ وقيل : يَعُودَ على  "  السَّلام  "  ]{[15]} ،   وعلى هذين القَوْلَيْن لا تكون الجُمْلَةُ من قوله :  "  يَهْدِي  "  حالاً ولا صِفَةً لعدم الرَّابط .  <br />
و  "  مَنْ  "  موصولةٌ أو نكرةٌ موصوفةٌ ،   ورَاعَى لَفْظَها في قولِهِ :  "  اتَّبَع  "  فلذلك [ أفرد الضَّمِير ،   ومَعْناها ،   فَلِذَلك ]{[16]} جمعه في قوله :  "  وَيُخْرِجُهُم  "  .  <br />
وقرأ عُبَيْد بن عُمَيْر ،   ومُسْلِم بن جُنْدُب ،   والزُّهَري{[17]}  "  بهُ  "  بضم الهاء حيث وقع ،   وقد تقدَّم أنَّه الأصْل .  <br />
وقرأ الحسن{[18]}  "  سُبْلَ  "  بسكون الباء ،   وهو تخْفيفٌ قياسي به كقولهم في عُنُق : عُنْق ،   وهذا أوْلَى لِكَوْنه جمعاً ،   وهو مَفْعُول ثانٍ ل  "  يَهْدِي  "  على إسقاط حَرْفِ الجرِّ ،   أي إلى  "  سُبْل  "  ،   وتقدَّمَ تحقيق نظيره ،   ويجُوزُ أن يَنْتَصِبَ على أنَّهُ بدَل من  "  رِضْوَانَهُ  "  إما بَدَل كُل من كلّ لأن سبل السلام [ هي رضوان الباري تعالى ،   وإما بدل اشتمال ؛ لأن الرضوان مشتمل على سبل السلام ؛ أو لأنها مشتملة على رضوان الله تعالى ،   وإما بدل ]{[19]} بعض من كل ؛ لأن سبل السلام بعض الرضوان .  <br />
فصل<br />
معنى { يَهْدِي بِهِ اللَّهُ } أي : بالكِتَابِ المُبِين { من اتَّبَعَ رِضْوَانَهُ } أي : كان مطلوبُهُ من طلب الدِّين{[20]} اتِّباع الدِّين [ الذي ]{[21]} يَرْتَضِيه اللَّه ،    "  سُبُلَ السلام  "  [ أي : طُرُق السَّلامَة ]{[22]} وقيل : السلام هو الله عزَّ وجلَّ ،   و  "  سُبُلَهُ  "  : دينَهُ الذي شرع لِعبَادِه ،   ويجُوزُ أن يكون على حَذْفِ مُضَافٍ ،   أي : سُبُل دار السلام ،   ونظيره قوله : { وَالَّذِينَ قُتِلُواْ فِي سَبِيلِ اللَّهِ فَلَن يُضِلَّ أَعْمَالَهُمْ سَيَهْدِيهِمْ وَيُصْلِحُ بَالَهُمْ } [ محمد : 4 - 5 ] ومعلومٌ أنَّه ليس المُرادُ هِدَايَة الاسْتِدْلاَل ،   بل الهدايَةُ إلى طَرِيق الجَنَّة .  <br />
ثم قال : { ويخْرِجُهُم من الظُّلُمَاتِ إلى النُّورِ بِإذْنِهِ } أي : من ظُلمات الكُفْر إلى نورِ الإيمان  "  بإذْنِهِ  "  بتوفيقه و  "  بإذنِهِ  "  متعلِّق ب  "  يُخْرِجُهُم  "  أي : بتَيْسِيره أو بأمْره ،   و  "  البَاءُ  "  للحال أي : مُصَاحِبِين لِتَيْسِيرِه أو للسَّبَبِيَّة ،   أي : بسبب أمْرِه المُنَزَّل على رسوله .  <br />
وقيل :  "  الباء  "  تتعلق بالاتِّبَاع ،   أي : يتَّبع رِضْوَانه بإذْنِه .  <br />
قال ابنُ الخَطِيب{[23]} : ولا يجُوزُ أن تتعلَّق بالهِدَاية ،   [ ولا بالإخْرَاج ؛ لأنَّه لا مَعْنَى له ،   فَدَلَّ ذلك على أنَّه لا يتبع رضوان الله إلا من أرَادَ اللَّه منه ذَلِكَ ] .  <br />
ثم قال : { يَهْدِيهِمْ إلى صِرَاطٍ مُسْتَقيم } وهو الدِّين الحقُّ .
