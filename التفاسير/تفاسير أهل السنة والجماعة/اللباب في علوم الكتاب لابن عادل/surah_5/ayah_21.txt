وقرأ ابن{[11334]} مُحَيْصِن هنا وفي جميع القُرْآن  "  يَا قَوْمُ  "  مضموم الميم .  <br />
وتروى قراءة{[11335]} عن ابن كثيرٍ [ ووَجْهُهَا أنَّها ]{[11336]} لُغَةٌ في المُنَادى المضاف إلى يَاءِ المُتَكَلِّم كَقِرَاءة { [ قل ]{[11337]} رَبُّ احْكُم بِالحَقِّ } [ الأنبياء : 112 ] ،   وقد تقدَّمَت هذه [ المسألة ]{[11338]} .  <br />
وقرأ ابن{[11339]} السمَيْفع : { يَا قَوْمِيَ ادْخُلُوا } بفتح الياء ،   ورُوِيَ أنَّ إبراهيم - عليه السلام - لما صعد [ جَبَل لبنان ]{[11340]} ،   فقال اللَّهُ تعالى له :  "  انظر فما أدركه بصرك فهو مقدس ،   وهو ميراث لذريتك " {[11341]} .  <br />
والأرضُ المقدَّسَةُ هي الأرْضُ المطهَّرَةُ من الآفات ؛ لأنَّ التَّقْدِيس هُو التَّطْهِيرُ ،   وقال المُفَسِّرُون{[11342]} طهِّرَت من الشِّرك ،   وجُعِلَت مَسْكَناً وقَرَاراً للأنْبِيَاء ،   وفيه نظرٌ ؛ لأنَّ تلك الأرْض لما أمَرهُمْ مُوسى بِدُخُولها ما كانَتْ مقدَّسة عن الشِّرْك ،   وما كَانَتْ مَقرًّا للأنْبيَاء ،   وقد يُجَابُ عَنْهُ بأنَّها كَانَتْ كذلك فيما قَبْل .  <br />
واختلفُوا في تلك الأرْض ،   فقال عِكْرِمَةُ ،   والسديُّ ،   وابنُ زَيْد : هي أريحا{[11343]} .  <br />
وقال الكَلْبِيُّ : هي دمشق وفلسْطِين وبعض الأرْدُن{[11344]} ،   وقال الضَّحَّاك : هي إيليَا وبَيْتُ المَقْدِسِ ،   وقال مُجَاهِد : هي الطُّور وما حَوْلَه{[11345]} .   وقال قتادةُ : هي الشَّامُ كُلُّها{[11346]} .   وقال كَعْبٌ : وجَدْتُ في كتاب اللَّه المُنَزَّل [ أنَّ الشَّام ]{[11347]} كَنْزُ الله من أرْضِه ،   وبِها كَثْرَةٌ من عِبَادِه .  <br />
وقوله : { كَتَبَ اللَّهُ لَكُمْ } يعني : في اللَّوْح المَحْفُوظِ أنَّها لكم مَسَاكِن .  <br />
وقال ابن إسحاق{[11348]} : وهب اللَّهُ لكم ،   وقيل : جعلها لكم [ قال السديّ : أمَرَكُم الله بِدُخُولها ]{[11349]} .  <br />
فإن قيل : لم قال { كتبَ اللَّه لَكُم } ،   ثم قال { فَإِنَّهَا مُحَرَّمَةٌ عَلَيْهِمْ }{[11350]} [ المائدة : 26 ] .  <br />
فالجوابُ : قال ابنُ عبَّاس : كانت هِبَةً ثُمَّ حرَّمها عليهم بشُؤم تَمَرُّدِهِم وعِصْيَانِهم{[11351]} ،   وقيل : اللَّفْظ وإن كان عامًّا لكنَّ المرادُ به الخُصُوصُ ،   فكَأَنَّها كُتِبَتْ لِبَعْضِهِم ،   وَحُرِّمَتْ على بَعْضِهِم .  <br />
وقيل : إنَّ الوَعْد بقوله : { كَتَبَ اللَّهُ لَكُمْ } مشروطٌ بقَيْد الطَّاعة ،   فلما لم يُوجَد الشَّرْط لم يُوجَد المَشْرُوط .  <br />
وقيل : إنَّها مُحَرَّمةٌ عليهم أرْبَعِين سَنَة ،   فلما مَضَى الأرْبَعُون حصل ما كتَبَ .  <br />
وفي قوله : { كَتَبَ اللَّهُ لَكُمْ } فَائِدَة ،   وهي{[11352]} أنَّ القوم وإن كانُوا جبَّارِين ،   إلاَّ أنَّ الله تعالى لمَّا وعد هؤلاءِ الضُّعَفَاء بأنَّ تلك الأرْضَ لهم ،   فإن كانوا مُؤمنين مُقَرَّبين بصدْق الأنْبِيَاء ،   عَلِمُوا قَطْعاً أنَّ اللَّه يَنْصُرهم عليهم ،   فلا بُدَّ وأن يَعْزِمُوا على قتالهم من غير خَوْفٍ ولا جُبْن .  <br />
قوله : { وَلاَ تَرْتَدُّوا عَلَى أَدْبَارِكُمْ } فالجار والمجرور [ حال من فاعل  "  تَرتَدوا  "  أي : لا ترتدوا مُنقلبين ،   ويجُوزُ أن ]{[11353]} يتعلَّق بِنَفْسِ الفِعْل قَبْلَهُ .  <br />
وقوله :  "  فَتَنْقَلِبُوا  "  فيه وجهان :<br />
أظهرُهُمَا : أنَّهُ مَجْزُومٌ عَطْفاً على فِعْل النَّهْي .  <br />
والثاني : أنَّهُ منصُوبٌ بإضْمَار  "  أنْ  "  بعد الفَاءِ في جواب النَّهي .  <br />
و  "  خَاسِرِين  "  حالٌ .  <br />
وفي المَعْنَى وجهان :<br />
أحدهما : لا يَرْجِعُوا عن الدِّين الصَّحيح في نُبُوَّة مُوسَى ؛ لأنَّهُ - عليه السلام - لما أخْبَرَ اللَّه تعالى جعل تِلْكَ الأرْضَ لَهُم ،   [ أو ]{[11354]} كان هذا وعْداً بأنَّ الله يَنْصُرهم{[11355]} عليهم ،   فلو لَمْ يقْطعُوا بِهَذِه النُّصْرَة ،   صارُوا شاكِّين في صِدْق مُوسى - عليه الصلاة والسلام - فيصيروا{[11356]} كافرين بالنُّبُوَّة والإلهِيَّة .  <br />
والثاني : لا ترجِعُوا عن الأرْضِ التي أمرتُم بِدُخُولِها إلى الأرْض الَّتِي خَرَجْتُم عنها ،   يُرْوَى أنَّهُم عَزَمُوا على الرُّجُوعِ إلى مِصْر .  <br />
وقوله :  "  فَتَنْقَلِبُوا خَاسِرِين  "  أي : في الآخِرَة يَفوتُكُم الثَّواب ويَلْحَقُكُم العِقَابُ .  <br />
وقيل : تَرْجِعُون إلى الذِّلَّة ،   وقيل : تُمَزَّقُون{[11357]} في التِّيه ،   ولا تَصِلُون{[11358]} إلى شَيْءٍ من مَطَالبِ الدُّنْيَا ومَنَافِع الآخِرَة .
