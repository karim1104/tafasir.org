قوله :  "  أإنَّكُمْ  "  قرأ نافعٌ{[16471]} وحفصٌ عن عاصم :  "  إنكم  "  على الخبر المستأنف ،   وهو بيان تلك الفاحشة ،   وقرأ الباقون بالاستفهام المقتضي للتّوبيخ ،   فقرأ ابنُ كثير بهمزة غير ممدودة وتليين الثَّانية ،   وقرأ أبُوا عمرو بهمزة ممدودة للتّخفيف وتليين الثانية ،   والباقون بهمزتين على الأصل .  <br />
قال الواحديُّ{[16472]} :  "  كان هذا استفهاماً معناه الإنكار لقوله تعالى :  "  أَتَأْتُونَ الْفَاحِشَةَ  "  ،   وكلُّ واحد من الاستفهامين جملة مستقلة غير محتاجة في تمامها إلى شيء آخر  "  .  <br />
قوله : { لَتَأْتُونَ الرجال شَهْوَةً } قيل : نصب  "  شَهْوَةٍ  "  على أنه مفعول من أجله ،   أي : لأجل الاشتِهَاءِ لا حامل لكم عليه إلاَّ مجرّد الشَّهوة لا غير .  <br />
وقيل : إنَّها مصدر واقعٌ موقع الحال ،   أي : مشتهين أو باق على مصدريَّته ،   ناصبة  " أتَأتُونَ  "   ؛ لأنَّهُ بمعنى أتشتهون .  <br />
ويقال : شَهِيَ يَشْهَى شَهْوَةً ،   [ وشَهَا يَشْهُو شَهْوَةٍ ]{[16473]} قال الشَّاعر : [ الطويل ]<br />
وَأشْعَثَ يَشْهَى النَّوْمَ قُلْتُ لَهُ : ارْتَحِلْ *** إذَا مَا النُّجُومُ أعْرَضَتْ واسْبَكَرَّتِ{[16474]}<br />
وقد تقدَّم ذلك في آل عمران{[16475]} .  <br />
قوله : { مِّن دُونِ النساء } فيه ثلاثةُ أوْجُهٍ :<br />
أحدها : أنَّهُ متعلق بمحذوف ،   لأنَّهُ حال من  "  الرِّجالِ  "  أي : أتأتونهم منفردين عن النِّساء .  <br />
والثاني : أنَّهُ متعلِّق ب  "  شَهْوَة  "  ،   قاله الحوفيُّ .   وليس بظاهر أن تقول :  "  اشتهيتُ من كذا  "  ،   إلاَّ بمعنى غير لائق هنا .  <br />
والثالث : أن يكُون صفة ل  "  شهوة  "  أي : شهوة كَائِنَة من دونهن .  <br />
قوله : { بَلْ أَنْتُمْ قَوْمٌ مُّسْرِفُونَ }  "  بل  "  للإضراب ،   والمشهور أنهُ إضراب انتقالٍ من قصّة إلى قصّة ،   فقيل : عن مذكور ،   وهو الإخبار بتجاوزهم عن الحدِّ في هذه الفاحشة ،   أو عن توبيخهم وتقريرهم ،   والإنكار عليهم .  <br />
وقيل : بل للإضراب عن شيء مَحْذُوفٍ .   واختلف فيه :<br />
فقال أبُو البقاء{[16476]} :  "  تقديرُهُ : ما عَدَلْتُم بل أنتم  "  .  <br />
وقال الكَرْمَانيُّ :  "  بل  "  ردٌّ لجواب زعموا أن يكون لهم عذراً أي :  "  لا عذر لكم بل "  .  <br />
وجاء هان بصفة القوم اسم الفاعل وهو  "  مُسْرِفُونَ  "   ؛ لأنَّهُ أدلُّ على الثُّبوت ولموافقة رءوس الآي ؛ فإنَّهَا أسماء .  <br />
وجاء في النمل [ 55 ] { تَجْهَلُونَ } دلالة على أنَّ جهلهم يتجدد كل وقت ولموافقة رءوس الآي فإنها أفعال .  <br />
فصل في الإسراف<br />
معنى  "  مُسْرِفُونَ  "  أي : يتجاوزون الحلال إلى الحَرَامِ .  <br />
قال الحسنُ :  "  كانوا لا ينكحون إلا الغرباء " {[16477]} .  <br />
وقال الكلبيُّ :  "  إنَّ أوَّل من عملَ عملَ قوم لوط إبليس ؛ لأنَّ بلادهم أخُصَبَتْ فانتجعها{[16478]} أهلُ البلدان ،   فتمثل لهم إبليس في صورة شابّ ،   ثم دعى إلى دُبرِهِ فنكح في دبره ،   فأمر الله - تعالى - السَّماءَ أن تحصبهم ،   والأرض أن تخسف بهم{[16479]} .
