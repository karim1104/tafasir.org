المراد ب  "  أهله  "  : أنصاره وأتباعه .  <br />
وقال ابن عباس :  "  المراد : ابنتاه  "  وقوله :  "  إلا امْرَأتَهُ  "  أي : زوجتهُ ،   يقال : امرأةُ الرَّجل أي زوجته ويقال : رجل المرأة بمعنى زوجها ؛ لأنَّ الزوج بمنزلة المالك لها ،   وليست المرأة بمنزلة المالك للرّجل .  <br />
وقوله :  "  مِنَ الغَابِرِينَ  "  يعني الباقين في العذاب .  <br />
وقيل : من الباقين المعمّرين قد أتى عليها دَهْرٌ طويل فهلكت ،   فهي مع من هلك .  <br />
يقال : غبر الشيء يغبر غبوراً إذا مكث وبقي .  <br />
وقوله : { كَانَتْ مِنَ الغابرين } جواب سؤال مقدَّر ،   وهذا كما تقدم في البقرة ،   وفي أوَّل هذه السُّورة في قصة  "  إبليس  "  .  <br />
والغابر : المقيم وهذا [ هو ] مشهور اللُّغة ،   وأنشدوا قول بي ذُؤَيْبٍ الهذلي : [ الكامل ]<br />
فَغَبَرْتُ بَعْدَهُمُ بِعَيْشٍ نَاصِبٍ *** وإخَالُ أنِّي لاحِقٌ مُسْتَتْبَعُ{[16482]}<br />
ومنه غُبَّرُ اللبن لبقيته في الضَّرْع ،   وغُبَّرُ الحَيْض أيضاً ،   قال أبو كبير الهُذَلِيُّ ،   ويروى لتأبَّط شَرّاً : [ الكامل ]<br />
ومُبَرَّأ مِنْ كُلِّ غُبَّرِ حَيْضَةٍ *** وَفَسَادِ مُرْضِعَةٍ وَدَاءٍ مُعْضِلِ{[16483]}<br />
ومعنى  "  مِنَ الغَابِرِينَ  "  في الآية أي : من المقيمين في الهلاكِ .  <br />
وقال بعضهم :  "  غبر بمعنى مضى وذهب  "  ومعنى الآية يساعده ؛ وأنشد للأعشى : [ السريع ]<br />
عَضَّ بِمَا أبْقَى المَوَاسِي لَهُ *** مِنْ أمِّهِ فِي الزَّمَنِ الغَابِرِ{[16484]}<br />
أي : الزمان الماضي .  <br />
وقال بعضهم : غَبَرَ : أي غاب ،   ومنه قولهم :  "  غَبَر عَنَّا زماناً  "  .  <br />
وقال أبُو عبيدة{[16485]} :  "  غبر : عُمِّر دَهْراً طويلاً حتّى هَرِمَ  "  ،   ويدلُّ له : { إِلاَّ عَجُوزاً فِي الغابرين } [ الصافات : 135 ] .  <br />
وقد تقدَّم .   والحاصل أنَّ الغبور مشتركٌ ك  "  عَسْعَسَ  "  ،   أو حقيقة ومجازٌ وهو المرجح .   والغبارُ : ما يبقى من التُّراب المُثَارِ ومنه : { وَوُجُوهٌ يَوْمَئِذٍ عَلَيْهَا غَبَرَةٌ } [ عبس : 40 ] .  <br />
تخييلاً لتغيرها واسودادها والغَبْراء : الأرْضُ ؛ قال طرفةُ : [ الطويل ]<br />
رَأيْتُ بَنِي غَبْرَاءَ لا يُنْكِرُونَنِي *** وَلاَ أهْلُ هَذَاكَ الطِّرَافِ المُمَدَّدِ{[16486]}
