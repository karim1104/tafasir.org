قوله : { فَلَماَّ نَسُواْ مَا ذُكِّرُواْ بِهِ } الضَّميرُ في نَسُوا للْمنهيِّينَ و  "  ما  "  موصولةٌ بمعنى  "  الذي  "  أي : فلمَّا نسُوا الوعظ الذي ذكَّرَهُم به الصَّالحون .  <br />
قال ابنُ عطيَّة{[16926]} : ويحتمل أن يُرادَ به الذِّكرُ نفسه ،   ويحتمل أن يراد به ما كان فيه الذكر .  <br />
قال أبُو حيان{[16927]} : ولا يظهرُ لي هذان الاحتمالان .  <br />
قال شهابُ الدِّين{[16928]} : يعني ابنُ عطية بقوله :  "  الذِّكرُ نفسُهُ  "  أي : نفسُ الموصول مُرادٌ به المصدر كأنه قال : فلمَّا نسُوا الذِّكْرَ الذي ذُكِّروا به ،   وبقوله :  "  مَا كان فيه الذِّكر  "  نَفَسُ الشيء المذكَّر به الذي هو متعلِّق الذكر ؛ لأن ابن عطيَّة لمَّا جعل  "  ما  "  بمعنى  "  الذي  "  قال : إنَّها تحتملُ الوقوع على هذين الشيئين المتغايرين .  <br />
فصل<br />
النِّسيان يطلق على السَّاهي ،   والعامد التَّارك لقوله : { فَلَماَّ نَسُواْ مَا ذُكِّرُواْ بِهِ } أي : تركوه عن قصد ،   ومنه قوله تعالى : { نَسُواْ الله فَنَسِيَهُمْ } [ التوبة : 67 ] .  <br />
فصل<br />
المعنى : فلمَّا تركوا ما وعظوا به ،   { أَنجَيْنَا الذين يَنْهَوْنَ عَنِ السواء وَأَخَذْنَا الذين ظَلَمُواْ } أي الذين أقدموا على المعصية .  <br />
واختلف المفسِّرون في الفرقة السَّاكتة .   فنقل عن ابن عبَّاسٍ : أنَّهُ توقَّف فيهم ،   ونقل عنه : هلكت الفرقتان ونجت النَّاهية ،   وكان ابنُ عباس إذا قرأ هذه الآية بكى وقال : إنَّ هؤلاء الذين سكتوا عن النَّهي عن المنكر هلكوا ،   ونحن نرى أشياء ننكرها ،   ثم نسكت ،   ولا نقول شيئاً .  <br />
وقال الحسنُ : نجت الفرقتانِ ،   وهلكت العاصية{[16929]} ،   لأنهم لما قالوا : { لِمَ تَعِظُونَ قَوْماً الله مُهْلِكُهُمْ أَوْ مُعَذِّبُهُمْ } دلَّ على أنَّهُمْ أنكروا أشد الإنكار ،   وأنَّهُمْ إنَّما تركوا وعظهم ؛ لأنه غلب على ظنهم أنهم لا يلتفتون إلى ذلك الوعظِ .  <br />
فإن قيل : إن ترك الوعظِ معصية ،   والنَّهي عنه أيضاً معصية ؛ فوجب دخول هؤلاء التَّاركين للوعظ النَّاهين عنه تحت قوله : { وَأَخَذْنَا الذين ظَلَمُواْ بِعَذَابٍ بَئِيسٍ } .  <br />
فالجوابُ : هذا غير لازمٍ ؛ لأنَّ النَّهي عن المنكر إنَّمَا يجب على الكفاية ،   ولو قام به البعض سقط عن الباقين .  <br />
وروي عن ابن عبَّاس أنه قال : أسمعُ الله يقول : { أَنجَيْنَا الذين يَنْهَوْنَ عَنِ السواء وَأَخَذْنَا الذين ظَلَمُواْ بِعَذَابٍ بَئِيسٍ } ،   فلا أدري ما فعلت الفرقة السَّاكتة{[16930]} ؟<br />
قال عكرمةُ : قلت له : جعلني اللَّهُ فداك ،   ألا تراهم قد أنكروا ،   وكرهوا ما هم عليه وقالوا : { لِمَ تَعِظُونَ قَوْماً الله مُهْلِكُهُمْ } ،   وإن لم يقل الله أنجيتهم لم يقل : أهلكتهم ،   فأعجبه قولي ورضي به ،   وأمر لي ببردين فكسانيهما ؛ وقال : نَجتِ السَّاكتةُ ،   وهذا قول يمان بن رباب ،   والحسن ،   وابن زيد .  <br />
قوله :  "  بعذابٍ بئيسٍ  "  .   أي : شديد .  <br />
قرأ نافعٌ ،   وأبو جعفر ،   وشيبةٌ بيْسٍ{[16931]} بياء ساكنة ،   وابن عامر بهمزة ساكنة .  <br />
وفيهما أربعة أوجه :<br />
أحدها : أنَّ هذا في الأصْلِ فعلٌ ماضٍ سُمِّيَ به فأعربَ كقوله عليه الصَّلاة والسَّلام :  "  أنهاكم عن قيل وقال  "  بالإعراب والحكاية ،   وكذا قولهم :  "  مُذ شبَّ إلى دبَّ ومُذ شبٍّ إلى دَبٍّ  "  ،   فلما نُقل إلى الاسميَّة صار وصفاً ك : نِضْو ونِقْض .  <br />
والثاني : أنَّهُ وصف وضع على فعل ك : حِلْف .  <br />
الثالث : أن أصله بَئيس كالقراءةِ المشهورة ،   فخفَّفَ الهمزة ؛ فالتقت ياءان ،   ثم كسر الياء إتباعاً ،   كرغيف وشهيد فاستثقل توالي ياءين بعد كسرةٍ ،   فحذفت الياء المكسورة ؛ فصار اللَّفظُ  "  بِيْسٍ  "  وهو تخريج الكسائيِّ .  <br />
الرابع : أن أصله بَئِس بوزن  "  كَتِف  "  ثم أتبعت الياءُ للهمزة في الكسر ثم سُكِّنت الهمزة ،   ثمَّ أبدلت ياء ك : بِيرٍ وذِيبٍ .  <br />
وأمَّا قراءة ابن عامر فتحتمل أن تكون فعلاً منقولاً ،   وأن تكون وصفاً ك : حِلْف .  <br />
وقرأ أبو بكر عن عاصم بَيْئَسٍ بياء ساكنة بين باء ،   وهمزة مفتوحتين ،   وهو صفةٌ على فَيْعَل ك : ضَيْغَم ،   وصَيْرَف ،   وصَيْقَل ،   وهي كثيرةٌ في الأوصافِ .  <br />
وقرأ امرؤُ القيسِ : [ الرجز ]<br />
كِلاهُمَا كَأنَ رَئِيساً بَيْئَسا *** يَضْرِبُ في يومِ الهِيَاجِ القَوْنَسَا{[16932]}<br />
وقرأ باقي السبعة  "  بَئِيسٍ  "  بزنه  "  رَئِيسٍ  "  وفيه وجهان :<br />
أحدهما : أنَّهُ وصفٌ على  "  فَعِيلٍ  "  ك : شَدِيدٍ ،   وهو للمبالغة وأصله فاعل .  <br />
والثاني : أنه مصدرٌ وصف به أي : بعذابٍ ذي بأس بَئِيس ،   ف  "  بَئِيسٍ  "  مصدر مثل : النكير والقدير ،   ومثر ذلك في احتمالِ الوجهين قول أبي الأصبع العدواني : [ مجزوء الكامل ]<br />
حَنَقاً عليَّ ولا أرَى *** لِيَ مِنْهُمَا شَرّاً بَئِيساً{[16933]}<br />
وهي أيضاً قراءةُ عليٍّ وأبي رجاء{[16934]} .  <br />
وقرأ يعقوبُ القارئ  "  بَئِسَ  "  بوزن  "  شَهِدَ  "  ،   وقرأها أيضاً عيسى بنُ عُمَرَ ،   وزيد بن علي .  <br />
وقرأ نصرُ بنُ عاصم  "  بَأسَ  "  بوزن  "  ضَرَبَ  "  فعلاً ماضياً .  <br />
وقرأ الأعمش ومالك بنُ دينار  "  بَأس  "  فعلاً ماضياً ،   وأصله  "  بَئِس  "  بكسر الهمزة ،   فسكَّنَهَا تخفيفاً ك : شَهْدَ في قوله : [ الرجز ]<br />
لَوْ شَهْدَ عَاد فِيَ زَمَانِ تُبَّعِ{[16935]} *** .   .   .  <br />
وقرأ ابنُ كثير وأهل مكة بِئِسٍ بكسر البَاءِ ،   والهمز همزاً خفيفاً ،   ولم يُبَيِّن هل الهمزة مكسورةٌ أو ساكنةٌ ؟<br />
وقرأ طلحة وخارجة عن نافع  "  بَيْسٍ  "  بفتح الباء ،   وسكون الياء مثل : كَيْلٍ ،   وأصله  "  بَيْئَس  "  مثل : ضَيْغَم فخفَّف الهمزة بقلبها ياءً ،   وإدغام الياء فيها ثم خفَّفهُ بالحذف ك : مَيْت في : مَيِّت .  <br />
وقرأ عيسى بن عمر والأعمش وعاصم في رواية  "  بَيْئٍسٍ  "  كقراءة أبي بكر عنه ،   إلاَّ أنَّهُ كسر الهمزة ،   وهذه قد ردَّها النَّاسُ ؛ لأن  "  فَيْعِلاً  "  بكسر العين في المعتلِّ ،   كما أن  "  فَيْعَلاً  "  بفتحها في الصحيح ك : سَيِّد وضَيْغَم ،   على أنه قد شذّ  "  صَيْقِل  "  بالكسر ،   و  "  عَيَّل  "  بالفتح .  <br />
وقرأ نصر في رواية مالك بن دينار عن  "  بَأسٍ  "  بفتح الباء والهمزة وجر السِّين ،   بزنة  "  جَبَلٍ  "  .  <br />
وقرأ أبو عبد الرحمن السلمي وطلحة بن مصرف  "  بَئِسٍ  "  مثل كبد وحذر .  <br />
قال عبيد الله بن قيس : [ المديد ]<br />
لَيْتَنِي ألْقَى رُقَيَّةَ فِي *** خَلْوَةٍ مِنْ غَيْرِ مَا بَئِسِ{[16936]}<br />
وقرأ نصر بنُ عاصم في رواية بَيِّسٍ بهمزةٍ مشددة .  <br />
قالوا : قلب الياء همةً وأدغمها في مثلها .   ورويت هذه عن الأعمش أيضاً .  <br />
وقرأت طائفة بَأسَ بفتح الثلاثة ،   والهمزة مشددة ،   فعلاً ماضياً ،   ك  " شَمَّرَ "  ،   وطائفة أخرى بَأسَ كالتي قبلها إلاَّ أنَّ الهمزة خفيفةٌ ،   وطائفة بَائسٍ بألف صريحة بين الباءِ والسِّينِ المجرورة ،   وقرأ أهل المدينة بِئيسٍ ك :  "  رَئِيسٍ  "  ،   إلاَّ أنهم كسروا الباء ،   وهذه لغةُ تميم في فعيل الحلقيِّ العين نحو : بِعِير ،   وشِعِير ،   وشِهيد ،   سواء أكان اسماً أم صفة .  <br />
وقرأ الحسن والأعمش  "  بِئْيَسٍ  "  بياء مكسورة ،   ثم همزة ساكنة ،   ثم يا مفتوحة ،   بزنة  "  حِذْيَم  "  ،   و  "  عِثْيَر  "  .  <br />
وقرأ الحسنُ بِئْسَ بكسر الباءِ ،   وسكون الهمزة وفتح السِّينِ ،   جعلها التي للذَّمِّ في نحو : بِئْسَ الرجل زيدٌ .  <br />
ورُويت عن أبي بكر .  <br />
وقرأ الحسن أيضاً كذلك ،   إلاَّ أنه بياءٍ صريحة ،   وتخريجها كالتي قبلها ،   وهي مرويةً عن نافع وقد ردَّ أبو حاتم هذه القراءةَ والتي قبلها بأنَّهُ لا يقالُ : مررت برجلٍ بِئْسَ ،   حتَّى يقال بِئْسَ الرجل ،   أو بئس رجلاً .  <br />
قال النَّحَّاس{[16937]} : وهذا مردودٌ - يعني قول أبي حاتم - حكى النحويون : إن فعلت كذا وكذا فبها ونعمت ،   أي : ونعمت الخصلة ،   والتقدير : بِئْسَ العذابُ .  <br />
قال شهابُ الدِّينِ{[16938]} : أبو حاتم معذورٌ في القراءة ،   فإنَّ الفاعل ظاهراً غير مذكور ،   والفاعل عمدةٌ لا يجوز حذفه ،   ولكنه قد ورد في الحديث مَنْ تَوضَّأ فبها ونعْمتُ ،   ومن اغتسَلَ فالغُسْلُ أفضلُ ففاعل  "  نِعْمَتْ  "  هنا مضمرٌ يفسِّرُهُ سياقُ الكلام .  <br />
قال أبُو حيَّان{[16939]} : فهذه اثنتان وعشرون قراءةً ،   وضبطُها بالتَّلخيص : أنَّها قُرِئتْ ثلاثية اللَّفْظِ ،   ورباعَّيتَهُ ،   فالثُّلاثي اسماً : بِئْسٍ ،   وَبِيْسٍ ،   وبَيْسٍ ،   وبَأْسٍ ،   وبَأَسٍ ،   وبَئِسٍ ،   وفعلاً بِيْسَ وبِئْسَ ،   وبَئِسَ ،   وبَأَسَ ،   وبَأْسَ ،   وبَيِسَ .  <br />
والرباعية اسماً : بَيْئَسٍ ،   وبِيْئِسٍ ،   وبَيْئِسٍ ،   وبَيِّسٍ ،   وبَئِيْسٍ ،   وبِئَيْسٍ ،   وبِئْيِسٍ ،   وبِئْيَسٍ ،   وفعلاً : بَأَّسَ  "  .  <br />
وقد زا أبو البقاء{[16940]} أربع قراءات أخر : بَيِس بباء مفتوحة وياء مكسورة .  <br />
قال : وأصلها همزة مكسورة فأبدلت ياء ،   وبَيَس بفتحهما .  <br />
قال : وأصلها ياء ساكنة وهمزة مفتوحة إلاَّ أنَّ حركة الهمزة ألقيت على الياء وحذفت ،   ولم تقلب الياء ألفاً ،   لأنَّ حركتها عارضةٌ .   وبَأْيَسٍ بفتح الباء ،   وسكون الهمزة وفتح الياء .  <br />
قال : وهو بعيد إذ ليس في الكلام  "  فَعْيَل  "  وبَيْآس على فَيْعَال .   وهو غريب .
