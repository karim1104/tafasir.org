لمَّا شَرَحَ وعيد الكفار وثواب أهل الإيمان أتبعه بذكر المُنَاظَراتِ التي تَدُورُ بين الفريقين في هذه الآية .  <br />
قوله : { أَن قَدْ وَجَدْنَا }  "  أن  "  يحتمل أن تكون تفسيرية للنِّدَاءِ ،   وأن تكون مخففة من الثَّقِيلَةِ ،   واسمها ضمير الأمر والشَّأنِ ،   والجملة بعدها خبرها ،   وإذَا كان الفعل مُتَصَرّفاً غير دعاء ،   فالأجود الفصل ب  "  قَدْ  "  كهذه الآية أو بغيرها .   وقد تقدَّم تحقيقه في المائدة .  <br />
قال الزَّمخشريُّ{[16158]} : فإن قلت : هلا قيل : ما وعدكم ربكم ،   كما قيل : { مَا وَعَدَنَا رَبُّنَا } .  <br />
قلت : حُذف ذلك تخفيفاً لدلالة  "  وَعَدَنَا  "  عليه .  <br />
ولقائل أن يقول : أطْلِقَ ليتناول كلَّ ما وعد الله من البعث والحساب والعقاب والثواب ،   وسائر أحوال القيامة ،   لأنَّهُم كانوا مكذِّبين بذلك أجمع ،   ولأنَّ الموعود كله ممَّا ساءهم ،   وما نعيم أهل الجنَّة إلاَّ عذاب لهم فأطلق لذلك .  <br />
قال شِهَابُ الدِّين{[16159]} : قوله :  "  ولقائل .   .   .   إلى آخره .  <br />
هذا الجواب لا يطابق سؤاله ؛ لأنَّ المدعي حذف المفعول الأوَّل ،   وهو ضمير المخاطبين .  <br />
والجوابُ وقع بالمفعول الثَّاني الذي هو الحِسَابُ والعقاب ،   وسائر الأحوال ،   [ فهذا ] إنَّما يناسب لو سئل عن حذف المفعول الثاني ،   لا المفعول الأول  "  .  <br />
وأجاب ابْنُ الخطيبِ{[16160]} عن السُّؤالِ بأن قوله : { مَا وَعَدَنَا رَبُّنَا } يدلُّ على أنَّهُ تعالى خاطبهم بهذا الوَعْدِ وكونهم مخاطبين من قبل الله - تعالى - بهذا الوعد يوجب مزيد التّشريف ومزيد التشريف لائق بحال المؤمنين .  <br />
أمّا الكَافِرُ فليس أهلاً لأن يخاطبه الله - تعالى - فلهذا السّبب لم يذكر الله تعالى أنَّهُ خاطبهم بهذا الخطاب بل ذكر الله - تعالى - أنَّهُ بيَّن هذا الحكم .  <br />
و  "  نعم  "  حرف جواب ك  "  أجل  "  و  "  إي  "  و  "  جير  "  و  "  بلى  "  ،   ونقيضتها  "  لا  "  .  <br />
و  "  نَعَمْ  "  تكون لتصديق الإخبار ،   أو إعلام استخبار ،   أو وَعْدِ طالب ،   وقد يُجَابُ بها النَّفي المقرون باستفهام وهو قليل جدّاً كقول جحْدَرٍ : [ الوافر ]<br />
أَليْسَ اللَّيْلُ يَجْمَعُ أمَّ عَمْروا *** وإيَّانَا فَذَاكَ بِنَا تَدَانِي<br />
نَعَمْ ،   وتَرَى الهِلالَ كَمَا أرَاهُ *** ويَعْلُوهَا النَّهَارُ كَمَا عَلاَنِي{[16161]}<br />
فأجاب قوله :  "  ألَيْسَ  "  ب  "  نعم  "  ،   وكان من حقه أن يقول : بلى ،   ولذلك يُرْوَى عن ابن عباس في قوله تعالى : { أَلَسْتَ بِرَبِّكُمْ قَالُواْ بلى } [ الأعراف : 172 ] : لو قالوا : نعم لكفروا ،   وفيه بحثٌ يأتي إنْ شَاءَ اللَّهُ - تعالى - قريباً .  <br />
وقرأ الكسائِيُّ{[16162]} والأعمشُ ويحيى بن وثَّابٍ بكسر عينها ،   وهي لغة  "  كنَانَة  "  ،   وطعن أبُو حَاتِمِ عليها وقال :  "  ليس الكسر بمعروف  "  .  <br />
واحتج الكِسَائِيُّ لقراءته بما يُحكى عن عمر بن الخطاب أنَّه سأل قوماً فقالوا : نعم بالفتح ،   فقال :  "  أمَّا النَّعَم فالإبل فقولوا : نَعِم  "  أي بالكَسْرِ .  <br />
قال أبو عبيد :  "  ولم نَرَ العرب يعرفون ما رَوَوْه عن عمر ونراه مُوَلَّداً  "  .  <br />
قال شهاب الدين : وهذا طَعْنٌ في المُتَواتِر فلا يُقبل ،   وتبدل عينها حاءً ،   وهي لغة فاشية ،   كما تبدل حاء  "  حتى  "  عيناً .  <br />
قوله : { فَأَذَّنَ مُؤَذِّنٌ } .  <br />
التأذين في اللُّغَةِ النداء والتّصويت الإعلام ،   والأذان للصّلاة إعلام بها وبوقتها .  <br />
وقالوا في  "  أذَن مؤذّن  "  : نادى مناد أسمع الفريقين .  <br />
قال ابن عباس :  "  وذلك المؤذن من الملائكة وهو صاحبُ الصُّورِ{[16163]}  "  .  <br />
قوله :  "  بينهم  "  يجوز أن يكون منصوباً ب  "  أذَّن  "  أو ب  "  مؤذن  "  فعلى الأول التقدير :<br />
أنَّ المؤذن أوقع ذلك الأذان بينهم أي في وسطهم .  <br />
وعلى الثَّاني التَّقديرُ : أنَّ مؤذِّناً من بينهم أذَّن بذلك الأذان ،   والأول أوْلَى .  <br />
وأن يكون مُتَعَلِّقاً بمحذوف على أنَّهُ صفة ل  "  مؤذّن  "  قال مكيّ{[16164]} - عند إجازته هذا الوَجْهِ - :  "  ولكن لا يعمل في  "  أنْ  "  مؤذِّن  "  إذ قد نعته  "  يعني أنَّ قوله :  " أَن لَّعْنَةُ الله "  لا يجوز أنْ يكون معمولاً ل  "  مؤذّن  "   ؛ لأنَّهُ موصوف واسم الفاعل متى وصف لم يعمل .  <br />
قال شهابُ الدِّين{[16165]} :  "  وهذا يوهم أنَّا إذا لم نجعل  "  بَيْنَهُمْ  "  نعتاً ل  "  مؤذِّن  "  جاز أن يعمل في  "  أنْ  "  ،   وليس الأمر كذلك ؛ لأنَّكَ لو قلت : ضرب ضَارِبٌ [ زيداً تنصب زيداً ب  "  ضرب  "  لا ب  "  ضارب  "  ] .  <br />
لكني قد رأيت الواحِدِي أجاز ما أجاز مكيّ من كون  "  مؤذّن  "  عاملاً في  "  أن  "  ،   وإذا وصفته امتنع ذلك ،   وفيه ما تقدّم وهو حسن .  <br />
قوله : { أَن لَّعْنَةُ الله عَلَى الظالمين }  "  أنْ  "  يجوز أن تكون المفسِّرة ،   وأن تكون المخففة ،   والجملة الاسميَّة بعدها الخبر ،   فلا حاجة هنا لفاصل .  <br />
وقرأ الأخوان{[16166]} ،   وابن عامر ،   والبزِّي :  "  أنَّ  "  بفتح الهمزة وتشديد النون ،   ونصب  " اللَّعنة  "  على أنَّهَا اسمها ،   و  "  على الظالمين  "  خبرها ،   وكذلك في [ النور 7 ] { أَن لَّعْنَةُ الله عَلَيْهِ } خفَّف  "  أنْ  "  ورفع اللّعنة نافع وحده ،   والباقون بالتشديد والنَّصب .  <br />
[ قال الواحِديُّ : مَنْ شدّد فهو الأصلُ ،   ومن خفَّف فهو مخففة من التشديد على إرادة إضمار القصّة والحديث تقديره : أنه لعنة الله ،   ومثله قوله تعالى : { وَآخِرُ دَعْوَاهُمْ أَنِ الحمد للَّهِ } [ يونس : 10 ] التقدير : أنَّهُ ،   ولا يخفف  "  أنْ  "  هذه إلا وتكون بعد إضمار الحديث والشأن ] .  <br />
وقرأ{[16167]} عصمةُ عن الأعمشِ :  "  إنَّ  "  بالكسر والتشديد ،   وذلك : إمَّا على إضمار القول عند البصريين ،   وإمَّا على إجراء النِّداء مُجْرى القول عند الكوفِيِّين .
