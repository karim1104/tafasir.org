قوله تعالى : { تِلْكَ القرى نَقُصُّ عَلَيْكَ } .  <br />
قال الزَّمَخْشَرِيُّ كقوله : { وهذا بَعْلِي شَيْخاً إِنَّ } [ هود : 72 ] في كونه مبتدأ وخبراً وحالاً يعني أن  "  تِلْكَ  "  مبتدأ مشارٌ بها إلى ما بعدها ،   و  "  القُرَى  "  خبرها ،   و  "  نَقُصُّ  "  حال أي قاصِّينَ كقوله : { فَتِلْكَ بُيُوتُهُمْ خَاوِيَةً } [ النمل : 52 ] .  <br />
قال الزمخشريُّ{[16613]} : فإن قلت : ما معنى  "  تِلْكَ القُرَى  "  حتى يكون كلاماً مفيداً ؟<br />
قلت : هو مفيدٌ ولكن بالصِّفةِ في قولك :  "  هو الرَّجُلُ الكريم  "  يعني أنَّ الحال هنا لازمة ليفيد التَّركيب كما تلزم الصِّفةِ في قولك :  "  هو الرَّجُلُ الكريمُ  "  ألا ترى أنَّكَ لو اقتصرت على  "  هو الرَّجُلُ  "  لم يكن مفيداً ،   ويجوزُ أن تكون  "  القُرَى  "  صفة لتلك ،   و  "  نقصُّ  "  الخبر ،   ويجوز أن يكون  "  نقصُّ  "  خبراً بعد خبر .  <br />
و  "  نَقُصُّ  "  يجوز أن يكون على حاله من الاستقبال أي : قد قصصنا عليك من أبْنَائِهَا ونحن نَقُصُّ عليك أيضاً بعض أبنائها [ ويجوز أن يكون عبر به عن الماضي ،   أي : قد قَصَصْنَا عليك من أبنائها ]{[16614]} وأُشير بالبُعْدِ تنبيهاً على بعد هلاكها وتقادمه عن زمن الإخبار فهو من الغيب ،   وأراد القصص المتقدمة .  <br />
وفي قوله :  "  القُرَى  "  ب  "  أل  "  تعظيم كقوله تعالى : { ذَلِكَ الكتاب } [ البقرة : 22 ] ،   وقول الرسول - عليه الصلاة والسلام :  "  أوْلَئِكَ المَلأُ مِنْ قُريشٍ  "  ،   وقول أمية : [ البسيط ]<br />
تِلْكَ المَكَارِمُ لا قَعْبَانِ مِنْ لَبَنٍ *** شِيبا بِمَاءٍ فَعَادَا بَعْدُ أبْوَالاً{[16615]}<br />
و  "  مِنْ  "  للتبعيض كما تقدَّم ؛ لأنَّهُ إنَّما قصَّ عليه - عليه الصلاة والسلام - ما فيه عظةٌ وانزجارٌ دون غيرهما ،   وإنَّما قصّ أنباء أهل القرى ؛ لأنَّهم اغتروا بطول الإمهال مع كثرة النِّعم فتوهموا أنَّهُم على الحقِّن فذكرها الله - تعالى - لقوم محمد - عليه الصلاة والسلام - ليحترزوا عن مثل تلك الأعمال ،   ثم قال : { وَلَقَدْ جَاءَتْهُمْ رُسُلُهُم بالبينات } .  <br />
قوله : { فَمَا كَانُواْ لِيُؤْمِنُواْ بِمَا كَذَّبُواْ } الظَّاهِرُ أنَّ الضَّمائرَ عائدة على أهل القرى .  <br />
وقال يمان بن رئاب :  "  إنَّ الضميرين الأوَّلين لأهل القرى ،   والضَّميرُ في  "  كذَّبُوا  "  لأسْلافِهم .   وكذا جوَّزهُ ابن{[16616]} عطية أيضاً أي :  "  فما كان الأبناء ليؤمِنُوا بما كذَّب به الآباء  "  ،   وقد تقدَّم الكلامُ على لام الجُحُودِ ،   وأنَّ نفي الفِعْلِ معها أبْلَغُ .  <br />
و  "  ما  "  موصولة اسميَّة ،   وعائدها مَحْذُوفٌ ؛ لأنَّه مَنصوبٌ متَّصل أي : بما كذبوه [ ولا يجوز أن يقدر به وإن كان الموصول مجرورا بالباء أيضاً لاختلاف المتعلق ،   وقال هنا  "  بما كذبوا  "  ]{[16617]} فلم يذكر متعلق التكذيب ،   وفي  "  يونس  "  ذكره فقال : { بِمَا كَذَّبُواْ بِهِ } [ يونس : 74 ] ،   والفرق أنَّهُ لمَّا حذفه في قوله :{ ولكن كَذَّبُواْ } [ الأعراف : 96 ] أستمرّ حذفه بعد ذلك ،   وأمَّا في يونس فقد أبرزه في قوله : { فَكَذَّبُوهُ فَنَجَّيْنَاهُ } [ يونس 74 ] .  <br />
{ كَذَّبُواْ بِآيَاتِنَا } [ يونس : 73 ] فناسب ذكره موافقة قال معناه الكرمَانِيّ .  <br />
فصل في معنى  "  ما كانوا ليؤمنوا  " <br />
قال ابن عباس والسُّدِّيُّ :  "  فما كان أولئك الكفار ليؤمنوا عند إرسالي بما كذبوا به يوم أخذنا مِيثاقَهُم حين أخرِجُوا من ظهر آدم فآمنوا كرهاً وأقرُّوا باللِّسانِ وأضمروا التَّكذيب  " {[16618]} .  <br />
وقال الزَّجَّاجُ{[16619]} :  "  فما كانوا ليؤمنوا بعد رؤيةِ تلك المعجزات بما كَذَّبُوا قبل رؤية المُعْجِزَاتِ  "  .  <br />
وقيل : معناه ما كانوا لو أحييناهم بعد إهلاكهم ورددناهم في دار التكليف ليؤمنوا بما كذَّبُوا به قبل إهلاكهم ،   ونظيره قوله تعالى : { وَلَوْ رُدُّواْ لَعَادُواْ لِمَا نُهُواْ عَنْهُ } [ الأنعام : 28 ] .  <br />
[ وقيل : إنه قبل مجيء الرسول كانوا مصرين على الكفر فهؤلاء ما كانوا ليؤمنوا بعد مجيء الرسل ]{[16620]} وقيل : ليؤمنوا في الزَّمانِ المستقبل<br />
قوله : { كَذَلِكَ يَطْبَعُ الله على قُلُوبِ الكافرين } .  <br />
قال الزَّجَّاجُ : والكاف في  "  كذلك  "  في محلِّ نصب [ أي : مثل ذلك الطبع على قلوب أهل القرى المنتفي عنهم الإيمان يطبع الله على قلوب الكفرة الجانين ]{[16621]} .  <br />
قوله تعالى : { وَمَا وَجَدْنَا لأَكْثَرِهِم مِّنْ عَهْدٍ وَإِن وَجَدْنَا أَكْثَرَهُمْ لَفَاسِقِينَ 102 } .  <br />
قوله  "  لأكْثَرِهِمْ  "  فيه وجوه :<br />
الظَّاهِرُ أنَّهُ متعلقٌ بالوجْدانِ كقولك : ما وجدتُ له مالاً أي : ما صَادَفْتُ له مالاً ولا لقيته .  <br />
الثاني : أن يكون حالاً من  "  عهد  "   ؛ لأنَّه في الأًصل صفة نكرةٍ فلما قُدِّم عليها نُصب على الحال ،   والأصْلُ : ما وجدنا عهداً لأكثرهم ،   وهذا ما لم يذكر أبُو البقاءِ غيره .  <br />
وعلى هذين الوجهين ف  "  وَجَدَ  "  متعدِّية لواحد وهو  "  من عَهْدٍ  "  ،   و  " منْ "  مزيدةٌ فيه لوجود الشرطين .  <br />
الثالث : أنَّهُ في محلِّ نصب مفعولاً ثانياً لوجَدَ إذ هي بمعنى علمية ،   والمفعول هو  "  مِنْ عَهْدٍ  "  .   وقد يترجَّحُ هذا بأنَّ  "  وَجَدَ  "  الثانية علمية لا وجدانيَّة بمعنى الإصابة ،   وسيأتي دليل ذلك .   وإذا تقرَّر هذا فينبغي أن تكون الأولى كذلك مطابقة للكلام ومناسبة له ،   ومن يرجّح الأوَّل يقولُ : إنَّ الأولى لمعنى : والثَّانية لمعنى آخر .  <br />
فصل في معنى الآية<br />
قال ابن عباس : يريدُ : وما وجدنا لأكثرهم من عهد ،   الوفاء بالعهد الذي عاهدهم عليه وهم في صلب آدم{[16622]} حيث قال : { أَلَسْتَ بِرَبِّكُمْ قالوا بلى } [ الأعراف : 172 ] .  <br />
وقال ابن مَسْعُودٍ :  "  المرادُ بالعهد هاهنا الإيمان ،   لقوله تعالى : { إِلاَّ مَنِ اتخذ عِندَ الرحمن عَهْداً } [ مريم : 87 ] أي قال : لا إله إلا الله  "  .  <br />
وقيل : المرادُ بالعهدِ وضع الأدلَّةِ على صِحَّةِ التَّوحيد والنَّبوةِ [ تقديره : ] وما وجدنا لأكثرهم من الوفاءِ بالعهد .  <br />
قوله :  "  وَإنْ وَجدْنَا  "   "  إنْ  "  هذه هي المخفَّفةُ وليس هنا عاملة لمباشرتها الفعل فَزَالَ اختصاصُهَا المُقْتَضِي لإعمالها .  <br />
وقال الزَّمخشريُّ{[16623]} :  "  وإنَّ الشَّأن والحديث وجدنا  "  .  <br />
فظاهِرُ هذه العبارة أنَّها مُعْملَة ،   وأنَّ اسمها ضميرُ الأمر والشَّأن ،   وقد صرَّحَ أبُو البقاء{[16624]} هنا بأنَّها معملةٌ ،   وأن اسمها محذوف ،   إلا أنَّهُ لم يقدِّره ضمير الحديث بل غيره فقال :  "  واسمها محذوفٌ أي : إنَّا وَجَدْنَا  "  .   وهذا مذهب النَّحويين أعني اعتقاد إعمال المخفَّف من هذه الحروف في  "  أن  "  المفتوحة على الصَّحيح ،   وفي  "  كأن  "  التَّشبيهية ،   وأمَّا  "  إنْ  "  المخففة المكسورة فلا .   وقد تقدَّم إيضاحه .  <br />
ووجدنا هنا متعدية لاثنين أولهما  "  أكْثَرَهُم  "  ،   والثاني  "  لفاسقين  "  ،   قال الزمخشريُّ : والوجود بمعنى العلم من قولك : وجدتُ زيداً ذا الحفاظ بدليل دخول  "  إنْ  "  المخفَّفة ،   واللاَّم الفارقة ولا يسوغ ذلك إلاَّ في المبتدأ والخبر والأفعالِ الدَّاخلة عليهما يعني أنها مختصة بالابتداء ،   وبالأفعال النَّاسخةِ له ،   وهذا مذهبُ الجُمْهُورِ ،   وقد تقدَّم الخلافُ عن الأخْفَشِ أنَّهُ يجوزُ على غيرها ،   وتقدَّم دليله على ذلك ،   واللاَّمُ فارقة وقيل : هي عوض من التَّشديد .  <br />
قال مَكيٌّ{[16625]}  "  ولزمت اللاَّمُ في خبرها عوضاً من التشديد والمحذوف الأوَّلِ ،   وقد تقدَّم أنَّ بعض الكوفيين يجعلون  "  إن  "  نافية ،   واللاَّم بمعنى  " إلاَّ  "  في قوله تعالى : { وَإِن كَانَتْ لَكَبِيرَةً } [ البقرة : 143 ] .  <br />
ومعنى فاسقين خارجين عن الطَّاعةِ ،   مارقين عن الدِّينِ ،   وقيل : ناقضين العَهْدَ .  <br />
وقوله :  "  لأكْثَرهُم  "  ،   و  "  أكْثَرهُم  "  ،   و  "  من بعدهم  "  : إنْ جعلنا هذه الضَّمائر كلَّها للأمم السَّالِفَةِ فلا اعتراضَ ،   وإن جعلنا الضَّمير في  " لأكثرهُم  "  و  "  أكْثَرهُمْ  "  لعموم النَّاسِ والضَّمير في  "  من بعدهم  "  للأمم السَّالِفَةِ كانت هذه الجملة - أعني ما وجدنا - اعتراضاً كذلك قاله الزمخشريُّ{[16626]} ،   وفيه نظر ؛ لأنَّهُ إذا كان الأوَّلُ عاماً ثم ذكر شيء يندرجُ فيه ما بعده وما قبله كيف يُجْعل ذلك العامُّ معترضاً بين الخاصين .  <br />
وأيضاً ،   فالنَّحْويُّون إنما يعرفون الاعتراض فيما اعترض به بين متلازمين ،   إلاَّ أنَّ أهل البيان عندهم الاعتراض أعمُّ من ذلك ،   حتَّى إذا أتي بشيء بين شيئين مذكورَيْنِ في قصَّةٍ واحدة سَمَّوه اعتراضاً{[16627]} .
