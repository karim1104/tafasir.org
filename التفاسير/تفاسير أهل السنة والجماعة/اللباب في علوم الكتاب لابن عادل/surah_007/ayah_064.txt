قوله :  "  فَكَذَّبُوهُ  "  : أي في ادِّعَاءِ النُّبوُّةِ والرِّسالةِ .  <br />
 "  فَأنْجَيْنَاهُ  "  من الطُّوفان ،   وأنجْينا من كان معه [ وكانوا أرْبَعِينَ رجلاً ،   وأربعين امرأة ]{[16393]} .  <br />
وقيل : عشرةٌ : بَنُوه : حَامٌ ،   وسامٌ ،   ويافث ،   وسبعة ممن آمن معه من المؤمنين .  <br />
 "  فِي الفُلْكِ  "  أي : في السَّفِينَةِ ،   وأغرقنا الكُفَّارَ والمكذِّبين ،   وبين العِلَّة في ذلك فقال : { إِنَّهُمْ كَانُواْ قَوْماً عَمِينَ } .  <br />
قوله :  "  فِي الفُلْكِ  "  يجوزُ أن يتعلق ب  "  أنْجَيْنَاهُ  "  ،   أي : أنجيناه في الفلك ،   [ ويجوز أن تكون  "  فِي  "  حينئذٍ سببيَّةً أي : بسبب الفُلْكِ ]{[16394]} كقوله :  "  إنَّ امْرَأةً دخلت النَّارَ في هِرَّةٍ{[16395]}  "  ،   ويجوزُ{[16396]} أن يتعلق في الفلك بما تعلَّق به الظَّرْفُ الواقع صلةً ،   أي : الذين استقرُّوا في الفلك معه .  <br />
 "  وعَمِيْنَ  "  جمع عَمٍ ،   وقد تقدَّم الكلامُ على هذه المادة{[16397]} .  <br />
وقيل : عمٍ هنا إذا كان أعْمَى البصيرة ،   [ قال ابن عباس : عَمِيَتْ قلوبُهُم عن معرفة التَّوْحِيد ،   والنَّبوة والمعَادِ{[16398]} قال أهل اللُّغَة ]{[16399]} : غير عارفٍ بأموره ،   وأعْمَى أي في البَصَرِ .  <br />
قال زُهَيْرٌ : [ الطويل ]<br />
وأعْلَمُ عِلْمَ اليَوْمِ والأمْسِ قَبْلَهُ *** ولَكِنَّنِي عَنْ عِلْمِ مَا فِي غَدٍ عَمٍ{[16400]}<br />
قاله اللَّيْثُ وقيل : عم وأعْمَى بمعنىً ،   كخَضِرٍ وأخْضَر .  <br />
وقال بعضهم :  "  عَم  "  فيه دلالةٌ على ثبوت الصِّفةِ واستقرارها [ كفَرِح ] وضيّق ،   ولو أريد الحدوث لَقِيلَ : عامٍ كما يقال : فارحٌ وضَائِقٌ .  <br />
وقال الزَّمَخْشَرِيُّ{[16401]} : قرئ  "  قَوْماً عَامِينَ  "  .
