لقوله بعد ذلك { وَمَا تَنقِمُ مِنَّا } فإنَّ الضَّميرَ في مِنَّا يَخُصُّهُمْ .  <br />
وجَوَّزُوا أن يعود عليهم ،   وعلى فرعون ،   أي : إنَّا - نحن وأنت - ننقلب إلى اللَّهِ ،   فيُجازي كلاًّ بعمله ،   وهذا وإن كان هو الواقعَ إلاَّ أنَّهُ ليس من هذا اللَّفظِ .  <br />
قوله وَمَا تَنقِمُ قد تقدَّم في المائدةِ أنَّ فيه لغتين وكيفية تعدِّيه ب  "  مِنْ  "  وأنَّهُ على التَّضمين .  <br />
وقوله : { إلاَّ أنْ آمنَّا } يجوز أن يكون في محلَِّ نصبٍ مفعولاً به ،   أي : ما تَعِيبُ علينا إلاَّ إيماننا ويجوزُ أن يكون مفعولاً من أجله ،   أي : ما تنال مِنَّا وتعذِّبنا لشيءٍ من الأشياء إلاَّ لإيماننا وعلى كلا القولينِ فهو استثناءٌ مفرغ .  <br />
قوله  "  لَمَّا جَاءَتْنَا  "  يجوزُ أن تكون ظرفيةً كما هو رأي الفارس ،   وأحد قولي سيبويه ،   والعامِلُ فيها على هذا آمَنَّا أي : آمنَّا حين مجيء الآيات ،   وأن تكون حرف وجوب لوجوب ،   وعلى هذا فلا بدَّ لها من جواب وهو محذوف تقديره : لما جاءتنا آمّنا بها من غير توقّف .  <br />
قوله :  "  ربنا أفرغ علينا صبرا  "  معنى الإفراغ في اللَّغَةِ : الصَّبُّ .   وأصله من إفراغ الإناء وهو صب ما فيه بالكليَّة ،   فكأنَّهُمْ طلبوا من اللَّه كلَّ الصَّبْرِ لا بعضه .  <br />
ونكَّرُوا  "  الصَّبْر  "  وذلك يدلُّ على الكمالِ والتَّمَامِ ،   أي : صبراً كَاملاً تاماً ،   كقوله تعالى : { وَلَتَجِدَنَّهُمْ أَحْرَصَ الناس على حَيَاةٍ } [ البقرة : 96 ] أي : على حياة كاملة تامَّةٍ .  <br />
وقوله : { وَتَوَفَّنَا مُسْلِمِينَ } أي : توفنا على الدِّين الحقِّ الذي جاء بِهِ موسى .   واحْتَجَّ القاضي بهذه الآية على أنَّ الإيمان والإسلامَ واحد .  <br />
فقال : إنَّهُم قالوا أوَّلاً : { آمَنَّا بِآيَاتِ رَبِّنَا } ،   ثمَّ ثانياً ،    "  وَتَوَفَّنَا مُسْلِمينَ  "  ،   فوجب أن يكون هذا الإسلام هو ذاك الإيمان .
