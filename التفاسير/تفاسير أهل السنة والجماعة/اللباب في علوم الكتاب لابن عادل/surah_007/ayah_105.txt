قرأ العامة  "  على أنْ  "  ب  "  عَلَى  "  التي هي حرف جر داخلة على أن وما في حيّزها .  <br />
ونافع قرأ{[16630]}  "  عليّ  "  ب  "  عَلَى  "  التي هي حرف جرّ داخلة على ياء المتكلِّم .  <br />
فأما قراءة العامة ففيها سِتَّةٌ أوْجُهٍ ،   ذكر الزَّمخشريُّ منها أربعة أوجه :<br />

 .   .   .   *** وتَشْقَى الرِّمَاحُ بالضَّيَاطِرَةِ الحُمْرِ{[16631]}<br />
معناه : وتشقى الضياطرة بالرِّمَاحِ .  <br />
قال أبُو حيَّان{[16632]} :  "  وأصحابنا يخصُّون القلب بالضَّرُورةِ ،   فينبغي أن يُنزَّه القرآن عنه  "  .  <br />
وللنَّاس فيه ثلاثةُ مذاهب : الجواز مطلقاً ،   [ المنع مطلقاً ]{[16633]} ،   التَّفصيل : بين أن يفيد معنىً بديعاً فيجوزُ ،   أو لا فيمتنع ،   وقد تقدَّم إيضاحه ،   وسيأتي منه أمثلة أخر في القرآن العظيم .  <br />
وعلى هذا الوجه تصيرُ هذه القراءة كَقِرَاءةِ نافع في المعنى ،   إذ الأصلُ : قول الحق حقيق عليَّ ،   فقلب اللفظ فصار :  "  أنَّ حقيق على قول الحقِّ  "  .  <br />
قال :  "  والثاني : أن ما لزمك فقد لزمته ،   فلمَّا كان قول الحقِّ حقيقاً عليه كان هو حقيقاً على قول الحقِّ أي لازماً له  "  .  <br />
والثالث : أن يضمَّنَ حقيق معنى حريص كما ضمت  "  هيجّني  "  معنى ذكْرني في البيتِ المذكور في كتاب سيبويه وهو قوله : [ البسيط ]<br />
إذَا تَغَنَّى الحَمَامُ الوُرْقُ هَيَّجَنِي *** وَلَوْ تَسَلَّيْتُ عَنْهَا أمَّ عَمَّارِ{[16634]}<br />
الرابع : أن تكون  "  عَلَى  "  بمعنى  "  الباء  "  ،   وبهذا الوجه قال أبو الحسن والفراء{[16635]} والفارسيُّ{[16636]} ،   قالوا : إنَّ  "  على  "  بمعنى الباء كما أن الباء بمعنى  "  على  "  في قوله : { وَلاَ تَقْعُدُواْ بِكُلِّ صِرَاطٍ } [ الأعراف : 86 ] أي : على كلٍّ .  <br />
وقال الفرَّاءُ : العرب تقول : رَمَيْتُ على القوس وبالقوس وجئتُ على حالٍ حسنة وبحال حسنة ،   وتؤيده قراءة أبيّ والأعمش  "  حقيق بأن لا أقول  "  إلاَّ أنَّ الأخفش قال :  "  وليس ذلك بالمطَّرد لو قلت :  "  ذهبتُ على زيْدٍ  "  تريد :  "  بزيدٍ  "  لم يجز  "  ،   وأيضاً فلأن مذهب البصريِّين عدم التجوُّزِ في الحُرُوفِ .  <br />
الخامس : - وهو الأوجه والأدخل في نكت القرآن - أن يغرق موسى - عليه الصلاة والسلام - [ في وصف نفسه ]{[16637]} بالصِّدْق في ذلك المقام لا سيما وقد رُوِيَ أنَّ فرعون - لعنه الله - لمَّا قال موسى : إنِّي رسول من رب العالمين قال له : كذبت فيقول : أنا حقيقٌ على قول الحقِّ أي : واجب عليَّ قول الحقِّ أن أكون أنا قائله والقائم به ،   ولا يَرْضى إلاَّ بمثلي ناطقاً به .  <br />
قال أبُو حيَّان{[16638]} : ولا يصحُّ هذا الوجه إلاَّ إن عنى أنه يكون  "  أن لا أقُول "  صفة له كما تقول : أنَا على قول الحقِّ أي : طريقتي وعادتي قول الحقّ .  <br />
السادس : أن تكون  "  على  "  متعلقة ب  "  رَسُول  "  .  <br />
قال ابن مقسم : حقيقٌ من نعت  "  رَسُول  "  أي رسول حقيق من ربِّ العالمين أرْسِلْتُ على ألاَّ أقولَ على الله إلا الحقَّ ،   وهذا معنى صحيح واضحٌ ،   وقد غفل أكثرُ المفسرين من أرباب اللُّغَة عن تعليق  "  على  "  ب  "  رسول  "  ،   ولم يخطر لهم تعليقه إلاَّ ب  "  حقيق  "  .  <br />
قال أبُو حيَّان{[16639]} : وكلامه فيه تَنَاقضٌ في الظَّاهِرِ ؛ لأنَّهُ قدَّر أولاً العامل في  "  عَلَى  "   "  أرسلت  "  وقال أخيراً :  "  لأنهم غفلوا عن تَعْليق  "  على  "  ب  " رسول  "  ،   فأمَّا هذا الأخير فلا يجوزُ عند البصريين ؛ لأنَّ رسولً قد وُصِف قبل أن يأخذ معموله ،   وذلك لا يَجُوزُ ،   وأمَّا تعليقه ب  "  أرسلت  "  مقدَّراً لدلالةِ لفظ  "  رَسُوله  "  عليه فهو تقديرٌ سائغ .  <br />
ويتأوَّل كلامه أنَّه أراد بقوله تُعَلَّقُ  "  على  "  ب  "  رسول  "  أنه لمَّا كان دالاًّ عليه صحَّ نسبة التَّعلق له .  <br />
قال شهابُ الدِّين{[16640]} :  "  وقال أبُو شامَةَ بعد ما ذكر هذا الوجه عن ابن مقسم : والأوْجهُ الأربعةُ التي للزمخشريِّ ولكن هذه وجوهٌ متعسِّفةٌ ،   وليس المعنى إلاَّ على ما ذكرته أوَّلاً ،   يعني وجه ابن مقسم ،   وهذا فيه الإشكال الذي ذكره الشيخ يعني أبا حيَّان يعني من إعمال اسم الفاعل أو الجاري مجراه وهو موصوفٌ  "  .  <br />
وأمَّا قراءة نافع فواضحةٌ وفيها ثلاثةُ أوجُهٍ :<br />
أحدها : أن يكون الكلامُ قد تم عند قوله :  "  حقيق  "  ،   و  "  عليَّ  "  خبر مقدَّمٌ ،    "  ألاَّ أقُولَ  "  مبتدأ مؤخر ،   كأنَّهُ قيل : عليَّ عدم قول غير الحقِّ أي : فلا أقُولُ إلا الحقَّ .  <br />
الثاني : أن يكون  "  حَقِيقٌ  "  خبراً مقدماً ،   و  "  ألاَّ أقولَ  "  مبتدأ على ما تقدَّم بيانه .  <br />
الثالث :  "  أن لا أقول  "  فاعلٌ ب  "  حقيقٌ  "  كأنَّهُ قيل : يحقُّ ويجبُ أن لا أقول ،   وهذا أغربُ الوُجُوهِ لوضُوحِهِ لفظاً ومعنى ،   وعلى الوجهين الأخيرين تتعلَّق  "  عليَّ  "  ب  "  حقيقٌ  "  لأنَّك تقول :  "  حقَّ عليهِ كذا  "  قال تعالى : { أولئك الذين حَقَّ عَلَيْهِمُ القول } [ الأحقاف : 18 ] .   وعلى الوجْهِ الأوَّلِ يتعلَّق بمحذوف على ما تقرر .  <br />
وأمّا رفع  "  حقيقٌ  "  فقد تقدَّم أنَّهُ يجوز أن يكون خَبَراً مقدَّماً ،   ويجوز أن يكون صفةً ل  "  رَسُول  "  ،   وعلى هذا فيضعف أن يكون  "  مِنْ رب  "  صفةً لئلا يلزم تقديم الصفة غير الصّريحَة [ على الصَّريحة ] ،   فينبغي أن يكون متعلّقاً بنفس  "  رَسُول  "  ،   وتكون  "  مِنْ  "  لابتداء الغاية مجازاً .  <br />
ويجوز أن يكون خبراً ثانياً .   ويجوز أن يكون مبتدأ وما بعده الخبر على قراءة من شددَّ الياء ،   وسوَّغَ الابتداء بالنكِرَةِ حينئذٍ تعلُّق الجارِّ بها .  <br />
فقد تحصَّل في رفعه أربعة أوجُهٍ ،   وهل هو بمعنى فاعل ،   أو بمعنى مفعول ؟ الظَّاهِرُ أنَّهُ يحتمل الأمرين مُطْلَقاً ،   أعني على قراءة نَافِعٍ وقراءة غيره .  <br />
وقال الوَاحِدِيُّ نَاقِلاً عن غيره :  "  إنَّهُ مع قراءةِ نافع محتمل للأمرين ،   ومع قراءة العامَّةِ بمعنى مفعول فإنَّهُ قال :  "  وحقيق على هذا القراءةِ - يعني قراءة نَافِعٍ - يجوز أن يكون بمعنى فاعل  "  .  <br />
