إن قيل : إن الحروف المقطّعة في أوائل السور أسماء حروف التهجِّي ،   بمعنى أن الميم اسم ل  " مَهْ "  ،   والعين ل  " عَهْ "  ،   وإن فائدتها إعلامهم بأن هذا القرآن منتظمٌ من جنس ما تنتظمون من كلامكم ،   ولكن عجزتم عنه ،   فلا محلّ لها حينئذ من الإعراب ،   وإنما جيء بها لهذه الفائدة ،   فألقيت كأسماء الأعداد ونحو :  " واحد اثنان "  ،   وهذا أصح الأقوال الثلاثة ،   أعني أن في الأسماء التي لم يقصد الإخبار عنها ولا بها ثلاثة أقوال :<br />
أحدها : ما تقدم .  <br />
والثاني : أنها معربة ،   بمعنى أنها صالحة للإعراب ،   وإنما فات شرطه وهو التركيب ،   وإليه مال الزمخشري رحمه الله .  <br />
والثالث : أنها موقوفة أي لا معربة ولا مبنيةٌ .  <br />
أو إن قيل : إنها أسماء السور المفتتحة بها ،   أو إنها بعض أسماء الله - تعالى - حذف بعضها ،   وبقي منها هذه الحروف دالّة عليها وهو رأي ابن عَبّاس - رضي الله تعالى عنهما - كقوله : الميم من  " عليهم "  ،   والصاد من  " صادق "  ،   فلها حينئذ محلّ من الإعراب ويحتمل الرفع والنصب والجر :<br />
فالرفع على أحد وجهين : إما بكونها مبتدأ ،   وإما بكونها خبراً كما سيأتي بيانه مفصلاً إن شاء الله تعالى .  <br />
والنصب على أحد وجهين أيضاً : إما بإضمار فعلٍ لائقٍ ،   تقديره : اقرءوا :  " الم "  ،   وإما بإسقاط حرف القسم ؛ كقول الشاعر : [ الوافر ]<br />
إِذا مَا الْخُبْزُ تَأْدِمُهُ بِلَحْمٍ *** فَذَاكَ أَمَانَةَ اللهِ الثَّرِيدُ{[430]}<br />
يريد : وأَمَانَةِ اللهِ .  <br />
وكذلك هذه الحروف أقسم الله بها .  <br />
وقد رد الزَّمَخْشَرِيّ هذا الوجه بما معناه : أنّ القرآن في : { ص وَالْقُرْآنِ ذِي الذِّكْرِ } [ ص : 1 ] والقلم في<br />
{ ن وَالْقَلَمِ وَمَا يَسْطُرُونَ } [ القلم : 1 ] محلوف بهما لظهور الجرّ فيهما ،   وحينئذ لا يخلو أن تُجْعَل الواو الداخلة عليها للقسم ،   أو للعطف ،   والأول يلزم منه محذور ،   وهو الجمع بين قسمين على مقسم قال : [ وهم يستكرهون ]{[431]} ذلك .  <br />
والثاني ممنوع ،   لظهور الجَرّ فيما بعدها ،   والفرض أنك قدرت المعطوف عليه من مَحَلّ نصب ،   وهو رَدّ واضح ،   إلا أن يقال : هي في محلّ نصب إلا فيما ظهر فيه الجر [ فيما بعده ]{[432]} كالموضعين المتقدمين ؛ { حم وَالْكِتَابِ } [ الزخرف : 1-2 ] ،   و{ ق وَالْقُرْآنِ } [ ق : 1 ] ولكن القائل بذلك لم يفرق بين موضع وموضع ،   فالرد لازم كله .  <br />
والجَرّ من وجهٍ واحدٍ ،   وهو أنها مقسم بها ،   حذف حرف القسم ،   وبقي عمله كقولهم :  " اللهِ لأفعلنَّ "  أجاز ذلك الزمخشري ،   وأبو البقاء رحمهما الله ،   وهذا ضعيف ؛ لأن ذلك من خصائص الجَلاَلَة المعظمة لا يشاركها{[433]} فيه غيرها .  <br />
فتخلص مما تقدم أن في  " الم "  ونحوها ستة أوجه وهي : أنها لا محل لها من الإعراب ،   أَوْ لَهَا محل ،   وهو الرفع بالابتداء ،   أو الخبر .  <br />
والنَّصْب بإضمار فعل ،   أو حذف حرف القسم .  <br />
والجَرّ بإضمار حرف القسم{[434]} .  <br />
فصل في الحروف المقطعة<br />
سئل الشعبي{[435]} - رحمه الله تعالى - عن هذه الحروف فقال : سرّ الله ،   فلا تطلبوه .  <br />
وروى أبو ظِبْيَانَ{[436]} عن ابن عباس - رضي الله تعالى عنهما - قال : عجزت العلماء عن إدْرَاكِهَا ،   وقال الشعبي وجماعة رحمهم الله سائر حروف التهجّي في أوائل السور من المتشابهة الذي استأثر الله بعلمه ،   وهي سرّ القرآن ؛ فنحن نؤمن بظاهرها ،   ونَكِلُ العلمَ فيها إلى الله تعالى .  <br />
قال أبو بكر الصديق رضي الله تعالى عنه :  " في كل كتاب سِرّ ،   وسرُّ الله - تعالى - في القرآن أوائل السور "  .  <br />
ونقل ابنُ الخَطِيْبِ رحمه الله أن المتكلمين أنكروا هذا القول ،   وقالوا : لا يجوز أن يرد في كتاب الله ما لاَ يَكُونُ مفهوماً للخلق ،   واحتجوا عليه بآيات منها :<br />
قوله تبارك وتعالى : { أَفَلاَ يَتَدَبَّرُونَ الْقُرْآنَ أَمْ عَلَى قُلُوبٍ أَقْفَالُهَا } [ محمد : 24 ] بالتدُّبر في القرآن ،   ولو كان غير مفهوم ،   فكيف يأمر بالتدبّر فيه .  <br />
وكذا قوله : { أَفَلاَ يَتَدَبَّرُونَ الْقُرْآنَ وَلَوْ كَانَ مِنْ عِندِ غَيْرِ اللَّهِ لَوَجَدُواْ فِيهِ اخْتِلاَفاً كَثِيراً } [ النساء : 82 ] فكيف يأمر بالتدبُّر لمعرفة نفي التَّنَاقض والاختلاف ،   وهو غير مفهوم للخلق ؟<br />
ومنها قوله تعالى : { نَزَلَ بِهِ الرُّوحُ الأَمِينُ عَلَى قَلْبِكَ لِتَكُونَ مِنَ الْمُنذِرِينَ بِلِسَانٍ عَرَبِيٍّ مُّبِينٍ } [ الشعراء : 193- 195 ] ،   فلو لم يكن مفهوماً بطل كون الرسول - عليه السلام - منذراً به ،   وأيضاً قوله : { بِلِسَانٍ عَرَبِيٍّ مُّبِينٍ } يدلّ على أنه نازلٌ بلغة العَرَبِ ،   وإذا كان كذلك وجب أن يكون مفهوماً .  <br />
ومنها قوله تعالى : { لَعَلِمَهُ الَّذِينَ يَسْتَنبِطُونَهُ مِنْهُمْ } [ النساء : 83 ] ،   والاستنباط منه لا يمكن إلاّ مع الإحاطة بمعناه .  <br />
ومنها قوله تعالى : { تِبْيَاناً لِّكُلِّ شَيْءٍ } [ النحل : 89 ] .  <br />
وقوله تعالى : { مَّا فَرَّطْنَا فِي الكِتَابِ مِن شَيْءٍ } [ الأنعام : 38 ] .  <br />
وقوله تعالى : { هُدًى لِّلنَّاسِ } [ البقرة : 185 ] ،   { هُدًى لِّلْمُتَّقِينَ } [ البقرة : 2 ] ،   وغير المعلوم لا يكون هُدًى .  <br />
وقوله تعالى : { وَشِفَاءٌ لِّمَا فِي الصُّدُورِ وَهُدًى وَرَحْمَةٌ لِّلْمُؤْمِنِينَ } [ يونس : 57 ] وكل هذه الصفات لا تحصر في غير المعلوم .  <br />
وقوله تبارك وتعالى : { قَدْ جَاءَكُم مِّنَ اللَّهِ نُورٌ وَكِتَابٌ مُّبِينٌ } [ المائدة : 15 ] .  <br />
وقوله تعالى : { أَوَلَمْ يَكْفِهِمْ أَنَّا أَنزَلْنَا عَلَيْكَ الْكِتَابَ يُتْلَى عَلَيْهِمْ إِنَّ فِي ذَلِكَ لَرَحْمَةً وَذِكْرَى لِقَوْمٍ يُؤْمِنُونَ } [ العنكبوت : 51 ] فكيف يكون الكتاب كافياً وكيف يكون ذكره مع أنه غير مفهوم ؟<br />
وقوله تعالى : { هَذَا بَلاَغٌ لِّلنَّاسِ وَلِيُنذَرُواْ بِهِ } [ إبراهيم : 52 ] ،   فيكف يكون بلاغاً ؟ وكيف يقع به الإنذار ،   مع أنه غير معلوم ؟<br />
وقال في آخر الآية : { وَلِيُنذَرُواْ بِهِ وَلِيَعْلَمُواْ أَنَّمَا هُوَ إِلَهٌ وَاحِدٌ وَلِيَذَّكَّرَ أُوْلُواْ الأَلْبَابِ } ،   وإنما يكون كذلك لو كان معلوماً .  <br />
وقوله تعالى : { إِنَّ هَذَا الْقُرْآنَ يَهْدِي لِلَّتِي هِيَ أَقْوَمُ } [ الإسراء : 9 ] فكيف يكون هاديا ،   مع أنه غير معلوم ،   ومن الأخبار قوله عليه الصَّلاة والسَّلام :  " إنِّي تَركْتُ فيكم ما إِنْ تَمَسَّكْتُمْ به لن تَضِلُّوا بَعْدِي أبداً كتابَ الله وسُنَّتي " {[437]} فكيف يمكن التمسُّك به وهو غير معلوم ؟<br />
وعن علي بن أبي طالب -رضي الله تعالى عنه- أنه -عليه الصلاة والسّلام- قال :  " عليكم بكتاب الله - فيه نَبَأُ ما قبلكم ،   وخبر مَنْ بَعْدَكم ،   وحُكْم ما بينكم هو الفَصْلُ ليس بالهَزْلِ ،   من تركه من جَبَّار قَصَمَهُ الله تعالى .   ومن اتَّبَعَ الهدى في غيره أضَلّه الله - تعالى - هو حَبْلُ الله المتين ،   والذكر الحكيم ،   والصِّراط المستقيمُ ،   هو الذي لا تزيغُ به الأهواء ،   ولا تشبع به العلماء ،   ولا يخلق عن كثرة الرَّد ولا تنقضي عَجَائبه ،   مَنْ قَالَ به صَدَقَ ،   ومن حكم به عَدَل ،   ومن خاصم به فَلَج ،   ومن دَعَا إليه هُدِيَ إلى صِرَاط مستقيم " {[438]} .  <br />
ومن المعقول أنه لو ورد شيء لا سبيل إلى العِلْم إلاّ به لكانت المُخَاطبة به نحو مُخَاطبة العرب باللّغة الزنجية ،   ولما لم يَجُزْ ذلك فكذا هذا .  <br />
وأيضاً المقصود من الكلام الإفهام ،   فلو لم يكن مفهوماً لكانت المُخَاطبة عبثاً وسفهاً ،   وهو لا يليق بالحكيم .  <br />
وأيضاً أنَّ التَّحدِّي وقع بالقرآن ،   وما لا يكون معلوماً لا يجوز وقوع التَّحدِّي به .  <br />
واحتجّ مخالفوهم بالآية ،   والخَبَرِ ،   والمعقول .  <br />
أما الآية فهو أن المُتشَابه من القرآن ،   وأنه غير معلوم ؛ لقوله تَعَالَى : { وَمَا يَعْلَمُ تَأْوِيلَهُ إِلاَّ اللَّهُ } [ آل عمران : 7 ] والوقف هاهنا ،   لوجوه :<br />
أحدها : أن قوله تَعَالَى :  " والراسخون في العلم "  لو كان معطوفاً على قوله تعالى  " إِلاَّ الله "  لبقي قوله :  " يقولون آمنا به "  منقطعاً عنه ،   وإنه غير جائزٍ ؛ لأنه لا يقال : إنّه حال ،   لأنا نقول : فحينئذ يرجع إلى كُلّ ما تقدم ،   فيلزم أن يكون الله تعالى قائلاً : { يَقُولُونَ آمَنَّا بِهِ كُلٌّ مِّنْ عِندِ رَبِّنَا } وهذا كفر .  <br />
وثانيها : أن الراسخين في العلم لو كانوا عالمين بتأويله لما كان لتخصيصهم بالإيمان به وجه ،   فإنّهم لما عرفوه بالدّلالة لم يكن الإيمان به إلاّ كالإيمان بالمُحْكَم ،   فلا يكون في الإيمان به مزيد مَدْحٍ .  <br />
وثالثها : أن تأويلها{[439]} كان مما يجب أن يعلم لما كان طلب ذلك التأويل ذمًّا ،   لكن قد جعله ذمًّا حيث قال : { فَأَمَّا الَّذِينَ في قُلُوبِهِمْ زَيْغٌ فَيَتَّبِعُونَ مَا تَشَابَهَ مِنْهُ ابْتِغَاءَ الْفِتْنَةِ وَابْتِغَاءَ تَأْوِيلِهِ } [ آل عمران : 7 ] وأما الخبر فروي أنه - عليه الصَّلاة والسَّلام - قال :  " إنَّ من العلم كهيئة المَكْنُون لا يعلمه إلاّ العُلَماء بالله تعالى فإذا نطقوا به أنكره أهل الغرّة بالله " {[440]} .  <br />
ولأن القول بأن هذه الفَوَاتح غير معلومة مروي عن أكابر الصَّحابة رضي الله عنهم فوجب أن يكون حقًّا ،   لقوله عليه الصلاة والسلام :  " أَصْحَابي كالنُّجُوم بأيّهم اقْتَدَيْتُمُ اهتديتم " {[441]} .  <br />
وأما المعقول فهو أنَّ الأفعال التي كلّفنا الله تعالى بها قسمان : منها ما يعرف وَجْه الحكمة فيه على الجُمْلة بعقولنا كالصَّلاة والزكاة والصَّوم ،   فإنَّ الصَّلاة تضرُّع محضٌ ،   وتواضعُ للخالق ،   والزكاة سَعْي في دَفْعِ حاجة الفقير ،   والصّوم سعي في كسر الشهوة .  <br />
ومنها ما لا يعرف وَجْهُ الحكمة فيها على الجملة بعقولنا كأفعال الحَجّ في رَمْيِ الجَمَرَات ،   والسَّعي بين الصفا والمروة ،   والرَّمل ،   والاضْطِبَاع .  <br />
ثم اتفق المحققون على أنه كما يحسن من الله - تعالى - أن يأمر عباده بالنوع الأول ،   فكذا يحسن الأمر بالنوع الثاني ؛ لأنّ الطَّاعة في النوع الأول ،   تدلُّ على كمال الانقياد والتسليم ،   لاحتمال أن المأمور إنما أتى به لما عرف بعقله من وَجْهِ المصلحة فيه .  <br />
أما الطاعة في النوع الثاني ،   فإِنها تدلّ على كمال الانقياد والتسليم ،   فإذا كان الأمر كذلك في الأفعال ،   فلم لا يجوز أيضاً أن يكون [ الأمر ] كذلك في الأقوال ؟ وهو أن يأمر الله - تعالى - تارةً أن نتكلم بما نقف على معناه ،   وتارةً بما لا نقف على معناه ،   ويكون المَقْصُود من ذلك ظهور الانقياد والتَّسليم .  <br />
القول الثَّاني : قول من زعم أنَّ هذه الفَوَاتح معلومةٌ ،   واختلفوا فيه ،   وذكروا وجوهاً :<br />
الأوّل : أنها أسماء السّور ،   وهو قول أكثر المتكلّمين ،   واختيار الخليل وسيبويه رحمهما الله تعالى .  <br />
قال القَفّال - رحمه الله تعالى - وقد سّمت العرب هذه الحروف أشياء فسموا ب  " لام "  : والد حارثة بن لام الطَّائي ،   وكقولهم للنَّخاس :  " صاد "  ،   وللنقد :  " عين "  ،   وللسحاب :  " غين "  .  <br />
وقالوا : جبل  " قاف "  ،   وسموا الحوت :  " نوناً "  .  <br />
الثاني : أنها أسماء الله تَعَالى ،   روي عن علي - رضي الله تَعَالى عنه - أنه كان يقول : يا حم عسق .  <br />
الثالث : أنها أبعاض أسماء الله تَعَالى .  <br />
قال سعيد بن جبير رحمه الله : قوله :  " الر ،   حم ،   ونون "  مجموعها هو اسم الرحمان ،   ولكنا لا نقدر على كيفية تركيبها في البَوَاقي .  <br />
الرابع : أنها أسماء القرآن ،   وهو قول الكَلْبِيّ{[442]} - رحمه الله تعالى - والسّدي وقتادة رضي الله تعالى عنهم .  <br />
الخامس : أن كلّ واحد منها دالّ على اسم من أسماء الله - تعالى - وصفة من صفاته .  <br />
قال ابن عباس - رضي الله تعالى عنهما - في  " الم "  :  " الألف إشارة إلى أنه [ أحد ،   أول ،   آخر ،   أزلي ،    " واللام "  إشارة إلى أنه لطيف ،    " والميم "  إشارة إلى أنه ]{[443]} مَلِك مَجِيد مَنَّان "  .  <br />
وقال في  " كهيعص "  : إنه ثناء من الله - تعالى - على نفسه ،    " والكاف "  يدل على كونه كافياً ،    " والهاء "  على كونه هادياً ،    " والعين "  على العالم ،    " والصاد "  على الصادق .   وذكر ابن جريرٍ عن ابن عباس - رضي الله تعالى عنهما - أنه حمل  " الكاف "  على الكبير والكريم ،    " والياء "  على أن الله يجير ،    " والعين "  على أن الله العزيز والعدل .  <br />
والفرق بين هذين الوجهين أنه في الأول خصّص كل واحد من هذه الحروف باسم معين ،   وفي الثاني ليس كذلك .  <br />
السادس : بعضها يدلّ على أسماء الذات ،   وبعضها على أسماء الصّفات .  <br />
قال ابن عباس - رضي الله عنهما - في  " الم "  أنا الله أعلم ،   وفي  " المص "  أنا الله أفصل ،   وفي  " الر "  أنا الله أرى ،   وهذه رواية أبي صالح ،   وسعيد بن جبير عنه .  <br />
قال الزَّجَّاج :  " وهذا أحسن ،   فإن العرب تذكر حرفاً من كلمة تريدها كقولهم : [ مشطور السريع ]<br />
قُلْنَا لَهَا : قِفِي لَنَا قََالَتْ : قَافْ *** .   .  …………… .   .  {[444]}<br />
وأنشد سيبويه لغيلان : [ الرجز ]<br />
نَادَوْهُمْ أَنَ الْجِمُوا ،   أَلاَ تَا *** قَالُوا جَمِيعاً كُلُّهُمْ أَلاَ فَا{[445]}<br />
أي : لا تَرْكَبُوا ،   قالوا : بَلَى فَارْكَبُوا .  <br />
وأنشد قُطْرب : [ الرجز ]<br />
جَارِيَةٌ قَدْ وَعَدَتْنِي أَنْ تَا *** تَدْهُنَ رَأْسي وَتُفَلِّيني وَتَا{[446]}<br />
السّابع : كلّ واحد منها يدلّ على صفات الأفعال ،   ف  " الألف "  آلاَؤه ،   و " اللاّم "  لُطْفه ،   و " الميم "  مَجْدُه ،   قاله محمد بن كَعْبٍ القُرَظي .  <br />
الثَّامن : بعضها يدلّ على أسماء الله - تعالى - وبعضها يدلّ على أسماء غير الله تعالى .  <br />
قال الضَّحاك :  " الألف "  من الله ،   و " اللام "  من جبريل ،   و " الميم "  من محمد عليه الصَّلاة والسلام [ أَي أنزل الله الكتاب على لسان جبريل عليه الصَّلاة والسَّلام ]{[447]} .  <br />
التاسع : ما قاله المبرّد : واختاره جمعٌ عظيم من المحقّقين - أنّ الله - تعالى - إنَّمَا ذكرها احتجاجاً على الكُفّار ،   وذلك أن الرَّسول - عليه الصَّلاة والسّلام - لما تحدّاهم أن يأتوا بِمِثْلِ القرآن ،   أو بِعَشْرِ سُوَرٍ ،   أو بسورة ،   فعجزوا عنه أنزلت هذه الأحرف تنبيهاً على أن القرآن ليس إلاّ من هذه الأحرف ،   وأنتم قادرون عليها ،   وعارفون بقوانين الفَصَاحة ،   فكان يجب أن تأتوا بِمِثْلِ هذا القرآن ،   فلما عجزتم عنه دلّ ذلك على أنه من عِنْدِ الله لا من البَشَرِ .  <br />
العاشر : قول أبي روق{[448]} وقُطْرب : إن الكُفَّار لما قالوا :<br />
{ لاَ تَسْمَعُواْ لِهَذَا الْقُرْآنِ وَالْغَوْاْ فِيهِ } [ فصلت : 26 ] وتواصوا بالإعراض عنه أراد الله - تعالى - لما أحب صلاحهم ونفعهم أن يُورِدَ عليهم ما لا يعرفونه ،   ليكون ذلك سبباً لإسْكَاتهم ،   واستماعهم لما يرد عليهم من القرآن ،   فأنزل الله - تعالى - عليهم هذه الأحرف ،   فكانوا إذا سمعوها قالوا كالمتعجبين : اسمعوا إلى ما يجيء به محمد عليه الصلاة والسلام ،   فإذا أصغوا هجم عليهم القُرآن فكان ذلك سبباً لاستماعهم ،   وطريقاً إلى انتفاعهم ،   فكان كالتنبيه لما يأتي بعده من الكلام كقوله الأول .  <br />
الحادي عشر : قول أبي العَالِيَةِ  " إنّ كل حرف منها في مُدّة أعوام وآجال آخرين "  .  <br />
قال ابن عباس رضي الله تعالى عنهما :  " سُرّ أبو ياسِر بن أَخْطَب برسول الله - صلى الله عليه وسلم - وهو يَتْلُو سورة البقرة { الم ذَلِكَ الكِتَابُ } ،   ثم أتى أخوه حُيَيُّ بن أَخْطَب{[449]} ،   وكَعْب بن الأَشْرَف{[450]} ،   وسألوه عن  " الم "  وقالوا : ننشدك الله الذي لا إله إلا هو أحقّ أنها أَتَتْكَ من السماء ؟ فقال النبي صلى الله عليه وسلم :  " نعم كذلك نزلت "  ،   فقال حُيَيّ : إن كنت صادقاً إني لأعلم أَجَلَ هذه الأمّة من السنين ،   ثم قال : كيف ندخل في دين رجل دلّت هذه الحروف بحساب الجمل على أن منتهى أَجَل مُدّته إحدى وسبعون سنةً ،   فضحك رسول الله - صلى الله عليه وسلم - فقال حُيَيّ : فهل غير هذا ؟ قال :  " نعم المص "  فقال حُيَيّ : هذا أكثر من الأولى هذه مائة وإحدى وثلاثون سنة ،   فهل غير هذا ؟ قال :  " نعم الر "  قال حُيَيّ : هذه أكثر من الأولى والثانية ،   فنحن نشهد إن كنت صادقاً ما ملكت أمتك إلا مائتين وإحدى وثلاثين سنةً ،   فهل غير هذا ؟ قال :  " نعم "  قال :  " المر "  قال : فنحن نشهد أنا من الذين لا يؤمنون ،   ولا ندري بأي أقوالك نأخذ .  <br />
فقال أبو ياسر : أما أنا فأشهد على أن أنبياءنا قد أخبرونا عن مُلْكِ هذه الأمّة ،   ولم يبينوا أنها كم تكون ،   فإن كان محمد صادقاً فيما يقول ،   إني لأراه يستجمع له هذا كله فقام اليهود ،   وقالوا اشتبه علينا أمرك ،   فلا ندري أَبِالْقَلِيْلِ نأخذ أم بالكثير ؟ فذلك قوله تعالى : { هُوَ الَّذِي? أَنزَلَ عَلَيْكَ الْكِتَابَ }{[451]} [ آل عمران : 7 ] الآية الكريمة .  <br />
ورُوي عن ابن عَبّاس - رضي الله تعالى عنهما - أنها أقسام .  <br />
وقال الأخفش : أقسم الله - تعالى - لشرفها وفضلها ؛ لأنها مبادئ كتبه المنزلة ،   ومباني أسمائه الحسنى .  <br />
وقيل فيها غير ذلك .  <br />
واعلم أن الله - تعالى - أورد في هذه الفَوَاتح نصف عدد أَسامي حروف المُعْجَم ،   وهي أربعة عشر : الألف ،   واللام ،   والميم ،   والصاد ،   والراء ،   والكاف ،   والهاء ،   والعين ،   والطاء ،   والسين ،   والحاء ،   والقاف ،   والباء ،   والنون في تسع وعشرين سورة .  <br />
وجاءت أيضاً مختلفة الأعداد ،   فوردت  " ص ق ن "  على حرف .  <br />
و "  طه وطس ويس وحم "  على حرفين .  <br />
و "  الم والر وطسم "  على ثلاثة أحرف .  <br />
و "  المص والمر "  على أربعة أحرف .  <br />
و "  كهيعص وحم عسق "  على خمسة أحرف ،   والسبب فيه أن أبنية كلامهم على حرف وحرفين إلى خمسة أحرف ،   فكذا هاهنا .
