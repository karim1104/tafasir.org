قوله تعالى : { لِّلَّذِينَ يُؤْلُونَ مِن نِّسَائِهِمْ تَرَبُّصُ } : هذه جملةٌ من مبتدأ وخبرٍ ،   وعلى رأي الأخفش من باب الفعلِ والفاعلِ ؛ لأنه لا يشترطُ الاعتماد ،   و " مِن نِّسَائِهِمْ "  في هذا الجارِّ ثمانيةُ أوجهٍ :<br />
أحدها : أنْ يتعَلَّقَ ب  " يُؤْلُونَ "  .  <br />
قال الزمخشريُّ :  " فإنْ قلتَ : كيف عُدِّي ب  " مِنْ "  وهو مُعَدى ب  " عَلَى "  ؟ قلتُ : قد ضُمِّنَ في هذا القَسَم المخصوصِ معنى البُعْد ،   فكأنه قيل : يَبْعُدُونَ من نسائِهم مُؤْلِينَ ،   أو مُقْسِمِين "  .  <br />
الثاني : أنَّ  " آلَى "  يتَعَدَّى ب  " عَلَى "  وب  " مِنْ "  ؛ قاله أبو البقاء{[3563]} نقلاً عن غيره ؛ أنه يقال : آلَى من امرأته ،   وعلى امرأته .  <br />
الثالث : أنَّ  " مِنْ "  قائمةٌ مقامَ  " عَلَى "  على رأي الكوفيِّين .  <br />
والرابع : أنها قائمةٌ مقامَ  " فِي "  ،   ويكونُ ثم مضافٌ محذوفٌ ،   أي : على تَرْكِ وَطْءِ نسائهم ،   أو في ترك وطء نسائهم .  <br />
والخامس : أنَّ  " مِنْ "  زائدةٌ ،   والتقدير : يُؤْلُون أَنْ يَعْتَزلوا نِسَاءَهُم .  <br />
والسادس : أَنْ تتعلَّقَ بمحذوف ،   والتقديرُ : والذين يُؤُلون لهم من نسائهم تربُّصُ أربعة أشهرٍ ؛ فتتعلَّق بما يتعلق به  " لَهُم "  المحذوفُ ،   هكذا قَدَّره أبو حيَّان وعزاه للزمخشريِّ قال شهاب الدين وفيه نظرٌ ؛ فإنَّ الزمخشريَّ قال : ويجوزُ أن يُرَادَ : لهم من نسائهم تَرَبُّصُ ؛ كقولك :  " لِي مِنْكَ كَذَا "  فقوله  " لَهُمْ "  لم يُرد به أن ثَمَّ شيئاً محذوفاً ،   وهو لفظُ  " لَهُمْ "  ،   إنما أرادَ أَنْ يعلِّق  " مِنْ "  بالاستقرار الذي تعلَّق به  " لِلَّذِينَ "  ،   غايةُ ما فيه : أنه أتى بضمير  " الَّذِينَ "  تبييناً للمعنى ،   وإِلَى هذا المنحَى نحا أبو البقاء{[3564]} ؛ فإنه قال : وقيل : الأصلُ  " عَلَى "  ،   ولا يَجُوزُ أن تقومَ  " مِنْ "  مقامَ  " عَلَى "  فَعَلَى ذلك تتعلَّقُ  " مِنْ "  بمعنى الاستقرار ،   يريدُ الاستقرارَ الذي تعلَّقَ به قوله  " لِلَّذِينَ "  ،   وعلى تقدير تسليم أنَّ لفظة  " لَهُمْ "  مقدرةٌ ،   وهي مُرادةٌ ،   فحينئذٍ : إنما تكونُ بدلاً من  " لِلَّذِينَ "  بإِعادةِ العاملِ ،   وإلاَّ يبقى قوله  " لِّلَّذِينَ يُؤْلُونَ "  مُفْلَتاً ،   وبالجملةِ فتعلُّقه بالاستقرار غيرُ ظاهرٍ ،   وأمَّا تقديرُ الشيخ :  " والذين يُؤْلُونَ لَهُمْ مِن نِّسَائِهِمْ تَرَبُّصُ "  ،   فليس كذلك ؛ لأنَّ  " الَّذِينَ "  لو جاء كذلك غيرَ مجرور باللام ،   سَهُلَ الأمرُ الذي ادَّعاه ،   ولكن إنما جاءَ كما تراه مجروراً باللام ،   سهُلَ الأمرُ الذي ادَّعاه ،   ولكن إنما جاءَ كما تراه مجروراً باللام ،   ثم قال أبو حيَّان : وهذا كلُّه ضعيفٌ يُنَزَّه القرآن عنه ،   وإنما يتعلَّق ب  " يُؤْلُونَ "  على أحد وجهين : إمَّا أنْ تكونَ  " مِنْ "  للسَّبَبِ ،   أي يَحْلِفُون بسبب نسائِهم ،   وإمَّا أَنْ يُضَمَّن معنى الامتناع ،   فيتعدَّى ب  " مِنْ "  فكأنه قيل  " لِلَّذِينَ يمتنعُونَ من نسائِهِم بالإِيلاَءِ "  فهذان وَجْهان مع الستة المتقدِّمة ؛ فتكونُ ثمانيةً ،   وإن اعتبرت مطلقَ التضمينِ فتجيءُ سبعةً .  <br />
والإِيلاءُ : الحَلفُ .   مصدرُ آلَى يُؤْلي ،   نحو : أَكْرَمَ يُكْرِم إِكْرَاماً ،   والأصل :  " إئْلاءٌ "  فأُبدِلت الهمزةُ الثانيةُ ياءً ؛ لسكونها وانكسار ما قبلها ؛ نحو :  " إِيمَان "  .  <br />
ويقال : تَأَلَّى وايتَلَى على افتَعَلَ ،   والأصلُ : ائْتَلَى ،   فقُلِبَت الثانيةُ ياءً ؛ لِما تقدَّم .  <br />
والحَلْفَةُ : يقال لها : الأَلِيَّةُ والألُوَّةُ والأَلْوَةُ والإِلْوَةُ ،   وتُجْمَعُ الأَليَّةُ على  " أَلايَا "  ؛ كعَشيَّة وعَشَايَا ،   ويجوزُ أن تُجْمَعَ الأَلُوَّة أيضاً على  " أَلاَيَا "  ؛ كرَكُوبَة ورَكَائِب ؛ قال كُثَيِّر عزَّة : [ الطويل ]<br />
قَلِيلُ الألاَيَا حَافِظٌ ليَمِينِهِ *** إِذَا صَدَرَتْ مِنْهُ الأَلِيَّةُ بَرَّتِ{[3565]}<br />
وقد تقدَّم كيف تصريفُ أَلِيَّة وَأَلاَيَا عند قوله : { نَّغْفِرْ لَكُمْ خَطَايَاكُمْ } [ البقرة :58 ] جمع خطيئة والإيلاء من عرف الشَّرع : هو اليمين على ترك الوطء ؛ كقوله : لا أُجامعك ،   أو لا أُباضعك ،   أو لا أُقاربك{[3566]} .  <br />
ومن المفسرين من قال في الآية حذف تقديره : للَّذين يؤلون من نسائهم ألاَّ يطئوهم ،   إلاَّ أنَّه حذف لدلالة الباقي عليه .  <br />
قال ابن الخطيب{[3567]} : هذا إذا حملنا لفظ  " الإِيلاَءِ "  على المفهوم اللُّغَوِيّ ،   أَمَّا إذا حملناه على المفهوم الشَّرعي ،   لم يحتج إلى هذا الإضمار .  <br />
وقرأ أُبيٌّ{[3568]} وابن عباس :  " للَّذِينَ يُقْسِمُونَ "  ،   نقله القرطبي .  <br />
وقرأ{[3569]} عبد الله :  " آلَوْا مِنْ نِسَائِهِم "  .  <br />
والتَّربصُ : الانتظارُ ،   وهو مقلوبُ التَّصبُّرِ ؛ قال : [ الطويل ]<br />
تَرَبَّصْ بِهَا رَيْبَ المَنُونِ لَعَلَّهَا *** تُطَلَّقُ يَوْماً أَوْ يَمُوتَ حَلِيلُهَا{[3570]}<br />
وإضافةُ التربُّص إلى الأشهرِ فيها قولان :<br />
أحدهما : أنَّه من باب إضافة المصدر لمفعوله ؛ على الاتساع في الظرف ؛ حتَّى صارَ مفعولاً به ،   فأُضيفَ إليه ،   والحالةُ هذه كقوله :  " بَيْنَهُمَا مَسِيرة يَوْمٍ "  أي : مَسِيرة في يَوْم .  <br />
والثاني : أنه أُضِيف الحَدَثُ إلى الظرف من غير اتِّساع ،   فتكونُ الإِضافةُ بمعنى  " فِي "  وهو مذهبٌ كوفيٌّ ،   والفاعلُ محذوفٌ ،   تقديره : تربُّصُهُمْ أربعةُ أشهرٍ .  <br />
فصل<br />
قال قتادة : كان الإيلاء طلاقاً لأهل الجاهلية وقال سعيد بن المُسيَّب{[3571]} : كان ذلك من ضرار أهل الجاهليَّة{[3572]} ،   وكان الرَّجُلُ لا يُحِبُّ امرأته ،   ولا يريد أن يتزوَّجها غيره ،   فيحلف ألاَّ يقربها أبداً ،   فيتركها لا أَيماً ولا ذات بعلٍ ،   وكانوا في ابتداء الإِسلام يفعلون ذلك أيضاً ؛ فأزال الله تعالى ذلك ،   وضرب للزَّوْج مُدَّة يَتَرَوَّى فيها ويتأمَّل ،   فإِنْ رأى المصلحة في تركِ هذه المضارَّة ،   فعلها ،   وإن رأى المصلحة في المفارقة ،   فارقها{[3573]} .  <br />
وقال القُرطبيّ{[3574]} : وقد آلَى النَّبِيُّ - صلى الله عليه وسلم - وطَلَّق ،   وسببُ إيلائه : سُؤال نسائه إِيَّاه من النَّفقةِ ما ليس عندهُ ؛ كذا في  " صَحِيح مُسْلِم " {[3575]} ،   وقيل : لأن زينب ردَّت عليه هديَّتهُ ،   فغضب - صلى الله عليه وسلم - فآلَى منهُنَّ ،   ذكره ابن ماجه{[3576]} .  <br />
فصل<br />
واختلف أهل العلم فيه :<br />
فذهب أكثرهم إلى أَنَّهُ إِن حلف لا يَقْرَبُ زوجته أبداً ،   أَو سَمَّى مُدَّة أكثر من أربعة أشهر يكون مؤْلياً ،   فلا يتعرَّض لها قبل مضي أربعة أشهر ،   وبعد مُضيِّها يُوقفُ ويُؤمرُ بالفيئة أو الطَّلاقِ بعد مُطالبةِ المرأة ،   والفَيْئَة : هي الرُّجُوع عمَّا قال بالوطء إن قَدَر عليه ،   وإن لم يَقْدِر فبالقَوْل ،   فإن لم يَفِ ولم يُطَلِّق ،   طَلَّق عليه الحاكم واحدة .  <br />
وذهب إلى الوَقْفِ بعد مُضِيِّ المُدَّةِ : عمر ،   وعثمان ،   وعليٌّ ،   وأبو الدَّرداء ،   وابن عمر{[3577]} .  <br />
قال سُليمان بن يسار : أدْرَكْتُ بضعة عشر من أصحاب النبي - صلى الله عليه وسلم - كُلُّهم يقول بوقف المُؤْلي{[3578]} ،   وإليه ذهب سعيد بن جُبير ،   وسُلَيْمان بن يسار ،   ومجاهد ،   وبه قال مالك والشَّافعي وأحمد وإِسحاق .  <br />
وقال بعض أهل العلم : إذا مضتْ عليها أربعةُ أشهر ،   يقع عليها طَلْقَةً بائنة ،   وهو قول ابن عبَّاس وابن مسعود ،   وبه قال سُفيانُ الثَّورِيُّ وأصحاب الرَّأي{[3579]} .  <br />
وقال سعيد بن المُسيَّب والزُّهري : تقع طلقة رجعيَّةً{[3580]} ،   ولو حلف ألاَّ يطأَها أَقَلَّ من أربعة أشهر ،   لا يكون مُؤْلياً ،   بل هو حانثٌ إذا وطئها قبل مُضِيِّ تلك المُدَّة ،   وتجب عليه كفَّارة يمين على الصَّحيح ،   ولو حلف ألاَّ يطَأَهَا أربعة أشهرٍ ،   لا يكون مُؤلياً عند من يقول بالوقف بعد مُضِيِّ المُدَّة ،   لأن بقاء المُدَّة شرط للوقف ،   وثبوت المُطَالبة بالفيئة أو الطَّلاق ،   وقد مضتِ المُدَّة ،   وعند من لا يقول بالوقف لا يكون مؤلياً ويقعُ الطَّلاَق بمُضِيِّ المُدَّة .  <br />
وقال ابن عبَّاس لا يكون مُؤْلياً حتى يحلف لا يَطأَها أبداً{[3581]} .  <br />
وقال الحسن البصريِّ وإسحاق : أي مُدَّة حَلَف عَلَيها ،   كان مُؤلياً وإن كان يَوْماً{[3582]} .  <br />
فصل<br />
قال القرطبيُّ{[3583]} : إذا حلف أَلاَّ يطأ امرأتهُ أكثر من أربعة أشهرٍ ؛ فانقضت الأربعة أشهرٍ ولم تُطالبه ،   ولم ترفعهُ إلى الحاكم ،   لم يلزمه شيء عند مالكٍ وأكثر أهل المدينة .  <br />
وقال بعض أصحابنا : يلزمه بانقضاء الأَربعَة أَشهر طلقة رجعيَّة ،   وقال بعضهم : طلقة بائِنَة .  <br />
والصَّحِيح : ما ذهب إليه مالِكٌ ،   وأكثر أَهْلِ المدينة وأصحابه .  <br />
فصل هل ينعقد الإيلاء في الغضب ؟ !<br />
قال ابن عبَّاس : لا يكون إيلاءً إلاَّ في حالِ الغَضَبِ{[3584]} ،   وهو المشهور عن عليٍّ{[3585]} - رضي الله عنه - وهو قول اللَّيث ،   والشَّعبي ،   والحسن ،   وعطاء قالوا : لا يكون الإيلاءُ إلاَّ على وجه مغاضبةٍ ومشادَّة{[3586]} .  <br />
وقال ابن سيرين : يكون في غضبٍ ،   وغير غضبٍ{[3587]} ،   وهو قول ابن مسعود ،   والثوري ومالك وأهل العراق والشَّافعيِّ وأحمد .  <br />
فصل<br />
والمدخول بها وغير المدخول بها سواءٌ في صِحَّة الإيلاء منها .  <br />
قال القرطبي{[3588]} : والذِّمِّيُّ لا يصحُّ إيلاؤه كما لا يصحُّ طلاقهُ ،   ولا ظِهَارُهُ ؛ لأن نكاح أهل الشِّرْكِ عندنا ليس بنكاحٍ صحيحٍ .  <br />
فصل<br />
ومُدَّة الإيلاء أربعة أشهر في حق الحُرِّ والعبدِ لأنَّها ضُرِبت لمعنى يرجع إلى الطَّبع ،   وهو قِلَّةُ صبر المرأَة عن الزَّوج ،   فيستوي فيه الحُرُّ والعبدُ ؛ كمُدَّة الفَيْئَةِ ومُدَّة الرِّضاع .  <br />
وعند مالكٍ وأبي حنيفة : ينتصف بالرِّقِّ ،   إلاَّ عند أبي حنيفة : ينتصف بِرِقِّ المرأة ،   وعند مالكٍ : برِقِّ الرَّجُل ؛ كقولهما في الطَّلاَق .  <br />
ولنا : ظاهر قوله تعالى : { لِّلَّذِينَ يُؤْلُونَ مِن نِّسَائِهِمْ } فتتناول الكُلَّ من غير تخصيصٍ ،   والتَّخصيص خلاف الظَّاهر ،   والمعنى المُتَقَدِّم يمنع التَّخْصِيص .  <br />
قال القرطبيُّ{[3589]} : وأَجَلُ المُؤلي من يوم حلف ،   لا مِنْ يوم المرافعة إلى الحاكم .  <br />
فصل فيمن يصح منه الإيلاء ومن لا يصح<br />
كُلُّ زَوْج يتصوَّر منه الوِقَاع ،   وكان تصرُّفه مُعْتبراً في الشَّرْع ،   صَحَّ منه الإيلاءُ .  <br />
وقال مالكٌ : لا يَصِحُّ الإيلاءُ إلاَّ في حال الغضب ،   وقال غيره : يصحُّ الإيلاءُ في حال الرِّضَى والغضب ،   ويصحُّ الإيلاءُ من الرَّجعيَّة ؛ لأنها زوجة ؛ بدليل أَنَّه لو قال : نسائي طوالِقٌ ،   وقع عليها الطَّلاَقُ .  <br />
وإيلاء الخَصِيِّ صحيحٌ ،   لأنه يُجَامع كالفحلِ ،   وإنما فُقِد في حَقِّهِ الإِنْزَالُ ،   والمجبوب إن بقي منه ما يمكنه أن يُجَامع به ،   صَحَّ إِيلاؤه ،   وإن لم يَبْقَ ،   فيه خلافٌ .  <br />
قال أبو حنيفة : لا يَصِحُّ إيلاؤه .  <br />
وقال غيره : يَصِحُّ لعموم الآية ،   ولا يصحُّ الإيلاء [ من أَجْنَبيَّة ] ،   فلو آلَى منها ثم تَزَوَّجها ،   لم يكن مُؤْلِياً .  <br />
فصل<br />
فإن امتنع من وطئها بغير يمين إضراراً بها ،   أُمر بوطئها ،   فإن امتنع إضراراً بها ،   فرَّق الحاكم بينه وبينها ،   من غير ضرب مُدَّة .  <br />
وقيل : يُضربُ له أجلُ الإيلاءِ ،   فإذا حلف لا يطؤها حتى يفطم ولدها ؛ لئلا يُمْغَلَ بولدها ،   ولم يرد إضراراً بها حتى يَنْقَضِي أَمَدُ الرِّضاع ،   لم يكُن لها مُطَالبته .  <br />
فصل<br />
المُوْلِي لا يخلُو إمّا أن يحلف تَرْك الوَطْءِ بالله  تعالى  أو بغيره ،   فإِن حَلَف بالله تعالى كان مُؤلياً ،   ثم إن جامعها في مُدَّة الإيلاءِ ،   خَرَجَ عن الإيلاءِ ،   وهل تجب عليه كَفَّارة ؟ فيه قَوْلاَن :<br />
أصحّهما : أن عليه الكَفَّارة كما قَدَّمناه ؛ لعموم الدَّلائِل الموجبةِ للكَفَّارةِ عند الحِنْثِ باليمين بالله - تعالى - لأنه لا فَرْقَ بين قوله :  " والله لا أَقْرَبُكِ "  ثم يَقْربُهَا ،   وبين قَوله :  " والله لا أُكَلِّمُكِ "  ثم يُكَلِّمُهَا .  <br />
والقول الثاني : لا كَفَّارة عليه ؛ لقول تعالى : { فَإِنْ فَاءُوا فَإِنَّ اللهَ غَفُورٌ رَّحِيمٌ } .  <br />
ولو كانت الكَفَّارة واجبةٌ لذكرها ،   والحاجة مناسبةٌ إلى معرفتها ،   ولا يجوزُ تأخِير البيان عن وقتِ الحاجة ،   والغُفران يوجبُ ترك المُؤَاخَذةِ .  <br />
وللأَوَّلِين أن يجيبوا : بأنه إنما ترك الكَفَّارة هاهنا لأنه  تعالى - بيَّنها على لسان رسوله - عليه الصَّلاة والسلام  في سَائِر المواضع ،   وتَرْكُ المُؤَاخذة بقوله :  " غَفُورٌ رَّحِيمٌ "  يدل على عدم العقاب ،   وهو لا يُنافي وجُوب الفِعْل ،   كما أن التَّائِب عن الزِّنا والقَتْلِ لا عِقاب عليه ،   ومع ذلك يجِبُ عليه الحَدُّ والقِصَاصُ .  <br />
وأما إن كان الحالف في الإيلاء بغير اللهِ ؛ كما إذا قال : إن وَطَأْتُكِ فعبدي حُرٌّ ،   أَوْ أَنْت طالِقٌ ،   أو ضرتك طَالِق ،   أو التزم أمراً في الذِّمَّة ،   فقال : إن وَطَأْتُكِ فَلِلَّه عليَّ عِتْق رَقبة ،   أو صدقة ،   أو صَوْم ،   أو حَجّ ،   أو صلاة ،   فقال الشَّافعي في  " القدِيمِ "  ،   وَأَحمدُ في ظاهر الرِّواية عنه : لا يكون مُؤْلِياً ؛ لأن الإِيلاَء المعهُود هو الحلفُ باللهِ ؛ ولقوله - عليه الصلاة والسلام - :<br />
مَنْ حَلَفَ فَلْيَحْلِفْ بِاللهِ " {[3590]} .  <br />
وقال الشَّافعيّ في  " الجَدِيدِ "  - وهو قول أبي حنيفة ومالك وجماعة - إنه يكُون مُؤْلِياً ؛ لأن لفظ الإيلاءِ يَتَناوَلُ الكُلَّ ،   وعلى القولين فيمينُهُ منعقدة ،   فإن كان قد عَلَّق به طلاقاً أو عِتْقاً ،   وقع بِوَطْئِه ذلك المُعَلَّق ،   وإن كان المُعَلَّق التزام قُرْبة في الذِّمَّة ،   فعليه ما في نذرِ اللَّجاج{[3591]} ،   إِمَّا كَفَّارة يمين ،   وإِمَّا الوَفَاء بما سَمَّى .  <br />
وفائدة هذين القولين : أنا إن قلنا : يكون مُؤْلياً ،   فبعد أَرْبَعَة أشهر يضيق الأَمْر عليه ،   حَتَّى يَفيء أو يُعَلِّق ،   وَإِنْ قلنا : لا يكون مُؤْلياً ،   لا يضيقُ عليه الأَمر .  <br />
قال القرطبي{[3592]} : فإن حلف بالنَّبِيّ أو الملائِكَة ،   أو الكَعبة ألا يَطَأَها أو قال : هو يَهُوديٌّ ،   أو نصرانيٌّ ،   أو زانٍ إن وطِئَها ،   فليس بِمُؤْلٍ ،   قال : وإن حلف ألا يَطَأَها ،   واستثنى فقال : إن شاء الله ،   فإنَّه يكون مُؤلياً ،   فإن وَطِئَهَا ،   فلا كَفَّارة عليه .  <br />
وقال ابن الماجشون في  " المَبْسُوطِ "  : ليس بِمؤلٍ ،   وهو الصَّحيحُ .  <br />
قوله : { فَإِنْ فَاءُوا } ألفُ  " فَاءَ "  منقلبةٌ عن ياء ؛ لقولهم : فَاءَ يَفِيءُ فَيْئَةً : رَجَعَ والفَيْءُ : الظلُّ ؛ لرجوعه من بعد الزوال ،   وقال عَلْقَمَةُ : [ الطويل ]<br />
فَقُلْتُ لَهَا فِيئِي فَمَا تَستَفِزُّنِي *** ذَوَاتُ العُيُون وَالبَنَانِ المُخَضَّبِ{[3593]}<br />
وفرّقوا بين الفَيْءِ والظِّل : فقالوا : الفَيء ما كان بالعَشِيِّ ؛ لأنه الذي نسختهُ الشَّمسُ ،   والظَّلُّ ما كان بالغداة ؛ لأنه لم تنسخه الشَّمس ،   وفي الجنة ظِلٌّ وليس فيها فَيْءٌ ؛ لأنه لا شَمْسَ فيها ؛ قال تعالى : { وَظِلٍّ مَّمْدُودٍ } [ الواقعة :30 ]  ؛ وأنشد : [ الطويل ]<br />
فَلاَ الظِّلُّ مِنْ بَرْدِ الضُّحَى تَسْتَطِيعُهُ *** وَلاَ الفَيْءُ مِنْ بَرْدِ العَشِيِّ تَذُوقُ{[3594]}<br />
وقيل : فُلاَن سريعُ الفيئة ،   أي : سريعُ الرُّجوع عن الغَضَبِ إلى الحَالَةِ المتقدِّمة ،   حكاه الفَرَّاء عن العرب{[3595]} .  <br />
وقيل لما رَدَّه الله على المُسلمين من مال المُشْرِكين : فَيْء ؛ كأنه كان لهم فرجع إِلَيهم ،   فقوله :  " فَاءُوا "  معناه : رَجَعُوا عَمَّا حَلَفُوا عليه من تَرْك الجِمَاع ،   فإنه غَفُورٌ رَحِيمٌ .  <br />
فصل في وجوب الكفَّارة<br />
قال القُرطبيّ{[3596]} : جمهور العُلماء أَوْجَبُوا الكَفَّارة على المُؤْلي إذا فاء بجماع امرأته .  <br />
وقال الحسن : لا كَفَّارة عليه{[3597]} .  <br />
قال القرطبي{[3598]} : إِذَا كَفَّر عن يمينه ،   سقط عنه الإيلاءَ .
