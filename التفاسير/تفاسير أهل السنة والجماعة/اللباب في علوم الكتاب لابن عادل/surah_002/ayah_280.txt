قوله تعالى : { وَإِن كَانَ ذُو عُسْرَةٍ } : في  "  كان  "  هذه وجهان :<br />
أحدهما : - وهو الأظهر - أنها تامةٌ بمعنى حدث ،   ووجد ،   أي : وإن حدث ذو عسرةٍ ،   فتكتفي بفاعلها كسائر الأفعال ،   قيل : وأكثر ما تكون كذلك إذا كان مرفوعها نكرةٌ ،   نحو :  "  قد كان مِنْ مَطَرٍ  "  .  <br />
والثاني : أنها الناقصة والخبر محذوفٌ .   قال أبو البقاء{[4752]} :  "  تقديره : وإن كان ذو عسرة لكم عليه حقٌّ ،   أو نحو ذلك  "  وهذا مذهب بعض الكوفيين في الآية ،   وقدَّر الخبر : وإن كان من غرمائكم ذو عسرةٍ .   وقدَّره بعضهم : وإن كان ذو عسرةٍ غريماً .  <br />
قال أبو حيَّان{[4753]} :  "  وَحَذْفُ خبرِ كَانَ لا يجيزه أصحابنا ؛ لا اختصاراً ؛ ولا اقتصاراً ،   لعلَّةٍ ذكروها في كتبهم .   وهي أنَّ الخبر تأكّد طلبه من وجهين :<br />
أحدهما : كونه خبراً عن مخبر عنه .  <br />
والثاني : كونه معمولاً للفعل قبله ،   فلما تأكدت مطلوبيته ،   امتنع حذفه .  <br />
فإن قيل : أليس أن البصريين لمَّا استدلَّ عليهم الكوفيون في أنَّ  "  ليس  "  تكون عاطفةً بقوله : [ الرمل ]<br />
 .   .   .   *** إِنَّمَا يَجْزِي الفَتَى لَيْسَ الجَمَلْ{[4754]}<br />
تأوَّلُوهَا على حَذْفِ الخَبَرِ ؛ وأَنْشدوا شَاهِداً على حَذْفِ الخبرِ قولَه : [ الكامل ]<br />
 .   .   .   *** يَبْغِي جِوَارَكِ حِينَ لَيْسَ مُجِيرُ{[4755]}<br />
وإذا ثبت هذا ،   ثبت في سائر الباب .  <br />
فالجواب أن هذا مختصٌّ بليس ؛ لأنها تشبه لا النافية ،   و "  لا  "  يجوز حذف خبرها ،   فكذا ما أشبهها  "  .  <br />
وتقوَّى الكوفيُّون بقراءة عبد الله ،   وأُبيّ{[4756]} ؛ وعثمان :  "  وَإِنْ كَانَ ذَا عُسْرَةٍ  "  أي : وإن كان الغريم ذا عُسْرَةٍ .   قال أبو عليّ : في  "  كان  "  اسمها ضميراً تقديره : هو ،   أي : الغريم ،   يدلُّ على إضماره ما تقدَّم من الكلام ؛ لأنَّ المرابي لا بدَّ له ممَّن يرابيه .  <br />
وقرأ{[4757]} الأعمش :  "  وإِنْ كان مُعْسِراً  "  قال الدَّاني ،   عن أحمد بن موسى :  "  إنها في مُصْحَفِ عبد الله كذلك  "  .  <br />
ولكنَّ الجمهور على ترجيح قراءة العامة وتخريجهم القراءة المشهورة .   قال مكي{[4758]} : وَإِنْ وقع ذُو عُسْرَةٍ ،   وهو سائغٌ في كلِّ الناس ،   ولو نصبت  "  ذا  "  على خبر  "  كان  "  ،   لصار مخصوصاً في ناس بأعيانهم ؛ فلهذه العلة أجمع القرَّاء المشهورون على رفع  "  ذو  "  .  <br />
وقد أوضح الواحديُّ هذا ،   فقال :  "  أي : وإنْ وقع ذو عسرةٍ ،   والمعنى على هذا يصحُّ ،   وذلك أنه لو نصب ،   فقيل : وإن كان ذا عسرة ،   لكان المعنى : وإن كان المشتري ذا عُسْرةٍ ،   فنظرةٌ ؛ فتكون النظرة مقصورةً عليه ،   وليس الأمر كذلك ؛ لأن المشتري ،   وغيره إذا كان ذا عسرةٍ ،   فله النظرة إلى الميسرة  "  .  <br />
وقال أبو حيَّان{[4759]} : مَنْ نصب  "  ذَا عُسْرَةٍ  "  ،   أو قرأ  "  مُعْسِراً  "  فقيل : يختصُّ بأهل الرِّبا ،   ومن رفع ،   فهو عامٌّ في جميع من عليه دينٌ ،   قال :  "  وليس بلازمٍ ،   لأنَّ الآية إنما سِيقَتْ في أهل الربا ،   وفيهم نزلت  "  قال شهاب الدين : وهذا الجواب لا يجدي ؛ لأنه وإن كان السياق كذا ،   فالحكم ليس خاصاً بهم .  <br />
وقرئ {[4760]} "  وَمَنْ كَانَ ذَا عُسْرَةٍ  "  ،   وقرأ{[4761]} أبو جعفرٍ  "  عُسُرَةٍ  "  بضم السين .  <br />
فصل<br />
قال ابن الخطيب{[4762]} : لما كنتُ ب  "  خَوَارِزْم  "  ،   وكان هناك جمعٌ من أكابر الأدباء ،   فأوردت عليهم إشكالاً في هذا الباب ؛ فقلت : إنكم تقولون : إنَّ  "  كان  "  إذا كانت ناقصةً ،   أنها تكون فعلاً ؛ وهذا محالٌ ؛ لأن الفعل ما دلَّ على اقتران حدثٍ بزمان ،   فقولك  "  كان  "  يدل على حصول معنى الكون في الزمان الماضي ،   وإذا أفاد هذا المعنى ،   كانت تامةً ،   لا ناقصةً ،   فهذا الدليل يقتضي أنها إن كانت فعلاً ،   كانت تامةً لا ناقصة ،   وإن لم تكن تامةً لم تكن فعلاً البتة ؛ بل كانت حرفاً ،   وأنتم تنكرون ذلك ؛ فبقوا في هذا الإشكال زماناً طويلاً ،   وصنَّفوا في الجواب عنه كتباً ،   وما أفلحوا فيه ،   ثم انكشف لي فيه سرٌّ أذكره - هاهنا - وهو : أنَّ  "  كانَ  "  لا معنى له إلاَّ أنه حدث ،   ووقع ،   ووجد إلاَّ أن قولك وجد ،   وحدث على قسمين :<br />
أحدهما : أن يكون المعنى وجد ،   وحدث الشيء ؛ كقولك : وجد الجوهر ،   وحدث العرض .  <br />
والثاني : أن يكون المعنى وجد ،   وحدث موصوفية الشيء بالشيء ،   فإذا قلت : كان زيدٌ عالماً ،   فمعناه : حدث في الزمان الماضي موصوفية زيد بالعلم .  <br />
والقسم الأول هو المسمَّى ب  "  كان  "  التامة .  <br />
والقسم الثاني : هو المسمَّى ب  "  الناقصة  "  وفي الحقيقة : فالمفهوم من  "  كان  "  في الموضعين هو الحدوث ،   والوقوع إلاَّ أنه في القسم الأول المراد حدوث الشيء في نفسه فلا جرم كان الاسم الواحد كافياً والمراد في القسم الثاني حدوث موصوفية أحد الأمرين بالآخر ،   فلا جرم لم يكن الاسم الواحد كافياً ،   بل لا بدَّ فيه من ذكر الاسمين حتى يمكنه أن يشير إلى موصوفية أحدهما بالآخر ،   وهذا من لطائف الأبحاث .  <br />
فأما إن قلنا إنه فعل ،   كان دالاًّ على وقوع المصدر في الزمان الماضي فحينئذٍ تكون تامةً لا ناقصةً ،   وإن قلنا إنه ليس بفعل بل حرفٌ ،   فكيف يدخل فيه الماضي والمستقبل ،   والأمر ،   وجميع خواصِّ الأفعال ؟ وإذا حمل الأمر على ما قلناه ،   تبيَّن أنه فعلٌ وزال الإشكال بالكلية .  <br />
المفهوم الثالث ل  "  كان  "  أن تكون بمعنى  "  صَارَ  "  ؛ وأنشدوا : [ الطويل ]<br />
بِتَيْهَاءَ قَفْرٍ وَالمَطِيُّ كَأَنَّهَا *** قَطَا الحَزْنِ قَدْ كَانَتْ فِرَاخاً بُيُوضُهَا{[4763]}<br />
وعندي أنَّ هذا اللفظ - هاهنا - محمولٌ على ما ذكرناه ،   فإنَّ معنى  "  صار  "  أنَّها حدثت موصوفية الذات بهذه الصفة ،   بعد أنها ما كانت موصوفة بذلك ؛ فتكون  "  كان  "  هنا أيضاً بمعنى حدث ،   ووقع ؛ إلاَّ أنه حدوثٌ مخصوصٌ وهو أنه حدث موصوفية الذات بهذه الصفة ،   بعد أن كان الحاصل موصوفية الذات بصفة أخرى .  <br />
المفهوم الرابع : أن تكون زائدة ؛ وأنشدوا : [ الوافر ]<br />
سَرَاةُ بَنِي أبي بَكْرٍ تَسَامَى *** عَلَى كَانَ المُسَوَّمَةِ الجِيَادِ{[4764]}<br />
و  "  العُسْرَةِ  "  اسم من الإعسار ،   ومن العسر ،   وهو تعذُّر الموجود من المال ؛ يقال : أعسر الرجل ،   إذا صار إلى حالة العسرة ،   وهي الحالة التي يتعسَّر فيها وجود المال .  <br />
قوله :  "  فَنَظِرَةٌ  "  الفاء جواب الشرط ،   و "  نَظِرَةٌ  "  خبر مبتدأ محذوف ،   أي : فالأمر أو فالواجب ،   أو مبتدأٌ خبرُه محذوفٌ ،   أي : فعليكم نظرةٌ .  <br />
وقرأ العامَّة :  "  نَظِرَةٌ  "  بزنة  "  نَبِقَة  "  .   وقرأ الحسن{[4765]} ،   ومجاهدٌ ،   وأبو رجاء :  "  فَنَظْرةٌ  "  بتسكين العين ،   وهي لغةٌ تميمةٌ يقولون :  "  كَبْد  "  في  "  كَبِد  "  و "  كَتْف  "  في  "  كَتِف{[4766]}  "  .  <br />
وقرأ عطاء  "  فَنَاظِرَةٌ  "  على فاعلة ،   وقد خرَّجها أبو إسحاق{[4767]} على أنها مصدر نحو : { لَيْسَ لِوَقْعَتِهَا كَاذِبَةٌ } [ الواقعة :2 ] { يَعْلَمُ خَائِنَةَ الأَعْيُنِ } [ غافر :19 ] { أَن يُفْعَلَ بِهَا فَاقِرَةٌ } [ القيامة :25 ] .   وقال الزمخشري :  "  فناظِرُهُ ،   أي : فصاحب الحقّ ناظره ،   أي : منتظره ،   أو صاحب نظرته على طريقة النسب ؛ كقولهم : مَكَانٌ عَاشِبٌ ،   وبَاقِلٌ ؛ بمعنى ذو عشبٍ ،   وذو بقلٍ ،   وعنه :  "  فناظِرْهُ  "  على الأمر بمعنى : فسامحه بالنظرة ،   وباشره بها  "  فنقله عنه القراءة الأولى يقتضي أن تكون قراءته  "  ناظِر  "  اسم فاعل مضافاً لضمير ذي العسرة ،   بخلاف القراءة التي قدمها عن عطاء ،   فإنها  "  نَاظِرَةٌ  "  بتاء التأنيث ،   ولذلك خرَّجها الزَّجَّاج على المصدر .   وقرأ{[4768]} عبد الله :  "  فناظِرُوه  "  أمراً للجماعة بالنظرة ،   فهذه ستُّ قراءات مشهورها واحدةٌ .  <br />
وهذه الجملة لفظها خبرٌ ،   ومعناها الأمر ؛ كقوله تعالى : { وَالْوَالِدَاتُ يُرْضِعْنَ أَوْلاَدَهُنَّ } [ البقرة :233 ] وقد تقدَّم .   والنظرة : من الانتظار ،   وهو الصبر والإمهال .  <br />
تقول : بعته الشيء بنظرة ،   وبإنظار .   قال : { قَالَ أَنظِرْنِي إِلَى يَوْمِ يُبْعَثُونَ قَالَ إِنَّكَ مِنَ المُنظَرِينَ } [ الأعراف :14-15 ] { إِلَى يَوْمِ الْوَقْتِ الْمَعْلُومِ } [ الحجر :38 ] .  <br />
قوله : { إِلَى مَيْسَرَةٍ } قرأ نافع وحده :  "  مَيْسُرَة  "  بضمِّ السِّين ،   والباقون بفتحها .   والفتح هو المشهور ؛ إذ مفعل ،   ومفعلة بالفتح كثيرٌ ،   ومفعُلٌ بالضم ،   معدومٌ ؛ إلا عند الكسائي ،   فإنه أورد منه ألفاظاً ،   وأمَّا مفعلةٌ ،   فقالوا : قليلٌ جدّاً وهي لغة الحجاز ،   وقد جاءت منها ألفاظٌ ،   نحو : المسرُقة ،   والمقبرة ،   والمشربة ،   والمسربة ،   والمقدرة ،   والمأدبة ،   والمفخرة ،   والمزرعة ،   ومعربة ،   ومكرمة ،   ومألكة .  <br />
وقد ردَّ النحاس{[4769]} الضمَّ ؛ تجرؤاً منه ،   وقال :  "  لم تَأْتِ مَفْعُلَةٌ إِلاَّ في حُرُوفٍ معدودةٍ ليس هذه منها ،   وأيضاً فإنَّ الهاء زائدةٌ ،   ولم يَأْتِ في كلامِهِم مَفْعُل الْبتةَ  "  انتهى .  <br />
وقال سيبويه{[4770]} :  "  لَيْسَ في الكَلاَمِ مَفْعُل  "  قال أبو عليّ :  "  يَعْنِي في الآحاد  "  .   وقد حكى سيبويه  "  مَهْلك  "  مثلَّث اللاَّم ،   وقال الكسائيُّ :  "  مَفْعُل  "  في الآحاد ،   وأورد منه مكرُماً في قول الشاعر : [ الرجز ]<br />
لِيَوْمِ رَوْعٍ أَوْ فَعَالِ مَكْرُمِ{[4771]} *** .   .   .  <br />
ومَعْوُن في قول الآخر - هو جميلٌ - : [ الطويل ]<br />
بُثَيْنُ ،   الْزَمِي  "  لاَ  "  ؛ إنَّ  "  لاَ  "  إِنْ لَزِمْتِهِ *** عَلَى كَثْرَةِ الوَاشِينَ أَيُّ مَعْوُنِ{[4772]}<br />
ومألكاً في قول عديّ : [ الرمل ]<br />
أَبْلِغِ النُّعْمَانَ عَنِّي مَأْلُكاً *** أَنَّهُ قَدْ طَالَ حَبْسِي وَانْتِظَارِي{[4773]}<br />
وهذا لا يرد على سيبويه لوجهين :<br />
أحدهما : أنَّ هذا جمعٌ لمكرمةٍ ،   ومعونة ،   ومألكة ،   وإليه ذهب البصريون ،   والكوفيون خلا الكسائي ،   ونقل عن الفراء أيضاً .  <br />
والثاني : أن سيبويه لا يعتدُّ بالقليل ،   فيقول :  "  لم يَرِدْ كذا  "  وإن كان قد ورد منه الحرف والحرفان ،   لعدم اعتداده بالنادر القليل .  <br />
وإذا تقرَّر هذا ،   فقد خطَّأ النحويون مجاهداً ،   وعطاءً في قراءتهما :  "  إلى مَيْسُرِهِ  "  بإضافة  "  مَيْسُر  "  مضموم السين إلى ضمير الغريم ؛ لأنهم بنوه على أنه ليس في الآحاد مفعل ،   ولا ينبغي أن يكون هذا خطأً ؛ لأنه على تقدير تسليم أنَّ مفعلاً ليس في الآحاد ،   فميسر هنا ليس واحداً ،   إنما هو جمع ميسرة ،   كما قلتم أنتم : إنَّ مكرماً جمع مكرمةٍ ،   ونحوه ،   أو يكون قد حذف تاء التأنيث للإضافة ؛ كقوله : [ البسيط ]<br />
إِنَّ الخَلِيطَ أَجَدُّوا البَيْنَ فَانْجَرَدُوا *** وأَخْلَفُوكَ عِدَ الأَمْرِ الَّذِي وَعَدُوا{[4774]}<br />
أي : عدة الأمر ؛ ويدلُّ على ذلك أنهم نقلوا عنهما ،   أنهما قرآ أيضاً :  "  إِلَى مَيْسَرِهِ  "  بفتح السين ،   مضافاً لضمير الغريم ،   وهذه القراءة نصٌّ فيما ذكرته لك من حذف تاء التأنيث للإضافة ؛ لتوافق قراءة العامَّة :  "  إلى مَيْسَرَةٍ  "  بتاء التأنيث .  <br />
وقد خرَّجها أبو البقاء{[4775]} على وجهٍ آخر ،   وهو أن يكون الأصل :  "  مَيْسُورِه  "  فخفِّف بحذف الواو ؛ اكتفاءً بدلالة الضمة عليها ،   وقد يتأيَّد ما ذكره على ضعفه ،   بقراءة عبد الله ،   فإنه قرأ : إلى  "  مَيْسُورِه  "  بإضافة  "  مَيْسورٍ  "  للضمير ،   وهو مصدرٌ على مفعول ؛ كالمجلود والمعقول ،   وهذا إنما يتمشَّى على رأي الأخفش ؛ إذ أثبت من المصادر زنة مفعول ،   ولم يثبته سيبويه .  <br />
 "  والمَيْسَرَةُ  "  : مفعلةٌ من اليسر ،   واليسار الذي هو ضدُّ الإعسار ،   وهو تيسر الموجود من المال ومنه يقال أيسر الرجل ،   فهو موسرٌ ،   أي : صار إلى حالة وجود المال فالميسرة ،   واليسر ،   والميسور : الغنى .  <br />
فصل في سبب نزول  "  وإن كان ذو عسرة  " <br />
لما نزل قوله : { وَإِنْ تُبْتُمْ فَلَكُمْ رُؤُوسُ أَمْوَالِكُمْ لاَ تَظْلِمُونَ وَلاَ تُظْلَمُونَ } [ البقرة :279 ] قال بنو عمرو الثَّقفي : بل نتوب إلى الله ،   فإنه لا طاقة لنا بحرب الله ورسوله ،   فرضوا برأس المال ،   فشكى بنو المغيرة العسرة ،   وقالوا : أخِّرونا إلى أن تدرك الغلاَّت ،   فأبوا أن يؤخروا ؛ فأنزل الله تعالى : { وَإِن كَانَ ذُو عُسْرَةٍ } يعني وإن كان الذي عليه الدَّين معسراً ،   { فَنَظِرَةٌ إِلَى مَيْسَرَةٍ } ،   يعن : أَنْظِرُوه إلى اليسار ،   والسَّعة .  <br />
فصل في بيان حكم الإنظار<br />
اختلفوا في حكم الإنظار : هل هو مختصٌّ بالربا ،   أو عامٌّ في كل دين ؟ فقال ابن عباس ،   وشريح ،   والضحاك ،   والسدي ،   وإبراهيم : الآية في الربا ،   وذكر عن شريح أنَّه أمر بحبس الخصم ،   فقيل له : إنه معسرٌ ،   فقال شريح إنما ذلك في الربا ،   والله تعالى قال في كتابه العزيز { إِنَّ اللهَ يَأْمُرُكُمْ أَن تُؤدُّواْ الأَمَانَاتِ إِلَى أَهْلِهَا } [ النساء :58 ] .  <br />
وقال جماعة منهم مجاهد : إنها عامَّة في كل دين{[4776]} ؛ لعموم قوله تعالى : { ذُو عُسْرَةٍ } ولم يقل ذا عسرة .  <br />
فصل<br />
والإعسار : هو ألاَّ يجد في ملكه ما يؤدِّيه بعينه ،   ولا يكون له ما لو باعه ،   لأمكنه أداء الدَّين من ثمنه خارجاً عن مسكنه وثيابه ،   ولا يجوز أن يحبس من لم يجد إلاَّ قوت يوم لنفسه وعياله ،   وما لا بدَّ لهم من كسوةٍ لصلاتهم ودفع البرد والحرِّ عنهم .  <br />
واختلفوا : إذا كان قويّاً ،   هل يلزمه أن يؤاجر نفسه من صاحب الدَّين ،   أو غيره ؟<br />
فقال بعضهم : يلزمه ذلك ،   كما يلزمه إذا احتاج لنفسه ،   أو لعياله .  <br />
وقال بعضهم : لا يلزمه ذلك ،   واختلفوا أيضاً إذا بذل للمعسر ما يؤدِّي به الدَّين ،   هل يلزمه قبوله والأداء ،   أو لا يلزمه ؟ فأما من له بضاعة كسدت عليه فواجبٌ عليه أن يبيعها بالنقصان إن لم يكن إلاَّ ذلك ،   ويؤدِّيه في الدَّين .  <br />
فصل في تحريم حبس المعسر<br />
إذا علم الإنسان أنَّ غريمة معسرٌ - حرم عليه حبسه ،   وأن يطالبه بما له عليه ووجب عليه إنظاره إلى يساره ،   فإن ارتاب في إعساره ،   جاز له حبسه إلى أن يظهر إعساره ،   فإذا ادَّعى الإعسار وكذَّبه الغريم ،   فإن كان الدَّين عن عوضٍ ،   كالبيع ،   والقرض ،   فلا بدَّ له من إقامة البيِّنة على أنَّ ذلك العوض قد هلك ،   وإن كان الدَّين عن غير عوض كالإتلاف ،   والصَّداق ،   والضَّمان ؛ فالقول قول المعسر ؛ لأن الأصل الفقر وعلى الغريم إقامة البينة .  <br />
فصل<br />
قال المهدويُّ{[4777]} : قال بعض العلماء : هذه الآية ناسخةٌ لما كان في الجاهلية من بيع من أعسر .  <br />
قال القرطبيُّ{[4778]} : وحكى مكيٌّ : أنَّ النبي - صلى الله عليه وسلم - أمر به في صدر الإسلام .  <br />
قال ابنُ عطية : فإن ثبت فِعْلُ النبي - صلى الله عليه وسلم - فهو نسخٌ ،   وإلاَّ فليس بنسخ .  <br />
قال الطَّحاوي{[4779]} : كان الحرُّ يباع في الدَّين أول الإسلام ،   إذا لم يكن له مالٌ يقضيه عن نفسه ؛ حتى نسخ الله ذلك بقوله : { وَإِن كَانَ ذُو عُسْرَةٍ فَنَظِرَةٌ إِلَى مَيْسَرَةٍ } .  <br />
واحتجُّوا بما رواه الدَّارقطنيُّ من حديث مسلم بن خالد الزَّنجي ،   قال : حدثنا زيد ابن أسلم عن ابن البيلمانيّ ،   عن سرَّقٍ ،   قال : كان لرجلٍ عليّ مالٌ - أو قال - دينٌ فذهب إلى رسول الله - صلى الله عليه وسلم - فلم يصب لي مالاً ،   فباعني منه ،   أو باعني له ؛ أخرجه البزَّار بإسناد طويل ،   ومسلم بن خالد الزنجي ،   وعبد الرحمن البيلماني لا يحتجُّ بهما .  <br />
قوله : { وَأَن تَصَدَّقُوا } مبتدأ وخبره  "  خير  "  وقرأ عاصم{[4780]} : بتخفيف الصاد ،   والباقون : بتثقيلها .   وأصل القراءتين واحدٌ ؛ إذ الأصل : تتصدَّقوا ،   فحذف عاصمٌ إحدى التاءين : إمَّا الأولى ،   وإمَّا الثانية ،   وتقدَّم تحقيق الخلاف فيه ،   وغيره أدغم التاء في الصاد ،   وبهذا الأصل قرأ عبد الله{[4781]} :  "  تَتَصَدَّقوا  "  .   وحذف مفعول التصدُّق للعلم به ،   أي : بالإنظار ؛ لقوله - صلى الله عليه وسلم - :  "  لاَ يَحِلُّ دَيْنُ رَجلٍ مُسْلم ،   فيؤخره ؛ إلاَّ كان له بِكُلِّ يَوْمٍ صدقةٌ{[4782]}  "  وهذا ضعيفٌ ؛ لأن الإنظار ثبت وجوبه بالآية ،   فلا بد من حمل هذه الآية على فائدةٍ جديدةٍ ،   ولأن قوله  "  خَيْرٌ لكُمْ  "  إنما يليق بالمندوب ،   لا بالواجب .   وقيل : برأس المال على الغريم ،   إذ لا يصحُّ التصدق به على غيره ؛ كقوله تعالى : { وَأَن تَعْفُواْ أَقْرَبُ لِلتَّقْوَى } [ البقرة :237 ] .  <br />
قوله : { إِن كُنْتُمْ تَعْلَمُونَ } جوابه محذوفٌ ،   و "  أَنْ تَصَدَّقُوا  "  بتأويل مصدرٍ مبتدأ ،   و "  خيرٌ لكم  "  خبره .  <br />
فصل في تقدير مفعول  "  تعلمون  "  ونصب  "  يوماً  " <br />
وتقدير مفعول  "  تَعْلَمُونَ  "  فيه وجوه :<br />
أحدها : إن كنتم تعلمون أنَّ هذا التصدُّق خير لكم إن عملتموه .  <br />
الثاني : إن كنتم تعلمون فضل التصدُّق على الإنظار والقبض .  <br />
الثالث : إن كنتم تعلمون أنَّ ما يأمركم به ربُّكم أصلح لكم .
