قوله تعالى : { صِرَاطَ الَّذِينَ أَنْعَمْتَ عَلَيْهِمْ غَيْرِ الْمَغْضُوبِ عَلَيْهِم وَلاَ الضَّآلِّينَ }<br />
قولُه تَعَالَى : { صِرَاطَ الَّذِينَ أَنْعَمْتَ عَلَيْهِمْ } .  <br />
 "  صِرَاطَ الذِيْنَ  "  بدل منه ،   بدل كُلِّ مِنْ كُلّ ،   وهو بَدَلُ مَعْرِفَةٍ مِنْ مَعْرِفَةٍ .  <br />
والبدلُ سبعةُ أَقْسَامٍ على خلاف في بعضها :<br />
بدلُ كُلّ من كُلّ ،   وبدل بَعْضٍ من كُلّ ،   وبدلُ اشْتِمَالٍ ،   وبدلُ غَلَطٍ ،   وبدل نِسْيَان ،   وبدل بَدَاء ،   وبدل كُلّ من بعض .  <br />
أما الأقسامُ الثلاثَةُ الأُوَلُ ،   فلا خلافَ فِيها .  <br />
وأما بدلُ البدَاء ،   فأثبته بعضُهم ؛ مستدلاًّ بقوله عليه الصَّلاة والسَّلام :  "  وإنَّ الرَّجُلَ لَيُصَلِّي الصلاَة ،   وما كتب له نِصْفُهَا ثُلُثُهَا رُبعُهَا إلى العُشُرِ  "  ولا يَرِدُ هذا في القرآن الكريمِ .  <br />
وأما الغَلَطُ والنسْيَانُ ،   فأثبتهما بعضُهم ؛ مُسْتَدِلاًّ بقول ذي الرُّمَّةِ : [ البسيط ]<br />
75- لَمْيَاءُ فِي شَفَتَيْهَا حُوَّةٌ لَعَسٌ- *** -وَفِي اللِّثَاتِ وَفِي أَنْيَابِهَا شَنَبُ{[323]}<br />
قال : لأنَّ  "  الحُوّة  "  السّوادُ الخالِصُ ،   و  "  اللَّعَسُ  "  سواد يشوبه حُمْرَة ،   ولا يرِدُ هذان البدلان في كَلاَمٍ فصيحٍ .  <br />
وأما بدل الكُلّ من البعض ،   فأثبته بعضهُم ،   مُسْتَدِلاًّ بظاهِر قوله : [ الخفيف ]<br />
76- نَضَرَ اللهُ أَعْظُماً دَفَنُوهَا- *** -بِسِجِسْتَانَ طَلْحَةَ الطَّلَحَاتِ{[324]}<br />
في رواية مَنْ نَصَبَ  "  طَلْحَةَ  "  ،   قال : لأنَّ  "  الأَعْظُمَ  "  بعضُ  "  طَلْحَةَ  "  ،   و  "  طَلْحَةَ  "  كُلّ وقد أُبْدِلَ منها ؛ واستدلّ - أيضاً - بقول امرىء القيس [ الطويل ]<br />
77- كَأَنِّي غَدَاةَ البَيْنِ يَوْمَ تَحَمَّلُوا- *** -لَدَى سَمُرَاتِ الحَيِّ نَاقِفُ حَنْظَلِ{[325]}<br />
ف  "  غَدَاةَ  "  بعضُ  "  اليوم  "  ،   وقد أُبْدِلَ  "  اليوم  "  منها .  <br />
ولا حُجَّةَ في البيتيْنِ ،   أما الأولُ : فإنَّ الأَصْلَ  "  أعظماً دفنوها أَعْظُمَ طلحة  "  ثم حُذِفَ المضافُ ،   وأقيم المُضَافُ إليهِ مُقَامه ؛ ويدلُّ على ذلك الروايةُ المشهورةُ وهي جَرُّ  "  طَلْحَةَ  "  على أنَّ الأصل :  "  أعظم طلحة  "  ولم يُقَم المضاف إليه مقامَ المضاف .  <br />
وأما الثاني : فإنَّ  "  اليَوْمَ  "  يُطلقُ على القطعةِ من الزمان ،   كما تقدّم ،   وليس هذا موضعَ البَحْثِ عَن دَلائِلِ المذهبيْن .  <br />
وقيل :  "  الصراط  "  الثاني غير الأول ،   والمرادُ به : العلمُ بالله تعالى .   قاله جَعْفَرُ بنُ محمد{[326]} رحمه الله تعالى : وعلى هذا فتخريجته أن يكونَ مَعْطُوفاً حُذِفَ منه حَرْفُ العَطْفِ ،   وبالجملة فهو مُشْكلٌ .  <br />
والبدلُ ينقسمُ أيضاً إلى :<br />
بدل مَعرفةٍ ،   ونكرةٍ منْ نكرةٍ ،   ومعرفةٍ منْ نكرةٍ ،   ونكرةٍ مِنْ معرفةٍ .  <br />
ويَنْقَسمُ أيضاً إلى :<br />
بدل ظاهِر من ظاهرٍ : ومُضْمَرٍ مِنْ مُضْمَرٍ ،   وظاهرٍ مِنْ مضمر ،   ومضمرٍ من ظاهر .  <br />
وفائدةُ البَدلِ : الإيضاحُ بعد الإبْهَامِ ؛ لأنهُ يُفِيدُ تأكيداً من حَيْثُ المعنى ،   إذ هو على نيّةِ تَكْرَار العامل .  <br />
و  "  الذين  "  في مَحَلِّ جرٍّ بالإضافة ،   وهو اسمُ موصولٍ ،   لافتقاره إلى صِلَةٍ وعائدٍ ،   وهو جمع  "  الذي  "  في المعنى ،   والمشهور فيه أن يكونَ بالياءِ ،   رفعاً ،   ونصباً ،   وجرًّا ؛ وبعضُهم يرفعُه بالواوِ ؛ جَرْياً له مَجْرَى جَمْعِ المذكَّر السَّالم ؛ ومنه : [ الرجز ]<br />
78- نَحْنُ الَّذُونَ صَبَّحُوا الصَّبَاحَا- *** -يَوْمَ الفَسَادِ غَارَةً مِلْحَاحَا{[327]}<br />
وقد تُحْذَفُ نُونُه اسْتِطَالةً بصلته ؛ كقوله : [ الطويل ]<br />
79- وَإِنَّ الَّذِي حَانَتْ بِفَلْجٍ دمَاؤُهُمْ- *** -هُمُ الْقَوْمُ كُلُّ القَوْمِ يَا أُمَّ خَالِدِ{[328]}<br />
ولا يقع إلاَّ على أُولي العلمِ ،   [ ولا يقع مجرى جمع المذكر السَّالم ،   بخلاف مجرده فإنه يقع على أولي العلم ]{[329]} وغيرهم .  <br />
و  "  أَنْعَمْتَ  "  : فِعْلٌ ،   وفاعلٌ ،   صِلَة المَوْصُولِ .  <br />
والتاء في  "  أنعمتَ  "  ضميرُ مرفوعٍ مُتَّصل .   و  "  عليهم  "  جار ومجرور متعلّق ب  "  أنعمتَ  "  ،   والضميرُ هو العائدُ ،   وهو ضمير جمع المذكرين العقلاء ،   ويستوي فيه لفظ مُتَّصِلِه ومُنْفَصِلِهِ .  <br />
والهمزةُ في  "  أنعمتَ  "  ؛ لجَعْلِ الشيءِ صَاحِبَ {[330]}ما صِيَغَ منه ،   فحقُّه أن يَتَعَدَّى بِنَفْسِه ،   ولكن ضُمِّنَ معنى  "  تَفَضَّلَ  "  فَتَعَدَّى تَعْدِيَتَهُ .  <br />
وقرأ عمر{[331]} بنُ الخَطّابِ ،   وابنُ الزُّبَيْرِ{[332]} رضي الله - تعالى - عنهما  "  صِرَاطَ مَنْ أَنْعَمْتَ  "  .  <br />
ول  "  أَفْعَلَ  "  أربعةٌ وعشرُونَ مَعْنى ،   تقدّمَ وَاحِدٌ .  <br />
والتعدِيَةُ ؛ نحو :  "  أَخْرَجتُه  "  .  <br />
والكثرةُ ؛ نحو :  "  أَظْبَى المَكَانُ  "  أَيْ :  "  كَثُرَ ظِبَاؤُه  "  .  <br />
والصَّيرورةُ ؛ نحو :  "  أَغَدَّ البَعِيرُ  "  صار ذا غُدّة .  <br />
والإعانة ؛ نحو :  "  أَحْلَبْتُ فُلاَناً  "  أي : أعنتُه على الحَلْبِ .  <br />
والتَّشْكِيَةُ ؛ نحو :  "  أَشكيتُه  "  أي : أزلتُ شِكَايَتَهُ .  <br />
والتَّعرِيضُ ؛ نحو :  "  أبعتُ المبتاعَ  "  ،   أي : عرضتُه للبيع .  <br />
وإصابةُ الشيءِ بمعنى ما صيغ منه ؛ نحو :  "  أحمدتُه  "  أي : وجدتُه محموداً .  <br />
وبلوغُ عَدَدٍ ؛ نحو :  "  أعْشَرتِ الدَّرَاهِمُ  "  ،   أي : بلغتِ العَشَرَة .  <br />
أو بلوغُ زَمانٍ ؛ نحو  "  أصبح  "  ،   أو مَكَانٍ ؛ نحو  "  أَشْأَمَ  "  .  <br />
وموافقَةُ الثّلاثي ؛ نحو :  "  أحزتُ المكانَ  "  بمعنى : حُزْتُهُ .  <br />
أَوْ أَغْنَى عن الثلاثي ؛ نحو :  "  أَرْقَلَ البعيرُ  "  .  <br />
ومطاوعةُ  "  فَعَلَ  "  ؛ نحو قَشَع الريح ،   فَأَقْشَع السّحابُ .  <br />
ومطاوعَةُ  "  فَعَّلَ  "  ؛ نحو :  "  فَطَّرْتُه ،   فَأَفْطَرَ  "  .  <br />
ونَفْيُ الغريزَةِ ؛ نحو :  "  أسرع  "  .  <br />
والتَّسميةُ ؛ نحو :  "  أخطأتهُ  "  ،   أَيْ : سَمَّيْتُه مخْطِئاً .  <br />
والدعاءُ ؛ نحو :  "  أسقيتُه  "  ،   أَي : قلتُ له : سَقَاكَ الله تعالى .  <br />
والاستحقاقُ ؛ نحو  "  أَحْصَدَ الزرعُ  "  ،   أَيْ : استحقَّ الحصادَ .  <br />
والوصولُ ؛ نحوه :  "  أَعْقَلْتُهُ  "  ،   أَيْ : وَصَّلْتُ عقلي إليه .  <br />
والاستقبالُ نحو :  "  أَفَفْتُه  "  ،   أَي : استقبلتُه بقول : أُفٍّ .  <br />
والمجيءُ بالشيء ؛ نحو :  "  أكثرتُ  "  أَيْ : جئتُ بالكثير .  <br />
والفرقُ بين أَفْعَلَ وفَعَل ،   نحو : أَشْرَقَتِ الشَّمسُ : أضاءتْ ،   وشَرَقَتْ : طَلَعَتْ .  <br />
والهجومُ ؛ نحو : أَطْلَعْتُ على القوم ،   أيْ : اطَّلعْتُ عَلَيْهِمْ .  <br />
و  "  على  "  حرف استعلاء حقيقةً أو مجازاً ؛ نحو : عليه دَيْنٌ : ولها معانٍ أُخَرُ ،   منها : المُجَاوزة ؛ كقوله : [ الوافر ]<br />
80- إِذَا رَضِيَتْ عَلَيَّ بَنُو قُشَيْرٍ- *** -لَعَمْرُ اللهِ أَعْجَبَنِي رِضَاهَا{[333]}<br />
أيْ : عَنِّي .  <br />
وبمعنى  "  الباءِ  "  { حَقِيقٌ عَلَى أَن لاَّ أَقُولَ } [ الأعراف : 105 ] ،   أي : بأَنْ ،   وبمعنى  "  فِي "  ؛ { الشَّيَاطِينُ عَلَى مُلْكِ سُلَيْمَانَ } [ البقرة : 102 ] أيْ : فِي [ مُلْكِ ] ،   { الْمَالَ عَلَى حُبِّهِ ذَوِي الْقُرْبَى } [ البقرة : 177 ] .  <br />
والتعليلُ : { وَلِتُكَبِّرُواْ اللَّهَ عَلَى مَا هَدَاكُمْ } [ البقرة : 185 ]  ؛ أي لأجلِ هِدَايَتِه إياكم .  <br />
وبمعنى  "  مِن  "  : { حَافِظُونَ إِلاَّ عَلَى أَزْوَاجِهِمْ }[ المؤمنون : 5 ،   6 ] ،   أيْ : إلاّ مِنْ أَزواجهم .  <br />
والزيادة كقوله : [ الطويل ]<br />
81- أبَى اللهُ إلاَّ أَنَّ سَرْحَةَ مَالِكٍ- *** -عَلَى كُلِّ أَفْنَانِ العِضَاهِ تَرُوقُ{[334]}<br />
لأنَّ  "  تُروقُ  "  يتعدى بنفسِه ،   ولكل موضع من هذه المواضعِ مَجَالٌ للنظر .  <br />
وهي مترددةٌ بين الحَرْفِيَّةِ ،   والاسْمِيَّةِ{[335]} ؛ فتكون اسماً في موضعين :<br />
أحدهُما : أن يدخلَ عليها حَرْفُ الجَرّ ؛ كقول الشاعر : [ الطويل ]<br />
82- غَدَتْ مِنْ عَلَيْهِ بَعْدَ مَا تَمَّ ظِمْؤُهَا- *** -تَصِلُّ وَعَنْ قَيْضٍ بِزَيزَاءَ مَجْهَلِ{[336]}<br />
ومعناها  "  فَوْق  "  ،   أيْ : من فوقه .  <br />
والثاني : أنْ يؤدي جعلُه حرفاً ،   إلى تعدِّي فعل المضمر المنفصل إلى ضمير المُتّصل في غيرِ المَوَاضِع الجَائِز فيها ؛ ومن ذلك قوله : [ المتقارب ]<br />
83- هَوِّنْ عَلَيْكَ فإنَّ الأُمُورَ- *** -بِكَفِّ الإِلهِ مَقَادِيرُهَا{[337]}<br />
ومثلُها في هذيْن الحُكْمَيْن  "  عَنْ  "  ،   وستأتي إنْ شاء الله تعالى .  <br />
وزعم بعضُهم أنَّ  "  على  "  مترددةٌ بين الاسم ،   والفِعْلِ ،   والحرفِ .  <br />
أما الاسمُ والحرفُ ،   فقد تقدما .  <br />
وأما الفعلُ : قال : فإنك تقولُ :  "  عَلاَ زيدٌ  "  أي : ارتفع .   وفي هذا نَظَرٌ ؛ لأن  "  عَلاَ  "  إذا كان فِعْلاً ،   مُشْتَقٌّ من العُلُوِّ ،   وإذا كان اسماً أو حرفاً ،   فلا اشتقاقَ له ،   فليس هو ذَاكَ ،   إلاَّ أنَّ هذا القَائِلَ يَرُدُّ هذا النظرَ ،   [ بقولهم : إنَّ  "  خَلاَ  "  ،    "  وَعَدا  "  مترددانِ بين الفعليَّةِ والحرفيَّةِ ،   ولم يلتفتوا إلى هذا النظر ]{[338]} .  <br />
والأصلُ في هاء الكِناية الضَّمُّ ،   فإنْ تقدمها ياءٌ ساكنة ،   أو كسرةٌ ،   كَسَرَها غيرُ الحِجازَيين ؛ نحو : عَلَيْهِم وفِيهِمْ وَبِهِمْ .  <br />
والمشهورُ في مِيمِهَا السكونُ قبل متحرك ،   والكسرُ قبلَ ساكن ،   هذا إذا كَسَرْتَ الهاء ،   أما إذا ضممتَ ،   فالكَسْرُ ممتنع إلاّ في ضَرُورة ؛ كقوله :  "  وفِيهُمِ الحكام  "  بِكَسْرِ المِيمِ .  <br />
وفي  "  عَلَيْهِمْ  "  عشرُ لُغاتٍ :<br />
قُرِئ{[339]} بِبَعْضِها :  "  عَلَيْهُِمْ  "  بكسر الهاء وضمها ،   مع سُكُون الميم .  <br />
 "  عَلَيْهِمِي  "  ،   بكسر الهاء ،   وزيادة الياء ،   وبكسر الميم فقط .  <br />
 "  عليهُمُو  "  بضم الميم ،   وزيادة واو ،   أو الضم فقط .  <br />
 "  عليهِمُو  "  بِكَسْرِ الهاءِ ،   وضم الميمِ ،   بزيادة الواو .  <br />
 "  عليهُمِي  "  بِضَمِّ الهاء ،   وزيادة ياء بعد الميم .  <br />
أو الكسر فقط  "  عليهِمُ  "  بكسر الهاء ،   وضم الميم ،   حكى ذلك ابنُ الأَنْبَارِي{[340]} .  <br />
والتفسيرُ ،   قال البَغَويُّ - رحمه الله تعالى - : صراط الذين أنعمت عليهم أي : مَنَنْتَ عليهم بِالهِدَايَةِ والتوفيق ،   وقال عِكْرَمة{[341]}- رضي الله تعالى عنه - : مَنَنْتَ عليهم بالثَّبات على الإيمان والاسْتِقَامَة وعلى الأنبياء عليهم السلام .  <br />
وقِيل : على كُلِّ مَنْ ثَبَتَهُ الله - تعالى - من النَّبِيِّين والمُؤْمنين الذين ذكرهم الله - تَعَالَى - في قوله :{ فَأُوْلَئِكَ مَعَ الَّذِينَ أَنْعَمَ اللَّهُ عَلَيْهِم مِّنَ النَّبِيِّينَ وَالصِّدِّيقِينَ وَالشُّهَدَآءِ وَالصَّالِحِينَ } [ النساء : 69 ]<br />
وقال ابنُ عباس - رضي الله تعالى عنهما - هُمْ قومُ مُوسَى ،   وعِيسَى - عليهما الصلاة والسلام ،   قبل أن غيروا دينهم .  <br />
وقال أَبُو العَالِيَةَ{[342]} : هم آلُ الرسولِ صلى الله عليه وسلم وأبو بكر ،   وعمر رضي الله عنهما .  <br />
وقال شَهْرُ بْنُ حَوْشَبٍ{[343]} - رضي الله عنه - : هم أصحابُ النبي صلى الله عليه وسلم ،   وأهل بَيْتِهِ .  <br />
وقرأ حَمْزَةُ {[344]} "  عَلَيْهُمْ  "  ،   و  "  إلَيْهُمْ  "  ،   و  "  لَدَيْهُمْ  "  بضم الهاء .  <br />
ويضم يَعْقُوب{[345]} كُلَّ هاءٍ قبلها ياءٌ ساكنة تثنيةً وجمعاً ،   إلاّ قولَه تعالى :<br />
{ بَيْنَ أَيْدِيهِنَّ وَأَرْجُلِهِنَّ } [ الممتحنة : 12 ] .  <br />
والآخرُونَ : بكسرها .   فَمَنْ ضَمَّها ردّها إلى الأصل ؛ لأنها مضمومة عند الانفراد .  <br />
ومَنْ كسرها ،   فالأصل الياءُ السَّاكنة ،   والياءُ أختُ الكسرة .  <br />
وضم ابنُ كَثِير{[346]} ،   وأَبُو جَعْفَر{[347]} كُلَّ ميم جمع مُشْبِعاً في الوَصْلِ ،   إذا لم يلقها ساكن ،   فإنْ لقيها ساكِنٌ فلا يُشْبِع .  <br />
ونَافِعٌ يُخَيِّرُ ،   ويضمُّ وَرْش{[348]} عند ألِفِ القطع .  <br />
وإذا تلقته ألفُ الوصلِ ،   وقبل الهاء كسرٌ ،   أو ياءٌ ساكنةٌ ،   ضمّ الهاءَ والمِيمَ حَمْزَةُ والكسائي - رحمهما الله - وكسرَهُما أَبُو عَمْرو ،   وكذلك يَعْقُوبُ إذَا انْكَسر ما قبله .  <br />
والآخرون : بضمّ الميم ،   وكسرِ الهاء ؛ لأجل الياء أو لكسر ما قبلها ،   وضمّ الميم على الأصل ،   وقرأ عمرُ بن الخَطَّاب{[349]} - رضي الله تعالى عنه - :  "  صرَاطَ مَنْ أنْعَمْتَ عَلَيْهِمْ  "  .  <br />
قال ابنُ الخَطِيب{[350]} - رحمه الله تعالى - : اخْتُلِفَ في حَدّ النّعْمَةِ :<br />
فقال بعضُهم : إنَّها عِبَارَةٌ عن المَنْفَعةِ المفعولة على جِهَةِ الإحسان إلَى الغيرِ .  <br />
[ ومنهم مَنْ يقولُ : المنفعة الحسنة المفعولة على جهة الإحْسَان إلى الغير ] .  <br />
قالوا : وإنما زدْنا على هذا القَيْدِ ،   لأن النعمةَ يستحقّ لها الشكر والإحسان [ والحقّ أن هذا القيد غير معتبر ؛ لأنه لا يجوز أن يستحق الشكر بالإحسان ]{[351]} ،   وإنْ كان فعله محظوراً ؛ لأن جهةَ استحقاقِ السكر غير جهةِ استحقاق الذَّنْب والعِقاب ،   فأيُّ امتناعٍ في اجتماعهما ؟ أَلاَ ترى أن الفاسِقَ يستحقُّ بإنعامه الشُّكْرَ ،   والذَّمَّ بمعصيةِ الله تعالى ،   فلا يجوزُ أنْ يَكُونَ الأمرُ ها هنا كذلك .  <br />
ولنرجع إلى تفسير الحَدّ : فنقول : أما قَولُنا :  "  المنفعة  "  ؛ فلأن المَضَرّةَ المحضةَ لا تكون نِعْمَةً .  <br />
وقولنا : المفعولة على جهة الإحسان ؛ لأنه لو كان نفعاً حقًّا وقَصَدَ الفاعلُ به نفعَ نفسه ،   نَفْعَ المفعولِ به ،   فلا يكون نِعْمَةً{[352]} ،   كَمَنْ أحسن إلى جَاِريَتِهِ ،   ليربَحَ عليها .  <br />
وها هنا فوائدُ :<br />
الفائِدَةُ الأُوْلَى : أَنَّ كلّ ما يصل إلى الخلق من النفع ،   ودفع الضَّرر ،   فهو من الله تعالى على ما قال تبارك وتعالى :{ وَمَا بِكُم مِّن نِّعْمَةٍ فَمِنَ اللَّهِ } [ النحل : 53 ] ،   ثمَّ إنَّ النعمةَ على ثلاثةِ أَقْسَامٍ :<br />
أحدُها : نِعمةٌ تَفَرَّدَ الله - تعالى - بإيجَادِهَا ،   نحو : أنْ خَلَق ورَزَقَ .  <br />
وثانيها{[353]} : نعمةٌ وصلت إلينا من جهةِ غير الله - تعالى - في ظاهرِ الأَمْرِ ،   وفي الحقيقة فهي - أيضاً - إنّما وصلتْ من الله تبارك وتعالى ؛ وذلك لأنه - تعالى - هو الخالقُ لتلك النعمةِ ،   والخالقُ لذلك المنعِِمِ ،   وخالقٌ لداعيةِ الإنْعَام بتلك النعمة في قلب ذلك المنعم ،   إلاّ أنه تبارك وتعالى لَمّا أَجْرَى تلك النعمة على يَدِ ذلك العَبْدِ ،   كان ذلك العبدُ مشكوراً ،   ولكن المشكور في الحقيقة هو الله - تعالى - ولهذا قال تعالى : { أَنِ اشْكُرْ لِي وَلِوَالِدَيْكَ إِلَيَّ الْمَصِيرُ } [ لقمان : 14 ] فبدأ بنفسِه ،   تنبيهاً على أن إنعامَ الخلقِ لا يتمّ إلاّ بإنعام الله تعالى .  <br />
وثالثها : نِعمٌ وصلت من الله إلينا بسبب طاعتنا ،   وهي أيضاً من الله تعالى{[354]} ؛ لأنه لولا أنَّ الله - سبحانه وتعالى - وَفّقنا للطاعات ،   وأعاننا عليها ،   وهدانا إليها ،   وأَزَاحَ الأعذار عَنا ،   وإِلاَّ لَمَا وَصَلْنَا إلى شَيْءٍ منها ،   فظهر بها التقرير أنَّ جَمِيعَ النعم في الحقيقة من الله تَعَالى .  <br />
الفائدةُ الثَّانيةُ : اختلفوا [ في أنه ]{[355]} هل لله - تعالى - نعمةً على الكافرِ أَم لاَ ؟ فقال بعضُ أصحابنا : ليس لله - تعالى - على الكافر نعمة .  <br />
وقالت المعتزلةُ : لله - تعالى - على الكافر نعمة دينية ،   ونعمة دنيوية .  <br />
واحتجَّ الأصحابُ على صحّةِ قولهم ،   بالقرآن [ الكريم ]{[356]} ،   والمعقول .  <br />
أما القرآنُ ؛ فقوله تبارك وتعالى : { صِرَاطَ الَّذِينَ أَنْعَمْتَ عَلَيْهِمْ }  ؛ وذلك لأنه لو كان لله على الكافر نعمةٌ ،   لكانوا داخِلِينَ تحت قوله : { أَنْعَمْتَ عَلَيْهِمْ } فيكون طلباً لصراطِ الكُفّارِ ،   وذلك بَاطِلٌ ،   فثبت بهذه الآيةِ أنه ليس لله - تعالى - على الكافر نعمةٌ .  <br />
فإن قَالُوا : إنَّ قَوْله : { الصِّرَاطَ الْمُسْتَقِيمَ } يدفعُ ذَلكَ .  <br />
قلنا : { صِرَاطَ الَّذِينَ أَنْعَمْتَ عَلَيْهِمْ } بدل مِنْ قَوْلِه : { الصِّرَاطَ الْمُسْتَقِيمَ }  ؛ فكان التَّقْدِير :  "  اهْدِنا صراطَ الذين أنعمت عليهم  "  ،   وحينئذٍ يَعُودُ المحذوفُ المذكورُ .  <br />
وقوله تبارك وتعالى : { وَلاَ يَحْسَبَنَّ الَّذِينَ كَفَرُواْ أَنَّمَا نُمْلِي لَهُمْ خَيْرٌ لأَِنفُسِهِمْ إِنَّمَا نُمْلِي لَهُمْ لِيَزْدَادُواْ إِثْماً } [ آل عمران : 178 ] .  <br />
وأما المَعْقُولُ : فهو أَنَّ نِعَمَ الدنيا الفانيةَ في مُقَابلةِ عَذَابِ الآخرة على الدوام ،   كالقَطْرة في البحر ،   ومثل هذا لا يكون نِعْمَةً ،   بدليل أنَّ مَنْ جعل السُّمَّ في الحَلْوَى لم يَعُدِ النفعُ الحاصلُ منه نعمةً ؛ لأجل أن ذلك النفع حقيرٌ في مُقابلةِ ذلك الضَّرر الكبير{[357]} ،   فكذا ههُنا .  <br />
وأما الَّذِين قالوا : إن لله على الكافر نعمةً ،   فقد احتجُّوا بقوله تعالى : { يَاأَيُّهَا النَّاسُ اعْبُدُواْ رَبَّكُمُ الَّذِي خَلَقَكُمْ وَالَّذِينَ مِن قَبْلِكُمْ لَعَلَّكُمْ تَتَّقُونَ الَّذِي جَعَلَ لَكُمُ الأَرْضَ فِرَاشاً وَالسَّمَاءَ بِنَآءً } [ البقرة : 21 ،   22 ] ،   على أنه يَجبُ على الكُلِّ طاعةُ الله - تعالى - لأجلِ هذه النعم ،   وإلاّ لما كانت هذه النعمُ العظيمةُ معتبرةً ؛ وقولِه تعالى : { كَيْفَ تَكْفُرُونَ بِاللَّهِ وَكُنتُمْ أَمْوَاتاً فَأَحْيَاكُمْ } [ البقرة : 28 ] ،   ذكر ذلك في معرض الامْتِنَانِ ،   وشرحِ النعم .  <br />
وقولِه تعالى : { يَابَنِي إِسْرَائِيلَ اذْكُرُواْ نِعْمَتِيَ الَّتِي أَنْعَمْتُ عَلَيْكُمْ } [ البقرة : 40 ] .  <br />
وقولِه تعالى : { وَقَلِيلٌ مِّنْ عِبَادِيَ الشَّكُورُ } [ سبأ : 13 ] .  <br />
وقول إبليس : { وَلاَ تَجِدُ أَكْثَرَهُمْ شَاكِرِينَ } [ الأعراف : 17 ] .  <br />
ولو لم تحصل النعمة ،   لم يلزمْ من عَدَمِ إقدامِهم على الشكر محذورٌ ؛ لأنّ الشكر لا يمكن إلاَّ عند حصول النعمة .  <br />
الفائدة الثالثةُ : قال ابنُ الخَطِيب{[358]} - رحمه الله - : قوله تعالى : { اهْدِنَا الصِّرَاطَ الْمُسْتَقِيمَ صِرَاطَ الَّذِينَ أَنْعَمْتَ عَلَيْهِمْ } يدل على إمامةِ أبي بكر - رضي الله عنه ؛ لأنا ذكرنا أن تقديرَ الآية :  "  اهدنا صراط الذين أنعمت عليهم  "  والله - تعالى - قد بيّن في آية أُخْرَى أَنَّ { الَّذِينَ أَنْعَمْتَ عَلَيْهِمْ } من هم ؛ بقوله تعالى : { فَأُوْلَئِكَ مَعَ الَّذِينَ أَنْعَمَ اللَّهُ عَلَيْهِم مِّنَ النَّبِيِّينَ وَالصِّدِّيقِينَ } [ النساء : 69 ] ورئيسهم أبو بكر الصّديق - رضي الله تعالى عنه - فكان معنى الآية أن الله - تعالى - أمرنا أن نطلب الهداية [ التي كان عليها أبو بكر الصديق ،   وسائر الصّديقين ،   ولو كان أبو بَكْرٍ - رضي الله تعالى عنه - غيرَ إمامٍ ،   لما جَازَ الاقتداء به ]{[359]} ،   فثبت [ بما ذكرناه دلالة هذه الآية على ]{[360]} إمامة أبي بكر رضي الله عنه .  <br />
الفائدة الرابعة : قوله تعالى : { أَنْعَمْتَ عَلَيْهِمْ } يتناول كُلّ من كان لله - تعالى - عليه نعمة ،   وهذه النعمة إما أن يكون المراد منها نعمة الدنيا ،   أو نعمة الدين ،   والأول باطل فثبت أن المراد منه نعمة الدين .  <br />
فنقول : كل نعمة ديِنِيّة سوى الإيمان فهي مشروطة بحصول الإيمان ،   وأمّا نعمة الإيمان فيمكن حصولها خالياً عن سائر النعم الدينية ،   وهذا يدلّ على أن المراد من قوله تعالى : { أَنْعَمْتَ عَلَيْهِمْ } هو نعمة الإيْمَان ،   فرجع حاصل القول في قوله تعالى : { هْدِنَا الصِّرَاطَ الْمُسْتَقِيمَ صِرَاطَ الَّذِينَ أَنْعَمْتَ عَلَيْهِمْ } أنه طلب لنعمة الإيمان ،   وإذا ثبت هذا الأصل ،   فيتفرع عليه أحكام :<br />
الأول : أنه لما ثبت أن المُرادَ من هذه النعمة نِعْمَةُ الإيمان ،   إذ لفظ الآية الكريمة صريح في أن الله - تعالى - هو المنعم بالنعمة ،   ثبت أنّ الخالق للإيمان ،   والمعطي للإيمان هو الله تعالى ،   وذلك يدلّ على فساد قول المعتزلة ،   وكان الإيمان أعظم النعم ،   فلو كان الفاعل للإيمان هو العبد لكان إنعام العبد أشرف وأعلى من إنعام الله تعالى ،   ولو كان كذلك لما حسن من الله - تعالى - أن يذكر إنعامه في معرض التعظيم .  <br />
الحكم الثاني : يجب ألاَّ يبقى المؤمن مخلداً في النار ؛ لأن قوله : { أَنْعَمْتَ عَلَيْهِمْ } مذكور في معرض التَّعْظيم بهذا الإنعام ،   ولو لم يكن له أثر في دفع العَذَاب المؤبّد لكان قليل الفائدة ،   فما كان يحسن من الله - تعالى - ذكره في معرض التَّعْظيم .  <br />
الحكم الثالث : دلّت الآية الكريمة على أنه لا يجب على الله - تعالى - رعاية [ الصلاح والأصلح ]{[361]} في الدين ؛ لأنه لو كان الإرْشاد على الله - تعالى - واجباً لم يكن ذلك إنعاماً ،   وحيث سماه الله - تعالى - إنعاماً علمنا أنه غير واجب .  <br />
الحكم الرابع : لا يجوز أن يكون المراد بالإنعام الإقدار على الإيمان ؛ لأن الله - تبارك وتعالى - قدر المكلف عليه ،   وأرشده إليه ،   وأزاح أعْذَارَهُ وعِلَلَهُ عَنْهُ ،   لأن كل ذلك حاصل في حَقّ الكفار ،   فلما خص - تعالى - بعض المكلفين بهذا الإنعام ،   مع أن الإقدار ،   وإزاحة العلل حاصل في حَقّ الكل ،   علمنا أن المراد ليس هو الإقدار ،   وإزاحة الموانع .  <br />
قوله تعالى : { غَيْرِ الْمَغْضُوبِ عَلَيْهِمْ وَلاَ الضَّآلِّينَ }<br />
 "  غير  "  بدل من  "  الذين  "  بدل نكرة من معرفة .  <br />
وقيل : نعت ل  "  الذين  "  ،   وهو مشكل ؛ لأن  "  غير  "  نكرة و  "  الذين  "  معرفة{[362]} ،   وأجابوا عنه بجوابين :<br />
أحدهما : أن  "  غير  "  إنما يكون نكرة إذا لم يقع بين ضدّين ،   فأما إذا وقع بين ضدين فقد انحصرت الغيرية ،   فيتعرف  "  غير  "  حينئذ بالإضافة ،   تقول :  "  مررت بالحركة غير السكون  "  والآية من هذا القبيل ،   وهذا إنما يتمشّى على مذهب ابن السّراج ،   وهو مرجوح .  <br />
والثاني : أن الموصول أَشْبَهَ النكرات في الإبْهَام الذي فيه ،   فعومل معاملة النكرات .  <br />
وقيل : إن  "  غير  "  بدل من المضمر المجرور في  "  عليهم  "  ،   وهذا يشكل على قول من يرى أن البدل يحل محلّ المبدل منه ،   وينوي بالأول الطّرح ؛ إذ يلزم منه خلو الصّلة من العائد ،   ألا ترى أن التقدير يصير :  "  صراط الذين أنعمت على غير المغضوب عليهم  "  .  <br />
و  "  المغضوب  "  خفض بالإضافة ،   وهو اسم مفعول ،   والقائم مقام الفاعل الجار والمجرور ،   ف  "  عليهم  "  الأولى منصوبة المَحَلّ ،   والثانية مرفوعته ،   و  "  أل  "  فيه موصولة ،   والتقدير :  "  غير الذين غُضِب عليهم  "  .  <br />
والصحيح في  "  أل  "  الموصولة أنها اسم لا حَرْفٌ .  <br />
واعلم أن لفظ  "  غير  "  مفرد مذكر أبداً ،   إلا أنه إن أريد به مؤنث جاز تأنيث فعله المسند إليه ،   نقول :  "  قامت غيرك  "  ،   وأنت تعني امرأة ،   وهي في الأصل صفة بمعنى اسم الفاعل ،   وهو مغاير ،   ولذلك لا تتعرف بالإضافة ،   وكذلك أخواتها ،   أعني نحو :  "  مِثْل وشِبْه وشَبِيه وخِدْن وتِرْب  "  .  <br />
وقد يستثنى بها حملاً على  "  إلاّ  "  كما يوصف ب  "  إلاّ  "  حملاً عليها ،   وقد يراد بها النفي ك  "  لا  "  ،   فيجوز تقديم معمول معمولها عليها ،   كما يجوز في  "  لا  "  تقول :  "  أنا زيداً غَيْرُ ضارب  "  أي : غير ضارب زيداً ؛ ومنه قول الشاعر : [ البسيط ]<br />
84- إِنَّ امْرَءًا خَصَّنِي عَمْداً مَوَدَّتْهُ- *** -عَلَى التَّنَائِي لَعِنْدِي غَيْرُ مَكْفُورِ{[363]}<br />
تقديره : غير مكفور عندي ،   ولا يجوز ذلك فيها إذا كانت لغير النَّفي .  <br />
لو قلت :  "  جاء القوم زيداً غير ضارب  "  ،   تزيد : غير ضارب زيداً لم يجز ؛ لأنها ليست بمعنى  "  لا  "  التي يجوز فيها ذلك على الصَّحيح من الأقوال في  "  لا  "  .  <br />
وفيها قول ثانٍ يمنع ذلك مطلقاً .  <br />
وقول ثالث : يفصل بين أن تكون جَوَاب قَسَمٍ ،   فيمتنع فيها ذلك ،   وبين ألاّ يكون فيجوز .  <br />
وهي من الألفاظ اللاَّزمة للإضافة لفظاً وتقديراً ،   فإدْخَال الألف واللام عليها خَطَأ .  <br />
واختلفوا هل يجوز دخول  "  أل  "  على  "  غير وبعض وكل  "  والصحيح جوازه .  <br />
قال البغوي - رحمه الله تعالى - :  "  غير  "  ها هنا بمعنى  "  لا  "  و  "  لا  "  بمعنى  "  غير  "  ،   ولذلك جاز العَطْفُ عليها ،   كما يقال :  "  فلان غير مُحْسن ولا مجمل  "  ،   فإذا كان  "  غير  "  بمعنى  "  لا  "  ،   فلا يجوز العَطْفُ عليها ب  "  لا  "  ؛ لا يجوز في الكلام :  "  عندي سوى عبد الله ولا زيد  "  .  <br />
وقرئ{[364]} :  "  غَيْرَ  "  نصباً ،   فقيل : حال من  "  الَّذِين  "  وهو ضعيف ؛ لمجيئه من المُضَاف إليه في غير المواضع الجائز فيها ذَلِكَ ،   كما ستعرفه إن شَاءَ اللهُ تعالى : وقيل : من الضمير في  "  عليهم  "  .  <br />
وقيل على الاستثناء المنقطع ،   ومنعه الفَرَّاء ؛ قال : لأن  "  لا  "  لا تُزَادُ إلاَّ إذا تقدمها نفي ،   كقول الشاعر : [ البسيط ]<br />

88- أَبَى جُودُهُ لاَ البُخْلَ واسْتَعْجَلَتْ بِهِ- *** -نَعَمْ مِنْ فَتًى لا يَمْنَعُ الجُودَ نَائِلُهْ{[368]}<br />
ف  "  لا  "  في هذه المواضع كلها صلةٌ .  <br />
وفي هذا الجواب نظر ؛ لأن الفَرَّاء لم يقل : إنها غير زائدة ،   وقولهم : إن  "  لا  "  زائدة في الآية ،   وتنظيرهم بالمَوَاضِعِ المتقدّمة لا تفيد ،   وإنّما تحرير الجواب أن يقولوا : وجدت  "  لا  "  زائدةً من غير تقدّم نفي ،   كهذه المواضع المتقدمة .  <br />
ويحتمل أن تكون  "  لا  "  في قوله :  "  لا البُخْلَ  "  مفعولاً به ل  "  أَبَى  "  ،   ويكون نصب  "  البُخْلَ  "  على أنه بدل من  "  لا  "  أي : أبى جُودُهُ قَوْلَ لا ،   وقول : لا هو البخل ،   ويؤيد هذا قوله :  "  واسْتَعْجَلَتْ بِهِ نَعَمْ  "  فجعل  "  نَعَمْ  "  فاعل  "  اسْتَعْجَلَتْ  "  ،   فهو من الإِسْنَادِ اللَّفْظي ،   أي : إلى جود هذا اللَّفظ ،   واستعجل به هذا اللفظ .  <br />
وقيل : إن نصب  "  غير  "  بإضمار أعني .   ويحكى عن الخليل ،   وقدّر بعضهم بعد  "  غير  "  محذوفاً قال : التقدير :  "  غير صِرَاط المَغْضُوب  "  ،   وأطلق هذا التَّقدير ،   فلم يقيده بِجَرّ  "  غير  "  ،   ولا نصبه ولا يتأتى ذلك إلاَّ مع نصبها ،   وتكون صفةً لقوله تعالى : { الصِّرَاطَ الْمُسْتَقِيمَ } وهذا ضعيف ؛ لأنه متى اجتمع البدل{[369]} والوصف قدم الوصف ،   فالأولى أن تكون صفةً ل  "  صراط الذين  "  ،   ويجوز أن تكون بدلاً من  "  الصراط المستقيم  "  ،   أو من  "  صراط الذين " {[370]} إلا أنه يلزم منه تكرار البدل ،   وفي جوازه نَظَر ،   وليس في المَسْألة نقل ،   إلاّ أنهم قد ذكروا ذلك في بَدَلِ البَدَاء خَاصّة ،   أو حالاً من  "  الصراط  "  الأول أو الثاني .  <br />
واعلم أنّه حيث جعلنا  "  غير  "  صفةً فلا بد من القول{[371]} بتعريف  "  غير  "  ،   أو إبهام الموصوف ،   وجريانه مجرى النكرة ،   كما تقدم تقريره ذلك في القراءة بجرّ  "  غير  "  .  <br />
و  "  لا  "  في قوله تعالى : { وَلاَ الضَّآلِّينَ } زائدة لتأكيد معنى النَّفي المفهوم من  "  غير  "  لئلا يتوهّم عطف  "  الضَّالين  "  على  "  الذين أنعمت  "  .  <br />
وقال الكوفيون : هي بمعنى  "  غير  "  وهذا قريبٌ من كونها زائدةً ،   فإنه لو صرح ب  "  غير  "  كانت للتأكيد أيضاً ،   وقد قرأ بذلك عمر بن الخَطَّاب وأبيٌّ رضي الله عنهما .  <br />
و  "  الضَّالين  "  مجرور عطفاً على  "  المغضوب  "  ،   وقرىء شاذاً " {[372]} الضَّأَلِّينَ  "  ،   بهمز الألف ؛ وأنشدوا : [ الطويل ]<br />

91- وَلَّى نَعَامُ بَنِي صَفْوَانَ زَوْزَأَةً- *** - .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .  {[375]}<br />
بهمز ألف  "  زَوْرَأَة  "  ،   والظَّاهر أنها لغةٌ مطَّردةٌ ؛ فإنهم قالوا في قراءة{[376]} ابن ذَكْوَان{[377]} :  "  مِنْسَأَتَهُ  "  بهمز ساكنة : إنّ أصلها ألف ،   فقلبت همزة ساكنة .  <br />
فإن قيل : لم أتى بصلة  "  الذين  "  فعلاً ماضياً ؟<br />
قيل : ليدلّ ذلك على ثبوت إنعام الله - تبارك وتعالى - عليهم وتحقيقه لهم ،   وأتى بصلة  "  أل  "  اسماً ليشمل سائر الأزمان ،   وجاء مبنيًّا للمفعول ؛ تحسيناً للفظ ؛ لأنّ من طلبت منه الهداية ،   ونسب الإنعام إليه لا يناسبه نسبة الغضب إليه ،   لأنه مقام تلطُّف ،   وترفّق لطلب الإحسان ،   فلا يحسن مواجهته بصفة الانتقام .  <br />
والإنعام : إيصال الإحسان إلَى الغير ،   ولا يقال إلا إذا كان الموصل إليه الإحْسَان من العُقَلاَء ،   فلا يقال : أنعم فلان على فَرَسِهِ ،   ولا حماره .  <br />
والغضب : ثَورَان دم القلب إرادة الانتقام ،   ومنه قوله عليه الصلاة والسلام :  "  اتَّقُوا الغَضَبَ فإنه جَمْرَةٌ تُوقَدُ في قَلْبِ ابْنِ آدَمَ ،   ألم تَرَ إلى انْتِفَاخِ أَوْدَاجِهِ وحُمْرَةِ عينيه{[378]} "  .  <br />
وإذا وصف به الباري - تبارك وتعالى - فالمراد به الانتقام لا غيره .  <br />
قال ابن الخطيب - رحمه الله تعالى - : هنا قاعدة كليةٌ ،   وهي أن جميع الأعراض النَّفْسَانية - أعني الرحمة ،   والفرح ،   والسُّرور ،   والغضب ،   والحَيَاء ،   والعُتُوّ ،   والتكبر ،   والاستهزاء -لها أوائل ولها غايات .  <br />
ومثاله : الغضب : فإنّ أول غليان دم القلب ،   وغايته : إرادة إيصال الضَّرَرِ إلى [ المغضوب عليه ،   فلفظ الغضب في حق الله لا يحمل على أوله الذي هو غليان دم القلب ،   بل على غايته الذي هو إرادة الإضرار ،   وأيضاً الحَيَاءُ ]{[379]} له أول وهو انكسار النفس ،   وهذه قاعدة شريفة في هذا الباب .  <br />
ويقال : فُلاَن غُضبَّة : إذا كان سريع الغَضَبِ .  <br />
ويقال : غضبت لفلان إذا كان حيًّا وغضبت به إذا كان ميتاً .  <br />
وقيل : الغضب تغيُّر القلب لمكروه .  <br />
وقيل : إن أريد بالغضب العُقُوبة كان صفة فعل ،   وإن أريد به إرادة العقوبة كان صفة ذاتٍ .  <br />
والضلال : الخَفَاء والغيبوبة .  <br />

92- أَلَمْ تَسْأَلْ فَتُخْبِرَكَ الدِّيَارُ- *** -عَن الحَيِّ المُضَلَّلِ أَيْنَ سَارُوا{[380]} ؟<br />
 "  والضَّلضلَة  "  : حجر أملس يَرُده السَّيْل في الوادي .  <br />
ومن الثاني :<br />
{ أَإِذَا ضَلَلْنَا فِي الأَرْضِ }[ السجدة : 10 ] ،   وقيل : الضّلال : العُدُول عن الطريق المستقيم ،   وقد يُعَبَّرُ به عن النِّسْيان كقوله تعالى :{ أَن تَضِلَّ إِحْدَاهُمَا } [ البقرة : 282 ] بدليل قوله :{ فَتُذَكِّرَ } [ البقرة : 282 ] .  <br />
التفسير : قيل :  "  المغضوب عليهم  "  هم اليهود .  <br />
وقيل :  "  الضالون  "  هم النصارى ؛ لأن الله - تعالى - حكم على اليهود بالغَضَبِ فقال تعالى :{ مَن لَّعَنَهُ اللَّهُ وَغَضِبَ عَلَيْهِ } [ المائدة : 60 ] ،   وحكم على النصارى بالضَّلال فقال تعالى : { وَلاَ تَتَّبِعُواْ أَهْوَآءَ قَوْمٍ قَدْ ضَلُّواْ مِن قَبْلُ } [ المائدة : 77 ] .  <br />
وقيل : هذا ضعيف ؛ لأن منكري الصَّانع والمشركين أَخْبَثُ ديناً من اليهود والنصارى ،   فكان الاحتراز من دينهم أولى .  <br />
وقيل :  "  المغضوب عليهم  "  : هم : الكُفّار ،   و  "  الضّالون  "  : هم المنافقون .  <br />
وقال سهل بن عبد الله رضي الله عنهما :  "  غير المغضوب عليهم  "  بالبِدْعَةِ ،    "  والضّالين  "  عن السُّنَّة .  <br />
والأَوْلَى أن يحمل  "  المغضوب عليهم  "  على كل من أَخْطَأَ في الاعتقاد ؛ لأن اللفظ عام ،   والتقييد خلاف الأصل .  <br />
فَصْلٌ في عصمة الأنبياء والملائكة<br />
قال ابن الخطيب- رحمه الله تعالى - :  "  غير المغضوب عليهم  "  يدلُّ على أن أحداً من الملائكة ،   والأنبياء - عليهم الصلاة والسلام - ما أَقْدَمَ على عملٍ مخالف قول الدين ،   ولا على اعتقاد مخالف اعتقاد دين الله ؛ لأنه لو صدر عنه ذلك لكان قد ضَلّ عن الحق ،   لقوله تعالى : { فَمَاذَا بَعْدَ الْحَقِّ إِلاَّ الضَّلاَلُ } [ يونس : 32 ] ،   ولو كانوا ضالين لما جاز الاقتداء بهم ،   ولا بطريقهم ،   ولكانوا خارجين عن قوله تعالى : { أَنْعَمْتَ عَلَيْهِمْ } ،   ولما كان ذلك باطلاً علمنا بهذه الآية عِصْمةَ الملائكة ،   والأنبياء عليهم الصلاة والسلام .  <br />
فَصْلٌ في إضافة الغضب لله<br />
قالت المعتزلة : غَضَبُ الله - تعالى - عليهم يدلُّ على كونهم فاعلين للقبائح باختيارهم ،   وإلاَّ لكان الغضب عليهم ظلماً من الله - تعالى - عليهم .  <br />
وقال أصحابنا - رحمهم الله تعالى - : لما ذكر غضب الله عليهم ،   وأتبعه بذكر كونهم ضالين دلّ ذلك على أن غضب الله - تعالى - عليهم علّة لكونهم ضالين ،   وحينئذ تكون صفة الله - تعالى - مؤثرةً في صفة العبد .  <br />
أما لو قلنا : إن كونهم ضالين يوجب غضب الله - تعالى - عليهم لزم أن تكون صفة العبد مؤثرة في صفة الله تعالى ،   وذلك مُحَال .  <br />
فَصْلٌ<br />
قال ابن الخطيب{[381]} - رحمه الله تعالى - : دلّت هذه الآية على أن المكلّفين ثلاث فرق :<br />
أهل الطاعة ،   وإليهم الإشارة بقوله تعالى : { أَنْعَمْتَ عَلَيْهِمْ } .  <br />
وأهل البغي والعدوان ،   وهم المراد بقوله تعالى : { غَيْرِ الْمَغْضُوبِ عَلَيْهِمْ } .  <br />
وأهل الجهل في دين الله ،   وإليهم الإشارة بقوله تعالى : { وَلاَ الضَّآلِّينَ } .  <br />
فإن قيل : لم قدم ذكر العُصَاة على ذكر الكَفَرَةِ ؟<br />
قلنا : لأن كل أحد يحترز عن الكفر ،   أما قد لا يحترز عن الفِسْق ،   فكان أهم فقدم لهذا السّبب ذلك .  <br />
قال ابن الخطيب - رحمه الله تعالى - : ها هنا سؤال ،   وهو أن غضب الله إنما تولَّد عن علمه بصدور القبيح والجناية{[382]} عنه ،   فهذا العلم إما أن يقال : إنه قديم ،   أو محدث ،   فإن كان قديماً فلم خلقه ،   ولم أخرجه من العَدَمِ إلى الوجود ،   مع علمه بأنه لا يستفيد من دخوله في الوجود إلا العَذَاب الدَّائم ،   ولأنه من كان غضبان على الشَّيء كيف [ يعقل ]{[383]} إقدامه على إيجَادِهِ وتكوينه ؟ فإن كان ذلك العلم حادثاً لكان الباري - تعالى - محلاًّ للحوادث ،   إلاَّ أنه يلزم أن يفتقر إحداث ذلك العلم إلى سَبْقِ علمٍ آخر ،   وتسلسل ،   وهو مُحَال .  <br />
والجواب : يفعل الله ما يشاء ويحكم ما يريد .  <br />
سؤال آخر<br />
وهو أن من أنعم الله - تعالى - عليه امتنع أن يكون مغضوباً عليه ،   وأن يكون من الضَّالين ،   فلما ذكر قوله : { أَنْعَمْتَ عَلَيْهِمْ } ،   فما الفائدة في أن ذكر عقيبه : { غَيْرِ الْمَغْضُوبِ عَلَيْهِم وَلاَ الضَّآلِّينَ } ؟<br />
والجواب : الإيمان إنما يكمل بالرَّجَاء والخوف ،   كما قال عليه الصلاة والسلام :  "  لَوْ وُزِنَ خَوْفُ المُؤْمِن وَرَجَاؤُهُ لاعتْدَلا " {[384]} ،   فقوله : { صِرَاطَ الَّذِينَ أَنْعَمْتَ عَلَيْهِمْ } يوجب الرَّجَاء الكامل ،   وقوله تعالى : { غَيْرِ الْمَغْضُوبِ عَلَيْهِمْ وَلاَ الضَّآلِّينَ } يوجب الخوف الكامل ،   وحينئذٍ يقوى الإيمان بركنيه وطرفيه ،   وينتهي إلى حَدِّ الكمال .  <br />
سؤال آخر<br />
ما الحكمة في أنه - تَعَالَى - جعل المقبولين طائفةً واحدةً ،   وهم الذين أنعم الله عليهم ،   والمردودين فريقين : المغضوب عليهم ،   والضَّالين ؟<br />
فالجواب : أنّ الذين كملت نعم الله - تَعَالَى - عليهم هم الذين جمعوا بين معرفة الحَقّ لذاته ،   والخير لأجل العمل به ،   فهؤلاء هم المُرَادون بقوله : { أَنْعَمْتَ عَلَيْهِمْ } ،   فإن اختلّ قيد العمل فهم الفَسَقَةُ ،   وهم المغضوب عليهم ،   كما قال تعالى : { وَمَن يَقْتُلْ مُؤْمِناً مُّتَعَمِّداً فَجَزَآؤُهُ جَهَنَّمُ خَالِداً فِيهَا وَغَضِبَ اللَّهُ عَلَيْهِ وَلَعَنَهُ }[ النساء :93 ]<br />
