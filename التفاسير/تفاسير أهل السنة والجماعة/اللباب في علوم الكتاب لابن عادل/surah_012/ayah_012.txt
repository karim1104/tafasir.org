قوله : { أَرْسِلْهُ مَعَنَا غَداً يَرْتَعْ وَيَلْعَبْ } في :  "  يرْتَعْ ويَلعَبْ  "  أربع عشرة قراءة :<br />
أحدها : قراءة نافع : بالياء من تحت ،   وكسر العين .  <br />
الثانية : قراءة البزِّي ،   عن ابن كثيرٍ :  "  نَرْتَعِ ونلعب  "  بالنُّون وكسر العين .  <br />
الثالثة : قراءة قنبل ،   وقد اختلف عليه ،   فنقل عه ثُبُوت الياء بعد العين وصلاً ووقفاً ،   وحذفها وصلاً ووقفاً ،   فيوافق البزِّي في أحد الوجهين عنه ،   فعنه قراءتان .  <br />
الخامسة : قراءة أبي عمرو ،   وابن عامر :  "  نَرتَعْ ونَلعَبْ  "  بالنُّون ،   وسكون العين ،   والباء .  <br />
السادسة : قراءة الكوفيين :  "  يَرْتَعْ ويَلعبْ  "  بالياء من تحت وسكون العين والباءِ .  <br />
وقرأ جعفر بن محمد :  "  نَرْتَعْ  "  بالنُّون ،    "  ويَلْعَبْ  "  بالياء ،   ورُويت عن ابن كثيرٍ .  <br />
وقرأ العلاء بن سيابة :  "  يَرْتَعِ ويَلْعَبُ  "  بالياء فيهما ،   وكسر العين وضمّ الباء .  <br />
وقرأ أبو رجاء كذلك ،   إلا أنَّه بالياء من تحت فيهما .  <br />
والنخعي ويعقوبك  "  نَرْتَع  "  بالنون ،    "  ويَلْعَب  "  بالياء .  <br />
وقرأ مجاهدٌ ،   وقتادةُ ،   وابن محيصِن :  "  يَرْتَعْ ويَلْعَب  "  بالياء ،   والفعلان في هذه القراءات كلها مبنيان للفاعل .  <br />
وقرأ زيد بن علي :  "  يُرْتَع ويُلْعَب  "  بالياء من تحت فيهما مبنيين للمعفول .  <br />
وقرىء :  "  نَرْتَعِي ونَلْعَبُ  "  بثبوت الياء ،   ورفع الباء .  <br />
وقرأ ابن أبي عبلة :  "  نَرْعَى ونَلْعَب  "  .  <br />
فهذه اربع عشرة قراءة منها ستٌّ في السَّبع المتواتر وثمان في الشواذٍّ .  <br />
فمن قرأ بالنُّون ،   فقد أسند الفعل إلى إخوة يوسف .  <br />
سُئل أبو عمرو بن العلاء : كيف قالوا : نلعب وهم أنبياء ؟ قال : كان ذلك قبل أن يُنَبِّئهُم الله عزَّ وجلَّ .  <br />
قال ابن الأعرابي : الرَّتْع : الأكل بشدة ،   وقيل : إنه الخَصْبُ .  <br />
وقيل : المراد من اللَّعب : الإقدام على المُباحات ،   وهذا يوصف به الإنسان ،   كما رُوِي عن النبي صلى الله عليه وسلم أنه قال لجابر :  "  هَلاَّ بِكْراً تُلاعِبُهَا وتُلاعِبُك  "  .  <br />
وقيل : كان لعبهم الاستباق ،   والغرض منه : تعليم المحاربة ،   والمقاتلة مع الكُفَّار ،   ويدلُّ عليه قولهم :  "  إِنَّا ذَهَبْنَا نَسْتبِقُ  "  وإنما سمَّوه لعباً ؛ لأنه في صُورة اللَّعِب .  <br />
وأما من قرأ بالياءِ ،   فقد أسند الفعل إليه دونهم ،   فالمعنى : أنه يبصر رَعْي الإبلِ ؛ لتيدرَّب بذلك ،   فمرَّة يرتع ،   ومرَّة يلعب ؛ كفعل الصِّبيان .  <br />
ومن كسر العين ،   اعتقد أنه جزم بحذف حرف العلَّة ،   وجعلهُ مأخُوذاً من يفتعِل من الرَّعي ؛ كيَرْتَمِي من الرَّمْي ،   ومن سكن العين ،   واعتقد أنه جزم بحذق الحركة ،   وجعلهُ مأخوذاً من : رَتَعَ يَرْتَعُ ،   إذا اتَّسع في الخِصْب قال :<br />
3058 .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   *** وإذَا يَحْلُولَهُ الحِمَى رَتَعْ<br />
ومن سكَّن الباء جعلهُ مَجزُوماً ،   ومن رفعها ،   جعله مرفوعاً على الاستئناف ،   أي : وهو يلعبُ ،   ومن غاير بين الفعلين ،   فقرأ بالباء من تحت في  "  يَلْعَب  "  دون  "  نَرْتَع  "   ؛ فلأن رَعْياً ،   إذا أكلته فالارتعاء للمواشِي ،   وأضافوه إلى أنفسهم ؛ لأنه السَّبب ،   والمعنى : نرتع إبلنا ،   فنسبُوا الارتعاء والقيام بحفظ المال إلى أنفسهم ؛ لأنهم بالغون .  <br />
ومن قرأ :  "  نُرْتع  "  رُباعياً ،   جعل مفعُوله محذوفاً ،   أي : يَرْعى مَواشينا ،   ومن بناها للمفعول ،   فالوجه : أنه أضمر المفعُول الذي لم يُسمَّ فاعله ،   وهو ضمير الغدِ ،   والأصل :<br />
نُرْتَع فيه ،   ونُلعَبُ فيه ،   ثم اتسع فيه ؛ فحذف حرف الجرِّ ،   فتعدى إليه الفعل بنفسه ،   فصار نُرْتعه ونَلْعَبُه ،   فلما بناهُ للمفعول ،   قام الضمير المنصُوب مقام فاعله ،   فانقلب مرفوعاً فاستتر في رافعه ،   فهو في الاتِّساع كقوله : [ الطويل ]<br />
3059 ويَوْمٍ شَهِدْنَا سَلِمياً وعَامِراً *** .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .  <br />
ومن رفع الفعلين ،   جعلهما حالين ،   وتكون مقدَّرة ،   وأمَّا إثبات الياء في  "  نَرْتَعي  "  مع جزم  "  يَلْعَب  "  وهي قراءة قنبل ،   فقد تجرَّأ بعضُ النَّاس وردَّها .  <br />
وقال ابن عطيَّة : هي قراءة ضعيفةٌ لا تجوز إلا في الشِّعْر ،   وقيل : هي لغة من يجز بالحركة المقدَّرة ،   وأنشد :<br />
3060 ألَمْ يَأتِيكَ والأنْبَاءُ تَنْمِي *** .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .  <br />
وقد تقدَّمت هذه المسألة .  <br />
و  "  نَرْتَع  "  يحتمل أن يكون وزنه :  "  نَفْتَعِل من الرَّعْي وهي أكلُ المرعى ؛ كما تقدَّم ،   ويكون على حذف مضاف ،   أي : نرتع مواشينا ،   أو من المراعاة للشيء ؛ قال : [ الخفيف ]<br />
3061 تَرْتعِي السَّفحَ فالكَثِيبَ فَذا قَارِ *** فَروضَ القَطَا فَذاتَ الرِّئالِ<br />
ويحتمل أن يكون وزنه  "  نَفْعَل  "  من رَتَعَ يَرْتَع : إذا أقام في خصب وسعة ،   ومنهُ قول الغضبان بن القبعثرى :  "  القَيْدُ والرَّتعة وقِلَّة المَنعَة  "   ؛ وقال الشاعر : [ الوافر ]<br />
3062 أكُفْراً بَعْدَ ردِّ المَوْتِ عَنِّي *** وبَعدَ عَطائِكَ المِائة الرِّتاعَا<br />
قوله : { وَإِنَّا لَهُ لَحَافِظُونَ } جلمة حالية ،   والعاملُ فيها أحد شيئين : إمَّا الأمر ،   وإمَّا جوابه .  <br />
فإن قيل : هل يجوز أن تكون المسألة من الإعمال ؛ لأن كلاًّ من العاملين يصح تسلُّطه على الحال ؟ .  <br />
فالجواب : لا يجُوز ذلك ؛ لأنَّ الإمال يستلزم الإضمار ،   والحالُ لا تضمر ؛ لأنَّها لا تكون إلا نكرةً ،   أو مؤولةً بها .
