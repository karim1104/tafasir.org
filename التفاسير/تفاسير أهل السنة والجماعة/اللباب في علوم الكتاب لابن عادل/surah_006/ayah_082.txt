قوله :  "  الَّذِينَ آمَنُوا  "  هل هو من كلام إبراهيم ،   أو من كلام قومه ،   أو من كلام اللَّهِ تعالى ؟ ثلاثة أقوالٍ ،   وعليها يَتَرتَّبُ الإعرابُ .  <br />
فإن قلنا : إنها من كلام إبراهيم كانت جواباً عن السؤال في قوله  "  فأيُّ الفَريقَيْنِ  "  .  <br />
وكذا إن قلنا : إنها كلام قومه ،   وأنهم أجابوا بما هو حُجَّة عليهم كأن الموصول خبر مبتدأ محذوف ؛ أي : هم الذين آمنوا ،   وإن جعلناه من كلام الل‍َّهِ تعالى ،   وأنَّهُ أمَرَ نَبِيَّهُ بأن يجيب به السُّؤال المتقدم ،   فكذلك أيضاً .  <br />
وإن جعلناه لِمُجَرَّدِ الإخبار من الباري - تعالى – كان الموصول مبتدأ ،   وفي خبره أوجه :<br />
أحدها : أنه الجملة بعده ،   فإن  "  أولئك  "  ،   و "  أولئك  "  مبتدأ ثانٍ ،   و "  الأمن  "  مبتدأ ثالث ،   و "  لهم  "  خبره ،   والجملةُ خَبَرُ  "  أولئك  "  ،   و "  أولئك  "  وخبره خبر الأوَّلِ .  <br />
الثاني : أن يكون  "  أولئك  "  بَدَلاً أو عطف بَيَان ،   و "  لهم  "  خبر الموصول ،   و "  الأمن  "  فاعلٌ به لاعتماده .  <br />
الثالث : كذلك ،   إلا أنَّ  "  لهم  "  خبرٌ مقدَّم ،   و "  الأمن  "  مبتدأ مؤخر ،   والجُمْلَةُ خبر الموصُول .  <br />
الرابع : أن يكون  "  أولئك  "  مبتدأ ثانياً ،   و "  لهم  "  خبره ،   و "  الأمن  "  فاعل به ،   والجملةُ خبر الموصول .  <br />
الخامس : وإليه ذهب أبو جَعْفَرٍ النحاسُ ،   والحوفي أن  "  لهم الأمن  "  خبر الموصول ،   وأن  "  أولئك  "  فَاصِلَةٌ ،   وهو غريب ؛ لأن الفَصْلَ من شَأنِ الضمائر لا من شَأنِ أسماء الإشارة .  <br />
وأمَّا على قولنا بأن  "  الذين  "  خبر مبتدأ محذوف ،   فيكون  "  أولئك  "  مبتدأ فقط ،   وخبره الجملة بعده ،   أو الجار وَحْدَهُ ،   و "  الأمْن  "  فاعل به ،   والجملة الأولى على هذا مَنْصُوبةٌ بقولٍ مُضْمَرٍ ،   أي : قُلْ لهم الذين آمنوا إن كانت من كلام الخليل ،   أو قالوا هم الذين إن كانت من كلام قومه .  <br />
قوله :  "  وَلَمْ يَلْبِسُوا  "  يجوز فيه وجهان :<br />
أحدهما : أنها مَعْطُوفَةٌ على الصِّلةِ ،   فلا مَحلَّ لها حينئذٍ .  <br />
والثاني : أن تكون الواو للحال ،   الجملة بعدها في محلِّ نصبٍ على الحال ،   أي : آمنوا غير مُلْبسينَ بِظُلْم .  <br />
وهو كقوله تعالى : { أَنَّى يَكُونُ لِي غُلاَمٌ وَلَمْ يَمْسَسْنِي بَشَرٌ } [ مريم :20 ] ،   ولا يُلْتَفَتُ إلى قول ابن عصفور ،   حيث جعل وقوع الجُمْلَةِ المنفية حالاً قليلاً ،   ولا إلى قَوْلِ ابن خَرُوفٍ ،   حيث جعل الواو واجِبَة الدخول على هذه الجملة ،   وإن كان فيها ضَمِيرٌ يعود على الحالِ .  <br />
والجمهور{[14393]} على  "  يَلْبِسُوا  "  بفتح الياء بمعنى  "  يخلطونه  "  .  <br />
وقرأ{[14394]} عكرمةُ بضمها من الإلْبَاسِ .    "  وهُمْ مُهْتَدُونَ  "  يجوز اسْتِئْنَافُهَا وحاليتها .  <br />
فصل في تفسير الآية<br />
روى عَلْقَمَةُ عن عَبْدِ اللَّهِ قال : لما نزلت { الَّذِينَ آمَنُواْ وَلَمْ يَلْبِسُواْ إِيمَانَهُمْ بِظُلْمٍ } شقَّ ذلك على المُسلمينَ ،   فقالوا : يا رسول اللَّهِ ،   فأيُّنَا لا يَظْلِمُ نفسه ،   فقال : لَيْسَ ذلِكَ ،   إنَّمَا هُوَ الشِّرْكُ ،   ألَمْ تَسْمَعُوا إلى ما قال لُقْمان لابنه : { يَا بُنَيَّ لاَ تُشْرِكْ بِاللَّهِ إِنَّ الشِّرْكَ لَظُلْمٌ عَظِيمٌ }{[14395]} [ لقمان :13 ] .
