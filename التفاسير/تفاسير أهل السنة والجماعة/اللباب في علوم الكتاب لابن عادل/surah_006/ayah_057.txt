قوله : { إِنِّي عَلَى بَيِّنَةٍ مِّن رَّبِّي } أي : على بيانِ أو بَصِيرةٍ وبُرهانٍ من ربي .  <br />
قوله :  "  وكَذَّبْتُم به  "  في هذه الجملة وجهان :<br />
أحدهما : أنها مُسْتَأنَفَةُ سِيقَتْ للإخبارِ بذلك .  <br />
والثاني : أنها في مَحَل نصبٍ على الحالِ ،   وحينئذٍ هل يحتاج إلى إضمار  "  قد  "  أم لا ؟<br />
و "  الهاء  "  في  "  به  "  يجوز أن تعود على  "  ربِّي  "  ،   وهو الظاهر .  <br />
وقيل : على القرآن ؛ لأنه كالمذكور .  <br />
وقيل : على اسْتِعْجَالهِمْ بالعذاب ؛ لأنهم كانوا يقولون : { إِن كَانَ هَذَا هُوَ الْحَقَّ مِنْ عِندِكَ فَأَمْطِرْ عَلَيْنَا حِجَارَةً } [ الأنفال :32 ] .  <br />
وقيل : على بيِّنةٍ ؛ لأنها في معنى البيانِ .  <br />
وقيل : لأن  "  التاء  "  فيها للمُبالغةِ ،   والمعنى على أمرٍ بيِّنٍ من ربي .  <br />
و "  مِنْ ربِّي  "  في محلِّ جَرِّ صِفَةً ل  "  بيِّنَةٍ  "  .  <br />
قوله :  "  ما عِنْدي مَا تَسْتَعْجِلُون بِهِ  "  كان عليه الصَّلاةُ والسَّلامُ يخوِّفهم نزول العذاب ،   فقال تعالى : قال يا محمَّد : ما عندي ما تَسْتَعْجِلُونَ به ،   يعني قولهم : { إِن كَانَ هَذَا هُوَ الْحَقَّ مِنْ عِندِكَ } الآية .  <br />
وقيل : أراد به القِيامَةَ ؛ لقوله تعالى : { يَسْتَعْجِلُ بِهَا الَّذِينَ لاَ يُؤْمِنُونَ بِهَا } [ الشورى :18 ] .  <br />
قوله : { إِنِ الْحُكْمُ إِلاَّ لِلَّهِ } أي : في تأخير عذابهم .  <br />
قوله :  "  يَقُصُّ الحقَّ  "  قرأ نافع ،   وابن كثير{[14055]} ،   وعاصم  "  يَقُصُّ  "  [ بصاد مهملة مشددة ]{[14056]} مرفوعة ،   وهي قراءة ابن عبَّاسٍ ،   والباقون بضادٍ{[14057]} معجمة مخففة مكسورة ،   وهاتان في المتواتر .  <br />
وقرأ عبد الله ،   وأبَيٌّ ،   ويحيى{[14058]} بن وثَّابٍ ،   والنخعي ،   والأعمش ،   وطلحة :  "  يَقْضِي بالحقِّ  "  من القضاءِ .  <br />
وقرأ سعيد بن جُبَيْرٍ{[14059]} ،   ومجاهد :  "  يقضي بالحقِّ وهو خير القاضين  "  .   فأمَّا قراءة  "  يقضي  "  فَمِنَ القضاء .  <br />
ويؤيده قوله :  "  وهُوَ خَيْرُ الفَاصِلِينَ  "  فإن الفَصْلَ يناسب القضاء ،   ولم يُرْسَمْ إلاَّ بضاد ،   كأن  "  الباء  "  حذفت خطَّاً كما حذفت لَفْظاً لالتقاء الساكنين ،   كما حُذِفَتْ من نحو : { فَمَا تُغْنِ النُّذُر } [ القمر :5 ] .  <br />
وكما حذفت  "  الواو "  في { سَنَدْعُ الزَّبَانِيَةَ } [ العلق :18 ] ،   { وَيَمْحُ اللَّهُ الْبَاطِلَ } [ الشورى :24 ] كما تقد‍َّم .  <br />
وأمَّا قراءةُ نَصْبِ  "  الحقّ  "  بعدهُ ،   ففيه أربعة أوجه :<br />
أحدها : أنه مَنْصُوبٌ على أنه صِفَةٌ لمصدر مَحْذُوفٍ ،   أي : يقضي القضاء الحقّ .  <br />
والثاني : أنه ضمَّن  "  يقضي  "  معنى  "  ينفذ  "  ،   فلذلك عدَّاهُ إلى المفعول به .  <br />
الثالث : أن  "  قضى  "  بمعنى  "  صَنَع  "  فيتعدَّى بنفسه من غير تَضْمينٍ ،   ويدُلُّ على ذلك قول الهُذَلِيّ شِعْراً : [ الكامل ]<br />
وَعَليْهِمَا مَسْرُودتانِ قَضَاهُمَا *** دَاوُدُ أوْ صَنَعُ السَّوابِغِ تُبَّعُ{[14060]}<br />
[ أي : صنعهما ]{[14061]} داود .  <br />
الرابع : أنه على إسْقَاطِ حَرْفِ الجرِّ ،   أي : يقضي بالحق ،   فلما حذف انْتَصَبَ مَجْرُورُهُ على حَدِّ قوله : [ الوافر ]<br />
تَمُرُّونَ الدِّيَار وَلَمْ تَعُوجُوا *** .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .  {[14062]}<br />
ويُؤيِّد ذلك القراءة بها الأصل .  <br />
وأمَّا قراءةُ  "  يَقُصُّ  "  فمن  "  قَصَّ الحديثَ  "   ،   أو مِنْ  "  قَصَّ الأثَرَ  "  أي : تتبَّعه .  <br />
قال تعالى : { نَحْنُ نَقُصُّ عَلَيْكَ أَحْسَنَ الْقَصَصِ } [ يوسف :3 ] .  <br />
ورجَّحَ أبُو عَمْرِو بْنُ العلاءِ القراءة الأولى بقوله :  "  الفَاصِلينَ  "  وحُكِيَ عنه أنه قال :  "  أهُوَ يَقُصُّ الحقَّ أوْ يَقْضِي الحقَّ  "  فقالوا :  "  يَقُصُّ  "  فقال : لو كان  "  يَقُصُّ  "  لقال :  "  وهو خير القاصِّين  "  أقَرَأ أحَدٌ بهذا ؟ وحيث قال : وهو خير الفاصلين فالفَصْلُ إنما يكون في القضاءِ .  <br />
وكأن أبا عمرو لم يبلغه  "  وهو خير القاصين  "  قراءة ،   وقد أجاب أبو علي الفارسي عما ذكره أبو العلاء ،   فقال :  "  القَصَصُ  "  هنا بمعنى القولِ ،   وقد جاء القول في الفَصْل أيضاً ،   قال تعالى<br />
{ إِنَّهُ لَقَوْلٌ فَصْلٌ } [ الطارق :13 ] .  <br />
وقال تعالى : { أُحْكِمَتْ آيَاتُهُ ثُمَّ فُصِّلَتْ } [ هود :1 ] .  <br />
وقال تعالى :  "  ونُفَصِّلُ الآياتِ  "  فقد حمل الفَصْلَ على القول ،   واستعمل معه كما جاء مع القضاءِ ،   فلا يلزم من الفاصل أن يكون معيناً ل  "  يقضي  "  .  <br />
فصل في الاحتجاج بالآية لأهل السُّنة<br />
أحتج أهل السُّنَّةِ بقوله : { إِنِ الْحُكْمُ إِلاَّ لِلَّهِ } على أنه لا يقدر العَبْدُ على أمر من الأمور إلاَّ إذا قَضَاهُ الله ،   فيمتنع منه فعلُ الكُفْرِ إلا إذا قضى اللَّهُ وحكم به ،   وكذلك في جميع الأفعال ؛ لأن قوله : { إِنِ الْحُكْمُ إِلاَّ للَّه } [ يفيد الحصر ]{[14063]} .  <br />
واحتج المعتزلة بقوله :  "  يقضي الحق  "  ،   ومعناه : أن كل ما قضى به فهو الحقّ ،   وهذا يقتضي ألاَّ يريد الكفر من الكافر ،   ولا المعصية من العاصي ؛ لأن ذلك ليس بحق{[14064]} ،   والله أعلم .
