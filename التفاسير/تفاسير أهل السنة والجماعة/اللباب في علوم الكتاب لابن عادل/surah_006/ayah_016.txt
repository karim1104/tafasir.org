"  مَنْ  "  شرطيةٌ ،   ومَحَلُّها يحتمل الرَّفْعَ والنصب ،   كما سيأتي بيانه .  <br />
وقرأ الأخوان{[13399]} ،   وأبو بكر عن عاصم :  "  يَصْرِف  "  بفتح الياء وكسر الراء على تسمية الفاعل .  <br />
والباقون{[13400]} بضمِّ الياء وفتح الراء على ما لم يُسَمَّ فاعله .  <br />
فأمَّا في القراءة الأولى ،   ف  "  مَنْ  "  فيها تَحْتَمِلُ الرفع والنصب ،   فالرفعُ من وجهٍ واحدٍ ،   وهو الابتداء ،   وخبرها فعل الشَّرطِ أو الجواب أو هما ،   على حَسَبِ الخلاف ،   وفي مفعول  "  يَصْرِف  "  حينئذ احتمالان :<br />
أحدهما : أنه مَذْكُورٌ وهو "  يومئذ  "  ،   ولا بُدَّ من حَذْفِ مُضَافٍ ،   أي : من يَصْرِفِ اللَّه عنه هَوْلَ يومئذ أو عذابَ يومئذ - فقد رحمه - فالضمير في  "  يَصْرِف  "  ،   يعود على اللَّهِ تعالى ،   ويدلُّ{[13401]} عليه قراءة أبَيِّ{[13402]} بن كعبٍ  "  مَنْ يَصْرِف اللَّهُ  "  بالتصريح به .  <br />
والضميران في  "  عنه  "  و "  رحمه  "  ل  "  مَنْ  "  .  <br />
والثاني : أنه محذوف لدلالةِ ما ذكر عليه قَبْلَ ذلك ،   أي : مَنْ يَصْرف اللَّهُ عنه العذاب  "  يومئذ  "  منصوب على الظرِفِ .  <br />
وقال مكيٌ{[13403]} :  "  ولا يَحْسُنُ أن تُقَدَّر هاء ؛ لأن الهاء إنما تُحْذَفُ من الصِّلاتِ  "  .  <br />
قتل شهابُ الدين{[13404]} : يعني أنه لا يُقَدَّر المَفْعُولُ ضميراً عائداً على عذاب يوم ؛ لأن الجملة الشرطية عنده صِفَةٌ ل  "  عَذَاب  "  ،   والعائِدُ منها محذوف ،   لكنَّ الحَذْفَ إنما يكون من الصِّلَةِ لا من الصِّفَةِ ،   وهذا معنى قول الواحديّ أيضاً{[13405]} ،   إلاَّ أنَّ قَوْلَ مَكي  "  إنما يُحْذَفُ من الصِّلاتِ  "  يريدُ في الأحسن ،   وإلاَّ فيحذف من الصِّفاتِ والأخبار والأحوال ،   ولكنَّه دون الصِّلة .  <br />
والنصبُ من وجهين :<br />
أحدهما : أنَّه مفعول مُقَدَّمٌ ل  "  يَصْرِف  "  والضمير في  "  عنه  "  على هذا يتعيَّنُ عودهُ على العذابِ المتقدمّ ،   والتقدير : أيَّ شخصٍ يصرفِ اللَّهُ عن العذاب .  <br />
والثاني : أنه مَنْصُوبٌ على الاشْتِغَالِ بفعلٍ مُضْمَرٍ لا يبرز ،   يفسره هذا الظَّاهِرُ من معناه لا من لَفْظِهِ ،   والتقدير : مَنْ نُكْرِمْ أو مَنْ نُنَجِّ يَصْرِف اللَّه .  <br />
والضمير في  "  عنه  "  للشرطية .  <br />
وأمَّا مفعول  "  يَصْرِفْ  "  على هذا فَيَحْتَمِلُ الوجهين المُتقدِّمينِ ،   أعني كونه مذكوراً ،   وهو "  يومئذٍ  "  على حَذْفِ مُضافٍ ،   أو محذوفاً اختصاراً .  <br />
وأمَّا القراءة الثَّانية ف  "  مَنْ  "  تحتمل وجهين :<br />
أحدهما : أنها في مَحَلّ رفع بالابتداء ،   وخبره ما بعده على ما تقدَّم والفاعل المَحْذُوفُ هو اللَّهُ -تعالى- يَدُلُّ عليه قراءةُ أبيّ المُتقدِّمةُ وفي القائم مقامه أربعة أوجه :<br />
أحدهما : أنه ضمير العذاب ،   والضمير في  "  عنه  "  يعود على  "  مَنْ  "  فقط ،   والظرف فيه حينئذ ثلاثة أوجه :<br />
أحدها : أنه منصوب ب  "  يصرف  "  .  <br />
والثاني : أنه منصوب بالعذاب ،   أي : الذي قام ضميره مقام الفاعل ،   قاله أبو البقاء{[13406]} - رضي الله عنه - .   ويلزم منه إعْمَالُ المصدر مضمراً ،   وقد يقال : يُغْتَفَرُ ذلك في الظروف .  <br />
الثالث : قال أبو البقاء{[13407]} :  "  إنه حال من الضمير  "  - يعني الضمير الذي قامَ مقامَ الفاعل ،   وجازَ وقوع الحال ظَرْفَ زمان ؛ لأنها في معنًى لا عن جُثّة .  <br />
الثاني من الأوجه الأربعة : أن القَائِمَ مقام الفاعل ضميره  "  مَنْ  "  والضمير في  "  عنه  "  يعُود على العذاب ،   والظَّرف منصوب ،   إمَّا ب  "  يُصْرف  "  وإمَّا على الحالِ من هاء  "  عنه  "  .  <br />
والثالث : من أوجه العامل في  "  يومئذٍ  "  متعذِّرٌ هنا وهو واضح ،   والتقديرُ : أي شخصٍ يُصْرف هو عن العذاب .  <br />
الثالث : أنَّ القائم مقام الفاعل  "  يومئذ  "  إمَّا على حذف مضاف أي : من يُصْرَف عنه فَزَعُ أو هَوْلُ يومئذ ،   وإمَّا على قيام الظرف دون مضاف ،   كقولك :  "  سير يوم الجمعة  "  ،   وإنما بُنِيَ  "  يومئذٍ  "  على الفَتْحِ لإضافته إلى غير مُتَمَكِّنٍ{[13408]} ،   ولو قُرِئَ بالرفع لكان جَائِزاً في الكلام ،   وقد قرئ : { وَمِنْ خِزْيِ يَوْمِئِذٍ } [ هود :66 ] فتحاً وجراً بالاعتبارين ،   وهما اعتبارانِ مُتَغَايِرَان .  <br />
فإن قيل : يلزمُ على عدم تقدير حَذْفِ المضاف إقامةُ الظَّرْفِ غير التام مقام الفاعل ،   وقد نصُّوا على أنَّ الظَّرْفِ المقطُوعَ عن الإضافة لا يُخبَرُ به ،   ولا يقوم مقام فاعل ،   ولو قلت :  "  ضُرب قبلُ  "  لم يَجُزْ ،   والظرف هنا في حكم المقطوع عن الإضافة فلا يجوز هنا قيامه مقام الفاعل ،   إلاَّ على حَذْفِ مضاف ،   فالجواب أن هذا في قُوَّة الظَّرْفِ المضاف ؛ إذ التنوين عِوَضٌ عنه ،   وهذا ينتهضُ على رَأي الجمهور أما الأخفش{[13409]} فلا ،   لأنَّ التنوين عنده تَنْوِينُ صَرْفٍ والكَسْرُ كَسْرُ إعراب .  <br />
والرابع : أنَّ القائم مقامَهُ  "  عنه  "  ،   والضميرُ في  "  عنه  "  يعودُ على  "  مَنْ  "  ،   و "  يومئذٍ  "  منصوب على الظَّرْفِ ،   والعامل فيه  "  يُصْرَفْ  "  ،   ولا يجوز الوجهان الأخيران ،   أعني نَصْبَهُ على الحالِ ،   لأن الضمير للجُثَّة والزَّمَان لا يقع حالاً عنهما ،   كما لا يَقَعُ خبراً ،   وأعني كونه مَعْمُولاً للعذاب ،   إذ ليس هو قائماً مقام الفاعلِ .  <br />
والثاني من وَجْهي{[13410]}  "  مَنْ  "  أنها في مَحَلِّ نصب بفعل مُضْمَرٍ يفسّره الظاهرُ بعده ،   وهذا إذا جعلنا  "  عنه  "  في مِحَلِّ نصب بأنْ يُجْعَلَ القائم مقامَ الفاعل : إمَّا ضميرَ العذاب ،   وإمَّا  "  يومئذ  "  .  <br />
والتقدير : مَنْ يكرم اللَّهُ ،   أو من يُنَجِّ يُصْرَفْ عنه العذابُ أو هولُ يومئذ ،   ونظيره :  "  زيدٌ به مُرُور حسن  "  ،   أقمت المصدر فبقي  "  عنه  "  منصوب المَحَلّ .  <br />
والتقدير : جاوزت زيداً مُرَّ به مُرُورُ حسن ،   وأمَّا إذا جُعل  "  عنه  "  قائماً مقام الفاعل تعيَّنَ رفعُه بالابتداء .  <br />
وأعلم أنه متى قلت : مَنْصُوبٌ على الاشتغال ،   فإنما يُقدَّر الفعل بعد  "  مَنْ  "  ؛ لأن لها صدر الكلام ،   ولذلك لم أظْهِره إلاَّ مؤخّراً ،   ولهذه العِلَّةِ منع بعضهم الاشتغال فيما له صَدْرُ الكلام كالاسْتِفهَامِ والشرط .  <br />
والتنوين في  "  يومئذٍ  "  عوضٌ عن جُمْلَةٍ مَحْذُوفةٍ تضمَّنها الكلام السَّابق .  <br />
والتقدير : يومئذٍ يكون الجزاء ،   وإنَّما قلنا ذلك ؛ لأنه لم يتقدَّم في الكلام جملةٌ مُصَرَّحٌ بها يكون التنوين عَوَضاً منها ،   وقد تقدَّم خلافُ الأخفش .  <br />
وهذه الجملة الشَّرطيَّةُ يجوز فيها وجهان : الاستئناف ،   والوصف ل  "  عذاب يوم  "  ،   فحيثُ جعلنا فيها ضميراً يعود على عذاب يوم ،   إمَّا مِنْ  "  يُصَرف  "  ،   وإمَّا مِنْ  "  عنه  "  جاز أن تكون صفةً وهو الظَّاهر ،   وأن تكون مُسْتأنفةً ،   وحَيْثُ لم نجعلْ فيها ضميراً يعود عليه - وقد عرفت كيفية ذلك - تعيَّنَ أن تكون مُسْتَأنَفَةً ،   ولا يجوز أن تكون صِفَةً لخلوِّها من الضمير .  <br />
وَرجَّح بعضهم إحْدى القراءَتَيْنِ على الأخرى ،   وذلك على عَادَتِهِمْ ،   فقال أبو عَلِيٍّ الفارسي : قراءة  "  يَصْرِف  "  يعني المبنيَّ للفاعل أحْسَنُ لمناسبة قوله :  "  رحمه  "  ،   يعني : أنَّ كُلاًّ منهما مَبْنيٌّ للفاعل ،   ولم يقل :  "  فقد رُحِمَ  "  واختارها أبو حَاتِم ،   وأبو عُبَيْد{[13411]} ،   ورجَّحَ بعضهم قراءة المبني للمفعول بإجماعهم على قراءة قوله : { لَيْسَ مَصْرُوفاً عَنْهُمْ } [ هود : 8 ] يعني في كونه أتى بصيغة اسم المَفْعُول المُسْنَدِ إلى ضمير العذابِ المذكور أوَّلاً .  <br />
ورجَّحَهَا محمد بن جرير{[13412]} بأنها أقَلُّ إضماراً ،   ومكي - رحمه الله - تَلَعْثَم في كلامه في ترجيحه لقراءة الأخوين ،   وأتى بأمثلةٍ فَاسِدَةٍ في كتاب  "  الهداية  "  له .  <br />
قال ابن عطية{[13413]} :  "  وقد تقدَّمَ أوَّلَ الكتاب{[13414]} عن ثَعْلبٍ وغيره من العلماء أنَّ ترجيح إحدى القراءاتِ المتواترة على الأخرى بحيث تُضَعَّفُ الأخرى لا يجوز  "  .  <br />
والجملة من قوله :  "  فقد رحمه  "  في محلّ جَزْمٍ على جواب الشرط والفاء واجبة .  <br />
قوله :  "  وذلِكَ الفَوزُ  "  مبتدأ وخبر جيء بهذه الجُمْلَةِ مقرِّرةً لما تقدَّم من مضمون الجملة قبلها ،   والإشارَةُ ب  "  ذلك  "  إلى المَصْدَرِ المفهوم من قوله :  "  يُصْرف  "  ،   أي : ذلك الصرف .  <br />
و "  المبين  "  يحتمل أن يكون مُتَعَدِّياً ،   فيكون المفعول مَحْذُوفاً ،   أي : المبين غيرَه ،   وأن يكون قاصراً بمعنى يبين ،   وقد تقدَّمَ أنَّ  "  أبان  "  ،   يكون قاصراً بمعنى  "  ظَهَرَ  "  ،   ومتعدّياً بمعنى  "  أظهر  "  .
