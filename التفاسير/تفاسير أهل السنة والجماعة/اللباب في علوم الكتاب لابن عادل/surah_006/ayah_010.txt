قرأ حمزة{[13287]} ،   وعاصمٌ ،   وأبو عمرو بكسر الدَّال على أصل الْتِقَاءِ السَّاكِنَيْنِ ،   والباقون{[13288]} بالضم على الإتباع ،   ولم يبالِ بالساكن ؛ لأنه حَاجِزٌ غير حصين وقد تَقَرَّرَتْ هذه القاعدة بدلائلها في سورة  "  البقرة  "  عند قوله تعالى : { فَمَنِ اضْطُرَّ } [ الآية :173 ] و "  برسلٍ  "  متعلّق ب  "  استهزئ  "  و "  مِنْ قبلك  "  صفة ل  "  رسل  "  ،   وتأويلهُ ما تقدَّم في وقوع  "  من قبل  "  صلة{[13289]} .  <br />
والمرادُ من الآية : التَّسْلية لِقَلْبِ الرسول صلى الله عليه وسلم أي : أن هذه الأنواع الكثيرة التي يعاملونك بها كانت موجودة في سائر القرون .  <br />
قوله :  "  فحاق بالذين سخِروا  "  ،   فاعل  "  حاق  "  :  "  ما كانوا  "  ،   و "  ما  "  يجوز أن تكون موصُولةً اسميةً ،   والعائد  "  الهاء  "  في  "  به  "  و "  به  "  يتعلَّق ب  "  يستهزئون  "  ،   و "  يستهزئون  "  خبر ل  "  كان  "  ،   و "  منهم  "  متعلّق ب  "  سخروا  "  على أنَّ الضمير يعود على الرُّسل ،   قال تبارك وتعالى : { إِن تَسْخَرُوا مِنَّا فَإِنَّا نَسْخَرُ مِنكُمْ } [ هود :38 ] .  <br />
ويجوز أن يتعدَّى بالباء نحو : سَخِرْت به ،   ويجوز أن يتعلّق  "  منهم  "  بمحذوفٍ على أنَّهُ حالٌ من فاعل  "  سَخِروا  "  والضمير في  "  منهم  "  يعود على الساخرين .  <br />
وقال أبو البقاء{[13290]} :  "  على المستهزئين  "  .  <br />
وقال : الحوفي{[13291]} :  "  على أمَمِ الرسل  "  .  <br />
وقد رَدَّ أبو حيَّان على الحوفي بأنه يَلْزَمُ إعادته على غير مذكور .  <br />
وجوابُهُ في قوة المذكور ،   وردَّ على أبي البقاء بأنه يصير المعنى : فحاق بالذين سَخِرُوا كائنين من المستهزئين ،   فلا حَاجَة إلى هذه الحال ؛ لأنها مفهومةٌ من قوله :  "  سخروا  "  وجوَّزوا أن تكون  "  ما  "  مصدريَّةً ،   ذكره أبو حيَّانَ ولم يتعرض{[13292]} للضمير في  "  به  "  .  <br />
والذي يظهر أنه يعود على الرسول الذي يَتَضَمَّنُهُ الجَمْعُ ،   فكأنه قيل : فَحَاقَ بهم عَاقِبَةُ استهزائهم بالرسول المُنْدَرجِ في جملة الرُّسُلِ ،   وأمَّا على رأى الأخْفَشِ ،   وابن السراج{[13293]} فتعود على  "  ما  "  المصدريّة ؛ لأنها اسم عندهما .  <br />
و "  حاق  "  ألفه مُنْقَلِبَةٌ عن  "  ياء  "  بدليل  "  يَحِيق  "  ،   ك  "  باع  "   "  يبيع  "  ،   والمصدر حَيْق وحُيُوق وحَيَقان كالغَلَيان والنَّزَوان .  <br />
وزعم بعضهم أنه من  "  الحَوْق  " {[13294]} ،   والمستدير بالشيء ،   وبعضهم أنه من  "  الحقّ  "  ،   فأبدلت إحدى القافين ياءً كَتَظَنَّنتُ ،   وهذان لَيْسَا بشيء .  <br />
أمَّا الأول فلاختلاف المَادَّةِ ،   إلاَّ أن يريدوا الاشتقاق{[13295]} الأكبر .  <br />
وأما الثاني : فلأنها دَعْوَى مُجَرَّدَةٌ من غير دليلٍ ،   ومعنى  "  حاق  "  أحاط .  <br />
وقيل : عاد عليه وبَالُ مَكْرهِ ،   قاله الفراء .   وقيل : دَارَ .  <br />
وقال الربيع بن أنس : نَزَلَ{[13296]} .  <br />
وقال عطاء : حَلَّ ،   والمعنى يدور على الإحاطة والشمول ،   ولا تستعمل إلا في الشر .  <br />
قال الشاعر : [ الطويل ]<br />
فأوْطَأ جُرْدَ الخَيْلِ عُقْرَ دِيَارِهِمْ *** وَحَاقَ بِهِمْ مِنْ بأسِ ضَبَّةَ حَائِقُ{[13297]}<br />
وقال الراغب{[13298]} :  "  قيل : وأصله : حَقَّ ،   فقلب نحو "  زَلَّ وزَالَ  "  وقد قرئ  "  فأزلهما وأزالَهُمَا  "  وعلى هذا ذَمَّهُ وذَامه  "  .  <br />
وقال الأزهري{[13299]} :  "  جعل أبو إسحاق  "  حاق  "  بمعنى  "  أحاط  "  ،   وكأنَّ مَأخَذَهُ من  "  الحَوْق  "  وهو ما اسْتَدَارَ بالكَمَرَة  " {[13300]} .  <br />
قال :  "  وجائز أن يكون الحَوْق فِعْلاً من  "  حاق يحيق  "  ،   كأنه في الأصل : حُيْق ،   فقلبت الياء واواً لانْضِمَامِ ما قلبها  "  .  <br />
وهل يحتاج إلى تقدير مضاف قبل  "  ما كانوا  "  ؟<br />
نقل الوَاحِدِيُّ عن أكثر المفسرين{[13301]} ذلك ،   أي : عقوبة ما كانوا ،   أو جَزَاء ما كانوا ،   ثم قال :  "  وهذا إذا جعلت  "  ما  "  عبارة عن القرآن والشريعة وما جاء به النَّبي صلى الله عليه وسلم ،   وإن جعلْتَ  "  ما  "  عبارة عن العذاب الذي كان صلى الله عليه وسلم يُوعدهم به إن لم يؤمنوا استَغْنَيْتَ عن تقدير المضاف ،   والمعنى : فَحَاقَ بهم العذابُ الذي يستهزئون به ،   وينكرونه  "  .  <br />
والسُّخْرِيَةُ : الاسْتِهْزَاءُ والتهَكُّمُ ؛ يقال : سَخِرَ منه وبه ،   ولا يُقَالُ إلاَّ اسْتِهْزَاءً به فلا يَتَعَدَّى ب  "  مِنْ  "  .  <br />
وقال الراغب{[13302]} :  "  سَخَرْتُهُ إذا سَخَّرْتَهُ للهُزْءِ منه ،   يقال : رجل سُخَرَة بفتح الخاء إذا كان يَسْخَرُ من غيره ،   وسُخْرَة بِسُكُونها إذا كان يُسْخَر منه ومثله :  "  ضُحَكة وضُحْكَة  "  ،   ولا ينقاس  "  .  <br />
وقوله : { فَاتَّخَذْتُمُوهُمْ سِخْرِيّاً } [ المؤمنون : 110 ] يحتمل أن يكون من التسخير ،   وأن يكون من السُّخْرية .  <br />
وقد قرئ سُخْرياً وسِخْرياً بضم السين وكسرها .  <br />
وسيأتي له مزيدُ بيان في موضعه إن شاء اللَّهُ تعالى .
