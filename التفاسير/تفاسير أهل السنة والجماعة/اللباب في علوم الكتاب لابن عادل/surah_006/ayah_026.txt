قوله : { وَهُمْ يَنْهَوْنَ عَنْهُ } في الضميرين- أعني  "  هم  "  وهاء  "  عنه  "  - أوجه :<br />
أحدها : أن المرفوع يعود على الكُفَّارِ ،   والمجرور يعود على القرآن الكريم ،   وهو أيضاً الذي عَادَ عليه الضَّميرُ المَنْصُوب من  "  يَفْقَهُوه  "  ،   والمُشَارُ إليه بقولهم :  "  إنْ هَذَا  "  .  <br />
والثاني : أنَّ  "  هم  "  يعود على من تَقدَّمَ ذكرهم من الكُفَّار ،   وفي  "  عنه  "  يعود على الرسول ،   وعلى هذا ففيه الْتِفَاتٌ من الخطاب إلى الغَيْبَةِ ،   فإن قوله : { جَاءُوَكَ يُجَادلونك } خطابٌ للرسول عليه الصَّلاةُ والسَّلام ،   فخرج من هذا الخطاب إلى الغَيْبَةِ .  <br />
وقيل : يعود المرفوع على أبي طالب وأتْبَاعِهِ .  <br />
وفي قوله :  "  يَنْهَوْنَ  "  و "  يَنْأوْنَ  "  تَجْنِيسُ التصريف ،   وهو عِبَارةٌ عن انفراد كل كلمة عن الأخرى بحرف ف  "  ينهون  "  انفردت بالهاء ،   و "  يَنْأوْن  "  بالهمزة ،   ومثله قوله تعالى : { وَهُمْ يَحْسَبُونَ أَنَّهُمْ يُحْسِنُونَ } [ الكهف :104 ] { بِمَا كُنتُمْ تَفْرَحُونَ فِي الأَرْضِ بِغَيْرِ الْحَقِّ وَبِمَا كُنتُمْ تَمْرَحُون } [ غافر :75 ] .  <br />
وقوله عليه الصلاة والسلام :  "  الخَيْلُ مَعْقُودٌ في نَوَاصِيْهَا الخَيْرُ  " {[13530]} ،   وبعضهم يسميه  "  تجنيس التَّحْرِيف  "  وهو الفرق بين كلمتين بحرف وأنشدوا في ذلك قول القائل : [ الكامل ]<br />
إنْ لَمْ أشُنَّ عَلَى ابْنِ حَرْبٍ غَارَةً *** لَمْ تَخْلُ يَوْماً مِنْ نِهابِ نُفُوسِ{[13531]}<br />
وذكر غيره أن  "  تجنيس التحريف  "  هو أن يكون الشَّكْلُ فرقاً بين كلمتين ،   وجعل منه  "  اللُّهَى تفتح اللَّهى  "  وقد تقدَّم تحقيقه .  <br />
وقرأ الحسن{[13532]}  "  ويَنَوْن  "  بإلقاء حركة الهمزة على النون وحذفها ،   وهو تخفيف قياسي .  <br />
و "  النَّأيُ  "  : البُعْدُ ،   قال : [ الطويل ]<br />
إذَا غَيَّرَ النَّأيُ المُحِبِّينَ لَمْ يَزَلْ *** رَسِيسُ الهَوَى مِنْ حُبِّ مَيَّةَ يَبْرَحُ{[13533]}<br />
وقال الآخر في ذلك ،   فأجَادَ ،   [ الطويل ]<br />
ألاَ حَبَّذا هِنْدٌ وأرْضٌ بِهَا هِنْدُ *** وهِنْدٌ أتَى مِنْ دُونِهَا النَّايُ والبُعْدُ{[13534]}<br />
عطف الشيء على نفسه للمُغَايَرَةَ اللَّفْظيَّة يقال : نَأى زيد يَنْأى نَأياً ،   ويتعدَّى بالهمزة ،   فيقال : أنْأيْتُهُ ،   ولا يُعَدَّى بالتضعيف ،   وكذا كل ما كان عينه همزةً .  <br />
ونقل الواحدي أنه يقال : نَأيْتهُ بمعنى نَأيْتُ عنهُ .  <br />
وأنشد المُبَرِّدُ : [ الطويل ]<br />
أعَاذِلُ إنْ يُصْبِحْ صَدَاي بِقَفْرَةٍ *** بَعيداً نآنِي صَاحِبِي وَقَريبِى{[13535]}<br />
أي : نَأى عَنِّي .  <br />
وحكى اللَّيْثُ :  "  نَأيْتُ الشيء  "  ،   أي : أبْعَدْتُهُ ،   وأنشد : [ الطويل ]<br />
إذَا مَا الْتَقَيْنَا سَالَ مِنْ عَبَرَاتِنَا *** شَآبيبُ يُنْأى سَيْلُهَا بالأصَابِع{[13536]}<br />
فَبَنَاهُ للمفعول ،   أي : يُنَحِّى ويُبْعَدُ .  <br />
والحاصلُ أنَّ هذه المادة تَدُلُّ على البُعْدِ ،   ومنه أتَنَأى أي : أفْعَلُ النَّأيَ .   والمَنْأى : الموضع البعيدُ .  <br />
قال النابغة : [ الطويل ]<br />
فَإنَّكَ كالمَوْتِ الَّذِي هُوَ مُدْرِكِي *** وَإنْ خِلْتُ أنَّ المُنْتَأى عَنْكَ وَاسِعُ{[13537]}<br />
و "  تَنَاءَى  "  أي : تباعد ،   ومنه النُّؤيُ للحُفَيْرَةِ التي حول الخِبَاءِ لتبعد عنه الماء .  <br />
وقُرِئ{[13538]} : { ونَاءَ بِجَانِبِهِ } [ فصلت :51 ] وهو مَقْلُوبٌ من  "  نأى  "  ،   ويَدُلُّ على ذلك أنَّ الأصل هو المَصْدَرُ وهو "  النَّأيُ  "  بتقديم الهمزة على حرف العِلَّة .  <br />
فصل في المراد بالآية وسبب نزولها<br />
معنى الآية الكريمة أنهم يَنْهَوْنَ النَّاسَ عن اتِّبَاع مُحَمَّدِ صلى الله عليه وسلم ويَنْأوْنَ عنه ،   أي : يتباعدون عنه بأنفسهم نزلت هذه الآية في كُفَّار  "  مكة  "  المشرفة ،   قال محمد بن الحَنَفِيَّةِ والس‍ُّدي والضَّحاك{[13539]} ،   وقال قتادةُ : يَنْهَوْنَ عن القرآن ،   وعن النَّبِيَّ صلى الله عليه وسلم ويتباعدون عنه{[13540]} .  <br />
واعلم أنَّ النهي عن النبي صلى الله عليه وسلم مُحَالٌ فلا بد أن يكون النهي عن فعل يتعلَّقُ به ،   فذكروا فيه قولين :<br />
الأول : ينهون عن تَدَبُّرِ القُرْآنِ واستماعه ،   وعن التَّصديقِ بنبوة مُحَمَّدٍ صلى الله عليه وسلم والإقرار برسالته .  <br />
الثاني : قال ابن عبَّاسِ - رضي الله عنه - ومقاتل : نزلت في أبي طالب كان ينهى النَّاس عن أذَى النبي صلى الله عليه وسلم ويمعنهم ويَنْأى عن الإيمان به أي : يَبْعَدُ ،   حتى روي أنه اجتمع إليه رؤوس المشركين ،   وقالوا : تَخَيَّرْ{[13541]} من أصْبَحِنَا وَجْهاً وادْفَعْ إلينا محمداً ،   فقال أبو طالب : ما أنْصَفْتُمُونِي أدْفَعُ إليكم ولدي لِتَقْتُلُوهُ وأرَبِّي وَلَدَكُمْ{[13542]} .  <br />
وروي أنَّ النبي صلى الله عليه وسلم دَعَاهُ إلى الإيمان فقال : لولا أن تُعِيِّرُني قُرَيْشٌ لأقررت بها عَينكَ ،   ولكن أذُبُّ عنك ما حييتُ{[13543]} ،   وقال فيه أبياتاً : [ الكامل ]<br />
واللَّهِ لَنْ يَصِلُوا إلَيْكَ بجَمْعِهِمْ *** حَتَّى أوَسَّدَ فِي التُّرابِ دَفِينَا<br />
فَاصْدَعْ بأمْرِكَ مَا عَلَيْكَ غَضَاضَةٌ *** وابْشِرْ وَقَرَّ بِذاكَ مِنْكَ عُيُونَا<br />
وَدَعَوْتَنِي وَعَرفْتُ أنَّكَ نَاصِحِي *** وَلَقَدْ صَدَقْتَ وَكُنْتَ ثَمَّ أمِينَا<br />
وَعرَضْتَ دِينا قَدْ عَلِمْتُ{[13544]} بأنَّهُ *** مِنْ خَيْرِ أدْيَانِ البَرِيَّةِ دِينَا<br />
لَوْلاَ الملامَةُ أو حذارُ مَسَبَّةٍ *** لَوَجَدْتَنِي سَمْحاً بِذَاكَ مُبِينَا{[13545]} {[13546]}<br />
واعلم أنَّ القول الأوَّلَ أشبه لوجهين :<br />
أحدهما : أنَّ جميع الآيات المتقدمة في ذَمِّ طريقتهم ،   فلذلك كان ينبغي أن يكون قولهم :  "  وهم ينهون عنه  "  مَحْمُولاً على أمْرٍ مذموم ،   وإذا حملناه على أنَّ أبا طالبٍ كان ينهى عن إيذائه لما حصل هذا النَّظْمُ{[13547]} .  <br />
وثانيهما : قوله تبارك وتعالى بعد ذلك : { وَإِن يُهْلِكُونَ إِلاَّ أَنفُسَهُم } يعني به ما تقدم ذكره ،   ولا يَليقُ ذلك النهي عن أذِيَّتِهِ ،   لأن ذلك حسن لا يوجب الهلاك{[13548]} .  <br />
فإن قيل : إنَّ قوله : { وَإِن يُهْلِكُونَ إِلاَّ أَنفُسَهُم } يرجع إلى قوله :  "  يَنْأوْنَ عَنْهُ  "  لا إلى قوله :  "  ينهون عنه  "  ؛ لأن المراد بذلك أنهم يبعدون عنه بمُفَارَقَةِ دينه وترك موافقته وذلك ذَمٌّ .  <br />
فالجوابُ أن ظاهر قوله : { وَإِن يُهْلِكُونَ إِلاَّ أَنفُسَهُمْ } يرجعُ إلى كل ما تقدَّمَ ذِكْرُهُ ،   كما يقال :  "  فلان يبعد عن الشيء الفلاني وينفر عنه ،   ولا يَضُرُّ بذلك إلاَّ نفسه  "  ،   فلا يكون{[13549]} هذا الضرر متعلّقاً بأحد الأمرين دون الآخر .  <br />
قوله :  "  وإن يهلكون  "   "  إنْ  "  نافية كالتي في قوله : { إِنْ هَذَا }<br />
[ الأنعام :25 ] و "  أنفسهم  "  مفعولٌ ،   وهو استثناءُ مُفَرَّغٌ ،   ومفعول  "  يَشْعرون  "  محذوف : إمَّا اقتصاراً ،   وإمَّا اختصاراً ،   أي : وما يشعرون أنهم يهلكون أنفسهم بتماديهم في الكُفْرِ وغُلُوِّهِمْ فيه ،   قاله ابن عباس{[13550]} .
