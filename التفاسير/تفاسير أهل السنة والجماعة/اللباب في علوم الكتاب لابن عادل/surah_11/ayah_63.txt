قوله : { أَرَأَيْتُمْ إِن كُنتُ على بَيِّنَةً } تقدَّم نظيره [ يونس : 50 ] ،   والمفعول الثَّاني هنا محذوف تقديره : أأعْصيه ويدل عليه  "  إنْ عَصَيْتُه  "  .   وقال ابنُ عطيَّة : هي مِنْ رؤيةِ القَلْبِ ،   والشَّرطُ الذي بعده وجوابه يَسُدُّ مسدَّ مفعولين ل  "  أرَأيْتُم  "  .  <br />
قال أبُو حيَّان{[18852]} :  "  والذي تقرَّر أنَّ  "  أرَأيْتَ  "  ضُمِّن معنى  "  أخْبِرْنِي  "  ،   وعلى تقدير أن لا يُضَمَّن ،   فجملةُ الشَّرط والجواب لا تسدُّ مسدَّ مفعولي  "  عَلِمْتُ  "  وأخواتها  "  .  <br />
قوله : { إِن كُنتُ على بَيِّنَةً مِّن رَّبِّي } ورد بحرف الشَّك ،   وكان على يقين تام في أمره إلاَّ أنَّ خطاب المخالف على هذا الوجه أقرب إلى القبُولِ ؛ فكأنه قال : قدِّرُوا أنِّي على بيِّنةٍ من ربِّي وأنِّي نبيٌّ على الحقيقةِ ،   وانظُرُوا إن تابعتكم ،   وعصيتُ أمر ربِّي ،   فمن يمنعني من عذابِ الله فما تزيدُونَنِي على هذا التقدير غير تَخْسِير .  <br />
قوله :  "  غَيْرَ تَخْسِير  "  الظاهرُ أنَّ  "  غَيْرَ  "  مفعولٌ ثانٍ ل  "  تَزِيدُونَنِي  "  .  <br />
قال أبُو البقاءِ{[18853]} :  "  الأقْوَى هنا أن تكُون  "  غير  "  استثناءً في المعنى ،   وهو مفعولٌ ثانٍ ل  "  تَزِيدُونَنِي  "  ،   أي : فمَا تَزِيدُونَنِي إلاَّ تَخْسِيراً .  <br />
ويجوز أن تكون  "  غير  "  صفةً لمفعولٍ محذوفٍ ،   أي شيئاً غير تخسير ،   وهو جيد في المعنى .  <br />
ومعنى التَّفْعِيل هنا النسبةُ ،   والمعنى : غير أن أخسركُم ،   أي : أنسبكم إلى التَّخْسير ،   قال الزمخشريُّ .  <br />
قال الحسنُ بن الفضلِ : لم يكنْ صالح في خسارةٍ حتى قال : { فَمَا تَزِيدُونَنِي غَيْرَ تَخْسِيرٍ } وإنَّما المعنى : فما تَزِيدُونَنِي بما تقُولُون إلا نسبتي إيَّاكم إلى الخسارةِ{[18854]} .  <br />
والتفسيقُ والتَّفْجِيرُ في اللغة : النسبة إلى الفِسْقِ والفُجُور ،   فكذلك التَّخْسيرُ هو النسبة إلى الخُسرانِ .  <br />
وقيل : هو على حذف مضافٍ ،   أي : غير بضارِّه تخسيركم ،   قالهُ ابن عبَّاسٍ{[18855]} .
