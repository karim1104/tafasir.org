قوله تعالى : { هُنَالِكَ تَبْلُواْ كُلُّ نَفْسٍ مَّآ أَسْلَفَتْ } الآية .  <br />
في  "  هُنالِكَ  "  وجهان :<br />
أظهرهما : بقاؤه على أصله ،   من دلالته على ظرف المكان ،   أي : في ذلك الموقف الدَّحض ،   والمكان الدَّهش .   وقيل : هو هنا ظرف زمان على سبيل الاستعارة ،   ومثله { هُنَالِكَ ابتلي المؤمنون } [ الأحزاب : 11 ] ،   أي : في ذلك الوقت ؛ وكقوله : [ الكامل ]<br />
وإذَا الأمُورُ تَعاظَمَتْ وتَشاكَلَتْ *** فهُناكَ يَعْترِفُون أيْنَ المفْزعُ{[18439]}<br />
وإذا أمكن بقاء الشيء على موضوعه ،   فهو أولى .  <br />
وقرأ الأخوان{[18440]}  "  تَتْلُو  "  بتاءين منقوطتين من فوق ،   أي : تطلب وتتبع ما أسلفته من أعمالها ،   ومن هذا قوله : [ الرجز ]<br />
إنَّ المُريبَ يتْبَعُ المُرِيبَا *** كَمَا رَأيْتَ الذِّيبَ يتلُو الذِّيبَا{[18441]}<br />
أي : يتبعه ويتطلَّبه ،   ويجوز أن يكون من التلاوة المتعارفة ،   أي : تقرأ كلُّ نفس ما عملته مسطَّراً في صحف الحفظة ،   لقوله - تعالى - : { يا ويلتنا مَا لهذا الكتاب لاَ يُغَادِرُ صَغِيرَةً وَلاَ كَبِيرَةً إِلاَّ أَحْصَاهَا } [ الكهف : 49 ] ،   وقوله : { وَنُخْرِجُ لَهُ يَوْمَ القيامة كِتَاباً يَلْقَاهُ مَنْشُوراً } [ الإسراء : 13 ] .  <br />
وقرأ الباقون :  "  تَبْلُوا  "  من البلاء ،   وهو الاختبار ،   أي : يعرف عملها : أخيرٌ هو أم شر ،   وقرأ عاصم في رواية  "  نَبلو  " {[18442]} بالنُّون والباء الموحَّدة ،   أي : نَخْتَبر نحنُ ،   و "  كُلَّ  "  منصُوب على المفعول به ،   وقوله  "  مَا أسْلفَتْ  "  على هذا القراءة يحتمل أن يكون في محلِّ نصبٍ ،   على إسقاطِ الخافض ،   أي : بما أسْلفَتْ ،   فلمَّا سقط الخافض ،   انتصب مَجْرُوره ؛ كقوله : [ الوافر ]<br />
تَمُرُّونَ الدِّيارَ فَلَمْ تَعُوجُوا *** كلامُكُمُ عليَّ إذنْ حَرَامُ{[18443]}<br />
ويحتمل أن يكون منصوباً على البدل من  "  كُلُّ نَفْسٍ  "  ويكون من بدل الاشتمال .   ويجُوزُ أن يكون  "  نَبْلُو  "  من البلاء ،   وهو العذاب .   أي : نُعَذبها بسبب ما أسلفت ،   و "  مَا  "  يجوز أن تكون موصولةً اسمية ،   أو حرفية ،   أو نكرة موصوفة ،   والعائدُ محذوفٌ على التقدير الأول ،   والآخر دون الثاني على المَشْهُور .  <br />
وقرأ ابنُ{[18444]} وثَّاب :  "  وَرِدُّوا  "  بكسر الرَّاء ،   تشبيهاً للعين المضعفة بالمعتلَّة ،   نحو :  "  قيل  "  و "  بيع  "  ،   ومثله : [ الطويل ]<br />
ومَا حِلَّ مِنْ جَهْلٍ حُبَا حُلمائِنَا  *** .   .   .  {[18445]}<br />
بكسر الحاء ،   وقد تقدَّم بيانُ ذلك [ البقرة : 11 ] .  <br />
قوله :  "  إِلَى الله  "  لا بُدَّ من مضاف ،   أي : إلى جزاءِ الله ،   أو موقف جزائه .  <br />
والجمهور على  "  الحقِّ  "  جرًّا ،   وقرئ{[18446]} منصوباً على أحد وجهين :<br />
إمَّا القطع ،   وأصله أنَّه تابعٌ ،   فقطع بإضمار  "  أمدح  "  ،   كقولهم : الحمدُ لله أهل الحمد .  <br />
وإمَّا أنَّه مصدرٌ مؤكِّدٌ لمضمونِ الجملة المتقدمة ،   وهو  "  رُدُّوا إلى الله  "  وإليه نحا الزمخشريُّ .   قال : كقولك :  "  هذا عبدُ الله الحَقُّ ،   لا الباطل  "  على التَّأكيد ؛ لقوله :  "  رُدُّوا إلى اللهِ  "  .  <br />
وقال مكِّي :  "  ويجوزُ نصبهُ على المصدر ،   ولم يُقرأ به  "  ،   وكأنَّه لم يطلع على هذه القراءة ،   وقوله :  "  مَّا كَانُواْ يَفْتَرُونَ  "   "  مَا  "  تحتمل الأوجُه الثلاثة .  <br />
المعنى :  "  هُنالِكَ  "  : في ذلك الوقت ،    "  نَبْلُو  "  أي : نختبر ،   والمعنى : يفعلُ بها فعل المختبر ،   وعلى القراءة الأخرى : أنَّ كلَّ نفس نختبر أعمالها ،   في ذلك الوقت .  <br />
{ وردوا إِلَى الله } أي : رُدُّوا إلى جزاءِ الله ،   قال ابن عبَّاسٍ :  "  مولاهُمُ الحقّ  "  أي : الذي يجازيهم بالحق ،   وقيل : جعلوا ملجئين إلى الإقرار بإلاهيته ،   { مَوْلاَهُمُ الحق } أي أعرضوا عن المولى الباطل ،   ورجعوا إلى المولى الحق ،   وقد تم تفسير { مولاهم الحق } [ الأنعام : 62 ] في الأنعام .    "  وضلَّ عَنْهُم  "  : زال وبطل ،    "  مَّا كانُوا يفْتَرون  "  أي : يعبدون ،   ويعتقدُون أنهم شفعاء ،   فإن قيل : قد قال : { وَأَنَّ الكافرين لاَ مولى لَهُمْ } [ محمد : 11 ] ،   قيل : المولى هناك هو الناصر ،   وههنا بمعنى الملك .
