قوله : { ثُمَّ نُنَجِّي رُسُلَنَا } قال الزمخشريُّ : هو معطوفٌ على كلامٍ محذوف يدلُّ عليه قوله : { إِلاَّ مِثْلَ أَيَّامِ الذين خَلَوْاْ مِن قَبْلِهِمْ } [ يونس : 102 ] كأنَّه قيل : نُهلكُ الأمم ثم نُنَجِّي رسلنا ،   معطوفٌ على حكايةِ الأحوالِ الماضية .  <br />
قرأ الكسائي{[18633]} في رواية  "  نصر  "  نُنْجِي "  خفيفة ،   والباقون : مشددة ،   وهما لغتان ،   وكذلك في قوله  "  نُنْجِ المُؤمنينَ  "  والمعنى : ننجي رسلنا ،   والذين آمنوا معهم عند نزول العذابِ .   معناه : نَجَّينَا ،   مستقبلٌ بمعنى الماضي ،   ونجَّيْنَا وأنْجَيْنَا بمعنى واحد  "  كذلِكَ  "  كما نَجَّيْناهم  "  حَقًّا  "  واجباً { عَلَيْنَا نُنجِ المؤمنين } .  <br />
قوله :  "  حقّاً  "  فيه أوجهٌ :<br />
أحدها : أن يكون منصوباً بفعل مُقَدَّر أي : حقَّ ذلك حقّاً .  <br />
والثاني : أن يكون بدلاً من المحذوف النَّائب عنه الكافُ تقديره : إنجاء مثل ذلك حقّاً .  <br />
والثالث : أن يكون  "  كذلك  "  و "  حقًّا  "  منصوبين ب  "  نُنْجِ  "  الذي بعدهما .  <br />
والرابع : أن يكون  "  كَذلِكَ  "  منصوباً ب  "  نُنَجي  "  الأولى ،   و حقّاً ب  "  نُنْجِ  "  الثَّانية .  <br />
وقال الزمخشري{[18634]} : مثل ذلك الإنجاء ننجي المؤمنين منكم ،   ونهلك المشركين ،   وحقّاً علينا اعتراضٌ ،   يعني حقَّ ذلك علينا حقّاً .  <br />
وقرأ الكسائيُّ{[18635]} وحفصٌ  "  نُنْجي المؤمنين  "  مخففاً من أنجى يقال : أنْجَى ونجَّى .  <br />
كأنزل ونزَّل ،   وجمهور القرَّاءِ لم ينقلوا الخلاف إلاَّ في هذا دون قوله : { فاليوم نُنَجِّيكَ بِبَدَنِكَ } [ يونس : 92 ] ودون قوله { ثُمَّ نُنَجِّي رُسُلَنَا } [ يونس : 103 ] .   وقد نقل أبو علي الأزهري الخلافَ فيهما أيضاً ،   ورسِمَ في المصاحف بجيم دون ياء .  <br />
فصل<br />
قال القاضي{[18636]} : قوله  "  حقًّا عليْنَا  "  المراد به الوجوب ؛ لأنَّ تخليصَ الرَّسُول - صلوات الله البر الرحيم وسلامه عليه - والمؤمنين من العذاب إلى الثَّواب واجبٌ ،   ولولاهُ ما حسن من الله تعالى أن يلزمهم الأفعال الشَّاقَّة ،   وهذا يجري مجرى قضاء الدَّين .  <br />
والجوابُ ،   بأن نقُول : إنَّه حقٌّ بحسب الوعْدِ والحُكْمِ ،   ولا نقُولُ إنَّهُ حقٌّ بحسب الاستحقاق لما ثبت أنَّ العبد لا يستحقُّ على خالقه شيئاً .
