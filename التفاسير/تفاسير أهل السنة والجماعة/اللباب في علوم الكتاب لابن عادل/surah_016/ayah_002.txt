قوله { يُنَزِّلُ الملاائكة } قد تقدم الخلاف في  "  يُنَزِّلُ  "  بالنسبة إلى التشديد والتخفيف في البقرة{[19682]} .  <br />
وقرأ زيد بن عليٍّ والأعمش وأبو بكر عن عاصم  "  تُنَزَّلُ  "  [ مشدداً ]{[19683]} مبنيًّا للمفعول وبالتاء من فوق .    "  المَلائِكَةُ  "  رفعاً لقيامه مقام الفاعل ،   وقرأ الجحدري{[19684]} : كذلك إلا أنه خفَّف الزَّاي .  <br />
وقرأ الحسن ،   والأعرج ،   وأبو العالية{[19685]} - رحمهم الله - عن عاصم بتاء واحدة من فوق ،   وتشديد الزاي مبنياً للفاعل ،   والأصل تتنزل بتاءين .  <br />
وقرأ ابن أبي{[19686]} عبلة :  "  نُنَزِّلُ  "  بنونين وتشديد الزَّاي  "  المَلائِكةَ  "  نصباً ،   وقتادة كذلك إلاَّ أنَّه بالتخفيف .  <br />
قال ابن عطية{[19687]} :  "  وفيهما شذوذٌ كبيرٌ  "  ولم يبين وجه ذلك .  <br />
ووجهه أنَّ ما قبله وما بعده ضمير غائبٌ ،   وتخريجه على الالتفات .  <br />
قوله :  "  بِالرُّوحِ  "  يجوز أن يكون متعلقاً بنفس الإنزالِ ،   وأن يكون متعلقاً بمحذوف على أنه حال من الملائكة ،   أي : ومعهم الروحُ .  <br />
قوله  "  مِنْ أمْرهِ  "  حال من الروح ،   و  "  مِنْ  "  إمَّا لبيانِ الجنس ،   وإما للتبعيض .  <br />
قوله  "  أنْ أنْذِرُوا  "  في  "  أنْ  "  ثلاثة أوجهٍ :<br />
أحدها : أنَّها المفسرة ؛ لأن الوحي فيه ضرب من [ القول ]{[19688]} ،   والإنزال بالروح عبارةٌ عن الوحي ؛ قال تعالى : { وَكَذَلِكَ أَوْحَيْنَآ إِلَيْكَ رُوحاً مِّنْ أَمْرِنَا } [ الشورى : 52 ] وقال : { يُلْقِي الروح مِنْ أَمْرِهِ على مَن يَشَآءُ مِنْ عِبَادِهِ } [ غافر : 15 ] .  <br />
الثاني : أنها المخففة من الثقيلة ،   واسمها ضمير الشأن محذوف ،   تقديره : أنَّ الشأن أقول لكم : أنه لا إله إلا أنا ،   قاله الزمخشري{[19689]} .  <br />
الثالث : أنها المصدرية التي من شأنها نصب المضارع ،   ووصلت بالأمر ؛ كقولهم : كتبت إليه بأن قُمْ ،   وتقدم البحث فيه .  <br />
فإن قلنا : إنَّها المفسرة فلا محلَّ لها ،   وإن قلنا : إنها المخففة ،   أو الناصبة ففي محلّها ثلاثة أوجهٍ :<br />
أحدها : أنَّها مجرورة المحل بدلاً من  "  الرُّوحِ  "  لأنَّ التوحيد روحٌ تحيا به النفوس .  <br />
الثاني : أنَّها في محل جرٍّ على إسقاطِ الخافض ؛ كما هو مذهب الخليل .  <br />
الثالث : أنَّها في محلِّ نصبٍ على إسقاطه ؛ وهو مذهب سيبويه{[19690]} .  <br />
والأصل : بأن أنذروا ؛ فلما حذف الجار جرى الخلاف المشهور .  <br />
قوله : { أَنَّهُ لاَ إله إِلاَّ أَنَاْ } هو مفعول الإنذار ،   والإنذار قد يكون بمعنى الإعلام ؛ يقال : أنْذرتهُ ،   وأنْذَرتهُ بكذا ،   أي : أعلموهم بالتوحيد .  <br />
وقوله :  "  فاتَّقُونِ  "  التفاتٌ إلى التكلم بعد الغيبة .  <br />
فصل<br />
وجه النَّظم : أنَّ الله - تعالى - لما أجاب الكفار عن شبهتهم ؛ تنزيهاً لنفسه - سبحانه وتعالى - عما يشركون ؛ فكأنَّ الكفار قالوا : هب أنَّ الله قضى على بعض عبيده بالشرّ ،   وعلى آخرين بالخير ،   ولكن كيف يمكنك أن تعرف هذه الأمور التي لا يعلمها إلا الله ؟ وكيف صرت بحيث تعرف أسرار الله وأحكامه في ملكه وملكوته ؟ .  <br />
فأجاب الله - تعالى - بقوله : { يُنَزِّلُ الملائكة بالروح مِنْ أَمْرِهِ على مَن يَشَآءُ مِنْ عِبَادِهِ أَنْ أنذروا أَنَّهُ لاَ إله إِلاَّ أَنَاْ فاتقون } وتقرير هذا الجواب : أنَّه - تعالى - ينزل الملائكة على من يشاء من عباده ،   ويأمر ذلك العبد أن يبلغ إلى سائر الخلق أن إله الخلق كلفهم بالتوحيد ،   وبالعبادة ،   وبين لهم أنَّهم إن فعلوا ،   فازوا بخيرِ الدنيا والآخرة ،   فهذا الطريق ضربٌ مخصوصٌ بهذه المعارف من دون سائر الخلق .  <br />
فصل<br />
روى عطاء عن ابن عباس - رضي الله عنهما - قال : يريد ب " المَلائِكة "  جبريل وحده{[19691]} .  <br />
وقال الواحديُّ : يسمَّى الواحد بالجمع ؛ إذا كان ذلك الواحد رئيساً مقدَّماً جائز ،   كقوله تعالى : { إِنَّآ أَرْسَلْنَا } [ القمر : 19 ] ،   و{ إنَّآ أَنزَلْنَا } [ النساء : 105 ] ،   و { إِنَّا نَحْنُ نَزَّلْنَا } [ الحجر : 9 ] .  <br />
والمراد بالروح الوحي كما تقدم ،   وقيل : المراد بالروح هنا النبوة ،   وقال قتادة رحمه الله تعالى : الرحمة{[19692]} ،   وقال أبو عبيدة : إنَّ الروح ههنا جبريل عليه السلام .  <br />
والباءُ في قوله  "  بِالرُّوحِ  "  بمعنى  "  مع  "  كقولهم :  " خَرجَ فلانٌ بِثيَابهِ "  أي : ومعه ثيابهُ .  <br />
والمعنى : نُنزِّلُ الملائكة مع الروح ؛ وهو جبريل ،   وتقرير هذا الوجه : أنَّه - تعالى - ما أنزل على محمدٍ - صلوات الله وسلامه عليه - جبريل وحدهُ في أكثر الأحوالِ ؛ بل كان يُنزِّلُ مع جبريل - عليه السلام - أقواماً من الملائكة ؛ كما في يوم بدرٍ ،   وفي كثيرٍ من الغزواتِ ،   وكان ينزل على رسول الله صلى الله عليه وسلم تارة ملك الجبال ،   وتارة ملك البحار ،   وتارة رضوان ،   وتارة غيرهم .  <br />
وقوله  "  مِنْ أمْرهِ  "  أي أنَّ ذلك النُّزُولَ لا يكون إلا بأمر الله ؛ كقوله : { وَمَا نَتَنَزَّلُ إِلاَّ بِأَمْرِ رَبِّكَ } [ مريم : 64 ] وقوله تعالى : { وَهُمْ بِأَمْرِهِ يَعْمَلُونَ } [ الأنبياء : 27 ] ،   وقوله : { وَيَفْعَلُونَ مَا يُؤْمَرُونَ } [ التحريم : 6 ] .  <br />
وقوله : { على مَن يَشَآءُ مِنْ عِبَادِهِ } يريد الأنبياء المخصوصين برسالته :  " أنْ أنْذِرُوا "  قال الزجاج :  "  أنْ  "  بدلٌ من  "  الرُّوحِ  "  .  <br />
والمعنى : ينزِّل الملائكة بأن أنذروا ،   أي : أعلموا الخلائق ،   أنَّه لا إله إلا أنا ،   والإنذار هو الإعلامُ مع التخويف .  <br />
 "  فاتَّقُون  "  فخافون .   يروى أن جبريل - صلوات الله وسلامه عليه - نزل على آدم - عليه الصلاة والسلام - اثنتي عشرة مرة ،   وعلى إدريس أربع مراتٍ ،   وعلى نوح - عليه الصلاة والسلام - خمسين مرَّة ،   وعلى إبراهيم اثنتين وأربعين مرة وعلى موسى أربع مرات ،   وعلى عيسى عشر مراتٍ ،   وعلى محمدٍ صلى الله عليه وسلم وعلى سائر الأنبياء أربعة وعشرين ألف مرَّةٍ .
