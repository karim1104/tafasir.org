قوله : { فَنَادَتْهُ الْمَلائِكَةُ } قرأ الأخوان{[9]}  " فَنَادَاهُ المَلاَئِكَةُ "  - من غير تأنيث - والباقون  " فَنَادَتْهُ "  بتاء التأنيث - باعتبار الجمع المُكَسَّر ،   فيجوز في الفعل المسند إليه التذكير باعتبار الجمع ،   والتأنيث باعتبار الجماعة ،   ولتأنيث لفظ  " الملائكة "  مع أن المذكور إذا تقدَّم فعلُهم - وهم جماعة - كان التأنيث فيه أحسن ؛ كقوله تعالى : { قَالَتِ الأَعْرَابُ } [ الحجرات : 14 ] .   ومثل هذا { إِذْ يَتَوَفَّى الَّذِينَ كَفَرُواْ الْمَلائِكَةُ } <br />
[ الأنفال : 50 ] تُقْرأ بالتاء والياء ،   وكذا قوله : { تَعْرُجُ الْمَلاَئِكَةُ } [ المعارج : 4 ] .   قال الزجاج : يلحقها التأنيث للفظ الجماعة ،   ويجوز أن يُعَبَّر عنها بلفظ التذكير ؛ لأنه - تعالى جمع الملائكة ،   وهكذا قوله : { وَقَالَ نِسْوَةٌ }[ يوسف : 30 ]وإنما حَسُنَ الحذفُ - هنا - للفصل بين الفعل وفاعله .  <br />
وقد تجرأ بعضُهم على قراءة العامة ،   فقال :  " أكره التأنيثَ ؛ لما فيه من موافقة دَعْوَى الجاهلية ؛ لأن الجاهلية زعمت أن الملائكة إناث "  .  <br />
روى إبراهيم قال : كان عبد الله بن مسعود يُذَكِّر الملائكةَ في كُلِّ القرآنِ .  <br />
قال أبو عُبَيْد :  " نراه اختار ذلك ؛ خلافاً على المشركين ؛ لأنهم قالوا : الملائكة بناتُ الله "  .  <br />
وروى الشعبيُّ أن ابن مسعود قال :  " إذا اختلفتم في الياء والتاء فاجعلوها ياءً "  .  <br />
وتجرأ أبو البقاء على قراءة الأخوين ،   فقال : وكره قوم قراءة التأنيث لموافقة الجاهلية ،   ولذلك قرأ  " فناداه "  بغير تاء - والقراءة غير جيِّدة ؛ لأن الملائكة جمع ،   وما اعتلوا ليس بشيءٍ ؛ لأن الإجماع على إثبات التاء في قوله : { وَإِذْ قَالَتِ الْمَلاَئِكَةُ يَمَرْيَمُ } [ آل عمران : 42 ] .  <br />
وهذان القولان - الصادران من أبي البقاء وغيره - ليسا بجيِّدَيْن ؛ لأنهما قراءتان متواترتان ،   فلا ينبغي أن ترد إحداهما ألبتة .  <br />
والأخوان على أصلهما من إمالة  " فَنَادَاهُ "  .   والرسم يحتمل القراءتين معاً - أعني : التذكير والتأنيث والجمهور على أن الملائكة المراد بهم واحد - وهو جبريل .  <br />
قال الزَّجَّاج : أتاه النداء من هذا الجنس الذين هم الملائكة ،   كقولك : فلان يركب السُّفُنَ - أي : هذا الجنس كقوله تعالى : { يُنَزِّلُ الْمَلائِكَةَ } <br />
[ النحل : 2 ] يعني جبريل  " بِالرُّوحِ "  يعني الوحي .   ومثله قوله : { الَّذِينَ قَالَ لَهُمُ النَّاسُ } [ آل عمران : 173 ] وهو نعيم بن مسعود ،   وقوله : { إِنَّ النَّاسَ قَدْ جَمَعُواْ لَكُمْ } [ آل عمران : 173 ] يعني أبا سفيان .   ولما كان جبريل - عليه السلام - رئيسَ الملائكة أخبر عنه إخبار الجماعة ؛ تعظيماً له .  <br />
قيل : الرئيس لا بدَّ له من أتباع ،   فلذلك أخبر عنه وعنهم ،   وإن كان النداء قد صدر منه - قاله الفضل بن سلمة - ويؤيد كون المنادي جبريل وحده قراءةُ عبد الله - وكذا في مصحفه - فناداه جبريل .  <br />
والعطف بالفاء - في قوله  " فَنَادَتْهُ "  - مُؤذِنٌ بأن الدعاء مُتَعقب بالتبشير .  <br />
والنداء : رفع الصوت ،   يقال : نادَى ندَاء - بضم النون وكسرها - والأكثر في الأصوات مجيئها على الضم ،   نحو البُكَاء ،   والصُّراخ ،   والدُّعاء ،   والرُّغاء .  <br />
وقيل : المكسور مصدر ،   والمضموم اسم .   ولو عُكِسَ هذا لكان أبْيَنَ ؛ لموافقته نظائره من المصادر .  <br />
قال يعقوب بن السكيت : إن ضمّيت نونه قصرته ،   وإن كسرتها مددته .  <br />
وأصل المادة يدل على الرفع ،   ومنه المنْتَدَى والنادي ؛ لاجتماع القوم فيهما وارتفاع أصواتهم .   وقالت قريش : دار الندوة ،   لارتفاع أصواتهم عند المشاورة والمحاورة فيها ،   وفلان أنْدَى صَوْتاً من فلان - أي : أرفع - هذا أصله في اللغة ،   وفي العرف : صار ذلك لأحسنها نَغَماً وصوتاً ،   والنَّدَى : المَطَر ،   ومنه : نَدِيَ ،   يَنْدَى ،   ويُعَبَّر به عن الجود ،   كما يُعَبَّر بالمطر والغيث عنه استعارةً .  <br />
قوله : { وَهُوَ قَائِمٌ } جملة حالية من مفعول النداء ،   و " يُصَلِّي "  يحتمل أوجهاً :<br />
أحدها : أن يكون خبراً ثانياً - عند مَنْ يرى تَعَدُّدَه مطلقاً - نحو : زيدٌ شاعرٌ فقيهٌ .  <br />
الثاني : أنه حال من مفعول النداء ،   وذلك - أيضاً - عند مَنْ يجوِّز تعدُّدَ الحال .  <br />
الثالث : أنه حال من الضمير المستتر في  " قَائِمٌ "  فيكون حالاً من حال .  <br />
الرابع : أن يكون صفة لِ  " قَائِمٌ "  .  <br />
قوله : { فِي الْمِحْرَابِ } متعلق ب  " يُصَلِّي "  ،   ويجوز أن يتعلق ب  " قَائِمٌ "  إذا جعلنا يُصَلِّي حالاً من الضمير في  " قَائِمٌ "  ؛ لأن العامل فيه - حينئذ - وفي الحال شيء واحد ،   فلا يلزم فيه فَصْل ،   أما إذا جعلناه خبراً ثانياً أو صفة لِ  " قَائِمٌ "  أو حالاً من المفعول لزم الفصلُ بين العاملِ ومعمولهِ بأجنبيٍّ .   هذا معنى كلام أبي حيّان .  <br />
قال شِهَابُ الدِّيْنِ{[10]} : والذي يظهر أنه يجوز أن تكون المسألة من باب التنازع ؛ فإن كُلاًّ من  " قَائِمٌ "  و " يصلِّي "  يصح أن يتسلَّط على  " فِي الْمِحْرَابِ "  وذلك على أي وجهٍ تقدم من وجوه الإعراب .  <br />
والمحراب - هنا - : المسجد .  <br />
قوله : { إِنَّ اللَّهَ } قرأ نافع وحمزة وابن عامر{[11]} بكسر  " إنَّ "  والباقون بفتحها ،   فالكسر عند الكوفيين ؛ لإجراء النداء مُجْرَى القولِ ،   فيُكْسر معه ،   وعند البصريين ،   على إضْمار القول - أي : فنادته ،   فقالت .   والفتح والحذف - على حذف حرف الجر ،   تقديره : فنادته بأن الله ،   فلما حُذِفَ الخافض جَرَى الوجهان المشهوران في مَحَلِّها .  <br />
وفي قراءة عبد الله{[12]} :  " فنادته الملائكة يا زكريا "  فقوله :  " يا زكريا "  هو مفعول النداء ،   وعلى هذه القراءة يتعين كسر  " إن "  ولا يجوز فتحُها ؛ لاستيفاء الفعلِ معموليه ،   وهما الضمير وما نُودي به زكريا .  <br />
قوله : { يُبَشِّرُكَ } قرأ نافع وابن كثير وأبو عمرو وابن عامر وعاصم{[13]} الخمسة في هذه السورة { أَنَّ اللَّهَ يُبَشِّرُكَ } - في موضعين - وفي سورة الإسراء : { وَيُبَشِّرُ الْمُؤْمِنِينَ } [ الإسراء : 9 ] وفي سورة الكهف : { وَيُبَشِّرُ الْمُؤْمِنِينَ } - بضم الياء ،   وفتح الباء ،   وكسر الشين مشددة - من بَشَّرَه ،   يُبَشِّرُه .  <br />
وقرأ نافع وابن عامر وعاصم - ثلاثتهم - كذلك في سورة الشورى ،   وهو قوله : { ذَلِكَ الَّذِي يُبَشِّرُ اللَّهُ عِبَادَهُ الَّذِينَ آمَنُواْ } [ الشورى : 23 ] .  <br />
وقرأ الجميع - دون حمزة{[14]} - كذلك في سورة براءة : { يُبَشِّرُهُمْ رَبُّهُم بِرَحْمَةٍ مِّنْهُ } [ التوبة : 21 ] وفي الحجر - في قوله : { إِنَّا نُبَشِّرُكَ بِغُلامٍ عَلِيمٍ } [ الحجر : 53 ] - ولا خلاف في الثاني - وهو قوله : { فَبِمَ تُبَشِّرُونَ } [ الحجر : 54 ] - أنه بالتثقيل .  <br />
وكذلك قرأ الجميع{[15]} - دون حمزة - في سورة مريم - في موضعين - { إِنَّا نُبَشِّرُكَ } [ مريم : 7 ] وقوله : { لِتُبَشِّرَ بِهِ الْمُتَّقِينَ }<br />
[ مريم : 97 ] .   وكل من لم يذكر من قرأ بالتقييد المذكور فإنه يقرأ بفتح حرف المضارعة ،   وسكون الياء وضم الشين .  <br />
وإذا أردت معرفة ضبط هذا الفَضل ،   فاعلم أن المواضع التي وقع فيها الخلاف المذكور تسع كلماتٍ ،   والقُرَّاء فيه على أربع مراتبٍ :<br />
فنافع وابن عامر وعاصم ثَقَّلُوا الجميعَ .  <br />
وحمزة خفّف الجميع إلا قوله : { فَبِمَ تُبَشِّرُونَ } [ الحجر : 54 ] .  <br />
وابن كثير وأبو عمرو ثقلا الجميعَ إلا التي في سورة الشورَى فإنهما وافقَا فيها حمزة .   والكسائي خفَّف خمساً منها ،   وثقَّل أربعاً ،   فخفَّفَ كلمتي هذه السورةِ ،   وكلمات الإسراء والكهفِ والشورَى .   وقد تقدم أن في هذا الفعل ثلاث لغاتٍ : بشَّر - بالتشديد - وبَشَرَ – بالتخفيف- .  <br />
وعليه ما أنشده الفراء قوله : [ الطويل ]<br />
بَشَرْتَ عِيَالِي إذْ رَأيْتَ صَحِيفَةً *** أتَتْكَ مِنَ الْحَجَّاجِ يُتْلَى كِتَابُهَا{[16]}<br />
الثالثة : أبْشَرَ - رباعياً - وعليه قراءة بعضهم  " يُبَشِّرُكَ "  - بضم الياء .  <br />
ومن التبشير قول الآخر : [ الكامل ]<br />
يَا بِشْرُ حُقَّ لِوَجْهِكَ التَّبْشِيرُ *** هَلاَّ غَضِبْتَ لَنَا وَأنْتَ أمِيْرُ ؟{[17]}<br />
وقد أجمع على مواضع من هذه اللغات نحو  " فَبَشِّرْهُمْ "  .  <br />
{ وَأَبْشِرُوا } [ فصلت : 30 ] { فَبَشَّرْنَاهَا بِإِسْحَاقَ } [ هود : 71 ] .   قالوا : { بَشَّرْنَاكَ بِالْحَقِّ } [ الحجر : 55 ] .   فلم يرد الخلاف إلا في المضارع دونَ الماضي .  <br />
وقد تقدم معنى البشارة واشتقاقها في سورة البقرة .  <br />
قوله : { بِيَحْيَى } متعلق ب { يُبَشِّرُكَ } ولا بد من حذف مضاف ،   أي : بولادة يحيى ؛ لأن الذوات ليست متعلقة للبشارة ،   ولا بد في الكلام من حذف معمول قاد إليه السياقُ ،   تقديره : بولادة يحيى منك ومن امرأتك ،   دلَّ على ذلك قرينةُ الحالِ وسياقُ الكلامِ .  <br />
و " يحيى "  فيه قولان :<br />
أحدهما - وهو المشهور عند المفسِّرين - : أنه منقول من الفعل المضارع وقد سَمُّوا بالأفعالِ كثيراً ،   نحو يعيش ويعمر ويموت .  <br />
قال قتادة :  " سُمِّي { يَحْيَى } لأن الله أحياه بالإيمان "  .  <br />
وقال الزَّجَّاج :  " حيي بالعلم "  وعلى هذا فهو ممنوع من الصرف للعلميَّة ووزن الفعل ،   نحو يزيد ويشكر وتغلب .  <br />
والثاني : أنه أعجميّ لا اشتقاق له - وهو الظاهر - فامتناعه للعلمية والعُجْمَة الشخصية .  <br />
وعلى كلا القولين يُجْمَع على  " يَحْيَوْنَ "  بحَذْف الألف وبقاء الفتحةِ تدلّ عليها .  <br />
وقال الكوفيون : إن كان عربيَّا منقولاً من الفعل فالأمر كذلك ،   وإن كان أعجمياً ضُمَّ ما قبل الواو ،   وكسر ما قبل الياء ؛ إجراءً له مُجْرَى المنقوص ،   نحو جاء القاضُون ،   ورأيت القاضِين ،   نقل هذا أبو حيّان عنهم .   ونقل ابنُ مالك عنهم أن الاسم إن كانت ألفه زائدةً ضُمَّ ما قبلَ الواو ،   وكُسِرَ ما قبلَ الياءِ ،   نحو : جاء حبلون ورأيت حُبلِين ،   وإن كانت أصليةً نحو دُجَوْن وجب فتح ما قبل الحرفين .  <br />
قالوا : فإن كان أعجمياً جاز الوجهان ؛ لاحتمال أن تكون ألفهُ أصليةً أو زائدة ؛ إذْ لا يُعْرَف له اشتقاق .   ويصغر يحيى على  " يُحَيَّى "  وأنشد للشيخ أبي عمرو بن الحاجب في ذلك : [ مجزوء الرمل ]<br />
أيُّها الْعَالِمُ بِالتَّصْرِ *** يفِ لا زِلْتَ تُحَيَّا<br />
قَالَ قَوْمٌ : إنَّ يَحْيَى *** إنْ يُصَغَّرْ فَيُحَيَّا<br />
وَأبَى قَوْمٌ فَقَالُوا *** لَيْسَ هَذَا الرَّأيُ حَيَّا<br />
إنَّمَا كَانَ صَوَاباً *** لَوْ أجَابُوا بِيُحَيَّا<br />
كَيْفَ قَدْ رَدُّوا يُحَيَّا *** وَالَّذِي اخْتَارُوا يُحَيَّا ؟<br />
أتُرَاهُمْ فِي ضَلالٍ *** أمْ تَرَى وَجْهاً يُحَيَّا ؟{[18]}<br />
وهذا جارٍ مَجْرَى الألْغاز في تصغير هذه اللفظة ،   وذلك يختلف بالتصريف والعمل ،   وهو أنه لما اجتمع في آخر الاسم المصَغَّر ثلاثُ ياءاتٍ جرى فيه خلافٌ بين النحاة بالنسبةِ إلى الحَذْف والإثبات ،   وأصل المسألة تصغير  " أحْوَى "  ويُنْسَب إلى  " يَحْيَى "   " يَحْيَى "  - بحذف الألف ،   تشبيهاً لها بالزائد - نحو حُبْلِيّ - في حُبْلَى - و " يَحْيَوِيّ "  - بالقلب ؛ لأنها أصل كألف مَلْهَوِيّ ،   أو شبيهة بالأصل إن كان أعجمياً - و " يَحْيَاوِيّ "  - بزيادة ألف قبل قَلْبِ ألفِهِ واواً .  <br />
وقرأ حمزة والكسائي  " يَحْيَى "  بالإمالة ؛ لأجل الياء والباقون بالتفخيم .  <br />
قال ابن عباس :  " سُمِّيَ "  يَحْيَى ؛ لأن اللهَ أحيا به عَقْرَ أمِّه {[19]} .  <br />
وقال قتادة : لأن الله أحيا قلبه بالإيمان {[20]} .  <br />
وقيل : لأن الله أحياه بالطاعة حتى إنه لم يَعْصِ اللهَ ،   ولم يَهِمّ بمعصيةٍ .  <br />
قال القرطبي :  " كان اسمه - في الكتاب الأول - حَيَا ،   وكان اسم سارة - زوجة إبراهيم - يسارة ،   وتفسيره بالعربية : لا تلد ،   فلما بُشِّرَت بإسحاق قيل لها : سارة ،   سمَّاها بذلك جبريل - عليه السلام - فقالت : يا إبراهيم ،   لم نقص من اسمي حرف ؟ فقال إبراهيم ذلك لجبريل - عليه السلام - فقال : إن ذلك حرف زيد في اسمِ ابنٍ لها من أفضل الأنبياء ،   اسمه حيا ،   فسُمِّي بيَحْيَى "  .  <br />
قوله : { مُصَدِّقًا } حال من  " يَحْيَى "  وهذه حال مقدرة .  <br />
وقال ابن عطية :  " هي حال مؤكدة بحسب حال هؤلاء الأنبياء عليهم الصلاة والسلام "  .   و " بِكلِمةٍ "  متعلق ب  " مُصَدِّقاً "  .  <br />
وقرأ أبو السّمال{[21]}  " بِكِلْمَةٍ "  - بكسر الكاف وسكون اللام - وهي لغة صحيحة ؛ وذلك أنه أتبع الفاء للعين في حركتها ،   فالتقى بذلك كسرتان ،   فحذف الثانيةَ ؛ لأجل الاستثقال .  <br />
فصل<br />
قيل : المراد بها الجمع ؛ إذ المقصود التوراة والإنجيل وغيرهما من كتب الله تعالى المُنَزَّلة فعبَّر عن الجمع ببعضه ،   ومثل هذا قول النبي صلى الله عليه وسلم :  " أصْدَقُ كَلِمَةٍ قَالَهَا شَاعِرٌ كَلِمَةُ لَبِيدٍ "  حيثُ قال : [ الطويل ]<br />
ألا كُلُّ شَيْءٍ مَا خَلاَ اللهَ بَاطِل *** .   .   .   {[22]}<br />
وذكر جابر - رضي الله عنه - الحُوَيْدِرَة فقال : لَعَنَ اللهُ كلمته - يعني قصيدته .  <br />
وقال الجمهور : الكلمة : هي عيسى عليه السلام .  <br />
قال السديُّ : لقيت أمُّ عيسى ،   أمَّ يَحيى - وهذه حامل بعيسى ،   وتلك حامل بيحيى - فقالت أم يحيى : أشعَرْتِ أني حُبْلَى ؟ : فقالت : مريم : وأنا - أيضاً - حُبْلَى ،   قالت امرأة زكريا : فإني وجدت ما في بطني يسجد لما في بطنك ،   فذلك قوله{[23]} : { مُصَدِّقاً بِكَلِمَةٍ مِّنَ اللَّهِ } .  <br />
قال القرطبيُّ :  " رُوِيَ أنها أحسَّت بجنينها يَخِرّ برأسه إلى ناحية بطن مريم "  .  <br />
وقال ابن عبّاسٍ : إن يحيى كان أكبر سِنًّا من عيسى بستةِ أشهر{[24]} .  <br />
وقيل : بثلاث سنين ،   وكان يحيى أول من آمن به وصدق بأنه كلمة الله وروحه .  <br />
وسمي عيسى عليه السلام كلمة .   قيل : لأنه خُلِقَ بكلمة من الله { كُنْ فَيَكُونُ } من غير واسطة أب فسمي لهذا كلمة - كما يسمى المخلوق خَلْقاً ،   والمقدور قُدْرة ،   والمرجُوُّ رجاءً ،   والمشتَهَى شهوةً - وهو باب مشهور في اللغة .  <br />
وقيل : هو بشارة اللهِ مريم بعيسى - بكلامه على لسان جبريل عليه السَّلام .  <br />
وقيل : لأنه تكلم في الطفوليَّة ،   وأتاه الكتاب في زَمَنِ الطفوليَّةِ ،   فلهذا كان بالغاً مبلغاً عظيماً ،   فسُمِّيَ كلمة كما يقال : فلان جود وإقبال - إذا كان كاملاً فيهما .  <br />
وقيل : لما وردت البشارةُ به في كتب الأنبياء قبله ،   فلما جاء قيل : هذا هو تلك الكلمة - كما إذا أخبر عن حدوث أمر ،   فإذا حَدَث ذلك الأمر ،   قال : قد جاء قولي ،   وجاء كلامي - أي : ما كنت أقول ،   وأتكلم به - ونظيره قوله تعالى : { وَكَذَلِكَ حَقَّتْ كَلِمَتُ رَبِّكَ عَلَى الَّذِينَ كَفَرُواْ أَنَّهُمْ أَصْحَابُ النَّارِ } [ غافر : 6 ] وقوله : { وَلَكِنْ حَقَّتْ كَلِمَةُ الْعَذَابِ عَلَى الْكَافِرِينَ } [ الزمر : 71 ] .  <br />
وقيل : لأن الإنسان قد يُسَمَّى ب  " فضل الله "  و " لطف الله "  وكذلك عيسى كان اسمه كلمة الله وروح الله .  <br />
واعلم أن كلمة الله - تعالى - كلامه ،   وكلامه - على قول أهل السنة - صفةٌ قديمةٌ قائمةٌ بذاته وفي قول المعتزلة : صفة يخلقها الله في جسم مخصوص ،   دالة بالوضع على معاني مخصوصة .  <br />
وضروريّ حاصل بأن الصفة القديمة ،   أو الأصوات التي هي أعراض غير باقية يستحيل أن يقال : إنها ذات عيسى ،   ولما كان ذلك باطلاً في بداهة العقول ،   لم يَبْقَ إلا التأويل .  <br />
قوله : { مِّنَ اللَّهِ } في محل جر ؛ صفة ل  " كَلِمَةٍ "  فيتعلق بمحذوف ،   أي : بكلمة كائنة من الله { وَسَيِّداً وَحَصُوراً وَنَبِيّاً } أحوال أيضاً - كمصَدِّقاً .   والسيد : فَيْعِل ،   والأصل سَيْود ،   ففُعِلَ به ما فعل ب  " ميت "  ،   كما تقدم ،   واشتقاقه من سَادَ ،   يَسُودُ ،   سِيَادَةً ،   وسُؤدُداً - أي فاق نظراءه في الشرف والسؤدد .  <br />
ومنه قوله : [ الرجز ]<br />
نَفْسٌ عِصَامٍ سَوَّدَتْ عِصَاما *** وَعَلَّمَتْهُ الْكَرَّ والإقْدَامَا<br />
وَصَيَّرَتْهُ بطلاً هُمَامَا *** .   .   .   {[25]}<br />
وجمعه على  " فَعَلَة "  شاذ قياساً ،   فصيح استعمالاً ؛ قال تعالى : { إِنَّآ أَطَعْنَا سَادَتَنَا وَكُبَرَآءَنَا } [ الأحزاب : 67 ] .  <br />
وقال بعضهم : سُمي سيِّداً ؛ لأنه يسود سَوَاد الناس أي : مُعْظَمهم وجُلَّهم .   والأصل سَوَدَة ،   و " فَعَلَة "  لِ  " فاعِل "  نحو كافِر وكفرة ،   وفاجِر وفَجَرَة ،   وبارّ وبررة .  <br />
وقال ابن عباس : السَّيِّد : الحليم {[26]} .  <br />
قال الجبائي : إنه كان سيداً للمؤمنين ،   ورئيساً لهم في الدين - أعني : في العلم والحلم والعبادة و الورع .  <br />
قال مجاهدٌ : السَّيِّد : الكريم على الله تعالى{[27]} .  <br />
وقال ابن المُسَيِّبِ : السيِّد : الفقيه العالم{[28]} .  <br />
وقال عكرمة : السيد : الذي لا يغلبه الغضبُ{[29]} .  <br />
وقيل : هو الرئيس الذي يتبع ،   ويُنتَهَى إلى قولهِ .  <br />
وقال المفضل : السيد في الدين .  <br />
وقال الضحاك : الحسن الخلق {[30]} .  <br />
وقال سعيد بن جبير : هو الذي يُطيع ربَّه .  <br />
ويقول عن الضَّحَّاكِ : السيد : التقِيّ{[31]} .  <br />
وقال سفيان : الذي لا يحسد .  <br />
وقيل : هو الذي يفوق قومَه في جميع خصال الخيرِ .  <br />
وقيل : هو القانع بما قسم الله له .  <br />
وقيل : هو السَّخِيّ .  <br />
قال رسول الله صلى الله عليه وسلم  " مَنْ سَيدُكُمْ يَا بَنِي سَلمةَ "  ؟ قالوا : جَد بن قَيْس على بُخْلِه ،   فقال :  " وأي دواء أدوى من البخل ،   لكن سَيِّدَكم عمرو بن الجموح{[32]} "  .   وفي الآية بذلك دليل على جواز تسمية الإنسان سيداً كما تجوز تسميته عزيزاً وكريماً .   وقال صلى الله عليه وسلم لبني قريظة :  " قوموا إلى سيِّدكم{[33]} .  <br />
وقال - في الحسن - :  " إن ابني هذا سَيِّدٌ ،   فلعلَّ اللهَ يُصْلِحُ بِهِ بَيْنَ فِئَتَيْن عَظِيْمَتين من المسلمين " {[34]} .  <br />
قال الكسائي : السيّد من المَعْز : [ الْمُسِّن ] .   وفي الحديث :  " الثَّنِيُّ من الضَّأن خير من السَّيِّد مِن الْمَعْزِ الْمُسِنّ " {[35]} .  <br />
وقال الشاعر : [ الطويل ]<br />
سَوَاءٌ عَلَيْهِ شَاةُ عَامٍ دَنَت لَهُ *** لِيَذْبَحَهَا للِضَّيْفِ أمْ شَاةُ سَيِّد{[36]}<br />
والحصور : فعول للمبالغة ،   مُحَوَّل من حاصر ،   كضروب .  <br />
وفي قوله : [ الطويل ]<br />
ضَرُوبٌ بِنَصْلِ السَّيْفِ سُوقَ سِمَانِهَا *** إذَا عَدِمُوا زاداً فَإنَّكَ حَاصِرُ{[37]}<br />
وقيل : بل هو فَعُول بمعنى : مفعول ،   أي : محصور ،   ومثله ركوب بمعنى : مركوب ،   وحلوب بمعنى : محلوب .  <br />
والحصور : الذي يكتم سره .  <br />
قال جرير : [ الكامل ]<br />
وَلَقَدْ تَسَقَّطَنِي الْوُشَاةُ فَصَادَفُوا *** حَصِراً بِسِرِّكِ يَا أمَيْمَ ضَنِينَا{[38]}<br />
وهو البخيل - أيضاً - قال : [ البسيط ]<br />
 .   .   .   *** لاَ بِالْحَصُورِ وَلاَ فِيهَا بِسَئَّارِ{[39]}<br />
وقد تقدم اشتقاق هذه المادة وهو مأخوذ من المنع ؛ وذلك لأن الحصور هو الذي لا يأتي النساء - إما لطبعه على ذلك ،   وإما لمغالبته نفسه - قال ابنُ مسعودٍ وابن عبَّاس وسعيد بن جبير وقتادة وعطاء والحسن : الحصور : الذي لا يأتي النساء ولا يقربُهُنَّ ،  {[40]} وهو - على هذا - بمعنى فاعل ،   يعني أنه يحصر نفسه عن الشهوات .  <br />
قال سعيد بن المُسيِّبِ هو العِنِّين الذي لا ماء{[41]} له ،   فيكون بمعنى  " مفعول "  كأنه ممنوع من النساء .  <br />
واختيار المحققين أنه الذي لا يأتي النساء لا للعجز بل للعفة والزهد - مثل الشروب والظلوم والغشوم - والمنع إنما يحصل إذا كان المقتضي قائماً ،   والدفع إنما يحصل عند قوة الداعية والرغبة والغِلْمَة .   والكلام إنما خرج مخرج الثناء وأيضاً فإنه أبعد من إلحاق الآفة بالأنبياء - والصفة التي ذكروها صفة نقص ،   وذكر صفة النقصان في معرِض المدحِ ،   لا يجوز ،   ولا يستحق به ثواباً ولا تعظيماً .  <br />
فصل<br />
احتجَّ بعضُهم - بهذه الآية - على أن ترك النكاح أفضل ؛ لأنه - تعالى - مدحه بترك النكاح ،   فيكون تركه أفضل في تلك الشريعة ،   فيجب أن يكون الأمر كذلك في شريعتنا ؛ للنص والمعقول أما النص فقوله تعالى :<br />
{ أُوْلَئِكَ الَّذِينَ هَدَى اللَّهُ فَبِهُدَاهُمُ اقْتَدِهْ } [ الأنعام : 90 ] .  <br />
وأمَّا المعقول فهو أن الأصل في الثابت بقاؤه على ما كان ،   والنسخ على خلاف الأصل .  <br />
وأجيبوا بأن هذا منسوخ بقوله –  " تَنَاكَحُوا تَنَاسَلُوا{[42]} "  وقوله :  " لا رَهْبَانِيَّةَ فِي الإسلامِ{[43]} "  .   وقوله عليه الصلاة والسلام : <br />
 " النِّكَاحُ سُنَّتِي وَسُنَّةُ الأنبِيَاءِ مِنْ قَبْلِي ،   فَمَنْ رَغِبَ عَن سُنَّتِي فَلَيْسَ مِنِّي " {[44]} .  <br />
وقولهم : النسخ على خلاف الأصل .  <br />
قلنا : مسلم إذا لم يُعْلَم الناسخُ ،   وقد علمناه .  <br />
قوله : { وَنَبِيًّا } اعلم أن السيادةَ إشارة إلى أمرين :<br />
أحدهما : القدرة على ضبط مصالح الخَلْق فيما يرجع إلى تعليمِ الدين .  <br />
والثاني : ضبط مصالحهم في تأديبهم ،   والأمر بالمعروف ،   والنهي عن المنكر .  <br />
والحصور إشارة إلى الزهد التام ،   فلما اجتمعا حصلت النبوة ؛ لأنه ليس بعدهما إلا النبوة .  <br />
قوله : { مِّنَ الصَّالِحِينَ } صفة لقوله : { وَنَبِيّاً } فهو في محل نصب ،   وفي معناه ثلاثة أوجهٍ :<br />
الأول : معناه من أولاد الصالحين .  <br />
الثاني : أنه خَيِّر - كما يقال للرجل الخَيِّر : إنه من الصالحين .  <br />
الثالث : أن صلاحه كان أتمَّ من صلاح سائر الأنبياء ؛ لقوله - عليه السلام - :  " مَا مِنْ نَبِيٍّ إلاَّ عَصَى وَهَمّ بِمَعْصِيَةٍ إلاَّ يحيَى بن زكريا ،   فإنه لم يَعْصِ ولم يَهِمَّ بمعصيةٍ "  .  <br />
فإن قيل : إن كان منصب النبوة أعلى من منصب الصلاح ،   فما الفائدة في ذكر منصب الصلاح بعد ذكر النبوة ؟<br />
فالجواب : أن سليمان - بعد حصول النبوة - قال : { وَأَدْخِلْنِي بِرَحْمَتِكَ فِي عِبَادِكَ الصَّالِحِينَ } [ النمل : 19 ]<br />
وتحقيقه أن للأنبياء قدراً من الصلاح لو انتقص لانتفت النبوة ،   فذلك القدر - بالنسبة إليهم - يجري مجرى حفظ الواجبات - بالنسبة إلينا - وبعد اشتراكهم في ذلك القدر تتفاوت درجاتُهم في الزيادة على ذلك القدر ،   فكلما كان أكثر نصيباً كان أعلى قَدْراً .  <br />
والله أعلم .
