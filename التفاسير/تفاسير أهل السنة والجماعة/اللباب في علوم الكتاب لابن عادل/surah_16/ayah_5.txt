قوله تعالى : { والأنعام خَلَقَهَا لَكُمْ } الآية هذه الدلالة الثالثة ؛ لأنَّ أشرف الأجساد الموجودة في العالم السفليِّ بعد الإنسان سائرُ الحيوانات لاختصاصها بالقوى الشريفة ،   وهي الحواسُّ الظاهرة والباطنة والشهوةُ والغضب .  <br />
قوله : { والأنعام خَلَقَهَا } العامة على النصب ،   وفيه وجهان :<br />
أحدهما : أنه نصب على الاشتغال وهو أرجح من الرفع لتقدم جملة فعليَّة .  <br />
والثاني : أنه نصب على عطفه على  "  الإنْسانَ  "  ،   قاله الزمخشريُّ ،   وابن عطيَّة فيكون  "  خَلقَهَا  "  على هذا مؤكداً ،   وعلى الأول مفسراً .  <br />
وقرئ{[19696]} شاذًّا  "  والأنْعَامُ  "  رفعاً وهي مرجوحةٌ .  <br />
قوله : { لَكُمْ فِيهَا دِفْءٌ } يجوز أن يتعلق  "  لكم  "  ب  "  خلقها  "  ،   أي : لأجلكم ولمنافعكم ،   ويكون  "  فيها  "  خبراً مقدماً ،   و  "  دفء  "  مبتدأ مؤخرٌ ،   ويجوز أن يكون  "  لَكُمْ  "  هو الخبر ،   أو يكون  " فيها "  هو الخبر ،   و " لكم "  متعلق بما تعلَّق به الخبر ،   أو يكون  " فيها "  حالاً من  " دفءٌ "  لأنه لو تأخر لكان صفة له ،   أو يكون  " فيها "  هو الخبر ،   و " لكم "  متعلق بما تعلق به ،   أو يكون حالاً من  "  دفء  "  قاله أبو البقاء{[19697]} .  <br />
وردَّه أبو حيَّان{[19698]} : بأنَّه إذا كان العاملُ في الحال معنويًّا ،   فلا يتقدم على الجملة بأسرها ،   ولا يجوز  "  قَائماً في الدَّارِ زيْدٌ  "  فإن تأخَّرت نحو  "  زَيْدٌ في الدَّارِ قَائِماً  "  جاز بلا خلافٍ ،   أو توسَّطت بخلاف أجازه الأخفش ومنعه غيره .  <br />
ولقائل أن يقول : لما تقدم العامل فيها ،   وهي معه جاز تقديمها عليه بحالها إلا أن نقول : لا يلزم من تقديمها وهو متأخر تقديمها عليه وهو متقدم لزيادة الفتح .  <br />
وقال أبو البقاء أيضاً :  "  ويجوز أن يرتفع  "  دِفْءٌ  "  ب  "  لَكُمْ  "  أو ب  " فِيهَا "  والجملة كلُّها حالٌ من الضمير المنصوب  "  .  <br />
قال أبو حيان  "  ولا يسمَّى جملة ،   لأنَّ التقدير : خلقها كائنٌ لكم فيها دفءٌ ،   أو خلقها لكم كائناً فيها دفءٌ  "  .  <br />
قال شهابُ الدِّين :  "  قد تقدم الخلاف في تقدير متعلق الجار إذا وقع حالاً أو صفة أو خبراً ،   هل يقدر فعلاً أو اسماً ،   ولعلَّ أبا البقاءِ نحا إلى الأول فتسميته له جملة صحيحٌ على هذا  "  .  <br />
والدِّفءُ : اسم لما يدفأ به ،   أي : يسخنُ .  <br />
قال الأصمعيُّ : ويكون الدفءُ السخونة ،   يقال : اقعد في دفء هذا الحائط ،   أي : في كنفه ،   وجمعه أدفَاء ،   ودَفِئَ يومنا فهو دَفيءٌ ،   ودَفِئَ الرَّجُل يَدْفأ فهو دَفْآنُ ،   وهي دَفْأى ،   كَسَكْران ،   وسَكْرَى .  <br />
والمُدفِّئَةُ بالتخفيف والتشديد ،   الإبل الكثيرة الوبر الكثيرة الوبر والشَّحم ،   وقيل : الدِّفْءُ : نِتاجُ الإبل وألبَانُهَا وما ينتفعُ به منها .  <br />
وقرأ زيد بن علي :  "  دِفٌ  "  بنقل حركة الهمزة إلى الفاءِ ،   والزهريُّ : كذلك إلاَّ أنَّه شدَّد الفاء ،   كأنَّه أجرى الوصل مجرى الوقف ،   نحو قولهم : هذا فرخٌّ بالتشديد وقفاً .  <br />
وقال صاحب اللَّوامحِ :  "  ومنهم من يعوض من الهمزة فيشدِّد الفاء وهو أحد وجهي حمزة بن حبيبٍ وقفاً  "  .  <br />
قال شهابُ الدِّين{[19699]} : والتشديد وقفاً : لغة مستقلة وإن لم يكن ثمَّ حذف من الكلمة الموقوف عليها .  <br />
قوله  "  ومَنافِعُ  "  أراد النَّسْل ،   والدَّرَّ ،   والركوب ،   والحملَ ،   وغيرها ،   فعبر عن هذا الوصف بالمنفعة ؛ لأنَّه الأعمُّ ،   والدر والنسل قد ينتفع به بالبيع بالنقودِ ،   وقد ينتفع به بأن تبدَّل بالثياب ،   وسائر الضَّرورياتِ ،   فعبَّر عن جملة الأقسامِ بلفظ المنافع ليعمَّ الكل .  <br />

الحيوانات قسمان :<br />
منها ما ينتفع به الإنسان ،   ومنها ما لا يكون كذلك ،   والقسم المنتفع به [ أفضل ]{[19700]} من الثاني ،   والمنتفع به إمَّا أن ينتفع به الإنسان في ضروراته ،   مثل الأكلِ واللبسِ أو في غير ضروراته ،   والأول أشرف وهو الأنعام ،   فلهذا يدأ بذكره فقال : { والأنعام خَلَقَهَا لَكُمْ } وهي عبارة عن الأزواج الثمانية ،   وهي الضَّأنُ والمعز والبقر والإبل .  <br />
قال الواحديُّ : تمَّ الكلام عند قوله : { والأنعام خَلَقَهَا } ثم ابتدأ وقال : { لَكُمْ فِيهَا دِفْءٌ } .  <br />
