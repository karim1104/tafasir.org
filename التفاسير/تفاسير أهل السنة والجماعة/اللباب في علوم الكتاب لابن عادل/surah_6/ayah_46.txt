{ أَرَأَيْتُمْ إِنْ أَخَذَ اللَّهُ } المفعول الأوَّل محذوف ،   تقديره : أرأيتم سَمْعَكُمْ وأبصاركم إن أخذها اللَّهُ ،   والجملة الاسْتِفْهَامِيَّةُ في موضع الثاني ،   وقد تقدَّم أن أبا حيَّان يجعله من التَّنازُعِ ،   وجوابُ الشرط مَحْذُوفٌ على نحو ما مَرَّ .  <br />
وقال الحوفي : وحَرْفُ الشَّرْطِ وما اتَّصَلَ به في موضع نَصْبٍ على الحالِ ،   والعاملُ في الحالِ  "  أرأيتم  "  كقولك :  "  اضْرِبْهُ إن خرج  "  أي : خارجاً ،   وجواب الشَّرْطِ ما تقدَّم مما دَخَلَتْ عليه هَمْزَةُ الاستفهام وهذا إعرابٌ لا يَظْهَرُ .  <br />
ولم يُؤتَ هنا ب  "  كاف  "  الخطاب ،   وأتِيَ به هناك ؛ لأن التَّهْديدَ هناك أعْظَمُ فَنَاسَبَ التأكيد بالإتيان ب  "  كاف  "  الخطاب ولمَّا لم يُؤتَ بالكافِ وجَبَ بروز علامةِ الجَمْعِ في التاء لئلا يَلْتَبِسَ ،   ولو جيء معها بالكافِ لاسْتُغْنِيَ بها كما تقدَّمَ ،   وتوْحِيدُ السَّمْعِ ،   وجَمْعُ الأبصارِ مفهومٌ مما تقدَّم في  "  البقرة  "  .  <br />
قوله :  "  مَنْ إلهٌ  "  مبتدأ وخبر ،   و "  مَنْ  "  استِفْهَامِيَّةٌ ،   و "  غيرُ الله  "  صِفَةٌ ل  "  إله  "  و "  يأتيكم  "  صِفَةٌ ثانية ،   و "  الهاء  "  في  "  به  "  تعود على  "  سمعكم  "  .  <br />
وقيل : تعود على الجميع ،   ووُحِّد ذهَاباً به مذهب اسم الإشارةِ .  <br />
وقيل : تعود على الهدى المدلول عليه بالمعنى .  <br />
وقيل : يَعُودُ على المَأخُوذِ والمختُومِ المدلول عليهما بالأخذ والخَتْمِ ،   والاستفهامُ هنا للإنكارِ .  <br />
والجمهور{[13907]} :  "  بِهِ انظر  "  بكسر الهاء على الأصل ،   وروى المُسَيَّبي عن نافعٍ  "  بهُ انظر  "  بضم الهاء [ وهي لغةُ من يقرأ  "  فخسفنا بهو وبدارهُو الأرض  "  فحذف  "  الواو "  لالتقاء الساكنين ،   فصارَ  "  به انظر  "  والباقون بكسرها{[13908]} .  <br />
وقرأ حمزة{[13909]} ،   والكسائي  "  يَصْدِفُونَ  "  بإشْمَامِ الزَّاي ،   والباقون بالصاد{[13910]} .  <br />
فصل في معنى الآية<br />
قال ابنُ عبَّاسٍ : المعنى : أيُّها المشركونُ إنْ أخَذَ الله ،   أي : أذهب وانْتَزَعَ سَمْعَكُمْ وأبْصَارَكُمْ ،   وختم على قلوبكم ،   أي : طبع على قلوبكم فلم تَعْقِلِ الهُدَى{[13911]} .  <br />
وقيل : معناه : أزال عقولكم حتى تصيروا كالمَجَانين .  <br />
وقيل : المُرَادُ من هذا الخَتْمِ الأمانة .  <br />
فصل في إثبات الصانع<br />
المرادُ من هذا الكلام الدِّلالةُ على وُجُودِ الصانع الحكيم المُخْتَار ؛ لأن أشْرَفَ أعضَاءِ الإنْسَانِ هو السَّمْعُ والبصر والقَلْبُ ،   والأذن محل القوة السَّامعة ،   والعَيْنُ مَحَلُّ القوة البَاصِرَةِ ،   والقَلْبُ مَحَلُّ الحياة والعِلْمِ والعَقْلِ ،   فلوْ زالَتْ هذهِ الصِّفَات عن هذه الأعضاء اخْتَلَّ أمْرُ الإنسان ،   وبَطَلَتْ مَصَالِحُهُ في الدنيا والدِّين .  <br />
ومن المعلوم بالضرورة أن القَادِرَ على تحصيل هذه القُوَى فيها ،   وصونها عن الآفَاتِ ليس إلا الله تعالى ،   وإذا كان الأمْرُ كذلك كان المُنَعِمُ بهذه النعم العظيمة هو الله سُبْحَانَهُ وتعالى .  <br />
قوله : { انْظُرْ كَيْفَ نُصَرِّفُ الآيَات } .  <br />
 "  كيف  "  مَعْمُول ل  "  نُصرِّفُ  "  ونَصبُها : إمَّا على التَّشْبيه بالحالِ ،   او التشبيه بالظَّرْفِ ،   وهي مُعْلِّقةٌ ل  "  انظر  "  فهي في مَحَلِّ نصب بإسْقَاطِ حرفِ الجرِّ ،   وهذا ظاهرٌ مما تقدَّم .  <br />
و "  نُصَرِّف  "  : نُبَيِّنُ ،   و "  يَصْدِفُون  "  معناه : يُعْرِضُونَ ،   يقال : صَدَف عن الشيء صَدفاً وصدوفاً وصدافيةً ،   وصادَفْتُهُ مُصادفَةً أي : لقِيتُهُ عن إعْراضٍ من جهتِهِ .  <br />
قال عِدِيُّ بْنُ الرقَاعِ : [ البسيط ]<br />
إذَا ذَكَرْنَ حَدِيثاً قُلْنَ أحْسَنَهُ *** وهُنَّ عَنْ كُلِّ سُوءٍ يُتَّقَى صُدُفُ{[13912]}<br />
 "  صُدُف  "  جمع  "  صَدُوف  "  ك  "  صُبُر  "  في جمع  "  صَبُور  "  .  <br />
وقيل : معنى  "  صَدَف  "  :  "  مال  "  ،   مأخوذة من الصَّدَفِ في البعيرِ ،   وهو أن يميل خُفُّهُ من اليد إلى الرِّجْلِ من الجانب الوَحْشِيّ .  <br />
و "  الصَّدَفُ  "  جمع  "  صَدَفَة  "  ،   وهي المحَارَةُ التي تكون فيها الدُّرَّة .  <br />
قال : [ البسيط ]<br />
وَزَادَهَا عَجَباً أنْ رُحْتُ فِي سُبُلٍ *** وَمَا دَرَتْ دَوَرانَ الدُّرِّ في الصَّدَفِ{[13913]}<br />
و "  الصَّدَفُ  "  و "  الصُّدُف  "  بفتح الصاد والدال وضمّهما ،   وضم الصاد وسكون الدال ناحية الجَبَلِ المُرْتفعِ ،   وسيأتي لهذا مزيدُ بيان .  <br />
فصل في دفع شبهة للمعتزلة<br />
قال الكعبي{[13914]} : دلَّت هذه الآية على أن اللَّه -تعالى- مَكَّنَهُمْ من الفَهْمِ ،   ولم يخلق فيهم الإعْرَاضَ والصَّدَّ ،   ولو كان تعالى هو الخَالِق للكفر فيهم لم يكن لهذا الكلامِ مَعْنَى .  <br />
واحْتَجَّ أهل السُّنَّةِ بعين هذه الآية قالوا : إنه -تعالى- بيَّن أنه بالغ في إظهار هذه الدلالة وفي تقريرها ،   وإزالة الشبهات عنها ،   ثم إنهم مع هذه المُبَالَغةِ القَاطِعَةِ للعُذْرِ ما زادوا إلاَّ تَمَادِياً في الكُفْرِ والعنادِ ،   وذلكَ يَدُلُّ على أن الهُدَى والإضْلالَ لا يحصلان إلاَّ بهداية الله -تعالى- وإضْلالِهِ ،   فَدَلالَةُ الآية على قولنا أقوى من دَلالتهَا على قولهم .
