قوله -تعالى- { حَمُولَةً وَفَرْشاً } مَنْصُوبان على أنَّهُما نُسِقَا على  "  جَنَّاتِ  "  أي : وأنشأ من الأنْعام حَمُولَة ،   و "  الأنْعَام  "  قيل : هي من الإبل خَاصَّة ،   وقيل : الإبل والبَقَر والغَنَم .  <br />
وقيل : ما أحَلَّه الله -تعالى- من الحيوانِ ؛ قاله أحمد بن يَحْيَى{[15403]} ،   قال القُرْطُبيُّ : وهذا أصَحُّهَا .  <br />
وقال القُرْطُبِي : فَعُولة بفتح الفَاءِ ،   إذَا كانت بمعنى الفَاعِل اسْتَوى فيها المُذَكَّر والمؤنَّث ؛ نحو قولك : رَجُل فَرُوقَة للجَبَان والخَائِفِ ،   ورجل صَرُورَة وامرأة صرورة إذا لم يَحُجَّا ؛ ولا جَمْع له فإذا كانت بِمَعْنَى المفعول ،   فرق بين المُذَكَّر والمؤنَّث بالهاء ،   كالحَلُوبة والرَّكُوبة ،   والحَمُولة بضم الحاءِ : أحْمَال وأما الحُمُول : بالضَّمِّ بغير هاء فهي الإبل الَّتِي عليها الهَوَادِجُ كان فيها نِساءٌ أو لم يَكُنّ ؛ قاله أبو زَيْد{[15404]} .  <br />
والحَمُولة : ما أطاق الحمل عَلَيْه من الإبل ،   والفَرْش : صِغَار هذا هو المشْهُور في اللُّغَة .  <br />
وقيل الحَمُولة : كبارُ الأنْعَام ،   أعني : الإبل والبَقَر والغَنَم ،   والفَرْشُ : صغارها قال :  "  ويدُلُ له أنَّهُ أبدل منه قوله بعد ذلك :  "  ثَمَانِيَة أزْوَاجٍ من الضَّأنِ  "  كما سيأتي لأنها دَانِية من الأرْض بسبب صغَر أجْرَامِها مثل الفَرْشِ وهي الأرْض المَفْرُوشُ عليها  "  .  <br />
وقال الزَّجَّاج{[15405]} : أجمع أهْل اللُّغَة على أنّ الفَرْشَ صِغَار الإبل ،   وأنشد القَائِل : [ الزجر ]<br />
أوْرَثَنِي حَمُولَةً وَفَرْشَا *** أمُشُّهَا فِي كُلِّ يَوْمٍ مَشَّا{[15406]}<br />
وقال الآخر : [ الرمل ]<br />
وَحَوَيْنَا الفَرْشَ مِنْ أنْعَامِكُمْ *** والحَمُولاتِ وَربَّاتِ الحِجَالْ{[15407]}<br />
قال أبو زَيْد :  "  يحتمل أن يكون سُمِّيَتْ بالمَصْدَر ؛ لأن الفَرْشَ في الأصْل مصدر  "  والفَرْش لفظ مُشْتَرك بين مَعَانٍ كثيرة : منها ما تقدَّم ،   ومنها : مَتَاع البَيْت والفَضَاء الوَاسِع ،   واتِّسَاع خُفِّ البَعير قليلاً ،   والأرْض الملساء ،   عن أبي عَمْرو بن العَلاء ،   ونباتٌ يلْتَصِق بالأرْضِ ،   ومنه قول الشاعر : [ الزجر ]<br />
كَمِشْفَرِ النَّابِ تلُوكُ الفَرْشَا{[15408]} *** .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .  <br />
وقيل الحَمُولة : كل ما حُمِل عليه من إبل وبَقَر وبَغْل وحِمَار .  <br />
والفَرْشُ هنا : ما اتُّخِذَ من صُوفه ووبَرِه وشَعْرِه ما يُفْتَرشُ ،   وأنشدوا للنَّابغة : [ الطويل ]<br />
وَحَلَّتْ بُيُوتِي فِي يَفَاعٍ مُمَنَّعٍ *** تَخَالُ بِهِ راعِي الحَمُولَةِ طَائِرا{[15409]}<br />
وقال عنترة : [ الكامل ]<br />
مَا رَاعَنِي إلاَّ حَمُولَةُ أهْلِهَا *** وَسْط الدِّيار تَسَفُّ حَبَّ الخِمْخِمِ{[15410]}<br />
قوله : { كُلُوا ممَّا رزَقُكم اللَّه } يريد : ما أحَلَّها لكم .  <br />
قالت المعتزلة{[15411]} : إنه -تبارك وتعالى- أمر بأكْل الرِّزْقِ ،   ومنع من أكل الحَرامٍ ينتج أن الرِّزْق ليس بَحَرَام .  <br />
ثم قال - تعالى- : { وَلاَ تَتَّبِعُواْ خُطُوَاتِ الشيطان } أي : التَّحلِيل والتَّحريم من عِنْد أنْفُسكُم ،   كما فعله أهْل الجَاهِليَّة ،   أي : لا تَسْلكُوا طرائق الشَّيْطَان وأبان  "  أنه لَكُم عَدُوٌّ مُبِينٌ  "  أي : بيِّن العداوة أخرج آدم من الجنَّة ،   وقوله : { لأَحْتَنِكَنَّ ذُرِّيَّتَهُ إَلاَّ قَلِيلاً } [ الإسراء : 62 ] .  <br />
قال الزجاج : في خُطُوات الشَّيْطَانِ ثلاثة أوجُه : ضم الطاء{[15412]} ،   وفتحها ،   وإسكَانها .
