"  إذا  "  منصوب بجوابه ،   أي : فقلْ ،   سلامٌ عليكم وَقْتَ مجيئهم أي : أوقع هذا القول كله في وقت مجيئهم إليك ،   وهذا معنى واضح .  <br />
وقال أبو البقاء{[14011]} :  "  والعاملُ في  "  إذا  "  معنى الجواب ،   أي : إذا جاءوك سَلِّمْ عليهم  "  ولا حَاجَة تدعو إلى ذلك مع فوات قوة المعنى ؛ لأن كونه يُبَلِّغُهُمُ السَّلام والإخبار بأنه كتب على نفسه الرَّحْمَةَ ،   وأنه من عَمِلَ سُوءاً بجَهَالَةٍ غفر له لا يقوم مقامه السَّلامُ فقط ،   وتقديره يفضي إلى ذلك .  <br />
وقوله :  "  سلامٌ  "  مبتدأ ،   وجاز الابتداء به وإن كان نَكِرَةً ؛ لأنه دُعَاءٌ ،   والدُّعَاءُ من المُسَوِّغَاتِ .  <br />
وقال أبو البقاء{[14012]} :  "  لما فيه من معنى الفِعْلِ  "  وهذا ليس من مذهب جمهور البصريين ،   وإنما هو شيء نُقِلَ عن الأخفش : أنه إذا كانت النكرة في معنى الفِعْل جاز الابتداء بها ورفعها الفاعل ،   وذلك نحو :  "  قائم أبواك  "  ونقل ابن مالك أن سيبويه{[14013]} أوْمَأ إلى جوازه ،   واستدلال الأخفش بقوله : [ الطويل ]<br />
خَبِيرٌ بَنُو لِهْبٍ فَلا تَكُ مُلْغِياً *** مَقَالَةَ لِهْبِيٍّ إذَا الطَّيْرُ مَرَّتِ{[14014]}<br />
ولا دليل فيه ؛ لأنَّ  "  فعيلاً  "  يقع بلفظ واحدٍ للمفرد وغيره ،   ف  "  خبير  "  خَبَرٌ مقدَّمٌ ،   واسْتَدَلَّ له أيضاً بقول الآخر : [ الوافر ]<br />
فَخَيْرٌ نَحْنُ عِنْدَ النَّاسِ مِنْكُمْ *** إذَا الدَّاعِي المُثَوِّبُ قَالَ : يَا لاَ{[14015]}<br />
ف  "  خير  "  مبتدأ ،   و "  نحن  "  [ فاعل ]{[14016]} سَدَّ مَسَدَّ الخبر .  <br />
فإن قيل : لِمَ لا يجوز أن يكون  "  خير  "  خبراً مُقدَّماً ،   و "  نحن  "  مبتدأ مؤخر ؟<br />
قيل : لئلا يَلْزَمُ الفَصْلُ بين  "  أفعل  "  و "  مِنْ  "  بأجنبي بخلاف جَعْلِه فاعلاً ،   فإن الفاعل كالخبر بخلاف المبتدأ .  <br />
و "  عليكم  "  خَبَرُهُ ،   و "  سلامٌ عليكم  "  أبلغ من  "  سَلاَماً عليكم  "  بالنصب ،   وقد تقررَّ هذا في أوَّلِ  "  الفاتحة  "  عند قراءة  "  الحَمْدُ  "  و "  الحَمْدَ  "  .  <br />
وقوله :  "  كَتَبَ رَبُّكُم  "  في مَحَلِّ نصب بالقولِ ،   لأنه كالتفسير لقوله :  "  سلامٌ عليكم  "  .  <br />
فصل في نزول الآية<br />
قال عكرمة : نزلت في الذين نَهَى اللَّهُ - عزَّ وجلَّ - نَبِيَّهُ صلى الله عليه وسلم عن طردهم ،   وكان النبي صلى الله عليه وسلم إذا رآهم بَدَأهُمْ بالسلام{[14017]} .  <br />
وقال عطاء : نزلت في أبي بكرٍ ،   وعُمَرَ ،   وعُثمانَ ،   وعلي ،   وبلال ،   وسالم ،   وأبي عُبَيْدةَ ،   ومُصْعَبِ بن عُمَيْرٍ ،   وحَمْزَة ،   وجعفر ،   وعُثْمانَ بْنِ مَظْعُون ،   وعمَّارِ بْنِ يَاسِرٍ ،   والأرقم بن أبي الأرقم وأبي سَلَمَة بْنِ عَبْدِ الأسَدِ{[14018]} .  <br />
قال ابن الخطيب{[14019]} :  "  وها هنا إشْكَالٌ ،   وهو أن النَّاسَ اتفقوا على أن هذه السُّورة نزلت دفعةً واحدةً ،   وإذا كان كذلك ،   فكيف يمكن أن يُقَالَ في كُلِّ واحدٍ من آيات هَذِهِ السُّورة : إن سبب نزول هذه الآية الأمْرُ الفلاني بِعَيْنِهِ ،   بل الأقْرَبُ أن تُحْمَلَ هذه الآية على عمومها ،   فكل من آمن باللِّهِ دخل تحت هذا التشريف  "  .  <br />
فصل فيما يطلق عليه لفظ  "  السلام  " <br />
قال المبرِّد{[14020]} : السَّلامُ في اللغة على أربعة أشياء :<br />
فمنها سلمت سلاماً ،   وهو معنى الدعاء .  <br />
ومنها أنه اسْمٌ من أسْمَاء اللَّهِ تعالى .  <br />
ومنها الإسْلام .  <br />
ومنها الشَّجَرُ العظيم أحْسَبُهُ مُسَمًّى بذلك لسلامتِهِ من الآفَاتِ .  <br />
ومنها أيضاً اسم للحِجَارَةِ الصَّلْبَةِ ،   وذلك أيضاً لسَلامتِهَا من الرَّخَاوَةِ .  <br />
ثم قال الزجَّاج{[14021]} :  "  سلام عليكم  "  هاهنا يحتمل أن يكون له تأويلان :<br />
أحدهما : أن يكون مَصْدر : سَلَّمت تسليماً وسلاماً ،   مثل  "  السَّراح  "  من  "  التَّسْرِيح  "  ،   ومعنى سلمت عليه سلاماً : دعوت بأن يَسْلَمَ من الآفات في دينِهِ ونَفْسِهِ ،   والسَّلامُ بمعنى التَّسْلِيم .  <br />
والثاني : أن يكون  "  السَّلامُ  "  جَمْعَ  "  السلامة  "  ،   فمعنى قولك : السَّلامُ عليكم : السَّلامةُ عليكم .  <br />
وقال ابن الأنباري{[14022]} : قال قومٌ : السلامُ هو الله تعالى ،   فمعنى السَّلامُ عليكم [ يعني الله عليكم ]{[14023]} أي : على حفظكم ،   وهذا بَعِيدٌ في هذه الآية لتنكير السَّلامِ ،   ولو كان مُعَرَّفاً لصحَّ هذا الوَجْهُ .  <br />
فصل في الكلام على  "  السلام  " <br />
قال قوم{[14024]} : إنَّه -تعالى- لمَّا أمَرَ الرسول - عليه الصلاة والسلام - بأن يقول لهم : { سلامٌ عَليْكُمْ كَتَبَ ربُّكُمْ على نَفْسِهِ الرَّحْمَة } كان هذا من قول الله فَيَدُلُّ على أنه -تعالى- قال لهم في الدُّنْيَا : سلامٌ عليكم كتب ربُّكم على نفسه الرحمة .  <br />
ومنهم من قال{[14025]} : بل هذا كلامِ الرَّسُولِ صلى الله عليه وسلم .  <br />
فصل في معنى  "  كتب  " <br />
كتب كذا [ على فلان ]{[14026]} يفيد الإيجاب ،   أي : بمعنى قَضَى ،   وكلمة  "  على  "  أيضاً تُفيدُ الإيجابَ ،   ومجموعهما مُبالغة في الإيجاب ،   وهذا يقتضي كونه -تعالى- راحماً لِعِبَادِهِ على سبيل الوُجُوبِ ،   واختلفوا في ذلك الوجوب ؟<br />
فقال أهْلُ{[14027]} السُّنَّةِ : له -سبحانه وتعالى- أن يتصرَّفَ في عبادِهِ كَيْفَ شَاءَ وأراد إلاَّ أنه أوجب الرَّحْمَة على نَفْسِهِ على سبيل الفَضْلِ والكرم .  <br />
وقالت المعتزلةُ{[14028]} : إنّ كونه عالماً بِقُبْحِ القَبَائِح ،   وعالماً بكونه غنيّاً عَنْهَا يمنعه من الإقْدامِ على القَبَائِحِ ،   ولو فعله كان ظَالِماً ،   والظُّلْمُ قَبِيحٌ ،   والقُبْحُ منه مُحَالٌ .  <br />
فصل في الدلالة في الآية<br />
دلَّتْ هذه الآية على جواز تسمية ذاتِ الله -تعالى- بالنفس ،   أيضاً قوله تعالى : { تَعْلَمُ مَا فِي نَفْسِي وَلاَ أَعْلَمُ مَا فِي نَفْسِكَ }<br />
[ المائدة :116 ] يَدُلُّ عليه ،   والنَّفْسُ هنا بمعنى الذَّاتِ والحقيقة ،   لا بمعنى الجِسْمِ ،   والدَّمِ ؛ لأنه -تعالى- مُقدَّسٌ عَنْهُ ؛ لأنه لو كان جِسْماً لكان مُرَكَّباً ،   والمُرَكَّب ممكن .  <br />
وأيضاً إنه أحَدٌ ،   والأحد لا يكون مُرَكَّباً ،   وما لا يكون مركباً لا يكون جسماً .  <br />
وأيضاً الأجْسَامُ متماثلةٌ في تمام الماهية ،   فلو كان جِسْماً لحصل له مِثْل ،   وذلك بَاطِلٌ ؛ لقوله تعالى : { لَيْسَ كَمِثْلِهِ شَيْءٌ } [ الشورى :11 ] .  <br />
فصل في دحض شبهة المعتزلة<br />
قالت المعتزلة{[14029]} :  "  كَتَبَ ربُكُمْ على نَفْسِهِ الرَّحْمَةَ  "  يُنَافِي كونه تعالى يخلق الكُفْرَ في الكَافِرِ ،   ثم يُعَذِّبُهُ عليه أبَد الآبَادِ ،   وينافي أن يقال : إنه يمنعه من الإيمان ،   ثم يأمره حال ذلك المَنْعِ بالإيمان ،   ثم يعذبه على ذلك .  <br />
وأجيب بأنه - تعالى- نَافِعٌ ضارُّ محيي مميت ،   فهو -تعالى- فعل تلك الرَّحْمَةَ البالغة ،   وفعل هذا القَهْرَ البالغ ولا مُنافَاة بين الأمرين .  <br />
قوله :  "  أنَّهُ ،   فأنَّهُ  "  قرأ ابن عامر{[14030]} ،   وعاصم بالفتح فيهما ،   وابن كثير وأبو عمرو ،   وحمزة{[14031]} ،   والكسائي بالكَسْرِ فيهما ،   ونافعٌ{[14032]} بفتح الأولى ،   وكسر الثانية ،   وهذه القراءاتُ الثلاثُ في المُتَواتِرِ ،   والأعرج بكسر{[14033]} الأولى وفتح الثانية عكس قراءة نافع ،   هذه رواية الزّهراوي عنه ،   وكذا الدَّاني .  <br />
وأمَّا سيبويه فروى قراءته كقراءة نافعٍ ،   فيحتمل أن يكون عنه رَوَايَتَانِ .  <br />
فأمَّا القرَاءةُ الأولَى فَفَتْحُ الأولَى فيها من أربعة أوجه :<br />
أحدها : أنها بدلٌ من  "  الرحمة  "  بدل شيء من شيء ،   والتقدير :  "  كتب على نفسه أنه من عمل  "  إلى آخره ،   فإنَّ نفس هذه الجمل المتضمنةِ للإخبار بذلك رَحْمَة .  <br />
والثاني : أنها في مَحَلِّ رَفْعٍ على أنها مبتدأ ،   والخبر محذوف ،   أي :  "  عليه أنه من عمل  "  إلى آخره .  <br />
والثالث : أنها [ فتحت ] على تقدير حَذْفِ حرف الجرَّ ،   والتقدير :  "  لأنه من عمل  "  ،   فلما حُذِفت  "  اللاَّمُ  "  جرى في مَحَلِّهَا الخلاف المشهور .  <br />
الرابع : أنها مَفْعُولٌ ب  "  كتب  "  ،   و "  الرحمة  "  مفعول من أجلِهِ ،   أي : أنه كتبَ أنَّهُ من عملَ لأجل رحمته إياكم .  <br />
قال أبو حيَّان{[14034]} : وينبغي ألاَّ يجوز ؛ لأنَّ فيه تَهْيِئَةَ العامل للعمل ،   وقطعه عنه .  <br />
وأمَّا فَتْحُ الثانية فمن خمسة أوجه :<br />
أحدها : أنها في مَحَلِّ رفع على أنها مبتدأ ،   والخبر محذوف ،   أي : فَغُفْرَانُهُ ورَحْمَتُهُ حاصلان أو كائنان ،   أو فعليه غفرانه ورحمته .  <br />
وقد أجمع القُرَّاءُ على فتح ما بعد  "  فاء  "  الجزاء في قوله :<br />
{ أَلَمْ يَعْلَمُواْ أَنَّهُ مَن يُحَادِدِ اللَّهَ وَرَسُولَهُ فَأَنَّ لَهُ نَارَ جَهَنَّمَ }<br />
[ التوبة :63 ] { كُتِبَ عَلَيْهِ أَنَّهُ مَن تَوَلاَّهُ فَأَنَّهُ يُضِلُّهُ } [ الحج :4 ] كما أجمعوا على كسرها في قوله : { وَمَن يَعْصِ اللَّهَ وَرَسُولَهُ فَإِنَّ لَهُ نَارَ جَهَنَّمَ } [ الجن :23 ] .  <br />
الثاني : أنها في محلِّ رفعٍ على أنها خبر مبتدأ محذوف ،   أي : فأمره أو شأنه أنه غفورٌ رحيم .  <br />
الثالث : أنها تكرير للأولى كُرِّرت لمَّا طال الكلامُ وعطفت عليها بالفاء ،   وهذا مَنْقُولٌ على أبي جَعْفَرٍ النحاس ،   وهذا وهمٌ فاحشٌ ؛ لأنه يَلْزَمُ منه أحدُ مَحْذُوريْنِ : إمَّا بقاءُ مبتدأ بلا خبر ،   أو شرطٍ بلا جواب .  <br />
وبيانُ ذلك أنَّ  "  مَنْ  "  في قوله :  "  أنه مَنْ عَمِلَ  "  لا تخلو : إمَّا أن تكون مَوْصُولَةً أو شرطية ،   وعلى كلا التقديرين ،   فهي في محلِّ رفعٍ بالابتداء ،   فلو جعلنا  "  أن  "  الثانية مَعْطُوفَةً على الأولى لَزِمَ عدمُ خبر المبتدأ ،   وجواب الشرط ،   وهو لا يجوز .  <br />
وقد ذكر هذا الاعتراض ،   وأجاب عنه الشيخ شهابُ الدين أبو شامة فقال :  "  ومنهم مَنْ جعل الثانية تكريراً للأولى لأجل طولِ الكلام على حَدِّ قوله : { أَيَعِدُكُمْ أَنَّكُمْ إِذَا مِتٌّمْ وَكُنتُمْ تُرَاباً وَعِظاماً أَنَّكُمْ مُّخْرَجُون } [ المؤمنون :35 ] ودخلت  "  الفاء  "  في  "  فأنه غفور  "  على حدِّ دخولها في { فَلاَ تَحْسَبَنَّهُمْ بِمَفَازَةٍ } [ آل عمران :188 ] على قول من جعلهُ تكريراً لقوله : { لاَ تَحْسَبَنَّ الَّذِينَ يَفْرَحُون } [ آل عمران :188 ] إلاَّ أن هذا ليس مثل  "  أيَعدكُمْ أنكم  "  ؛ لأن هذه لا شرط فيها ،   وهذه فيها شَرْطٌ ،   فيبقى بغير جواب .  <br />
فقيل : الجواب محذوف لدلالة الكلام عليه تقديره : غفر لهم  "  انتهى .  <br />
وفيه بُعْدٌ ،   وسيأتي هذا الجواب أيضاً في القراءة الثانية منقولاً عن أبي البقاءِ ،   وكان ينبغي أن يجيب به هنا ،   لكنه لم يفعل ولم يظهر فَرْقٌ في ذلك .  <br />
الرابع : أنها بدلٌ من الأولى ،   وهو قول الفرَّاء{[14035]} والزَّجَّاج{[14036]} وهذا مَرْدُودٌ بشيئين :<br />
أحدهما : أنَّ البدل لا يدخل فيه حَرْفُ عطفٍ ،   وهذا مقترن بحرف العطف ،   فامتنع أن يكون بدلاً .  <br />
فإن قيل : نجعل  "  الفاء  "  زائدة ،   فالجوابُ أن زيادتها غير زائدة ،   وهو شيء قال به الأخفش{[14037]} .  <br />
وعلى تقدير التَّسْليم فلا يجُوزُ ذلك من وَجْهٍ آخر ،   وهو خُلُوُّ المبتدأ ،   أو الشرط عن خبرٍ أو جواب .  <br />
والثاني من الشيئين : خُلُوُّ المبتدأ ،   أو الشرط عن الخبر ،   أو الجواب كما تقدَّم تقريره ،   فإن قيل : نجعل الجواب مَحْذُوفاً - كما تقدَّم نقلهُ عن أبي شامة - قيل : هذا بعيد عن الفَهْمِ .  <br />
الخامس : أنها مرفوعة بالفاعليَّةِ ،   تقديره :  "  فاسْتَقَرَّ أنَّهُ غفورٌ رحيمٌ  "  أي : اسْتَقَرَّ وثبت غُفْرَانُهُ ،   ويجوز أن يُقدَّر في هذا الوجه جَارّاً رافعاً لهذا الفاعل عند الأخْفَشِ تقديره : فعليه أنه غفورٌ ؛ لأنه يرفع به وإن لم يعتمد ،   وقد تقدَّم تحقيقه مِرَاراً .  <br />
وأمَّا القراءة الثانية : فكسر الأولى من ثلاثة أوجه :<br />
أحدها : أنها مُسْتَأنَفَةٌ ،   وأن الكلام تامُّ قبلها ،   وجيء بها وبما بعدها كالتَّفْسير لقوله : { كتبَ ربُّكُم على نَفْسهِ الرَّحْمَةَ } .  <br />
والثاني : أنَّها كُسِرت بعد قَوْلٍ مُقدَّرٍ ،   أي : قال الله ذلك ،   وهذا في المعنى كالذي قبله .  <br />
والثالث : أنه أجري  "  كتب  "  مُجْرَى  "  قال  "  ،   فَكُسِرَتْ بعده كما تُكْسَرُ بعد القَوْلِ الصريح ،   وهذا لا يَتَمَشَّى على أصول البصريين .  <br />
وأمَّا كَسْرُ الثانية فمن وجهين :<br />
أحدهما : أنها على الاسْتِئْنَافِ بمعنى أنها في صَدْرِ جملةٍ وقعتْ خبراً ل  "  من  "  الموصُولةِ ،   أو جواباً لها إن كانت شرطاً .  <br />
والثاني : أنها عُطِفَتْ على الأولى ،   وتكريرٌ لها ،   ويعترض على هذا بأنه يَلْزَمُ بقاءُ المبتدأ بلا خبرٍ ،   والشرط بلا جزاءٍ ،   كما تقدَّم ذلك في المفتُوحَتَيْنِ .  <br />
وأجاب أبو البقاء{[14038]} عن ذلك بأن خبر  "  من  "  محذوف دلَّ عليه الكلامُ ،   وقد تقدَّم أنه كان ينبغي أن يجيب بهذا الجواب في المفتوحتين عند من جعل الثانية تَكْرِيراً للأولى ،   أو بَدَلاً منها ثم قال : ويجوز أن يكون العائدُ مَحْذُوفاً ،   أي : فإنه غفورٌ له .  <br />
قال شهاب الدين{[14039]} : قوله :  "  ويجوز  "  ليس بجيِّدٍ ،   بل كان ينبغي أن يقول : ويجب ؛ لأنه لا بُدَّ من ضميرٍ عائدٍ على المبتدأ من الجملة الخبرية ،   أو ما يقوم مُقَامَهُ إن لم يكن نفس المبتدأ .  <br />
وأمَّا القراءةُ الثالثة : فيُؤخَذُ فتْحُ الأولى وكَسْرُ الثانية مما تقدَّم من كسرها وفتحها بما يليق من ذلك ،   وهو ظاهر .  <br />
وأمَّا القراءة الرابعة : فكذلك .  <br />
وقال أبو شامة :  "  وأجاز الزَّجَّاج{[14040]} كَسْرَ الأولى ،   وفَتْحَ الثانية ،   وإن لم يقرأ به  "  .  <br />
قال شهاب الدين{[14041]} : وقد قدَّمْتُ أنَّ هذه قراءة الأعرج وأن الزهراوي وأبا عمرو الدَّاني نقلاها عنه ،   فكأن الشَّيْخَ لم يَطَّلِعُ عليها .  <br />
وتقدَّم أن سيبويه لم يَرْو عن الأعْرَجِ إلاَّ كقراءة نافعٍ فهذا مما يصلح أن يكون عُذْراً للزَّجَّاج ،   وأمَّا أبو شَامَةَ فإنه مُتَأخرٌ ،   فعدم اطِّلاعِهِ عَجيبٌ .  <br />
و "  الهاء  "  في  "  أنه  "  ضمير الأمر والقِصَّةِ ،   و "  مَنْ  "  يجوز أن تكون شرطيَّة ،   وأن تكون موصولة ،   وعلى كل تقدير فهي مُبْتَدَأةٌ ،   و "  الفاءُ  "  ما بعدها في محلِّ جَزْم جواباً إن كانت شرطاً ،   وإلاَّ ففي محلِّ رفعٍ خبراً إن كانت موصُولة ،   والعائدُ محذوفٌ ،   أي : غفورٌ له .  <br />
و "  الهاء  "  في  "  بعده  "  يجوز أن تعود على  "  السُّوء  "  ،   وأن تعود على العمل المفهوم من الفعل كقوله : { اعْدِلُواْ هُوَ أَقْرَبُ } [ المائدة :8 ] والأوَّل أوْلَى ؛ لأنه أصْرَحُ ،   و "  منكم  "  مُتعلِّقٌ بمحذوفٍ إذ هو حالٌ من فاعل  "  عمل  "  ،   ويجوز أن تكون  "  مِنْ  "  للْبَيَانِ ،   فيعمل فيها  "  أعني  "  مقدراً .  <br />
وقوله :  "  بجهالةٍ  "  فيه وجهان :<br />
أحدهما : أنه مُتَعلِّق ب  "  عمل  "  على أن  "  الباء  "  للسَّبَبيَّة ،   أي : عمله بسبب الجَهْلِ ،   وعبَّر أبو البقاء{[14042]} في هذا الوجه عن ذلك بالمفعول به وليس بواضحٍ .  <br />
والثاني : وهو الظَّاهِرُ أنه للحالِ ،   أي : عمله مُصَاحباً للجَهَالَةِ ،    "  ومِنْ  "  في  "  مِنْ بعده  "  لابتداء الغاية .  <br />
فصل في تحرير معنى الآية<br />
قال الحسنُ : كل من عمل مَعْصِيَةً فهو جَاهِلٌ{[14043]} ،   ثُمَّ اختلفُوا ؛ قال مُجاهد : لا يعلمُ حلالاً من حرامٍ فمن جهالته ركب الأمر{[14044]} .   وقيل : جاهلٌ بما يورثه ذلك الذَّنْبُ .  <br />
وقيل : جهالتُهُ من حيث إنه آثر المَعْصِيَةَ على الطَّاعةِ ،   والعاجل القليل على الآجل الكثير ،   ثُمَّ تاب من{[14045]} بعد ورجع عن ذنْبِهِ ،   وأصلح عمله .  <br />
قيل : وأخْلَصَ توْبَتَهُ فإنه غفورٌ رحيمٌ .
