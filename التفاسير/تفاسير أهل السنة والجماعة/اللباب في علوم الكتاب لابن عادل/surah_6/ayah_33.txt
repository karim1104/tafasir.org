قوله تعالى : { قَدْ نَعْلَمُ إِنَّهُ لَيَحْزُنُكَ الَّذِي يَقُولُونَ }  "  قد  "  هنا حرف تَحْقِيقٍ .  <br />
وقال الزمخشري{[13721]} والتبريزي :  "  قد نعلم  "  بمعنى رُبَّمَا التي تجيء لزيادة الفِعْلِ وكثرته ،   نحو قوله : [ الطويل ]<br />
 .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   *** [ وَلَكِنَّهُ ]{[13722]} قَدْ يُهْلِكُ المَالَ نَائِلُهْ{[13723]}<br />
قال أبو حيَّان{[13724]} : وهذا القول غَيْرُ مَشْهورٍ للنحاة ،   وإن قال به بعضهم مُسْتَدِلاً بقول [ القائل ]{[13725]} : [ البسيط ]<br />
قَدْ أتْرُكُ القِرْنَ مُصْفَرًّا أنَامِلُهُ *** كَأنَّ أثْوَابَهُ مُجَّتُ بِفِرْصَادِ{[13726]}<br />
وقال الآخر في ذلك : [ الطويل ]<br />
أخِي ثِقَةٍ لا تُتْلِفُ الخَمْرُ مَالَهُ *** ولَكنَّهُ قَدْ يُهْلِكُ المَالَ نَائِلُهْ{[13727]}<br />
والذي يظهر أن التكثير لا يفهم من  "  قد  "  ،   وإنما فهم من سياق الكلام ؛ إذ التمدُّح بقتل قرن واحد غيرُ طائل ،   وعلى تقدير ذلك فو متعذّر في الآية ؛ لآن علمه -تبارك وتعالى- لا يَقْبَلُ التكثير .  <br />
قال شهابُ الدين : قَدْ يُجَابُ عنه بأن التكثير في متعلِّقات العِلْمِ لا في العِلْمِ ،   [ ثم قال ]{[13728]} : وقوله بمعنى  "  رُبَّمَا  "  التي تجيء لزيادة الفعل وكثرته المشهور أنَّ  "  رُبَّ  "  للتقليل{[13729]} لا للتَّكْثير ،   وزيادةُ  "  ما  "  عليها لا يخرجها عن ذلك ،   بل هي مُهيِّئَةٌ لدخولها على الفعِل ،   و "  ما  " {[13730]} المهيِّئةُ لا تزيل الكَلِمَة عن معناها الأصلي ،   كما لا تزيل{[13731]}  "  لَعَلَّ  "  ،   عن الترجي ،   ولا  "  كأنَّ  "  عن التشبيه ،   ولا  "  ليت  "  عن التمني .  <br />
وقال ابن مالك :  "  قد  "  ك  "  رُبَّمَا  "  في التقليل والصَّرْفِ إلى معنى المضيّ ،   وتكون في حينئذٍ للتَّحقيق والتوكيد ،   نحو{ قَدْ نَعْلَمُ إِنَّهُ لَيَحْزُنُكَ } [ الأنعام : 33 ] { وَقَد تَّعْلَمُونَ أَنِّي رَسُولُ اللَّهِ } [ الصف :5 ] .  <br />
وقوله : [ الطويل ]<br />
وَقَدْ تُدْرِكُ الإنْسَانَ رَحْمَةُ رَبِّهِ *** وَلَوْ كَانَ تَحْتَ الأرْضِ سَبْعِينَ وَادِيَا{[13732]}<br />
وقد تخلو من التَّقلِيل ،   وهي صَارِفَةٌ لمعنى المُضِيَّ ،   نحو قوله :<br />
{ قَدْ نَرَى تَقَلُّبَ وَجْهِكَ } [ البقرة :144 ] .  <br />
وقال مكي : و "  قد  "  هنا وشبهه تأتي لتأكيد الشيء ،   وإيجابه ،   وتصديقه ،   و "  نَعْلَمُ  "  بمعنى عَلِمْنَا .  <br />
وقد تقدم الكلام في هذه الحروف وأنها مُتَردِّدَةٌ بين الحَرْفيَّةِ والاسميَّةِ .  <br />
وقال أبو حيَّان{[13733]} : هُنَا  "  قَدْ  "  حرف تَوَقُّع إذا دخلت على مُسْتَقْبَلِ الزمان كان التوقُّعُ من المتكلّم ؛ كقولك : قد ينزل المَطَرُ شَهْر كذا ،   وإذا كان مَاضِياً أو فِعْلَ حَالٍ بِمَعْنَى المضيّ كان التوقع عند السَّامِعِ .  <br />
وأمَّا المتكلِّم فهو مُوجب ما أخبر به ،   وعبَّر هنا بالمُضَارع إذ المُرَادُ الاتِّصَافُ بالعِلْمِ واسْتِمْرَارُهُ ،   ولم يُلْحَظْ فيه الزمانُ كقولهم :  "  هو يُعْطِي ويَمْنَعُ  "  .  <br />
 "  ليحزنك  "  سَادٌّ مَسَدّ المفعولين ،   فإنها معلِّقَةٌ عن العمَلِ ،   وكُسِرَتْ لدخول  "  اللام  "  في حَيِّزِهَا ،   وتقدَّمَ الكلامُ في  "  ليحزنك  "  ،   وأنه قُرِئَ بفتح{[13734]} الياءِ وضَمِّهَا من  "  حَزَنَهُ  "  و "  أحْزَنَهُ  "  في آل عمران .  <br />
و "  الَّذي يَقُولُون  "  فاعِلٌ ،   وعائدُهُ محذوفٌ ،   أي : الذي يَقُولونَهُ من نِسْبَتِهِمْ له إلى ما لا يَلِيقُ به ،   والضَّميرُ في  "  إنه  "  ضمير الشَّأن والحديث والجُمْلَةُ بعدهُ خَبَرُهُ مُفَسِّرةٌ له ،   ولا يجوزُ في هذا المُضَارع أن يقدر باسمِ فاعلٍ رافعٍ لفاعلٍ كما يُقَدَّر في قولك :  "  إن زيداً يقوم أبوه  "  لئلاَّ يلزمُ تفسيرُ ضمير الشأن بمفردٍ .  <br />
وقد تقدَّمَ أنه مَمْنوعٌ عند البصرييّنَ .  <br />
فصل في سبب نزول الآية<br />
قال السُّدِّيِّ : التقى الأخْنَسُ بن شريق ،   وأبو جَهَل بن هشامٍ ،   فسأله الأخنس أبَا جَهْلِ فقال : يا أبا الحكم ،   أخبرني عن مُحَمَّدٍ أصادق هو أم كاذب ،   فإنه ليس ها هنا أحَدٌ يسمعُ كلامك غيري ؟ .  <br />
فقال أبو جَهْلٍ : والله إن محمَّداً لَصَادِقٌ ،   وما كذَب قَطّ ،   ولكن إذا ذَهَبَتْ بَنُو قُصَيِّ باللّوَاء والسِّقاية والحِجَابَةِ والندْوةِ ،   والنُّبُوَّة ،   فماذا يكون لِسَائِرِ قريشٍ ،   فأنزل اللَّهُ هذه الآية{[13735]} .  <br />
وقال ناجيةُ بن كَعْبَ : قال أبو جَهْلِ للنبي صلى الله عليه وسلم : ما نَتَّهِمُكَ ولا نُكذِّبُكَ ،   ولكنَّا نُكَذِّبُ الذي جئت به ،   فأنزل الله -تعالى- هذه الآية{[13736]} .  <br />
وقيل : إنَّ الحارث بن عامر من قريشٍ قال : يا مُحَمَّدُ ،   والله إن اتَّبَعْنَاكَ نُتَخَطَّفُ من أرْضِنَا ،   فنحنُ لا نُؤمِنُ بك لهذا السَّببِ .   واعلم أن فِرَقِ الكفار كانوا كثيرين ،   فمنهم من ينكر نُبُوَّتَهُ ؛ لأنه ينكر أن يكون الرسولُ من البَشَرِ ،   وقد تقدَّمَ شُبْهَتُهُمْ ،   وأجابَ اللَّهُ عنها{[13737]} .  <br />
ومنهم من ينكر البَعْثَ ،   ويقول : إن محمَّداً يخبر بالحَشْرِ والنَّشْرِ بعد الموت ،   وذلك مُحَالٌ ،   فيطعن في رسالة محمَّدٍ من هذا الوَجْهِ ،   وقد ذكر الله شبههُمْ في هذه السُّورَةِ ،   وأجاب عَنْهَا .  <br />
ومنهم من كان يُشَافِهُهُ بالسَّفَاهَةِ وهُوَ المذكورُ في هذه الآية .  <br />
واختلفوا في ذلك المُحْزِنِ .  <br />
فقيل : كانوا يَقُولوُنَ : ساحرٌ ،   وشاعرٌ ،   وكاهن ،   ومجنون وهو قول الحسن{[13738]} .  <br />
وقيل : كانوا يُصَرِّحُونَ بأنهم لا يؤمنون به{[13739]} .  <br />
وقيل : كانوا ينسبونه إلى الكذبِ{[13740]} .  <br />
قوله :  "  فإنهم لا يكذبُونَكَ  "  .  <br />
قرأ{[13741]} نافع ،   والكسائي  "  لا يكذبونك  "  مُخَفَّفاً من  "  أكْذَب  "  .  <br />
والباقون{[13742]} مثقَّلاً من  "  كذَّبَ  "  وهي قراءة عَلِيٍّ ،   وابن عبَّاسٍ .  <br />
واختلف الناسُ في ذلك ،   فقيل : هُمَا بمعنًى واحدٍ ،   مثل : أكثر وكَثَّرَ وأنْزَلَ ونزَّلَ ،   وقيل : بينهما فَرْقٌ .  <br />
قال الكسائي : العَرَبُ تقول : كَذّبت الرجل بالتَّشْديد إذا نُسِبَ الكذب إليه ،   وأكذبته إذا نَسَبْتَ الكذب إلى ما جَاءَ بِهِ دُونَ أن تَنْسِبَهُ إليه ،   ويقولون أيضاً : أكذبت الرَّجُلَ إذا وجدته كَاذِباً ،   ك  "  أحْمَدْتُهُ  "  إذا وجدته محمُوداً ،   فَمَعْنَى لا يُكذبونك مُخَفَّفاً : لا يَنْسِبُون الكَذِبَ إلَيْكَ ولا يجدونك كاذباً وهو واضحٌ .  <br />
وأمَّا التَّشديد فيكون خبراً مَحْضاً عن عدم تكذيبهم إيَّاه .  <br />
فإن قيل : هذا مُحالٌ ؛ لأن بَعْضَهُم قد وُجِدَ من تكذيب ضَرْورَةً .  <br />
فالجوابُ من وُجُوه :<br />
الأول : أنَّ وإن كان مَنْسُوباً إلى جميعهم أعْنِي عدم التكذيب ،   فهو إنما يُرَادُ بعضهم مجازاً ،   كقولك : { كَذَّبَتْ قَوْمُ نُوحٍ } [ الشعراء :105 ] { كَذَّبَتْ قَوْمُ لُوطٍ } [ الشعراء :160 ] وإن كان فيهم من لم يكذبه ،   فهو عامٌّ يرادُ به الخَاصُّ .  <br />
والثاني : أنه نفي للتكذيب لانْتِفَاءِ ما يَتَرَتَّبُ عليه من المَضَارِّ ،   فكأنه قيل : فإنهم لا يكذبونك تكذيباً يُبَالَى به ويضرك ؛ لأنك لَسْتَ بكاذبٍ ،   فتكذيبُهُمْ كلا تَكْذِيبٍ ،   فهو من نَفْيِ السَّبَبِ لانتفاء مسببه .  <br />
وقال الزمخشري{[13743]} : والمعنى أن تكذيبك أمْرٌ راجع إلى اللَّه تعالى ؛ لأنك رسُولُهُ المصدَّق ،   فهم لا يكذبونك في الحقيقةِ ،   إنَّما يكذِّبون اللَّهَ بجحود آياته ،   فانْتَهِ عن حُزْنِكَ ،   كقول السَّيِّدِ لغُلامِهِ وقد أهَانَهُ بعض الناسِ لم يُهينوك وإنما أهَانُونِي ،   فهو نظير قوله : { إِنَّ الَّذِينَ يُبَايِعُونَكَ إِنَّمَا يُبَايِعُونَ اللَّهَ } [ الفتح :10 ] .  <br />
الثالث : أن القوم ما كانوا يُكذِّبُون به في السِّرِّ كما تقدَّمَ في سبب النزول ،   فيكون تقدير الآية : فإنَّهُمْ لا يُكَذِّبُونَكَ بقلوبهم ،   بل بظاهر قولهم .  <br />
ونظيره : قوله تعالى : { وَجَحَدُواْ بِهَا وَاسْتَيْقَنَتْهَا أَنفُسُهُمْ } [ النمل :14 ] في قصة موسى عليه الصلاة والسلام .  <br />
الرابع : أنهم لا يقولون : أنْتَ كَذَّابٌ ؛ لأنهم جَرَّبُوكَ الدَّهْرَ الطويل وما وَجدُوا منك كذباً ألْبَتَّة ،   وسَمَّوْك بالأمين ،   وإنما جَحَدُوا صِحّةَ نُبُوَّتك ؛ لأنهم اعْتَقَدُوا أنَّ محمَّداً عرض له نوع خَبَلٍ ونُقْصَانٍ ،   فلأجله تَخَيَّلَ في نفسه كونه رَسُولاً من عند الله وبهذا التقدير لا ينسبونه إلى الكذبِ ،   بل هو أمينٌ في كُلِّ الأمورِ إلاَّ في هذا الوجهِ الواحد .  <br />
الخامس : قال ابن الخطيب{[13744]} : المرادُ أنَّهُمْ لا يخصُّونك بالتكذيب ،   بل ينكرُون دلالةَ المعجزة الظَّاهرة على الصدق مطلقاً لقوله : { وَلَكِنَّ الظَّالِمِينَ بِآيَاتِ اللَّهِ يَجْحَدُونَ } .  <br />
والمُرَادُ أنهم يقولون في كُلِّ معجزةٍ : إنها سِحْرٌ ،   فالتقدير : أنهم لا يكذِّبونك على التِّعيين ،   بل القَوْمُ يُكذِّبون جَمِيعَ الأنبياء والرُّسُلِ .  <br />
قوله :  "  بآيات اللَّهِ  "  يجوز في هذا الجَارِّ وجهان :<br />
أحدهما : أنه مُتَعَلِّقٌ ب  "  يجحدونَ  "  وهو الظَّاهر ،   وجوَّز أبُو البقاء{[13745]} أن يتعلق ب  "  الظَّالمين  "  قال : كقوله تعالى : { وَآتَيْنَا ثَمُودَ النَّاقَةَ مُبْصِرَةً فَظَلَمُواْ بِهَا } [ الإسراء :59 ] وهذا الذي قال ليس بجيِّدٍ لأن  "  الباء  "  هناك سَبَبِيَّةٌ ،   أي : ظلموا بسببها ،   و "  الباء  "  هنا معناها التعدية ،   وهنا شيء يتعلَّق به تعلُّقاً واضحاً ،   فلا ضَرُورَةَ تَدْعُوا إلى الخروج عَنْهُ ،   وفي هذه الآية إقامةُ الظاهر مُقَامَ المضمر ،   إذا الأصل :  "  ولكنهم يَجْحَدُونَ بآياتِ الله  "  ،   ولكنَّهُ نَبَّهَ على أن الظلم هو الحامل لهم على الجُحُودِ .  <br />
والجحود والجَحْدُ نفي ما في القَلْبِ ثَبَاتُهُ ،   وأو إثْبَاتُ ما في القلبِ نَفْيُهُ .  <br />
وقيل : الجَحْدُ إنْكَارُ المعرفةِ ،   فليس مُرَادفاً للنفي من كُلِّ وجهٍ .
