قوله تعالى : { قَدْ خَسِرَ الَّذِينَ كَذَّبُواْ بِلِقَاءِ اللَّهِ } [ الآية :31 ] وصف أحوال منكري البَعْثِ بأمرين :<br />
أحدهما : حصول الخُسْرَانِ ،   أي : خسروا أنفسهم بتكذيبهم المصير إلى اللَّهِ -تبارك وتعالى- بالبَعْثِ بعد الموت .  <br />
والثاني : حَمْلُ الأوْزَارِ العظيمة ،   فأمَّا خسرانهم فهو حسرتهم على تفريطهم وفوات الثواب وحُصُول العقاب .  <br />
قوله : { حَتَّى إِذَا جَاءَتْهُمُ السَّاعَةُ بَغْتَةً } في نصب  "  بَغْتَةً  "  أربعة أوجه :<br />
أحدها : أنها مصدرٌ في موضع الحال من فاعل  "  جاءَتْهُمْ  "   ،   أي : مُبَاغتةً ،   وإمَّا من مفعوله أي : مبغوتين .  <br />
الثاني : أنها مصدرٌ على غير الصَّدر{[13645]} ؛ لأنَّ معنى  "  جاءتهم  "  بَغَتَتْهُمْ بغتة ،   فهو كقولهم :  "  أتيته رَكْضاً  "  .  <br />
الثالث : أنَّها منصوبةٌ بفعل محذوف من لفظها ،   أي : تبغتهم بَغْتَة .  <br />
الرابع : بفعل [ من غير لفظها ،   أي : أتتهم بغتة ،   والبغت والبغتة مفاجأة الشيء بسرعة من ]{[13646]} غير اعتدادٍ به ،   ولا جَعْلِ بالٍ منه حتَّى لو استشعر الإنسانُ به ،   ثم جاء بسرعة من غير اعتدادٍ به لا يُقَالُ فيه : بَغْتَة ،   وكذلك قول الشاعر في ذلك : [ الطويل ]<br />
إذَا بَغَتَتْ أشْيَاءُ قَدْ كَانَ قَبْلَهَا *** قَدِيماً فَلاَ تَعْتَدَّهَا بَغَتَاتِ{[13647]}<br />
والألف واللام في  "  السَّاعة  "  للغَلَبَةِ كالنَّجْمِ والثُّرَيَّا ؛ لأنها غلبت على يوم القيامة ،   وسِّمَيتِ القيامَةُ سَاعةً لسرعة الحِسَابِ فيها على الباري تبارك وتعالى .  <br />
وقيل : لأنَّ السَّاعة من الوَقْتِ الذي تقوم فيه القيامة ؛ لأنها تَفْجأ الناس في ساعة لا يعلمها [ أحدٌ ]{[13648]} إلاَّ اللَّهُ تعالى .   وقوله :  "  قالوا  "  هو جواب  "  إذا  "  .  <br />
قوله :  "  يَا حَسْرَتَنَا  "  هذا مجازٌ ؛ لأن الحَسْرَةَ لا يتأتى منها الإقْبَالُ ،   وإنَّما المعنى على المُبَالغَةِ في شِدَّةِ التَّحَسُّرِ ،   وكأنهم نادوا التحسُّر ،   وقالوا : إن كان لك وَقْتٌ ،   فهذا أوانُ حضورك .  <br />
ومثله :  "  يا ويلتا  "  والمقصودُ التنبيهُ على خطأ المنادي ،   حيث ترك ما أحْوَجَهُ تركه إلى نداء هذه الأشياء .  <br />
قال سيبويه{[13649]} - رحمه الله - : فيكون المنادى هو نفس الحَسْرَةِ ،   والمُرَادُ بالحَسْرَةِ النَّدَامَةُ .  <br />
قال الزَّجَّاج{[13650]} - رحمه الله تعالى - : هذا النِّدَاءُ ينبِّهُ الناس على ما سيحصل لهم من الحَسْرَةِ ،   والعربُ تعبر عن تعظيم أمثال هذه الأمور باللَّفظَةِ كقوله تبارك وتعالى : { يَا حَسْرَةً عَلَى الْعِبَادِ } [ يس :30 ] { يَا حَسْرَتَى عَلَى مَا فَرَّطَتُ فِي جَنبِ اللَّهِ }<br />
[ الزمر :56 ] { يَا وَيْلَتَى ءَأَلِدُ } [ هود :72 ] و{ يَا أَسَفَى } [ يوسف :84 ] والمعنى : يا أيها النَّاس تَنَبَّهُوا على ما وَقَعَ من الأسَفِ ،   فوقع النداءُ على غير المنادى في الحقيقة .  <br />
قوله :  "  عَلَىَ مَا فَرَّطْنَا  "  متعلّق{[13651]} بالحسرة و "  ما  "  مَصْدريَّةٌ ،   أي : على تفريطنا ،   والضمير في  "  فيها  "  يجوز أن يعود على السَّاعِةِ ،   ولا بد من مضاف ،   أي في شأنها والإيمان بها ،   وأن يعود على الصَّفَقَةِ المتضمِّنة في قوله : { قّدْ خَسِرَ الَّذين } ،   قاله الحسن ،   أو يعود على الحياة الدنيا ،   وإن لم يَجْرِ لها ذِكُرٌ لكونها مَعْلُومَةَ ،   قاله الزمخشري{[13652]} - رحمه الله تعالى - .  <br />
وقيل : يعود على مَنَازِلِهْم في الجنَّةِ إذا رأوها ،   وهو بَعِيدٌ .  <br />
والتفريطُ : التقصير في الشيء مع القُدْرَةِ على فعله .  <br />
وقال أبو عُبَيْدِ{[13653]} هو التَّضْييعُ .  <br />
وقال ابن بَحْرٍ : وهو السَّبْقُ ومنه الفارط ،   أي : السَّابق للقوم ،   فمعنى فَرَّط بالتشديد خَلَّى السَّبْقَ لغيره ،   فالتضعيف فيه للسَّلْبِ ،   ك  "  جلَّدت البعير  "  ومنه { فَتَهَجَّدْ بِهِ نَافِلَةً لَّكَ } [ الإسراء :79 ] .  <br />
فصل في تحرير معنى الخسران<br />
اعلم أن [ كلمة  "  حتى  "  ]{[13654]} غاية [ لقوله :  "  كذبوا  "  ]{[13655]} لا لقوله :  "  قد خَسِرَ  "  ،   لأن خسرانهم لا غَايَةَ له ،   ومعنى  "  حتى  "  ها هنا أنَّ مُنْتَهَى تكذيبهم الحَسْرةُ يوم القيامة والمعنى : أنهم كذبوا بالبَعْثِ إلى أن ظهرت السَّاعةُ بَغْتَةً ،   فإن قيل : إنما يَتَحَسَّرُونَ عن موتهم .  <br />
فالجواب : لما كان المَوْتُ وُقُوعاً في [ أحوال الآخرة و ]{[13656]} مقدماتها جُعِلَ من جنس السَّاعة ،   وسُمِّيَ باسمها ،   فلذلك قال عليه الصلاة والسلام :  "  مَنْ مَاتَ فَقَدْ قَامَتْ قِيَامَتُهُ  "  والمراد بالساعة : القيامة .  <br />
قوله :  "  وَهُمْ يَحْمِلُونَ  "   "  الواو "  للحال ،   وصَاحِبُ الحال  "  الواو "  في  "  قالوا  "  أي : قالوا : يا حَسْرَتَنَا في حالةِ حَمْلِهِمْ أوْزَارَهُمْ .  <br />
وصُدِّرت هذه الجملة بضمير مبتدأ ؛ ليكون ذِكْرُهُ مَرَّتين فهو أبْلَغُ .  <br />
والحَمْلُ هنا قيل : مجازٌ عن مُقََاساتِهِمُ العذابَ الذي سَبَبُهُ الأوْزَارُ .  <br />
[ قال الزَّجَّاجُ : كما يُقَال :  "  ثقل عليّ كلام فلان  "  ]{[13657]} والمعنى : كرهته .  <br />
وقيل : هو حقيقة وفي الحديث :  "  إنَّهُ يُمَثَّلُ لَهُ عَمَلُهُ بِصُورَةٍ قَبِيحَةٍ مُنْتِنَةِ الرِّيح فَيَحمِلُهَا  "  وهو قول قتادَةَ ،   والسُّدي{[13658]} ،   وخُصَّ الظَّهْرُ ،   لأنه يُطيق [ من الحَمْل ]{[13659]} ما لا يُطِيقُهُ غيره من الأعْضَاءِ كالرأس والكَاهِلِ ،   وهذا كما تقدم{[13660]} في قوله : { فَلَمَسُوهُ بِأَيْدِيهِمْ }<br />
[ الأنعام :7 ] { فَنَبَذُوهُ وَرَاءَ ظُهُورِهِمْ } [ آل عمران :187 ] لأن اليد أقوى في الإدراك اللَّمْسِيِّ من غيرها .  <br />
والأوزَارُ : جمع  "  وِزْر  "  ك  "  حِمْل  "  وأحمال وعِدْل وأعْدَال .  <br />
والوِزْر في الأصل الثقل ،   ومنه : وَزَرْتُهُ ،   أي : حَمَّلته شَيْئاً ثقيلاً{[13661]} ،   ووزير الملك من هذا ؛ لأنه يَتَحَمَّلُ أعْبَاء ما قَلَّدَهُ المَلِكُ من مئونة رعيَّته وحَشَمَتِه{[13662]} ومنه أوْزَارُ الحرب لسلاحها وآلاتها ،   قال [ القائل في ذلك ]{[13663]} : [ المتقارب ]<br />
وأعْدَدْتُ لِلْحَرْبِ أوْزَارَهَا *** رِمَاحاً طِوَالاً وخَيْلاً ذُكُورَا{[13664]}<br />
وقيل : الأصل في ذلك الوَزَرُ بفتح الواو والزاي ،   وهو المَلْجأ الذي يُلْتَجَأُ إليه من الجَبَلِ ،   قال تعالى : { كَلاَّ لاَ وَزَرَ } [ القيامة :11 ] ثمَّ قيل للثقل : وِزْرٌ تَشْبيهاً بالجَبَلِ ،   ثم اسْتُعِيرَ الوِزْرُ إلى الذَّنْبِ تشْبيهاً به في مُلاقَاةِ المَشَقَّةِ ،   والحاصلُ أنَّ هذه المادة تَدُلُّ على الرَّزَانَةِ والعَظَمَةِ .  <br />
قوله : { ألا سَاءَ ما يزِرُونَ }  "  ساء  "  هنا تحتمل أوجهاً ثلاثة :<br />
أحدها : أنها  "  ساء  "  المُتَصَرِّفَةُ المتعدِّيَةُ ،   ووزنها حينئذٍ  "  فَعَل  "  بفتح العين ،   ومفعولها حينئذٍ محذوفٌ ،   وفاعلها  "  ما  "  .  <br />
و "  ما  "  تحتمل ثلاثة أوجه :<br />
أن تكون موصولةً اسميةً ،   أو حرفية ،   أو نكرة موصوفة ،   وهو بعيد ،   [ وعلى جعلها اسمية أو نكرة موصوفة تقدّر ]{[13665]} لها عائداً ،   والحرفية غير محتاجة إليه عند الجمهور .  <br />
والتقدير : ألا سَاءَهُمُ الذي يَزِرُونَهُ ،   أو شيء يزرونه ،   أو وزْرُهُمْ [ وبدأ ابن عطية بهذا الوجه ؛ قال{[13666]} : كما تقول : ساءني هذا الأمر ،   والكلام خبر مجرد كقوله : [ البسيط ]<br />
رَضيتَ خِطَّةَ خَسْفٍ غَيْرَ طَائِلَةٍ *** فَسَاءَ هَذَا رِضًى يا قَيْسَ عَيْلانَا{[13667]}<br />
قال أبو حيان{[13668]} : ولا يتعين أن تكون  "  ما  "  في البيت خبراً مجرداً ،   بل تحتمل الأوجه الثلاثة ]{[13669]} وهو ظاهر .  <br />
الثاني : أن يكون للتعجُّب ،   فتنتقل{[13670]} من  "  فَعَل  "  بفتح العين [ إلى ]{[13671]}  "  فعُل  "  بضمها ،   فتعطى حكم فعل التَّعَجُّب من عدمِ التصرف ،   والخروج من الخبر المَحْضِ إلى الإنشاء إن قلنا : إن التعجُّب إنشاء ،   وهو الصحيح ،   والمعنى : ما أسْوَأَ ،   أي : أقبح الذي يزرونه ،   أو شيئاً يزرونه ،   أو وِزْرُهم .  <br />
الثالث : أنها بمعنى  "  بئس  "  فتكون للمُبَالَغَةِ في الذَّمِّ فتعطى أحكامها أيضاً ،   ويجري الخِلافُ في  "  ما  "  الواقعةِ بعدها حَسْبَما ذكر في { بِئْسَمَا اشْتَرَوْا } [ البقرة : 90 ] وقد ظهر الفَرْقُ بين هذه الأوجه الثلاثة ،   فإنها في الأوَّل متعدّية متصرّفة ،   والكلام معها خَبَرٌ مَحْضٌ ،   وفي الأخيرين قَاصِرَةٌ جامدة إنشائية .  <br />
والفرق بين الوجهين الأخيرين أنَّ التعجبيَّة لا يُشْتَرَطُ في فاعلها ما يشترط في فاعل  "  بئس  "  .  <br />
وقال أبو حيَّان{[13672]} : والفَرْقُ بين هذا الوجه يعني كونها بمعنى  "  بئس  "  ،   والوجه الذي قبله - يعني كونها تعجبيَّةً - أنه لا يُشْتَرَطُ فيه ما يشترط في فاعل  "  بئس  "  من الأحكام ،   ولا هو جملةٌ منعقدةٌ من مبتدأ وخبر ،   [ إنما هو منعقد من فعل أو فاعل انتهى ،   وظاهره لا يظهر إلاَّ بتأويل ،   وهو أن الذم لا بد فيه من مَخْصوص بالذَّمِّ ،   وهو مبتدأ ،   والجملة الفعلية قبله خبره فانعقد من هذه الجملة مبتدأ وخبر ]{[13673]} .  <br />
إلا أنَّ لقائلٍ أن يقول : إنما يَتَأتَّى هذا على أحَدِ الأعَارِيبِ في [ المخصوص ]{[13674]} وعلى تقدير التَّسْليم ،   فلا مَدْخَلَ للمخصوص بالذَّمِّ في جملة الذَّمِّ بالنسبة إلى كونها فِعْلِيَّة ،   فحينئذٍ لا يظهر فَرْقٌ بينها وبين التَّعجبية في أنَّ كُلاَّ منهما منعقدةٌ من فِعلٍ وفاعل .  <br />
قال ابن عباس - رضي الله عنهما - : بئْسَ الحمْلُ حَمَلُوا  "  .
