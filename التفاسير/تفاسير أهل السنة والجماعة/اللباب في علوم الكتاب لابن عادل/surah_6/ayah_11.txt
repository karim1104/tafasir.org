قوله تعالى : { قُلْ سِيرُواْ فِي الأَرْضِ } كما صبر رسول الله صلى الله عليه وسلم في الآية الأولى{[13303]} ،   حَذَّرَ القوم في هذه الآية ،   وقال لرسوله : قل لهم : لا تغتروا بما وَصَلْتُمْ إليه من الدنيا ولذَّاتها ،   بل سيروا في الأرض لتعرفُوا صحة ما أخبركم الرسولُ عنه من نزول العذاب بمن كذب الرسل من الأمم السَّالفة قبلكم .   يحذر كفار  "  مكّة  "  ،   ويحتمل هذا السير أنْ يكون بالعقول والفِكَرِ ،   ويحتمل السَّيْرَ في الأرضِ .  <br />
قوله :  "  ثُمَّ انْظُرُوا  "  : عطف على  "  سِيرُوا  "  ولم يجئ في القرآن العطف{[13304]} في مثل هذا الموضوع إلاَّ بالفاء ،   وهنا جاء ب  "  ثم  "  فيحتاج إلى فَرْقٍ .  <br />
فذكر{[13305]} الزمخشري الفرق وهو : أنْ جَعَل النَّظَرَ مُسَبَّباً عن السَّيْرِ في قوله :  "  فانْظُرُوا  "  كأنه{[13306]} قيل : سِيُروا لأجْلِ النظرِ ،   ولا تسيروا سَيْرَ الغافلين .  <br />
وهنا معناه إبَاحَةُ السَيَّرِ في الأرض للتجارة وغيرها من المَنَافِع ،   وإيجاب النظر في آثار الهالكين ،   ونبَّه على ذلك ب  "  ثمَّ  "  لِتَبَاعُدِ ما بين الواجب والمباح{[13307]} .  <br />
قال أبو حيَّان{[13308]} - رضي الله عنه - : وما ذكر أوَّلاً مُتَنَاقض ؛ لأنه جعل النظر مُتَسَبِّباً عن السَّيْرِ ،   فكان السَّيْرُ سبباً للنَّظَرِ ،   ثم قال : فكأنه قيل : سيروا لأجْلِ النَّظَرِ ،   فجعل السَّيْرَ مَعْلُولاً بالنَّظَرِ ،   والنَّظَرِ سَبَبٌ له فَتَنَاقَضَا ،   ودعوى أن  "  الفاء  "  سببيةٌ دعوى لا دَلِيلَ عليها ،   وإنَّما مَعْنَاها التَّعْقِيبُ فقط ،   وأمَّا  "  زَنَى ماعِز فَرُجم  "  فَفَهْمُ السببية من قَرِينَةٍ غيرها .  <br />
قال :  "  وعلى تقدير [ تَسْلِيم ]{[13309]} إفادتها السَّبَبَ ،   فَلِمَ كان السيرُ هنا سَيْرَ إباحةٍ ،   وفي غيره سَيْرَ إيجاب ؟  "  .  <br />
وهذا اعتراض صحيح إلاَّ قوله :  "  إنَّ  "  الفاء  "  لا تفيد السَّبَبِيَّةَ  " {[13310]} فإنه غيرُ{[13311]} مُرْضٍ ،   [ ودليله في غير هذا الموضع ]{[13312]} ومثلُ هذا المكان في كون الزَّمَخْشَرِيّ جعل شيئاً عِلَّةً ،   ثم جعله مَعْلولاً{[13313]} ،   كما{[13314]} سيأتي إن شاء الله -تعالى- في أوَّلِ  "  الفتح  "  ،   ويأتي هناك جوابه .  <br />
قوله :  "  كَيْفَ كَانَ عَاقِبَةُ  "  كَيْفَ  "  خبرٌ مقدَّمٌ ،   و "  عاقبة  "  اسمها ،   ولم يُؤنَّثْ فعلْها ؛ لأنَّ تأنيثها غير حقيقي ؛ ولأنها بتأويل المآلِ والمُنْتَهَى ،   فإنَّ العاقبة مَصْدَرٌ على وزن  "  فاعلة  "  وهو محفوظ في ألْفَاظ تقدَّمَ ذِكْرُها وهي منتهى الشيء وما يَصيرُ إليه .  <br />
والعاقبةُ إذا أطْلِقَتْ اختصت بالثواب قال تعالى : { وَالْعَاقِبَةُ لِلْمُتَّقِينَ } [ الأعراف :128 ] وبالإضافة قد تستعمل في العُقُوبةِ كقوله تبارك وتعالى : { ثُمَّ كَانَ عَاقِبَةَ الَّذِينَ أَسَاءُواْ السُّوأى }<br />
[ الروم :10 ] ،   { فَكَانَ عَاقِبَتَهُمَا أَنَّهُمَا فِي النَّارِ خَالِدِينَ فِيهَا }<br />
[ الحشر :17 ] فَصَحَّ أن تكون اسْتِعَارةً من ضدِّهِ كقوله تعالى :<br />
{ فَبَشِّرْهُم بِعَذَابٍ أَلِيمٍ } [ آل عمران :21 ] .  <br />
و "  كَيْفَ  "  مُعَلِّقة للنظر ،   فهي في مَحَلِّ نصبٍ على إسْقَاط الخافض ،   لأنَّ معناه هنا التَّفَكُّرُ والتدبُّرُ .   والله أعلم .
