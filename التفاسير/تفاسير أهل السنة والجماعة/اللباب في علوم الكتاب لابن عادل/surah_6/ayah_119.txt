قوله :  "  ومَا لَكُمْ  "  مُبْتَدأ وخبر ،   وقوله :  "  ألاَّ تَأكُلوا  "  فيه قولان :<br />
أحدهما : هو على حَذْف حَرْف الجرِّ ،   أي : أيُّ شيء اسْتَقَرَّ في مَنْع الأكْل ممَّا ذكر اسْم اللَّه عليه ؛ وهو قول أبي إسْحَاق{[15065]} الزَّجَّاج ،   فلما حُذِفَتْ  "  في  "  جَرَى القولان المَشْهُوران ،   ولم يذكر الزَّمَخْشَرِيُّ غير هذا الوجه{[15066]} .  <br />
الثاني : أنَّها في محل نَصْبٍ على الحالِ ،   والتَّقْدير : وأيُّ شَيْء لَكُم تَاركين للأكْل ،   ويؤيِّد ذلك وُقوع الحالِ الصَّريحة في مِثْل هذا التَّركيب كَثِيراً ،   نحو : { فَمَا لَهُمْ عَنِ التذكرة مُعْرِضِينَ } [ المدثر : 49 ] إلاَّ أن هذا مَرْدُود بِوَجْهَيْن :<br />
أحدهما : أنَّ  "  أنْ  "  تُخَلِّص الفِعْل للاسْتِقْبَال ،   فكيف يَقَعُ ما بَعْدَها حالاً ؟<br />
والثاني : أنَّها مع ما بعدها مُؤوَّلة بالمصدر ،   وهو أشْبَه بالمُضْمَرَات كما تقدَّم تحريره ،   والحال إنَّما تكُون نكرة .  <br />
قال أبُو البقاء{[15067]} : إلاَّ أن يُقَدَّر حَذْفُ مُضاف ،   فَيَجُوز ،   أي :  "  وما لَكُم ذَوِي ألا تأكُلوا  "  وفي تَكَلُّف ،   فمفعول  "  تَأكُلُوا  "  مَحْذوف بَقِيتْ صفَته ،   تقديره :  "  شَيْئاً مما ذُكِر اسْمُ اللَّه  "  ويجُوز ألاَ يُراد مَفْعُول ،   بل المُراد : ومَا لكُم ألا يقع منكم الأكْل ،   وتكون  "  مِنْ  "  لابْتِدَاء الغَاية ،   أي : أن لا تَبْتَدِئُوا بالأكْل من المَذْكُور عليه اسم اللَّه ،   وزُعِم ،   أنَّ  "  لاَ  "  مَزِيدة ،   وهذا فَاسِدٌ ؛ إذا لا داعِي لِزيَادتها .  <br />
قوله :  "  وقد فصَّل لَكُم ما حرَّم  "  قرأ{[15068]} ابْنُ كَثِير ،   وأبُو عَمْرو ،   وابنُ عَامِر : ببنائهما للمفعُول : ونافع ،   وحفصٌ عن عاصم : ببنَائِهَما للفاعل ،   وحمزة ،   والكسَائِيُّ ،   وأبُو بكر عن عاصم{[15069]} : ببناء الأوَّل للفاعل ،   وبناء الثّانِي للمَفْعُول ،   ولم يأتي عكس هذه ،   وقرأ عطيَّة العُوفيُّ كقراءة الأخَويْن ،   إلاَّ أنَّه خفف الصَّاد من  "  فَصَّل  "  والقَائِم مقام الفاعل : هو المَوْصُول ،   وعائده من قوله :  "  حرَّم عَلَيْكُم  "  .   والفَاعِل في قراءة مَنْ بَنَى للفَاعِل ضمير اللَّه -تعالى- ،   والجُمْلَة في محلِّ نَصْبٍ على الحالِ .  <br />
فصل في المراد من الآية<br />
قوله :  "  فَصَّل لَكُم ما حرَّم عَلَيْكُم  "  قال أكثر المُفَسِّرين : هو المُراد من قوله تعالى : { حُرِّمَتْ عَلَيْكُمُ الميتة والدم وَلَحْمُ الخنزير } في أوَّل المائدة [ الآية : 3 ] ،   وفيه إشْكَالٌ ،   وهو أنَّ سُورة الأنْعَام مَكيَّة ،   وسُورة المائدة من آخر ما أنْزَل اللَّه -تعالى- بالمدينة ،   فقوله :  "  فصَّل  "  يَجِبُ أن يكُون ذلك المُفَصَّل متقدِّماً على هذا المُجْمَل ،   والمَدَنِيّ متأخِّر عن المَكِيّ ،   فيمتنع كونه مُتقدِّماً ،   ولقَائِل أن يقول : المُفَصّل : هو قوله - تبارك وتعالى- بعد هذه الآية الكريمة : { قُل لاَّ أَجِدُ فِي مَا أُوْحِيَ إِلَيَّ مُحَرَّماً على طَاعِمٍ يَطْعَمُهُ } [ الأنعام : 145 ] ،   الآية ،   وهي وإن كانت مذكورة بعد هذه الآية بقليل ،   إلا أن هذا القدر من التأخير لا يمنع أن يكون هو المُرَاد ،   خُصُوصاً أن السُّورة نزلت دَفْعَة واحِدَة بإجْماع المُفَسِّرين على ما تقدَّم ،   فيكون في حُكْم المُقارن .  <br />
قوله :  "  إلاَّ ما اضْطُرِرْتُم إليه  "  في الاسْتِثْنَاء وجهان :<br />
أحدهما : أنَّه مُنْقَطِع ،   قاله ابن عطيَّة والحُوفِي .  <br />
والثاني : أنه [ اسْتِثْنَاء ]{[15070]} متَّصِل .  <br />
قال أبو البقاء{[15071]} :  "  ما  "  في مَوْضِع نَصْبٍ على الاسْتِثْنَاء من الجِنْس من طريق المَعْنى ؛ لأنه وبَّخَهُم بترك الأكل مِمَّا سُمِّي عليه ،   وذلك يَتَضَمّن الإباحة مُطْلَقاً .  <br />
قال شهاب الدِّين{[15072]} : الأوَّل أوْضَح والاتِّصال قلق المَعْنَى ،   ثم قال :  "  وقوله :  "  وقد فصَّل لَكُم ما حرَّم عليكم  "  أي : في حَالِ الاخْتِيَار ،   وذلك حلالٌ حال الاضْطِرارِ  "  .  <br />
قوله :  "  وإنَّ كَثِيراً لَيُضِلُّون  "  قرأ الكوفيُّون{[15073]} بضمِّ الياء ،   وكذا الَّتِي في يُونس : { رَبَّنَا لِيُضِلُّواْ } [ الآية : 88 ] والباقون : بالفَتْح ،   وسيأتي لذلك نَظَائِر في إبْراهيم وغيرها ،   والقراءتان واضِحَتَان ؛ فإنه يٌقال : ضلَّ في نَفْسَه ،   وأضَلَّ غيره ،   فالمَفْعُول مَحْذُوف على قراءة الكُوفيين : وهي أبْلَغ في الذَّمِّ ،   فإنها تتضَّمن قُبْحَ فِعْلِهم ،   حَيْث ضلوا في أنْفُسِهِم ،   وأضَلُّوا غيرهم ؛ كقوله - تعالى- : { وَأَضَلُّواْ كَثِيراً وَضَلُّواْ عَن سَوَاءِ السبيل } [ المائدة : 77 ] .  <br />
قيل المُراد بِه : عمرو بن لُحَيّ فمن دُونه من المشركين الَّذين اتخذوا البَحَائِر والسَّوَائِب وقراءة الفَتح لا تُحوِجُ إلى حذف ،   فرجَّحها بَعْضُهم بهذا وأيضاً : فإنهم أجْمَعُوا على الفَتْح في  "  ص  "  عند قوله : { إِنَّ الذين يَضِلُّونَ عَن سَبِيلِ الله } [ 26 ] .  <br />
وقوله :  "  بِأهْوَائِهِم  "  متعلِّق ب  "  يَضِلُّونَ  "  والباءُ سَبَبيَّة ،   أي : اتِّباعهم أهْواءَهم ،   وشهواتهم .  <br />
وقوله :  "  بغير عِلْم  "  متعلِّق بِمَحْذُوف ،   لأنه حالٌ ،   أي : يَضِلُّون مُصَاحِبِين للجَهْلِ ،   أي : مُلْتَبِسين بغير علمٍ .  <br />
فصل في المراد بالآية<br />
قيل : المُرَاد : عمرو بن لُحَيّ كما تقدَّم ؛ لأنَّه أول من غير دين إسماعيل .  <br />
وقال الزَّجَّاج{[15074]} : المراد منه الَّذِين يُحَلِّلُون المَيْتَة ،   ويناظِرُونكم في إحلالها ،   ويَحْتَجون عليهخا بقولهم لما أحَلَّ ما تذبحونه أنْتُم ،   فَبأن يحلَّ ما يَذْبَحُه الله أوْلَى ،   وكذلك كل ما يَضِلُّون فيه من عبادة الأوثان ،   والطَّعن في نُبُوّة محمد صلى الله عليه وسلم وإنما يتِّبِعُون فيه الهوى والشَّهوة [ بغير عِلْمن وهذه الآية تدلُّ على أن التقليد حَرَام ؛ لأنَّه قول بمحض الَوَى والشَّهوة ]{[15075]} ثم قال :  "  إنَّ رَبَّك هُوَ أعْلم بالمُهْتدين  "  أي : هو العالم بما في ضَمَائِرِهم من التَّعَدِّي ،   وطلب نُصْرة البَاطِل ،   والسَّعي في إخْفَاء الحقِّ ،   وإذا كان عَالِماً بأحْوَالهم وقَادِراً على مجازاتهم فهو تعالى يجازيهم عليها والمقصُود منه التَّهْديد والتخويف .
