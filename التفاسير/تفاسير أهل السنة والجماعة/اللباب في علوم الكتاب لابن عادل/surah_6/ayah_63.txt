وهذا نوع آخر من الدلالة على كمالِ القُدرةِ الإلهية ،   وكمال الرحمة والفَضْلِ والإحسان .  <br />
وقرأ السبعة هذه{[14134]} مشدّدة : { قُلِ اللَّهُ يُنَجِّيكُم } [ الأنعام :64 ] قرأها الكوفيون وهشام بن عامر عن ابن عامر مشددة كالأولى .   وقرأ الثِّنْتينِ{[14135]} بالتخفيف من  "  أنْجَى  "  حُمَيْدُ بن قيس ،   ويعقوب ،   وعلي بن نَصْرٍ عن أبي عمرو ،   وتحصَّل من ذلك أن الكوفيين وهشاماً يثقلون في الموضعين ،   وأن حميداً ومن مَعَهُ يُخَفِّفُونَ فيهما ،   وأن نافعاً ،   وابن كثير ،   وأبا عمرو ،   وابن ذكوان عن ابن عامرٍ يُثَقِّلُون الأولى ،   ويُخَفِّفُون الثانية ،   والقراءات واضحة ،   فإنها من : نجَّى وأنْجى ،   فالتضعيف والهمزة كلاهما للتَّعديَةِ .  <br />
فالكوفيون وهشام التَزَمُوا التَّعْديةَ بالتضعيف ،   وحميد وجماعته التَزَمُوهَا بالهمزة .   والباقون جمعوا بين التَّعديتين جمعاً بين اللُّغَتَيْنِ كقوله تعالى : { فَمَهِّلِ الْكَافِرِينَ أَمْهِلْهُمْ رُوَيْداً } [ الطارق :17 ] .  <br />
والاستفهام للتقرير والتَّوْبيخ ،   وفي الكلام حَذفُ مضاف ،   أي : مِنْ مهالِكِ ظُلُمات ،   أو من مخاوفها ،   والظلمات كِنَايةٌ عن الشدائد والأهوال إذا سافروا في البرِّ والبَحْرِ .  <br />
قوله :  "  تَدْعُونَهُ  "  في مَحَلِّ نصب على الحال ،   إما من مفعول  "  ينجيكم  "  ،   وهو الظاهر ،   أي : ينجيكم داعين إيَّاه ،   وإما من فاعله ،   أي : مدعُوَّاً من جهتكم .  <br />
قوله : { تَضَرُّعاً وخُفْيَةٌ } يجوز فيها وجهان :<br />
أحدهما : أنهما مصدران في موضع الحالِ ،   أي : تدعونه مُتَضَرِّعين ومُخْفِينَ .  <br />
والثاني : أنها مصدارن من معنى العامل لا من لفظه كقولك : قعدت جُلُوساً .  <br />
وقرأ الجمهور{[14136]} :  "  خُفْيَةً  "  بضم الخاء ،   وقرأ{[14137]} أبو بكر بكسرها ،   وهما لغتانِ ،   كالعُدْوةِ والعِدْوةِ ،   والأسْوَة والإسْوَة .  <br />
وقرأ الأعمش{[14138]} :  "  وخيفة  "  كالتي في  "  الأعراف  "  وهي من الخَوْفِ ،   قُلِبَتْ  "  الواو "  ياء لانكسار ما قبلها وسكونها ،   ويظهر على هذه القراءة أن يكون مفعولاً من أجله لولا ما يَأبَاهُ  "  تَضرُّعاً  "  من المعنى .  <br />
قوله :  "  لَئِنْ أنْجَيْتَنَا  "  الظاهر أن هذه الجملة القسميَّةَ تفسير للدُّع‍اءِ قبلها .  <br />
ويجوز أن تكون مَنْصُوبَةً المَحلِّ على إضمار القول ،   ويكون ذلك القول في محلِّ نصب على الحال من{[14139]} فاعل  "  تدعونه  "  أي : تدعونه قائلين ذلك ،   وقد عرف مما تقد‍َّم غير مرَّةٍ كيفية اجتماع الشرط والقسم .  <br />
وقرأ الكوفيون{[14140]}  "  أنْجَانَا  "  بلفظ الغَيْبَةِ مُرَاعَاةً لقوله  "  تَدْعُونَهُ  "  والباقون{[14141]}  "  أنجيتَنَا  "  بالخطاب حكاية لخطابِهِمْ في حالة الدعاء ،   وقد قرأ كُلٌّ بما رسم في مصحفه ،   فإن في مصاحف  "  الكوفة  "  : أنْجَانَا  "  ،   وفي غيرها :  "  أنْجَيْتَنَا  "  .  <br />
قوله :  "  مِنْ هَذِهِ  "  متعلِّقٌ بالفعل قَبْلَهُ ،   و "  مِنْ  "  لابتداء الغاية ،   و "  هذه  "  إشارةٌ إلى الظُّلماتِ ،   لأنها تجري مجرى المؤنثة الواحدة ،   وكذلك في  "  منها  "  تعود على الظلمات .  <br />
وقوله : { ومِنْ كُلِّ كَرْبٍ } عطف على الضمير المجرور بإعادةِ حرف الجر ،   وهو واجب عند البصريين ،   وقد تقدَّم .  <br />
و "  الكَرْبُ  "  غاية الغَمِّ الذي يأخذ النَّفْسَ .  <br />
قوله :  "  ثُمَّ أنْتُمْ تُشْرِكُونَ  "  يريد أنهم يُقِرُّونَ أن الذي يدعونه عند الشدة هو الذي يُنَجِّيهم ،   ثم يشركون معه الأصنامَ التي علموا أنها لا تضر ولا تنفع .
