"  من آية  "  فاعل زيدت فيه  "  مِنْ  "  لوُجُودِ الشرطين{[13196]} ،   فلا تَعَلُّقَ لها .  <br />
و "  من آيات  "  صفة ل  "  آية  "  ،   فهي في مَحَلِّ جرٍ على اللَّفْظِ ،   أو رفعٍ على الموضع{[13197]} .  <br />
وقال الوَاحِدِيِّ{[13198]} :  "  مِنْ  "  في قوله :  "  مِنْ آية  "  صفةٌ ل  "  آية  "  أي : آية لاستغراق الجنْسِ الذي يقع في النَّفْيِِ ،   كقولك : ما أتاني من أحدٍ  "  .  <br />
والثانية : في قوله :  "  من آياتِ رَبِّهِم  "  للتبعيض .  <br />
والمعنى : وما يظهر لهم دَلِيلٌ قط من الدَّلائِل التي يجب فيها النَّظَرُ والاعتبار ،   إلاَّ كانوا عنها مُعْرِضينَ ،   والمُرادُ بهم أهل  "  مكة  "  ،   والمرادُ بالآيات : إنْشِقاقُ القمر وغيره .  <br />
وقال عطاء : يريد : من آيات القرآن .  <br />
قوله :  "  إلاَّ كَانُوا  "  هذه الجملة الكَوْنِيَّةُ في مَحَلِّ نَصْبٍ على الحال ،   وفي صاحبها وجهان :<br />
أحدهما أنَّهُ الضميرُ في  "  تأتيهم  "  .  <br />
والثاني : أنَّهُ  "  مِنْ آيةٍ  "  وذلك لتخصيصها{[13199]} بالوَصْفِ .  <br />
و "  تأتيهم  "  يحتمل أن يكون ماضي المعنى{[13200]} لقوله :  "  كَانُوا  "  ،   ويحتمل أنْ يكون مُسْتَقْبَلَ المعنى ؛ لقوله  "  تَأتيهِمْ  "  .  <br />
واعلم أنَّ الفعْلَ الماضي لا يَقَعُ بَعْدَ  "  إلاَّ  "  بأحد شَرْطَيْنِ : إمَّا وقوعه بَعْدَ فِعْلٍ كهذه الآية ،   أو يقترن ب  "  قد  "  نحو :  "  ما زيدٌ إلاَّ قد قام  "  وهنا الْتِفَاتٌ من خطابه بقوله  "  خلقكم  "  إلى آخره إلى الغيبةِ بقوله :  "  وَمَا تَأتِيهم  "  .
