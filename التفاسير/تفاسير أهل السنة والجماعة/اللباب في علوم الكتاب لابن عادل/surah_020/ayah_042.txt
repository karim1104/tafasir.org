قوله : { اذهب أَنتَ وَأَخُوكَ بِآيَاتِي{[24373]} } لما قال : { واصطنعتك لِنَفْسِي } عقبه بذكر ما له اصطنعته ،   وهو الإبلاغ والأداء ،   و  "  الياء  "  في  " بِآيَاتِي "   بمعنى ( مع ){[24374]} ،   لأنهما لو ذهبا إليه بدون آيةٍ معهما لم يلزمه الإيمان ،   وذلك من أقوى الدلائل على فساد التقليد{[24375]} .  <br />
قال ابن عباس : يعني الآيات التسع التي بعث الله بها موسى{[24376]} .   وقيل{[24377]} : إنها{[24378]} العصا واليد ،   لأنهما{[24379]} اللذان جرى ذكرهما في هذا الموضع ،   ولم يذكر أنه -عليه السلام-{[24380]} أوتي قبل مجيئه إلى فرعون ،   لا بعد مجيئه حتى لقي فرعون فالتمس منه آية غير هاتين الآيتين ،   قال تعالى حكاية عن فرعون { إِن كُنتَ جِئْتَ بِآيَةٍ فَأْتِ بِهَآ إِن كُنتَ مِنَ الصادقين فألقى عَصَاهُ فَإِذَا هِيَ ثُعْبَانٌ مُّبِينٌ وَنَزَعَ يَدَهُ فَإِذَا هِيَ بَيْضَآءُ لِلنَّاظِرِينَ }{[24381]} ،   وقال : { فَذَانِكَ بُرْهَانَانِ مِن رَّبِّكَ إلى فِرْعَوْنَ وَمَلَئِهِ }{[24382]} .  <br />
فإن قيل : كيف يطلق لفظ الجمع على الاثنين ؟<br />
فالجواب من وجوه :<br />
أحدها{[24383]} : أن العصا كانت آيات ،   انقلابُها{[24384]} حيواناً ،   ثم إنها كانت في أول الأمر{[24385]} صغيرة ،   لقوله{[24386]} تعالى : { تَهْتَزُّ كَأَنَّهَا جَآنٌّ }{[24387]} ،   ثم كانت تعظم وهذه آية أخرى ،   ثم إنه كان عليه السلام يدخل في يده في فمها قلم تضره وهذه آية أخرى{[24388]} ،   ثم كانت تنقلب{[24389]} عصا وهذه آية أخرى ،   وكذلك اليد فإن بياضها آية ،   وشُعَاعَها آية أخرى ،   ثم زوالهما بعد ذلك آية أخرى ،   فدل ذلك على أنهما كانتا{[24390]} آيات كثيرة .  <br />
وثانيها{[24391]} : هَبْ أن العصا أمرٌ واحدٌ ولكن فيها آيات ،   لأن{[24392]} انقلابها حيةً يدل على وجود إله{[24393]} قادر على الكل عالم بالكل حكيم ،   ويدل على نبوة موسى ،   ويدل على جواز الحشر حيث انقلب الجماد حيواناً ،   فهذه آيات كثيرة ،   ولذلك قال{[24394]} : { إِنَّ أَوَّلَ بَيْتٍ وُضِعَ لِلنَّاسِ لَلَّذِي بِبَكَّةَ مباركا }{[24395]} .   .   .   إلى قوله .   .   .   { فِيهِ آيَاتٌ بَيِّنَاتٌ مَّقَامُ إِبْرَاهِيمَ وَمَن دَخَلَهُ كَانَ آمِناً }{[24396]} فهاهنا أولى .  <br />
وثالثها{[24397]} : قال بعضهم : أقل الجمع اثنان{[24398]} .  <br />
وقيل : معنى قوله :  "  بِآيَاتِي  "  أمُدكُّما بآياتي ،   وأظهر على أيديكما من الآيات ما تزاح به العلل من فرعون وقومه ،   والمعنى : فإن آياتي معكما كما يقال : اذهب فإن{[24399]} جندي معك{[24400]} أي : إنِّي أمدك بهم من احتجت{[24401]} .  <br />
وقيل : الآيات : العصا ،   واليد ،   وحل العقدة من لسانه ،   وذلك أيضاً معجزة{[24402]} .  <br />
قوله : { وَلاَ تَنِيَا } يقال :  "  وَنَى يَنِيَ وَنْيَاً كَوَعَد يَعِدُ وَعْداً ،   إذا فَتَرَ{[24403]} .  <br />
والوَنْيُ{[24404]} الفُتُور ،   ومنه : امرَأةٌ أَنَاةٌ{[24405]} ،   وصفوها بفتور القيام كناية عن ضخامتها .   قال زهير{[24406]} :<br />
مَنَّا الأَنَاةُ وَبَعْضُ القَوْمِ يَحْسَبُنَا *** أَنَّا بِطَاءٌ وفي إبْطَائِنَا سِرْعُ{[24407]}<br />
بكسر السين وفتح الراء مصدر ( سَرُع ) بفتح السين وضم الراء .  <br />
تقول : سَرُعَ سِرَعاً{[24408]} كصَغُرَ صِغَراً .  <br />
والأصل : ونَاةٌ ،   فأبدلوا الهمزة{[24409]} من الواو كأحَد وليس بالقياس{[24410]} ،   وفي الحديث :  "  إنَّ فِيكَ لَخَصْلَتَيْنِ يُحِبُّهُمَا اللهُ الحِلْمُ والأَنَاةُ  " {[24411]} .  <br />
والوَانِي : المقصِّر في أمره ،   قال الشاعر :<br />
فَمَا اَنَا بالوَانِي وَلاَ الضَّرْع الغُمْرِ{[24412]} *** .   .   .  <br />
ووَنَى{[24413]} فعلٌ لازم يتَعَدى{[24414]} وزعم بعضهم{[24415]} أنه يكون من أخوات ( زال وانفك ) فيعمل بشرط النفي{[24416]} أو شبهه{[24417]} عمل ( كان ) ،   فيقال{[24418]} :  " مَا وَنِيَ زيدٌ قائماً ،   وأنشد ابن مالك{[24419]} شاهداً على ذلك قوله :<br />
لاَ يَنشي الحُبُّ{[24420]} شِيمَةَ الحُبِّ مَا دَا *** مَ فَلاَ يَحْسَبَنَّهُ ذَا ارْعِوَاءِ{[24421]}<br />
أي : لا يزال الحُبُّ{[24422]} بضم الحاء شيمةَ الحِبِّ{[24423]} أي : بكسرها وهو المحب{[24424]} .   ومن منع ذلك يتأول{[24425]} البيت على حذف حرف الجر{[24426]} ،   لأنَّ{[24427]} هذا الفعل{[24428]} يتعدى تارة ب ( عَنْ ) وتارة ب ( في ){[24429]} يقال : ما ونَيْتُ عن حاجتك ،   أو : في حاجتك فالتقدير : لا يفتر الحب في شيمة المحب ،   وفيه مجاز بليغ وقد عدي في الآية الكريمة ب ( في ) .  <br />
قرأ يحيى بن وثَّاب  "  وَلاَ تِنِيَا  "  بكسر التاء{[24430]} إتباعاً لحركة النون ،   وسكن الياء في  "  ذِكْرِي  " {[24431]} .  <br />
وقرأ أهل الحجاز{[24432]} وأبو عمرو  "  لِنَفْسِيَ اذْهَبْ  " {[24433]} ذِكْرِي اذْهَبَا  " {[24434]} و { إِنَّ قَوْمِيَ اتخذوا }{[24435]} و{[24436]} { مِن بَعْدِيَ اسمه }{[24437]} بفتح الياء فيهن وافقهم{[24438]} أبو بكر في { مِن بَعْدِيَ اسمه } .  <br />
وقرأ الآخرون بإسكانها{[24439]} .  <br />
والمراد بالذكر تبليغ الرسالة .   وقيل : لا تفترا عن ذكر الله .   ( والحكمة فيه{[24440]} ){[24441]} أنَّ مَنْ ذكر جلالَ الله استخف غيره ،   فلا يخاف أحداً ،   ويقوى روحه بذلك الذكر فلا يضعف في{[24442]} مقصوده ،   ومن ذكر الله فلا بد وأن يكون ذاكراً إحسانه ( وذاكرُ إحسانه ){[24443]} لا يفتر في أداء أوامره .  <br />
وقيل : لاَ تَنِيَا في ذِكْرِي عند فرعون ،   وكيفية الذكر أن يذكرا{[24444]} لفرعون وقومه أنَّ الله تعالى لا يرضى منهم{[24445]} الكفر ،   ويذكرا{[24446]} لهم أمر الثواب والعقاب ،   والترغيب والترهيب{[24447]} .
