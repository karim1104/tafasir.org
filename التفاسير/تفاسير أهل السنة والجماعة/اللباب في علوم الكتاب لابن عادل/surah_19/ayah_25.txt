قوله تعالى : { وهزي إِلَيْكِ بِجِذْعِ النخلة } : يجوز أن تكون الباءُ في  "  بِجِذْعٍ  "  زائدة ،   كهي في قوله تعالى : { وَلاَ تُلْقُواْ بِأَيْدِيكُمْ } [ البقرة : 195 ] [ وقوله ] :<br />
 .   .   .   *** .   .   .   لا يَقْرَأْنَ بالسُّوَرِ{[21537]}<br />
وأنشد الطبريُّ -رحمة الله تعالى- : [ الطويل ]<br />
بِوَادٍ يمانٍ يُنْبِتُ السِّدْرَ صدْرُهُ *** وأسْفَلُهُ بالمَرْخِ والشَّبَهَانِ{[21538]}<br />
أي : ينبت المرخ أي : هُزِّي جذْعَ النَّخلةِ .  <br />
أو حركي جذْعَ النَّخلة .   قال الفرَّاء : العربُ تقول : هزَّهُ ،   وهزَّ به ،   وأخذ الخطام وأخذ بالخطام ،   وزوَّجتُك فلانة ،   وبفُلانةٍ ويجوز أن يكون المفعول محذوفاً ،   والجارُّ حالٌ من ذلك المحذوف ،   تقديره : وهُزِّي إليك رُطباً كائناً بجذع النخلة ،   ويجوز أن يكون هذا محمولاً على المعنى ؛ إذ التقدير : هُزِّي الثمرةَ بسبب هزِّ الجِذْع ،   أي : انفُضِي الجِذْع ،   وإليه نحا الزمخشريُّ ؛ فإنه قال :  "  أو افْعَلِي الهَزَّ  "   ؛ مقوله : [ الطويل ]<br />
 .   .   .   *** يَجْرَحْ فِي عَراقِيبِهَا نَصْلِي{[21539]}<br />
قال أبو حيَّان : وفي هذه الآية ،   وفي قوله تعالى : { واضمم إِلَيْكَ جَنَاحَكَ } [ القصص : 32 ] ما يردُّ على القاعدة المقرَّرةِ في علم النَّحو : من أنَّه لا يتعدَّى فعلُ المضمرِ المتَّصل إلى ضميره المتَّصلِ ،   إلاَّ في باب  "  ظنَّ  "  وفي لَفْظَتَيْ  "  فَقَدَ ،   وعدِمَ  "  لا يقالُ : ضربْتُكَ ،   ولا ضَرَبْتُني ،   أي : ضَربْتَ أنْتَ نَفْسَكَ ،   وضَربْتُ أنَا نفسي ،   وإنما يؤتى في هذا بالنَّفْسِ ،   وحكمُ المجرور بالحرف المنصوب ؛ فلا يقال : هَزَزْتَ إليك ،   ولا زيدٌ هزَّ إليه ؛ ولذلك جعل النحويُّون  "  عَنْ  "  و  "  عَلَى  "  اسمين في قول امرئ القيس : [ الطويل ]<br />
دَعْ عَنْكَ نَهْباً صيحَ في حُجُراتِهِ *** ولكِنْ حدياً ما حَدِيثُ الرَّواحِلِ{[21540]}<br />
وقول الآخر : [ المتقارب ]<br />
هَوِّنْ عليمَ فإنَّ الأمُورَ *** بكَفِّ الإلهِ مقاديرُهَا{[21541]}<br />
وقد ثبت بذلك كونهما اسمين ؛ لدُخُولِ حرفِ الجرِّ عليهما في قوله : [ الطويل ]<br />
غَدَتْ من عليْهِ بعدما تمَّ ظمؤهَا *** تَصِلُ وعن قَيْضٍ بِبَيْدَاءَ مَجْهَلِ{[21542]}<br />
وقول الآخر : [ البسيط ]<br />
فَقُلْتُ للرَّكْبِ لمَّا أن عَلا بِهِمُ *** مِنْ عَنْ يمينِ الحُبَيَّا نظرةٌ قَبْلُ{[21543]}<br />
وأمَّا  "  إلى  "  فحرفٌ بلا خلافٍ ،   فلا يمكنُ فيها أن تكون اسماً ؛ ك  "  عَنْ  "  و  " عَلَى  "  ثم أجاب : بأنَّ  "  إليكِ  "  في الآيتين لا تتعلقُ بالفعل قبله ،   إنما تتعلَّقُ بمحذوفٍ على جهةِ البيان ،   تقديره :  "  أعني إليك  "  قال :  "  كما تأولوا ذلك في قوله : { إِنِّي لَكُمَا لَمِنَ الناصحين } [ الأعراف : 21 ] في أحد الأوجه  "  .  <br />
قال شهاب الدين -رضي الله تعالى عنه- : وفيه ذلك جوابان آخران :<br />
أحدهما : أن الفعل الممنوع إلى الضمير المتصل ،   إنما هو من حيث يكون الفعلُ واقعاً بذلك الضمير ،   والضمير محلٌّ له ؛ نحو :  "  دَعْ عنْكَ  "  و  "  هوِّنْ عليْكَ  "  وأمَّا الهزُّ والضمُّ ،   فليسا واقعين بالكاف ،   فلا محذور .  <br />
والثاني : أنَّ الكلام على حذف مضافٍ ،   تقديره : هُزِّي إلى جهتك ونحوك واضمم إلى جهتك ونحوك .  <br />
فصل في المراد بجذع النخلة<br />
قال [ القفال ]{[21544]} {[21545]} : الجِذْعُ من النَّخلة : هو الأسفل ،   وما دُون الرَّأس الذي عليه الثَّمرة .  <br />
وقال قطربٌ : كُلُّ خشبة في أصل شجرة ،   فهي جذعٌ .  <br />
قوله :  "  تُسَاقِطْ  "  قرأ حمزةُ{[21546]}  "  تَسَاقَطْ  "  بفتح التاء ،   وتخفيف السين ،   وفتح القاف ،   والباقون -غير حفص- كذلك إلا أنَّهم شدَّدُوا السِّين ،   وحفصٌ ،   بضم التاء ،   وتخفيف السين ،   وكسر القاف .  <br />
فأصلُ قراءةِ غير حفص  "  تتساقطْ  "  بتاءين ،   مضارع  "  تَساقَطَ  "  فحذف حمزة إحدى التاءين تخفيفاً ؛ نحو : { تَنَزَّلُ } [ القدر : 4 ] و { تَذَكَّرُونَ } [ الأنعام : 152 ] ،   والباقون أدغمُوا في السِّين ،   وقراءة حفص مضارعُ  "  سَاقَطَ  "  .  <br />
وقرأ الأعمش ،   والبراء [ بنُ عازبٍ ]  "  يَسَّاقَطْ  "  كالجماعة ،   إلا أنه بالياء من تحتُ ،   أدغم التاء في السِّين ؛ إذ الأصلُ :  "  يتساقَط  "  فهو مضارعُ  "  اسَّاقَطَ  "  وأصله  "  يَتَساقَطُ  "  فأدغمَ ،   واجتلبتْ همزة الوصل ؛ ك  "  ادَّارَأ  "  في  "  تَدَارَأ "  .  <br />
ونْقل عن أبي حيوة ثلاثُ قراءاتٍ :<br />
وافقهُ مسروقٌ في الأولى ،   وهي  "  تُسْقِطْ  "  بضم التاء ،   وسكون السين ،   وكسر القاف من  "  أسْقَطَ  "  .  <br />
والثانية : كذلك إلا أنه بالياء من تحتُ .  <br />
الثالثةُ كذلك إلاّ أنه رفع  "  رُطَباً جَنِيًّا  "  بالفاعلية .  <br />
وقُرِئَ{[21547]}  "  تَتَسَاقَط  "  بتاءين من فوقُ ،   وهو أصل قراءة الجماعة ،   وتَسْقُط ويَسْقُط ،   بفتح التاء والياء ،   وسكون السين ،   وضمِّ القاف ،   فرفعُ الرطب بالفاعلية ،   وتعطي من الأفعال ما يوافقه في القراءات المتقدمة ،   ومن قرأ بالتاء من فوق ،   فالفعل مسندٌ : إمَّا للنَّخلة ،   وإمَّا للثمرةِ المفهومة من السياق ،   وإمَّا للجذْعِ ،   وجاز تأنيثُ فعله ؛ لإضافته إلى مؤنَّث ؛ فهو كقوله : [ الطويل ]<br />
 .   .   .   *** كَمَا شَرقَتْ صَدْرُ القناةِ مِن الدَّمِ{[21548]}<br />
وكقراءة { يَلْتَقِطْهُ بَعْضُ السيارة } [ يوسف : 10 ] ومن قرأ بالياء من تحت ،   فالضميرُ للجذْعِ ،   وقيل : للثَّمر المدلول عليه بالسيِّاق .  <br />
وأمَّا نصب  "  رُطَباً  "  فلا يخرجُ عن كونه تمييزاً ،   أو حالاً موطِّئة ،   إن كان الفعل قبله لازماً ،   أو مفعولاً به ،   إن كان الفعل متعدِّياً ،   [ والذَّكِيُّ ] يردُّ كلَّ شيءٍ إلى ما يليقُ به من القراءات ،   وجوَّز المبرِّد في نصبه وجهاً غريباً : وهو أن يكون مفعولاً به ب  "  هُزِّي  "  وعلى هذا ،   فتكون المسألةُ من باب التنازع في بعض القراءات : وهي أن يكون الفعل فيها متعدِّياً ،   وتكون المسألةُ من إعمال الثاني ،   للحذف من الأوَّل .  <br />
وقرأ{[21549]} طلحة بن سليمان  "  جنيًّا  "  بكسرِ الجيم إتباعاً لكسرةِ النون .  <br />
والرُّطبُ : اسم جنسٍ برطبة ؛ بخلاف  "  تُخَم  "  فإنَّه جمعٌ لتخمة ،   والفرقُ : أنهم لزمُوا تذكيرهُ ،   فقالوا : هو الرطبُ ،   وتأنيث ذاك ،   فقالوا : هي التُّخَمُ ،   فذكَّرُوا  " الرُّطَب  "  باعتبار الجنس ،   وأنَّثُوا  "  التُّخَمَ  "  باعتبار الجمعيَّة ،   وهو فرقٌ لطيفٌ ،   ويجمعُ على  "  أرطابٍِ  "  شذوذاً كربع وأرباع ،   والرُّطب : ما قطع قبل يبسه وجفافه ،   وخصَّ الرُّطب بالرُّطبِ من التَّمر ،   وأرطبَ النَّخْلُ ؛ نحو : أتْمَرَ وأجْنَى .  <br />
والجَنِيُّ : ما طاب ،   وصلح للاجتناء ،   وهو  "  فَعِيلٌ  "  بمعنى مفعول أي رُطَباً مَجنيًّا ،   وقيل : بمعنى فاعلٍ ،   أي : طريًّا ،   والجنى والجنيُّ أيضاً : المُجْتَنَى من العسلِ ،   وأجْنَى الشَّجَرُ : أدْرَكَ ثمرهُ ،   وأجنتِ الأرضُ : كَثُرَ جناها ،   واستعير من ذلك  "  جنى فلانٌ جنايةً  "  كما استعير  "  اجْتَرَمَ جَريمَةً  "  .
