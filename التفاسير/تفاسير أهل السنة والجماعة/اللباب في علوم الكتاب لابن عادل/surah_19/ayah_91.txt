قوله : { أنْ دَعَواْ }{[22711]} في محله{[22712]} خمسة أوجه :<br />
أحدها{[22713]} : أنه في محل نصب على المفعول من أجله ،   قاله أبو البقاء{[22714]} ،   والحوفي{[22715]} ،   ولم يُبيِّنَا ما العامل فيه ،   ويجوز أن يكون العامل  "  تَكَادُ  "  ،   أو  " تَخُرُّ  "  ،   أو  "  هَدَّا  "  ،   أي : تَهُدُّ لأن دعوا ،   ولكن شرطُ النصب هنا{[22716]} مفقود ،   وهو اتحاد{[22717]}  "  الفاعل في المفعول له والعامل فيه{[22718]} ،   فإن عنيا على أنه على إسقاط اللام مطرد في  "  أنْ  "  فقريب  " {[22719]} .   وقال الزمخشري : وأن يكون منصوباً بتقدير سقوط اللام  " {[22720]} وإفضاء الفعل ،   أي هدَّا أن دعوا  " {[22721]} ،   علل الخرور بالهدِّ ،   والهدُّ{[22722]} بدعاء الولد للرحمن{[22723]} .  <br />
فهذا تصريح منه على أنه بإسقاط{[22724]} الخافض .    "  وليس مفعولاً له صريحاً{[22725]}  "  .  <br />
الوجه الثاني : أن يكون مجروراً بعد إسقاط الخافض  " {[22726]} كما هو مذهب الخليل{[22727]} والكسائي{[22728]} .  <br />
والثالث : أنه بدل من الضمير في  "  مِنْهُ  " {[22729]} كقوله :<br />
عَلَى حَالةٍ لوْ أنَّ في القَوْمِ حَاتِماً *** عَلى جُودهِ لضن{[22730]} بالمَاءِ حاتمِ{[22731]}<br />
 "  بجر  "  حاتم  " {[22732]} الأخير بدلاً من الهاء في  "  جوده  "  .  <br />
قال أبو حيان : وهو بعيد لكثرة الفصل بين البدل والمبدل منه بجملتين{[22733]} .  <br />
الوجه{[22734]} الرابع : أن يكون مرفوعاً ب  "  هَدَّا  "  .   قال الزمخشري : أي هدَّهَا دعاءُ الولدِ للرحمن{[22735]} .   قال أبو حيان : وفيه بعدٌ ،   لأن الظاهر في  "  هَدَّا  "  أن يكون مصدراً توكيدياً ،   والمصدر التوكيدي لا يعمل{[22736]} ،   ولو فرضناه غير توكيدي لم يعمل بقياس إلا إذا كان أمراً ،   أو مستفهماً عنه نحو ضرباً زيداً ،   وأضربا{[22737]} زيداً ؟ على خلاف فيه{[22738]} ،   وأما إن كان خبراً كما قدَّره الزمخشري ،   أي : هدَّها دعاء الولد للرحمن{[22739]} .  <br />
فلا ينقاس{[22740]} ،   بل ما جاء من ذلك هو نادر كقول امرئ القيس{[22741]} :<br />
وقُوفاً بِهَا صَحْبِي عَليَّ مَطِيُّهُمْ *** يقُولونَ لا تَهْلَكْ أسّى وتجمَّلِ{[22742]}<br />
أي : وقَفَ صَحْبِي{[22743]} .  <br />
الخامس : أنَّه خبر مبتدأ محذوفٍ ،   تقديره : الموجب لذلك دعاؤهم .   كذا قدره أبو البقاء{[22744]} .   و  "  دَعَا  "  يجوز أن يكون بمعنى سمَّى ،   فيتعدى لاثنين ،   ويجوز جر ثانيهما بالباء{[22745]} ،   قال الشاعر :<br />
دَعَتْنِي أخَاهَا أمُّ عَمْرٍو ولمْ أكُنْ *** أخَاهَا ولَمْ أرْضَعْ لهَا بِلبَانِ<br />
دَعَتْنِي أخَاهَا  "  بَعْدَمَا كَانَ بَيْنَنَا *** مِنَ الفِعْلِ ما لا يفعلُ الأخوانِ{[22746]}  " {[22747]}<br />
وقول الآخر :<br />
ألا رُبَّ مَنْ يُدْعَى نَصِيحاً وإن تَغِبْ *** تَجِدْهُ بِغَيْبٍِ مِنْكَ غَيْرَ نَصِيحِ{[22748]}<br />
وأولهما في الآية محذوف ،   قال الزمخشري : طلباً للعموم والإحاطة بكل ما يدعى له ولد ،   ويجوز أن يكون من  "  دَعَا  "  بمعنى نسب الذي مطاوعه ما في قوله- عليه السلام- :  "  مَنِ ادَّعى إلى غَيْرِ موالِيهِ  " {[22749]} ،   وقول الشاعر :<br />
إنَّا بَنِي نَهْشَلٍ لا نَدَّعِي لأب *** عَنْهُ ولا هُوَ بالأبْنَاءِ يَشْرِينَا{[22750]}<br />
لأي : لا ننتَسِبُ إليه{[22751]} .  <br />
 "  يَنْبَغِي  " {[22752]} مضارع انْبَغَى ،   وانْبَغَى مطاوعٌ لبغى ،   أي : طلب ،   و  "  أنْ يتَّخِذَ  "  فاعله{[22753]} .   وقد عد ابن مالك{[22754]}  "  يَنْبَغِي  "  في الأفعال التي لا تتصرف{[22755]} .  <br />
وهو مردودٌ عليه ،   لأنه قد سُمِعَ فيه الماضي قالوا : انْبَغَى{[22756]} .   وكرَّر لفظ  " الرَّحْمَنِ  "  تنبيهاً على أنه - تعالى- هو الرحمنُ وحدهُ ،   لأن أصول النعم وفروعها ليست إلا منه{[22757]} .  <br />
