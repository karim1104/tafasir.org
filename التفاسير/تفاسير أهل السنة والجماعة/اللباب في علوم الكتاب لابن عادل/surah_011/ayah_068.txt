ثم قال تعالى : { كَأَن لَّمْ يَغْنَوْاْ فِيهَآ } كأنَّهُم لم يُوجدوا .   والمغنى المقام الذي يقيمُ الحي فيه يقال : غني الرَّجُلُ بمكان كذا إذا أقام به .  <br />
قوله : { أَلاَ إِنَّ ثَمُودَ } قرأ حمزة{[18867]} وحفص هنا { أَلاَ إِنَّ ثَمُودَ } ،   وفي الفرقان : { وَعَاداً وَثَمُودَاْ } [ الآية : 38 ] وفي العنكبوت : { وَعَاداً وَثَمُودا وَقَدْ تَبَيَّنَ لَكُم } } [ الآية : 37 ] ،   وفي النجم : { وَثَمُودا فَمَآ أبقى } [ الآية : 51 ] جميعُ ذلك بمنع الصرف ،   وافقهم أبو بكر على الذي في النَّجْم .  <br />
وقوله : { أَلاَ بُعْداً لِّثَمُودَ } منعه القراء الصرفَ إلاَّ الكسائيَّ فإنَّهُ صرفه{[18868]} ،   وقد تقدَّم أنَّ من منع الصرف جعله اسماً للقبيلةِ ،   ومن صرف جعله اسماً للحيِّ ،   أو إلى الأبِ الأكبرِ ؛ وأنشد على المنع : [ الوافر ]<br />
ونَادَى صالحٌ يَا ربِّ أنْزِلْ *** بآلِ ثمُودَ مِنْكَ غداً عَذَابَا{[18869]}<br />
وأنشد على الصَّرف قوله : [ الطويل ]<br />
دَعَتْ أمُّ عَمْرٍو أمْرَ شرٍّ عَلِمْتُهُ *** بأرْضِ ثمُودٍ كُلِّهَا فأجَابَهَا{[18870]}<br />
وقد تقدَّم الكلامُ على اشتقاق هذه اللفظة في سورةِ الأعراف .
