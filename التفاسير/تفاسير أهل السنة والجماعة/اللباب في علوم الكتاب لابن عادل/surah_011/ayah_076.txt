فقالت الرُّسُلُ عند ذلك لإبراهيم - عليه الصلاة والسلام - : { يا إِبْرَاهِيمُ أَعْرِضْ عَنْ هذآ } أي : أعرض عن هذا المقال ،   ف { إِنَّهُ قَدْ جَآءَ أَمْرُ رَبَّكَ } ،   أي : عذابُ ربِّك وحُكم ربِّك { إِنَّهُمْ آتِيهِمْ } : نازلٌ بهم { عَذَابٌ غَيْرُ مَرْدُودٍ } ،   أي : غير مصروف عنهم .  <br />
قوله : { آتِيهِمْ عَذَابٌ } يجوز أن يكون جملة من مبتدأ وخبر في محلِّ رفع خبراً ل  "  إنَّهُمْ  "  ،   ويجوز أن يكون  "  آتِيهِمْ  "  الخبر  "  عَذابٌ  "  المبتدأ ،   وجاز ذلك لتخصُّصِه بالوصفِ ،   ولتنكير  "  آتِيهِمْ  "  ؛ لأنَّ إضافته غيرُ محصنةٍ .  <br />
ويجُوزُ أن يكون  "  آتِيهِمْ  "  خبر  "  إنَّ  "  ،   و "  عذابٌ  "  فاعلٌ به ،   ويدلُّ على ذلك قراءةُ عمرو بن هرم  "  وإنَّهُمْ أتاهُمْ  "  بلفظ الفعل الماضي .  <br />
