قوله : { وَلَقَدْ جَآءَتْ رُسُلُنَآ إِبْرَاهِيمَ بالبشرى } القصة .  <br />
قال النحويون : دخلت كلمة  "  قَدْ  "  ها هنا لأنَّ السَّامع لقصص الأنبياء - عليهم الصلاة والسلام - يتوقع قصة بعد قصة ،   و "  قَدْ  "  للتوقع ،   ودخلت اللاَّم في  "  لَقَدْ  "  تأكيداً للخبر .  <br />
فصل<br />
لفظ  "  رُسُلُنَا  "  جمع وأقله ثلاثة ،   فهذا يدلُّ على أنهم كانوا ثلاثة ،   والزَّائِدُ على هذا العددِ لا يثبتُ إلاَّ بدليل آخر ،   وأجمعُوا على أنَّ الأصلَ فيهم كان جبريلُ - عليه السَّلامُ - .  <br />
قال ابنُ عبَّاس - رضي الله عنهما - :  "  كانوا ثلاثة جبريل ،   وميكائيل ،   وإسرافيل{[18871]} ،   وهم المذكورون في الذَّاريات { هَلْ أَتَاكَ حَدِيثُ ضَيْفِ إِبْرَاهِيمَ } [ الذاريات : 24 ] وفي الحجر { وَنَبِّئْهُمْ عَن ضَيْفِ إِبْرَاهِيمَ } [ الحجر : 51 ] .  <br />
وقال الضحَّاكُ :  " كانوا تسعة " {[18872]} .   وقال محمد بن كعب - رضي الله عنه -  "  كان جبريل ومعه سبعة{[18873]}  "  وقال السُّديُّ :  "  أحد عشر ملكاً " {[18874]} وقال مقاتل :  "  كانوا اثني عشر ملكاً على صور الغلمان الوضاء وجوههم " {[18875]} .  <br />
 "  بالبشرى  "  بالبشارة بإسحاق ويعقوب وقيل : بسلامة لوطٍ ،   وإهلاك قومه .  <br />
قوله : { قَالُواْ سَلاَماً } : في نصبه وجهان :<br />
أحدهما : أنَّه مفعولٌ به ،   ثم هو محتملٌ لأمرينِ :<br />
أحدهما : أن يراد قالوا هذه اللفظ بعينه ،   وجاز ذلك لأنَّه يتضمَّن معنى الكلام .  <br />
الثاني : أنَّهُ أراد قالوا معنى هذا اللفظ ،   وقد تقدَّم نحو ذلك في قوله تعالى : { وَقُولُواْ حِطَّةٌ } [ البقرة : 58 ] .  <br />
وثاني الوجهين : أن يكون منصوباً على المصدر بفعل محذوفٍ ،   وذلك الفعلُ في محلِّ نصب بالقول ،   تقديره : قالوا : سَلَّمْنَا سلاماً ،   وهو من باب من ناب فيه المصدرُ عن العامل فيه ،   وهو واجب الإضمار .  <br />
قوله :  "  سَلاَمٌ  "  في رفعه وجهان :<br />
أحدهما : أنَّهُ مبتدأ وخبره محذوفٌ ،   أي : سلامٌ عليكم .  <br />
والثاني : أنَّه خبر مبتدأ محذوف أي : أمْرِي أو قَوْلِي سلام .  <br />
وقد تقدَّم أنَّ الرفع أدلُّ على الثُّبوتِ من النَّصْبِ [ الفاتحة : 2 ] ،   والجملة بأسرها - وإن كان أحد جزأيها محذوفاً - في محلِّ نصب بالقول ؛ كقوله : [ الطويل ]<br />
إذَا ذَقْتُ فَاهَا قُلْتُ : طَعْمُ مُدَامَةٍ *** .   .   .  {[18876]}<br />
وقرأ الاخوان{[18877]} :  "  قَالَ سِلْم  "  هنا وفي سورة الذَّاريات بكسر السين وسكون اللاَّم .  <br />
ويلزم بالضرورة سقوطُ الألف ،   قال الفرَّاءُ :  "  هُما لغتان كحِرْم وحَرَامٍ وحِلٍّ وحلالٍ  "  ؛ وأنشد : [ الطويل ]<br />
مَرَرْنَا فَقُلْنَا : إيهِ سِلْمٌ فَسلَّمْتْ *** كَمَا اكْتَلَّ بالبَرْقِ الغَمَامُ اللَّوَائِحُ{[18878]}<br />
يريد : سلامٌ ؛ بدليل : فسلَّمَتْ .  <br />
وقال الفارسي :  "  السِّلْم  "  بالكسر ضد الحربِ ،   وناسبَ ذلك لأنَّهم لمَّا امتنعوا من تناول ما قدَّمهُ إليهم ،   أنكرهم ،   وأوجس منهم خيفة ،   فقال : أنا سِلْم ،   أي : مُسَالمكم غيرُ محارب لكم ،   فلم تمتنعوا من تناول طعامي ؟<br />
قال ابنُ الخطيب{[18879]} - رحمه الله - : وهذا بعيدٌ ؛ لأنَّ على هذا التقدير ينبغي أن يكون تكلُّم إبراهيم - عليه الصلاة والسلام - بهذا اللفظ بعد إحضار الطَّعام ،   والقرآن يدل على أنَّ هذا الكلام قبل إحضار الطَّعام ؛ لأنَّه تعالى قال : { قَالُواْ سَلاَماً قَالَ سَلاَمٌ فَمَا لَبِثَ أَن جَآءَ بِعِجْلٍ حَنِيذٍ } والفاءُ للتَّعقيب ،   فدلَّ على أنَّ مجيئَهُ بالعجل الحنيذِ بعد السَّلام .  <br />
فصل<br />
أكثر ما يستعمل  "  سلامٌ عليكم  "  منكّراً ؛ لأنَّهُ في معنى الدُّعاءِ كقولهم : خير بين يديك .  <br />
فإن قيل : كيف جاز الابتداء بالنَّكرةِ ؟ .  <br />
فالجوابُ : إذا وصفت جاز ،   فإذا قلت :  "  سلامٌ عليكم  "  فالتَّنكير هُنا يدلُّ على الكمال والتَّمام ،   فكأنه سلام كامل تام عليك ،   ونظيره قوله تعالى : { سَلاَمٌ عَلَيْكَ سَأَسْتَغْفِرُ لَكَ ربي } [ مريم : 47 ] وقوله : { سَلاَمٌ قَوْلاً مِّن رَّبٍّ رَّحِيمٍ } [ يس : 58 ] وقوله : { وَالمَلاَئِكَةُ يَدْخُلُونَ عَلَيْهِمْ مِّن كُلِّ بَابٍ سَلاَمٌ عَلَيْكُم } [ الرعد : 23 ،   24 ] .  <br />
وأما قوله : { والسلام على مَنِ اتبع الهدى } [ طه : 47 ] فالمراد منه الماهية والحقيقة .  <br />
قال ابنُ الخطيب : قوله :  "  سَلاَمٌ عَلَيْكُمْ  "  أكملُ من قوله :  "  السَّلامُ عليْكُم  "  ؛ لأنَّ التَّنْكيرَ يُفيدُ المبالغة والتَّمام ،   والتعريف لا يفيدُ إلاَّ الماهية .  <br />
قوله :  "  فَمَا لَبِثَ  "  يجُوزُ في  "  ما  "  هذه ثلاثة أوجه :<br />
أظهرها : أنَّها نافيةٌ ،   وفي فاعل  "  لَبِثَ  "  حينئذٍ وجهان :<br />
أحدهما : أنَّه ضميرُ إبراهيم - عليه الصلاة والسلام - أي : فَمَا لَبِثَ إبراهيم ،   وإن جاء على إسقاطِ الخافض ،   فقدَّرُوه بالباء وب  "  عَنْ  "  وب  "  في  "  ،   أي : فمَا تأخَّر في أنْ ،   أو بأن ،   أو عن أن .  <br />
والثاني : أنَّ الفاعل قوله :  "  أنْ جَاءَ  "  ،   والتقدير : فَما لبثَ ،   أي : ما أبْطَأ ولا تأخَّر مجيئُه بعجل سمين .  <br />
وثاني الأوجه : أنَّها مصدريةٌ .  <br />
وثالثها : أنَّها بمعنى الذي .   وهي في الوجهين الأخيرين مبتدأ ،   وإن جاء خبرهُ على حذفِ مضاف تقديره : فلُبْثُه - أو الذي لبثه - قَدْرَ مجيئه .  <br />
قال القرطبيُّ{[18880]} : قوله :  "  أنْ جَاءَ  "  معناه : حتَّى جَاءَ .  <br />
والحَنِيذُ : المَشْوِيُّ بالرَّضْفِ في أخدود كفعل أهل البادية يشوون في حفرةٍ من الأرض بالحجارة المُحَمَّاة ،   يقال : حَنَذْتُ الشَّاة أحْنِذُهَا حَنْذاً فهي حَنِيذ ،   أي : مَحْنُوذة .  <br />
وقيل : حَنيذ بمعنى يَقْطُرُ دَسَمُه من قولهم : حَنَذْتُ الفرس ،   أي : سُقْتُه شَوْطاً أو شَوْطَيْن وتضع عليه الجُلَّ في الشمس ليعرق .
