قال تعالى : { وَقَالَ اركبوا فِيهَا بِسْمِ الله مَجْرَاهَا } .  <br />
يجُوزُ أن يكون فاعلُ  "  قَالَ  "  ضمير نوح - عليه الصلاة والسلام - ،   ويجوزُ أن يكون ضمير الباري - تعالى جل ذكره - أي : وقال الله لنوح ومنه معهُ .   و "  فِيهَا  "  متعلقٌ ب  "  ارْكَبُوا  "  وعُدِّي ب  "  في  "  لتضَمُّنه معنى  "  ادخلوا فيها راكبين  "  أو سيروا فيها .  <br />
وقيل : تقديره : اركبوا الماء فيها .   وقيل :  "  في  "  زائدة للتَّوكيد .  <br />
والركوب : العلو على ظهر الشيءِ ،   ومنه ركوب الدَّابة ،   وركوب السَّفينة ،   وركوب البحر ،   وكل شيء علا شيئاً ،   فقد ركبه ،   ويقال : ركبه الدَّين .   قال الليثُ - رحمه الله - : وتسمي العربُ من يركبُ السَّفينة : رُكَّابَ السَّفينة ،   وأمَّا الركبانُ ،   والأركُوبُ ،   والرَّكْبُ : فركَّابُ الدَّوابِّ .  <br />
قال الواحدي : ولفظة  "  فِي  "  في قوله  "  ارْكَبُوا فيهَا  "  لا يجوز أن تكون من صلة الركوب ؛ لأنَّه يقال : ركبت السفينة ولا يقال : ركبت في السَّفينة ،   بل الوجهُ أن يقول مفعول  "  ارْكَبُوا  "  محذوف والتقدير :  "  اركبوا الماء في السَّفينة  "  .  <br />
وأيضاً يجوز أن تكون فائدة هذه الزيادة ،   أنَّه أمرهم أن يكونوا في جوف الفلك لا على ظهرها فلو قال : اركبوها ،   لتوهَّمُوا أنَّه أمرهم أن يكونوا على ظهر السَّفينة .  <br />
قال قتادةُ : ركبُوا السَّفينة يوم العاشر من شهر رجب ؛ فسارُوا مائةً وخمسين يوماً ،   واستقرَّتْ على الجُودي شهراً ،   وكان خروجهم من السفينة يوم عاشوراء من المحرَّمِ{[18802]} .  <br />
قوله : { بِسْمِ الله مَجْرَاهَا وَمُرْسَاهَا } .  <br />
يجوز أن يكون هذا الجار والمجرور حالاً من فاعل  "  اركبوا  "  أو من  "  ها  "  في  "  فيها  "  ويكونُ  "  مجراها  "  ،   و "  مرساها  "  فاعلين بالاستقرار الذي تضمَّنهُ الجارُّ لوقوعه حالاً .   ويجوز أن يكون  "  بِسْمِ اللهِ  "  خبراً مقدَّماً ،   و "  مَجْراها  "  مبتدأ مؤخراً ،   والجملة أيضاً حالٌ ممَّا تقدَّم ،   وهي على كلا التقديرين حالٌ مقدرةٌ كذا أعربه أبو البقاء ،   وغيره .   إلاَّ أنَّ مكيّاً منع ذلك لخلو الجملة من ضمير يعود على ذي الحال إذا أعربنا الجملة أو الجارَّ حالاً من فاعل  "  ارْكَبُوا  "  قال : ولا يَحْسُنُ أن تكون هذه الجملة حالاً من فاعل  "  اركبُوا  "  ؛ لأنَّه لا عائد في الجملةِ يعودُ على الضمير في  "  اركبُوا  "  لأن المضمر في  "  بِسْمِ اللهِ  "  إنْ جعلته خبراً ل  "  مَجْراهَا  "  فإنَّما يعودُ على المبتدأ ،   وهو مجراها ،   وإن رفعت  "  مَجْرَاهَا  "  بالظَّرفِ لم يكن فيه ضمير الهاءِ في  "  مَجْراهَا  "  وإنما تعودُ على الضمير في  "  فِيهَا  "  .  <br />
وإذا نصبت  "  مَجْرَاهَا  "  على الظرف عمل فيه  "  بِسْمِ الله  "  وكانت الجملةُ حالاً من فعل  "  ارْكَبُوا  "  .  <br />
وقيل : بِسْمِ اللهِ  "  حال من فاعل  "  ارْكَبُوا  "  و "  مَجْراهَا ومُرْسَاهَا  "  في موضع الظرف المكاني ،   أو الزماني .   والتقدير : اركبوا فيها مُسَمِّين موضع جريانها ،   ورُسُوِّها ،   أو وقت جريانها ورسوِّها .  <br />
والعامل في هذين الظرفين حينئذٍ ما تضمَّنه  "  بِسْمِ اللهِ  "  من الاستقرار ،   والتقدير : اركبوا فيها مُتبرِّكين باسم الله في هذين المكانين ،   أو الوقتين .  <br />
قال مكي : ولا يجوز أن يكون العاملُ فيهما  "  ارْكَبُوا  "  ؛ لأنه لم يُرِدْ : اركبُوا فيها في وقتِ الجَرْي ،   والرسُوِّ ،   إنَّما المعنى : سمُّوُا اسم الله في وقت الجَرْيِ والرُّسُوِّ .  <br />
ويجُوزُ أيضاً أن يكون  "  مَجْرَاهَا ومُرْسَاها  "  مصدرين ،   و "  بِسْمِ الله  "  حالٌ كما تقدَّم ،   رافعاً لهذين المصدرين على الفاعليَّة أي : استقرَّ بسم الله إجراؤها ،   وإرساؤها ،   ولا يكونُ الجارُّ حينئذٍ إلاَّ حالاً من  "  هَا  "  في  "  فيها  "  لوجود الرابط ،   ولا يكونُ حالاً من فاعل  "  اركبُوا  "  لعدم الرَّابط .   وعلى هذه الأعاريب يكونُ الكلامُ جملةً واحدةً .  <br />
ويجوز أن يكون { بِسْمِ الله مَجْرَاهَا وَمُرْسَاهَا } جملة مستأنفة لا تعلُّق لها بالأولى من حيث الإعراب ،   ويكون قد أمرهم في الجملة الأولى بالرُّكُوب ،   وأخبر أنَّ مجراها ومرساها باسم الله .  <br />
فجرت ،   وإذا أراد أن ترسو قال : بسم الله .  <br />
قال بعضُ المفسِّرين : كان نوح عليه الصلاة والسلام إذا أراد إجراء السفينة قال :  "  بِسْمِ اللهِ مجرَاها  "  فتجري ،   وإذا أراد أن ترسُو قال :  "  بِسْمِ الله مرْساهَا  "  فترسو ،   فالجملتان محكيتان ب  "  قَالَ  "  .  <br />
وقرأ الأخوان وحفص  "  مَجْرَاها  "  بفتح الميم ،   والباقون{[18803]} بضمها .   واتَّفق السَّبعة على ضمِّ ميم  "  مُرْسَاها  "  .   وقرأ ابن مسعود ،   وعيسى{[18804]} الثقفي وزيد بن علي ،   والأعمش  "  مَرْسَاها  "  بفتح الميم أيضاً .  <br />
فالضمُّ فيهما ،   لأنهما من  "  أجْرَى وأرْسَى  "  ،   والفتح لأنَّهُما من  "  جَرَتْ ورَسَتْ  "  وهما : إمَّا ظرفا زمان أو مكان أو مصدران على ما سبق من التقادير .  <br />
وقرأ الضحاك ،   والنخعي ،   وابن وثابٍ ،   ومجاهدٌ ،   وأبو رجاء ،   والكلبي{[18805]} ،   والجحدري ،   وابن جندب مجريها ومرسيها بكسر الراء ،   والسين بعدهما ياء صريحة ،   وهما اسما فاعلين من  "  أجرى وأرسى  "  ،   وتخريجهما على أنهما بدلان من اسم الله .  <br />
قال ابنُ عطيَّة وأبو البقاء ،   ومكي : إنَّهما نعتان للهِ - تعالى - ،   وهذا الذي ذكروه إنَّما يتمُّ على تقدير كونهما معرفتين بتمحض الإضافة ،   وقال الخليلُ :  "  إنْ كُلَّ إضافةٍ غيرُ محضةٍ قد تُجعل محضة إلاَّ إضافة الصفةِ المشبهة ،   فلا تتمحَّض  "  .  <br />
وقال مكي :  "  ولو جعلت  "  مَجْراهَا  "  ،   و "  مُرْسَاها  "  في موضع اسم الفاعل لكانت حالاً مقدرة ،   ولجاز ذلك وكانت في موضع نصبٍ على الحال من اسم الله تعالى  "  .  <br />
والرُّسوُّ : الثَّبات ،   والاستقرار ،   يقال : رَسَا يَرْسُو وأرْسَيْتُهُ أنَا ؛ قال : [ الكامل ]<br />
فَصَبرْتُ نفْساً عند ذلكَ حُرَّةً *** تَرْسُو إذَا نَفْسُ الجبانِ تَطَلَّع{[18806]}<br />
أي : تَثْبُتُ وتَسْتَقِرُّ عندما تضطربُ وتتحرك نفسُ الجبان .  <br />
قوله : { إِنَّ رَبِّي لَغَفُورٌ رَّحِيمٌ } فيه سؤال ،   وهو إن كان ذلك وقت إظهارِ الضر ،   فكيف يليق به هذا الذكر ؟ .  <br />
والجوابُ : لعل القوم الذين ركبوا السَّفينة اعتقدُوا في أنفسهم أنَّا إنَّما نجونا ببركةِ علمنا فالله تعالى نبَّههم بهذا الكلام لإزالة ذلك العجب ،   فإنَّ الإنسان لا ينفكُّ من أنواع الزلاتِ وظلمات الشبهات ،   وفي جميع الأحوال ،   فهو محتاجٌ إلى إعانةِ الله ،   وفضله ،   وإحسانه ،   وأن يكون غفوراً لذنوبه رحيماً لعقوبته .
