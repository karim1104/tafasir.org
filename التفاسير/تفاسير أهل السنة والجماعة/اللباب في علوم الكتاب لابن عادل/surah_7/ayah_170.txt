قوله :  "  والَّذينَ يُمَسِّكُون  "  : فيه وجهان : أظهرهما : أنَّهُ مبتدأ ،   وفي خبره حينئذ وجهان : أحدهما : الجملة من قوله { إِنَّا لاَ نُضِيعُ أَجْرَ المصلحين } وفي الرَّابط حينئذ أقوال :<br />
أحدها : أنَّهُ ضميرٌ محذوف لفهم المعنى ،   والتقدير : المُصْلحين منهم ،   وهذا على قواعد جمهور البصريين ،   وقواعد الكوفيين تقتضي أنَّ  "  ألْ  "  قائمةٌ مقام الضمير ،   تقديره : أجْرَ مَصْلحيهمْ ؛ كقوله : { فَإِنَّ الجنة هِيَ المأوى } [ النازعات : 41 ] أي : مَأواهُ ،   وقوله { مُّفَتَّحَةً لَّهُمُ الأبواب } [ ص : 50 ] أي : أبوابُها ،   وقوله : { في أَدْنَى الأرض } [ الروم : 3 ] أي : أرضهم ،   إلى غير ذلك .  <br />
والثاني : أنَّ الرَّابطَ تكرُّرُ المبتدأ بمعناه ،   نحو : زيد قام أبو عبد الله ،   وهو رأي الأخفش ،   وهذا كما يُجيزه في الموصول ،   نحو : أبُو سعيدٍ الذي رويتُ عنه الخُدريُّ ،   والحجَّاجُ الذي رأيْتُ ابنُ يُوسُفَ ،   وقد تقدَّم من ذلك شواهدُ .  <br />
الثالث : أنَّ الرَّابط هو العموم في  "  المُصْلحينَ  "  قاله أبُو البقاء .  <br />
قال :  "  وإن شئْتَ قلت : لمَّا كان المصلحون جنساً والمبتدأ واحد منه استغنيت عن ضمير  "  .  <br />
قال شهاب الدين{[16967]} : العمومُ ربطٌ من الروابط الخمسة ؛ وعليه قول الشاعر : [ الطويل ]<br />
ألاَ لَيْتَ شِعْرِي هَلْ إلَى أمِّ سالمٍ *** سَبِيلٌ ؟ فأمَّا الصَّبْرُ عَنْهَا فَلاَ صَبْرا{[16968]}<br />
ومنه : نعم الرجُل زيدٌ ،   على أحد الأوجه :<br />
والوجه الثاني - من وجهي الخبر - : أنَّهُ محذوف ،   تقديره : والذين يمسكون مأجورون ،   أو مثابُونَ ونحوه .  <br />
وقوله تعالى : { إِنَّا لاَ نُضِيعُ } جملة اعتراضية ،   قاله الحوفيُّ .   ولا ضرورة تدعو إليه .  <br />
الثاني من وجْهَيْ  "  الذينَ يُمَسِّكُون  "  : أنَّه محل جر نسقاً على :  "  الَّذين يَتَّقُونَ  "  أي : والدّار الآخرةُ خير للمتقين ،   وللمتمسكين ،   قاله الزمخشريُّ .  <br />
إلاَّ أنه قال : ويكون قوله : { إنَّا لا نُضيعُ } اعتراضاً سيق لتأكيد ما قبله .   وفيه نظرٌ ؛ لأنَّه لَمْ يقعْ بين شيئين متلازمين ولا بين شيئين بينهما تعلُّقٌ معنويُّ ،   فكان ينبغي أن يقول : ويكون على هذا مستأنفا .  <br />
وقرأ العامَّةُ  "  يُمَسِّكُونَ  "  بالتشديد مِنْ  "  مَسَّك  "  بمعنى  "  تَمَسَّكَ  "  حكاه أهلُ التصريف أي : إنَّ :  "  فَعَّلَ  "  بمعنى  "  تفَعَّل  "  ،   وعلى هذا فالباءُ للآلة ،   كهي في  "  تَمَسَّكْتُ بالحبل  "  .  <br />
يقال : مَسَّكْتُ بالشَّيء ،   وتَمَسَّكْتُ ،   واسْتَمْسَكْتُ به ،   وامتسَكْتُ به .  <br />
وقرأ أبو بكر عن عاصم{[16969]} ،   ورويت عن أبي عمرو وأبي العالية  " يُمْسِكُونَ  "  بسكون الميم وتخفيف السين من  "  أمْسَكَ  "  وهما لغتان يقال : مَسَكْتُ ،   وأمْسَكْتُ .  <br />
وقد جمع كعبُ بنُ زهير بينهما في قوله : [ البسيط ]<br />
ولا تُمَسِّكُ بالعهدِ الذي زَعَمَتْ *** إلاَّ كَمَا تُمْسِكُ الماءَ الغَرَابِيلُ{[16970]}<br />
ولكن  "  أمْسَك  "  مُتعدٍّ .  <br />
قال تعالى : { وَيُمْسِكُ السماء } [ الحج : 65 ] فعلى هذا مفعوله محذوف ،   تقديره : يُمسِكون دينهم وأعمالهم بالكتاب ،   فالباء يجوزُ أن تكون للحال وأن تكون للآلة أي : مصاحبين للكتابِ ،   أي : لأوامره ونواهيه .  <br />
وحجة عاصم قوله تعالى : { فَإِمْسَاكٌ بِمَعْرُوفٍ } [ البقرة : 229 ] ،   وقوله : { أَمْسِكْ عَلَيْكَ زَوْجَكَ } [ الأحزاب : 37 ] ،   وقوله : { فَكُلُواْ مِمَّا أَمْسَكْنَ عَلَيْكُمْ } [ المائدة : 4 ] .  <br />
قال الواحديُّ : والتشديدُ أقوى ؛ لأنَّ التشديد للكثرة ،   وههنا أريد به الكثرة ؛ ولأنَّهُ يقال : أمسكته ،   وقَلَّمَا يقال : أمسكت به .  <br />
وقرأ عبد الله{[16971]} والأعمش :  "  اسْتَمْسَكُوا  "  ،   وأبي{[16972]} :  "  تَمَسَّكُوا  "  على الماضي ،   وهو جيّد لقوله تعالى : { وَأَقَامُواْ الصلاة } إذ قل ما يعطف ماض على مستقبل إلاَّ في المعنى .  <br />
فصل<br />
أراد والذين يعملون بما في الكتاب .  <br />
قال مجاهدٌ : هم المؤمنون من أهل الكتاب ك : عبد الله بن سلام وأصحابه تمسَّكوا بالكتاب الذي جاء به موسى ،   فلم يحرفوه ولم يكتموه ولم يتخذوه مأكلةً{[16973]} .  <br />
وقال عطاء : أمة محمد صلى الله عليه وسلم .  <br />
{ وَأَقَامُواْ الصلاة إِنَّا لاَ نُضِيعُ أَجْرَ المصلحين } أي : لا نضيع{[16974]} أجرهم ،   كقوله تعالى : { إِنَّ الذين آمَنُواْ وَعَمِلُواْ الصالحات إِنَّا لاَ نُضِيعُ أَجْرَ مَنْ أَحْسَنَ عَمَلاً } [ الكهف : 30 ] .  <br />
فإن قيل : التمسك بالكتاب يشتمل على كل عبادة ،   ومنها إقامة الصلاة فكيف أفردها بالذِّكر ؟ فالجوابُ : أفردها لعلو مرتبتها ،   فإنَّهَا أعظم العبادات بعد الإيمان .
