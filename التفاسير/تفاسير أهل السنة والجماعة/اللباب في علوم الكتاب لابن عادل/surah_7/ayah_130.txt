لما قال موسى لقومه : { عسى رَبُّكُمْ أَن يُهْلِكَ عَدُوَّكُمْ } بدأ بذكر ما أنزل بفرعون وقومه من المِحَنِ حالاً بعد حال ،   إلى أن وصل الأمرُ إلى الهلاكِ تنبيهاً للمكلَّفين على الزَّجْرِ عن الكفر .  <br />
 "  والسِّنينَ  "  : جمعُ سنة ،   وفيها لغتان أشهرهما : إجْرَاؤُهُ مُجرَى المُذكر السَّالم فيُرفع بالواو ويُنْصَبُ ويُجَرُّ بالياء ،   وتُحْذَفُ نُونُه للإضافة .  <br />
قال النُّحَاةُ : إنَّمَا جرى ذلك المجرى جَبْراً لما فاته من لامه المحذوفةِ ،   وسيأتي في لامه كلامٌ ،   واللغة الثانيةُ : أن يُجْعَلَ الإعرابُ على النُّون ولكن مع الياء خاصَّةً .   نقل هذه اللُّغة أبُو زيد والفراءُ .   ثم لك فيها لغتان : إحداهما : ثبوتُ تنوينها .   والثانية : عدمهُ .  <br />
قال الفرَّاءُ : هي في هذه اللُّغة مصروفة عند بني عامر ،   وغير مصروفة عند بني تميم ،   ووجه حذف التنوين التَّخفيف ،   وحينئذ لا تُحْذَفُ النُّون للإضافة وعلى ذلك جاء قوله : [ الطويل ]<br />
دَعَانِيَ مِنْ نَجْدٍ فإنَّ سِنِينَهُ *** لَعْبِنَ بِنَا شيباً وشَيِّبْنَنَا مُرْدَا{[16717]}<br />
وجاء في الحديث :  "  اللَّهُمَّ اجْعَلْهَا عليهم سنينَ كَسِنِي يُوسُفَ  " {[16718]}  "  وسِنيناً كسِنِينِ يُوسفُ  "  باللُّغتين ،   وفي لام سَنَةٍ لغتان ،   أحدهما : أنَّهَا واو لقولهم : سنوات وسَانَيْتُ ،   وسُنَيَّةٌ .   الثانية : أنَّها هاء لقولهم : سانَهْتُ وسَنَهَات وسُنَيْهَة ،   وليس هذا الحكمُ المذكور أعني : جريانه مجرى جمع المذكر أو إعرابه بالحركات مقتصراً على لفظ  "  سِنينَ  "  بل هو جارٍ في كلِّ اسم ثلاثي مؤنث حُذِفَتْ لامُهُ ،   وعُوِّضَ منها تَاءُ التَّأنيثِ ،   ولم يُجْمَع جمع تكسير نحو : ثُبَةٍ وثُبين ،   وقُلة ،   وقُلينَ .  <br />
فصل<br />
قال شهابُ الدِّين{[16719]} :  "  وتَحرَّزْتُ بقولي : حُذِفَتْ لامُه مِمَّا حُذِفَتْ فاؤه ،   نحو : لِدة وعِدَة .  <br />
وبقولي ولم يُجْمع جمع تكسير من ظُبَةٍ وظُبّى ،   وقد شَذَّ قولهم : لِدُونَ في المحذوف الفاء ،   وظِبُونَ في المكسَّر .  <br />
يَرَى الرَّاءُونَ بالشَّفرَاتِ مِنْهَا *** وقُودَ أبِي حُبَاحِبَ والظُّبِينَا{[16720]}<br />
واعلم أنَّ هذا النَّوَع إذَا جَرَى مَجْرَى الزيدينَ فإنْ كان مكسورَ الفاء سَلِمَتْ ،   ولم تُغَيَّر نحو : مائة ومئين ،   وفئة وفئين ،   وإنْ كان مفتوحها كُسِرَتْ نحو : سنين ،   وقد نُقِلَ فَتْحُها وهو قليلٌ جدّاً ،   وإن كان مضمومَهَا جاز في جمعها الوجهان : أعني السَّلامة ،   والكسر نحو : ثُبين وقُلين .  <br />
قال أبُو عَلِي : السَّنة على معنيين : أحدهما : يراد بها العام .   والثاني : يراد بها الجدْب .  <br />
وقد غلبت السَّنَةُ على زمانِ الجدْبِ ،   والعام على زمان الخصب حتى صَارَا كالعلم بالغلبة ولذلك أشتقُّوا من لفظ السَّنَةِ فقالوا : أسْنَتَ القَوْمُ .  <br />
قال : [ الكامل ]<br />
عَمْرُوا الذي هَشَمَ الثَّرِيدَ لِقَوْمِهِ *** ورِجَالُ مَكَّةَ مُسْنتُونَ عِجَافُ{[16721]}<br />
وقال حاتم الطائِيُّ : [ الطويل ]<br />
فإنَّا نُهِينُ المَالَ مِنْ غَيْرِ ضِنَّةٍ *** وَلاَ يَشْتَكِينَا في السِّنينَ ضَرِيرُهَا{[16722]}<br />
ويُؤيِّدُ ذلك في سورة يوسف [ الآية 47 ] :  "  { تَزْرَعُونَ سَبْعُ سِنِينَ دَأَباً } .  <br />
ثم قال : { سَبْعٌ شِدَادٌ } فهذا في الجدب .  <br />
وقال : { ثُمَّ يَأْتِي مِن بَعْدِ ذلك عَامٌ فِيهِ يُغَاثُ الناس } .  <br />
وقوله :  "  مِنَ الثَّمَرَاتِ  "  متعلِّق ب  "  نَقْصٍ  "  .  <br />
قال قتادةُ : أمَّا السنُونَ فلأهل البوادي ،   وأمَّا نقص الثَّمراتِ فلأهل الأمصار .  <br />
 "  لَعَلَّهُمْ يَذَّكَّرُّونَ  "  يتَّعظون ،   وذلك لأنَّ الشدة ترقق القلوب ،   وترغب فيما عند اللَّهِ .  <br />
قال تعالى : { وَإِذَا مَسَّهُ الشر فَذُو دُعَاءٍ عَرِيضٍ } .  <br />
فصل<br />
قال القاضي : هذه الآية تدلُّ على أنَّهُ تعالى فعل ذلك لإرادة أن يذكَّرُوا وأنْ لا يقيموا على كفرهم ،   وأجاب الواحديُّ :  "  بأنَّهُ قد جاء لفظا الابتلاء ،   والاختبار في القرآن لا بمعنى أنَّه تعالى يمتحنهم ،   لأنَّ ذلك على اللَّهِ مُحَالٌ ،   بل إنَّه تعالى عاملهم معاملة تشبه الابتلاء ،   والامتحان ،   فكذا ههنا "  .  <br />
ثم بيَّن أنَّهُم عند نزول تلك المحن عليهم يزيدون في الكُفْرِ ،   والمعصيةِ .
