قوله تعالى : { وَكَتَبْنَا لَهُ فِي الألواح مِن كُلِّ شَيْءٍ مَّوْعِظَةً } .   قوله : في الألْوَاحِ يجوز أن تكون لتعريف الماهيَّةِ ،   وأن تكون للعهد ؛ لأنَّهُ يروى في القصَّة أنَّهُ هو الذي قطَّعَهَا وشقَّقَهَا .  <br />
وقال ابنُ عطيَّة{[16797]} أل عوض من الضمير ،   تقديره : في ألواحه ،   وهذا كقوله : { فَإِنَّ الجنة هِيَ المأوى } [ النازعات : 41 ] أي : مأواه .   أمَّا كون أل عوضاً من الضَّمير فلا يعرفه البصريون .   وأمَّا قوله : { فَإِنَّ الجنة هِيَ المأوى } فإنَّا نحتاجُ فيه إلى رابطٍ يَرْبِطُ بين الاسم والخبر ،   والكوفيون : يجعلون أل عوضاً من الضمير .   والبصريون : يُقَدِّرُونَهُ ،   أي : هي المأوى له ،   وأما في هذه الآية فلا ضرورة تدعو إلى ذلك .  <br />
وفي مفعول  "  كتبنا  "  ثلاثةُ أوجهٍ ،   أحدها : أنَّهُ  "  موعِظَةً  "  ،   أي : كتبنا له مَوْعِظَةٌ وتفْصِيلاً .   و { مِنْ كُلِّ شَيْءٍ } على هذا فيه وجهان ،   أحدهما : متعلِّقٌ ب  "  كَتَبْنَا  "  والثاني : أنَّهُ متعلِّقٌ بمحذوف ؛ لأنَّهُ في الأصلِ صفةٌ ل  " مَوْعِظَةً  "  فلمَّا قُدِّم عليها نُصِبَ حالاً ،   و  "  لِكُلِّ شيءٍ  "  صفة ل  " تفصيلاً "  .  <br />
والثاني : أنَّهُ { مِنْ كُلِّ شَيْءً } .  <br />
قال الزمخشريُّ { مِنْ كُلِّ شَيْءٍ } في محل نصب مفعول  "  كَتَبْنَا  "  ،   و  " مَوعِظَةً وتفْصِيلاً  "  بدل منه ،   والمعنى : كَتَبْنَا له كُلَّ شيءٍ كان بنو إسرائيل يَحْتَاجُونَ إليه في دينهم من المواعظِ ،   وتفصيل الأحكامِ وتفصل الحلالِ والحرامِ .  <br />
الثالث : أنَّ المفعول محل المجرور .  <br />
وقال أبُو حيَّان{[16798]} - بعد ما حكى الوجه الأول عن الحوفي والثَّاني عن الزمخشري - : ويُحْتَمَلُ عندي وجهٌ ثالثٌ ،   وهو أن يكونَ مفعولُ  "  كَتَبْنَا  "  موضع المجرور ،   كما تقولُ :  "  أكلت من الرغيف  "  و  "  مِنْ  "  للتبعيض ،   أي : كتبنا له أشياء من كُلِّ شيء ،   وانتصب  "  مَوْعظَةً وتَفْصِيلاً  "  على المفعول من أجله ،   أي : كتبنا له تلك الأشياءَ للاتِّعاظِ وللتفصيل .  <br />
قال شهابُ الدِّينِ{[16799]} :  "  والظَّاهِرُ أنَّ هذا الوجه هو الذي أراده الزَّمخشري ،   فليس وجهاً ثالثاً  "  .  <br />
قوله :  "  بِقُوَّةٍ  "  حالٌ : إمَّا من الفاعل ،   أي : ملتبساً بقوة ،   وإمَّا من المفعول ،   أي : ملتبسه بقوة ،   أَي : بقوَّةِ دلائلها وبراهينها ،   والأول أوضح .   والجملةُ من قوله :  "  فَخُذْهَا  "  يُحتمل أن تكون بدلاً من قوله { فَخُذْ مَا آتَيْتُكَ } وعاد الضَّميرُ على معنى  "  ما  "  لا على لفظها .   ويحتمل أن تكونَ منصوبة بقول مضمر ،   ذلك القولُ منسوقٌ على جملة  "  كَتَبْنَا  "  والتقديرُ : وكتبنا فقلنا : خُذْهَا ،   والضَّميرُ على هذا عائدٌ على الألواحِ أو على التَّوراةِ ،   أو على الرِّسالاتِ ،   أو على كُلِّ شيءٍ ؛ لأنَّهُ في معنى الأشياء .  <br />
قال القرطبيُّ :  "  فكأنَّ اللَّوحَ تلوح فيه المعاني .   ويقال : رجل عظيم الألواح إذا كان كبيرَ عظم اليدين ،   والرِّجليْنِ  "  .  <br />
فصل<br />
قال الكلبيُّ : خَرَّ مُوسَى صَعِقاً يوم الخميس يوم عرفة ،   وأعطي التَّوراة يوم الجمعةِ يوم النَّحر{[16800]} ،   واختلفوا في عدد الألواح وجوهرها فقيل : كانت عشرة ،   وقيل سبعة .  <br />
وقيل : إنها لوحان .  <br />
وقال الواحديُّ : كانت من زُمُردَة .  <br />
وقيل : من زبرجدة خضراء ،   وقيل : ياقوتة ،   وقيل : من خشب سور الجنَّةِ طول كل لوح اثني عشرة ذراعاً .  <br />
وقال وهْبٌ : من صخرة صَمَّاء لَيَّنَهَا اللَّهُ لِمُوسَى .  <br />
قيل : رفع سبعها وبقيت ستة أسباعها ،   وكان في الذي رفع تفصيل كلِّ شيء وفي الذي بقي الهدى والرحمة ،   وليس في الآية ما يدلُّ على شيء من ذلك ،   ولا على كيفية الكتابة فإن ثبت في ذلك شيء بدليل منفصل قويٍّ وجب القولُ به ،   وإلاّ وجب السُّكوت عنه .   وأمَّا قوله مِن كُلِّ شيءٍ فليس على العموم ،   بل المراد من كلِّ شيء يحتاجُ موسى وقومه إليه في دينهم .  <br />
وقوله : { مَّوْعِظَةً وَتَفْصِيلاً لِّكُلِّ شَيْءٍ } فهو كالبيانِ للجملة التي قدمها بقوله : { مِن كُلِّ شَيْءٍ } ثم قال :  "  فَخُذْهَا بِقُوَّةٍ  "  أي : بعزيمةٍ قويةٍ ونيَّةٍ صادقةٍ .  <br />
قوله : { وَأْمُرْ قَوْمَكَ يَأْخُذُواْ بِأَحْسَنِهَا } .   الظَّاهِرُ أنَّ يأخُذُوا مَجْزُومٌ جواباً للأمر في قوله وأمْرُ ولا بدَّ من تأويله ،   لأنَّه لا يلزمُ مِنْ أمره إيَّاهم بذلك أن يأخُذُوا ،   بدليل عصيانِ بعضهم له في ذلك ،   فإنَّ شَرْطَ ذلك انحلال الجملتين إلى شرطٍ وجزاءٍ .  <br />
وقيل : الجزم على إضمار اللام تقديره : ليأخذوا ؛ كقوله : [ الوافر ]<br />
مُحَمَّدُ تَفْدِ نَفْسَكَ كُلُّ نَفْسٍ *** إذَا مَا خِفْتَ مِنْ أمْر تبالاَ{[16801]}<br />
وهو مذهبُ الكسائي .  <br />
وابن مالك يرى جوازه إذا كان في جواب  "  قُلْ  "  ،   وهنا لم يُذْكَرْ  "  قُلْ  "  ،   ولكن ذُكر شيءٌ بمعناه ؛ لأنَّ معنى  "  وأمُرْ  "  و  "  قُلْ  "  واحد .  <br />
قوله :  "  بِأحْسَنِهَا  "  يجوزُ أن يكونَ حالاً كما تقدَّم في :  "  بِقُوَّةٍ  "  ،   وعلى هذا فمفعولُ  "  يَأخُذُوا  "  محذوفٌ تقديرُه : يَأخُذُوا أنفسهم ،   ويجُوزُ أن تكون الباء زائدة ،   وأحسنها مفعول به ،   والتقديرُ : يأخُذُوا أحسنها كقوله : [ البسيط ]<br />
 .   .   .   *** سُودُ المَحَاجِرِ لا يَقْرَأنَ بالسُّوَرِ{[16802]}<br />
وقد تقدَّم تحقيقه في قوله تعالى : { وَلاَ تُلْقُواْ بِأَيْدِيكُمْ إِلَى التهلكة } [ البقرة : 195 ] .  <br />
و  "  أحسن  "  يجوز أن تكون للتَّفضيل على بابها ،   وأن لا تكون بل بمعنى  "  حَسَنَة  "  .  <br />
كقول الفرزدق : [ الكامل ]<br />
إنَّ الذي سَمَكَ السَّماءَ بَنَى لَنَا *** بَيْتاً دَعَائِمُهُ أعَزُّ وأطْوَلُ{[16803]}<br />
أي : عزيزةٌ طويلةٌ .  <br />
فإن قيل : إنَّه تعالى لمَّا تعهد بكلِّ ما في التَّوراة ،   وجب أن يكون الكلُّ حسناً .  <br />
وقوله : { يَأْخُذُواْ بِأَحْسَنِهَا } يقتضي أن يكون فيه ما ليس بأحسن ،   وأنَّهُ لا يجوزُ لهم الأخذ به وهو متناقض .  <br />
وأجابُوا بوجوه : منها : أنَّ تلك التَّكاليفَ منها ما هو حسن ،   ومنها ما هو أحسن كالقصاص والعفو ،   والانتصار ،   والصبر ،   أي : فمرهم أن يأخُذُوا بالأفضل فإنَّه أكثر ثواباً ،   لقوله : { واتبعوا أَحْسَنَ مَا أُنزِلَ إِلَيْكُم مِّن رَّبِّكُمْ } [ الزمر : 55 ] وقوله : { الذين يَسْتَمِعُونَ القول فَيَتَّبِعُونَ أَحْسَنَهُ } [ الزمر : 18 ] .  <br />
قالوا : فيحمل الأخذ بالأحسن على النَّدب .  <br />
ومنها : قال قُطْرُبٌ :<br />
{ يَأْخُذُواْ بِأَحْسَنِهَا } أي : بحسنها ،   وكلها حسن ؛ كقوله تعالى : { وَلَذِكْرُ الله أَكْبَرُ } [ العنكبوت : 45 ] وأنشد بيت الفرزدق المتقدم .   ومنها : أن الحسن يدخلُ تحته الواجب ،   والمندوب ،   والمباح وأحسن هذه الثلاثة : الواجبُ ،   والمندوبُ .  <br />
قوله : { سَأُوْرِيكُمْ دَارَ الفاسقين } جَوَّزُوا في الرُّؤية هنا أن تكون بصريَّةً ،   وهو الظَّاهِرُ فتتعدَّى لاثنين ،   أحدهما : ضمير المخاطبين ،   والثاني : دَارَ .  <br />
والثاني : أنَّها قلبية ،   وهو منقولٌ عن ابن زيد وغيره ،   والمعنى : سأعْلِمُكُمْ سَيْرَ الأولين وما حَلَّ بهم من النَّكَالِ : وقيل :  "  دَارَ الفاسِقِينَ  "  ما دَارَ إليه أمرهم ،   وذلك لا يُعْلم إلا بالإخبار والإعلام .  <br />
قال ابنُ عطيَّة{[16804]} - معترضاً على هذا الوجه - : ولَوْ كَانَ من رؤية القلب ،   لتعدَّى بالهمزة إلى ثلاثةِ مفاعيل .  <br />
ولو قال قائل : المفعولُ الثالثُ يتضمنه المعنى ،   فهو مُقَدَّرٌ أي : مذمومة أو خربة أو مُسَعَّرة - على قول من قال : إنَّهَا جهنم - قيل له : لا يَجُوزُ حذفُ هذا المفعولِ ،   ولا الاقتصارُ دُونَهُ ،   لأنَّهَا داخلةٌ على الابتداءِ والخبرِ ،   ولو جُوِّزَ لكان على قبح في اللِّسان ،   لا يليق بكتاب الله تعالى .  <br />
قال أبُو حيان{[16805]} :  "  وحَذْفُ المفعُول الثَّالث في باب  "  أعْلَمَ  "  لدلالة المعنى عليه جائزٌ ،   فيجوزُ في جواب : هل أعلمتَ زَيْداً عمراً منطلقاً ؟ أعلمتُ زيداً عمراً ،   وتحذف  "  منطلقاً  "  لدلالة الكلام السَّابق عليه  "  .  <br />
فصل<br />
قال شهابُ الدِّين{[16806]} : هذا مُسَلَّمٌ ،   لكن أيْنَ الدَّليل عليه في الكلام ،   كما في المثال الذي أبرزه الشَّيْخُ ؟<br />
ثم قال :  "  وأمَّا تَعْليلُهُ بأنَّهَا داخلةٌ على الخَبَرِ لا يدلُّ على المنع ،   لأنَّ خبر المبتدأ يجوزُ حَذْفُهُ اختصاراً ،   والثانِي ،   والثَّالِثُ في باب  "  أعْلَمَ  "  يَجُوزُ حذفُ كُلٍّ منهما اختصاراً  "  .  <br />
قال شهابُ الدِّين :  "  حذفُ الاختصار لدليلٍ ،   ولا دليلَ هَنَا  "  .  <br />
ثم قال :  "  وفي قوله لأنَّهَا - أي :  "  سَأريكُمْ  "  - داخلةٌ على المبتدأ ،   والخبر تجوُّزٌ  "  ويعني أنَّها قبل النَّقْل بالهمزة داخلة على المبتدأ والخبر .   وقرأ الحسن{[16807]} :  "  سَأوريكُمْ  "  بواو خالصة بعد الهمزة وفيها تخريجان : أحدهما قاله الزمخشريُّ - :  "  وهي لغةٌ فاشية بالحجاز يُقَالُ : أوْرَنِي كَذَا وأوْرَيْتُهُ ،   فوجهه أن يكون من أوْرَيْتُ الزَّنْدَ ،   فإن المعنى : بَيِّنْهُ لي وأنِزْهُ لأستبينَه  "  .  <br />
والثاني : - ذكره ابنُ جنيٍّ - وهو أنَّهُ على الإشباع ،   فيتولَّد منها الواو ،   قال  "  وناسَبَ هذا كونُهُ موضعَ تهديدٍ ووعيدٍ فاحتمل الإتيان بالواو  "  .  <br />
قال شهابُ الدِّين : وهذا كقول الشاعر : [ البسيط ]<br />
اللَّهُ يَعْلَمُ أنَّا فِي تَلَفُّتِنَا *** يَوْمَ اللِّقاءِ إلى أحْبَابِنَا صًورُ<br />
وأنِّنِي حَيْثُمَا يُثْني الهَوَى بَصَرِي *** مِنْ حَيْثُمَا سَلَكُوا أدْنُوا فأنْظُورُ{[16808]}<br />
لكن الإشباعَ بابُهُ الضَّرُورَةُ عند بعضهم .   وقرأ ابنُ{[16809]} عبَّاسٍ ،   وقسامة بن زيد  "  سأورثُكُمْ  "  قال الزمخشريُّ : وهي قراءة حسنةٌ ،   يصحِّحُهَا قوله تعالى : { وَأَوْرَثْنَا القوم } [ الأعراف : 137 ] .  <br />
فصل<br />
في قوله : { سَأُوْرِيكُمْ دَارَ الفاسقين } وجهان : الأول : أنَّ المُرادَ به التهديدَ والوعيد وعلى هذا ففيه وجهان :<br />
أحدهما : قال ابن عباس{[16810]} والحسنُ ومجاهدٌ : هي : جهنم وهي مصيرهم في الآخرة ،   فاحْذَرُوا أن تكونوا منهم .  <br />
وثانيهما : قال قتادةُ وغيره : سأدخلكم الشَّام ؛ فأريكم منازل القُرُون الماضيةِ مثل الجبابرةِ ،   والعمالقةِ ،   ومنازل عادٍ وثَمُودَ الذين خالفُوا أمْرَ اللَّهِ لتعتبروا بها{[16811]} .  <br />
الوجه الثاني : المُرَادُ به الوعد والبشارة بأنَّ الله تعالى سيوِّرثُهم أرض أعدائهم وديارهم وهي أرض مصر ،   قالهُ عطيةُ العوفيُّ ؛ ويدلُّ عليه قراءة قسامة .  <br />
وقال السُّدِّيُّ : هي مصارع الكفار{[16812]} .
