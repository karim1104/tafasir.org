يجوز في  "  أن  "  : أن تكُون المفسِّرة لمعنى الإيحَاءِ .  <br />
ويجوزُ أنْ تكون مصدريّةً ؛ فتكونُ هي ،   وما بعدها مفعول الإيحَاءِ .  <br />
قوله : { فَإِذَا هِيَ تَلْقَفُ } قرأ العامَّةُ :  "  تَلَقَّفُ  "  بتشديد القافِ ،   من  "  تَلَقَّفَ  "  والأصلُ :  "  تَتَلَقَّفُ  "  بتاءيْنِ ،   فحذفت إحداهُمَا ،   إمَّا الأولى ،   وإمَّا الثانية وقد تقدَّم ذلك في نحو { تَتَذَكَّرُونَ } [ الأنعام : 152 ] .  <br />
والبزِّيُّ : على أًصلها في إدغامِهَا فيما بعدها ،   فيقرأُ :  "  فإذا هي تَّلَقَّفُ  "  بتشديد التاء أيضاً ،   وقد تقدم تحقيقه عند قوله : { وَلاَ تَيَمَّمُواْ الخبيث } [ البقرة : 267 ] .  <br />
وقرأ حفصٌ{[16696]}  "  تَلْقَفُ  "  بتخفيف القافِ من  "  لَقِفَ  "  ك  "  عَلِمَ يَعْلَمُ ،   ورَكِب يَرْكَبْ  "  .  <br />
يقال : لَقِفْتُ الشَّيءَ ألْقَفُهُ لَقْفاً ،   ولَقَفَاناً ،   وتَلقفتُهُ أتَلقَّفُهُ تَلَقُّفاً : إذا أخَذْتهُ بِسُرعةٍ ،   فأكَلْتَهُ أو ابْتَلَعْتَهُ .  <br />
وفي التفسير : أنها ابتلعَتْ جميع ما صَنَعُوه ،   وأنشدُوا على : لَقِفَ يَلْقَفُ ،   ك  "  عَلِمَ يَعْلَمُ  "  قول الشَّاعِر : [ السريع ]<br />
وأنتَ عَصَا مُوسَى الَّتي لَم تَزَلْ *** تَلْقَفُ مَا يَصْنَعُهُ السَّاحِرُ{[16697]}<br />
ويُقَالُ : رَجُلٌ ثقفٌ لقفٌ ،   وثَقِيفٌ لَقِيفٌ ،   بَيِّن الثَّقافة واللَّقَافة .   ويُقَالُ : لَقِفَ ولَقِمَ بمعنى واحدٍ ،   قاله أبُو عُبيدٍ .  <br />
ويقالُ : تَلْقَفُ ،   وتَلْقَمُ ،   وتَلْهَمُ : بمعنىً واحدٍ .  <br />
والفَاءُ في  "  فإذَا هِيَ  "  يجوزُ أن تكُون العاطفة ،   ولا بُدَّ من حَذْفِ جملةٍ قَبْلهَا ليترتَّبَ ما بعد الفاءِ عليها ،   والتقديرُ :  "  فألْقَاهَا فإذا هِيَ  "  .  <br />
وَمَنْ جوَّز أن تكون الفاءُ زائدةً في نحو :  "  خَرَجْتُ فإذا الأسَدُ حَاضِرٌ  "  جوَّز زيادتها هُنَا .  <br />
وعلى هذا فتكونُ هذه الجملةُ قد أوحيت إلى موسى كالَّتي قَبْلَهَا .  <br />
وأمَّا على الأوَّل - أعني كون الفاءِ عاطفةً - فالجملةُ غير موحى بها إليه .  <br />
قوله :  "  مَا يَأفكونَ  "  يجوزُ في  "  ما  "  أن تكون بمعنى  "  الذي  "  والعائدُ محذوفٌ ،   أي : الذي يأفِكُونهُ .  <br />
ويجوز أن تكُون  "  ما  "  مصدرية ،    "  والمصدر  "  حينئذٍ واقعٌ موقعَ المفعُولِ به ،   وهذا لا حَاجةَ إلَيْهِ .  <br />
وذلك قولُهُ : { مَا كَانُواْ يَعْمَلُونَ } يَجُوز أن تكون  "  ما  "  بمعنى  "  الذي "  ،   فيكونُ المعنى : بَطَلَ الحبالُ والعِصيُّ الذي عملوا به السِّحر : أي : زَالَ ،   وذهب بِفُقْدانِهَا ،   وأن تكون مصدرية ،   أي : وبطل الذي كانوا يعملونه ،   أو عملهم .  <br />
وهذا المصدرُ يجوزُ أن يكون على بابه .  <br />
وأن يكون واقعاً موقع المفعول به .   بخلاف  "  مَا يَأفكُون  "  فإنَّ يتعيَّنُ أن يكُونُ واقعاً موقع المفعُول به ليصحَّ المعنى ؛ إذ اللَّقْفُ يستدعي عَيْناً يصحُّ تسلُّطُه عليها .  <br />
ومعنى الإفكِ في اللُّغةِ : قلبُ الشَّيءِ عن وجْههِ ،   ومنه قِيلَ للكذبِ إفْكٌ ،   لأنَّهُ مقلْوبٌ عن وجهه .  <br />
قال ابنُ عبَّاسٍ :  "  مَا يَأفِكُونَ  "  يُريدُ : يَكْذِبُونَ ،   والمعنى : أنَّ العصا تلقَفُ ما يأفِكُونَهُ ،   أي : يَقْلِبُونَهُ عن الحَقِّ إلى البَاطِلِ .
