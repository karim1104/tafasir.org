قوله تعالى : { وَإِذْ تَأَذَّنَ رَبُّكَ } الآية .  <br />
لمَّا شرح قبائح أعمال اليهود ذكر هنا حكمه عليهم بالذل والصَّغارِ إلى يوم القيامة ،   و  "  تَأذَّنَ  "  فيه أوجه ،   أحدها : أنَّهُ بمعنى :  "  آذَنَ  "  أي : أعْلَمَ .  <br />
قال الواحديُّ : وأكثر أهل اللغة على أنَّ :  "  التَّأذَُّن  "  بمعنى الإيذان ،   وهو الإعلامُ .  <br />
قال الفارسي :  "  آذَنَ  "  أعْلَمَ ،   و  "  أذَّنَ  "  نادى وصاح للإعلام ،   ومنه قوله { فَأَذَّنَ مُؤَذِّنٌ بَيْنَهُمْ } [ الأعراف : 44 ] .  <br />
قال : وبعض العرب يُجْرِي  "  آذَنْتُ  "  مجرى  "  تَأذَّنْتُ  "  فيجعل  "  آذَنَ وتَأذَّنَ  "  بمعنى فإذا كان  "  أذَّنَ  "  أعلم في لغة بعضهم ،   ف  "  أذَّنَ  "  تفعَّل من هذا .  <br />
وقيل : معناه : حَتَّمَ وأوْجبَ وهو معنى قول مجاهد : أمر{[16944]} ربك ،   وقول عطاء : حكم ربّك{[16945]} .  <br />
وقال الزمخشري :  "  تأذَّن  "  عزم ربك ،   وهو تفعَّل من الإيذانِ وهو الإعلامُ ؛ لأنَّ العازمَ على الأمر يُحدِّثُ به نفسه ويؤذنها بفعله ،   وأجري مُجْرَى فعل القسم ك : عَلِمَ الله ،   وشَهِدَ الله ،   ولذلك أجيب بما يجابُ به القسم وهو :  "  لَيَبْعَثَنَّ  "  .  <br />
وقال الطبريُّ وغيره  "  تَأذَّنَ  "  معناه  "  أعْلَمَ  "  ،   وهو قلقٌ من جهة التصريف ،   إذ نسبةُ  "  تأذَّنَ  "  إلى الفاعل غيرُ نسبة  "  أعْلَمَ  "  ،   وبين ذلك فرقٌ من التعدي وغيره .  <br />
وقال ابن عباس : تأذَّنَ ربُّك قال ربُّكَ{[16946]} .  <br />
قوله : { إلى يَوْمِ القيامة } فيه وجها : أصحهما : أنَّهُ متعلقٌ ب : لِيَبَعْثَنَّ .  <br />
والثاني : أنَّهُ متعلقٌ ب : تَأذَّنَ نقله أبو البقاء ،   ولا جائزٌ أن يتعلق ب : يَسُومُهُمْ ؛ لأن  "  مَنْ  "  إمَّا موصولةٌ ،   وإمَّا موصوفةٌ ،   والصلةُ والصفة لا يعملان فيما قبل الموصول والموصوف .  <br />
فصل<br />
الضمير في عليهم يقتضي رجوعه إلى الذين : { عتَوْاْ عَن مَّا نُهُواْ عَنْهُ } لكنَّهم قد مسخوا ،   فلم يستمر عليهم التَّكليف ؛ فلذلك اختلفوا :<br />
فقيل : المراد نسلهم .  <br />
وقيل : المراد سائر اليهود ،   فإنَّ أهل القرية كانوا فرقتين ،   فالمتعدِّي مسخ ،   وألْحَقَ الذُّلَّ بالبقيَّةِ .  <br />
وقال الأكثرون : هُمُ اليهُودَ الذين كانُوا في زمن الرسول - عليه الصَّلاة والسَّلامُ - ؛ لأن المقصودَ من الآية تخويفهم وزجرهم ،   وهذا العذابُ في الدُّنْيَا ،   لأنه نص على أنَّ ذلك العذاب ممدود إلى يوم القيامة ثمَّ اختلفوا فيه :<br />
فقيل : هو أخذ الجزيةِ .  <br />
وقيل : الاستخفافُ والإهانةُ لقوله تعالى { ضُرِبَتْ عَلَيْهِمُ الذلة أَيْنَ مَا ثقفوا } [ آل عمران : 112 ] وقيل : القتل والجلاء الذي وقع بأهل خيبر وبني قريظة والنضير .  <br />
فصل<br />
دلَّت هذه الآية على أنَّ اليهود لا دولة لهم ولا عزَّ ،   وأن الذُّلَّ والصغار لا يفارقهم ،   وقد ورد في الحديث : أن أتباع الدَّجَّالِ هم اليهود{[16947]} فإن صحَّ فمعناه : أنهم كانوا قبل خروجه يهوداً ،   ثم دانوا بإلهيته ؛ فذكروا بالاسم الأول ،   ولوا ذلك لكانوا في وقت اتباعهم الدَّجَّال قد خرجوا عن الذلةِ والقهرِ ،   وهو خلاف الآيةِ .  <br />
ثم قال : { إِنَّ رَبَّكَ لَسَرِيعُ العقاب } والمراد التَّحذيرُ من عذابه في الآخرة مع الذِّلةِ في الدُّنيا { وَإِنَّهُ لَغَفُورٌ رَّحِيمٌ } لمن تاب من الكُفْرِ ،   واليهوديَّةِ ،   وآمن باللَّهِ وبرسوله .
