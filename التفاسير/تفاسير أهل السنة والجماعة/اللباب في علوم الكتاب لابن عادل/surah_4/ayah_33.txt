"  جعلنا  "  فيه سِتَّةُ أوْجُهٍ ،   وذلك يَسْتَدْعِي مقدِّمَةً قبله ،   وَهُوَ أنّ  "  كُلّ  "  لا بدَّ لَهَا مِنْ شَيْءٍ تُضَافُ إليْهِ .  <br />
قال القُرْطُبِيُّ{[7686]} :  "  كُلّ  "  في كلام العربِ مَعْناهَا : الإحَاطَةُ والعموم ،   فإذا جَاءَتْ مُفْرَدَة ،   فلا بدَّ وأن يكُونَ في الكَلاَمِ حَذْفٌ عند جميع النحويين  "  .  <br />
واختلفوا في تقديرهِ : فقيل تقدِيرُهُ : ولكلِّ إنسان .  <br />
وقيل : لِكُلِّ مال ،   وقيل : لِكُلِّ قوم ،   فإنْ كانَ التَّقْديرُ : لكل إنسان ،   ففيه ثلاثة أوجه :<br />
أحدُهَا : وَلِكُلِّ إنسانٍ موروثٍ{[7687]} جعلنا موالي ،   أي : وُرَّاثاً مِمَّا تَرَكَ ،   ففي  "  تَرَكَ  "  ضميرٌ عائد على  "  كُلّ  "  ،   وهنا تمّ الكلام .  <br />
وقيل : تَقْدِيرُهُ : ويتعلق  "  مِمَّا تَرَك  "  ب  "  مَوَالي  "  لما فيه من معنى الوراثة ،   و  " موالي "  : مَفْعُولٌ أوَّ ل  "  جَعَلَ  "  ،   و  "  جَعَلَ  "  بمعنى :  "  صَيَّر  "  ،   و  "  لِكُلّ  "  جار ومجرور هو المفعول الثَّاني ،   قُدِّم على عامِلِهِ ،   ويرتفع  "  الوِلْدَان  "  على خبر مبتدأ محذوف ،   أو بفعل مقدّر ،   أي : يرثون مما [ ترك ]{[7688]} ،   كأنه قيل : ومَنْ الوارثُ ؟ فقيل : هم الوَالِدَان والأقْرَبُون ،   والأصل :  "  وجعلنا لكل ميت وراثاً يرثون مما تركه هم الوالدان والأقربون  "  .  <br />
والثَّانِي : أنَّ التَّقديرَ : ولكلِّ إنْسَانٍ موروث ،   جعلنا وراثاً مما ترك ذلك الإنسان .   ثُمَّ بين الإنْسَان المضاف إليه  "  كُلّ  "  بقوله : { الْوَالِدَانِ } ،   كأنه قيل : ومن هو هَذَا الإنسان الموروث ؟ فقيل{[7689]} : الوالدان والأقربُونَ ،   والإعراب كما تقدَّمَ في الوَجْهِ قَبْلَهُ ،   إنَّمَا الفرقُ بينهما أنَّ الوالِدَيْنِ في الأوَّلِ وارثون ،   وفي الثانِي مورثون ،   وعلى هذيْنِ الوجْهَيْنِ فالكلامُ جُمْلَتَانِ ،   ولا ضميرَ ،   محذُوف في  "  جعلنا  "  ،   و  "  موالي  "  مفعول أول ،   و  " لكل  "  مفعول ثان .  <br />
الثَّالِثُ : أن يكُونَ التَّقدِيرُ : ولكل إنسان وارِث ممَّن{[7690]} تركُ الولِدَانِ والأقْرَبُون جعلنا موالي ،   أي : موروثين ،   فَيُراد بالمَولى : الموْرُوثُ ،   ويرتفع  "  الوالدان  "  ب :  "  ترك  "  ،   وتكون  "  مَا  "  بمعنى  "  مَنْ  "  ،   والجارّ ،   والمجرورُ صِفَةٌ للمضاف إليه  "  كُلّ  "  ،   والكلامُ على هذا جُمْلَةٌ واحِدَةٌ ،   وفي هذا بُعْدٌ كبير .  <br />
الرَّابعُ : إذا كان التَّقديرُ وَلِكُلِّ قوْمٍ ،   فالمعنى : ولكل قوم جعلناهم مَوَالي نصيبٌ مِمَّا تَرَكَهُ والدُهم{[7691]} وأقربوهم ،   ف  "  لكل  "  خبر مقدّم ،   و  "  نَصِيب  "  مُبْتَدَأٌ مُؤخَّرٌ ،   و  " جعلناهم  "  صفة لقوم ،   والضَّمِيرُ العَائِدُ عليهم مفعولُ :  "  جعل  "  ،   و  "  موالي  "  : إما ثانٍ وإمّا حالٌ ،   على أنَّهَا بمعنى  "  خلقنا  "  ،   و  "  مما ترك  "  صفةٌ للمبتدأ ،   ثم حُذف المُبْتَدَأ ،   وبقيت صفته ،   [ وحُذِفَ المُضَافُ إليه  "  كُلّ  "  وبقيت صفته أيضاً ]{[7692]} ،   وحُذف العَائِدُ على المَوْصُوفِ .  <br />
ونظيره : لِكُلِّ خَلَقَهُ اللَّه إنْسَاناً مِنْ رِزْقِ اللَّه ،   أي : لِكُلِّ أحدٍ خلقه اللَّه إنْسَاناً نَصِيبٌ من رزقِ اللَّهِ .  <br />
الخَامِسُ : إنْ كَانَ التَّقدِيرُ : ولكلِّ مالٍ ،   فقالوا : يكون المعنى : ولكلِّ مال مِمَّا تركه الوالدانِ والأقربون جعلنا موالي ،   أي : وُرَّاثاً يلونه ،   ويحوزونه ،   وجعلوا  "  لِكُلّ  "  متعلقة : ب  "  جَعَلَ  "  ،   و  "  مِمَّا ترك  "  صفة ل  "  كُلّ  "  ،   والوالدان فَاعِلٌ ب  "  تَرَكَ  "  ،   فيكونُ الكلامُ على هذا ،   وعلى الوجهين قبله كلاماً واحداً ،   وهذا وإنْ كَانَ حَسَناً ،   إلاّ أنَّ فيه الفَصْلَ بين الصِّفَةِ والموْصُوفِ بجملةٍ عامِلَةٍ في الموْصُوفِ .  <br />
قاتل أبُو حَيَّان{[7693]} :  "  وهو نظير قولك : بكُلِّ رَجُلٍ مَرَرْتُ تميميٍّ وفي جواز ذلك نَظَرٌ  "  .  <br />
قال شهَابُ الدِّينِ{[7694]} :  "  ولا يحتَاجُ إلى نَظَرٍ ؛ لأنَّهُ قد وُجِدَ الفصلُ بَيْنَ الموْصُوفِ والصِّفَةِ بالجملةِ العَامِلَةِ في المُضَافِ إلى المَوصُوفِ ،   كقوله تعالى :<br />
{ قُلْ أَغَيْرَ اللَّهِ أَتَّخِذُ وَلِيّاً فَاطِرِ السَّمَاوَاتِ وَالأَرْضِ } [ الأنعام : 14 ] ف { فَاطِرِ } صفة ل { اللَّه } ،   وقد فُصِلَ [ بينهما ]{[7695]} ب { أَتَّخِذُ } العامل في { أَغَيْرَ } فهذا أولى "  .  <br />
السَّادسُ : أنْ يكُونَ لكلِّ [ مال ]{[7696]} مفعولاً ثانياً ل  "  جعَلَ  "  على أنَّها تصييرية{[7697]} ،   و  " مَوَالي  "  مفعول أوَّل ،   والإعرابُ على ما تقدَّمَ .  <br />
فصل<br />
 "  المَولى "  لفظ مُشْتَرَكٌ بيْنَ مَعَانٍ :<br />
أحدها : المعتِقُ ؛ لأنَّهُ ولي نعمة من أعتقه ،   ولذلك سمي مولى النعمة .   ثانيها : الْعَبْدُ المُعْتَقُ لاتِّصَالِ ولايَةِ مَوْلاَهُ به في إنْعَامِه عليه ،   وهذا كما سُمِّيَ الطَّالِبُ غرِيماً ؛ لأنَّ له اللُّزُوم والمطالبة بحقِّه ،   ويسمَّى المطلوب غريماً ،   لِكونِ الدِّينِ لازِماً له .  <br />
وثالثها : الحليفُ ؛ لأنَّ المحالف يلي أمْرَهُ بِعَقْدِ اليَمينِ .  <br />
ورابعُهَا : ابْنُ العَمِّ ؛ لأنَّهُ يليه بالنُّصْرَةِ .  <br />
وخامسها : المولى{[7698]} لأنَّ يليه بالنُّصْرَةِ ،   قال تعالى : { ذَلِكَ بِأَنَّ اللَّهَ مَوْلَى الَّذِينَ آمَنُواْ وَأَنَّ الْكَافِرِينَ لاَ مَوْلَى لَهُمْ } [ محمد : 11 ] .  <br />
سادسُهَا : العَصَبَةُ ،   وهو المُرادُ بهذه الآية ؛ لقوله عليه السلامُ :  "  أنا أوْلَى بالمؤمنينَ ،   مَنْ مَات وتَرَكَ مالاً ،   فَمَالُهُ لمَوَالِي الْعَصَبَةِ ،   ومَنْ ترك ديناً ؛ فأنَا وَلِيُّه  " {[7699]} .  <br />
وقال عليه السلامُ :  "  ألْحِقُوا الفَرَائِضَ بأهْلِهَا ،   فَمَا بَقِيَ فللأوْلَى عصبَةٍ ذكر  " {[7700]} .  <br />
قوله { وَالَّذِينَ عَقَدَتْ } في محلّهِ أربعة أوجهٍ :<br />
أحدُهَا : أنَّهُ مُبْتدأ والخبر قوله :  "  فآتوهم "  [ ودخلت الفاء في الحيز لتضمن الذي معنى الشرط ]{[7701]} .  <br />
الثَّاني : أنَّهُ منصوبٌ على الاشْتِغالِ بإضمار فعلٍ ،   وهذا أرجحُ مِنْ حَيْثُ إنَّ بَعْدَهُ طلباً .  <br />
والثَّالِثُ : أنَّهُ مرفوعٌ عطفاً على { الْوَالِدَانِ وَالأَقْرَبُونَ } ،   فإن أريدَ بالوالدين أنَّهُم موروثون ،   عادَ الضَّميرُ من  "  فآتوهم "  على  "  موالي "  وإن أُريد أنَّهُم وَارِثُون جازَ عودُه على  "  موالي "  وعلى الوالدَيْنِ وما عُطِفَ عليهم .  <br />
الرَّابِعُ : أنَّهُ منصوب عطفاً على  " موالي "  .  <br />
قال أبُو البَقَاءِ{[7702]} : [ أي : ]{[7703]}  "  وجعلنا الذين عاقدتْ وُرّاثاً ؛ وكان ذلك ونسخ  "  ،   وردّ عليه أبُو حَيَّان{[7704]} بِفَسَادِ العطْفِ ،   قال : فإن جُعِل من عطْفِ الجُمَل ،   وحُذِفَ المفعولُ الثَّاني لدلالة المعنى عليه أمكن ذلك أيْ : جعلنا وُرَّاثاً لكلِّ شَيْءٍ من المال ،   أو لِكُلِّ إنسان ،   وجَعلنَا الذِينَ عاقَدَتْ أيمانكم وراثاً وفيه بعد ذلِكَ تَكَلُّفٌ .  <br />
انتهى .  <br />
وقرأ عاصمٌ وحمْزَةُ والكسَائِيُّ{[7705]} :  "  عقدت  "  والباقون :  "  عاقدت  "  بألف وروي عن حمزة التَّشديد في  "  عقدت  "  ،   والمفاعلة هنا ظَاهِرَةٌ ؛ لأنَّ المَرَادَ المخالفة .  <br />
والمفعولُ محذوفٌ على كُلٍّ من القِرَاءاتِ ،   أي : عاقدْتَهم أو عَقَدْتَ حِلْفهم ،   ونسبة المُعاقَدَةِ ،   أو العَقْدِ إلى الأيمان مجاز ،   سوَاءٌ أُريد بالأيْمَانِ الجَارِحَة ،   أم القَسمُ .  <br />
وقيل : ثمَّ مُضافٌ محذوفٌ ،   أي : عقدت ذَوُو أيْمَانِكُم .  <br />
فصل في :  "  معنى المعاقدة والأيمان  " <br />
المُعَاقَدَةُ المُحالَفَةُ ،   والأيْمَانُ جمع يَمينٍ من اليد والقسَمِ ،   وذلك أنَّهُم كانُوا عند المُحالَفَةُ يأخذُ بعضهُمُ يدَ بَعْضٍ ،   على الوفَاءِ [ والتمسك ]{[7706]} بالعهد .  <br />
فصل الخلاف في نسخ الآية<br />
قال بعضهم : إنَّ هذه الآية مَنْسُوخَةٌ ،   واسْتدلُّوا على ذلك{[7707]} بوجوه :<br />
أحدها{[7708]} : أنَّ الرَّجُلَ كان في الجاهلِيَّةِ يُعَاقِدُ غيْرَهُ ،   فيقُولُ :  "  دَميَ دمُكَ وسِلْمِي سِلْمُك ،   وَحَرْبِي حَرْبُكَ ،   وترثُنِي{[7709]} وَأرِثُك ،   وَتَعْقِلُ عَنِّي ،   وأعْقِلُ عنك  "  ،   فيَكُونُ لهذا الحليف السّدس [ من ]{[7710]} الميراثِ ،   فذلك قوله :  "  فآتوهم نصيبهم  "  ،   فنُسِخ ذلك بقوله :<br />
{ وَأْوْلُواْ الأَرْحَامِ بَعْضُهُمْ أَوْلَى بِبَعْضٍ فِي كِتَابِ اللَّهِ } [ الأنفال : 75 ] ،   وبقوله :<br />
{ يُوصِيكُمُ اللَّهُ فِي أَوْلاَدِكُمْ } [ النساء : 11 ] .  <br />
وثانيها : أنَّ الرَّجُلَ كانَ يتَّخِذُ أجنبياً فيجعله{[7711]} ابْناً له ،   وَهُمُ المُسَمُّوْنَ بالأدْعِيَاء في قوله تعالى : { أَدْعِيَاءَكُمْ أَبْنَاءَكُمْ } ،   وكانوا يتوارثون بذلك ،   ثم نُسِخَ .  <br />
وثالِثُهَا : أنَّ النَّبيَّ صلى الله عليه وسلم كان يُثْبِتُ المؤاخاة بيْنَ الرَّجلين مِنْ أصحابِهِ ،   وكان ذلك سَبَباً للِتَّوَارُثِ ،   ثم نسخ .  <br />
وقال آخرُونَ : الآيةُ غير مَنْسُوخَةٍ .  <br />
وقال إبْراهيمُ ومُجاهِدٌ : أرادَ :  "  فآتوهم نصيبهم من النصر والرفادة ولا ميراث  " {[7712]} .  <br />
وقال الجبَّائِيُّ{[7713]} : تقدير الآية :  "  ولكل شيء مما ترك الوالدان والأقربون  "  ،   { والذين عاقدت أيمانكم } معطوف على قوله : { الوالدان والأقربون } ،   وسمى اللَّهُ تعالى الوارث مولى ،   والمعنى : لا تَدفعُوا المالَ إلى الحليفِ ،   بل للمولى ،   والْوَارِثِ .  <br />
وقال آخرون{[7714]} : المُرادُ ب { الذين عاقدت أيمانكم } الزَّوْجُ ،   والزَّوْجَةُ ،   فأراد عقد النِّكاح قال تعالى { وَلاَ تَعْزِمُواْ عُقْدَةَ النِّكَاحِ } [ البقرة : 235 ] وهو قول أبِي مُسْلِم الأصفهانيَّ قال : ونظيره آية المواريث ،   لما بَيَّنَ آية ميراث الوالدَيْنِ ،   ذكر معهم ميراثَ الزَّوْجِ ،   والزَّوْجَةِ .  <br />
وقيل : أراد بقوله : { الذين عاقدت أيمانكم } : الميراث بِسَبَبِ [ الوَلاَء ]{[7715]} وقيل :  "  نزَلَتِ الآيةُ في أبِي بكرٍ الصِّدِّيق ،   وابنه عبد الرَّحْمن ،   أمره اللَّه أن يؤتيه نَصِيبَهُ  " {[7716]} .  <br />
وقال الأصمُّ{[7717]} : المُرادُ بهذا النَّصِيب على سبيل الهِبَةِ ،   والهديَّة بالشيءِ القَلِيلِ كأمره تعالى لمن{[7718]} حَضَرَ القِسمَةَ أن يجعل لَهُ نصيباً كما تقدّم .  <br />
فصل الخلاف في إرث المولى الأسفل من الأعلى<br />
قال جمهور الفُقَهَاءِ{[7719]} :  "  لا يَرثُ المَولى الأسْفَل من الأعلى  "  .  <br />
وحكى الطَّحَاوِيُّ عن الحسن{[7720]} بنِ زيادٍ أنَّهُ قال :  "  يَرِثُ  "  ،   لما روى ابن عباس - رضي الله عنه - أنَّ رجُلاً أعتق عبداً له ؛ فَمَاتَ المُعْتِق ،   ولم يترك إلا المُعتَق{[7721]} ،   فجعل رسُول اللَّه صلى الله عليه وسلم ميراثه للغلام المعتق ولأنَّهُ داخلٌ{[7722]} في عموم قوله : { والذين عاقدت أيمانهم فآتوهم نصيبهم } .  <br />
وأجيب بأنه لَعَلّ{[7723]} ذلك لما صار لبيت المال دفعه النبي صلى الله عليه وسلم إلى الغُلامِ لحاجته ،   وفقره ؛ لأنه كانَ مالاً لا وارث لَهُ ،   فَسَبيلُهُ أن يُصرف إلى الفُقَرَاءِ .  <br />
ثم قال تعالى [ <br />
{ إِنَّ اللَّهَ لاَ يَغْفِرُ أَن يُشْرَكَ بِهِ } [ النساء : 48 ] ثم قال ]{[7724]} { إِنَّ اللَّهَ كَانَ عَلَى كُلِّ شَيْءٍ شَهِيداً } ،   وهذه كلمة وعد للمطيعين ،   ووعيد لِلْعُصَاةِ ،   والشَّهِيدُ الشَّاهد ،   والمرادُ إمّا{[7725]} علمه تعالى بجميع المَعْلُومَاتِ ،   فيكونُ المُرَادُ بالشَّهيدِ العالم ،   وإمّا شهادته على الخلق يَوْمَ القِيامَةِ ،   فالمُرَادُ بالشَّهيدِ المخبر .
