{ يَا أَيُّهَا الَّذِينَ آمَنُواْ خُذُواْ حِذْرَكُمْ فَانفِرُواْ ثُبَاتٍ أَوِ انْفِرُواْ جَمِيعاً } * [ وَإِنَّ مِنْكُمْ لَمَن لَّيُبَطِّئَنَّ فَإِنْ أَصَابَتْكُمْ مُّصِيبَةٌ قَالَ قَدْ أَنْعَمَ اللَّهُ عَلَيَّ إِذْ لَمْ أَكُنْ مَّعَهُمْ شَهِيداً * وَلَئِنْ أَصَابَكُمْ فَضْلٌ مِنَ الله لَيَقُولَنَّ كَأَن لَّمْ تَكُنْ بَيْنَكُمْ وَبَيْنَهُ مَوَدَّةٌ يالَيتَنِي كُنتُ مَعَهُمْ فَأَفُوزَ فَوْزاً عَظِيماً } ]{[8748]}<br />
قال القًرْطُبِي{[8749]} : إنه - تعالى - لمّا ذكر طاعة اللَّه وطَاعَة رسُولِهِ ،   أمر أهْل الطَّاعة بالقِيَام بإحْيَاء دينهِ وإعْلاءِ دَعْوَته ،   وأمَرَهُم ألاَّ يقْتَحِمُوا على عُدوِّهِم على جَهَالةٍ ،   حتى يَتَحَسَّسُوا إلى ما عِنْدَهُم ويَعْلَمُون كيف يَرُدُّون عَلَيْهم ،   لأن ذَلِكَ أثْبَت لَهُم .  <br />
وقال ابن الخطيب{[8750]} : لما رغَّب في طاعة اللَّهِ وطاعةِ رسُوله ،   عَادَ إلى ذِكْرِ الجِهَاد ؛ لأنه أشَقُّ الطَّاعَات ،   وأعْظَم الأمُور الَّتِي بها يَحْصُل تَقْوِية الدِّين ،   والحَذَر والحِذْر بمعنى ؛ كالأثر والإثْر ؛ والمَثَل والمِثْل ،   والشَّبَه والشِّبْه .  <br />
قيل : ولم يُسْمَع في هذا التَّرْكِيب إلا خُذْ بالكَسْرِ لا حَذَرَك .  <br />
يقال : أخَذَ حِذْرَهُ ؛ إذا تَيَقَّظَ واحْتَرَزَ من المَخُوفِ ؛ كأنه جَعَلَ الحِذْرَ آلَتَهُ التي يقي بها نَفْسَه ،   والمَعْنَى : احْذَرُوا واحْتَذِرُوا من العَدُّوِّ ،   ولا تمكِّنُوه من أنفُسِكم .  <br />
وقال الوَاحِدِيُّ{[8751]} : فيه قَوْلاَنِ :<br />
أحدهما : المُرَاد بالحِذْرِ [ ها ]{[8752]} هُنَا السِّلاح ،   والمعنى : خُذُوا سِلاَحَكُم ،   والسِّلاح يسمى حِذْراً ؛ لأنَّه يُتَّقَى ويُحْذَر .  <br />
والثاني :  "  خذوا حذركم  "  بمعنى : احْذَرُوا عَدُوَّكم ،   فعلى الأوَّل : الأمْر بأخذ السِّلاح مُصَرَّحٌ به ،   وعلى الثَّانِي : أخذ السِّلاح مَدْلُول{[8753]} عليه بِفَحْوَى{[8754]} الكَلامِ .  <br />
فإن قيل : إنَّ الَّذِي أمَر اللَّه - تعالى - بالحذْرِ عَنْهُ إن كان يُفْضي إلى الوُجُودِ ،   لم يَنْعَدِم ،   وإن كان الحذْر يُفْضِي إلى العَدَمِ ،   فلا حَاجَة إلى الحذْر ،   فعلى التَّقْديرَيْن الأمْر بالحذْر عَبَث{[8755]} ،   قال - عليه السلام - :  "  المَقْدُورُ كَائِنٌ  "  وقيل{[8756]} : الحذر لا يُغْنِي عن القَدَر .  <br />
فالجوابُ : أن هذا الكَلاَم يُبْطِل القَوْل بالشَّرَائِع ؛ فإنه يُقَالُ : إن كان الإنْسَان من أهْلِ السَّعَادة في قَضَاءِ اللَّه وقدرِه ،   فلا حَاجَة إلى الإيمَانِ ،   وإن كان من أهْل الشَّقَاءِ ،   لم ينْفَعْه [ الإيمانُ و ]{[8757]} الطَّاعَة ،   فهذا يفضي إلى سُقُوط التَّكْلِيف بالكُلِّيّة .  <br />
واعلم أنه لما كَان الكُلُّ بِقَضَاء اللَّه - تعالى - كان الأمْر بالحَذَرِ أيْضاً دَاخلاً بالقَدَر ،   وكان قَوْل القَائِل : أي فَائِدة بالحَذَر كَلاَماً مُتَنَاقِضاً ؛ لأنه لما كان هذا الحَذَر مُقَدَّراً ،   فأيُّ فائدة في هَذَا السُّؤال الطَّاعِن في الحَذَرِ .  <br />
قوله :  "  فانفروا [ ثبات ]{[8758]}  "  يقال : نَفَر القَوْم يَنْفِرُون نَفْراً ونَفِيراً ،   إذا نَهَضُوا لِقِتَال عَدُّوِّ{[8759]} ،   وخَرَجُوا للحَرْبِ ،   واستنْفَر الإمَامُ النَّاس لجِهَاد العَدُوّ ،   فَنَفَرُوا يَنْفِرُون : إذا حَثَّهُم على النَّفِير{[8760]} وَدَعَاهُم إلَيْه ؛ ومنه قوله - عليه السلام - :  " [ و ]{[8761]} إذا اسْتُنْفِرْتُم فانفرُوا  "  والنَّفِير{[8762]} : اسم للقَوْمِ الَّذِين يَنْفِرُون ؛ ومنه يُقال : فلان لاَ فِي العِيرِ ولا فِي النَّفِيرِ .  <br />
وقال النُّحَاة{[8763]} : أصْلُ هذا الحَرْف{[8764]} من النُّفُور والنِّفَارِ ؛ وهو الفَزَع ،   يقال : [ نَفر ]{[8765]} إليه ؛ ونَفَر مِنْهُ ؛ إذا فَزع منه وكَرِهَهُ ،   وفي مُضَارعه لُغَتَان : ضمُّ العَيْنِ وكَسْرِهَا ،   وقيل : يُقَال : نَفر الرَّجُل يَنْفِرُ بالكَسْرِ ،   ونَفَرَت الدَّابَّة تَنْفُر بالضَّمِّ [ ففرَّقُوا بَيْنَهُما في المُضَارع ،   وهذا الفَرْق يردُّه قِرَاءَة الأعْمَش :  " فانفُروا  "   "  أو انفُروا  "  بالضم ]{[8766]} فيهما ،   والمَصْدَر النَّفِير ،   والنُّفُور ،   والنَّفْر : الجماعة كالقَوْم والرَّهْط .  <br />
[ قوله ]{[8767]} :  "  ثبات  "  نصب على الحَالِ ،   وكذا  "  جميعاً  "  والمَعْنَى  "  انْفِرُوا جَمَاعاتٍ [ متفرِّقَة ]{[8768]} [ أي ]{[8769]} سَرِيّة بعد سَرِيّة ،   أو مُجْتَمِعِين كَوْكَبَةً وَاحِدَة ،   وهذا{[8770]} المَعْنَى الَّذي أراد الشَّاعِر في قوله : [ البسيط ]<br />
 .   .   .   *** طَارُوا إلَيْه زَرَافاتٍ وَوُحْدَانَا{[8771]}<br />
ومثله قوله :<br />
{ فَإنْ خِفْتُمْ فَرِجَالاً أَوْ رُكْبَاناً } أي{[8772]} : على أيّ الحَالَتَيْن كُنْتُم فَصَلُّوا .  <br />
قال أبُو حَيَّان{[8773]} : ولم يُقْرَأ  "  ثبات  "  فيما عَلِمْت{[8774]} إلا بكَسْر التَّاء .   انتهى .  <br />
وهذه هي اللُّغَةُ الفَصيحَة ،   وبَعْض العَرَب يَنْصِب جَمْع المُؤنَّث السَّالم إذا كان مُعْتَلَّ اللام مُعوضاً منها تاء التأنيث بالفَتْحَة{[8775]} ،   وأنشد الفرَّاء : [ الطويل ]<br />
فَلَمَّا جَلاَهَا بالأيَّام تَحَيَّزَتْ *** ثُبَاتاً عَلَيْهَا ذُلُّهَا واكْتئابُهَا{[8776]} {[8777]}<br />
وقرئ{[8778]} شاذاً : { وَيَجْعَلُونَ لِلَّهِ الْبَنَاتِ } [ النحل : 57 ] [ بالفتحة ]{[8779]} ،   وحكي : سَمِعْتُ لغاتَهُم ،   وزعم الفَارِسي أن الوَارِدَ من ذلك مُفْردٌ ردت لامهُ ؛ لأن الأصْل  " لُغَوَة  "  ؛ فلما رُدَّت اللام ،   قُلِبَت ألفاً ،   وقد رُدَّ على الفَارِسي : بأنّه يلْزَمُهُ الجَمع بين العِوَض والمُعَوَّضِ مِنْه ،   ويَرُدُّ عليه أيْضاً القِرَاءة المتقَدِّمة في الثبات ؛ لأن المُفْرَد منه مكْسُورُ الفَاءِ .  <br />
[  "  وثبات  "  جَمْعُ ثُبَة ،   ووزنها في الأصْل : فُعَلَة ،   كَحُطَمة ،   و ]{[8780]} إنما حُذِفَت لامُها وعُوض عنها تاءُ التَّأنِيثِ ،   وهل لامها واواً أو يَاءً ؟ قولان :<br />
حُجَّة القَوْل الأول : أنها مُشتقَّة من [ ثَبَا يُثْبُو ؛ كَخَلا يَخْلُو ،   أي : اجْتَمع .  <br />
وحُجَّةُ القول الثاني : أنها مُشْتَقة من ]{[8781]} ثبيت على الرجل إذا أثْنيت{[8782]} عليه ؛ كأنك جمعت مَحاسنه ،   وتجمع بالألفِ والتَّاءِ ،   وبالوَاوِ والنَّونِ ،   ويجوز في فَائِهَا{[8783]} حين تُجْمَع على  "  ثُبين  "  الضَّم والكَسْر ،   وكذا ما أشْبَهَهَا ،   نحو : قُلة{[8784]} ،   وبُرة{[8785]} ،   ما لم يُجْمَع جَمْع تكْسِير .  <br />
والثُّبَة : الجَمَاعة من الرِّجَال تكُون فَوْقَ العَشرة ،   وقيل : الاثْنَانِ والثَّلاثة ،   وتُصَغَّر على  "  ثُبْيَة  "  ،   بردِّ المَحْذُوف ،   وأما  "  ثُبة الحَوْضِ  "  وهي وَسطُهُ ،   فالمحذُوفُ عَيْنُها ،   لأنَّها من باب المَاء ،   أي : يَرْجِع ،   تُصَغِّر على  "  ثُوَيْبَةٍ  "  ؛ كقولك في تَصْغيرِ سَنَة : سُنَيْهَة .  <br />
فصل<br />
قال القرطبي{[8786]} : قيل إن هذه الآية مَنْسُوخة بقوله : { انْفِرُواْ خِفَافاً وَثِقَالاً }<br />
[ التوبة : 41 ] وبقوله { إِلاَّ تَنفِرُواْ يُعَذِّبْكُمْ عَذَاباً أَلِيماً } [ التوبة : 39 ] [ ولأن يكُون ]{[8787]} { انْفِرُواْ خِفَافاً وَثِقَالاً } [ التوبة : 41 ] مَنْسُوخاً بقوله : { فَانفِرُواْ ثُبَاتٍ أَوِ انْفِرُواْ جَمِيعاً } ،   وبقوله : { وَمَا كَانَ الْمُؤْمِنُونَ لِيَنفِرُواْ كَآفَّةً } [ التوبة : 122 ] أولى ؛ لأن فرض الجِهَادِ على الكِفْاية ،   فمتى سَدَّ الثُّغُورَ{[8788]} بَعْضُ المسْلِمِين ،   أسْقِطَ الفَرْضُ{[8789]} عن البَاقِينَ .  <br />
قال : والصَّحِيحُ أن الآيَتَيْنِ محكْمَتَانِ ،   إحَدَاهما : في الوَقْتِ الذي يُحْتَاجُ فيه إلى تعيُّن الجَميع ،   والأخْرَى : عند الاكْتِفَاء بِطَائِفَةٍ دُون غَيْرِهَا .
