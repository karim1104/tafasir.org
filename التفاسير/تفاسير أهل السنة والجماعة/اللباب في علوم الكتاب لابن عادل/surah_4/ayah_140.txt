قرأ الجماعة :  "  نُزِّل  "  مبنياً للمفعول ،   وعاصم{[10114]} ويعقوب قَرَآه  "  نَزَّلَ  "  مبنياً للفاعِل ،   وأبو حَيْوة وحُميد{[10115]} :  "  نَزَل  "  مخففاً مَبْنياً للفاعِل ،   والنخعي{[10116]} :  "  أُنْزِل  "  بالهَمْزَة مبنياً للمفعُول .  <br />
والقائمُ مقامَ الفاعِل في قراءة الجَمَاعة والنَّخعي ،   هو  "  أنْ  "  وما في حيِّزها ،   أي : وقد نَزَّل عليْكُم المَنْعَ من مُجَالستِهِم عند سَماعِكم الكُفْر بالآيَات ،   والاسْتِهْزَاء بها .  <br />
وأمّا في قراءة عاصمٍ : ف  "  أنْ  "  مع ما بعدها في مَحَلِّ نصبٍ مفعولاً به ب  "  نزَّل  "  ،   والفاعل ضميرُ الله - تعالى - كما تقدَّم .  <br />
وأما في قِرَاءة أبي حَيْوة وحمَيد : فمحَلُّها رفعٌ بالفاعِليّة ل  "  نزل  "  مخففاً ،   فمحَلُّها : إمّا نَصْب على قِرَاءة عَاصِمٍ ،   أو رَفْع على قِراءة غيره ،   ولكن الرَّفْع مختلف .  <br />
فصل<br />
قال المفسِّرون : المَعْنَى : وقد نَزَّل عليكم يا معشر المُؤمنين{[10117]} ،   { أَنْ إِذَا سَمِعْتُمْ آيَاتِ اللَّهِ } يعني : القُرْآن { يكفر بها ويستهزأ بها فلا تقعدوا معهم } يعني : مع المُسْتَهْزِئين { حتى يخوضوا في حديث غيره } ،   وذلك أنَّ المُشركِينَ كانُوا يخوضُونَ في مُجَالَسَتِهم في ذِكْر القُرْآن ،   يستَهْزِئُون به ،   فأنْزَل الله - تعالى - { وَإِذَا رَأَيْتَ الَّذِينَ يَخُوضُونَ فِي آيَاتِنَا فَأَعْرِضْ عَنْهُمْ حَتَّى يَخُوضُواْ فِي حَدِيثٍ غَيْرِهِ } [ الأنعام : 68 ] وهذه الآية نَزَلَتْ في مَكَّة .  <br />
ثم إن أحْبَار اليهُود بالمدينَة ،   كانوا يَفْعَلُون فعل المُشْرِكِين ،   وكان المُنافِقُون يَقْعُدون معهُم ،   ويُوافِقُونهم على ذلك ،   فقال - تعالى - مُخَاطِباً لَهُم : { أنه قد نزل عليكم في الكتاب أن إذا سمعتم آيات الله يكفر بها ويستهزأ بها } والمعنى : إذا سَمِعْتُم الكُفْر بآيَات اللَّه والاستهزاء بِهَا ،   لكنَّه أوْقع فعل السماع على الآية ،   والمُراد بها : سَمَاع الاسْتِهْزَاء .  <br />
قال الكسَائِيُّ{[10118]} : هو كما يُقَال : سَمِعْت عَبْد اللَّه يُلاَم{[10119]} .  <br />
قال ابن الخطِيب{[10120]} : وعندي فيه وَجْه آخَر : وهو أنْ يَكُون المَعْنَى : إذا سَمِعْتُم آيَاتِ اللَّه حَالَ ما يُكْفَر بها ويستهزأ بها ،   وعلى هذا فلا حَاجَة لما قاله الكسَائِيُّ .  <br />
قوله :  "  أنْ إذَا  "   "  أن  "  هذه هي المُخَفَّفةُ من الثَّقيلة ،   واسمُهَا : ضِمِير الأمْر والشَّأن ،   أي : أنَّ الأمْر والشأن إذا سَمعْتُم الكُفْر والاسْتهْزَاء ،   فلا تَقْعُدُوا .  <br />
قال أبو حيان : وما قَدَّره أبو البقاء{[10121]} من قوله :  "  أنَّكُم إذا سَمِعْتُم  "  ليس بجَيِّد ،   لأن  "  أن  "  المخففة لا تَعْمَل إلاَّ في ضِمِير الشَّأن ،   إلا في ضرورةٍ ؛ كقوله : [ الطويل ]<br />
فَلَوْ أنْكِ فِي يَوْمِ الرَّخَاءِ سألْتِني *** طَلاَقَكِ لَمْ أبْخَلْ وَأنْتِ صَدِيق{[10122]}<br />
قال شهاب الدين : هكذا قال ،   ولم أره أنا في إعراب أبِي البَقَاءِ إلا أنَّه بالهَاءِ دون الكَافِ والمَيم ،   والجملةُ الشَّرْطِية المُنْعَقدةُ من  "  إذا  "  وجوابها في مَحَلِّ رَفْع ،   خَبَراً ل  "  أنْ  "  ،   ومن مَجِيء الجُمْلة الشرطيَّة خبراً ل  "  أنْ  "  المُخَفَّفَة : قوله : [ الكامل ]<br />
فَعَلِمْتُ أنْ مَا تَتَّقُوهُ فَإنَّهُ *** جَزْرٌ لِخَامِعَةٍ وفَرْخِ عُقَابِ{[10123]}<br />
ف  "  مَا  "  شَرْطيةٌ ،   و  "  فإنه  "  جَوابُها ،   والجُمْلَةُ خبرٌ ل  "  أنْ  "  المخفَّفَةِ .  <br />
قوله :  "  يُكْفَرُ بِهَا  "  في محلِّ نَصْب على الحَالِ من الآيات ،   و  "  بها  "  في محلِّ رفع ؛ لقيامِه مقامَ الفاعلِ ،   وكذلِك في قوله :  "  يُسْتَهْزَأ بها  "  والأصل : يكفر بها أحدٌ ،   فلمَّا حُذِف الفاعلُ ،   قام الجارُّ والمَجْرُورُ مقامَه ،   ولذلك رُوعِي هذا الفَاعِلُ المَحْذُوف ،   فعاد عليه الضَّمِيرُ من قوله : { مَعَهُمْ حتى يخوضوا } كأنه قِيل : إذا سَمِعْتُم آياتِ الله يَكْفرُ بها المُشْرِكُون ،   ويَسْتَهزِئُ بها المُنَافِقُون ،   فلا تَقْعدوا مَعَهُم حتى يخُوضُوا في حَديثٍ غيره ،   أي : غير حَدِيث الكُفْر والاستهزاء ،   فعاد الضَّمير في  "  غيره  "  على ما دَلَّ عليه المَعْنَى .  <br />
وقيل : الضَّمير في  "  غيره  "  يجُوزُ أنْ يعودَ على الكُفْر والاستِهْزَاء المفهُومَيْن من قوله :  "  يُكْفَر بِهَا  "  و  "  وَيُسْتَهْزَأُ بِهَا  "  ،   [ وإنما أفْرَد الضَّمِير وإن كان المُرَاد به شَيْئين ؛ لأحد أمرين : ]<br />
إمَّا لأنَّ الكُفر والاستِهْزَاء شيءٌ واحدٌ في المعْنَى .  <br />
وإمَّا لإجراءِ الضَّميرِ مُجْرى اسم الإشَارةِ ،   نحو : { عَوَانٌ بَيْنَ ذَلِكَ } [ البقرة : 68 ] .   وقوله : [ الرجز ]<br />
كَأنَّهُ فِي الجِلْدِ تَوْلِيعُ البَهَقْ{[10124]} *** .   .   .  <br />
وقد تقدَّم تَحْقِيقُه في البقرة ،   و  "  حتى  "  : غايةٌ للنَّهْي ،   والمعنى : أنه يجُوز مُجَالستهم عند خَوْضِهم في غير الكُفْر والاستِهْزَاء .  <br />
قال الضَّحاك : عن ابن عبَّاسٍ : دخل في هذه الآيَة كل مُحْدِث في الدِّين ،   وكل مُبْتَدِع إلى يَوْم القِيَامَةِ{[10125]} .  <br />
قوله : { إنكم إذاً مثلهم }  "  إذاً  "  هنا : مُلغَاةٌ ؛ لوقوعها بين مُبْتَدأ وخبر ،   والجمهور على رفعِ اللام في  "  مثلُهم  "  على خَبَرِ الابتداء{[10126]} ،   وقرئ شاذاً بفتحها ،   وفيها تَخْريجَان :<br />
أحدهما : - وهو قولُ البصْريِّين - أنه خبر أيضاً ،   وإنما فُتِح لإضافته إلى غير مُتَمَكِّن ؛ كقوله - تعالى - : { إِنَّهُ لَحَقٌّ مِّثْلَ مَا أَنَّكُمْ تَنطِقُونَ } [ الذاريات : 23 ] بفتح اللاَّم ،   وقول الفرَزْدَق : [ البسيط ]<br />
 .   .   .   *** وإذْ مَا مِثْلَهُمْ بَشَرُ{[10127]}<br />
في أحدِ الأوجه .  <br />
والثَّاني : - وهو قولُ الكُوفيِّين - إن  "  مِثْل  "  يَجُوز نصبها على المَحَلِّ ،   أي : الظرف ،   ويُجيزُون :  "  زيد مِثْلَك  "  بالنَّصب على المحلِّ أي : زيدٌ في مثل حالك ،   وأفرد  "  مثل  "  هُنَا ،   وإن أخْبَرَ به عن جَمْع ولم يُطابق به كما طابق ما قبله في قوله : { ثُمَّ لاَ يَكُونُواْ أَمْثَالَكُم } [ محمد : 38 ] ،   وقوله : { وَحُورٌ عِينٌ كَأَمْثَالِ } [ الواقعة : 22 ،   23 ] .  <br />
قال أبُو البقاء{[10128]} وغيره : لأنه قصد به هُنَا المصدر ،   فوحَّد كما وحَّد في قوله : { أَنُؤْمِنُ لِبَشَرَيْنِ مِثْلِنَا } [ المؤمنون : 47 ] .   وتحرير المَعْنَى : أن التقدير : إنَّ عصيانكم مثل عصيانهم ،   إلا أنَّ تقدير المصدريّة في قوله :  "  لِبَشَريْنِ مِثْلِنَا  "  قلق .  <br />
فصل في معنى الآية<br />
والمعنى : أنكم إذاً مِثْلُهُم ،   إن قعدْتُم عندهم وهُم يَخُوضُون ويَسْتهزِئُون ،   ورضيتم بِهِ ،   فأنتم كُفَّار مِثْلُهم ،   وإن خَاضُوا في حَدِيث غَيْرِه ،   فلا بأس بالقُعُود مَعَهم مع الكَرَاهَة .  <br />
قال الحَسَن : لا يجوز القُعُود معهم وإن خَاضُوا في حَدِيث غَيْره{[10129]} ؛ لقوله{[10130]} - تعالى - :<br />
{ وَإِمَّا يُنسِيَنَّكَ الشَّيْطَانُ فَلاَ تَقْعُدْ بَعْدَ الذِّكْرَى مَعَ الْقَوْمِ الظَّالِمِينَ } [ الأنعام : 68 ] والأكثرون على الأوَّل ،   وآية الأنعام مَكِّية وهذه مَدَنِيَّة ،   والمتأخِّر{[10131]} أوْلَى .  <br />
فصل<br />
قال بعض العُلَمَاء : هذا يدل على أنَّ من رَضِي بالكُفْرِ ،   فهو كافِرٌ ،   ومن رَضِيَ بمنكر يَرَاه ،   وخالط أهْلَه وإن لَمْ يُبَاشِر ذلك ،   كان في الإثْمِ بمَنْزِلة المُبَاشِر لهذه الآية ،   وإن لم يَرْض وحَضَر خَوْفاً وتقية ،   فلا .  <br />
قوله : { إِنَّ اللَّهَ جَامِعُ الْمُنَافِقِينَ وَالْكَافِرِينَ فِي جَهَنَّمَ جَمِيعاً } أي : كما اجتمعوا على الاسْتِهْزَاء بآيَاتِ اللَّه في الدُّنْيا ،   فكذلك يَجْتِمَعُون{[10132]} في العَذَاب يوم القِيَامَةِ ،   وأراد : جامعٌ بالتنوين ؛ لأنَّه بعدما جَمَعَهُم حذف التَّنْوِين ؛ استخفافاً من اللَّفْظ ،   وهو مرادٌ في الحَقِيقَةِ .
