لما ذَكَرَ الجِهَاد ذَكَرَ أحد الأمُور التي يَحْتَاج إليها المُجَاهِد ،   وهو مَعْرِفَة كَيْفِيَّة أداء الصَّلاةِ في الخَوْف ،   والاشْتِغَال بمُحارَبَة العَدُوِّ .  <br />
 "  أن تقصروا  "  : هذا على حَذْفِ الخَافِضِ ،   أي : في أن تَقْصُرُوا ،   فيكونُ في مَحَلِّ  " أنْ "  الوَجْهَان المَشْهُوران ،   وهذا الجَارُّ يتعلَّقُ بلفظِ  "  جُنَاح  "  أي : فََليْسَ عَلَيْكُم جُنَاحٌ في قَصْرِ الصَّلاة .  <br />
قال الوَاحِدِيُّ : يُقَال : قَصَرَ فلان صَلاَتهُ ،   وأقصرها وقَصَّرَها ،   وكُلُّ جَائِزٌ .  <br />
والجمهور على  "  تَقْصُروا  "  من  "  قَصَرَ  "  ثلاثياً{[9511]} ،   وقرأ ابن عبَّاس :  "  تُقْصِروا  "  من  " أقْصر  "  ،   وهما لُغَتَان : قَصَر وأقْصَر ،   حكاهما الأزْهَرِيُّ ،   وقرأ الضَّبِيُّ عن رجاله بقراءة ابن عبَّاسٍ ،   وقرأ الزُّهري :  "  تُقَصِّروا  "  مشدّداً على التَّكْثِيرِ .  <br />
قوله :  "  من الصَّلاة  "  في  "  مِنْ  "  وَجْهَان :<br />
أظهرهُما : إنها تَبْعِيضيَّةٌ ،   وهذا مَعْنَى قول أبي البقاء{[9512]} ،   وزعم أنه مَذْهَب سيبويه ،   وأنَّها صفةٌ لمَحْذُوفٍ ،   تقديرُه : شيئاً من الصلاة .  <br />
والثاني : أنَّها زائدةٌ ،   وهذا رأيُ الأخْفَش فإنه لا يَشْتَرِطُ في زِيَادتها شيئاً ،   و  "  أن يَفْتِنَكم  "  : مفعُول  "  خفِْتُم  "  .  <br />
وقرأ عبد الله{[9513]} بن مَسْعُود ،   وأبَيُّ :  "  من الصَّلاة أن يفتنكُم  "  بإسقاط الجُمْلة الشَّرْطيّة ،   و  "  أن يفتِنَكُم  "  على هذه القراءة ،   مَفْعُولٌ من أجْله ،   ولغةُ الحِجَاز :  "  فَتَنَ  "  ثُلاثياً ،   وتميم وقَيْس :  "  أفْتَن  "  رُبَاعياً .  <br />
فصل<br />
لفظ القَصْر مُشْعِرٌ بالتَّخْفِيف ؛ لأنه لَيْس صريحاً في أنَّ المُرادَ : هو القَصر في عَددِ الركعَات ،   أي : في كيْفِيَّة أدائِها ،   فلا جَرَم حصل في الآيَة قولان :<br />
الأوّل : قَوْل الجُمْهُور أنَّ المراد مِنْه : القَصْر في عدد الرَّكَعَات والقَائِلون بهذا القَوْلِ اختلفُوا على قَوْلَيْن :<br />
الأوّل : أن المراد مِنْهُ : صلاة المُسَافِر ؛ وهو أنَّ كُلَّ صَلاَةٍ تكُونُ في الحَضَر أرْبَع رَكَعَاتٍ ،   فإنها تَصِير{[9514]} في السَّفَرِ رَكْعَتِيْنِ ،   وعلى هَذَا إنَّما يَدْخُل القَصْر في الرُّبَاعِيَّة خاصَّة .  <br />
الثاني : أنَّ المراد : صلاَة الخَوْف في السَّفَر ،   وهو قَوْل ابن عبَّاسٍ ،   وجَابِر بن عبد الله ،   وجماعة ،   قال ابن عبَّاس : فَرَضَ اللَّه صلاة الحَضَر أرْبَعاً ،   وصلاة [ السَّفر ركعتَيْن ]{[9515]} ،   وصلاة الخَوْفِ رَكْعَةً على لسان نَبِيِّكُم صلى الله عليه وسلم{[9516]} .  <br />
القول الثاني : أن المُرَاد من القَصْر : التَّخْفِيف في كيْفِيَّة أداء الرَّكَعَات ،   وهو أن يُكتفى في الصَّلاة بالإيمَاءِ والإشَارَة بدل الرُّكُوع والسُّجُود ،   وأن يَجُوز المَشْيُ في الصَّلاة ،   وأن تجُوز الصلاة عند تلَطُّخ الثَّوْب بالدَّمِ{[9517]} وهو الصَّلاة حال التِحَام القِتَال ،   [ وهو مَرْوِيٌّ عن ابْن عَبَّاسِ وطاووس{[9518]} ،   واحتَجُّوا : بأنَّ خوف فِتْنَة العَدُوِّ لا تَزُول فيمَا يُؤتَى بِرَكْعَتِيْن على تَمَامِ أوْصَافِهَا ،   وإنما عَيَّن ذَلِكَ فيما يَشْتَدُّ فيه الخَوْف حَالَ التِحام القِتَال ]{[9519]} ،   وهذا ضَعِيفٌ ؛ لأنه يُمْكِن إن يُقَال : إن المُسَافر إذا كانَت الصَّلاة قَلِيلَة الرَّكَعَاتِ ،   فيمكنه{[9520]} أنْ يَأتِيَ بِهَا على وَجهٍ لا يَكُون خَصْمُه عَالِماً بَِكَوْنِهِ مُصَلِّياً أما إذا كثُرت الرَّكَعَات ،   طالَت الصَّلاة ،   ولا يُمْكِنُه أن يَأتِي بها على حين غَفْلَةٍ مع العَدُوِّ ،   وحَمْل لفظِ القصْر على إسْقَاط [ بَعْض ]{[9521]} الرَّكَعَاتِ أوْلَى لوجوه :<br />
أحدُها : ما رُوِيَ عن يعلى{[9522]} بن أمَيَّة أنَّه قال : قلت لعمر بن الخَطَّاب - رضي الله عنه - : كيف نَقْصُر وقد أمِنَّا ،   وقد قال الله - تعالى - : { فَلَيْسَ عَلَيْكُمْ جُنَاحٌ أَن تَقْصُرُواْ مِنَ الصَّلاَةِ إِنْ خِفْتُمْ } فقال : عَجِبْتُ مِمَّا عَجِبْتَ مِنْهُ ؛ فَسَأَلْتُ الرَّسُولَ - عليه الصلاة والسلام - فَقَالَ :  "  صَدَقَةٌ تَصَدَّقَ اللَّهُ بِهَا عَلَيْكُم ،   فاقْبَلُوا صَدَقَتَهُ{[9523]}  " {[9524]} ،   وهذا يدلُّ على أن القَصْر المذكُور في الآية{[9525]} ،   وهو القصْرُ في عَدَدِ الركَعَاتِ .  <br />
الثاني : أن القَصْر عبارةٌ عن أن يؤتى بِبَعْض الشَّيْء ويقتصر عَلَيْه ،   فإمَّا أن يُؤتَى بشَيٍْ آخَرَ ،   فذلك لا يُسَمَّى قَصراً ،   ومعْلُوم : أن إقامَة الإيماء{[9526]} [ مَقَامَ ]{[9527]} الرُّكُوع والسُّجُود ،   وتَجويز المَشْي في الصَّلاة ،   وتَجْوِيز الصَّلاة مع الثَوْب المُلَطَّخ بالدَّم ،   ليس شيء من ذلك قَصْراً ؛ بل كُلُّها إثباتٌ لأحْكَامِ جديدةٍ ،   وإقَامَة لشَيْءٍ مَقَامَ شَيْء آخَرَ .  <br />
الثالث : أن  "  مِنْ  "  في قوله :  "  مِنَ الصّلاة  "  للتَّبْعيضِ ،   وذلك يُوجِبُ جَوازَ الاقْتِصَارِ على بَعْضِ الصَّلاة .  <br />
الرابع : أن لَفْظَ القَصْر كان في عُرْفِهِم مَخصُوصاً بتنْقِيصِ عددِ الرَّكَعَاتِ ،   ولهذا لمَّا صلَّى النَّبِيُّ صلى الله عليه وسلم الظُّهر رَكْعِتَيْنِ ،   قال ذو اليَدَيْنِ :  "  أقصُرَتِ الصلاةُ أم نَسِيتَ ؟  "  .  <br />
الخامس : القَصْر بمعنى : تغير هَيْئَةٍ [ الصَّلاة ]{[9528]} المَذْكُورة في الآيةِ الَّتِي بَعْدَهَا توجب أن يكُون المُراد من هذه الآيَة بَيَان القَصْر ،   بمعنى حذف بَعْض الرَّكعَات ،   لئلا يَلْزَم التَّكْرَار .  <br />
فصل هل الأفضل الإتمام أم القصر ؟<br />
قصر الصَّلاة في السَّفر{[9529]} جَائِز بالإجْمَاعِ ،   واخْتَلَفُوا في جواز الإتْمَام .  <br />
فذهب أكْثَرُهُم إلى أن القَصْرَ واجِبٌ ،   وهو قَوْل عُمَر وعَلِيِّ ،   وابن عُمَر ،   وجابر ،   وابن عبَّاس ،   وبه قال الحَسَن وعُمَر بن عَبْد العَزِيز ،   وقتادة ،   وهو قول مالِكٍ وأصحاب الرأي بما روتْ{[9530]} عائِشَة - رضي الله عنها - ،   قالت :  "  الصَّلاةُ أوَّل ما فُرِضت [ رَكْعَتَيْن في الحَضَر والسَّفَر فأقِرَّتْ ]{[9531]} صَلاَة السَّفَر ،   وأتِمَّت صَلاةُ الحَضَر  "  .  <br />
وذهب قَومٌ إلى جواز الإتْمَام ،   رُوِيَ ذلك عن عُثْمَان [ وسَعْد ]{[9532]} بن أبِي وَقَّاصٍ ،   وبه قال الشَّافِعِيُّ إن شاءَ أتَمَّ ،   وإن شاءَ قَصَر ،   والقَصْر أفْضَل .  <br />
فصل<br />
قال أهل الظَّاهِر : قَلِيلُ السَّفَر وكَثِيرُه{[9533]} سَوَاء ؛ لِظَاهِر الآية ،   فإن الآيَة مرتَّبَةٌ من شَرْط وجَزَاء ،   فإذا وُجِدَ الشَّرْط وهو الضَّرْب في الأرْض ،   ترتَّبَ{[9534]} عَلَيْه [ الجزاء ]{[9535]} سواء كَانَ طَويلاً أوْ قَصِيراً ،   وذلك مَرْوِيٌّ عن أنَسٍ ،   وقال عَمْرو بن دِينَار : قال لي جَابر بن زَيد : أقْصِر بعَرَفَة .  <br />
فإن قيل : هذا يَقْتَضِي حُصُول الرُّخْصَة عند انْتِقَال الإنْسَان من مَحَلَّة إلى مَحَلَّة .   فالجواب : لا نُسَلِّم أنَّ هذا ضَرْب في الأرْضِ ،   وإن سُلِّم ،   فنقول : الإجْمَاع مُنْعَقِدٌ على أنَّه غير مُعْتَبَر ،   فَهَذا تَخْصِيصٌ بالإجْمَاع ،   والعامُّ بعد التَّخْصيص حُجَّة .  <br />
وقال الجُمْهُور : إن السَّفر ما لم يتقدَّر بِمقْدَار مَخْصُوصٍ ،   لم تَحْصُل فيه الرُّخْصَة ،   وقالوا : أجْمَع السَّلَف على أنَّ أقَل السَّفِر مقدَّرٌ ،   لأنه رُوِيَ عن عُمَر أنَّه يَقْصِر في يَوْم تامٍّ{[9536]} ؛ وبه قَالَ الزُّهرِي والأوْزَاعِيُّ .  <br />
وقال ابْن عَبَّاس : يَقْصرُ إذا زَادَ علي يَوْم ولَيْلَة{[9537]} .  <br />
قال أنَس : المُعْتَبَر خَمْسَة فَرَاسِخ{[9538]} ،   وقال الحَسَن : مَسِيرة لَيْلَتَيْن{[9539]} .  <br />
وقال الشَعْبِي ،   والنَّخعِي ،   وسعيد بن جُبَيْر : من الكُوفَة إلى المَدَائِن مسيرة ثلاثةَ أيّام{[9540]} ،   وهو قول أبِي حَنِيفَة ،   وروى الحَسَن بن زِيَاد ،   عن أبي حنيفة : أنَّهُ إذا سَافَر إلى موضع يكون مَسيرة يَوْمَيْن ،   وأكثر اليوم الثَّالِث ،   جاز القَصْر ،   وهكذا رَوَاهُ ابن سماعة ،   عن أبِي يُوسُف ومحمَّد .  <br />
وقال مَالِكٌ : أمْيَال بأمْيَال هَاشِم جَدَّ رسُول الله صلى الله عليه وسلم ،   وهو الذي قدَّر أمْيَال البَادِية ؛ كل ميلٍ اثْنَا عَشَر ألف قدم ،   وهي أرْبَعَةٌ آلاف خطوة ،   فإن كل ثَلاثَة أقْدام خُطْوة ،   قالوا : واختِلاَف النَّاس يدل على انْعِقَاد الإجْمَاع ،   على أن الحُكْم غير مَرْبُوط بِمُطْلَق السَّفَر .  <br />
قال أهل الظاهر : اضْطِرَابُهم يَدُلُّ على أنَّهم لم يَجِدُوا دَلِيلاً في تَقْدِير المُدَّة ،   إذ لو{[9541]} وَجَدُوه لما حَصَل الاضْطِرَاب ،   وأما سُكُوت [ سَائِر ]{[9542]} الصَّحَابَة ؛ فلعَلَّه كان لاعْتِقَادِهم أنّ الآية دَالَّة على ارتِبَاط الحُكْمِ بِمُطْلَقِ السَّفَر ،   وإذا كان الحُكْم مَذْكُوراً في نَصِّ القُرْآن ،   لم يكن بِهِم حَاجَةٌ إلى الاجْتِهَاد والاستِنْبَاطِ ؛ فلهذا سَكَتوا ،   فاستدل الحَنفيَّة على تقدير المُدَّة بقوله - عليه الصلاة والسلام :  "  يَمْسَحُ المُسَافِر ثَلاَثَة أيّامٍٍ  "   ؛ وهو يدل على أنَّه إذَا لَمْ يَحْصُل المَسْح ثلاثة أيَّام ،   لا يسَمَّى مسافِراً .  <br />
واستدل الشَّافعيَّة بما رَوَى مُجَاهِدٌ وعطاء ،   عن ابن عبَّاس ،   عن النبيّ صلى الله عليه وسلم قال :  "  يا أهْل مَكَّة ،   لا تَقْصُروا في أدْنَى من أرْبَعة بُرُد من مَكَّة إلى عُسْفَان  " {[9543]} قال أهل الظَّاهِر : وهذا تَخْصِيص لعُمُوم القُرْآن{[9544]} بخَيْر الوَاحِد ،   وهو لا يَجُوز ؛ لأن القُرْآن مَقْطُوع به والخَبَر مَظْنُونٌ ،   وقال عليه الصلاة والسلام :  "  إذا رُوِيَ عَنِّي حديثٌ فاعْرِضُوه على كِتَاب اللَّه - [ تعالى ]{[9545]} - ،   فإن وَافق ،   فَاقْبَلُوه ،   وإلا فَرُدُّوهُ  "  وهذا مُخَالِفٌ لعموم الكِتَاب ،   وأيْضاً فإنها أخْبَار وردَتْ في وَاقِعَةٍ تَعُمُّ الحاجَةُ إلى مَعْرفتها ؛ لأن الصَّحابة - رضي الله عنهم - كانُوا في أكْثَر الأوْقَات في السَّفَر والغَزْو ،   فلو كَانَت الرُّخْصَة مَخْصُوصَة بِسَفَرٍ مقدَّر ،   لعرفوها ونَقَلُوها نقلاً متواتراً ،   لا سِيَّمَا وهو عَلَى خِلاف ظَاهِر القُرْآن ،   وأيضاً : فدلائل الشَّافعيَّة ودلائل الحَنَفِيَّة متقابِلة مُتَدافعة فسقَطَت{[9546]} ووجبَ الرُّجُوع لِظَاهِرِ القُرْآنِ .  <br />
فصل<br />
خصَّ أهلُ الظَّاهر جواز القصر بِحَال الخَوْف ؛ لقوله - تعالى - : { فَلَيْسَ عَلَيْكُمْ جُنَاحٌ أَن تَقْصُرُواْ مِنَ الصَّلاَةِ إِنْ خِفْتُمْ } والمَشْرُوط بالشَّيْءِ عدمٌ ،   عند عَدَمِ ذلك الشَّيءِ ،   ولا يَجُوز دفع هذا الشَّرط بأخْبَار الآحَادِ ؛ لأن نَسْخَ القُرْآنِ بِخَبر الوَاحِدِ لا يَجُوز .  <br />
قوله : { إِنْ خِفْتُمْ أَن يَفْتِنَكُمُ الَّذِينَ كَفَرُواْ } فقيل : إن يَفْتِنُوكم عن إتْمَام الرُّكُوع [ والسُّجُود ]{[9547]} ،   وقيل :  "  أن يفتنكم  "  أي يغلبكم الَّذين كَفَرُوا في الصَّلاة ،   ونَظِيرُه قوله : [ تعالى ]{[9548]} : { على خوف من فرعون وملإيهم أن يفتنهم } عن الصلاة ؛ لأن كُلَّ مِحْنَة ،   وبَلِيّة ،   وشِدَّة فهي فِتْنَة ،   وجَواب الشَّرْط مَحْذُوف يَدُلُّ عليه ما قَبْلَه .  <br />
وقيل : الكَلاَم تَمَّ [ عند قوله :  "  مِنَ الصَّلاة  " {[9549]} ] والجملة الشَّرْطيةُ مُسْتَأنَفةٌ حتى قيلَ : إنها نَزَلَت بعد سَنَةٍ عن نُزُول ما قَبْلَها ،   وجوابُه حينئذٍ{[9550]} أيْضاً مَحْذُوف ،   ولكن يُقَدَّرُ من جِنْسِ ما بَعْدَه ،   وهذا قَوْلٌ ضَعِيفٌ ،   وتأخير نُزُولِهَا لا يَقْتَضِي استِئْنَافَهَا .  <br />
فصل<br />
اخْتَلَفُوا متى يَقْصُر :<br />
فالجُمْهُور على أنَّ المُسَافِر لا يَقْصُر حتَّى يَخْرُج من بُيُوت القَرْيَة [ وحينئذٍ ]{[9551]} هو ضَارِبٌ في الأرْضِ ،   وهو قول مَالِكٍ في المُدَوَّنة{[9552]} ،   وروي عنه : أنَّه إذا كَانَت قَرْية تَجمع أهْلُها لا يَقْصُرُ حتى يُجَاوِزَها بثلاَثَةِ أمْيَالٍ ،   وكذلك في الرُّجُوع ،   وعن الحَارِث بن أبي رَبيعة : إذا أرَادَ السَّفَر ،   يَقْصُرُ في مَنْزِلِه ؛ فيكون مَعْنَى قوله - [ تعالى ]{[9553]} - : { إِذَا ضَرَبْتُمْ [ فِي الأَرْضِ } معناه ]{[9554]} : إذا أرَدْتُم السَّفَر .   وعن مجاهد : لا يَقْصُر يَوْمَه الأوَّل حتى اللَّيْل ،   وهذا شَاذٌّ ؛ لأن النَّبِيَ صلى الله عليه وسلم صلى الظُّهْر بالمَدِينَة [ أرْبعاً ]{[9555]} ،   وصلَّى العَصْرَ بذي الحُلَيْفَةِ رَكْعَتَيْن ،   وبين المَدينَة وذي الحُلَيْفَة سِتَّة أمْيالٍ ،   أو سَبْعَة .  <br />
فصل<br />
وعلى المَسَافِر أن يَنْوِيَ القَصْرَ حين الإحْرَام ،   فإن افْتَتَحَ الصَّلاة بنيَّة{[9556]} القصر ،   ثُمَّ عزم على المُقَام في أثناء{[9557]} الصَّلاة ،   جعلها نَافِلَةً ،   فإن كان ذَلِكَ بَعْد أنْ صَلَّى منها رَكْعَةً [ واحِدَة ]{[9558]} ،   أضَاف إليها أخْرى [ وسلَّم ]{[9559]} ثم صلى صلاة مُقِيم ،   وقال الأبْهَرِيّ ،   وابن الجَلاَّب : هذا - والله أعْلم - اسْتِحْبَابٌ ،   ولو بَنَى على صَلاَته وأتمّها ،   أجْزأتْهُ .  <br />
قوله : { إِنَّ الْكَافِرِينَ كَانُواْ لَكُمْ عَدُوّاً مُّبِيناً } [ و ]{[9560]} المعنى : إن العدَاوَة بَيْنَكُم وبَيْنَ الكَافِرين قَديمَةٌ ،   والآن قد أظْهَرْتُم خلافَهُم في الدِّين فازْدَادَت{[9561]} عَدَاوتُهم لَكُم ،   فمن شِدَّة{[9562]} العَدَاوة ،   حارَبُوكم وقَصَدُوا إتلافكُم إن قَدَرُوا ،   فإن طَالَتْ صَلاتُكُم ،   فرُبَّمَا وَجَدُوا الفُرْصَة في قَتْلِكُم ؛ فلهذا رَخَّصْتُ لَكُم في قَصْر الصَّلاة .  <br />
قوله  "  لكم  "  متعلِّقٌ بمَحْذُوف : لأنه حالٌ من  "  عَدُوّاً  "  ،   فإنه في الأصْل صِفَةُ نَكِرَةٍ ،   ثم قُدِّم عَلَيْها ،   وأجاز أبُو البَقَاء{[9563]} أن يتعلَّق ب  "  كَانَ  "  ،   [ وفي المَسْألة ] كَلاَمٌ مرَّ تَفْصِيلُه .   وأفْرد  "  عَدُوّاً  "  وإن كَانَ المُرَادُ به الجَمْعَ لأنَّ العدُوَّ يسْتَوِي فيه الوَاحِد والجمع ؛ قال - تعالى - : { فَإِنَّهُمْ عَدُوٌّ لِي } وقد تقدّم تَحْقِيقُه في البَقَرة .  <br />
فصل في معنى الآية<br />
قوله : { إِنْ خِفْتُمْ أَن يَفْتِنَكُمُ الَّذِينَ كَفَرُواْ } متَّصِل بما بَعْدَه من صَلاَةِ الخَوْفِ ،   منفصل عَمَّا قَبْلَه ،   رُوِيَ عن أبي أيُّوبٍ الأنْصَارِي ،   أنَّه قال : نَزل قَوْلُه : { فَلَيْسَ عَلَيْكُمْ جُنَاحٌ أَن تَقْصُرُواْ مِنَ الصَّلاَةِ } هذا القَدْر ،   ثمَّ بعد حَوْلٍ سألُوا رسُول الله صلى الله عليه وسلم عن صَلاَةِ الخَوْفِ ؛ فنزل :  "  إن خفتم  "  أي : وإن خِفْتُم { أَن يَفْتِنَكُمُ الَّذِينَ كَفَرُواْ إِنَّ الْكَافِرِينَ كَانُواْ لَكُمْ عَدُوّاً مُّبِيناً ،   وَإِذَا كُنتَ فِيهِمْ } .  <br />
ومثله في القُرْآن كَثِيرٌ [ أن ]{[9564]} يجيء الخبر بِتَمَامِه ،   ثم يُنسق عَلَيْه خبرٌ آخَر ،   وهو في الظَّاهِر كالمُتَّصِل به ،   وهو مُنْفَصِلٌ عنه ؛ كقوله - تعالى - : { الآنَ حَصْحَصَ الْحَقُّ أَنَاْ رَاوَدْتُّهُ عَن نَّفْسِهِ وَإِنَّهُ لَمِنَ الصَّادِقِينَ } [ يوسف : 51 ] هذه حكاية عن امْرَأة العَزِيز ،   وقوله : { ذلِكَ{[9565]} لِيَعْلَمَ أَنِّي لَمْ أَخُنْهُ بِالْغَيْبِ } [ يوسف : 52 ] ،   إخبارٌ عن يُوسُف - عليه الصلاة والسلام - .
