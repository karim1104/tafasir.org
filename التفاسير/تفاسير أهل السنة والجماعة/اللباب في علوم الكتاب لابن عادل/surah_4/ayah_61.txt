لما بين - [ تعالى ]{[8548]} - رغبتَهُم في التَّحَاكُم إلى الطَّاغُوتِ [ بين ]{[8549]} نَفْرتَهم عن التَّحَاكُم إلى الرَّسُولِ ،   وقد تَقَدَّم الكَلاَمُ على { تَعَالَوْاْ } في آل عمران .  <br />
قوله : { رَأَيْتَ } فيها وجهان :<br />
أحدهما : [ أنها من رُؤيَة البَصَرِ ،   أي : مُجَاهَرَة وتصْرِيحاً ]{[8550]} .  <br />
[ والثاني : ]{[8551]} أنَّها من رُؤيَة القَلْب ،   أي عَلِمْتَ ؛ ف { يَصُدُّونَ } في محلِّ نَصْبٍ على الحَالِ على القَوْلِ الأوَّلِ ،   وفي مَحَلِّ المَفْعُول الثَّانِي على الثَّانِي .  <br />
وقوله : { صُدُوداً } فيه{[8552]} وجهان :<br />
أحدهما : أنه اسْمُ مَصْدَرٍ ،   والمصْدَر إنما هو الصَّدُّ ،   وهذا اخْتِيَار ابن عطيَّة{[8553]} ،   وعزَاه مكِّي للخَلِيل بن أحمد .  <br />
والثَّاني : أنه مَصْدر بِنَفْسِه ؛ يقال : صَدَّ صَدّاً وصُدُوداً ،   وقال بَعْضُهم : الصُّدُود مَصْدَرُ صَدَّ اللازم ،   والصَّدُّ مصدرُ صَدَّ المُتَعَدَّي ،   نحو : { فَصَدَّهُمْ عَنِ السَّبِيلِ }<br />
[ النمل : 24 ] والفعْل هنا مُتَعَد بالحَرْفِ لا بِنَفْسِه ،   فلذلك جاءَ مصدَرُه على  " فُعُول  "  ؛ لأن  "  فُعُولا  "  غالباً اللازِم ،   وهذا فيه نَظَر ؛ إذا لِقَائِل أن يقُول : هو هُنا مُتَعَدٍّ ،   غاية ما فيه أنه حَذَفَ المَفْعُول ،   أي : يَصُدُّون غيرَهُم ،   أو المُتَحاكِمين عنك صُدُوداً ،   وأما  "  فُعُول  "  فجاء في المُتَعَدِّي ،   نحو : لَزِمَهُ لُزُوماً ،   وفَتَنَهُ فُتُوناً .  <br />
ومعنى الآية : يعرضون عنك ،   وذكر المَصْدَر للتَّأكيد والمُبَالَغَة ؛ كأنه قال{[8554]} : صُدُوداً أيَّ صُدُودٍ .
