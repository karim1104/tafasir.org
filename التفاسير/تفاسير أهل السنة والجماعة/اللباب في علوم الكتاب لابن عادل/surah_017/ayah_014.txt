قوله تعالى : { اقرأ كَتَابَكَ } : على إضمار القول ،   أي : يقال له : اقرأ ،   وهذا القول : إمَّا صفةٌ أو حالٌ ،   كما في الجملة قبله .   وهذا القائل هو الله تعالى .  <br />
قال الحسن :  "  يَقْرءُوهُ أمِّيًّا كان ،   أو غير أمِّيٍّ  " {[20260]} .  <br />
وقال أبو بكر بن عبد الله - رضي الله عنه- : يؤتى المؤمنُ يوم القيامةِ بصحيفته ،   وحسناته في ظهرها ،   يغبطه الناس عليها ،   وسيئاته في جوف صحيفته ،   وهو يقرؤها ،   حتَّى إذا ظنَّ أنها قد أوبقته ،   قال الله له :  "  قَدْ غَفَرْتُ لَكَ فِيمَا بَينِي وبَينكَ  "  فيعظم سروره{[20261]} ويصير من الَّذين قال الله - عزَّ وجلَّ - في حقهم : { وُجُوهٌ يَوْمَئِذٍ مُّسْفِرَةٌ ضَاحِكَةٌ مُّسْتَبْشِرَةٌ } [ عبس : 38 ،   39 ] .  <br />
قوله تعالى : { كفى بِنَفْسِكَ } فيه ثلاثة أوجه :<br />
المشهور عند المعربين : أنَّ  "  كَفَى  "  فعل ماضٍ ،   والفاعل هو المجرور بالباء ،   وهي فيه مزيدة ،   ويدلُّ عليه أنها إذا حذفت ارتفع ؛ كقوله : [ الطويل ]<br />
ويُخْبِرنِي عَن غَائبِ المَرْءِ هَديهُ *** كَفَى الهَدْيُ عَمَّا غَيَّبَ المَرءُ مُخْبِرا{[20262]}<br />
وقوله : [ الطويل ]<br />
 .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   *** كَفَى الشَّيْبُ والإسلامُ للمَرْءِ نَاهِيَا{[20263]}<br />
وعلى هذا ؛ فكان ينبغي أن يؤنَّث الفعل ؛ لتأنيث فاعله ،   وإن كان مجروراً ؛ كقوله { مَآ آمَنَتْ قَبْلَهُمْ مِّن قَرْيَةٍ } [ المؤمنون : 6 ] { وَمَا تَأْتِيهِم مِّنْ آيَةٍ } [ الأنعام : 4 ] .   وقد يقال : إنه جاء على أحد الجائزين ؛ فإن التأنيث مجازيٌّ .  <br />
والثاني : أن الفاعل ضمير المخاطب ،   و  "  كفى  "  على هذا اسم فعل أمر ،   أي : اكتفِ ،   وهو ضعيف ؛ لقبولِ  "  كَفَى  "  علاماتِ الأفعالِ .  <br />
الثالث : أن فاعل  "  كََفَى  "  ضمير يعود على الاكتفاء ،   وتقدَّم الكلام على هذا .   و  "  اليَوْمَ  "  نصبٌ ب  "  كَفَى  "  .  <br />
قوله :  "  حَسِيباً  "  فيه وجهان :<br />
أحدهما : أنه تمييزٌ ،   قال الزمخشريُّ :  "  وهو بمعنى حاسبٍ ؛ كضريب القداح ؛ بمعنى ضاربها ،   وصريم بمعنى صارم ،   ذكرهما سيبويه{[20264]} ،   و  "  على  "  متعلقة به من قولك : حَسِبَ عليه كذا ،   ويجوز أن يكون بمعنى الكافي ووضع موضع الشَّهيد ،   فعدِّي ب  "  عَلَى  "  لأنَّ الشاهد يكفي المدَّعي ما أهمَّه ،   فإن قلت : لِمَ ذكر  "  حَسِيباً  "  ؟ قلت : لأنه بمنزلةِ الشاهد ،   والقاضي ،   والأمين ،   وهذه الأمور يتولاَّها الرجال ؛ فكأنَّه قيل : كفى بنفسك رجلاً حسيباً ،   ويجوز أن تتأوَّل النفس بمعنى الشخص ،   كما يقال : ثلاثةُ أنفسٍ  "  .   قلت : ومنه قول الشاعر : [ الوافر ]<br />
ثَلاثةُ أنْفُسٍ وثَلاثُ ذَودٍ *** لقَد جَارَ الزَّمانُ على عِيَالِي{[20265]}<br />
والثاني : أنه منصوب على الحال ،   وذكر لما تقدم ،   وقيل : حسيبٌ بمعنى محاسب ؛ كخليطٍ وجليسٍ بمعنى : مخالطٍ ومجالسٍ .  <br />
قال الحسن - رضي الله عنه - :  "  عَدلَ ،   والله ،   في حقّك من جعَلكَ حَسِيبَ نَفْسكَ  " {[20266]} .  <br />
وقال السديُّ :  "  يقول الكافر يومئذٍ : إنَّك قَضيْتَ أنَّك لست بظلام للعبيد ،   فاجعلني أحاسِبُ نفسِي فيقال له : اقرأ كتابك ،   كفى بنفسك اليوم عليك حسيباً  " {[20267]} .
