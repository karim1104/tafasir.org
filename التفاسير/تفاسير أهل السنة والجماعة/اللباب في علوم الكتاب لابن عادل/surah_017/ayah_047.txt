قوله تعالى : { بِمَا يَسْتَمِعُونَ } : الباء في  "  بما  "  متعلق ب  "  أعْلَمُ  "  .   وما كان من باب العلم والجهل في أفعل التفضيل ،   وأفعل في التعجب تعدَّى بالباء ؛ نحو : أنت أعلمُ به ،   وما أَعلمك به ! ! وهو أجهل به ،   وما أجهله به ! ! ومن غيرهما يتعدَّى في البابين باللام ؛ نحو : أنت أكسى للفقراء ،   و  "  مَا  "  بمعنى الذي ،   وهي عبارةٌ عن الاستخفاف والإعراض ،   فكأنه قال : نحن أعلم بالاستخفاف ،   والاستهزاءِ الذي يستمعون به ،   قاله ابن عطيَّة .  <br />
قوله :  "  به  "  فيه أوجه{[20459]} :<br />
أحدها : أنه حال ،   فيتعلق بمحذوف .  <br />
قال الزمخشري :  "  وبه في موضع الحالِ ،   كما تقول : يستمعون بالهزءِ ،   أي : هازئين  "  .  <br />
الثاني : أنها بمعنى اللامِ ،   أي : بما يستمعون له .  <br />
الثالث : أنها على بابها ،   أي : يستمعون بقلوبهم أو بظاهر أسماعهم ،   قالهما أبو البقاء .  <br />
الرابع : قال الحوفيُّ :  "  لم يقلْ يستمعونه ،   ولا يستمعونك ؛ لمَّا كان الغرضُ ليس الإخبار عن الاستماعِ فقط ،   وكان مضمَّناً أنَّ الاستماع كان على طريق الهزء بأن يقولوا : مجنونٌ أو مسحورٌ ،   جاء الاستماع بالباء وإلى ،   ليعلم أنَّ الاستماع ليس المراد به تفهُّم المسموعِ دون هذا المقصد  "  فعلى هذا أيضا تتعلَّق الباء ب  "  يَسْتمِعُونَ  "  .  <br />
قوله تعالى : { إِذْ يَسْتَمِعُونَ } فيه وجهان :<br />
أحدهما : أنه معمولٌ ل "  أعْلَمُ  "  .   قال الزمخشري :  "  إذ يستمعون نصب ب "  أعْلَمُ  "  أي : أعلم وقت استماعهم بما به يستمعون ،   وبما يتناجون ؛ إذ هم ذوو نجوى  "  .  <br />
والثاني : أنه منصوبٌ ب  "  يَسْتمِعُونَ  "  الأولى .  <br />
قال ابن عطيَّة - رحمه الله - :  "  والعامل في  "  إذ  "  الأولى ،   وفي المعطوف  "  يَسْتمِعُونَ  "  الأولى  "  .  <br />
وقال الحوفيُّ : و  "  إذ  "  الأولى تتعلق ب  "  يَسْتمِعُونَ  "  وكذا  "  وإذْ هُمْ نجْوَى  "  لأن المعنى : نحن أعلم بالذي يستمعون إليك ،   وإلى قراءتك وكلامك ،   إنما يستمعون لسقطك ،   وتتبُّع عيبك ،   والتماسِ ما يطعنون به عليك ،   يعني في زعمهم ؛ ولهذا ذكر تعديتهُ بالباء و  "  إلى  "  .  <br />
قوله - عز وجل- :  "  نَجْوَى  "  يجوز أن يكون مصدراً ،   فيكون من إطلاق المصدر على العين مبالغة ،   أو على حذف مضاف ،   أي : ذوو نجوى ،   كما قاله الزمخشري ،   ويجوز أن يكون جمع نجيِّ ،   كقتيلٍ وقتلى ،   قاله أبو البقاء .  <br />
قوله تعالى : { إِذْ يَقُولُ } بدل من  "  إذ  "  الأولى في أحد القولين ،   والقول الآخر : أنَّها معمولة ل  "  اذْكُرْ  "  مقدَّراً .  <br />
قوله تعالى :  "  مَسْحُوراً  "  الظاهر أنَّه اسم مفعول من  "  السِّحرِ  "  بكسر السين ،   أي : مخبول العقل ،   أو مخدوعه ،   وقال أبو عبيدة : معناه أنَّ له سَحْراً ،   أي : رئة بمعنى أنه لا يستغني عن الطَّعام والشَّراب ،   فهو بشرٌ مثلكم ،   وتقول العرب للجبان :  "  قد انتفخَ سَحرهُ  "  بفتح السين ،   ولكلِّ من أكل وشرب : مسحورٌ ،   ومسحرٌ ،   فمن الأول قول امرئ القيس : [ الوافر ]<br />
أرَانَا مُوضَعِينَ لأمْرِ غَيْبٍ *** ونُسْحَرُ بالطَّعامِ وبالشَّرابِ{[20460]}<br />
أي : نُغذَّى ونُعَلَّلُ ،   ومن الثاني قول لبيدٍ : [ الطويل ]<br />
فَإنْ تَسْألِينَا فيمَ نَحْنُ فَإنَّنَا *** عَصَافِيرُ مِنْ هذا الأنَامِ المُسحَّرِ{[20461]}<br />
وردَّ الناس على أبي عبيدة قوله ؛ لبعده لفظاً ومعنًى .   قال ابن قتيبة :  "  لا أدري ما الذي حمل أبا عبيدة على هذا التَّفسير المستكرهِ مع ما فسَّره السَّلف بالوجوهِ الواضحة  "  .  <br />
قال شهاب الدين : وأيضاً فإن  "  السَّحْر  "  الذي هو الرِّئة لم يضرب له فيه مثلٌ ؛ بخلاف  "  السِّحْر  "  فإنهم ضربوا له فيه المثل ،   فما بعد الآية من قوله { انظر كَيْفَ ضَرَبُواْ لَكَ الأمثال } [ الإسراء : 48 ] لا يناسب إلا  "  السِّحْر  "  بالكسرِ .  <br />
فصل في معنى قوله : { نَّحْنُ أَعْلَمُ بِمَا يَسْتَمِعُونَ بِهِ } .  <br />
قال المفسرون : معنى الآية { نَّحْنُ أَعْلَمُ بِمَا يَسْتَمِعُونَ بِهِ } أي يطلبون سماعه ،   { إِذْ يَسْتَمِعُونَ إِلَيْكَ } وأنت تقرأ القرآن ،   { وَإِذْ هُمْ نجوى } يتناجون في أمرك ،   فبعضهم يقول : هذا مجنونٌ ،   وبعضهم يقول : شاعرٌ { إِذْ يَقُولُ الظالمون } يعني الوليد بن المغيرة وأصحابه : { إِن تَتَّبِعُونَ إِلاَّ رَجُلاً مَّسْحُوراً } مطبوباً .  <br />
وقال مجاهد - رحمه الله- : مخدوعاً{[20462]} ؛ لأنَّ السِّحر حيلة وخديعة ،   وذلك لأنَّ المشركين كانوا يقولون : إنَّ محمداً صلى الله عليه وسلم يتعلَّم من بعض النَّاس هذه الكلمات ،   وأولئك النَّاس يخدعونه بهذه الكلمات ،   فلذلك قالوا :  "  مَسْحُوراً  "  أي : مخدوعاً .  <br />
وأيضاً : كانوا يقولون : إنَّ الشيطان يتخيَّل له ،   فيظنُّ أنه ملكٌ ،   فقالوا : إنه مخدوع من قبل الشَّيطان .  <br />
وقيل : مصروفاً عن الحقِّ{[20463]} ،   يقال : ما سحرك عن كذا ،   أي : ما صرفك ،   وقيل : المسحور هو الشَّيء المفسود ،   يقال : طعام مسحور ،   إذا فسد ،   وأرض مسحورة ،   إذا أصابها من المطر أكثر ممَّا ينبغي فأفسدها .  <br />
فإن قيل : إنَّهم لم يتبعوا رسول الله صلى الله عليه وسلم فكيف يصحُّ أن يقولوا : { إِن تَتَّبِعُونَ إِلاَّ رَجُلاً مَّسْحُوراً } .  <br />
فالجواب أنَّ معناه : إن اتَّبعْتُموهُ ،   فقد اتَّبعْتُمْ رجلاً مسحوراً .
