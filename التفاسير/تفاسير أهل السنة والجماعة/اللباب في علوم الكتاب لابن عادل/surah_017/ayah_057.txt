قوله تعالى : { أولئك الذين يَدْعُونَ } :  "  أولئك  "  مبتدأ ،   وفي خبره وجهان :<br />
أظهرهما : أنّه الجملة من  "  يَبْتَغُونَ  "  ويكون الموصولُ نعتاً ،   أو بياناً أو بدلاً ،   والمراد باسم الإشارة الأنبياء أو الملائكة الذين عبدوا من دون الله ،   والمراد بالواو العبَّاد لهم ،   ويكون العائدُ على  "  الَّذينَ  "  محذوفاً ،   والمعنى : أولئك الأنبياء الذين يدعونهم المشركون ،   لكشف ضرِّهم - أو يدعونهم آلهة ،   فمفعولها أو مفعولاها محذوفان - يبتغون .  <br />
ويجوز أن يكون المراد بالواو ما أريد بأولئك ،   أي : أولئك الأنبياء الذين يدعون ربَّهم أو النَّاس إلى الهدى يبتغون ،   فمفعول  "  يَدْعُونَ  "  محذوف .  <br />
والثاني : أن الخبر نفسُ الموصول ،   و  "  يَبْتَغُون  "  على هذا حالٌ من فاعل  "  يَدْعُون  "  أو بدلٌ منه .   وقرأ{[20493]} العامة  "  يَدعُونَ  "  بالغيب ،   وقد تقدَّم الخلاف في الواو ؛ هل تعود على الأنبياء أو على عابديهم ،   وزيد بن عليٍّ بالغيبة أيضاً ،   إلا أنه بناه للمفعول ،   وقتادة ،   وابن مسعود بتاء الخطاب ،   وهاتان القراءتان تقوِّيان أنَّ الواو للمشركين ،   لا للأنبياء في قراءة العامة .  <br />
فصل<br />
إذا أعدنا  "  يَدْعُونَ  "  للعابدين ،   و  "  يَبْتَغُونَ  "  للمعبودين ،   فالمعنى : أولئك المعبودون يبتغون إلى ربِّهم الوسيلة ؛ لأنَّ الملائكة يرجعون إلى الله في طلب المنافع ،   ودفع المضارِّ ،   يرجون رحمته ،   ويخافون عذابه ،   وإذا كانوا كذلك ،   كانوا عاجزين محتاجين ،   والله - تعالى أغنى الأغنياء ،   فكان الاشتغال [ بعبادته ]{[20494]} أولى .  <br />
فإن قيل : لا نسلِّم أنَّ الملائكة محتاجون إلى رحمة الله تعالى ،   وخائفون من عذابه .  <br />
فالجواب : أنَّ الملائكة : إمَّا أن يقال : إنَّها واجبة الوجود لذواتها ،   أو يقال : إنَّها ممكنة الوجود لذواتها ،   والأول باطلٌ ؛ لأن جميع الكفَّار كانوا معترفين بأن الملائكة عبادُ الله ،   ومحتاجون إليه .  <br />
وأما الثاني : فهو يوجب القول بأنَّ الملائكة محتاجون في ذواتها ،   وفي كمالاتها إلى الله تعالى ،   فكان الاشتغالُ بعبادة الله تعالى أولى من الاشتغال بعبادةِ الملائكة .  <br />
وإن أعدنا  "  يَدْعُونَ  "  إلى الأنبياء - عليهم السلام - المذكورين في قوله تعالى : { وَلَقَدْ فَضَّلْنَا بَعْضَ النبيين على بَعْضٍ } فالمعنى هو أنَّ الذي عظمت منزلتهم - وهم الأنبياء - لا يعبدون إلا الله تعالى ،   ولا يبتغون الوسيلة إلاَّ إليه ،   فأنتم بالاقتداءِ بهم أحق ،   فلا تعبدوا غير الله - عزَّ وجلَّ - والمراد بالوسيلةِ : الدَّرجة العليا .  <br />
وقيل : كل ما يتقرَّب إلى الله تعالى .  <br />
واحتجُّوا على صحَّة هذا القول بأنَّ الملائكة لا يعصون الله ما أمرهم ،   فلا يخافون عذابه ،   فثبت أنَّ هذا غير لائقٍ بالملائكةِ ،   وإنما هو لائقٌ بالأنبياء - صلوات الله عليهم- .  <br />
وأجيب بأنَّ الملائكة يخافون من عذاب الله ،   لو أقدموا على الذنب ،   قال تعالى : { وَمَن يَقُلْ مِنْهُمْ إني إله مِّن دُونِهِ فذلك نَجْزِيهِ جَهَنَّمَ } [ الأنبياء : 29 ] وقال تعالى : { يَخَافُونَ رَبَّهُمْ مِّن فَوْقِهِمْ } [ النحل : 50 ] ثم قال عزَّ وجلَّ : { إِنَّ عَذَابَ رَبِّكَ كَانَ مَحْذُوراً } أي من حقِّه أن يحذر ،   فإن لم يحذره بعض الناس لجهله ،   فإنَّه لا يخرج عن كونه يجب الحذر عنه .  <br />
قوله تعالى : { أَيُّهُمْ أَقْرَبُ } في  "  أيُّ  "  هذه وجهان :<br />
أحدهما : أنها استفهامية .  <br />
والثاني : أنها موصولة بمعنى  "  الَّذي  "  وإنما كثر كلام المعربين فيها من حيث التقدير ،   فقال الزمخشريُّ :  "  وأيُّهم بدلٌ من واو  "  يَبْتَغُونَ  "  و  "  أيُّ  "  موصولة ،   أي : يبتغي من هو أقربُ منهم وأزلفُ ،   أو ضمِّن  "  يَبْتغُونَ الوسيلة  "  معنى يحرصون ،   فكأنه قيل : يحرصون أيهم يكون أقرب  "  .   فجعلها في الوجه الأول موصولة ،   وصلتها جملة من مبتدأ وخبر ،   حذف المبتدأ ،   وهو عائدها ،   و  "  أقْرَبُ  "  خبرٌ .   واحتملت  "  أيُّ  "  حينئذٍ أن تكون مبنية ،   وهو الأكثر فيها ،   وأن تكون معربة ،   وسيأتي موضعه في مريم : [ 69 ] إن شاء الله تعالى وفي الثاني جعلها استفهامية ؛ بدليل أنه ضمَّن الابتغاء معنى شيء تعلق ،   وهو يحرصون ،   فيكون  "  أيُّهُمْ  "  مبتدأ و  "  أقربُ  "  خبره ،   والجملة في محلِّ نصب على إسقاط الخافض ؛ لأنَّ  "  تحْرِصُ  "  يتعدَّى ب  "  على  "  قال تعالى : { إِن تَحْرِصْ على هُدَاهُمْ } [ النحل : 37 ] ،   { أَحْرَصَ الناس على حَيَاةٍ } [ البقرة : 96 ] .  <br />
وقال أبو البقاء{[20495]} :  "  أيُّهُمْ  "  مبتدأ ،   و  "  أقْرَبُ  "  خبره ،   وهو استفهامُ في موضع نصبٍ ب  "  يَدْعُونَ  "  ،   ويجوز أن يكون  "  أيُّهُمْ  "  بمعنى الذي ،   وهو بدلٌ من الضمير في  "  يَدعُونَ  "  .  <br />
قال أبو حيان :  "  علَّق  "  يَدْعُونَ  "  وهو ليس فعلاً قلبيًّا ،   وفي الثاني فصل بين الصلة ومعمولها بالجملة الحاليَّة ،   ولا يضرُّ ذلك ،   لأنَّها معمولة للصِّلة  "  .   قال شهاب الدين : أمَّا كون  "  يَدْعُونَ  "  لا يعلق ،   هو مذهب الجمهور ،   وقال يونس : يجوز تعليق الأفعال مطلقاً ،   القلبية وغيرها ،   وأمَّا قوله  "  فصل بالجملة الحالية  "  يعني بها  "  يَبْتَغُونَ  "  فصل بها بين  "  يَدْعُونَ  "  الذي هو صلة  "  الَّذينَ  "  وبين معموله ،   وهو  "  أيُّهم أقْرَبُ  "  لأنه معلَّقٌ عنه ،   كما عرفته ،   إلا أنَّ الشيخ لم يتقدَّم في كلامه إعرابُ  "  يَبْتغُونَ  "  حالاً ،   بل لم يعربها إلاَّ خبراً للموصول ،   وهذا قريبٌ .  <br />
وجعل أبو البقاء أيًّا الموصولة بدلاً من واو  "  يَدْعُونَ  "  ،   ولم أرَ أحداً وافقه على ذلك ،   بل كلُّهم يجعلونها من واو  "  يَبْتَغُونَ  "  وهو الظاهر .  <br />
وقال الحوفي- رحمه الله- :  "  أيُّهم أقربُ  "  ابتداء وخبر ،   والمعنى : ينظرون أيُّهم أقرب ،   فيتوسَّلون به ،   ويجوز أن يكون  "  أيهم أقرب  "  بدلاً من واو  "  يَبْتَغُون  "  .  <br />
قال شهاب الدين : فقد أضمر فعلاً معلقاً ،   وهو ينظرون فإن كان من نظر البصرِ ،   تعدَّى ب  "  إلى  "  وإن كان من نظر الفكر ،   تعدَّى ب  "  في  "  فعلى التقديرين : الجملة الاستفهامية في موضع نصبٍ بإسقاطِ الخافض ،   وهذا إضمارُ ما لا حاجة إليه .  <br />
وقال ابن عطية :  "  وأيُّهُمْ ابتداء ،   و  "  أقْرَبُ  "  خبره ،   والتقدير : نظرهم ووكدهم أيهم أقرب ،   ومنه قول عمر بن الخطاب - رضي الله عنه- :  "  فبات النَّاس يدُوكُونَ أيُّهمْ يُعطَاهَا  "  ،   أي : يتبارون في القرب  "  .   قال أبو حيان :  "  فَجَعلَ المحذوف  "  نظرُهمْ ووكْدهُمْ  "  وهذا مبتدأ ،   فإن جعلت  "  أيُّهمْ أقربُ  "  في موضع نصب ب  "  نَظرُهُمْ  "  بقي المبتدأ بلا خبر ،   فيحتاج إلى إضمار خبر ،   وإن جعلت  "  أيُّهم أقربُ  "  الخبر ،   لم يصحَّ ؛ لأنَّ نظرهم ليس هو  "  أيُّهم أقربُ  "  وإن جعلت التقدير :  "  نَظرهُمْ في أيهم أقربُ  "  أي : كائنٌ أو حاصلٌ ،   لم يصحَّ ذلك ؛ لأنَّ كائناً وحاصلاً ليس ممَّا يعلَّق  "  .  <br />
فقد تحصَّل في الآية الكريمة ستَّة أوجه :<br />
أربعة حال جعل  "  أيّ  "  استفهاماً :<br />
الأول : أنها معلِّقة للوسيلة ،   كما قرَّره الزمخشريُّ .  <br />
الثاني : أنها معلّقة ل  "  يَدعُونَ  "  كما قاله أبو البقاء .  <br />
الثالث : أنها معلقة ل  "  يَنْظُرونَ  "  مقدراً ،   كما قاله الحوفيُّ .  <br />
الرابع : أنها معلقة ل  "  نَظرُهمْ  "  كما قدَّره ابن عطيَّة .  <br />
واثنان حال جعلها موصولة :<br />
الأول : البدل من واو  "  يَدعُونَ  "  كما قاله أبو البقاء .  <br />
الثاني : أنها بدلٌ من واو  "  يَبْتغُونَ  "  كما قاله الجمهور .
