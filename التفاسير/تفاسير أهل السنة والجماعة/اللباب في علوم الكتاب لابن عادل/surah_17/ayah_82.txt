قوله تعالى : { وَنُنَزِّلُ مِنَ القرآن } : في  "  مِنْ  "  هذه ثلاثة أوجه :<br />
أحدها : أنها لبيان الجنس ،   قاله الزمخشري ،   وابن عطيَّة ،   وأبو البقاء ،   وردَّ عليهم أبو حيان : بأنَّ التي للبيان ،   لا بد وأن يتقدَّمها ما تبينه ،   لا أن تتقدم هي عليه ،   وهنا قد وجد تقدُّمها عليه .  <br />
الثاني : أنها للتبعيض ،   وأنكره الحوفي ؛ قال :  "  لأنه يلزم ألاَّ يكون بعضه شفاء  "  وأجيب عنه : بأنَّ إنزاله إنَّما هو مبعضٌ ،   وهذا الجواب ليس بظاهرٍ ،   وأجاب أبو البقاء بأنَّ منه ما يشفي من المرض .   وهذا يؤيده الدَّليلُ المتقدِّم ،   وأجاز الكسائيُّ :  "  ورَحْمةً  "  بالنصب عطفاً على ما تظاهر وهذا قد وجد بدليل رقية بعض الصحابة سيِّد الحيِّ الذي لدغ ،   بالفاتحة ؛ فشفي .  <br />
الثالث : أنها لابتداءِ الغاية ،   وهو واضحٌ .  <br />
والجمهور على رفع  "  شفاء ورحمةٌ  "  خبرين ل  "  هُوَ  "  ،   والجملة صلة ل  "  مَا  "  وزيد بن عليٍّ{[20658]} بنصبهما ،   وخرِّجت قراءته على نصبهما على الحال ،   والخبر حينئذ  "  لِلمُؤمنينَ  "  وقدِّمت الحال على عاملها المعنوي ،   كقوله تعالى : { والسماوات مَطْوِيَّاتٍ } [ الزمر : 67 ] في قراءة من نصب  "  مَطْويَّاتٍ  "  ،   وقول النابغة :<br />
رَهْطُ ابنِ كُوزٍ مُحقِبِي أدْرَاعهُمْ *** فِيهمْ ورَهْطُ رَبيعَة بْنِ حُذارِ{[20659]}<br />
وقيل : منصوبان بإضمار فعلٍ ،   وهذا عند من يمنع تقديمها على عاملها المعنوي ،   وقال أبو البقاء{[20660]} : وأجاز الكسائي :  "  ورحْمَةً  "  بالنصب عطفاً على  "  مَا  "  فظاهر هذا أن الكسائيَّ [ بقَّى ]  "  شِفاءٌ  "  على رفعه ،   ونصب  "  رَحْمةً  "  فقط عطفاً على  "  ما  "  الموصولة ؛ كأنه قيل : ونُنزِّل [ من القرآن رحمة ،   وليس في نقله ما يؤذن بأنه تلاها قرآناً ،   وتقدَّم الخلاف في ]  "  ونُنزِّلُ  "  تخفيفاً وتشديداً ،   والعامة على نون العظمة .  <br />
ومجاهد  " {[20661]} ويُنزِلُ  "  بياء الغيبة ،   أي : الله .  <br />
فصل في المراد ب  "  مِنْ  "  في الآية<br />
قال المفسِّرون : إنَّ  "  من  "  هنا للجنسِ ؛ كقوله تعالى : { فاجتنبوا الرجس مِنَ الأوثان } [ الحج : 30 ] .  <br />
أي : ونُنزِّل من هذا الجنس الذي هو قرآنٌ ما هو شفاءٌ ،   فجميع القرآن شفاء للمؤمنين ،   أي : بيانٌ من الضلالة والجهالة يتبيَّن به المختلف ،   ويتَّضح به المشكل ،   ويستشفى به من الشُّبهة ،   ويهتدى به من الحيرة ،   وهو شفاء للقلوب بزوال الجهل عنها .  <br />
واعلم أنَّ القرآن شفاء من الأمراض الرُّوحانيَّة ،   والأمراض الجسمانية .  <br />
أما كونه شفاء من الأمراض الروحانية ؛ لأنَّ المرض الروحانيَّ قسمان :<br />
أحدهما : الاعتقادات الباطلة ،   وأشدُّها فساداً الاعتقادات الفاسدة في الإلهيَّات ،   والنبوَّات ،   والمعاد ،   والقضاءِ ،   والقدر ؛ والقرآن كلُّه مشتملٌ على دلائل الحقِّ في هذه المطالب .  <br />
والثاني : الأخلاق المذمومة ؛ والقرآن مشتمل على تفاصيلها ،   وتعريف ما فيها من المفاسد ،   والإرشاد إلى الأخلاق الفاضلة ،   وكان القرآن شفاء من الأمراض الروحانيَّة .  <br />
وأما كونه شفاء من الأمراض الجسمانية ؛ فلأنَّ التبرُّك بقراءته يدفع كثيراً من الأمراض ؛ ويؤيده ما روي عن النبي صلى الله عليه وسلم أنه قال :  "  مَنْ لَمْ يَستشْفِ بالقرآنِ ،   فلا شَفاهُ الله تعالى  " {[20662]} .  <br />
وما ورد في حديث الرقية بالفاتحة .  <br />
ثم قال : { ولا يَزيدُ الظَّالِمِينَ إلاَّ خَسَاراً }{[20663]}  المراد بالظالمين ها هنا : المشركون ؛ لأنَّ سماع القرآن يزيدهم غضباً ،   وغيظاً ،   وحقداً ،   وكلَّما نزلت آيةٌ يتجدَّد تكذيبٌ ؛ فتزداد خسارتهم .  <br />
قال مجاهد وقتادة : لم يجالس هذا القرآن أحدٌ إلا قام عنه بزيادةٍ أو نقصان : قضاء الله الذي قضاه شفاءٌ ورحمةٌ للمؤمنين ،   ولا يزيد الظَّالمين إلا خساراً{[20664]} .
