قوله تعالى : { وَكَيْفَ يُحَكِّمُونَكَ } كقوله : { كَيْفَ تُحْيِ الْمَوْتَى } [ البقرة : 260 ] وقد تقدم .  <br />
[ قولُه : ]{[11815]}  "  وعنْدَهُمُ التَّوْرَاةُ  "   "  الواوُ  "  للحالِ ،   و  "  التوراة  "  يجوزُ أن تكون مُبْتدأ والظرفُ خَبَرُه ،   ويجوزُ أنْ يكونَ الظرفُ حالاً ،   و  "  التوراةُ  "  فاعلٌ بِهِ لاعتمادِهِ على ذِي الحالِ .  <br />
والجملةُ الاسميَّةُ أو الفعليَّةُ في محلِّ نَصْبٍ على الحالِ .  <br />
وقوله : { فِيهَا حُكْمُ اللَّهِ } ،    "  فيها  "  خَبَرٌ مقدمٌ ،   و  "  حُكْمُ  "  مبتدأ ،   أو فاعِلٌ كما تقدم في  "  التوراةِ  "  ،   والجملةُ حالٌ من  "  التوراةِ  "  ،   أو الجار وحدَهُ ،   و  "  حُكْمُ  "  مصدرٌ مضافٌ لفاعلِهِ .  <br />
وأجاز الزمخشريُّ{[11816]} : ألاَّ يكون لها مَحَل من الإعراب ،   بل هي مُبَيِّنةٌ ؛ لأنَّ عندَهُم ما يُغنيهم عن التحكيمِ ،   كما تقولُ :  "  عندك زيدٌ يَنْصَحُك ،   ويُشيرُ عليك بالصَّوابِ ،   فما تصنعُ بِغَيْره ؟  "  .  <br />
وقولُه تعالى :  "  ثُمَّ يَتولّونَ  "  معطوفٌ على  "  يُحَكِّمونَكَ  "  ،   فهو في سياقِ التعجُّب المفهُومِ مِن  "  كَيْفَ  "  وذلك إشارةٌ إلى حُكْمِ الله الذي في التوراة ،   ويجوزُ أن يَعُودَ إلى التحكيم والله أعلم .  <br />
فصل<br />
هذا تعجُّبٌ من اللَّهِ لنبيه [ عليه الصلاةُ والسلامُ ] مِنْ تحكيمِ اليهُودِ إياهُ{[11817]} بعد علمهم بما في التوراةِ مِنْ حدِّ الزَّانِي ،   ثُم تركِهِمْ قبولَ ذلك الحُكْم فيتعدلُونَ عما يعتَقدُونَه حُكْماً [ حقاً ]{[11818]} إلى ما يعتقدُونه باطِلاً طلباً للرخْصةِ فظهر جهلهم وعنادهُم من وُجُوه :<br />
أحدُها : عُدُولهُمْ عن حُكْمِ كتابِهِم .  <br />
والثاني : رجوعُهم إلى حكمِ مَنْ كانوا يعتقدون أنه مُبطلٌ .  <br />
والثالث : إعراضُهم عن حكمه بعد أن حكَّموه ،   فبينَ الله تعالى حال جهلهم وعنادهم ؛ لَئِلا يَغْتَرَّ مُغْترٌّ أنهم أهلُ كتابِ الله ومن المحافظِينَ على أمْر الله .  <br />
ثُمَّ قال تعالى : { وَمَآ أُوْلَئِكَ بِالْمُؤْمِنِينَ } أيْ بالتوراةِ وإنْ كانوا يُظْهِرُون الإيمانَ بها ،   وقِيلَ : هذا إخبارٌ بأنهم لا يُؤمنونَ أبَداً ،   وهو خَبَرٌ عن المستأنفِ لا عَنِ الماضِي .  <br />
وقِيلَ : إنَّهمَ وَإنْ طلبُوا الحكمَ مِنْك فما هُم بمؤمِنينَ بِكَ ،   ولا بالمعتقدينَ في صِحّةِ حُكْمكَ .
