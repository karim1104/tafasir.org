وقوله عز وجلا : { حُرِّمَتْ عَلَيْكُمُ الْمَيْتَةُ وَالْدَّمُ وَلَحْمُ الْخِنْزِيرِ } الآية .  <br />
وهذا هو المُسْتَثْنَى مِنَ الإبَاحَةِ في قوله : { إِلاَّ مَا يُتْلَى عَلَيْكُمْ } [ المائدة : 1 ] وهو أحَدَ عَشَرَ نَوْعاً ،   وقد تقدم إعرابُ : { حُرِّمَتْ عَلَيْكُمُ الْمَيْتَةُ } وأصْلُهَا .  <br />
واعلمْ أنَّ تحريمَ الميتةِ موافقٌ للمعقولِ ؛ لأنَّ الدمَ جوهرٌ لَطِيفٌ جِداً فإذا ماتَ الحيوانُ حَتْفَ أنْفِهِ احْتَبَسَ الدمُ في عُرُوقِهِ وتعفَّن وفَسَدَ ؛ وحصل من أكْلِهِ مَضَرَّةٌ .  <br />
وأمَّا الدمُ فقال الزَّمَخْشَرِيُّ : كانوا يَمْلَؤُونَ المِعَى مِنَ الدَّمِ وَيَشْوونَهُ{[10719]} ويُطْعمُونَه للضيفِ ،   فحرّم الله ذلك عليهم .  <br />
وأما الخنزيرُ فقال العلماءُ{[10720]} : الغذاءُ يصيرُ جُزْءاً من جوهرِ المتغذِّي ولا بُدَّ أنْ يَحْصُلَ لِلْمُتغذِّي أخلاقٌ{[10721]} وصفاتٌ مِنْ جِنْسِ ما كان حَاصِلاً في الغذاء ،   والخنزيرُ مَطْبوعٌ على حِرْصٍ عَظيمٍ ورغبة شديدةٍ في المُشْتهياتِ{[10722]} ،   فحرَّمَ اللَّهُ أكلهُ على الإنسانِ لِئلاَّ يَتَكيّفَ بِتلْكَ الكيفيّةِ ؛ وذلك لأنَّ الفرنج لما وَاظَبُوا على أكْلِ لَحْمِ الخنزيرِ أوْرَثَهُمُ الحرص العظيم والرغبة الشديدة في المُشْتهياتِ ،   وأورثهم عدم الغيرةِ ،   [ فإنّ الخِنْزِيرَ يَرَى الذّكرَ من الخنزيرِ يَنْزُو على الأنْثَى التي لَهُ لا يتعرض إليه ؛ لعدم الغَيْرةِ ،   وقد ]{[10723]} تقدمَ الكلام على الخنزيرِ واشتقاقِهِ في سورة البقرة .  <br />
وأمَّا الشاةُ فإنها حيوانٌ في غَايَةِ السَّلاَمَةِ ،   وكأنَّها ذَاتُ عَارِيَةٍ عَنْ جَمِيعِ الأخْلاقِ ،   فلذلك لا يَحْصُلُ للإنسانِ بِسَبَبِ أكْلِها كَيْفِيَّةٌ أجْنَبِيَّةٌ عن أحوالِ الإنسانِ .  <br />
وأمَّا ما { أُهِلَّ لِغَيْرِ اللَّهِ بِهِ } ،   والإهلالُ رَفْع الصوتِ ،   ومنه يقالُ : فلانٌ أهَلَّ بالحجِّ إذَا لَبَّى ،   ومِنْهُ استهلالُ الصَّبِي وهو صراخُهُ إذا وُلِد ،   وكانوا يقولُونَ عند الذَّبْح باسم اللاَّتِ والعُزَّى ،   فحرم الله ذلك ،   	وقدِّمَ هنا لَفْظُ الجلالةِ في قوله : { وَمَا أُهِلَّ لِغَيْرِ اللَّهِ بِهِ } وأخِّرتْ{[10724]} في البقرة [ آية 173 ]  ؛ لأنَّها هناك فاصلةٌ ،   أو تُشْبِهُ الفاصِلَةَ بخلاف هاهنا ،   فإنَّ بَعْدَها مَعْطُوفاتٍ .  <br />
والمنخنِقَةُ وهي التي تموتُ خَنْقاً ،   وهو حَبْسُ النَّفَسِ سَواءٌ فعل بها ذلك [ آدمِيٌّ أو ] اتفق لها ذلك في حَبْلٍ أوْ بين عُودَيْنِ أو نحوه .  <br />
وذكر قتادةُ أنَّ أهل الجاهليةِ كانوا يَخْنُقُونَ الشاةَ وغيرَها ،   فإذا ماتَتْ أكَلُوها ،   وذكر نحوه ابنُ عباسٍ{[10725]} .  <br />
والمَوْقُوذَةُ : وهي التي وُقذَتْ أي : ضُربَتْ حتى ماتت من وَقَذَه أيْ ضَرَبَهُ حتى اسْتَرْخَى ،   ومنه وَقَذَهُ النُّعَاسُ أيْ : غلبه ووقذه الحُلُم : أيْ : سكنه وكأنَّ المادة دالَّةٌ على سُكُونٍ واسْتِرخاءٍ ،   ويدل في الموقوذةِ ما رُمِيَ بالبندقِ فماتَ ،   وهي - أيضاً - في مَعْنى الميتةِ وفي معنى المنخنقةِ ،   فإنها ماتَتْ ولم يَسِلْ دَمُهَا .  <br />
والمتردِّيَةُ : من تَرَدَّى ،   أي : سَقَطَ مِنْ عُلُوٍّ فَهَلَكَ ،   ويُقالُ : ما يُدْرَى أين{[10726]}  "  ردَى  "  ؟ أيْ : ذَهَبَ ،   ورَدَى وتَرَدَّى بمعنى هَلكَ .  <br />
قال تعالى : { وَمَا يُغْنِي عَنْهُ مَالُهُ إِذَا تَرَدَّى } [ الليل : 11 ] ،   والمتردِّي أنْوَاعٌ : فالمترديةُ هي التي تسقطُ مِنْ جَبَلٍ أو مَوْضِعٍ مُشْرِفٍ أو في بِئْر فتموت ،   فهذه مَيْتَةٌ ،   لأنَّها ماتَتْ وما سَال منها الدمُ ،   ويدخل فيه إذا أصابه سَهْمٌ وهو في الجبل فَسَقَطَ على الأرض ؛ فإنه يُحَرَّمُ أكْلُه ؛ لأنَّه لا يُعْلَمُ ماتَ بالتَّرَدِّي أوْ بالسَّهْم .  <br />
ودخلتِ الهاءُ في هذه الكلمةِ{[10727]} ؛ لأنّ المنخنقةَ هي الشاةُ المنخنقةُ ،   كأنه قِيلَ : حُرِّمَتْ عليكم الشاةُ المنخنقةُ والموْقوذَةُ والمترديةُ ،   وخَصَّ الشاة ؛ لأنَّها من أعَمِّ [ ما يَأكُلُ ]{[10728]} الناسُ ،   والكلامُ يُخَرَّجُ{[10729]} على الأعَمِّ الأغْلَبِ{[10730]} ،   ويكُونُ المرادُ هو الكُلَّ .  <br />
و  "  النَّطِيحَةُ  "   "  فَعيلَة  "  بمعنى  "  مَفْعُولة  "  ،   وكان مِنْ حَقِّها ألاَّ تَدْخُلَها تاءُ التأنيثِ كقَتِيل وجَريح ،   إلاَّ أنَّها جَرَتْ مَجْرَى الأسماءِ ،   أوْ لأنَّها لم يُذْكَرْ مَوصُوفُها ؛ لأنك إنْ لَمْ تُدْخِل الهاءَ لَمْ يُعرَفْ أرَجُلٌ هُوَ أمِ امْرَأةٌ ،   ومثلُهُ : الذَّبِيحَةُ والنَّسيكَةُ .   كذا قاله أبُو البَقاءِ ،   وفيه نظر ؛ لأنَّهُم [ إنَّما ]{[10731]} يلحقون التاءَ إذا لم يذكر الموصوفُ{[10732]} لأجل اللَّبْسِ ،   نَحْوَ : مررت بِقَتيلةِ بَنِي فُلانٍ ،   لئلا يلتبس بالمؤنثِ وهنا اللَّبْسُ مُنْتفٍ ،   وأيضاً فحكم الذكرِ والأنْثَى في هذا سَوَاءٌ .  <br />
والنَّطِيحَةُ هي التي تَنْطَحُها الأخْرَى فتموت ،   وهذه - أيضاً - لأنها ماتت من غير سَيَلانِ الدَّم .  <br />
قوله تعالى : { وَمَا أَكَلَ السَّبُعُ إِلاَّ مَا ذَكَّيْتُمْ }  "  مَا  "  الأولى بمعنى  "  الَّذِي  "  ،   وعائِدُهُ محذُوفٌ ،   أيْ : وما أكلَهُ السّبُعُ ،   ومحلُّ هذا الموصولِ الرفعُ عَطْفاً على مَا لَمْ يُسَمَّ فَاعِلُه ،   وهذا غيرُ ماشٍ على ظاهره ؛ لأنَّ ما أكله السبعُ وفرغ منه لا يُذَكَّى ،   فلا بُدَّ من حذف .   ولذلك قال الزَّمَخْشَرِيُّ : وما أكَلَ بَعْضَهُ السبعُ .  <br />
وقرأ الحسنُ{[10733]} والفياض وأبو حَيْوة :  "  السَّبْعُ  "  بسكون الباءِ ،   وهو تسكين المضموم ،   ونقل فتح السين والباء معاً .  <br />
والسَّبُعُ : كُلُّ ذِي نَابٍ ومخْلب كالأسدِ والنَّمرِ ،   ويطلقُ على ذِي المِخْلب من الطيور قال : [ الخفيف ]<br />
وسبَاعُ الطَّيْر تَغْدُو بِطَاناً *** [ تَتَخَطَّاهُمُ فَمَا تَسْتقلُّ{[10734]} ]{[10735]}<br />
فصل<br />
معنى الكلام ما يُريدُ ما بَقِيَ مما أكل السبعُ .   قال قتادةُ : كان أهْلُ الجاهلية إذا جَرَحَ السبعُ شَيْئاً فقتله وأكل بعضَه أكلُوا ما بَقِيَ ،   فحرمه الله{[10736]} .  <br />
قوله تعالى : { إِلاَّ مَا ذَكَّيْتُمْ } فيه قولان :<br />
أحدهما : أنه استثناءٌ متصلٌ ،   والقائلون بهذا اختلَفُوا ،   فقال عَلِيٌّ ،   وابنُ عباسٍ ،   والحسنُ وقتادةُ : هو مُسْتَثْنَى من قوله :  "  والمُنْخَنِقَة  "  إلى قوله : { وَمَا أَكَلَ السَّبُعُ } وعلى هذا إن أدْرَكْتَ ذكَاتَهُ{[10737]} بأنْ وَجَدْتَ عَيْناً تَطْرِفُ ،   أوْ ذَنَباً يَتَحرَّكُ ،   أوْ رجلاً تَرْكُضُ فاذْبَحْ فإنَّهُ حلالٌ ،   فإنَّ هذه الحالَ تدلُ على بقاءِ الحياةِ فيه بتمامها .  <br />
وقال أبُو البَقاءِ : والاستثناءُ راجعٌ على المتردِّيةِ ،   والنَّطِيحَةِ وأكيلة السبُع ،   وليس إخراجُه المُنْخَنِقَة [ منه بجَيِّدٍ ]{[10738]} .  <br />
ومنهم من قال : هو مستثنى مما أكل السبعُ خاصَّة .  <br />
والقولُ الثاني : أنَّه منقطعٌ ،   أيْ : ولكنْ{[10739]} ما ذَكيْتم مِنْ غيرها فحلالٌ ،   أو فكُلُوه ،   كأنَّ هذا القائِلَ رأى أنَّها وَصَلَتْ بهذه الأسبابِ إلى الموتِ ،   أو إلى حالةٍ قَرِيبةٍ فلم تُفِدْ تَزْكِيتَها عِنْدَه شيئاً .  <br />
والتَّذْكِيَةُ : الذَّبْحُ ،   وذَكَتِ النَّارُ : ارتفعتْ ،   وأصلُ الذَّكاةِ تمامُ الشيء ومنه الذَّكاءُ في الفهمِ ،   وهو تمامُهُ [ والذكاء ]{[10740]} في السِّن ،   وهو النهايةُ في الشباب ،   ذَكَى الرجُل أيْ : أسَنَّ ،   قال : [ الوافر ]<br />
عَلَى أعْرَاقِهِ تَجْرِي الْمَذَاكِي *** وَلَيْسَ عَلَى تَقَلُّبِهِ{[10741]} وجُهْدِهْ{[10742]}<br />
وقيل : الاستثناءُ من التحريم لا مِنَ المحرماتِ ،   يَعْنِي ،   حَرَّمَ عَلَيْكُمْ ما مَضَى إلاَّ مَا ذَكَّيْتُم فإنه لكم حلالٌ ،   فيكون الاستثناءُ منقطعاً - أيضاً - .  <br />
وإذا قيل{[10743]} : أصلُ التذكيةِ الإتمامُ ،   فالمرادُ ههنا إتمامُ فَرْي الأوَداجِ وإنْهارِ الدَّم .  <br />
قال - عليه الصلاة والسلام - :  "  مَا أنْهَرَ الدَّمَ ،   وَذُكِرَ{[10744]} اسْمُ اللَّه عَلَيْهِ فَكُلْ لَيْسَ السِّنَّ وَالظُّفُرَ  " {[10745]} .  <br />
قال القُرْطُبِيُّ{[10746]} : جمهورُ العلماءِ على أنَّ كلَّ ما أفرى الأوْدَاجَ فأنْهَر الدمَ فهو مِنْ آلاتِ الذَّكاةِ ما خلَى السِّنَّ والظُّفُرَ والعَظْمَ ،   وعلى هذا تَوَاترتِ الأخبارُ .  <br />
وقال به{[10747]} فقهاءُ الأمصارِ ،   والسنُّ والظُّفْرُ المنهيُّ عنهما في التذكية{[10748]} هما غَيْرُ المنزوعَيْنِ ؛ لأنَّ ذلك يَصيرُ خَنْقاً ،   ولذلك قال ابنُ عباسٍ : ذلك الخنقُ .  <br />
فأمّا المنزُوعانِ إذا فَرَيا الأوْدَاجَ فالذكاةُ جائِزَةٌ بِهِمَا{[10749]} عِنْدَهُمْ .  <br />
وكره قومٌ السنَّ والظفرَ والعظمَ على كُلِّ حالٍ مَنْزُوعانِ كانا أو غَيْر منزُوعَيْنِ ،   منهم إبراهيمُ والحسنُ واللّيْثُ بنُ سَعْدٍ ،   وهُوَ مَرْوِيٌّ عَنِ الشَّافِعي .  <br />
وأقلُّ الذَّكاةِ في الحيوانِ المقدُورِ عليه قطعُ الحُلْقُومِ والمري ،   وكمالهُ أن يقطعَ الودجيْنِ معهما ،   ويجوزُ بكل محدَّدٍ يجرح من حَديدٍ أو قصبٍ أو زجاجٍ أو حجرٍ أو غيره إلاَّ السِّنَّ والظفرَ للحديث المتقدم .  <br />
وإنَّما يحلُّ ما ذكيته بعدما جرحه السبعُ فأكل منه شيئاً إذا أدْرَكْتَهُ والحياةُ فيه مستقرةٌ فذبحتَهُ ،   فأمَّا ما يجرحُ السبعُ فيخرجه إلى حالةِ المذبوحِ فهو في حُكمِ الميتَةِ فلا يكونُ حَلاَلاً ،   والمُترديةُ والنَّطيحةُ إذا أدركتهما حيَّةً ،   قَبْلَ أنْ تصيدَ إلى حالة المذبُوحِ فذبحتها تكونُ حلالاً ،   ولو رُمِي صيدٌ في الهواءِ فأصابَهُ فسقَطَ على الأرضِ [ وماتَ كانَ حلالاً ؛ لأنَّ الوقوعَ على الأرضِ ضرورتُهُ ،   فإنْ سقَطَ على شجرٍ أو جَبَلٍ فتردَّى منه ]{[10750]} فمات فلا يَحِلُّ ؛ لأنَّه من المتردية ،   إلاَّ أنْ يكونَ السهمُ ذَبَحَهُ في الهواءِ فيَحِلّ كيفما وقع ؛ لأنَّ الذبحَ قد حصل قبل التردِيَةِ .  <br />
فصل<br />
واختلفُوا [ فيمنْ رَفَعَ ]{[10751]} [ يَدَهُ ]{[10752]} قبل تمامِ الذَّكاةِ ثُم رجع [ على الفور ]{[10753]} وأكْمَلَ الذَّكَاة فقيل : يُجْزِئُهُ ،   وقيل : لا يُجْزِئه .  <br />
فالأولُ أصَحُّ ؛ لأنَّه جَرَحَهُ{[10754]} ثُم ذَكَّاهُ بعدُ وحياته مُسْتَجْمعةٌ فيه .  <br />
قوله : { وَمَا ذُبِحَ عَلَى النُّصُبِ } رَفْعٌ - أيضاً - عطفاً على  "  الميتةِ  "  .  <br />
واختلفُوا في النُّصُبِ ،   فَقِيل : هي حجارةٌ ،   كانوا يَذْبَحُون عليها ،   ف  "  على  "  هنا وَاضِحَةُ .  <br />
وقيل : هي الأصنَامُ ؛ لأنَّها تُنْصَبُ لتُعْبَد ،   فعلى هذا في  "  على  "  وَجْهانِ :<br />
أحدُهما : أنها بمعنى اللام{[10755]} ،   أيْ : وما ذُبِحَ لأجْلِ الأصنامِ ،   كذا ذكره أبُو البقاءِ وفيه نَظَرٌ ،   وهو كونُه قَدَّرَ المتعلَّق شيئاً خاصاً .  <br />
والجمهورُ على  "  النُّصب  "  بضَمتين ،   فقيل : هو جمعُ  "  نِصاب  " {[10756]} .  <br />
وقيل : هو مُفْردٌ ويدل له قَوْلُ الأعْشَى : [ الطويل ]<br />
وَذَا النُّصُبَ المَنْصُوبَ لا تَقْربَنَّهُ *** ولا تَعْبُدِ الشَّيْطَانَ واللَّهَ فاعْبُدَا{[10757]}<br />
وفيه احتمالٌ .  <br />
وقرأ طلحةُ بنُ مُصَرِّفٍ{[10758]} بضم النُّونِ وإسْكَانِ الصَّادِ ،   وهو تخفيفُ القراءة الأولى .  <br />
وقرأ عيسى{[10759]} بنُ عُمَرَ :  "  النَّصَب  "  بِفَتْحتين .  <br />
قال أبُو البقاء : وهو اسمٌ بمعنى : المنصُوبِ ،   كالقَبْضِ والنَّقَصِ ،   بمعنى : المقبُوض والمنقُوصِ .  <br />
والحسنُ{[10760]} النَّصْب بفتح النون وسكُون الصادِ ،   وهو مَصْدَرٌ واقعٌ مَوْقِع المفعولِ به ،   ولا يجوزُ أنْ تكونَ تخفيفاً كقراءة عِيسَى بنِ عُمر ؛ لأنَّ الفتحةَ لا تُخَفَّفُ .  <br />
فصل<br />
 "  النُّصُب  "  يَحْتَمِلُ أنْ يكونَ جَمْعاً وأنْ يكونَ واحداً ،   فإنْ كان جَمْعاً ففي واحدِه وُجُوهٌ :<br />
أحدها : أنَّ واحدَهُ نصابٌ ونُصُبٌ ،   كحِمار وحُمُر .  <br />
وثانيها : أنَّ وَاحِدَهُ نَصْبٌ فقولُكَ : نَصْبٌ ونُصُب كسَقْفٍ وسُقُف ،   وهو قولٌ لابْن الأنْبَارِي{[10761]} .  <br />
وثالثها : أنَّ واحدهُ النَّصْبَةُ .   قال اللَّيْثُ : [ النَّصْبُ ]{[10762]} جمعُ النَّصْبَةِ ،   وهي علامةٌ تُنْصَبُ للقومِ ،   وإنْ قُلْنا : النصبُ وَاحِدٌ ،   فجمعُهُ أنْصَابٌ ،   مثل عُنُق وأعْنَاق .  <br />
قال الأزْهَرِيُّ{[10763]} : وقد جعل الأعْشَى النُّصُبَ واحداً ،   وذكر البيتَ المُتَقَدِّم لَكِنْ رَوَاهُ عَلَى وَجْهٍ آخر ،   قال : [ الطويل ]<br />
وَلاَ النُّصُبَ المَنْصُوبَ لا تَعْبُدَنَّهُ *** لِعَافِيَةٍ واللَّهَ رَبَّكَ فَاعْبُدَا{[10764]}<br />
فصل<br />
قال بَعْضُهمْ{[10765]} : النُّصُبُ الأوْثانُ ،   واسْتَبْعَدَهُ قَوْمٌ ؛ لأنَّ هذا مَعْطُوفٌ على قولِهِ : { وَمَا أُهِلَّ لِغَيْرِ اللَّهِ بِهِ } وذلك هو الذَّبْحُ على اسمِ الأوْثانِ ،   والمعطوفُ يَجِبُ أنْ يكُونَ مُغَايِراً للمعطوفِ عَلَيْهِ .  <br />
وقال ابنُ زَيْدٍ : { وَمَا ذُبِحَ عَلَى النُّصُبِ } ،   { وَمَا أُهِلَّ لِغَيْرِ اللَّهِ بِهِ } هُمَا وَاحِدٌ{[10766]} .  <br />
وقال مُجاهدٌ وقتادةُ وابن جُرَيح : كانَتْ حولَ البيتِ ثلاثمائة وستُّونَ [ حَجَراً مَنْصُوبَة ]{[10767]} كان أهلُ الجاهِليّةِ يَعْبُدُونَها وَيُعَظِّمُونها ويَذْبَحُونَ{[10768]} لها وليست هي بأصْنامٍ ،   إنَّما الأصنامُ هي المصَوَّرةُ المنقوشَةُ ،   وكانوا يُلَطِّخُونَها بتلك الدماءِ{[10769]} ،   ويضعُون اللحومَ عليها .   فقال المسلمون{[10770]} يا رسول الله : كان أهلُ [ الجاهلية ]{[10771]} يُعظمون البيتَ بالدَّمِ ،   فنحنُ أحَقُّ أنْ نُعَظِّمَهُ ،   وكأن النبي عليه الصلاة والسلام لم يكَره ذلك ،   فأنزل الله تعالى : { لَن يَنَالَ اللَّهَ لُحُومُهَا وَلاَ دِمَاؤُهَا } [ الحج : 37 ] .  <br />
وقوله : { وَمَا ذُبِحَ عَلَى النُّصُبِ } .  <br />
[ فيه وجهان :<br />
أحدهما : وما ذبح على الاعتِقادِ وتَعْظِيم النُّصب ]{[10772]} .  <br />
والثاني : ومَا ذُبِح لِلنُّصبِ ،   و  "  اللاَّمُ  "  و  "  عَلَى  "  يتعاقَبَانِ .   قال تعالى : { فَسَلاَمٌ لَّكَ مِنْ أَصْحَابِ الْيَمِينِ } [ الواقعة : 91 ] [ أي : فَسَلامٌ عَلَيْكَ ]{[10773]} ،   وقال : { وَإِنْ أَسَأْتُمْ فَلَهَا } [ الإسراء : 7 ] أي فعليها .  <br />
قولُه سبحانَهُ : { وَأَنْ تَسْتَقْسِمُواْ بِالأَزْلاَمِ }  "  أنْ  "  وما في حَيِّزهَا في مَحَلِّ رَفْعٍ عَطْفاً على  "  الميْتَةِ  "  .  <br />
والأزْلامُ : القِدَاحُ ،   واحدُها : زَلْم ،   وزُلْم بِفَتْح الزَّاي وضَمِّها ذكره الأخْفَشُ .  <br />
وإنَّما سُمِّيتِ القِدَاحُ بالأزلامِ ؛ لأنَّها زُلِمَتْ أي : سُوِّيَتْ ،   ويقال : رجلٌ مُزْلِمٌ ،   وامرأةٌ مُزْلِمَةٌ إذَا كان خَفِيفاً قَلِيلَ العلائِقِ ،   ويقال : قَدَحٌ مُزْلَمٌ وزَلْم{[10774]} إذا حُرِّرَ وأُجِيدَ قَدُّهُ وَصِفَتُهُ ،   ومَا أحْسَنَ ما زلم سهمه ،   أيْ : سَوَّاهُ ،   ويقالُ لِقَوائمِ البَقَرِ : أزْلاَمٌ شُبِّهَتْ بالقِدَاحِ{[10775]} لِلَطَافَتِهَا .  <br />
وَفِي الاسْتِقْسامِ بالأزلامِ قولانِ :<br />
الأوَّلُ : كان أحدُهُم إذا أراد سَفَراً أو غَزْواً أو تِجَارَةً [ أو نِكَاحاً ]{[10776]} أو أمْراً آخَرَ ضرب بالقِدَاحِ ،   وكانوا قد كتبوا على بعضها أمَرَنِي رَبِّي ،   وعلى بعضها نَهَانِي رَبِّي ،   وتركوا بَعْضَهَا خَالِياً عن الكِتَابَةِ ،   فإنْ خرجَ الأمرُ أقدْمَ على العملِ ،   وإنْ خرجَ النَّهْيُ أمْسَكَ وأعادَ ،   وإنْ خرج الغَفْل أعاد العملَ مَرَّةً أخْرَى .  <br />
وذكر البَغَوِيُّ{[10777]} أنَّ أزلامَهم كانت سبعةَ أقْدَاحٍ مُسْتَوِيةً مِنْ شوحط يكونُ عند [ سادن ]{[10778]} الكعبة ،   مكتوب على واحدٍ منها : نَعمْ ،   وعلى واحد : لاَ ،   وعلى واحدٍ منها : مِنْكم ،   وعلى واحدٍ مِنْ غَيْرِكم ،   وعلى واحد : مُلْصَق ،   وعلى واحد : العَقْل ،   وواحد غفل ليس عليه شيءٌ ،   وكانوا إذا أرادُوا أمْراً أو تداوَرُوا في نَسَبٍ أو اختلفوا في تَحَمُّلِ عَقْلٍ جاءوا إلى هُبَل ،   وهو أعظمُ أصْنامِ قُرَيشٍ ،   وجاءوا بِمائَةِ دِرْهَمٍ وجَزُورٍ فأعطَوْها صاحبَ القِداح حتى يُجيلَ القومُ ويقولُون : يا إلَهنَا إنَّا أرَدْنَا كذا وكذا ،   فإنْ خرج نَعَمْ فعلُوا ،   وإنْ خرج : لا ،   لمْ يَفْعَلُوا ذلك ،   ثُم عادُوا إلى القِدَاح ثانيةً ،   وإذا أجالُوا على نسبٍ ،   فإنْ خرج منكُم [ كان وسيطاً منهم ،   وإن خرج من غيركم كان حليفاً ،   وإن خرج مُلْصقٌ كان على منزلته لا ]{[10779]} نسب له ولا حِلْفَ ،   وإذا اختلفُوا في عَقْلٍ فمن خرج عليه قَدَحُ العَقْلِ حَمَلَهُ ،   وإنْ خَرَجَ الغَفْل أجَالُوا ثانياً حتى يخرج المكتوبُ فنهى اللهُ تعالى عن ذلك وحرَّمهُ .  <br />
قال القُرطبيُّ{[10780]} : وإنّما قِيل لهذا الفعلِ اسْتِقْسام{[10781]} ؛ لأنَّهُمْ كانوا يَسْتَقْسِمُون به [ الرِّزْقَ ]{[10782]} فِيما يُريدون ،   كما يُقال : الاستقسامُ في الاستدعاءِ للسقي ،   ونظيرُ هذا الذي حرَّمه [ اللَّهُ ]{[10783]} قولُ المنَجِّم : لا يخرج من أجل نجم{[10784]} كذا ،   وأخرج من أجل نَجْم كذا .  <br />
وقال المؤرِّجُ وكثيرٌ مِنْ أهْلِ اللُّغَةِ : الاستقسامُ هاهنا هو المَيْسِرُ والقمارُ ،   ووجهُ ذِكْرِها مع هذه المطاعِمِ أنَّها كانت تقعُ{[10785]} عند البيتَ مَعَها .  <br />
وقال سَعِيدُ بنُ جُبَيْرٍ : الأزلامُ حَصًى بيضٌ يَضْرِبُونَ بها{[10786]} ،   وقال مُجاهِدٌ هي كعابُ فارسٍ والرُّومِ التي يَتَقَامَرُونَ بها{[10787]} .  <br />
وقال الشَّعْبِيُّ : الأزلامُ للعرب والكعابُ للعَجَمِ .  <br />
وقال سُفيانُ بنُ وكِيع : هي الشَّطَرَنج{[10788]} ،   قال عليه الصلاة والسلام :  "  مَنْ تَكَهَّنَ أو اسْتَقْسَمَ أو تَطَيَّرَ طيرةً تردّه عن سَفَرِهِ لَنْ يَلِجَ الدرجاتِ العُلَى مِنَ الجنَّةِ  " {[10789]} .  <br />
قوله :  "  ذَلِكُمْ فِسْقٌ  "  مُبْتَدأ وخَبَرٌ ،   واسمُ الإشارَةِ راجعٌ إلى الاستقسام بالأزلامِ خاصّةً ،   وهو مَرْوِيٌّ عن ابْنِ عبَّاسٍ ؛ لأنَّ معناه : حَرَّمَ عليكم تَناوُلَ الميْتَة [ وكذا ]{[10790]} ،   فرجع اسمُ الإشارةِ إلى هذا المُقَدَّرِ ،   فإنْ قيل : لم صار الاستقسامُ بالأزلامِ فِسْقاً والنبيُّ صلى الله عليه وسلم كان يُحِبُّ الفَألَ [ الحَسَن ]{[10791]} ؟<br />
فالجوابُ : قال الواحِدِي{[10792]} : إنَّما حرّم ذلك ؛ لأنَّه طلبٌ لمعرفةَ الغَيْبِ ،   وذلك حرام لِقَوْلِه تعالى : { وَمَا تَدْرِي نَفْسٌ مَّاذَا تَكْسِبُ غَداً } [ لقمان : 34 ] وقوله تعالى : { لاَّ يَعْلَمُ مَن فِي السَّمَاواتِ وَالأَرْضِ الْغَيْبَ إِلاَّ اللَّهُ } [ النمل : 65 ] ،   والحديث المتقدم .  <br />
قوله تعالى : { الْيَوْمَ يَئِسَ الَّذِينَ كَفَرُواْ مِن دِينِكُمْ }  "  اليَوْمَ  "  ظرف مَنْصُوبٌ ب  "  يَئِسَ  "  ،   والألفُ واللامُ فيه للْعَهْدِ .  <br />
قيل : أراد به يَوْمَ  "  عَرَفَة  "  وهو يَوْمُ  "  الجُمُعَةِ  "  عام حَجَّةِ الوَدَاعِ نزلت هذه الآيةُ فيه بَعْد العَصْرِ .  <br />
[ وقيل : هو يوم ]{[10793]} دخوله صلى الله عليه وسلم  "  مَكَّةَ  "  سنةَ تِسْعٍ .  <br />
وقِيل : [ سنة ]{[10794]} ثَمَانٍ .  <br />
وقال الزَّجَّاجُ - وتبعه الزَّمَخْشَرِيُّ - : إنَّها لَيْسَتْ للْعَهْدِ ،   ولم يُرِدْ ب  "  اليوم  "  [ يوماً ]{[10795]} مُعَيَّناً ،   وإنَّما أراد به الزمانَ الحاضِرَ وما يُدانيه من الأزْمِنَةِ الماضِيَةِ والآتِيَةِ كقولك : كُنْتَ بالأمْس شابًّا{[10796]} ،   وأنْتَ اليومَ أشْيَب ،   لا تُرِيدُ بالأمسِ الذي قَبْلَ يَوْمِك ،   و [ لا ]{[10797]} باليومِ الزَّمَن الحاضِر فَقَطْ ،   ونحوه  "  الآن  "  في قَوْلِ الشَّاعِرِ : [ الكامل ]<br />
الآنَ لَمَّا ابْيَضَّ مَسْرُبَتِي *** وعَضَضْتُ مِنْ نَابِي عَلَى جذْم{[10798]}<br />
ومثله أيضاً قوله زهير : [ الطويل ]<br />
وأعْلَمُ مَا في اليَوْمِ والأمْسِ قَبْلَهُ *** وَلكنَّنِي عَنْ عِلْمِ مَا فِي غدٍ عَمِ{[10799]}<br />
لم يُرِدْ بهذه حقائقها .  <br />
والجمهورُ على  "  يَئسَ  "  بالهمزة ،   وقرأ يَزِيدُ{[10800]} بنُ القَعْقَاع  "  يَيسَ  "  بِيَاءيْنِ مِنْ غَيْرِ همزة .  <br />
ورُوِيَتْ - أيضاً - عَنْ أبِي عَمْرو ،   ويقال : يَئِسَ يَيْئَسُ ويَئْيِسُ بفتح عَيْنِ المضارعِ وكسرها ،   فهو شاذّ .  <br />
ويقال : أيس [ أيضاً ]{[10801]} مقلوبٌ من  "  يَئِسَ  "  فوزنُهُ  "  عَفِل  "  ويدلُ على القَلْبِ كَوْنُهُ لم يُعَلَّ ،   إذ لو لم يقدر ذلك لَلَزِمَ إلغاءُ المقتضى ،   وهو تَحَرُّكُ حَرْفِ العِلّة ،   وانفتاحُ ما قبله ،   لكنَّه لما كان في مَعْنَى ما لم يُعَلَّ صَحَّ .  <br />
واليَأسُ : انقطاعُ الرَّجاءِ ،   وهو ضدُّ الطَّمَعِ .  <br />
 "  مِن دِينِكُمْ  "  مُتَعلقٌ ب  "  يئس  "  ،   ومعناها ابتداءُ الغَايَةِ ،   وهو على حَذْفِ مُضَاف ،   أي : منْ إبْطَالِ أمْرِ دِينكم .  <br />
فصل<br />
لَمَّا حَرَّم وحَلَّلَ فِيما تقدم ،   وخَتَمَ الكلام بقوله :  "  ذَلِكُمْ{[10802]} فِسْقٌ  "  ثُمَّ حَرَّضَهُم على التمسُّكِ بما شَرَع لهم ،   فقال : { الْيَوْمَ يَئِسَ الَّذِينَ كَفَرُواْ مِن دِينِكُمْ فَلاَ تَخْشَوْهُمْ وَاخْشَوْنِ } أي : فلا تخافُوا المُشْرِكين في خلافكم لهم{[10803]} في الشَّرائعِ والأدْيَانِ ،   فإنِّي أنْعَمْتُ عليكم بالدولةِ القاهِرةِ ،   وصارُوا مَقْهُورِينَ لكم ذَلِيلينَ عندكم ،   وحصل لهم اليأسُ مِنْ أنْ يَصِيروا قاهِرين لكم مُسْتَوْلِين عليكم ،   وإذا صار الأمرُ كذلك فيجبُ عليكم أنْ لا تَلْتَفِتُوا إليهم وأنْ تُقْبِلُوا على طاعَةِ اللَّهِ تعالى ،   والعملِ بشرائعِهِ .  <br />
وفي قوله : { الْيَوْمَ يَئِسَ الَّذِينَ كَفَرُواْ مِن دِينِكُمْ } قولان :<br />
الأولُ : يَئِسُوا مِنْ أنْ يُحَلِّلُوا الخبائِثَ بعد أنْ جعلها الله محرمةً .  <br />
والثاني : يَئسُوا مِنْ أنْ يَغْلِبُوكم على دينكُم ؛ لأنَّ الله تعالى قد وعد بإعلاء هذا الدين على كُلِّ الأديانِ بقوله : { لِيُظْهِرَهُ عَلَى الدِّينِ كُلِّهِ } [ التوبة : 33 ] فَحَقَّقَ ذلك النَّصْرَ ،   وأزالَ الخوفَ .  <br />
واسْتَدَلُّوا بهذه الآيةِ على أنَّ التقيَّةَ جائِزةٌ عِنْد الخوفِ ؛ لأنَّ اللَّه تعالى أمرهم بإظهارِ الشَّرائعِ عند زوال الخَوْفِ مِنَ الكفَّارِ ،   فدلَّ على جوازِ تركِهَا عند الخوف{[10804]} .  <br />
قوله سبحانه وتعالى : { الْيَوْمَ أَكْمَلْتُ لَكُمْ دِينَكُمْ وَأَتْمَمْتُ عَلَيْكُمْ نِعْمَتِي } في قوله :  "  الْيَوْمَ  "  [ كالكلام في  "  اليوم  "  ]{[10805]} قبلَهُ .  <br />
نزلت هذه الآيةُ يَوْمَ الجُمُعَةِ يَوْمَ عَرَفَةَ بعد العصر فِي حجَّةِ الوداع ،   والنبيُّ صلى الله عليه وسلم [ وشرف وكرم ومجد وعظم ]{[10806]} وَاقِف بِعَرَفات على ناقَتِهِ العَضْبَاءِ فكاد عَضُدُ الناقةِ يَنْقَدُّ من ثِقلها ،   فَبَرَكَتْ .  <br />
عن عُمَر بن الخطَّابِ رضي الله عنه أنَّ رَجُلاً من اليهودِ قال له : يا أميرَ المؤمِنينَ آيةٌ في كتابكم تقرؤونها ،   لو علينا مَعْشَرَ يَهُودٍ نَزَلَتْ لاتَّخَذْنَا ذلِكَ اليومَ عِيداً ،   قال : أيُّ آيةٍ ؟ قال : { الْيَوْمَ أَكْمَلْتُ لَكُمْ دِينَكُمْ وَأَتْمَمْتُ عَلَيْكُمْ نِعْمَتِي وَرَضِيتُ لَكُمُ الإِسْلاَمَ دِيناً } قال عُمَرُ : قَدْ عرفْنَا ذلك اليوم والمكانَ الذي أنْزِلَتْ فيه على النبي صلى الله عليه وسلم وهو قائِمٌ بِعَرَفَةَ يوم جمعة{[10807]} .  <br />
أشار عُمَرُ إلى أن ذلك اليومَ كان لنا عِيداً .  <br />
قال ابنُ عباسٍ : كان ذلك اليوم خمسةَ أعْيادٍ ،   جُمْعَةً وعَرفَةَ وعِيدَ اليَهُودِ والنَّصارَى والمَجُوسِ ،   ولم يَجْتَمِعْ أعيادُ أهْلِ المِلَلِ في يوم قبله ولا بعده{[10808]} .  <br />
وروى هارونُ بنُ عنترةَ عن أبيه قال :  "  لما نزلت هذه الآيةُ بَكَى عُمَرُ فقال له النبي صلى الله عليه وسلم  "  ما يُبْكِيكَ يا عُمَرُ ؟  "  فقال : أبْكانِي أنا كنا في زيادَةٍ مِنْ ديننَا ،   [ فأما إذْ ]{[10809]} كمُلَ فإنَّهُ لم يكمل{[10810]} شيءٌ إلا نَقَصَ ،   قال :  "  صَدَقْتَ  "  ،   فكانت هذه الآيةُ نَعْيَ النبي صلى الله عليه وسلم{[10811]} .  <br />
وعاش بعدها إحْدَى وثمانِينَ يَوْماً ،   ومات يَوْم الاثنَيْنِ بعدما زَاغَتِ الشمسُ لليلتين خَلَتَا من شهرِ ربيع الأوَّلِ سَنَة إحْدَى عَشَر مِنَ الْهِجْرَةِ ،   وقيل : يومَ الاثنين يَوْمَ الثاني عشر مِنْ ربيع الأولِ ،   وكانت هجرتُه في الثاني عشر مِنه .  <br />
فقوله : { الْيَوْمَ أَكْمَلْتُ لَكُمْ دِينَكُمْ } يَعْنِي يوم نُزولِ هذه الآيةِ أكملتُ لكُمْ دينكمْ الفَرائِضَ والسُّنَنَ ،   والحدُودَ والجهادَ ،   والحلالَ والحرامَ ،   فلم ينزل بعد هذه الآيةِ حلالٌ ولا حرامٌ ولا شيءٌ من الفرائِضِ ،   وهذا [ معنى ]{[10812]} قولِ ابْنِ عباسٍ{[10813]} .  <br />
ورُوي عنه أنَّ آيةَ الرِّبَا نزلت بعدها ،   وقال سَعِيدُ بنُ جُبَيْر وقتادةُ : { الْيَوْمَ أَكْمَلْتُ لَكُمْ دِينَكُمْ } فلم يَحُجَّ معكم مشْرِكٌ ،   وقيل : أظهرت دينَكُمْ وأمَّنْتكم من العدُوِّ{[10814]} ،   وقوله تعالى : { الْيَوْمَ أَكْمَلْتُ لَكُمْ دِينَكُمْ } يَقْتَضِي أن الدين كان نَاقِصاً قَبْلَ ذلك ،   وذلك يُوجِب أنَّ الدينَ الذي كان عليه محمدٌ صلى الله عليه وسلم مُواظِباً عليه أكْثَرَ عُمُرِه كان ناقصاً ،   وإنما وُجِدَ الدينُ الكامل في آخر عمرِهِ مُدَّةً قليلةً .  <br />
وأجَابُوا عنه بوجُوهٍ :<br />
أحدها : أنَّ المرادَ ما تقدم من إزالةِ الخوفِ عنهم ،   كما يقولُ الملِكُ إذا اسْتَوْلى على عدوه وقَهَرَهُ قَهْراً كُلِّياً : كَمُلَ ملكُنَا ،   وهذا ضَعيفٌ ؛ لأنَّ مُلْكَ ذلك المَلِك كان قبل قَهْرِ العدو ناقصاً فينبغي على هذا أن يُقال : إنَّ دينَ محمد صلى الله عليه وسلم كان ناقصاً قبل ذلك اليوم .  <br />
ثانيها : [ أنَّ ]{[10815]} المراد أكملتُ لكُم ما تَحْتَاجُون إليه في تكاليفكم من تعاليم الحلالِ والحرامِ ،   وهذا - أيضاً - ضعيفٌ ؛ لأنه لو لم يُبَيّن قبل هذا اليوم ما كانوا مُحْتاجين إليه من الشرائعِ ،   كان ذلك تأخيراً للبيانِ عن وقت الحاجة ،   وأنه لا يَجُوزُ .  <br />
وثالثها : وهو المختارُ ما ذكره القفَّالُ{[10816]} وهو أن الدينَ ما كان ناقصاً ألْبَتَّةَ ،   بل كان كامِلاً أبداً ،   وكانت الشرائِعُ النازِلةُ من عند الله تعالى في كل وقت كافية في ذلك إلا أنه تعالى كان عالماً في أوّل وقتِ المبعثِ في أن هذا اليوم ليس بكاملٍ في الغَدِ ،   ولا مَصْلَحَةَ فيه ،   فلا جَرَمَ كان يُنْسَخ بعد الثبوت ،   وكان ينزل بعد العدم ،   وأما في آخر زمانِ المبْعَثِ فأنزل الله شريعةً كامِلَةً ،   وحكم ببقائها إلى يوم الدِّين{[10817]} .  <br />
فالشرعُ أبداً [ كَانَ ]{[10818]} كامِلاً ،   إلاَّ أنَّ الأوّلَ كمال إلى زمان مَخْصُوصٍ والثاني : كمالٌ إلى يوم القيامة ،   فلهذا قال : { الْيَوْمَ أَكْمَلْتُ لَكُمْ دِينَكُمْ } .  <br />
وأجاب القُرْطبي{[10819]} : يقال : لِمَ قُلتَ إن كل نقص فهو عَيْبٌ ،   أرَأيْتَ نُقْصانَ الشَّهْرِ عَيْباً ؟ ونُقْصان صلاةِ المسافرِ أهُوَ عَيْبٌ ونقصان العمر الذي أراده الله بقوله : { وَمَا يُعَمَّرُ مِن مُّعَمَّرٍ وَلاَ يُنقَصُ مِنْ عُمُرِهِ } [ فاطر : 11 ] أهو عيب ؟ وكذلك نُقصَانُ أيّامِ الحَيْضِ عن المعهود ؟ ونقصان أيام الحمل ؟ ونقصان المال بسرقَةٍ أو حَرِيقٍ أو غَرَقٍ إذا لم يَفْتقر صاحبه ؟ فنقصان الدين في الشرع قَبْلَ أن يلحق الله الأجزاء الباقية في عِلْمِ الله تعالى لَيْسَ بعَيْبٍ ،   فمعنى قوله : { الْيَوْمَ أَكْمَلْتُ لَكُمْ دِينَكُمْ } يخرج على وجهين :<br />
أحدهما : أن المراد بلَّغْتُه أقْصَى الحد الذي كان له عندي فيما قضيتُهُ وقدَّرْتُهُ ،   وذلك لا يُوجِبُ أنْ يكونَ ما قَبْلَ ذلك ناقصاً عما كان عند الله تعالى لكنه يُوصَفُ بنُقصان مُقَيّد ،   فيقال : أكمل الله ناقِصاً عما كان عند الله تعالى أنه مُلحِقه به وضامه إليه كالرجل يُبلغه الله تعالى مائةَ سَنَةٍ ،   فيُقالُ : أكمل الله عُمُرَهُ [ فلا يلزم من ذلك أنْ يكون عُمُره ]{[10820]} ناقصاً حِين كان ابْن ستِّين سنة نَقْصَ قُصُورٍ وخللٍ{[10821]} ،   فإن النبيَّ صلى الله عليه وسلم يَقُولُ{[10822]} :  "  مَنْ عمَّره الله سِتين سَنَةً فقد أعْذَرَ إليه في العُمُر " {[10823]} .   وقد بلغ الله بالظُّهر والعصر والعشاء أربع ركعات ،   فلو قيل : أكملها كان الكلامُ صحيحاً ،   ولا يَلْزَمُ من ذلك أنَّها حينَ كانت ركعتين كانت ناقصةً نَقْصَ قُصُورٍ وخللٍ ،   ولو قيل : كانت ناقصة عما عند الله أنه ضامَّه إليها وزائده عليها لكان ذلك صحيحاً ،   فهكذا هذا في شرائع الإسلام وما كان شرع منها شيئاً فشيئاً إلى أنْ أنْهى الله الدِّين مُنْتَهَاهُ الذي كان له عنده .  <br />
الثاني : أنَّ المراد بقوله : { الْيَوْمَ أَكْمَلْتُ لَكُمْ دِينَكُمْ } أنَّهُ وفَّقَهُمْ إلى الحجِّ الذي لم يكن بقي عليهم مِنْ أرْكان دينهم{[10824]} غَيْرُه ،   فحَجُّوا فاستجمع{[10825]} لهم الدِّين أداء لأرْكَانِه ،   ولقوله - عليه الصلاة والسلام - :  "  بُنِيَ الإسلامُ على خَمْسٍ  "  الحديثَ ،   وقد كانُوا تشَهَّدُوا وصَلُّوا وزكوا وصامُوا وجاهَدُوا واعْتَمرُوا ولم يكونوا حجُّوا ،   فلما حجُّوا ذلك اليوم أنزل الله تعالى وهم بالموقف هذه الآية .  <br />
فصل رد شبه الاستدلال بهذه الآية على بطلان القياس<br />
استَدلُّوا بهذه الآيةِ على بُطْلان القياسِ ،   لأنَّ الآيةَ دلت على أنه تعالى قد نَصَّ على الحُكْمِ في جميع الوقائعِ ،   [ إذْ لَوْ بَقِيَ بَعْضُهَا غَيْرَ مُبَيَّنِ الحكم لم يكن الدينُ كامِلاً ،   وإذا حصلَ النص في جميع الوقائعِ ]{[10826]} فالقياسُ إنْ كان [ على ]{[10827]} وفْقِ النصِّ كان عَبَثاً ،   وإنْ كان خلافَهُ كان باطِلاً .  <br />
وأجيبُ{[10828]} بأن المرادَ بإكْمالِ الدينِ أنَّهُ تعالى بيَّن حُكْمَ جَميعِ الوَقَائِعِ بعضها بالنص ،   وبعضُها بيَّن طريقَ الحكمِ فيها بالقياسِ فإنَّهُ تعالى لما جَعَل الوقائع قِسْمَيْن :<br />
أحدهما : التي{[10829]} نَصَّ على أحكامهما{[10830]} .  <br />
والثاني : أنواعٌ يمكنُ استنباطُ{[10831]} الحكمِ فيها بواسطةِ قياسِها على القسم الأولِ ،   ثم إنَّه تعالى أمَرَ بالقياسِ ،   وتعبَّد المُكَلفِين به فكان ذلك في الحقيقة بياناً لِكُل الأحْكامِ .  <br />
قال نُفاةُ القياسِ : الطريقُ المقتضيةُ لإلحاقِ غَيْرِ المنصُوصِ بالمنصُوصِ ،   إمَّا أنْ تكونَ قَطْعِيَّةً أو غَيْرَ قَطْعِيَّةٍ .  <br />
فإنْ كانت قطعيّةً فلا نِزاعَ في صحته ،   فإنَّا نسلم أن القياسَ المبنيّ على المقدمات اليَقِينيَّةِ حُجةٌ ،   وهذا القياسُ يكون المصِيبُ فيه واحداً ،   ومخالِفهُ يَسْتَحِقُّ العقاب وينْقَصُ به قضاءُ القاضِي ،   وأنتم لا تقولون بذلك ،   وإنْ كانت طريقة ظنية{[10832]} كان كل واحدٍ يُمْكِنُه أنْ يَحْكُمَ بما غلب على ظَنِّه مِنْ غَيْرِ أنْ يَعْلَم [ هل ]{[10833]} هو دين اللَّهِ أمْ لا ؟ وهل هو الحكمُ الذي حكم [ به الله ]{[10834]} أم لا ؟ ومثل هذا لا يكون إكْمالاً للدّين ،   بل يكون ذلك{[10835]} إلقاءً للخلقِ في وَرطَة الظُّنونِ ،   وأجيب{[10836]} بأنه إذَا كان كُلُّ مُجْتَهِدٍ مُكَلَّفاً بالعمل بمقتضى ظَنِّه [ كان ]{[10837]} ذلك إكْمَالاً ويكون كُلُّ مكلفٍ قاطعاً بأنَّه عامل بحكم الله تعالى .  <br />
قوله سبحانه : { وَأَتْمَمْتُ عَلَيْكُمْ نِعْمَتِي }  "  عليكم  "  متعلقٌ ب  "  أتممت  "  ،   فلا يجوزُ [ تعلُّقهُ ]{[10838]} ب  "  نعمتي  "  ،   وإن كان فعلها يتعدَّى ب  "  على  "  نحو : { أَنعَمَ اللَّهُ عَلَيْهِ وَأَنْعَمْتَ عَلَيْهِ } [ الأحزاب : 37 ] ؛ لأن المصدر لا يتقدَّمُ عليه معمولُه{[10839]} ،   إلا أن ينُوبَ مَنَابَهُ .  <br />
قال أبُو البقاءِ : فإنْ جعلتهُ على التَّبِيين أيْ :  "  أتْمَمْتُ  "  أعْنِي  "  عَلَيْكُمْ  "  جاز ولا حاجة إلى ما ادّعَاهُ .  <br />
ومعنى { أتممت عليكم نعمتي } أيْ : أنجزتُ وَعْدِي في قوله : { وَلأُتِمَّ نِعْمَتِي عَلَيْكُمْ } ،   وكان مِنْ تَمَامِ نعمته أنْ دَخَلُوا مكةَ آمنينَ ،   وعليها ظَاهِرِينَ ،   وحَجُّوا آمنين مُطْمَئِنِينَ لمْ يخالِطْهُمْ أحَدٌ من المشركين .  <br />
قال ابنُ الخطيبِ{[10840]} : وهذا المعنى قد عُرِفَ بقوله : { الْيَوْمَ يَئِسَ الَّذِينَ كَفَرُواْ مِن دِينِكُمْ } فَحَمْلُهُ على هذا تَكْرِيرٌ ،   وإنما معنى قوله : { وَأَتْمَمْتُ عَلَيْكُمْ نِعْمَتِي وَرَضِيتُ لَكُمُ الإِسْلاَمَ دِيناً } أي : بسبب ذلك الإكمالِ ؛ لأنه لا نِعْمَةَ أتم مِنْ نعمة الإسلام .  <br />
قوله سبحانه : { وَرَضِيتُ لَكُمُ الإِسْلاَمَ دِيناً } في  "  رَضِيَ  "  وجهان :<br />
أحدهما : أنه مُتعد لِواحدٍ ،   وهو  "  الإسلامَ  "  ،   و  "  دِيناً  "  على هذا حال .  <br />
وقيل : هو مُضَمَّنٌ معنى صَيَّرَ وجعل ،   فيتعدَّى لاثْنَيْن ؛ أولهما :  "  الإسْلاَمَ  "  والثاني :  "  دِيناً  "  .  <br />
 "  لكم  "  يجُوز فيه{[10841]} وجهان :<br />
أحدهما : أنه متعلق ب  "  رَضَيَ  "  .  <br />
والثاني : أنه متعلقٌ بمحذوفٍ ؛ لأنَّه حالٌ من الإسلامِ ،   ولكنَّه قُدِّمَ عليه .  <br />
ومعنى الكلام أنَّ هذا هو الدينُ المرضِيُّ عند الله ،   ويُؤكِّدُهُ قوله تعالى : { وَمَن يَبْتَغِ غَيْرَ الإِسْلاَمِ دِيناً فَلَن يُقْبَلَ مِنْهُ } [ آل عمران : 85 ] .  <br />
وقوله تعالى : { فَمَنِ اضْطُرَّ فِي مَخْمَصَةٍ غَيْرَ مُتَجَانِفٍ لإِثْمٍ } قد تقدم الكلامُ عليها في البقرة [ آية : 173 ] .  <br />
و  "  فِي مَخْمَصَةٍ  "  متعلقٌ ب  "  اضْطُرَّ  "  ،   [ ومعنى :  "  اضْطُرَّ  "  ]{[10842]} أُصِيبَ بالضر الذي لا يمكنه الامتناعُ معه من الميتةِ .  <br />
و  "  المخمصةُ  "  : المَجَاعَةُ ؛ لأنها تَخْمُصُ [ لها ]{[10843]} البطونُ : أيْ : تَضْمُرُ .  <br />
قال أهل اللُّغَةِ{[10844]} : الخَمْصُ والمَخْمَصَةُ : خَلاَءُ البطن من الطعامِ ،   وأصْلُهُ مِنَ الخَمْصِ الذي هو ضُمُور البَطْنِ .   يقال : رجل خميصٌ وخُمْصَان ،   وامرأة خَمِيصَةٌ وخُمْصَانة ،   والجمع خَمَائِصُ وخُمْصَانَاتٌ ،   وهي صِفَةٌ مَحْمُودَةٌ في النساء .  <br />
ويقال : رجل خُمْصَان وامرأة خُمْصَانة ،   ومنه أخْمَصُ القدمِ لدِقَّتِهَا ،   ويستعمل في الجوع والغَرْث .  <br />
قال : [ الطويل ]<br />
تَبِيتُونَ في المَشْتَى مِلاءً بُطُونكُمْ *** وَجَارَاتُكُمْ غَرْثَى يَبِتْنَ خَمَائِصَا{[10845]}<br />
وقال آخر : [ الوافر ]<br />
كُلُوا فِي بَعْضِ بَطْنِكُمُ تَعِفُّوا *** فإنَّ زَمَانَكُمْ زَمَنٌ خَمِيصٌ{[10846]}<br />
وُصِفَ الزمانُ بذلك مبالغةً كقولهم : نَهَارُهُ صَائِمٌ ،   ولَيْلُهُ قائم .   و  "  غَيْرَ  "  نُصبَ على الحال .  <br />
قال بعضُهم : يَنْتَصِبُ بمحذوفٍ مُقَدَّرٍ على معنى : فيتناول{[10847]} غَيْر مُتجانِفٍ ،   ويجوز أن يَنْتَصِبَ بقوله :  "  اضْطُرَّ  "  ويكونُ المقدرُ متأخراً .  <br />
والجمهور على  "  مُتجَانِفٍ  "  بألف وتخفيف النون من  "  تَجَانَفَ  "  .  <br />
وقرأ أبُو عبدِ الرحمن والنَّخَعِيُّ{[10848]}  "  مُتَجنِّفٍ  "  بتشديدِ النُّونِ دُونَ ألفٍ .  <br />
قال أبو مُحمدِ بن عَطِية : وهي أبْلَغُ من  "  متجانف  " {[10849]} في المعنى ؛ لأنَّ شِدّةَ العينِ تدلُ على مُبالغةٍ وتوغُّلٍ في المعنى .  <br />
و  " لإثم  "  متعلّق ب  "  متجانف  "  ،   واللام على بابها .  <br />
وقيل : هي بمعنى  "  إلى  " {[10850]} أي غَيْرَ مائِلٍ إلى إثْمٍ ولا{[10851]} حاجة إليه .  <br />
وقد تقدم معنى هذا واشتقاقها عند قوله : { فَمَنْ خاف من مُوصٍ جَنَفاً } .  <br />
قال القُرْطبي{[10852]} : هو معنى قوله : { غَيْرَ بَاغٍ وَلاَ عَادٍ } [ البقرة : 173 ] ،   ومعنى  "  الإثم  "  هاهنا أنْ تأكُلَ فوق الشِّبَع تلذُّذاً في قولِ أهل  "  العراق  "  .  <br />
وفي قولِ أهْلِ  "  الحجازِ  "  : أنْ تكُون عَاصِياً{[10853]} .  <br />
قال قتادةُ{[10854]} : غَيْرَ مُتَعرِّضٍ لمعْصِية في مقصده .  <br />
وقوله : { فَإِنَّ اللَّهَ غَفُورٌ } جُمْلةٌ ،   إمَّا في محلِ جَزْمٍ ،   أو رفعٍ على حَسَبِ ما قيل في  "  مَنْ  "  ،   وكذلك القول في الفاء إما واجبةٌ أو جائزةٌ ،   والعائدُ على كِلاَ التقديرين محذوفٌ ،   أيْ :  "  فَإِنَّ اللَّهَ غَفُورٌ  "  له ،   [ يعني ]{[10855]} : يَغْفِرُ له أكْلَ المُحَرَّمِ عند الاضطرارِ  "  رَحِيمٌ  "  بعباده ،   حَيْثُ أحَلَّ لهم ذلك المحرمَ عند احتياجهم إلى أكْلِهِ ،   وهذا مِنْ تَمام ما تقدم ذكره في المطاعِمِ التي حَرَّمَها الله تعالى ،   يَعْنِي : إنها وإنْ كانت مُحرمةً إلا أنها تحل في حالِ الاضطرارِ ،   ومن قوله :  "  فِسْق  "  إلى هاهنا - اعتراضٌ وقع في النَّسَقِ ،   والغَرَضُ منه تأكيدُ ما ذُكِر مِنْ مَعْنى التحريم ،   فإن تحريم هذه الخبائثِ من جملةِ الدِّينِ الكاملِ والنعْمةِ الثابتةِ والإسلامِ الذي هو الدينُ المرضيُّ عند الله تعالى .
