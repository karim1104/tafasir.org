لمَّا نَهَى عن مُوالاةِ الكُفَّار في الآيَات المُتقدِّمة بيَّن هاهُنَا من يجب مُوالاته .  <br />
قوله { إِنَّما وَلِيُّكم اللَّه } مُبْتَدأ وخبر .  <br />
و  "  رَسُولُه  "  و  "  الَّذين  "  عَطْف على الخبر .  <br />
قال الزَّمَخْشَرِي{[12101]} : قد ذُكِرَتْ جماعةُ فهلاَّ قيل :  "  إنَّما أوْلِياؤُكم  "  ؟<br />
وأجاب أنَّ الولاية بِطَرِيق الأصَالَة للَّه - تعالى - ثم نظم في سلك إثْباتها لِرَسُوله وللمُؤمِنين ،   ولَوْ جِيءَ به جَمْعاً ،   فقيل :  "  إنَّما أوْلِيَاؤُكم  "  لم يكن في الكلام أصْل وتبع .  <br />
قال شهاب الدِّين{[12102]} : ويحتمل وَجْهاً آخر ،   وهو أن  "  وَلي  "  بِزِنَةِ  "  فَعِيل  "  ،   و  "  فَعِيل  "  قد نصَّ عليه أهْلُ اللِّسَان أنَّهُ يقع لِلْواحد والاثْنَين والجماعة تَذْكيراً وتَأنِيثاً بلفظ واحدٍ ،   يُقَال :  "  الزَّيْدُون صَدِيق  "  و  "  هِنْد صَدِيق  "  ،   وهذا مِثْله غاية ما فِيهِ أنَّهُ مقدَّم في التَّرْكِيب ،   وقد أجَابَ الزَّمَخْشَري وغَيْره في قوله تعالى : { وَمَا قَوْمُ لُوطٍ مِّنكُم بِبَعِيدٍ } [ هود : 89 ] وذكر وَجْهَ ذلك ،   وهو شَبَهُهُ بالمصادر ،   وسَيَأتِي تحقِيقُه - إن شَاء اللَّه تعالى - .  <br />
وقرأ ابُن{[12103]} مسعُود :  "  إنَّما مولاكم الله  "  ،   وهي تَفْسِير لا قِرَاءة .  <br />
فصل في سبب نزول الآية<br />
روى ابن عباس - رضي الله عنهما - أنَّها نَزَلت في عُبَادَة بن الصَّامِت - رضي الله عنه - وعبد الله بن أبيِّ ابن سلُول - لَعَنَهُ اللَّه - ،   حين تَبَرأ عُبَادة من اليَهُود وقال : أتوَلَّى الله ورسوله والذين آمَنُوا ،   فنزل فِيهِم من قوله تعالى : { يَأَيُّهَا الَّذِينَ آمَنُواْ لاَ تَتَّخِذُواْ الْيَهُودَ وَالنَّصَارَى أَوْلِيَاءَ } [ المائدة : 51 ] ،   إلى قوله تعالى : { إِنَّمَا وَلِيُّكُمُ اللَّهُ وَرَسُولُهُ وَالَّذِينَ آمَنُواْ } يَعْني : عُبَادة بن الصَّامِت وأصْحَاب رسُول الله - صلَّى الله عليه وعلى آله وسلَّم - ،   وقال جَابِر بن عبد الله - رضي الله عنهما - : جاء عَبْدُ الله بن سلام - رضي الله عنه - إلى رسول الله - صلَّى الله عليه وعلى آله وسلَّم - فقال : يا رسول الله :  "  إن قَوْمنا قُرْيَظَة والنَّضِير قد هَجَرُونا وفَارَقُونا وأقْسَمُوا ألاَّ يُجَالِسُونا  "  فنزلت هذه الآية الكريمة ،   فَقَرأهَا عليه رسول الله - صلَّى الله عليه وعلى آله وسلَّم - فقال : يا رسول الله رَضِينا باللَّه وبرسُوله وبالمُؤمنين أوْلياء{[12104]} ،   وعلى هذا التَّأويل أراد بقوله - تعالى - :  "  وَهُمْ رَاكِعُون  "  صلاة التَّطَوُّع باللَّيْل والنَّهَار .  <br />
قال ابن عباس - رضي الله عنهما - والسُّديُّ - رحمه الله - : قوله تعالى : { وَالَّذِينَ آمَنُواْ الَّذِينَ يُقِيمُونَ الصَّلاَةَ وَيُؤْتُونَ الزَّكَاةَ وَهُمْ رَاكِعُونَ } ،   أراد به على بن أبي طالب - رضي الله عنه - مرَّ به سَائِلٌ وهو رَاكِعٌ في المَسْجِد فأعطاه خاتمه{[12105]} ،   وقال جُوَيْبِر عن الضَّحَّاك في قوله تعالى : { إِنَّمَا وَلِيُّكُمُ اللَّهُ وَرَسُولُهُ وَالَّذِينَ آمَنُواْ } [ قال : هم المُؤمِنُون بعضهم أوْلِيَاء بعض ،   وقال أبو جعفر محمد بن علي البَاقِر : { إنما وليكم الله ورسوله والذين آمنوا } نزلت ]{[12106]} في المُؤمنين ،   فقيل له : إن ناساً يقولون : إنَّها نزلت في عَلِيٍّ - رضي الله تعالى عنه - قال : هو من المُؤمنين{[12107]} .  <br />
قوله تعالى : { الَّذِينَ يُقِيمُونَ الصَّلاَةَ } فيه خَمْسَة أوْجُه :<br />
أحدها : أنَّهُ مَرْفُوع على الوَصْف ،   لقوله تعالى :  "  الَّذِين آمَنُوا  "  .  <br />
وصف المؤمنين بإقام الصلاة ،   وإيتاء الزَّكاة ،   وذكر هَاتَيْن العبادَتيْن دون سَائِر فُروع الإيمان ؛ لأنَّهما أفْضَلُهُمَا .  <br />
الثاني : أنَّه مرفوع على البَدَل من  "  الَّذِين آمَنُوا  "  .  <br />
الثالث : أنه خَبَرُ مُبْتَدأ مَحْذُوف ،   أي : هُمُ الذين .  <br />
الرابع : أنه عَطْف بَيَان لما قبله ؛ فإنَّ كل ما جاز أنْ يَكُون بدلاً جاز أن يكون بَيَاناً ،   إلا فيما اسْتُثْنِيَ كما تقدَّم .  <br />
الخامس : أنه مَنْصُوب بإضمار فِعْل ،   وهذا الوجه والَّذِي قَبْله من باب القَطْعِ عن التَّبِعيَّة .  <br />
قال أبو حيَّان{[12108]} - بعد أن نَقَل عن الزَّمَخْشَرِي وَجْهَي البدل ،   وإضمار المُبْتَدأ فقط - :  "  ولا أدْرِي ما الَّذِي مَنَعَهُ من الصِّفة ،   إذ هو المُتَبَادَرُ إلى الذِّهن ،   ولأنَّ المُبْدَلَ منه على نِيَّةِ الطَّرْح ؛ وهو لا يصحُّ هنا ؛ لأنَّه هو الوَصْف المترتِّبُ عليه صحّة ما بعده من الأوْصَاف  "  .  <br />
قال شهاب الدِّين{[12109]} : لا نسلِّم أنَّ المتَبَادر إلى الذِّهْنِ الوَصْف ،   بل البَدَل هو المُتبَادر ،   وأيضاً فإن الوَصْف بالموصول على خِلاف الأصْل ؛ لأنَّه مؤوَّل بالمُشْتَقِّ وليس بمُشْتَقٍّ ،   ولا نُسلِّم أن المُبْدَل مِنْه على نِيَّة الطَّرْح ،   وهو المَنْقُول عن سِيبوَيْه{[12110]} .  <br />
قوله :  "  وَهُمْ رَاكِعُون  "  في هذه الجُمْلَة وجهان :<br />
أظهرهما : أنَّها معطوفة على ما قَبْلَها من الجُمَلِ ،   فتكون صِلَةً للمْوصُول ،   وجاء بهذه الجملة اسميَّةً دون ما قَبْلَها ،   فلم يَقُلْ :  "  يَرْكَعُون  "  اهتماماً بهذا الوَصْف ،   لأنَّه أظهر أرْكَان الصلاة .  <br />
والثاني : أنَّها  "  واو  "  الحال ،   وصاحبُها هو واو  "  يُؤتُون  "  .  <br />
والمراد بالرُّكُوع الخُضُوع أي : يُؤتُون الصَّدَقة ،   وهم مُتُواضِعُون للفُقَراء الَّذين يتصَدَّقُون عليهم .  <br />
ويجُوز أن يُراد به الرُّكُوع حَقِيقةً ؛ كما تقدَّم عن عَلِيّ - رضي الله عنه - .  <br />
وقال أبو مُسلمِ{[12111]} : المراد من الرُّكوع : الخُضُوع ،   أي : يُصَلُّون ويَركَعُون وهم مُنْقَادُون خَاضِعُون لجميع أوَامِرِ اللَّه ونواهيه .
