قوله تعالى : { فَبِمَا نَقْضِهِم مِّيثَاقَهُمْ } الآية .  <br />
قد تقدَّم الكلامُ على نَظِيره .  <br />
قال القُرْطُبِي{[11213]} : والمعنى : فبنقضهم ميثاقهم ،   و  "  مَا  "  زائدة للتَّوْكيد .  <br />
قال قتادةُ وغيرُه{[11214]} : وذلك أنَّها تؤكِّد الكلام بمعنى تمكنه في النَّفْس من جهة حسن النَّظْم ،   ومن جهة تَكْثِيرِه [ للتَّوْكِيد ]{[11215]} ،   كقوله [ الوافر ]<br />
 .   .   .   *** لِشَيْءٍ مَّا يُسَوَّدُ مَنْ يَسُودُ{[11216]}<br />
وقيل : نَقْضُ الميثاق تكْذيبُ الرُّسل ،   وقيل{[11217]} : الأنْبِيَاء عليهم الصلاة والسلام ،   وقيل : كِتْمَانهم صفة مُحَمَّد صلى الله عليه وسلم ،   وقيل : المَجْمُوع .  <br />
وقال عطاء :  "  لَعَنَّاهم  "  أبعدناهم من رحمتنا ،   وقال الحسن ومُقاتل : مسَخْنَاهم قِرَدة وخنازير .   وقال ابن عبَّاس : ضربنا الجِزْية عَلَيْهِم{[11218]} .  <br />
قوله تعالى : { وَجَعَلْنَا قُلُوبَهُمْ قَاسِيَةً } .  <br />
قرأ الجمهور  "  قَاسِيَة  "  اسم فاعل من قَسَا يَقْسُو .  <br />
وقرأ{[11219]} الأخوان وهي قراءة عَبْد الله  "  قَسِيَّة  "  بفتح القاف وكَسْرِ السِّين وتشديد اليَاء ،   واختلف الناس في هذه القراءة .  <br />
فقال الفَارِسي : لَيْسَت من ألْفَاظِ العرب [ في الأصل ]{[11220]} ،   وإنَّما هي كَلِمَةٌ أعْجَمِيَّةٌ معرَّبة ،   يعني أنها مأخوذَةٌ من قولهم : درهم قَسِيّ ،   أي : مَغْشُوش ،   شبَّه قلوبهم في كونها غير صَافِية من الكَدَرِ بالدَّراهم المَغْشوشة غير الخَالِصَة ،   وأنْشَدُوا قول أبي زُبَيْد : [ البسيط ]<br />
لها صَوَاهِلُ فِي صُمِّ السِّلامِ كَمَا *** صَاحَ القَسِيَّاتُ في أيْدِي الصَّيَارِيفِ{[11221]}<br />
وقول الآخر : [ الطويل ]<br />
وَمَا زَوَّدُونِي غَيْرَ سَحْقِ عمَامَةٍ *** وَخَمْسِ مِىءٍ مِنْهَا قَسِيٌّ وَزَائِفُ{[11222]}<br />
وقال الزَّمَخْشَرِي{[11223]} ،   وقرأ عبد الله  "  قسيَّة  "  ،   أي : رديئة مغشوشة من قولهم :  "  درهم قَسيّ  "  ،   وهو من القَسْوة ؛ لأن الذَّهَبَ والفضَّةَ الخالِصَيْن فيهما لين ،   والمغشوش فيه صلابةٌ ويُبْسٌ والقَاسِي والقاسِح بالحاء المهملة أخوان في الدَّلالة على اليُبْس .  <br />
وهذا القول سبقه إليه  "  المُبَرِّد  "  ،   فإنه قال :  "  يسمى الدِّرْهم المَغْشُوش قَسِيًّا لصلابته وشدَّتهِ للغِشِّ الذي فيه  "  ،   وهو يرجعُ للمعنى الأوَّل ،   والقاسِي والقَاسِح بمعنى واحد .  <br />
وعلى هذين القَولَيْن تكون اللَّفظة عَرَبيَّةً .  <br />
وقيل : بل هذه القِرَاءَةُ توافق قِراءَة الجَمَاعَة في المعنى والاشْتِقَاق ؛ لأنَّه  "  فَعِيل  "  للمُبَالغة ك  "  شاهد  "  ،   و  "  شهيد  "  ،   فكذلك قاسٍ وقَسِيّ ،   وإنما أنِّث على معنى الجماعةِ في المعنى والاشْتِقَاق .  <br />
وقرأ الهَيْصَم{[11224]} بن شداخ :  "  قُسِيَّة  "  بضم القَافِ وتشديد اليَاء .  <br />
وقرئ {[11225]} "  قِسِيَّة  "  بكسر القاف إتْبَاعاً ،   وأصْلُ القراءَتَيْن :  "  قاسِوَة  "  ،   و  "  قَسِيوة  "  لأن الاشْتِقَاق من القَسْوة .  <br />
فصل<br />
والمعنى أنَّ قلوبهم ليست بخالصة الإيمان ،   بل إيمانُهُم مشوبٌ بالكُفْرِ والنِّفَاق ،   وقيل : نائيةٌ عن قُبُول الحقِّ ،   مُنْصَرِفة عن الانقيادِ للدَّلائل .  <br />
وقالت المُعْتَزِلَة{[11226]} : أخْبِرْنا عنها بأنَّها صارت قَاسِيَة ،   كما يُقَال : فلانٌ جعل فُلاناً قاسِياً وعدْلاً .   ثم إنَّه تعالى ذكر بَعْضَ نتائج تلك القَسْوَة ،   فقال : { يُحَرِّفُونَ الكَلِمَ عَن مَوَاضِعِهِ } وهذا التَّحْرِيفُ هو تَبْدِيلُهم نعت النَّبيِّ - صلَّى الله عليه وآله وسلَّم - ،   وقيل : التَّأويل البَاطل .  <br />
والجملة من قوله :  "  يُحَرِّفُونَ  "  فيها أرْبَعَةُ أوْجُه :<br />
أحدها : أنَّها مستأنَفَةٌ بيان لِقَسْوة قُلُوبهم ؛ لأنَّه لا قَسْوَةَ أعظم من الافتراء{[11227]} على اللَّه تعالى .  <br />
والثاني : أنَّها حَالٌ من مَفْعُول  "  لَعَنَّاهُمْ  "  أي : لَعَنَّاهم حال اتِّصافهم بالتَّحْريف .  <br />
والثالث : قال أبو البقاء{[11228]} : أنَّها حال من الضَّمِير المسْتَتِر في  "  قَاسِيَة  "  .  <br />
وقال : ولا يجُوزُ أن يكون حَالاً من  "  القُلُوب  "  ؛ لأنَّ الضَّمير في  "  يُحَرِّفُون  "  لا يرجع إلى  "  القُلُوب  "  ،   وفي هذا نَظَرٌ من حيث جواز أن يكون حَالاً من الضَّمِير في  "  قَاسِيَة  "  ،   يَلزَمُهُ أن يُجوِّزَ أن يكون حَالاً من  "  القُلُوبِ  "  ،   لأنَّ الضَّمير المستتر في  "  قَاسِيَة  "  يعود على  "  القُلُوبِ  "  ،   فكما يمتَنِعُ أن يكون حالاً من ظاهره{[11229]} [ يمتنع أن يكون حالاً من ضميره ]{[11230]} ،   وكأنَّ المانع الذي توهمه كون الضَّمير وهو الواو في  "  يُحَرِّفُون  "  إنَّما يعُود على اليَهُودِ بِجُمْلَتِهِم لا على قُلُوبهم خاصَّةً ،   فإن القُلُوب لا تحرَّف ،   إنما يُحرَّف أصْحَابُ القُلوب ،   وهذا لازِمٌ له في تجويزه الحالِيَّة من الضَّمِير في  "  قَاسِيَة  "  .  <br />
ولقائِل أن يَقُول : المراد ب  "  القُلُوب  "  نفس الأشْخَاص ،   وإنما عبَّر عنهم بالقُلُوب ؛ لأن هذه الأعضاء هي محلُّ التَّحْريف ،   أي : إنَّه صَادِرٌ عَنْها بتَفَكُّرِها فيه ،   فيجُوزُ على هذا أن يكُون حَالاً من  "  القُلُوب  "  .  <br />
والرابع : أن يكون حالاً من  "  هم  "  في  "  قُلُوبِهم  "  .  <br />
قال أبو البَقَاء{[11231]} : وهو ضعيف يعني : لأنَّ الحال من المُضَاف إليه لا تَجُوزُ ،   وغيره يجوِّزُ ذلك في مثل هذا الموضع ؛ لأن المُضافَ بَعْضُ المُضَافِ إليه .  <br />
وقرأ الجمهور بفتح الكاف{[11232]} وكَسْرِ اللاَّم ،   وهو جمع  "  كَلِمَة  "  .  <br />
وقرأ{[11233]} أبو رجاء  "  الكِلْم  "  بكَسْرِ الكَافِ وسُكُون اللاَّم ،   وهو تَخْفيفُ قِرَاءة الجَمَاعَة ،   وأصْلُها أنَّه كَسَرَ الكاف إتْبَاعاً ،   ثم سكن العَيْنَ تَخْفِيفاً .  <br />
وقرأ السُّلمي{[11234]} والنَّخْعي :  "  الكَلاَم  "  بالألِف ،   و  "  عَنْ مَوَاضِعهِ  "  وتقدَّم مثله في  "  النِّسَاء  "  [ النساء : 46 ] .  <br />
قوله تعالى : { وَنَسُواْ حَظَّا مِّمَّا ذُكِّرُواْ بِهِ } .  <br />
قال ابن عباسٍ{[11235]} : تركوا نَصِيباً ممَّا أمِرُوا بِهِ من الإيمَان بِمُحَمَّد - عليه الصلاة والسلام - .  <br />
قوله - سبحانه - : { وَلاَ تَزَالُ تَطَّلِعُ عَلَى خَآئِنَةٍ مِّنْهُمْ } .  <br />
في  "  خَائِنَةٍ  "  [ ثَلاثةُ أوْجُه ]{[11236]} :<br />
أحدها : أنَّها اسم فاعل ،   والهاء لِلْمُبَالغة ك  "  راوية وعلاَّمة [ ونسَّابة ]{[11237]}  "  ،   أي : على شخص خائن .  <br />
قال الشَّاعر : [ الكامل ]<br />
حَدَّثْتَ نَفْسَكَ بِالْوَفَاءِ وَلَمْ تَكُنْ *** لِلْغَدْرِ خَائِنَةً مُغِلَّ الإصْبَعِ{[11238]}<br />
الثاني : أن التَّاء للتَّأنيث ،   وأنِّث على معنى : طائفة أو فرقة أو نفس أو فَعْلة خائنة .  <br />
قال ابن عبَّاسٍ :  "  على خَائِنَةٍ  "  ،   أي : على مَعْصِية ،   وكانت خِيَانَتُهُم نقض العَهْدِ ،   ومظاهرتُهمْ المُشْرِكين على حَرْبِ رسُول الله - صلَّى الله عليه وعلى آله وسلَّم - وهَمّهم بِقَتْلِهِ وسمه ونحوها من الخِيَانَاتِ الَّتي ظَهَرَتْ منهم .  <br />
الثالث : أنها بمعنى المَصْدَرِ ك  "  العافِيَة والعَاقِبَة{[11239]} والكاذِبَة واللاغية والواقِيَة  "  ،   قال تعالى : { فَأُهْلِكُواْ بِالطَّاغِيَةِ } [ الحاقة : 5 ] أي : بالطُّغيان ،   وقال : { لَيْسَ لِوَقْعَتِهَا كَاذِبَةٌ } [ الواقعة : 2 ] أي كذب ،   وقال : { لاَّ تَسْمَعُ فِيهَا لاَغِيَةً } [ الغاشية : 11 ] أي : لغواً .  <br />
وتقول العرب : سمعت رَاغِيَة الإبل ،   وثاغية الشاء ،   يعنون رُغَاءَها وثُغَاءَها .  <br />
قال الزَّجَّاج{[11240]} : [ ويقال ]{[11241]} عافاه اللَّهُ عافِيَةً ،   ويؤيِّد هذا الوجْهُ قراءة الأعمش{[11242]}  "  على خِيَانَة  "  وأصل  "  خَائِنة  "  خاوِنَة وخِيَانة وخوانة [ لقولهم : تَخوَّنَ ،   وخَوَّان ]{[11243]} وهو [ أخْوَن ،   وإنما أُعِلاَّ إعلال  "  قائمة  "  ،   و  "  قيام  "  ]{[11244]} .  <br />
و  "  مِنْهُم  "  صفة ل  "  خَائِنة  "  إن أُرِيد بها الصِّفَة ،   وإن أُرِيد بها المَصْدر قُدِّر مضَاف ،   أي : من بَعْض خيَانَاتِهِم .  <br />
قوله تعالى :  "  إلاَّ قَلِيلاً  "  منصُوب على الاسْتِثْنَاء ،   وفي المُسْتَثْنَى منه أربَعَةُ أقْوال :<br />
أظهرُهَا ،   أنَّه لفظ  "  خَائِنة  "  ،   وهُمَ الأشخاص المَذْكورُون في الجُمْلَةِ قَبْلَه ،   أي : لا تَزالُ تطَّلِع على مَنْ يَخُون منهم إلاَّ القليل ،   فإنَّه لا يَخُون فلا تَطلعْ عليه ،   وهم الذين أسْلَمُوا من أهْلِ الكِتَاب كعَبْدِ الله بن سلام وأصْحَابه .  <br />
قال أبُو البَقَاء{[11245]} :  "  ولو قُرئ بالجَرِّ على البَدَلِ لكان مُسْتَقِيماً  "  ،   يعني على البَدَل من  "  خَائِنَة  "  ،   فإنَّهُ في حيز{[11246]} كلامٍ غير مُوجب .  <br />
والثاني ذكره ابْنُ عطيَّة{[11247]} : أنَّهُ الفعل أي : لا تَزَالُ تَطَّلِعُ على فِعْلِ الخيانة إلا فَعْلاً قَلِيلاً ،   وهذا واضِحٌ إن أُريد بالخِيَانَة أنَّها صِفَةٌ لِلْفَعْلَةِ المقدَّرَة كما تقدَّم ،   ولكن يبْعد ما قاله ابنُ عطيَّة قوله بعده :  "  مِنْهُم  "  ،   وقد تقدَّم لَنَا نَظِيرُ ذلك في قوله : { مَّا فَعَلُوهُ إِلاَّ قَلِيلٌ مِّنْهُمْ } [ النساء : 66 ] من حَيْث جَوَّز الزَّمَخْشَرِيّ فيه أنْ يكون صِفَةً لمَصْدرٍ مَحْذُوفٍ .  <br />
الثالث : أنَّه  "  قُلُوبهُم  "  في قوله تعالى : { وَجَعَلْنَا قُلُوبَهُمْ قَاسِيَةً } ،   قال صاحب هذا القول : والمُرادُ بِهِمْ :  "  المُؤمِنُون ؛ لأنَّ القَسْوَة زالت عن قُلُوبِهِم  "  ،   وهذا بَعِيدٌ جدًّا ؛ لقوله :  "  لَعَنَّاهُمْ  "  .  <br />
الرابع : أنَّهُ الضَّمِير في  "  مِنْهُم  "  من{[11248]} قوله : { عَلَى خَائِنَةٍ مِنْهُمْ } قال مَكيّ{[11249]} .  <br />
قوله [ تعالى ] : { فَاعْفُ عَنْهُمْ وَاصْفَحْ } .  <br />
قيل :  "  العَفْوُ  "  نُسِخَ بآية السَّيْف ،   وقيل : لَمْ يُنْسَخُ ،   وعلى هذا فيه وجهان :<br />
أحدهما : معناه : فاعْفُ عن مُؤمنيهم ،   ولا تُؤاخِذْهُم بما سلَفَ منهم .  <br />
الثاني : أنَّا إن حملنا القَلِيل على الكُفَّار [ منهم الذين بَقُوا على الكُفْر ] ،   فالمعنى : أنَّ الله تعالى أمر رسوله بالعَفْوِ عنهم في صَغَائِر زَلاَّتِهِم ما داموا بَاقِين على العهد ،   وهو قَوْل أبِي مُسْلِم{[11250]} .  <br />
ثم قال تعالى : { إِنَّ اللَّهَ يُحِبُّ الْمُحْسِنِينَ } .  <br />
قال ابنُ عبَّاسٍ : إذا عَفَوْت فأنْتَ مُحْسِنٌ ،   وإذا كنْتَ مُحْسناً فقد أحبَّك اللَّه{[11251]} .  <br />
وقيل : المراد بهؤلاء المُحْسنين : هم المَعْنيّون بقوله تعالى : { إِلاَّ قَلِيلاً مِّنْهُمْ } وهمُ الذين ما نَقَضُوا الْعَهْد .
