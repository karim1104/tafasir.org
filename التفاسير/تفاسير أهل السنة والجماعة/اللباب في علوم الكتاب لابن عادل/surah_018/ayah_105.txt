قوله تعالى : { أولئك الذين كَفَرُواْ بِآيَاتِ رَبِّهِمْ وَلِقَائِهِ } .  <br />
لقاء الله عبارة عن رؤيته ؛ لأنَّه يقال : لقيتُ فلاناً ،   أي : رأيته .  <br />
فإن قيل : اللُّقيا عبارةٌ عن الوصول ؛ قال الله تعالى : { فَالْتَقَى المآء على أَمْرٍ قَدْ قُدِرَ } [ القمر : 12 ] .  <br />
وذلك في حقِّ الله محالٌ ؛ فوجب حمله على ثواب الله .  <br />
فالجواب : أن لفظ اللقاء ،   وإن كان في الأصل عبارة عن الوصول{[21351]} إلاَّ أنَّ استعماله في الرؤية مجازٌ ظاهرٌ مشهورٌ ،   ومن قال بأنَّ المراد منه : لقاء ثواب الله ،   فذلك لا يتمُّ إلا بالإضمار ،   وحمل اللفظ على المجاز المتعارفِ المشهور أولى من حمله على ما يحتاج إلى الإضمار .  <br />
واستدلَّت المعتزلة بقوله تعالى : { فَحَبِطَتْ أَعْمَالُهُمْ } على أن الإحباط حقٌّ ،   وقد تقدَّم ذلك في البقرة وقرأ ابن عباس{[21352]}  "  فَحبَطَتْ  "  بفتح الباء والعامة بكسرها .  <br />
قوله : { فَلاَ نُقِيمُ لَهُمْ يَوْمَ القيامة وَزْناً } .  <br />
قرأ العامة  "  نُقِيمُ  "  بنون العظمة ،   من  "  أقَامَ  "  ومجاهدٌ وعبيد بن عميرٍ :  "  فَلا يُقِيمُ  "  بياءِ الغيبة ،   لتقدُّم قوله :  "  بآيَاتِ ربِّهِمْ  "  فالضمير يعود عليه ،   ومجاهدٌ أيضاً  "  فلا يقُومُ لَهُمْ  "  مضارع  "  قَامَ  "  متعدٍّ ،   كذا قال أبو حيَّان ،   والأحسن من هذا : أن تعرب هذه القراءة على ما قاله أبو البقاء{[21353]} : أن يجعل فاعل  "  يَقومُ  "   "  صنيعُهمْ  "  أو  "  سَعْيهُم  "  وينتصب حينئذٍ  "  وزْناً  "  على أحد وجهين : إمَّا على الحال ،   وإمَّا على التَّمييز .  <br />
فصل في معنى الآية<br />
المعنى : أنَّا نزدري بهم ،   وليس لهم عندنا وزنٌ ومقدارٌ ،   تقول العرب : ما لفلانٍ عندي وزنٌ ،   أي : قدرٌ ؛ لخسَّتهِ ،   وروى أبو هريرة عن النبي صلى الله عليه وسلم قال :  "  إنَّهُ ليَأتِي يَومَ القيامةِ الرَّجلُ العَظيمُ السَّمينُ ،   فلا يَزنُ عند الله جَناحَ بَعُوضةٍ  "  قال :  " اقرءوا : فلا نقيم لهم يوم القيامة وزنا "  .  <br />
وقيل : المعنى : فلا نقيم لهم يوم القيامة ميزانا ؛ لأن الميزان إنما يوضع ؛ لأجل الحسنات والسيئات من الموحدين ؛ لتمييز مقدار الطاعات ،   ومقدار السيئات .  <br />
قال أبو سعيد الخدري{[21354]} : يأتي ناس بأعمال يوم القيامة من عندهم في التعظيم كجبال تهامة ،   فإذا وزنوها ،   لم تزن شيئا ،   فذلك قوله : { فلا نقيم يوم القيامة وزنا } .  <br />
وقال القاضي{[21355]} : إن من غلبت معاصيه ،   صارت طاعاته{[21356]} كأن لم تكن ،   فلم يدخل في الوزن شيء من طاعته ،   وهذا التفسير بناه على قول الإحباط والتكفير .  <br />
قوله : { ذلك جزاؤهم جهنم } : فيه أوجه كثيرة :<br />
الأول : أن يكون  " ذلك "  خبر مبتدأ محذوف ،   أي : الأمر ذلك والمعنى : الذي ذكرتُ من حبوط أعمالهم وخسَّة أقدارهم و  " جزاؤهم جهنم "  جملة برأسها .  <br />
الثاني : أن يكون  " ذلك "  مبتدأ أول ،   وجزاؤهم "  مبتدأ ثان ،   و " جهنم "  خبره ،   و هو [ وخبره "  ] خبر الأول ،   والعائد محذوف ،   أي : جزاؤهم به ،   كذا قال أبو البقاء{[21357]} .   ،   فالهاء في  " به "  تعود على  " ذلك "  و " ذلك "  مشار به إلى عدم إقامة الوزن .  <br />
قال أبو حيان :  " ويحتاج هذا التوجيه إلى نظر "  قال شهاب الدين : إن عنى النظر من حيث الصناعة ،   فمسلَّم ،   ووجه النظر : أن العائد حذف من  غير مسوغ إلا بتكلف ؛ فإن العائد على المبتدأ ،   إذا كان مجرورا ،   لا يحذف إلا إذا جر بحرف تبعيض ،   أو ظرفية ،   أو يجر عائدا جر قبله بحرف ،   جر به المحذوف ،   كقوله : [ الطويل ] .  <br />

أي : مفلح به ،   وإن عنى من حيث المعنى ،   فهو معنى جيد .  <br />
الثالث : أن يكون  " ذلك "  مبتدأ ،   و  " جزاؤهم "  بدل ،   أو بيان ،   و  " جهنم "  خبره .  <br />
الرابع : أن يكون  " ذلك "  مبتدأ أيضا ،   و " جزاؤهم "  خبره ،   و  " جهنم "  بدل ،   أو بيان ،   أو خبر ابتداء مضمر .  <br />
الخامس : أن يجعل  " ذلك "  مبتدأ ،   و  " جزاؤهم "  بدل ،   أو بيان ،   و  " جهنم "  خبر ابتداء مضمر ،   و  " بما كفروا "  خبر الأول ،   والجملة اعتراض .  <br />
السادس : أن يكون  " ذلك "  مبتدأ ،   والجار : الخبر ،   و " جزاؤهم جهنم "  جملة معترضة ،   وفيه بعدٌ .  <br />
السابع : أن يكون  " ذلك "  إشارة إلى جماعة ،   وهم مذكورون في قوله :  " بالأخسرين "  ،   وأشير إلى الجمع ؛ كإشارة الواحد ؛ كأنه قيل : أولئك جزاؤهم جهنم ،   والإعراب المتقدم يعود على هذا التقدير .  <br />
ومعنى الكلام : أن ذلك الجزاء جزاء على مجموع أمرين : كفرهم ،   واتخاذهم آيات الله ورسله هزوا ،   فلم يقتصروا على الرد عليهم وتكذيبهم ،   حتى استهزؤوا بهم .  <br />
قوله :  " واتخذوا "  فيه وجهان :<br />
أحدهما : أنه عطف على  " كفروا "  فيكون محله الرفع ،   لعطفه على خبر  " إن "  .  <br />
الثاني : أنه مستأنف ،   فلا محل له ،   والباء في قوله :  " بما كفروا "  لا يجوز تعلقها ب  " جزاؤهم "  للفصل بين المصدر ومعموله .
