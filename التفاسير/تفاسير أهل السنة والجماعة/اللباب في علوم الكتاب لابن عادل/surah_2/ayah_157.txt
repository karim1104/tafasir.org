قولهُ تعالى :  "  أُولَئِكَ  "  مبتدأٌ ،   و  "  صَلَوَاتٌ  "  مبتدأٌ ثان ،   و  "  عَلَيْهِمْ  "  خبرهُ مُقَدَّمٌ عليه ،   والجملةُ خبر قوله :  "  أُولَئِكَ  "  .  <br />
ويجوز أن تكون  "  صلوات  "  فاعلاً بقوله :  "  عليهم  "  .  <br />
قال أبو البقاء : لأنه قد قوي بوقوعه خبراً .  <br />
والجملة من قوله  "  أولئك  "  وما بعده خبر  "  الذين  "  على أحد الأوجه المتقدمة ،   أو لا محلّ لها على غيره من الأوجه .  <br />
و  "  قالوا  "  هو العامل في  "  إذا  "  ؛ لأنه جوابها وتقدم الكلام في ذلك وأنها هل تقتضي التكرار أم لا ؟<br />
قولهُ تعالى :  "  وَرَحْمَةٌ  "  عطف على الصلاة ،   وإن كانت بمعناها ،   فإن الصلاة من الله رحمة ؛ لاختلاف اللفظين كقوله : [ الوافر ]<br />
848 - وَقَدَّمَتِ الأَدِيمَ لِرَاهِشَيْهِ*** وَأَلْفَى قَوْلَها كَذِباً وَمَيْنَا{[11]}<br />
وقوله : [ الطويل ]<br />
849 - أَلاَ حَبَّذَا هِندُ وَأَرْضٌ بِهَا هِنْدٌ *** وَهِنْدٌ أَتَى مِنْ دُونِهَا النَّأْيُ وَالْبُعْدُ{[12]}<br />
قولُه تعالى :  "  مِنْ رَبِّهِمْ  "  فيه وَجْهَانِ :<br />
أَحدُهما : أنه متعلق بمحذوف ؛ لأنه صَفةٌ ل  "  صلوات  "  و  "  من  "  للابتداءِ ،   فهو في مَحَلِّ رفع ،   أيْ : صلوات كائنة مِنْ رَبِّهم .  <br />
والثَّانِي : أنه يتعلق بما تضمنه قولُه  "  عَلَيْهِمْ  "  من الفعل إذَا جعلناه رَافعاً ل  "  صلوات  "  رفع الفاعل ،   فعلى الأول ،   يكون قد حذف الصفة بعد  "  رَحْمَة  "  أَيْ : ورحمة منه .  <br />
وعلى الثَّانِي : لاَ يَحْتَاجُ إلى ذلك .  <br />
وقولُه :  "  وأُولَئِكُ هُمْ الْمُهْتَدُونَ  "  نَظيرُ : { وَأُوْلَئِكَ هُمُ الْمُفْلِحُونَ }<br />
[ البقرة : 5 ] وفيه وجوهٌ :<br />
أَحَدُهَا : أنهم هم المهتدون لهذه الطَّرِيقَةِ المُوصّلَةِ بصاحبها إلى كل خير .  <br />
وثَانِيهَا : المُهْتدُونَ إلى الجنَّةِ الفائزون بالثواب .  <br />
وثَالِثُها : المُهْتدُونَ لسائِر ما لزمهم .  <br />
فَصْلٌ في الكلام في الآية<br />
قال أَبُو الْبَقَاءِ :  "  هُمُ المُهْتَدُونَ  "  هُمْ : مُبْتَدأٌ أو توكيد أو فصل .  <br />
فإن قِيلَ : لِمَ أَفْرَدَ الرحْمَةَ وجَمَعَ الصَّلَواتِ ؟<br />
فالجوابُ : قال بعضُهم : إن الرحمَةَ مصدرٌ بمعنى التعطُّف والتحنُّن ،   ولا يجمعُ و  "  التَّاءُ  "  فيها بمنزلتها في الملّة والمحبّةِ والرأْفَةِ ،   والرحمةُ ليست للتحذيرِ ،   بل مَنْزِلتُها في مرية وثمرة ،   فكما لا يُقالُ : رقات ولا خلات ولا رأفات ،   لا يُقال : رَحَمات ،   ودخول الجمعُ يُشْعرُ بالتحذِيرِ والتقييد بعده ،   والإفْرَادُ مُطْلقاً مِنْ غَيْر تَحْدِيدٍ ،   فالإفْرَادُ - هنا - أَكْملُ وأكثرُ مَعْنًى من الجمع ؛ لأنه زيد بمدلول المفرد أكثر مِنْ مدلولِ الجَمْعِ ،   ولهذا كان قولُه تَعَالى :<br />
{ فَلِلَّهِ الْحُجَّةُ الْبَالِغَةُ } [ الأنعام : 149 ] أَعَمَّ وأَتَمَّ مَعْنًى مِنْ أَنْ يُقالَ : لِلَّهِ الحُجَجُ البَوالِغُ ،   وكذا قولُه : { وَإِن تَعُدُّواْ نِعْمَةَ اللَّهِ لاَ تُحْصُوهَا }<br />
[ إبراهيم : 34 ] أتمُّ مَعْنًى مِنْ أنْ يُقالَ : وإنْ تَعَدُّوا نِعَمَ الله لا تُحْصُوها ،   وقولُه سُبْحَانَهُ وتَعَالَى : { رَبَّنَآ آتِنَا فِي الدُّنْيَا حَسَنَةً }<br />
[ البقرة : 201 ] أتمُّ مَعْنًى مِنْ قوله : حَسَناتٍ ،   وقولُه : { بِنِعْمَةٍ مِّنَ اللَّهِ وَفَضْلٍ } [ آل عمران : 174 ] ،   أتَمُّ معنى من قوله : بنعمٍ ،   ونظائِرهُ كَثِيرةٌ .  <br />
وأما الصّلوات فالمراد بها درجات الثَّوَاب ،   وهي إنما تحصل شيئاً بَعْدَ شَيْءٍ ،   فكأنه دلّ على الصِّفَةِ المقصوُدَةِ .
