واعلم أنه  تعالى  لما وصف اليهود بالعِنَادِ ،   وأزال الطمع عن إيمانهم بين فرقهم ،   فالفرقة الأولى هي الضَّالة المضلّة ،   وهم الذين يحرّفون الكلم عن مواضعه .  <br />
والفرقة الثانية : المنافقون .  <br />
والفرقة الثالثة : الذين يُجَادلون المنافقين .  <br />
والفرقة الرابعة : هم المذكورون هنا ،   وهم العامة الأمّيون الذين لا يعرفون القراءة ،   ولا الكتابة ،   وطريقتهم التقليد وقَبُول ما يقال لهم ،   فبين الله  تعالى  أن المُمْتَنِعِينَ عن الإيمان ليس سبب ذلك واحداً ،   بل لكل قسم سبب .  <br />
وقال بعضهم : هم بعض اليهود والمنافقين .  <br />
وقال عكرمة والضَّحاك :  " هم نصارى العرب "  .  <br />
وقيل :  " هم قوم من أهل الكتاب رفع كتابهم لذنوب ارتكبوها ،   فصاروا آمنين "  .   وعن علي  رضي الله عنه  هم المجوس .  <br />
قوله :  " مِنْهُمْ "  خبر مقدم ،   فيتعلّق بمحذوف .   و  " أمّيون "  مبتدأ مؤخر ،   ويجوز على رأي الأخفش أن يكون فاعلاً بالظرف قبله ،   وإن لم يعتمد .  <br />
وقد بنيت على ماذا يعتمد فيما تقدم .  <br />
و  " أُمِّيُّونَ "  جمع  " أمّي "  وهو من لا يَكْتب ولا يَقْرأ .  <br />
واختلف في نسبته فقيل : إلى  " الأمّ "  وفيه معنيان :<br />
أحدهما : أنه بحال أُمّه التي ولدته من عدم معرفة الكتابة ،   وليس مثل أبيه ؛ لأن النساء ليس من شُغْلهن الكتابة .  <br />
والثاني : أنه بحاله التي ولدته أمه عليها لم يتغير عنها ،   ولم ينتقل .  <br />
وقيل : نسب إلى  " الأُمَّة "  وهي القَامَة والخِلْقَة ،   بمعنى أنه ليس له من النَّاس إلا ذلك .  <br />
وقيل : نسب إلى  " الأُمَّة "  على سَذَاجتها قبل أن يَعْرِف الأشياء ،   كقولهم : عامي أي : على عادة العامة .  <br />
وعن ابن عَبَّاس :  " قيل لهم : أميون ؛ لأنهم لم يصدقوا بأم الكتاب "  .  <br />
وقال أبو عبيدة :  " قيل لهم : أُميون ،   لإنْزَال الكتاب عليهم ،   كأنهم نسبوا لأم الكتاب "  .  <br />
وقرأ ابن أبي عَبْلة{[1410]} :  " أُمِّيُون "  بتخفيف الياء كأنه استثقل توالي تضعيفين .  <br />
وقيل : الأمي : من لا يُقِرّ بكتاب ولا رسول .  <br />
قوله : { لاَ يَعْلَمُونَ } جملة فعلية في محلّ رفع صفة ب  " أميون "  ،   كأنه قيل : أميون غير عالمين .  <br />
قوله : { ِلاَّ أَمَانِيَّ } هذا استثناء منقطع ؛ لأن  " الأماني "  ليست من جنس  " الكتاب "  ،   ولا مندرجة تحت مدلوله ،   وهذا هو المُنْقطع ،   ولكن شرطه أن يتوهّم دخوله بوجه ما ،   كقوله : { مَا لَهُمْ بِهِ مِنْ عِلْمٍ إِلاَّ اتِّبَاعَ الظَّنِّ } [ النساء : 157 ] وقول النَّابغة : [ الطويل ]<br />
601- حَلَفْتُ يَمِيناً غَيْرَ ذِي مَثْنَوِيَّةٍ *** وَلاَ عِلْمَ إِلاَّ حُسْنَ ظَنٍّ بِصَاحِبِ{[1411]}<br />
لأن بذكر العلم اسْتُحضِر الظن ،   ولهذا لا يجوز : صَهَلَتِ الخيل إلاَّ حماراً .  <br />
واعلم أن المنقطع على ضربين : ضرب يصحّ توجه العامل عليه ،   نحو : جاء القوم إلاّ حماراً .   وضرب لا يتوجه نحو ما مثل به النحاة :  " ما زاد إلا ما نقص "  ،   و  " ما نفع إلا ما ضر "  فالأول فيه لُغتان : لغة  " الحجاز "  وجوب نصبه ،   ولغة  " تميم "  أنه كالمتّصل ،   فيجوز فيه بعد النفي وشبهه النصب والإتباع .  <br />
والآية الكريمة من الضرب الأول ،   فيحتمل في نصبها وَجْهان :<br />
أحدهما : على الاستثناء المنقطع .  <br />
والثاني : أنه بدل من  " الكتاب "  .  <br />
و  " إلاّ "  في المنقطع تقدر عند البصريين ب  " لَكِنَّ "  ،   وعند الكونفيين ب  " بل "  ،   وظاهر كلام أبي البقاء أن نصبه على المصدر بفعل مَحْذُوف ،   فإنه قال :  " إلا أماني "  استثناء منقطع ؛ لأن الأماني ليس من جنس العلم ،   وتقدير  " إلاّ "  في مثل هذا ب  " لكن "  أي : لكن يتمنونه أماني ،   فيكون عنده من باب الاستثناء المفرّغ المنقطع ،   فيصير نظير :  " ما علمت إلا ظنّاً "  .   [ وفيه نظر ] .  <br />
الأَمَاني جمع  " أُمْنِيَّة "  بتشديد الياء فيها .  <br />
وقال أبو البَقَاءِ :  " يجوز تخفيفها فيها "  .  <br />
وقرأ أبو جعفر{[1412]} بتخفيفها ،   حَذَفَ إحدى الياءين تخفيفاً .  <br />
قال الأخفش :  " هذا كما يقال في جمع مفتاح : مفاتح ومفاتيح "  .  <br />
قال النَّحَّاس :  " الحذف في المعتلّ أكثر "  ؛ وأنشد قول النابغة : [ الطويل ]<br />
602- وَهَلْ يُرْجِعُ التَّسْلِيمَ أو يَكْشِفُ العَمَى *** ثَلاثُ الأَثَافِي والرُّسُومُ البَلاَقِعُ{[1413]}<br />
وقال أبو حَاتِمٍ :  " كلّ ما جاء واحده مشدداً من هذا النوع فلك في الجمع الوجهان "  .   وأصله يرجع إلى ما قال الأخفش .  <br />
ووزن  " أمنية "  :  " أُفْعُولَة "  من تَمَنَّى{[1414]} يَتَمَنَّى : إذا تلا وقرأ ؛ قال : [ الطويل ]<br />
603- تَمَنَّى كِتَابَ اللَّهِ آخِرَ لَيْلِهِ *** تَمَنِّيَ دَاوُدَ الزَّبُورَ عَلَى رِسْلِ{[1415]}<br />
وقال كعب بن مالك : [ الطويل ]<br />
604- تَمَنَّى كِتَابَ اللَّهِ أَوَّلَ لَيْلِهِ *** وآخِرَهُ لاَقَى حِمَامَ المقَادِرِ{[1416]}<br />
وقال تعالى : { إِلاَّ إِذَا تَمَنَّى أَلْقَى الشَّيْطَانُ فِي أُمْنِيَّتِهِ } [ الحج : 52 ] أي : قرأ وتلا .  <br />
والأصل على هذا :  " أُمْنُويَة "  فأعلت إعلال  " ميت "  و  " سيد "  وقد تقدم .  <br />
وقيل : الأمنية : الكذب والاختلاق .  <br />
قال أعرابي لابن دَأْب{[1417]} في شيء يحدث به : أهذا الشيء رويته أم تمنيته أي : اختلقته .  <br />
وقيل : ما يتمنّاه الإنسان ويشتهيه .  <br />
وقيل : ما يقدره ويحزره من مَنَّى : إذا كذب ،   أو تمنى ،   أو قدر ؛ كقوله : [ البسيط ]<br />
605- لاَ تَأْمَنَنَّ وَإِنْ أَمْسَيْتَ في حَرَمٍ *** حَتَّى تُلاَقِيَ مَا يَمْنِي لَكَ المَانِي{[1418]}<br />
أي : يقدر لك المقدر .  <br />
قال الراغب : والمني : التقدير ،   ومنه  " المَنَا "  الذي يُوزَن به ،   ومنه  " المنية "  وهو الأجل المقدر للحيوان ،    " والتَّمَنِّي "  : تقدير شيء في النفس وتصويره فيها ،   وذلك قد يكون عن ظنّ وتخمين ،   وقد يكون بناء على رَوِيَّةٍ وأصل ،   لكن لما كان أكثره عن تخمين كان الكذب أملك له ،   فأكثر التمني تصوّر ما لا حقيقة له [ والأمنية : الصورة الحاصلة في النفس من تمني الشيء ]{[1419]} .   ولما كان الكذب تصوّر ما لا حقيقة له وإيراده باللفظ الدَّال صار التمني كالمبدأ للكذب فعبر عنه .  <br />
ومنه قوله عثمان :  " ما تغنيت ولا تمنيت منذ أسلمت "  .  <br />
وقال الزمخشري{[1420]} : والاشتقاق من منَّى : إذا قدَّر ؛ لأن المتمني يُقَدِّرُ في نفسه ،   ويَحْزر ما يتمناه ،   وكذلك المختلق ،   والقارئ يقدر أن كلمة كذا بعد كذا ،   فجعل بين هذه المعاني قدراً مشتركاً وهو واضح .  <br />
وقال أبو مسلم : حَمْله على تمني القلب أولى بقوله تعالى : { وَقَالُواْ لَن يَدْخُلَ الْجَنَّةَ إِلاَّ مَن كَانَ هُوداً أَوْ نَصَارَى تِلْكَ أَمَانِيُّهُمْ } [ البقرة : 111 ] .  <br />
وقال الأكثرون : حمله على القراءة أليق ؛ لأنا إذا حَمَلْنَاه على ذلك كان له به تعلّق ،   فكأنه قال : لا يعلمون الكتاب إلاَّ بقدر ما يتلى عليهم فيسمعونه ،   وبقدر ما يذكر لهم فيقبلونه ،   ثم إنَّهُمْ لا يتمكّنون من التدبُّر والتأمل ،   وإذا حمل على أن المراد الأحاديث والأكاذيب أو الظن والتقدير وحديث النفس كان الاستثناء فيه نادراً .  <br />
قوله : { وَإِنْ هُمْ إِلاَّ يَظُنُّونَ }  " إن "  نافية بمعنى  " ما "  وإذا كانت نافية المشهور أنها لا تعمل عمل  " ما "  الحِجَازية .  <br />
وأجاز بعضهم ذلك ونسبه لسيبويه ،   وأنشد : [ المنسرح ]<br />
606- إنْ هُوَ مُسْتَوْلِياً عَلَى أَحَدٍ *** إلاَّ عَلَى أَضْعَفِ المَجَانِينِ{[1421]}<br />
 " هو "  اسمها ،   و  " مستولياً "  خبرها .  <br />
