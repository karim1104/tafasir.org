قوله :  " قَالُوا : ادْعُ لَنَا رَبَّكَ يُبَيِّنْ "  ف  " يبيّن "  مجزوم على جواب الأمر كقوله : { فَادْعُ لَنَا رَبَّكَ يُخْرِجْ لَنَا } [ البقرة : 61 ] .  <br />
قوله :  " مَا هِيَ "  ،    " ما "  استفهامية في محلّ رفع بالابتداء ،   تقديره : أي شيء هي ؟ وما [ الاستفهامية ]{[12]} يطلب بها شرح الاسم تارة ،   نحو : ما العَنْقَاء ؟ وماهية المسمى أخرى ،   نحو : ما الحركة ؟ .  <br />
وقال السكاكي{[13]} :  " يسأل ب "  ما  " عن الجنس ،   تقول : ما عندك ؟ أي : أيّ أجناس الأشياء عندك ؟ وجوابه "  كتاب  " ونحوه ،   أو عن الوصف ،   تقول : ما زيد ؟ وجوابه : "  كريم  "  ،   وهذا هو المراد في الآية .  <br />
و "  هي  " ضمير مرفوع منفصل في محلّ رفع خبراً ل "  ما  " والجملة في محلّ نصب ب "  يبيّن  "  ؛ لأنه معلّق عن الجملة بعده ،   وجاز ذلك ،   لأنه شبيه بأفعال القلوب .  <br />
قوله : "  لاَ فَارِضٌ وَلاَ بِكْرٌ  " لا "  نافية  "  ،   و "  فارض  " صفة ل "  بقرة  "  .  <br />
واعترض ب "  لا  " بين الصفة والموصوف ،   نحو : "  مررت برجل لا طويل ولا قصير  "  .  <br />
وأجاز أبو البقاء : أن يكون خبر المبتدأ محذوف ،   أي : لا هي فَارِضٌ .  <br />
وقوله : "  ولا بِكْر  " مثل ما تقدّم .  <br />
وتكررت "  لا  " لأنها متى وقعت قبل خبر ،   أو نعت ،   أَو حال وجب تكريرها تقول : "  زيد لا قائم ولا قاعد  "  ،   و "  مررت به لا ضحاكاً ولا باكياً  "  .  <br />
ولا يجوز عدم التكرار إلا في ضرورة خلافاً للمبرِّد وابن كَيْسَان ؛ فمن ذلك : [ الطويل ]<br />
568 وَأَنْتَ امْرُؤٌ مِنَّا خُلِقُتَ لِغَيْرنا *** حَيَاتُكَ لاَ نَفْعٌ ومَوْتُكَ فَاجِعُ<br />
وقوله : [ الطويل ]<br />
569 قَهَرْتُ الْعِدَا لاَ مُسْتَعِيناً بِعُصْبَةٍ *** وَلَكِنْ بَأَنْواعِ الْخَدائِعِ وَالمَكْرِ<br />
فلم يكررها في الخبر ،   ولا في الحال .  <br />
و "  الفارض  "  : المُسِنّة الهَرِمَة ،   قال الزمخشريُّ : كأنها سميْت بذلك ؛ لأنها فَرَضَتْ سِنَّها ،   أي : قَطَعَتْهَا وبلغت آخرها ؛ قال : [ الطويل ]<br />
570 لَعَمْرِي لَقَدْ أَعْطَيْتَ ضَيْفَكَ فَارِضاً *** تُسَاقُ إِلَيْهِ مَا تَقُومُ عَلَى رِجْلِ<br />
ويقال لكل ما قَدُمَ : فارض ؛ قال [ الرجز ]<br />
571 شَيَّبَ أَصْدَاعِيْ فَرَأْسِي أَبْيَضُ *** مَحَامِلٌ فِيهَا رِجَالٌ فُرَّضُ<br />
أي : كبار قدماء .  <br />
وقال آخر : [ الرجز ]<br />
572 يَا رُبَّ ذِي ضِغْنٍ عَلَيَّ فَارِضِ *** لَهُ قُرُوءٌ كَقُروءِ الْحَائِضِ<br />
وقال الرَّاغب : سميت فارضاً ؛ لأنها تقطع الأرض ،   والْفَرْضُ في الأصل القطع ،   وقيل : لأنها تحمل الأعمال الشاقة .  <br />
وقيل : لأن فريضة البقر تَبِيعٌ ومُسِنَّةٌ ،   قال : فعلى هذا تكون الفارض اسماً إسلاميّاً .  <br />
وقيل : "  الفَارِضُ  "  : التي ولدت بطوناً كثيرة فيتسّع جوفها لذلك ؛ لأن معنى الفارض في اللغة الواسع ،   نقله القرطبي عن بعض المتأخرين .   ويقال : فَرَضَتْ  تَفْرِضُ  بالفتح  فُرُوضاً .  <br />
وقيل : فَرُضَتْ  بالضم  أيضاً .  <br />
وقال المفضَّل بن سَلَمَةَ : الفارض : المُسِنَّة .   و "  البِكْر  "  : ما لم تحمل .  <br />
وقيل : ما ولَدَتْ بطناً واحداً ،   وذلك الولد بِكْرٌ أيضاً .   قال : الرجز ]<br />
573 يَا بِكْرَ بَكْرَين وَيَا خِلْبَ الْكَبِدْ *** أَصْبَحْتَ مِنِّي كِذِرَاع مِنْ عَضُدْ<br />
و "  البِكْر  " من الحيوان : من لم يطرقه فحل ،   و "  البَكْر  " - بالفتح - الْفَتِيُّ من الإِبِل ،   والبَكَارة  بالفتح  المصدر .  <br />
وقال المفضَّل بن سلمة : البِكْر : الشابة .  <br />
قال القفال - رحمه الله تعالى  : اشتقاق البكر يدل على الأول ،   ومنه الباكورة لأول الثمرة ،   ومنه : بُكْرَة النهار ،   ويقال : بكرت عليه البارحة ،   إذا جاء في أول الليل .  <br />
والأظهر أنها هي التي لم تلد ؛ لأن المعروف من اسم البكر من إناث بني آدم : ما لم يَنْزُ عليها الفحل .  <br />
قوله : عَوَانٌ صفة ل "  بقرة  "  ،   ويجوز أن يكون خبر المبتدأ محذوف أي : هي عَوانٌ كما تقدم في { لاَّ فَارِضٌ } [ البقرة : 68 ] والعوان : النَّصَف ،   وهو التوسُّط بين الشيئين ،   وذلك أقوى ما يكون وأحسنه ؛ قال : [ الوافر ]<br />
574- .   .   .   *** نَوَاعِمُ بَيْنَ أَبْكَارِ وَعُون<br />
وقال الشاعر يصف فرساً : [ الطويل ]<br />
575 كُمَيْتِ بَهِيمِ اللَّونِ لَيْسَ بِفَارِضٍ *** وَلاَ بِعَوانٍ ذَاتِ لَوْنٍ مُخَصَّفِ<br />
 "  فرس أخصف  " إذا ارتفعَ الْبَلَق من بطنه إلى جنبه ،   ويقال للنخلة الطويلة : عوان ،   وهي فيما زعموا لغة عَانِيَة ،   حكاه القرطبي .  <br />
وقيل : هي التي ولدت مرة بعد أخرى ومنه "  الحرب العوان  " أي التي جاءت بعد حرب أخرى ؛ قال زهير : [ الطويل ]<br />
576 إِذَا لَحِقَتْ حَرْبٌ عَوَانٌ مُضِرَّةٌ *** ضَرُوسٌ تُهِرُّ النَّاسَ أَنْيَابُهَا عُصْلُ<br />
والعُوْن  بسكون الواو  الجمع ،   وقد بضم ضرورة كقوله : [ السريع ]<br />
577- .   .   .   *** في الأَكُفِّ اللاَّمِعَاتِ سُوُرْ<br />
بضم الواو .   ونظيره في الصحيح "  قَذال وقُذُل  " و "  حِمَار وحُمُر  "  .  <br />
قوله : "  بين ذلك  " صفة ل "  عوان  " [ فهي ] في محلّ رفع ،   ويتعلق بمحذوف ،   أي : كائن بين ذلك ،   و "  بين  " إنما تضاف لشيئين فصاعداً ،   وجاز أن تضاف  هنا  إلى مفرد ؛ لأنه يشار به إلى المُثَنَّى والمجموع ؛ كقوله : [ الرمل ]<br />
578- إِنَّ لِلْخَيْرِ وَلِلشَّرِّ مَدًى *** وَكِلاَ ذَلِكَ وَجْهٌ وَقَبَلْ<br />
كأنه قيل : بين ما ذكر من الفارض والبكر .   قال الزمخشري :<br />
فإن قلت : كيف جاز أن يشار به إلى مؤنثين ،   وإنما هو لإشارة المذكر ؟<br />
قلت : لأنه في تأويل ما ذكر وما تقدم ،   وقال : وقد يجرى الضمير مجرى اسم الإشارة في هذا ،   قال أبو عبيدة : قلت لرؤبة في قوله : [ الرجز ]<br />
579- فِيهَا خُطُوطٌ مِنْ سَوَادٍ وَبَلَقْ *** كَأَنَّهُ فِي الْجِلْدِ تَولِيعُ الْبَهَقْ<br />
إن أردت الخطوط فقل : كأنها ،   وإن أردت السواد والبَلَق فقل : كأنهما ،   فقال : أردتُ : ذَاكَ وذَلِكَ .   والذي حسن منه أن أسماء الإشارة تثنيتها ،   وجمعها ،   وتأنيثها ليست على الحقيقة ،   وكذلك الموصولات ،   ولذلك جاء "  الذي  " بمعنى الجمع واحتج بعض العلماء بقوله : "  عَوَانٌ بَيْنَ ذَلِكَ  " على جواز الاجتهاد ،   واستعمال غلبة النَّص في الأحكام ،   إذ لا يعلم أنها بين الفَارِض والبِكْر إلا من طريق الاجتهاد .  <br />
قوله : { مَا تُؤْمَرونَ } "  ما  " موصولة بمعنى "  الذي  "  ،   والعائد محذوف تقديره : تؤمرون به ،   فحذفت الباء ،   وهو حذف مطرد ،   فاتصل الضمير فحذفت "  الهاء  "  ،   وليس هو نظير : { كَالَّذِي خَاضُواْ } [ التوبة : 69 ] فإن الحذف  هناك  غير مقيس .  <br />
ويضعف أن تكون "  ما  " نكرة موصوفة .  <br />
قال أبو البقاء : لأن المعنى على العموم ،   وهو ب "  الذي  " أشبه ،   ويجوز أن تكون مصدرية أي : أمركم بمعنى مأموركم ،   تسمية للمفعول بالمصدر ك "  ضَرْب الأمير  " قاله الزمخشري .  <br />
و "  تؤمرون  " مبني للمفعول ،   و "  الواو  " قائم مقام الفاعل ،   ولا محلّ لهذه الجملة لوقوعها صلة .  <br />
فصل في الغاية من وصف البقرة<br />
والمقصود كون البقرة في أكمل أحوالها ،   وذلك لأن الصَّغيرة تكون ناقصة ؛ لأنها لم تصل إلى حالة الكمال ،   والمُسِنّة صارت ناقصةً ؛ لتجاوزها حَدّ الكمال ،   والمتوسط هو الذي يكون في حال الكمال .   قاله الثعلبي .
