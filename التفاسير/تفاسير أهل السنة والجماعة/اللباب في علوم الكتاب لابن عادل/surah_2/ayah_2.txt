قوله : { ذَلِكَ الْكِتَابُ } .  <br />
يجوز في  " ذلك "  أن تكون مبتدأ ثانياً ،   و " الكتاب "  خبره ،   والجملة خبر  " الم ،   وأغنى الربط باسم الإشَارة ،   ويجوز أن يكون  " الم "  مبتدأ .  <br />
و " ذلك "  خبره ،   و " الكتاب "  صفة ل  " ذلك "  ،   أو بدل منه ،   أو عطف بيان ،   وأن يكون  " الم "  مبتدأ ،   و " ذلك "  مبتدأ ثانٍ ،   و " الكتاب "  : إما صفة له ،   أو بدل منه ،   أو عطف بيان له .  <br />
و "  لا ريب فيه "  [ خبر ]{[452]} عن المبتدأ الثاني ،   وهو خبره خبر عن الأول .  <br />
ويجوز أن يكون  " الم "  خبر مبتدأه مضمر ،   تقديره :  " هذا الم "  ،   فتكون جملة مستقلة بنفسها ،   ويكون  " ذلك "  مبتدأ ثانياً ،   و " الكتاب "  خبره .  <br />
ويجوز أن يكون صفة له ،   أو بدلاً ،   أو بياناً ،   و " لا ريب فيه "  هو الخبر عن  " ذلك "  أو يكون  " الكتاب "  خبراً ل  " ذلك "  ،   و " لا ريب فيه "  خبر ثان ،   وفيه نظر من حيث إنه تعدّد الخبر ،   وأحدهما جملة ،   لكن الظاهر جوازه ؛ كقوله تعالى : { فَإِذَا هِيَ حَيَّةٌ تَسْعَى } [ طه : 20 ] ،   إذا قيل بأن  " تسعى "  خبر .  <br />
وأما إن جُعِلَ صفة فلا{[453]} .  <br />
و " ذلك "  اسم إشارة : الاسم منه  " ذا "  ،   و " اللاَّم "  للبعد ،   و " الكاف "  للخطاب ،   ولها ثلاث رُتَب :<br />
دُنْيَا : ولها المُجَرّد من اللام والكاف نحو :  " ذا وذي "  و " هذا وهذي "  .  <br />
وَوُسْطَى : ولها المتّصل بحرف الخطاب ،   نحو " ذاك وذيك وتيك "  .  <br />
وقُصْوَى : ولها المتّصل ب  " اللام "  و " الكاف "  نحو :  " ذلك وتلك "  .  <br />
ولا يجوز أن تأتي ب  " اللام "  إلاّ مع  " الكاف "  ،   ويجوز دخول حرف التَّنبيه على سائر أسماء الإشارة إلاّ مع  " اللاَّم "  ،   فيمتنع للطول .  <br />
وبعض النحويين لم يذكر إلاّ رتبتين : دُنْيَا وغيرها{[454]} .   واختلف النحويون في  " ذا "  هل هو ثلاثي الوضع أم أصله حرف واحد ؟<br />
الأول قول البصريين ،   ثم اختلفوا على عينه ولامه ياء ،   فيكون من باب  " حيي "  ،   أو غينه واو ولامه ياء فيكون من باب  " غويت "  ثم حذفت لامه تخفيفاً ،   وقلبت العين ألفاً لتحركها ،   وانفتاح ما قبلها ،   وهذا كله على سبيل التّمرين .  <br />
وأيًّا فهذا مبني ،   والمبني لا يدخله تصريف ،   وإنما جيء هنا بإشارة البعيد تعظيماً للمُشَار إليه ومنه : [ الطويل ]<br />
أَقُولُ لَهُ والرُّمْحُ يَأْطِرُ مَتْنَهُ *** تَأَمَّلْ خُفَافاً إِنَّنِي أنا ذَلِكَا{[455]}<br />
أو لأنه لما نزل من السَّماء إلى الأرض أشير إليه بإشارة البعيد .  <br />
أو لأنه كان موجوداً به بنبيّه عليه الصلاة والسلام .  <br />
أو أنه أشير به إلى ما قضاه وقدّره في اللَّوح المحفوظ .  <br />
وفي عبارة المفسرين أشير بذلك إلى الغائب يعنون البعيد ،   وإلا فالمشار إليه لا يكون إلا حاضراً ذِهْناً أو حِسّاً ،   فعبروا عن الحاضر ذِهْناً بالغائب أي حسّاً وتحريراً لقول ما ذكرته لك .   وقال الأصَمّ وابن كيْسَان{[456]} : إن الله - تعالى - أنزل قبل سورة  " البقرة "  سوراً كذب بها المشركون ،   ثم أنزل سورة  " البقرة "  فقال :  " ذلك الكتاب "  يعني ما تقدم  " البقرة "  من السور لا شك فيه .  <br />
قال ابن الخَطِيب رحمه الله تعالى : سلّمنا أن المُشَار إليه حاضر ،   لكن لا نسلّم أن لفظة  " ذلك "  لا يشار بها إلا إلى البعيد .  <br />
بيانه : أن  " ذلك "  و " هذا "  حرف إشارة ،   وأصلهما  " ذا "  لأنه حرف الإشارة ،   قال تعالى : { مَّن ذَا الَّذِي } [ البقرة : 245 ] .  <br />
ومعنى  " ها "  تنبيه ،   فإذا قرب الشيء أشير إليه فقيل : هذا ،   أي : تَنَبَّهْ أيّها المُخَاطب لما أشرت إليه ،   فإنه حاضر معك{[457]} بحيث تَرَاه ،   وقد تدخل  " الكاف "  على  " ذَا "  للمخاطبة ،   و " اللام "  لتأكيد معنى الإشارة ،   فقيل :  " ذلك "  ،   فكأن المتكلّم بالغ في التنبيه لتأخّر المُشَار إليه عنه ،   فهذا يدلّ على أن لفظة  " ذلك "  لا تفيد البُعْد في أصل الوَضْع ،   بل اختص في العُرْف بالإشارة إلى البعيد للقرينة التي ذكرناها ،   فصارت كالدابة ،   فإنها مختصة في العرف بالفرس ،   وإن كانت في أصل الوضع متناولة لكل ما يدبّ على الأرض .  <br />
وإذا ثبت هذا فنقول : إنا نحمله ها هنا على مقتضى الوضع اللّغوي ،   لا على مقتضى الوضع العرفي ،   وحينئذ لا يفيد البُعْد ،   ولأجل هذه المُقَارنة قام كلّ واحد من اللَّفْظَين مقام الآخر .  <br />
قال تعالى : { وَاذْكُرْ عِبَادَنَا إِبْرَاهِيمَ وَإِسْحَاقَ } إلى قوله : { وَكُلٌّ مِّنَ الأَخْيَارِ } [ ص : 45-48 ] ثم قال : { هَذَا ذِكْرٌ } [ ص : 49 ] وقال : { وَعِندَهُمْ قَاصِرَاتُ الطَّرْفِ أَتْرَابٌ هَذَا مَا تُوعَدُونَ } [ ص : 52-53 ] .  <br />
وقال : { وَجَاءَتْ سَكْرَةُ الْمَوْتِ بِالْحَقِّ ذَلِكَ مَا كُنتَ مِنْهُ تَحِيدُ } [ ق : 19 ] .  <br />
وقال تعالى : { فَأَخَذَهُ اللَّهُ نَكَالَ الآخِرَةِ وَالأُوْلَى إِنَّ فِي ذَلِكَ لَعِبْرَةً لِّمَن يَخْشَى } [ النازعات : 24-25 ] .  <br />
وقال تعالى : { وَلَقَدْ كَتَبْنَا فِي الزَّبُورِ } [ الأنبياء : 105 ] وقال تعالى : { إِنَّ فِي هَذَا لَبَلاَغاً } [ الأنبياء : 106 ] .  <br />
وقال : { فَقُلْنَا اضْرِبُوهُ بِبَعْضِهَا كَذَلِكَ يُحْيِي اللَّهُ الْمَوْتَى } [ البقرة : 73 ] [ أي هكذا يحيي الموتى ]{[458]} .  <br />
وقال : { وَمَا تِلْكَ بِيَمِينِكَ يا مُوسَى } [ طه : 18 ] أي ما هذه التي بيمينك .  <br />
و " الكتاب "  في الأصل مصدر ؛ قال تعالى : { كِتَابَ اللَّهِ عَلَيْكُمْ } [ النساء : 24 ] وقد يراد به المكتوب ،   قال الشاعر : [ الطويل ]<br />
بَشَرْتُ عِيَالِي إِذْ رأَيْتُ صَحِيفَةً *** أَتَتْكَ مِنَ الحَجَّاجِ يُتْلَى كِتَابُهَا{[459]}<br />
ومثله [ الوافر ]<br />
تُؤَمِّلُ رَجْعَةً مِنِّي وفِيهاَ *** كِتَابٌ مِثْلُ مَا لَصِقَ الْغِرَاءُ{[460]}<br />
وأصل هذه المادة الدّلالة على الجَمْعِ ،   ومنه كتيبة الجَيْش ،   وكَتَبْتُ القربة ،   وكَتَبْتُ القربة : خَرَزْتُها ،   وَالكُتَبة -بضم  " الكاف "  الخرزة ،   والجمع كَتَب ،   قال : [ البسيط ]<br />
وَفْرَاءَ غَرْفِيَّةٍ أَثْأَى خَوَارِزُهَا *** مُشَلْشِلٌ ضَيَّعَتْهُ بَيْنَهَا الْكُتَبُ{[461]}<br />
وكَتَبْتُ الدَّابة [ إذا جَمَعْتَ بين شُفْرَي رَحِمِها بِحَلْقَةٍ أَو سَيْرٍ ]{[462]} قال الشاعر : [ البسيط ]<br />
لاَ تَأْمَنَنَّ فَزَارِيَّا حَلَلْتَ بِهِ *** عَلَى قَلُوصِكَ واكتُبْهَا بِأسْيَارِ{[463]}<br />
والكتابة عرفاً ضمّ بعض حروف الهجاء إلى بعض .  <br />
قال ابن الخطيب :  " واتفقوا على أنَّ المراد من الكتاب القرآن ،   قال تبارك وتعالى :{ كِتَابٌ أَنزَلْنَاهُ إِلَيْكَ مُبَارَكٌ }<br />
[ ص : 29 ] .  <br />
والكتاب جاء في القرآن على وجوه :<br />
أحدها : الفرض { كُتِبَ عَلَيْكُمُ الْقِصَاصُ } [ البقرة : 178 ] ،   { كُتِبَ عَلَيْكُمُ الصِّيَامُ } [ البقرة : 183 ] ،   { إِنَّ الصَّلاَةَ كَانَتْ عَلَى الْمُؤْمِنِينَ كِتَاباً مَّوْقُوتاً } [ النساء : 103 ] .  <br />
ثانيها : الحُجَّة والبُرْهان : { فَأْتُواْ بِكِتَابِكُمْ إِن كُنتُمْ صَادِقِينَ } [ الصافات : 157 ] أي : بِبُرْهانكم وحجّتكم .  <br />
ثالثها : الأَجَل : { وَمَا أَهْلَكْنَا مِن قَرْيَةٍ إِلاَّ وَلَهَا كِتَابٌ مَّعْلُومٌ } [ الحجر : 4 ] أي : أَجَل .  <br />
رابعها : بمعي مُكَاتبة السيد عبده : { وَالَّذِينَ يَبْتَغُونَ الْكِتَابَ مِمَّا مَلَكَتْ أَيْمَانُكُمْ } [ النور : 33 ] وهذا المصدر  " فِعَال "  بمعنى  " المُفَاعلة "  كالجِدَال والخِصَام والقِتَال بمعنى : المُجَادلة والمُخَاصمة والمُقَاتلة .  <br />
والكتاب - هنا - المُرَاد به القرآن ،   وله أسماء :<br />
أحدها : الكِتَاب كما تقدم .  <br />
وثانيها : القُرْآن : { إِنَّا جَعَلْنَاهُ قُرْآناً } [ الزخرف : 3 ] ،   { شَهْرُ رَمَضَانَ الَّذِي أُنْزِلَ فِيهِ الْقُرْآنُ } [ البقرة : 185 ] .  <br />
وثالثها : الفُرْقَان : { تَبَارَكَ الَّذِي نَزَّلَ الْفُرْقَانَ } [ الفرقان : 1 ] .  <br />
ورابعها : الذِّكْر ،   والتَّذْكِرَة ،   والذِّكْرَى : { وَهَذَا ذِكْرٌ مُّبَارَكٌ أَنزَلْنَاهُ } [ الأنبياء : 50 ] ،   { وَإِنَّهُ لَتَذْكِرَةٌ لِّلْمُتَّقِينَ } [ الحاقة : 48 ] وقوله تعالى : { وَذَكِّرْ فَإِنَّ الذِّكْرَى تَنفَعُ الْمُؤْمِنِينَ } [ الذاريات : 55 ] .  <br />
وخامسها : التَّنْزِيل : { وَإِنَّهُ لَتَنزِيلُ رَبِّ الْعَالَمِينَ } [ الشعراء : 192 ] .  <br />
وسادسها : الحديث : { اللَّهُ نَزَّلَ أَحْسَنَ الْحَدِيثِ } [ الزمر : 23 ] .  <br />
وسابعها : المَوْعِظَة : { قَدْ جَاءَتْكُمْ مَّوْعِظَةٌ مِّن رَّبِّكُمْ } [ يونس : 57 ] .  <br />
وثامنها : الحُكْم ،   والحِكْمَة ،   والحَكِيم ،   والمُحْكَم : { وَكَذلِكَ أَنزَلْنَاهُ حُكْماً عَرَبِيّاً } [ الرعد : 37 ] ،   { حِكْمَةٌ بَالِغَةٌ } [ القمر : 5 ] ،   { يس وَالْقُرْآنِ الْحَكِيمِ } [ يس : 1-2 ] ،   { كِتَابٌ أُحْكِمَتْ آيَاتُهُ } [ هود : 1 ] .  <br />
وتاسعها : الشِّفَاء : { وَنُنَزِّلُ مِنَ الْقُرْآنِ مَا هُوَ شِفَاءٌ } [ الإسراء : 82 ] .  <br />
وعاشرها : الهُدَى ،   والهَادِي { هُدًى لِّلْمُتَّقِينَ } [ البقرة : 2 ] ،  <br />
{ إِنَّ هَذَا الْقُرْآنَ يهْدِي لِلَّتِي هِيَ أَقْوَمُ } [ الإسراء : 9 ] ،  <br />
{ قُرْآناً عَجَباً يَهْدِي إِلَى الرُّشْدِ } [ الجن : 1-2 ] .  <br />
وذكروا له أسماء أُخَر منها :<br />
 " الصِّرَاط المستقيم ،   والعِصْمَة ،   والرَّحْمَة ،   والرُّوح ،   والقَصَص ،   والبَيَان ،   والتِّبْيَان ،   والمُبِين ،   والبَصَائر ،   والفَصْلُ ،   والنُّجُوم ،   والمَثَاني ،   والنّعْمَة ،   والبُرْهَان ،   والبَشِير ،   والنَّذِير ،   والقَيِّم ،   والمُهَيْمِن ،   والنور ،   والحق ،   والعزيز ،   والكريم ،   والعظيم ،   والمبارك "  .  <br />
قوله تعالى : { لاَ رَيْبَ فِيهِ } .  <br />
يجوز أن يكون خبراً كما تقدّم بيانه .  <br />
قال بعضهم : هو خبر بمعنى النّهي ،   أي : لا ترتابوا فيه كقوله تعالى : { فَلاَ رَفَثَ وَلاَ فُسُوقَ } [ البقرة : 197 ] أي : لا ترفثوا ولا تفسقوا .  <br />
قرأ{[464]} ابن كثير :  " فِيهِ "  بالإشباع في الوَصْلِ ،   وكذلك كل هاء كناية قبلها ساكن يشبعها وصلاً ما لم يَلِهَا ساكن ،   ثم إن كان السَّاكن قبل الهاء ياء يشبعها بالكسر ياء ،   وإنْ كان غيرها يشبعها بالضم واواً ،   ووافقه حفص{[465]} في قوله : { فِيهِ مُهَاناً } [ الفرقان : 69 ] فأشبعه .  <br />
ويجوز أن تكون هذه الجملة في محلّ نصب على الحال ،   والعامل فيه معنى الإشارة ،   و " لا "  نافية للجنس محمولة في العمل على نقيضها .    " إِنَّ "  ،   واسمها معرب ومبني :<br />
فيبنى إذا كان مفرداً نكرة على ما كان ينصب به ،   وسبب بنائه تضمنّه معنى الحرف ،   وهو " من "  الاسْتِغْرَاقِيَّة يدلّ عليه ظهورها في قول الشاعر : [ الطويل ]<br />
فَقَامَ يَذُودُ النَّاس عَنْهَا بِسَيْفِهِ *** فَقَالَ أَلاَ لاَ مِنْ سَبِيلٍ إِلَى هِنْدِ{[466]}<br />
وقيل : بني لتركُّبه معها تركيب خمسةَ عَشَرَ ،   وهو فاسد وبيانه في غير هذا الكتاب .   وزعم الزَّجَّاج أن حركة  " لاَ رَجُلَ "  ونَحْوِه حركة إعراب ،   وإنما حذف التنوين تخفيفاً ،   ويدل على ذلك الرجوع إلى هذا الأصل في الضرورة كقوله : [ الوافر ]<br />
أَلاَ رَجُلاً جَزَاهُ اللهُ خَيْراً *** يَدُلُّ عَلَى مُحَصِّلَةٍ تَبِيتُ{[467]}<br />
ولا دليل له لأن التقدير : أَلاَ تَرَوْنَنِي رَجُلاً ؟<br />
فإن لم يكن مفرداً - وأعني به المضاف والشبيه به - أُعْرِبَ نَصْباً نحو :  " لا خيراً من زيد "  ،   ولا عمل لها في المعرفة ألبتة ،   وأما نحو قوله : [ الطويل ]<br />
تُبَكِّي عَلَى زيْدٍ وَلاَ زَيْدَ مِثْلُهُ *** بَرِيءٌ مِنَ الْحُمَّى سَلِيمُ الجَوَانِحِ{[468]}<br />
وقول الآخر : [ الوافر ]<br />
أَرَى الْحَاجَاتِ عِنْدَ أَبِي خُبَيْبٍ *** نَكِدْنَ وَلاَ أُمَيَّةَ فِي الْبِلاَدِ{[469]}<br />
وقول الآخر : [ الرجز ]<br />
لاَ هَيْثَمَ اللَّيْلَةَ لِلْمَطِيِّ{[470]} *** .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .  <br />
وقوله عليه الصَّلاة والسَّلام :  " لا قُرَيْشَ بعد اليَوْمِ ،   إذا هلك كِسْرَى ،   فلا كسرى بَعْدَه " {[471]} فمؤوَّل .  <br />
و " رَيْبَ "  اسمها ،   وخبرها يجوز أن يكون الجار والمجرور وهو " فيه "  ،   إلاَّ أن بني  " تميم "  لا تكاد تذكر خبرها ،   فالأولى أن يكون محذوفاً تقديره : لا ريب كائن ،   ويكون الوَقْفُ على  " ريب "  حينئذ تامًّا ،   وقد يحذف اسمها ويبقى خبرها ،   قالوا :  " لا عليك "  أي : لا بأس عليك .  <br />
ومذهب سيبويه رحمه الله : أنها واسمها في مَحَلّ رفع بالابتداء ،   ولا عمل لها في الخبر .  <br />
ومذهب الأخفش : أن اسمها في مَحَلّ رفع ،   وهي عاملة في الخبر .  <br />
ولها أحكامٌ كثيرة ،   وتقسيمات منتشرة مذكورة في كتب النحو{[472]} .  <br />
واعلم أن  " لا "  لفظ مُشْتَرَك بين النَّفي ،   وهي فيه على قسمين :<br />
قسم تنفي فيه الجنس فتعمل عمل إنَّ كما تقدم ،   وقسم تنفي فيه الوحدة ،   وتعمل حينئذ عمل  " ليس "  ،   ولها قسم آخر ،   وهو النهي والدُّعاء فتجزم فعلاً واحداً ،   وقد تجيء زيادة كما تقدم في قوله : { وَلاَ الضَّالِّينَ } [ الفاتحة : 7 ] .  <br />
و " الرَّيْب "  : الشّك مع تهمة ؛ قال في ذلك : [ الخفيف ]<br />
لَيْسَ فِي الْحَقِّ يَا أُمَيْمَةُ رَيْبٌ *** إِنَّمَا الرَّيْبُ مَا يَقُولُ الكَذُوبُ{[473]}<br />
وحقيقته على ما قال الزَّمخشري :  " قلق النفس واضطرابها "  .  <br />
ومنه الحديث :  " دَعْ ما يُرِيبُكَ إلى ما لا يُرِيبك " {[474]} .  <br />
ومنه أنه مَرّ بظبي خائفٍ فقال :  " لاَ يُرِبهُ أَحَدٌ بشيءٍ "  .  <br />
فليس قول من قال :  " الرَّيب الشك مطلقاً "  بجيّد ،   بل هو أخصّ من الشَّك كما تقدم .  <br />
وقال بعضهم : في  " الرّيب "  ثلاثة معانٍ :<br />
أحدها : الشّك ؛ قال ابن الزِّبَعْرَى{[475]} : [ الخفيف ]<br />
لَيْسَ فِي الْحَقِّ يَا أُمَيْمَةُ رَيْبٌ *** .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .  {[476]}<br />
وثانيها : التُّهمةُ ؛ قال جميل بُثَيْنَةَ{[477]} : [ الطويل ]<br />
بُثَيْنَةُ قَالَتْ يَا جَمِيلُ أَرَيْتَنِي *** فَقُلْتُ : كِلاَنا يَا بُثَيْنُ مُرِيبُ{[478]}<br />
وثالثها : الحاجات ؛ قال : [ الوافر ]<br />
قَضَيْنَا مِنْ تِهَامَةَ كُلَّ رَيْبٍ *** وَخَيْبَرَ ثُمَّ أَجْمَعْنَا السُّيُوفَا{[479]}<br />
قال ابن الخطيب{[480]} : الريب قريب من الشك ،   وفيه زيادة ،   كأنه ظن سوء ،   كأنه ظن سوء ،   تقول : رَابَني أمر فلان إذا ظننت به سوءاً .  <br />
فإن قيل : قد يستعمل الريب في قولهم :  " ريب الدهر "  و " ريب الزمان "  أي : حوادثه ،   قال تعالى : { نَّتَرَبَّصُ بِهِ رَيْبَ الْمَنُونِ } [ الطور : 30 ] ويستعمل أيضاً فيما يختلج في القلب من أسباب الغيظ ،   كقول الشاعر : [ الوافر ]<br />
قَضَيْنَا مِنْ تِهَامَةَ كُلَّ رَيْبٍ *** .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .  {[481]}<br />
قلنا : هذان يرجعان إلى معنى الشك ،   لأن من يخاف من ريب المنون محتمل ،   فهو كالمشكوك فيه ،   وكذلك ما اختلج بالقلب فهو غير متيقن ،   فقوله تعالى : { لاَ رَيْبَ فِيهِ } المراد منه : نفي كونه مَظَنّةً للريب بوجه من الوجوه ،   والمقصود أنه لا شُبْهَة في صحته ،   ولا في كونه من عند الله تعالى ولا في كونه معجزاً .   ولو قلت : المراد لا ريب في كونه معجزاً على الخصوص كان أقرب لتأكيد هذا التأويل بقوله تعالى : { وَإِن كُنتُمْ فِي رَيْبٍ مِّمَّا نَزَّلْنَا عَلَى عَبْدِنَا } [ البقرة : 23 ] .  <br />
فإن قيل : لم تأت ،   قال ها هنا :  " لاَ رَيْبَ فِيهِ "  وفي موضع آخر : { لاَ فِيهَا غَوْلٌ } [ الصافات : 47 ] قلنا : لأنهم يقدمون الأهمّ ،   وهاهنا الأهم نفي الريب بالكليّة عن الكتاب .  <br />
ولو قلت :  " لا فيه ريب "  لأَوْهَمَ أن هناك كتاباً آخر حصل فيه الريب لا هاهنا ،   كما قصد في قوله تعالى<br />
{ لاَ فِيهَا غَوْلٌ } تفضيل خمر الجنّة على خمر الدنيا ،   بأنها لا تَغْتَال العقول كما تغتالها خمر الدنيا .   فإن قيل : من أين يدلّ قوله : { لاَ رَيْبَ فِيهِ } على نفي الريب بالكلية ؟ قلنا : القراءة المشهورة توجب ارتفاع الريب بالكلية ،   والدّليل عليه أن قوله :  " لا ريب "  نفي لماهيّة الريب ؛ ونفي الماهية يقتضي نفي كل فرد من أفراد الماهية ؛ لأنه لو ثبت فرد من أفراد الماهيّة لثبتت الماهية ،   وذلك مُنَاقض نفي الماهية ،   ولهذا السّر كان قولنا :  " لا إله إلا الله "  نفياً لجميع الآلهة سوى الله تعالى .  <br />
وقرأ{[482]} أبو الشعثاء{[483]} :  " لاَ رَيْبُ فِيهِ "  بالرفع ،   وهو نقيض لقولنا :  " ريب فيه "  ،   وهذا يفيد ثبوت فرد واحدٍ ،   وذلك النفي يوجب انتفاء جميع الأفراد ،   فيتحقق التناقض .  <br />
والوقف على  " فيه "  هو المشهور .  <br />
وعن نافع وعاصم{[484]} أنهما وَقَفَا على  " ريب "  ،   ولا بد للواقف من أن ينوي خبراً ،   ونظيره قوله : { لاَ ضَيْرَ } [ الشعراء : 50 ] وقول العرب :  " لا بأس "  .  <br />
واعلم أن الملحدة طعنوا فيه وقالوا : إن عني أنه لا شَكّ فيه عندنا ،   فنحن قد نشك فيه ،   وإن عني أنه لا شكّ فيه عنده فلا فائدة فيه .  <br />
الجواب : [ المراد ]{[485]} أنه بلغ في الوضوح إلى حيث لا ينبغي لمرتاب أن يرتاب فيه ،   والأمر كذلك ؛ لأن العرب مع بلوغهم في الفَصَاحة إلى النهاية عجزوا عن معارضة أَقْصَرِ سورة من القرآن ،   وذلك يشهد بأنه لقيت هذه الحُجَّة في الظهور إلى حيث لا يجوز للعاقل أن يرتاب فيه .  <br />
وقيل : في الجواب وجوه أخر :<br />
أحدها : أن النفي كونه متعلقاً للريب ،   المعنى : أنه منعه من الدلالة ،   ما إن تأمله المُنْصِف المحق لم يرتب فيه ،   ولا اعتبار بمن وجد فيه الريب ؛ لأنه لم ينظر فيه حَقّ النظر ،   فريبه غير معتدٍّ به .  <br />
والثاني : أنه مخصوص ،   والمعنى : لا ريب فيه عند المؤمنين .  <br />
والثالث : أنه خبر معناه النهي .   والأول أحسن .  <br />
قوله تعالى : { هُدًى لِّلْمُتَّقِينَ } .   يجوز فيه عدة أوجه :<br />
أحدها : أن يكون مبتدأ ،   وخبره فيه متقدماً عليه إذا قلنا : إن خبر  " لا "  محذوف .  <br />
وإن قلنا :  " فيه "  خبرها ،   كان خبره محذوفاً مدلولاً عليه بخبر  " لا "  ،   تقديره : لا ريب فيه ،   فيه هدى ،   وأن يكون خبر مبتدأ مضمر تقديره : هو هدى ،   وأن يكون خبراً ثانياً ل  " ذلك "  ،   على أن يكون  " الكتاب "  صفةً أو بدلاً ،   أو بياناً ،   و " لا ريب "  خبر أول ،   وأن يكون خبراً ثالثاً ل  " ذلك "  ،   على أن يكون  " الكتاب "  خبراً أول ،   و " لا ريب "  ،   خبراً ثانياً ،   وأن يكون منصوباً على الحال من  " ذلك "  ،   أو من  " الكتاب "  ،   والعامل فيه على كلا التقديرين اسم الإشارة ،   وأن يكون حالاً من الضَّمير في  " فيه "  ،   والعامل ما في الجَارِّ والمجرور من معنى الفعْلِ ،   وجعله حالاً مما تقدم : إما على المُبَالغة ،   كأنه نفس الهُدَى ،   أو على حذف مضاف ،   أي : ذا هُدَى ،   أو على وقوع المصدر موقع اسم الفاعل ،   وهكذا كلُّ مصدر وقع خبراً ،   أو صفة ،   أو حالاً فيه الأقوال الثلاثة ،   أرجحها الأول .  <br />
وأجازوا أن يكون  " فيه "  صفةً ل  " ريب "  ،   فيتعلّق بمحذوف ،   وأن يكون متعلقاً ب  " ريب "  ،   وفيه إشكال ؛ لأنه يصير مطولاً ،   واسم  " لا "  إذا كان مطولاً أعرب إلاّ أن يكون مُرَادهم أنّه معمول لِمَا دَلّ عليه  " ريب "  لا لنفس  " ريب "  .  <br />
وقد تقدّم معنى  " الهدى "  عند قوله تبارك وتعالى : { اهْدِنَا الصِّرَاطَ الْمُسْتَقِيمَ } [ الفاتحة : 6 ] .  <br />
و " هُدًى "  مصدر على وزن  " فُعَل "  قالوا : ولم يجئ من هذا الوزن في المَصَادر إلا  " سُرًى "  و " بُكًى "  و " هُدًى "  ،   وقد جاء غيرها ،   وهو " لَقِيتُهُ لُقًى "  ؛ قال الشَّاعر : [ الطويل ]<br />
وَقَدْ زَعَمُوا حُلْماً لُقَاكِ وَلَمْ أَزِدْ *** بحَمْدِ الَّذِي أَعْطَاكِ حِلْماً وَلاَ عَقْلا{[486]}<br />
و " الهدى "  فيه لغتان : التذكير ،   ولم يذكر اللّحْياني{[487]} غيره .  <br />
وقال الفراء : بعض بني أسد يؤنثه ،   فيقولون : هذه هدى .  <br />
و " في "  معناها الظرفية حقيقةً أو مجازاً ،   نحو :  " زيد في الدار " {[488]} ،   { وَلَكُمْ فِي الْقِصَاصِ حَيَاةٌ } [ البقرة : 179 ] ولها معان آخر :<br />
المصاحبة : نحو : { ادْخُلُواْ فِي أُمَمٍ } [ الأعراف : 38 ] .  <br />
والتعليل :  " إن امرأة دَخَلتِ النَّارَ في هِرَّةٍ " {[489]} وموافقة  " على "  : { وَلأُصَلِّبَنَّكُمْ فِي جُذُوعِ النَّخْلِ } [ طه : 71 ] [ أي : على جذوع ]{[490]} ،   والباء : { يَذْرَؤُكُمْ فِيهِ } [ الشورى : 11 ] أي : بسببه .  <br />
والمقايسة نحو قوله تعالى : { فَمَا مَتَاعُ الْحَيَاةِ الدُّنْيَا فِي الآخِرَةِ } [ التوبة : 38 ] .  <br />
و " الهاء "  في  " فيه "  أصلها الضم كما تقدم من أن  " هاء "  الكناية أصلها الضم ،   فإن تقدمها ياء ساكنة ،   أو كسرة كسرها غير الحجازيين ،   وقد قرأ حمزة :  " لأَهْلِهُ امْكُثُوا "  وحفص في :  " عَاهَدَ عَلَيْهُ الله "  [ الفتح : 10 ] ،   { وَمَا أَنسَانِيهُ إِلاَّ } [ الكهف : 63 ] بلغة أهل الحِجَاز ،   والمشهور فيها - إذا لم يلها ساكن وسكن ما قبلها نحو :  " فيه "  و " منه "  - الاختلاس ،   ويجوز الإشْبَاع ،   وبه قرأ ابن كثير{[491]} ،   فإن تحرّك ما قبلها أُشْبِعَتْ ،   وقد تختلس وتسكن ،   وقرئ ببعض ذلك كما سيأتي مفصلاً إن شاء الله تعالى .  <br />
و " للمتقين "  جارّ ومجرور متعلّق ب  " هدى "  .  <br />
وقيل : صفة ل  " هدى "  ،   فيتعلّق بمحذوف ،   ومحله حينئذ : إما الرفع أو النصب بحسب ما تقدم في موصوفه ،   أي هدى كائن أو كائناً للمتقين .  <br />
والأحسن من هذه الوجوه المتقدمة كلها أن تكون كل جملة مستقلّة بنفسها ،   ف  " الم "  جملة إن قيل : إنها خبر مبتدأ مضمر ،   و " ذلك الكتاب "  جملة ،   و " لا ريب "  جملة ،   و " فيه هدى "  جملة ،   وإنما ترك العاطف لشدّة الوصل ؛ لأن كلّ جملة متعلّقة بما قبلها آخذة بعنقها تعلقاً لا يجوز معه الفصل بالعطف .  <br />
قال الزمخشري :  " وقد أصيب بترتيبها مفصل البلاغة حيث جيء بها مُتَنَاسقة هكذا من غير حرف نسق .   [ وذلك لمجيئها مُتَتَابعة بعضها بعنق ]{[492]} بعض ،   والثانية متحدة بالأولى ،   وهلم جَرّاً إلى الثالثة والرابعة .  <br />
بيانه : أنه نبّه أولاً على أنه الكلام المتحدى به ،   ثم أشير إليه بأنه الكتاب المَنْعُوت بنهاية الكَمَال ،   فكان تقريراً لجهة التحدي .   ثم نفى عنه أن يتشبث{[493]} به طرف من الريب ،   فكان شهادة بكماله .  <br />
ثم أخبر عنه بأنه  " هدى للمتقين "  ،   فقرر بذلك كونه يقيناً ،   لا يحوم الشّك حوله ،   ثم لم تَخْلُ كل واحدة من هذه الأربع بعد أن رتبت هذه الترتيب الأنيق [ من ]{[494]} نُكْتَةٍ ذات جَزَالة : ففي الأولى الحذف ،   والرمز إلى الغرض بألطف وجه .  <br />
وفي الثانية ما في التعريف من الفَخَامة .  <br />
وفي الثانية ما في تقديم الريب على الظَّرف .  <br />
وفي الثالثة ما في تقديم  " الريب "  على الظرف .  <br />
وفي الرابعة الحذف ،   ووضع المصدر الذي هو " هدى "  موضع الوصف الذي هو " هاد "  وإيراده منكراً .  <br />
 " المتقين "  جمع  " مُتَّقٍ "  ،   وأصله : مُتَّقْيِينَ بياءينِ ،   الأولى : لام الكلمة ،   والثانية علامة الجمع ،   فاستثقلت الكسرة على لام الكلمة ،   وهي الياء الأولى فحذفت ،   فالتقى ساكنان ،   فَحذف إحداهما وهي الأولى .  <br />
و " متقٍ "  من اتقى يتّقي وهو مُفْتَعِل من الوِقَايَةِ ،   إلا أنه يطرد في الواو والياء إذا كانتا فاءين ،   ووقعت بعدهما  " تاء "  الافتعال أن يبدلا  " تاء "  نحو :  " اتَّعَدَ "  من الوَعْدِ ،   و " اتَّسَرَ "  من اليُسْرِ .   وَفِعْلُ ذلك بالهمزة شاذ ،   قالوا :  " اتَّزَرَ "  و " اتَّكَلَ "  من الإِزَارِ ،   والأَكْلِ .  <br />
ول  " افتعل "  اثنا عشر معنى :<br />
الاتِّخَاذ نحو :  " اتقى "  .  <br />
والتسبب نحو :  " اعمل "  .  <br />
وفعل الفاعل بنفسه نحو :  " اضطرب "  .  <br />
والتخير نحو :  " انتخب "  .  <br />
والخطف نحو :  " استلب "  .  <br />
ومطاوعة  " أَفْعَل "  نحو :  " انتصف "  .  <br />
ومطاوعة  " فَعَّل "  نحو : عمّمته فاعتمّ .  <br />
وموافقة  " تَفَاعَلَ "  و " تَفَعَّل "  و " اسْتَفْعَل "  نحو : احتور واقتسم واعتصم ،   بمعنى تحاور وتقَسَّم واستعصم .  <br />
وموافقة المجرد ،   نحو " اقتدر "  بمعنى : قَدَرَ .  <br />
والإغناء عنه نحو :  " اسْتَلَم{[495]} الحجر "  ،   لم يُلْفَظْ له بمجرد .  <br />
و " الوقاية "  : فرط الصيانة ،   وشدة الاحتراس من المكروه ،   ومنه  " فرس وَاقٍ "  : إذا كان يقي حافِرُه أدنى شيء يصيبه .  <br />
وقيل : هي في أصل اللَّغة قلّة الكلام .  <br />
وفي الحديث :  " التَّقِيُّ مُلْجَمٌ "  .  <br />
ومن الصيانة قوله : [ الكامل ]<br />
سَقَطَ النَّصِيفُ وَلَمْ تُرِدْ إِسْقَاطَهُ *** فَتَنَاوَلَتْهُ وَاتَّقَتْنَا بِاليَدِ{[496]}<br />
وقال آخر : [ الطويل ]<br />
فَأَلْقَتْ قِنَاعاً دُونَهُ الشَّمْسُ واتَّقَتْ *** بِأَحْسَنِ مَوْصُولَيْنِ كَفٍّ وَمِعْصَمِ{[497]}<br />
قال أبو العباس المقرئ : ورد لفظ  " الهدى "  في القرآن بإزاء ثلاثة عشر معنى :<br />
الأول : بمعنى  " البَيَان "  قال تعالى : { أُوْلَئِكَ عَلَى هُدًى مِّن رَّبِّهِمْ } [ البقرة : 5 ] أي : على بيان ،   ومثله : { وَإِنَّكَ لَتَهْدِي إِلَى صِرَاطٍ مُّسْتَقِيمٍ } [ الشورى : 52 ] أي : لتبين ،   وقوله تبارك وتعالى : { وَأَمَّا ثَمُودُ فَهَدَيْنَاهُمْ } [ فصلت : 17 ] أي : بَيّنّا لهم .  <br />
الثاني : الهُدَى : دين الإسلام ،   قال تَعَالَى : { قُلْ إِنَّ الْهُدَى هُدَى اللَّهِ } [ آل عمران : 73 ] أي : دين الحق هو دين الله .  <br />
وقوله : { إِنَّكَ لَعَلَى هُدًى } [ الحج : 67 ] أي : دين الحق .  <br />
الثالث : بمعنى  " المَعْرِفَة "  قال تعالى : { وَبِالنَّجْمِ هُمْ يَهْتَدُونَ } [ النحل : 16 ] أي : يعرفون ،   وقوله تعالى : { نَكِّرُواْ لَهَا عَرْشَهَا نَنظُرْ أَتَهْتَدِي أَمْ تَكُونُ مِنَ الَّذِينَ لاَ يَهْتَدُونَ } [ النمل : 41 ] أي : أتعرف .  <br />
الرابع : بمعنى  " الرسول "  قال تعالى : { فَإِمَّا يَأْتِيَنَّكُم مِّنِّي هُدًى } [ البقرة : 38 ] أي : رسول .  <br />
الخامس : بمعنى  " الرشاد "  قال تعالى : { وَاهْدِنَا إِلَى سَوَاءِ الصِّرَاطِ } [ ص : 22 ] أي أرشدنا .  <br />
وقوله : { عَسَى رَبِّي أَن يَهْدِيَنِي سَوَاءَ السَّبِيلِ } [ القصص : 22 ] .  <br />
وقوله تعالى : { اهْدِنَا الصِّرَاطَ } [ الفاتحة : 6 ] .  <br />
السادس : بمعنى :  " القرآن "  قال تعالى : { وَلَقَدْ جَاءَهُم مِّن رَّبِّهِمُ الْهُدَى } [ النجم : 23 ] أي : القرآن .  <br />
السابع : بمعنى : بعثة النبي - صلى الله عليه وسلم - قال تبارك وتعالى : { وَإِنَّكَ لَتَهْدِي } [ الشورى : 52 ] .  <br />
الثامن : بمعنى  " شرح الصدور "  قال تعالى : { فَمَن يُرِدِ اللَّهُ أَن يَهْدِيَهُ يَشْرَحْ صَدْرَهُ لِلإِسْلاَمِ } [ الأنعام : 125 ] .  <br />
التاسع : التوراة ،   قال تبارك وتعالى : { وَلَقَدْ آتَيْنَا مُوسَى الْهُدَى } [ غافر : 53 ] يعني : التوراة .  <br />
العاشر :  " الجنة "  قال تعالى : { إِنَّ الَّذِينَ آمَنُواْ وَعَمِلُواْ الصَّالِحَاتِ يَهْدِيهِمْ رَبُّهُم } [ يونس : 9 ] أي : يدخلهم الجنة .  <br />
الحادي عشر :  " حج البيت "  قال تعالى : { إِنَّ أَوَّلَ بَيْتٍ وُضِعَ لِلنَّاسِ لَلَّذِي بِبَكَّةَ مُبَارَكاً وَهُدًى لِّلْعَالَمِينَ } [ آل عمران : 96 ] أي الحج .  <br />
الثاني عشر :  " الصلاح "  قال تَعَالى : { وَأَنَّ اللَّهَ لاَ يَهْدِي كَيْدَ الْخَائِنِينَ } [ يوسف : 52 ] أي : لا يصلح .  <br />
الثالث عشر :  " التوبة "  قال تعالى : { إِنَّا هُدْنَا إِلَيْكَ } [ الأعراف : 156 ] أي : تُبْنَا ورجعنا .  <br />
فصل في المقصود بالهدى<br />
قال ابن الخطيب{[498]}  " رضي الله تعالى عنه "  : الهُدَى عبارة عن الدلالة .  <br />
وقال صاحب  " الكشاف "  : الهدى هو الدلالة الموصّلة للبغية .  <br />
وقال آخرون : الهدى هو الاهتداء والعلم والدليل على صحة الأول أنه لو كان كون الدلالة موصلة إلى البغية معتبراً في مسمى الهدى لامتنع حصول الهدى عند عدم الاهتداء ؛ لأن كون الدلالة موصلة إلى الاهتداء حال عدم الاهتداء مُحَال ،   وقد ثبت الهدى على عدم حال الاهتداء قال الله تعالى : { وَأَمَّا ثَمُودُ فَهَدَيْنَاهُمْ فَاسْتَحَبُّواْ الْعَمَى عَلَى الْهُدَى } [ فصلت : 17 ] أثبت الهدى مع عدم الاهتداء .   واحتج صاحب  " الكشَّاف "  بأمور ثلاثة :<br />
[ أوّلها ] : وقوع الضلالة في مُقَابلة الهدى ،   قال تعالى :<br />
{ أُوْلَئِكَ الَّذِينَ اشْتَرَوُاْ الضَّلاَلَةَ بِالْهُدَى } [ البقرة : 16 ] ،   وقال تعالى : { وَإِنَّا أَوْ إِيَّاكُمْ لَعَلَى هُدًى أَوْ فِي ضَلاَلٍ مُّبِينٍ } [ سبأ : 24 ] .  <br />
وثانيها : يقال : مهديّ في موضع المدح كالمهتدي ،   فلو لم يكن من شرط الهدى كَوْن الدلالة موصلة إلى البغية لم يكن الوصف بكونه مهدياً مدحاً ؛ لاحتمال أنه هدي ،   فلم يَهْتَدِ .  <br />
وثالثها : أن  " اهتدى "  مطاوع  " هَدَى "  يقال : هَدَيْتُه فَاهْتَدَى ،   كما يقال : كسرته فانكسر ،   وقطعته فانقطع ،   فكما أن الانكسار والانقطاع لاَزِمَانِ للكسر والقطع ،   وجب أن يكون الاهتداء من لوازم  " الهدى "  .  <br />
والجواب عن الأوَّلِ : أن الفرق بين الهدى والاهتداء معلوم بالضرورة ،   فمقابل  " الهدى "  هو " الإضلال " {[499]} ومقابل  " الاهتداء "  هو " الضلال "  فجعل  " الهدى "  في مقابلة  " الضلال "  ممتنع .  <br />
وعن الثاني : المنتفع{[500]} بالهدى سمي مهدياً ؛ لأن الوسيلة إذا لم تُفْضِ إلى المقصود كانت نازلة منزلة المعدوم .  <br />
وعن الثالث : أن{[501]} الائتمار مُطَاوع الأمر يقال : أمرته فائتمر ،   ولم يلزم منه أن يكون من شرط كونه آمراً حصول الائتمار ،   وكذا لا يلزم من كونه هذه أن يكون مفضياً إلى الاهتداء ،   على أنه معارض بقوله : هديته فلم يهتد .  <br />
ومما يدل علة فساد قول من قال : الهدى هو العلم خاصة أن الله - تعالى - وصف القرآن بأنه هدى ،   ولا شك أنه في نفسه ليس بعلم ،   فدلّ على أن الهدى هو الدلالة لا الاهتداء والعلم .  <br />
فصل في اشتقاق المتقي<br />
والمتقي في اللغة : اسم فاعل من قولهم : وقاه فاتَّقَى ،   والوقاية : فرط الصيانة .  <br />
قال ابن عباس رضي الله عنهما : التقيّ : من يتقي الشّرْك والكبائر والفواحش{[502]} ،   وهو مأخوذ من الاتقاء ،   وأصله : الحجز بين شيئين .  <br />
وفي الحديث :  " كان إذا احْمَرّ البأسُ اتَّقينا برسول الله صلى الله عليه وسلم " {[503]} .  <br />
أي : إذا اشتد الحَرْبُ جعلنا بيننا وبين العدو ،   فكأن المتقي جعل الامتثال لأمر الله ،   والاجتناب عما نَهَاه حاجزاً بينه وبين العذاب .   وقال عمر بن الخَطّاب لكعب الأحبار :  " حدثني عن التقوى ،   فقال : هل أخذت طريقاً ذا شَوْكٍ ؟ قال : نعم ،   قال : فما عملت فيه ؟ قال : حذرت وشَمّرت ،   قال كعب : ذلك التَّقوى "  .   وقال عمر بن عبد العزيز : التقوى تَرْكُ ما حَرَّمَ الله ،   وأداء ما افترض الله ،   فما رزق الله بعد ذلك فهو خير{[504]} إلى خير .  <br />
وقال ابن عمر : التَّقْوَى ألا ترى نفسك خيراً من أحد .  <br />
إذا عرفت هذا فنقول : إن الله - تعالى - ذكرَ المتقي هاهنا في معرض المدح ،   [ ولن يكون ذلك ]{[505]} بأن يكون متقياً في أمور الدنيا بل بأن يكون متقياً فيما يتصل بالدّين ،   وذلك بأن يكون آتياً بالعبادات ،   محترزاً عن المحظورات .   واختلفوا في أنه هل يدخل اجتناب الصَّغائر في التقوى ؟ فقال بعضهم : يدخل كما تدخل الصّغائر في الوعيد .  <br />
وقال آخرون : لا يدخل ،   ولا نزاع في وجوب التوبة عن الكُلّ ،   إنما النزاع في أنه إذا لم يتوقّ الصغائر هل يستحق هذا الاسم ؟<br />
فروي عنه - عليه الصلاة والسلام - أنه قال :  " لا يَبْلُغُ العَبْدُ درجة المتّقين حتى يَدَعَ ما لا بَأْسَ به حَذَراً مما به بَأْسُ " {[506]} وعن ابن عباس - رضي الله تعالى عنهما - أنهم الذين يَحْذَرُون من الله العُقُوبَة في تَرْكِ ما يميل الهَوَى إليه ،   ويرجون رحمته بالتَّصديق بما جاء منه .  <br />
واعلم أن حقيقة التقوى ،   وإن كانت هي التي ذكرناها إلاَّ أنها قد جاءت في القرآن ،   والغرض الأصلي منها الإيمان تارة ؛ كقوله تعالى : { وَأَلْزَمَهُمْ كَلِمَةَ التَّقْوَى } [ الفتح : 26 ] أي : التوحيد { أُوْلَئِكَ الَّذِينَ امْتَحَنَ اللَّهُ قُلُوبَهُمْ لِلتَّقْوَى } [ الحجرات : 3 ] ،   { قَوْمَ فِرْعَوْنَ أَلاَ يَتَّقُونَ }<br />
[ الشعراء : 11 ] أي : لا يؤمنون .  <br />
وتارة التوبة كقوله تبارك وتعالى : { وَلَوْ أَنَّ أَهْلَ الْقُرَى آمَنُواْ وَاتَّقَوْاْ } [ الأعراف : 96 ] ،   { وَأَنَاْ رَبُّكُمْ فَاتَّقُونِ } [ المؤمنون : 52 ] .  <br />
وتارة ترك المعصية كقوله تعالى : { وَأْتُواْ الْبُيُوتَ مِنْ أَبْوَابِهَا وَاتَّقُواْ اللَّهَ } [ البقرة : 189 ] أي : فلا تعصوه .  <br />
وتارة الإخلاص كقوله : { فَإِنَّهَا مِن تَقْوَى الْقُلُوبِ } [ الحج : 32 ] أي : من إخلاص القلوب .  <br />
وهاهنا سؤالات :<br />
السؤال الأول : كون الشَّيء هدى ودليلاً لا يختلف بحسب شخص دون شخص فلماذا جعل القرآن هدى للمتّقين فقط ؟ وأيضاً فالمتقي مهتدي ؟ والمهتدي لا يهتدي ثانياً ،   والقرآن لا يكون هدى للمتقين ؟<br />
والجواب : أن القرآن كما أنه هدى للمتقين ،   ودلالة لهم على وجود الصانع ،   وعلى صدق رسوله ،   فهو أيضاً دلالة للكافرين ؛ إلا أن الله تبارك وتعالى ذكر المتقين مدحاً ليبين أنهم هم الذين اهتدوا ،   وانتفعوا به كما قال :<br />
{ إِنَّمَا أَنتَ مُنذِرُ مَن يَخْشَاهَا } [ النازعات : 45 ] وقال :<br />
{ إِنَّمَا تُنذِرُ مَنِ اتَّبَعَ الذِّكْرَ } [ يس : 11 ] .  <br />
وقد كان عليه الصلاة والسلام منذراً لكلّ الناس ،   فذكر هؤلاء الناس لأجل أن هؤلاء هُمُ الذين انتفعوا بإنذاره .  <br />
وأما من فسر الهُدَى بالدلالة الموصلة إلى المقصود ،   فهذا السؤال زائل عنه ؛ لأن كون القرآن موصلاً إلى المقصود ليس إلاَّ في حق المتقين .  <br />
السّؤال الثاني : كيف وصف القرآن كله بأنه هدى ،   وفيه مجمل ومتشابه كثير ،   ولولا دلالة العقل لما تميز المُحْكم عن المُتَشَابه ،   فيكون الهدى في الحقيقة هو الدلالة العقلية لا القرآن ؟<br />
ونقل عن علي بن أبي طالب أنه قال لابن عباس حيث بعثه رسولاً إلى الخوارج : لا تَحْتَجَّ عليهم بالقرآن ،   فإنه خَصْمٌ ذو وجهين ،   ولو كان هدى لما قال علي بن أبي طالب - رضي الله عنه - ذلك فيه ،   ولأنا نرى{[507]} جميع فرق الإسلام يحتجون به ،   ونرى القرآن مملوءاً من آيات بعضها صريح في الجبر ،   وبعضها صريح في القدر ؛ فلا يمكن التوفيق بينهما إلاّ بالتعسُّف الشديد ،   فكيف يكون هدى ؟<br />
الجواب : أن ذلك المُتَشَابه والمُجْمل لما لم ينفك عما هو المراد على التعيين{[508]} - وهو إما دلالة العقل ،   أو دلالة السمع - صار كله هُدًى .  <br />
السؤال الثالث : كل ما يتوقّف كون القرآن حُجّة على صحته لم يكن القرآن هدى فيه ،   فإذا استحال كون القرآن هدى في معرفة ذات الله - تعالى - وصفاته ،   وفي معرفة النبوة ،   فلا شَكَّ أن هذه المَطَالب أشرفُ المطالب ،   فإذا لم يكن القرآن هدى فيها ،   فكيف جعله الله هدى على الإطلاق ؟<br />
الجواب : ليس من شرط كونه هدى أن يكون هدى في كل شيء ،   بل يكفي فيه أن يكون هدى في بعض الأشياء ،   وذلك بأن يكون هدى في تعريف الشَّرَائع ،   أو يكون هدى في تأكيد ما في العقول ،   وهذه الآية من أَقْوَى الدلائل على أن المُطْلق لا يقتضي{[509]} العموم ،   فإن الله - تعالى - وصفة بكونه هُدًى من غير تقييد في اللّفظ ،   مع أنه يستحيل أن يكون هدى في إثبات الصَّانع ،   وصفاته ،   وإثبات النبوة ،   فثبت أنّ المطلق لا يفيد العموم .  <br />
السّؤال الرابع : الهُدَى هو الذي بلغ في البيان والوضوح إلى حيث بين غيره ،   والقرآن ليس كذلك ،   فإن المفسّرين ما ذكروا آية إلاّ وذكروا فيها أقوالاً كثيرة متعارضة ،   ويؤيد هذا قوله تعالى : { لِتُبَيِّنَ لِلنَّاسِ مَا نُزِّلَ إِلَيْهِمْ } [ النحل : 44 ] .  <br />
وما يكون كذلك لا يكون مبيناً في نفسه ،   فضلاً عن أن يكون مبيناً لغيره ،   فكيف يكون هدى ؟ قلنا : من تكلم في التفسير بحيث يورد الأقوال المُتَعَارضة ،   ولا يرجح واحداً منها على الباقي يتوجه عليه السؤال ،   وأما من رجح واحداً على البواقي فلا يتوجّه عليه السؤال
