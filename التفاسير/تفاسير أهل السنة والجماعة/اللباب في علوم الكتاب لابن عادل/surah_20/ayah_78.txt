قوله : { فَأَتْبَعَهُمْ فِرْعَوْنُ بِجُنُودِهِ } قال أبُو مسلم : يزعم رواة اللغة أنَّ  "  أتْبَعَهُمْ وتَبعَهُمْ  "  واحد ،   وذلك جائز{[25879]} ويحتمل أن تكون الباء زائدة ،   أي أتبَعَهُم فرعونُ جنوده{[25880]} كقوله : { لاَ تَأْخُذْ بِلِحْيَتِي }{[25881]} ( أسْرَى بِعَبْدِهِ ){[25882]} {[25883]} {[25884]} .  <br />
وقال غيره{[25885]} : في بَاء  "  بجنوده  "  أوجه :<br />
أحدها{[25886]} : أن تكون الباء للحال ،   وذلك على أن  "  أَتْبَعَ  "  متعد{[25887]} لاثنين حذف ثانيهما ،   والتقدير : فَأتْبَعَهُمْ فرعونُ عقابَه{[25888]} ،   وقدَّره أبو حيَّان : رُؤَسَاءَه وحشَمَهُ{[25889]} .  <br />
قال شهاب الدين : والأول أحسن{[25890]} .  <br />
والثاني{[25891]} : أن الباءَ زائدة في المفعول الثاني{[25892]} .   والتقدير : فَأتْبَعَهُمْ فِرعون جنوده ،   كقوله تعالى : { وَلاَ تُلْقُواْ بِأَيْدِيكُمْ }{[25893]} .  <br />
(  .   .   .   *** .   .   .   لاَ يَقْرَأْنَ بالسُّوَرِ{[25894]} ){[25895]}<br />
وأتبع قد جئتها متعدياً{[25896]} لاثنين مصرح بهما قال تعالى : وَأَتْبَعْنَاهُمْ ذُرِّيَاتِهِمْ{[25897]} .  <br />
والثالث{[25898]} : أنها معدية{[25899]} على أن  "  أتْبَعَ  "  ،   قد يتعدى لواحد بمعنى{[25900]} تَبع ويجوز على هذا الوجه{[25901]} أن تكون الباء للحال أيضاً ،   بل هو الأظهر .   وقرأ أبو عمرو في رواية والحسن  "  فاتْبَعَهُمْ  "  بالتشديد ،   وكذلك قراءة الحسن في جميع القرآن إلا في قوله : { فَأَتْبَعَهُ شِهَابٌ ثَاقِبٌ }{[25902]} قوله : { مَا غَشِيَهُمْ }{[25903]} فاعل  "  غَشِيَهُمْ  " {[25904]} وهذا من باب الاختصار وجوامع الكلم أي{[25905]} : ما يقل لفظها ويكثر معناها ،   أي فَغَشِيَهُمْ مَا لا يعلم كنهه إلا الله تعالى{[25906]} وقراءة الأعمش  "  فَغَشَّاهُمْ  "  مضعَّفاً{[25907]} ،   وفي الفاعل حينئذ{[25908]} ثلاثة أوجه :<br />
أحدها : أنه  "  مَا غَشَاهُمْ  "  كالقراءة قبله ،   أي غطَّاهم من اليَمِّ ما غطَّاهُم .  <br />
والثاني{[25909]} : هو ضمير الباري تعالى .   أي : فَغَشَّاهُم{[25910]} الله .  <br />
والثالث{[25911]} : هو ضمير فرعون ،   لأنه السبب في إهلاكهم{[25912]} .  <br />
وعلى هذين الوجهين : ف  "  مَا غَشَّاهُمْ  "  في محل نصب مفعولاً ثانياً{[25913]} .  <br />
فصل<br />
قيل{[25914]} : أمرَ فرعونُ جنوده أن يَتْبَعُوا{[25915]} موسى وقومه{[25916]} ،   وكان هو فيهم  "  فَغَشِيَهُمْ  "  أصابهم  "  مِنَ اليَمِّ مَا غَشِيَهُمْ  "  ،   وهو الغرق .  <br />
وقيل :  "  غَشِيَهُمْ  "  ،   علاهم وسترهم{[25917]} { مِّنَ اليم مَا غَشِيَهُمْ } يريد بعض ماء اليم لا كلّه .  <br />
وقيل : غَشِيَهُم من اليَمِّ ما غشي{[25918]} قوم موسى فغرقوا هم ونَجَا موسى وقومه{[25919]} .
