قوله : { [ ومن يعمل ] من الصالحاتَ من ذكر } "   " من "  الأولَى : للتَّبعيضِ ؛ لأنَّ المكلَّف لا يطيق عَمَل كل الصَّالِحَات .  <br />
وقال الطَّبَرِي :  "  هي زائدةٌ عند قَوْم  "  وفيه ضعفٌ ،   لعدم الشَّرْطَيْن ،   و  "  مِنْ  "  الثانية للتبيين ،   وأجاز أبو البقاء{[9848]} أن تكُونَ حالاً ،   وفي صَاحِبها وجهان :<br />
أحدهما : أنه الضَّميرُ المَرْفُوع ب  "  يَعْمَل  "  .  <br />
والثاني : أنه الصَّالحات ،   أي : الصالحات كَائِنةً من ذكر أو أنثى ،   وقد تقدَّم إيضاح هذا في قوله : { لاَ أُضِيعُ عَمَلَ عَامِلٍ مِّنكُم مِّن ذَكَرٍ أَوْ أُنثَى } [ آل عمران : 195 ] والكلامُ على  "  أوْ  "  أيضاً ،   وقوله :  "  وَهُوَ مُؤْمِنٌ  "  جملة حَالِيّة من فَاعِل  "  يَعْمَل  "  .  <br />
[ قوله  "  يدخلون  "  ] قرأ أبو عَمْرو ،   وابن كَثير ،   وأبُو بَكْر عن عَاصِم{[9849]} :  "  يُدْخَلون  "  هُنَا ،   وفي مَرْيم ،   وأوَّل غَافِر بضم حَرْف المُضَارَعة ،   وفتح الخَاء مبنياً للمَفْعُول ،   وانفردَ ابنُ كَثِيرٍ وأبو بكْرٍ بثانية غَافِر ،   وأبو عمرو بالتي في فَاطِر ،   والبَاقُون : بفتحِ حَرْفِ المُضَارعة ،   وضمِّ الخَاءِ مبنياً للفاعِل ،   وذلك للتفنُّنِ في البلاغَةِ .   والأوّل أحْسن ؛ لأنَّهُ أفْخَم ،   ولِكَوْنه مُوافقاً لقوله : { وَلاَ يُظْلَمُونَ فَتِيلاً } .  <br />
وأما القِرَاءة الثَّانية : فهي مُطَابِقَةٌ لقوله : { ادْخُلُواْ الْجَنَّةَ أَنتُمْ وَأَزْوَاجُكُمْ } [ الزخرف : 70 ] ،   ولقوله { ادْخُلُوهَا بِسَلامٍ } [ الحجر : 46 ] .  <br />
والنقير : النَّقْرَة في ظَهْر النَّوَاة ،   مِنْها تَنْبُت النَّخْلَة ،   والمَعْنَى : أنَّهم لا يُنْقَصُون قدر مَنْبَت النَّوَاة .  <br />
فإن قِيلَ : لم خصَّ اللَّه الصالِحينَ بأنَّهم لا يُظْلَمُونَ ،   مع أنَّ غيرهم كَذَلِك ؛ كما قال :<br />
{ وَمَا رَبُّكَ بِظَلاَّمٍ لِّلْعَبِيدِ } [ فصلت : 46 ] ،   وقوله { وَمَا اللَّهُ يُرِيدُ ظُلْماً لِّلْعَالَمِينَ } [ آل عمران : 108 ] .  <br />
فالجواب : من وجهين :<br />
الأول : أن يَكُون الرَّاجع في قوله :  "  ولا يُظْلَمُونَ  "  عائداً إلى عُمَّال السُّوءِ ،   وعُمَّال{[9850]} الصَّالِحَات جَمِيعاً .  <br />
والثَّاني : أن من لا يُنْقِصُ من الثَّواب ،   أولى بأن لا يَزِيدَ في العِقَابِ .  <br />
روى الأعْمَش ،   عن أبي الضُّحَى{[9851]} عن مَسْرُوقٍ ،   قال : لمَّا نَزَل { لَّيْسَ بِأَمَانِيِّكُمْ وَلا أَمَانِيِّ أَهْلِ الْكِتَابِ مَن يَعْمَلْ سُوءًا يُجْزَ بِهِ } قال أهْل الكِتَاب : نَحْنُ وأنْتُم سَوَاءٌ ،   فَنَزَلتْ : { وَمَن يَعْمَلْ مِنَ الصَّالِحَاتَ } الآية ،   ونزل أيْضاً : { وَمَنْ أَحْسَنُ دِيناً مِّمَّنْ أَسْلَمَ }<br />
[ النساء : 125 ] .  <br />
فصل : صاحب الكبيرة لا يخلد في النار<br />
وهذه الآية من أدَلِّ الدَّلائِل على أن صَاحِبَ الكَبِيرَة لا يُخَلَّد [ في النَّارِ ]{[9852]} بل ينقل{[9853]} إلى الجَنَّة ؛ لأنَّا بينَّا أن صاحِبَ الكبِيرَة مؤمن ،   وإذا ثَبَتَ ذَلِك ،   وكان قد عَمِلَ الصَّالِحَات ،   وجب أنْ يدْخُل الجَنَّة ،   لهذه الآية ،   ولزم بِحُكْم الآيَات الدَّالَّة على وَعِيد الفُسَّاق أن يَدْخُل النَّار ،   فإمَّا{[9854]} أن يدْخُل النَّار{[9855]} ،   ثم يَنْتَقِل إلى النَّارِ ،   وذلك بَاطِلٌ بالإجْماع ،   أو يَدْخُل النَّار ،   ثم ينتقل إلى الجَنَّة ،   وذلك هُو المَطْلُوب .
