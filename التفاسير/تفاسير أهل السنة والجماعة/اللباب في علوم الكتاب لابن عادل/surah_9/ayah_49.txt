قوله تعالى : { وَمِنْهُمْ مَّن يَقُولُ ائذن لِّي وَلاَ تَفْتِنِّي } الآية .  <br />
{ مَّن يَقُولُ ائذن لِّي } كقوله :  "  يَا صالحُ ائتنا  "  من أنه يجوزُ تحقيقُ الهمزة ،   وإبدالها واواً ،   لضمة ما قبلها ،   وإن كانت منفصلةً من كلمةٍ أخرى .   وهذه الهمزةُ هي فاءُ الكلمة ،   وقد كان قبلها همزة وصلٍ سقطت درجاً .  <br />
قال أبو جعفرٍ  "  إذا دخلت  "  الواو  "  و "  الفاء  "  على  "  ائذن  "  فهجاؤها : ألف وذال ونون ،   بغير ياء .   أو  "  ثم  "  فالهجاءُ : ألفٌ وياءٌ وذالٌ ونونٌ .   والفرق : أنَّ  "  ثُمَّ  "  يوقف عليها وينفصل ،   بخلافهما  "  .  <br />
قال شهابُ الدِّين  "  يعني إذا دخلت واوُ العطف ،   أو فاؤه ،   على هذه اللفظة اشتدَّ اتصالهما بها ،   فلم يُعْتَدَّ بهمزة الوصل المحذوفة دَرْجاً ،   فلم يُرْسَمْ لها صورةٌ ،   فتكتب  "  فأذَنْ  "  ،   و "  أذَنْ  "  فهذه الألف هي صورة الهمزة ،   التي هي فاء الكلمة  "  .  <br />
وإذا دخلت عليها  "  ثم  "  كُتِبَتْ كذا : ثم ائتُوا ،   فاعتدُّوا بهمزة الوصل فرسموا لها صورة .  <br />
قال شهابُ الدين : وكأنَّ هذا الحُكْمَ الذي ذكره مع  "  ثم  "  يختصُّ بهذه اللَّفظة ،   وإلاَّ فغيرها مما فاؤه همزةٌ ،   تسقط صورة همزة وصله خطّاً ،   فيكتب الأمرُ من الإتيان مع  "  ثم  "  هكذا :  "  ثُمَّ أتُوا  "  ،   وكان القياس على  "  ثُم ائْذَانْ  "   "  ثم ائْتُوا  "  ،   وفيه نظرٌ ،   وقرأ{[17859]} عيسى بن عمر ،   وابن السَّميفع ،   وإسماعيل المكي ،   فيما روى عنه ابن مجاهد  "  ولا تُفْتِنِّي  "  بضم حرف المضارعة ،   من  "  أفتنه  "  رباعياً .   قال أبُو حاتم  "  هي لغة تميمٍ  "  وقيل : أفتنه : أدخله فيها ،   وقد جمع الشاعر بين اللغتين ،   فقال : [ الطويل ]<br />
لَئِنْ فَتَنَتْني لَهْيَ بالأمْسِ أفْتَنَتْ *** سَعِيداً فأمْسَى قَدْ قَلاَ كُلَّ مُسْلِمِ{[17860]}<br />
ومتعلق  "  الإذن  "  : القعودُ ،   أي : ائذن لي في القعود والتخلف عن الغزو ،   ولا تَفْتِنِّي بخروجي معك .   أي : لا تهلكني بخروجي معك ،   فإنَّ الزمانَ شديد الحرّ ،   ولا طاقة لي به .   وقيل : لا تفتنيِّي ؛ لأنِّي إذا خرجت معك هلك مالي وعيالي ،   وقيل : نزلت في جد بن قيس المنافق وذلك أنَّ النبي صلى الله عليه وسلم لمَّا تجهَّز لغزو تبوك ،   قال له :  "  يا أبا وهب ،   هل كل في جلاد بني الأصفر ؟ يعني : الروم - تتخذ منهم سراري ووصفاء  "  فقال جدُّ : يا رسول الله ،   لقد عرف قومي أني رجل مغرمٌ بالنِّساءِ ،   وإنِّي أخشى إن رأيت بنات الأصفر ألاَّ أصبر عنهنَّ ،   ائذن لي في القعود ،   ولا تفتنِّي بهنّ ،   وأعينكم بمالي{[17861]} .  <br />
قال ابن عباسٍ :  "  اعتلَّ جدُّ بن قيس ،   ولم تكن علته إلا النفاق ،   فأعرض عنه النبي صلى الله عليه وسلم وقال :  "  قد أذنت لك  "  ،   فأنزل الله عز وجل : { وَمِنْهُمْ مَّن يَقُولُ ائذن لِّي } الآية{[17862]} .  <br />
قوله : { أَلا فِي الفتنة سَقَطُواْ } أي : في الشّرك والإثم وقعوا بنفاقهم ،   وخلافهم أمر الله ورسوله .  <br />
وفي مصحف أبي  "  سَقَطَ  "  لأنَّ لفظة  "  مِنْ  "  موحد اللفظ ،   مجموع المعنى ،   وفيه تنبيه على أنَّ مَنْ عصى الله لغرض ،   فإنَّهُ تعالى يبطل عليه ذلك الغرض ؛ لأنَّ القوم لمَّا اختاروا القعود لئلا يقعوا في الفتنة ،   بيَّن اللهُ تعالى أنهم واقعون ساقطون في عينِ الفتنة - ثم قال { وَإِنَّ جَهَنَّمَ لَمُحِيطَةٌ بالكافرين } مطبقة عليهم .
