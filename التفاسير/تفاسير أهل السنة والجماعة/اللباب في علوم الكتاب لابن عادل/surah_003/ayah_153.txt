العامل في  "  إذْ  "  قيل : مُضْمَر ،   أي : اذكروا .  <br />
وقال الزمخشريُّ :  "  صَرَفَكُمْ  "  أوْ  "  لِيَبْتَلِيَكُمْ "  .  <br />
وقال أبو البقاء : ويجوز أن يكون ظرفاً لِ  "  عَصَيْتُمْ  "  أوْ  "  تَنَازَعْتُمْ  "  أو فَشِلْتُمْ .  <br />
وقيل : هو ظرف لِ  "  عَفَا عَنْكُمْ  "  أي : عفا عنكم إذْ تُصْعِدُون هاربين .  <br />
وكل هذه الوجوهِ سائغةٌ ،   وكونه ظرفاً لِ  "  صَرَفَكُمْ  "  جيدٌ من جهةِ المعنى ،   ولِ  "  عَفَا  "  جيدٌ من جهة القُرْبِ ،   وعلى بعض هذه الأقوالِ تكون المسألةُ من باب التنازعِ ،   ويكون على إعمال الأخيرِ منها ،   لعدم الإضمارِ في الأول ،   ويكون التنازعُ في أكثرِ من عاملينِ .  <br />
والجمهور على { تُصْعِدُونَ } - بضم التاء وكسر العين - من : أصْعَدَ في الأرض ،   إذا ذهب فيها .   والهمزةُ فيه للدخول ،   نحو أصبح زيدٌ ،   أي : دخل في الصباح ،   فالمعنى : إذ تدخلون في الصعود ،   ويُبيِّن ذلك قراءة أبيٍّ{[6077]}  "  تصعدون في الوادي "  .  <br />
وقرأ الحسنُ ،   والسُّلمي ،   وقتادةُ :  "  تَصْعَدُونَ  "  بفتح التاء والعين{[6078]} - من : صعد في الجبل ،   أي : رقي ،   والجمع بين القراءتين أنهم - أولاً - أصعدوا في الوادي ،   ثم لما هزمهم العدو - صعدوا في الجبل ،   وهذا على رَأى مَنْ يُفَرِّق بين صعد وأصْعد ،   وقرأ أبو حَيْوَةَ :  "  تَصَعَّدُون  "  بالتشديد{[6079]} - وأصله : تَتَصَعَّدُونَ ،   حذفت إحدى التاءين ،   إما تاء المضارعة ،   أو تاء  "  تَفَعَّل  "  والجمع بين قراءتِهِ وقراءة غيره كما تقدم .  <br />
والجمهورُ { تُصْعِدُونَ } بتاء الخطاب ،   وابن مُحَيْصن - ويُرْوَى عن ابن كثيرٍ - بياء الغيبة{[6080]} ،   على الالتفاتِ ،   وهو حسنٌ .  <br />
ويجوز أن يعودَ الضمير على المؤمنين ،   أي : { وَاللَّهُ ذُو فَضْلٍ عَلَى الْمُؤْمِنِينَ إِذْ تُصْعِدُونَ } فالعاملُ في  "  إذْ  "   "  فَضْلٍ  "  ويقال : أصْعَدَ : أبعد في الذهاب ،   قال القُتَبِيُّ أصعد : إذا أبْعَد في الذهاب ،   وأمعن فيه ،   فكأن الإصعادَ إبعادٌ في الأرض كإبعاد الارتفاعِ .  <br />
قال الشاعرُ : [ الطويل ]<br />
ألاَ أيُّهَذَا السَّائِلِي ،   أيْنَ أصْعَدَتْ ؟ *** فَإنَّ لَهَا مِنْ بَطْنِ يَثْرِبَ مِوْعِدا{[6081]}<br />
وقال آخرُ : [ الرجز ]<br />
قَدْ كُنْتِ تَبْكِينَ عَلَى الإصْعَادِ *** فَالْيَوْمَ سُرِّحْتِ ،   وَصَاحَ الْحَادِي{[6082]}<br />
وقال الفرَّاءُ وأبو حاتم : الإصعاد : في ابتداء السفر والمخارج ،   والصعود : مصدر صَعَدَ : رقي من سُفْلٍ إلى عُلُو ،   ففرَّق هؤلاء بين صَعَد وأصْعَد .  <br />
وقال المفضَّلُ : صعد وأصعد بمعنًى واحدٍ ،   والصعيد : وجْهُ الأرضِ .  <br />
قال بعضُ المفسّرين :  "  وكلتا القراءتين صوابٌ ،   فقد كان يومئذ من المنهزمين مُصْعِد وصاعد "  .  <br />
قوله : { وَلاَ تَلْوُونَ } الجمهور على { تَلْوُونَ } - بواوين - وقُرِئَ بإبدال الأولى همزة ؛ كراهية اجتماع واوين ،   وليس بقياسٍ ؛ لكون الضمة عارضة ،   والواو المضمومة تُبْدَل همزة بشروط تقدمت في  "  البقرة "  .  <br />
منها : ألا تكون الضمة عارضة ،   كهذه ،   وأن لا تكون مزيدة ،   نحو ترهوك .  <br />
وألا يمكن تخفيفها ،   نحو سُور ونور - جمع سوار ونوار - لأنه يمكن تسكينُها فتقول : سور ونور ،   فيخف اللفظ بها .  <br />
وألا يُدْغم فيها ،   نحو تعوَّذ - مصدر تعوذ .  <br />
ومعنى { وَلاَ تَلْوُونَ } ولا ترجعون ،   يقال : لَوَى به : ذهب به ،   ولَوَى عليه : عطف .  <br />
قَالَ الشاعرُ : [ الطويل ]<br />
 .   .   .   ***  أخُو الْجَهْدِ لا يَلْوِي عَلَى مَنْ تَعَذَّرَا{[6083]}<br />
وأصله أن المعرِّجَ على الشيءِ يلوي عليه عنقه ،   أو عنان دابته ،   فإذا مضى - ولم يعرِّج - قيل : لن يلوي ،   ثم استعمل في ترك التعريجِ على الشيء وترك الالتفاتِ إليه ،   يقال : فلانٌ لا يلوي على كذا أي : لا يلتف إليه ،   وأصل { تَلْوُونَ } تلويون ،   فَأعِلَّ بحذفِ اللام ،   وقد تقدم في قوله : { يَلْوُونَ أَلْسِنَتَهُمْ } [ آل عمران : 78 ] وقرأ الأعمشُ ،   وأبو بكر بنُ عَيَّاشٍ - ورويت عن عاصم  "  تُلوون  "  بضم التاء - من ألوى وهي لغةٌ في لوى{[6084]} .  <br />
وقرأ الحسن  "  تَلُون  "  - بواو واحدة{[6085]} - وخرجوها على أنه أبدل الواو همزةً ،   ثم نقل حركة الهمزة على اللام ،   ثم حذف الهمزة ،   على القاعدة ،   فلم يبق من الكلمة إلا الفاء - وهي اللام - وقال ابنُ عطيةَ :  "  وحذفت إحدى الواوين للساكنين ،   وكان قد تقدم أن هذه القراءة هي قراءة مركبةٌ على لغة من يهمز الواوَ ،   وينقل الحركة "  .  <br />
وهذا عجيبٌ ،   بعد أن يجعلها من باب نقل حركةِ الهمزةِ ،   كيف يعودُ ويقول : حذفت إحدى الواوين للساكنين ؟ ويُمكن تخريجُ هذه القراءة على وجهين آخرينِ :<br />
أحدهما : أن يقالَ : استُثقلت الضمةُ على الواو ؛ لأنها أختها ،   فكأنه اجتمع ثلاثُ واواتٍ ،   فنُقِلت الضمةُ إلى اللامِ ،   فالتقى ساكنانِ - الواو التي هي عينُ الكلمةِ ،   والواو التي هي ضميرٌ - فحُذفت الأولى ؛ لالتقاء الساكنين ،   ولو قال ابن عطيةَ هكذا لكان أولى .  <br />
الثاني : أن يكون  "  تَلُونَ  "  مضارع وَلِي - من الولاية - وإنما عُدِّي ب  "  على  "  لأنه ضُمِّن معنى العطف .   وقرأ حُميد{[6086]} بن قيس :  "  على أُحُدٍ  "  - بضمتين - يريد الجَبَل ،   والمعنى : ولا تلوون على مَنْ فِي جبل أُحُد ،   وهو النبي صلى الله عليه وسلم قال ابن عطية : والقراءة الشهيرة أقْوى ؛ لأن النبي صلى الله عليه وسلم لم يكن على الجبل إلا بعدما فَرَّ الناسُ عنه ،   وهذه الحالُ - من إصعادهم - إنما كانت وهو يدعوهم .  <br />
ومعنى الآيةِ : تعرجون ،   ولا يلتفت بعضٌ إلى بعضٍ .  <br />
قوله :  "  والرسول يدعوكم  "  ،   مبتدأ وخبر في محل نصب على الحالِ ،   العامل فيها  "  تلوون "  .  <br />
أي : والرسول يدعوكم في أخراكم ومن ورائكم ،   يقول :  "  إليَّ عِبَاد الله ؛ فأنا رسولُ اللهِ ،   من يكر فله الجنَّةُ "  .  <br />
ويحتمل أنه كان يدعوكم إلى نفسه ،   حتى تجتمعوا عنده ،   ولا تتفرقوا .   و  "  أخراهم  "  آخر الناس كما يقال في أولهم ،   ويقال : جاء فلانٌ في أخريات الناس .  <br />
قوله : { فَأَثَابَكُمْ } فيه وجهان :<br />
أحدهما : أنه معطوف على  "  تصعدون  "  و  "  تلوون  "  ،   ولا يضر كونهما مضارعين ؛ لأنهما ماضيان في المعنى ؛ لأن  "  إذ  "  المضافة إليهما صيرتهما ماضيين ،   فكأن المعنى إذا صعدتم ،   وألويتم .  <br />
الثاني : أنه معطوفٌ على  "  صرفكم "  .  <br />
قال الزمخشريُّ { فَأَثَابَكُمْ } عطف على صرفكم ،   وفيه بُعْدٌ ؛ لطول الفصل وفي فاعله قولان :<br />
أحدهما : أنه الباري تعالى .  <br />
والثاني : أنه النبيُّ صلى الله عليه وسلم .  <br />
قال الزَّمَخْشَرِيُّ : ويجوز أن يكون الضمير في { فَأَثَابَكُمْ } للرسول أي : فآساكم من الاغتمام ،   وكما غمكم ما نزل به من كسر رباعيته غمه ما نزل بكم من فوت الغنيمة .  <br />
و  "  غماً  "  مفعول ثانٍ .  <br />
وقوله : { فَأَثَابَكُمْ } هل هو حقيقة أو مجاز فقيل : مجاز كأنه جعل الغم قائماً مقام الثواب الَّذِي كان يحصل لولا الفرارُ فهو كقوله : [ الطويل ]<br />
أخَافُ زِيَاداً أنْ يَكُونَ عَطَاؤُهُ *** أدَاهِم سُوداً أوْ مُحَدْرَجَة سُمْرَا{[6087]}<br />
وقول الآخر :<br />
تحية بينهم ضرب وجيع{[6088]} *** .   .   .  <br />
جعل القيود والسياط بمنزلة العطاء ،   والضرب بمنزلة التحية .  <br />
وقال الفرّاءُ :  "  الإثابة - هاهنا - بمعنى المعاقبة  "  وهو يرجع إلى المجاز ؛ لأن الإثابة أصلها في الحسنات .  <br />
قوله : { بِغَمٍّ } يجوز في الباء أوجهٌ :<br />
أحدها : أن تكون للسببيةِ ،   على معنى أن متعلِّق الغَمْ الأول الصحابة ،   ومتعلق الغَمِّ الثاني قيل المشركين يوم بدرٍ .  <br />
قال الحَسَنُ : يريد غَم يوم أحدٍ للمسلمين بغمّ يوم بدرٍ للمشركينَ ،   والمعنى : فأثابكم غماً بالغم الذي أوقعه على أيديكم بالكفار يوم بدرٍ .  <br />
وقيل متعلَّق الغَمِّ الرسول ،   والمعنى : أذاقكم الله غمًّا بسبب الغَمِّ الذي أدخلتموه على الرسول والمؤمنين بفشلكم ومخالفتكم أمره ،   أو فأثابكم الرسول غماً بسبب غم اغتممتموه لأجله ،   والمعنى أن الصحابة لما رأوا النبي صلى الله عليه وسلم شُجَّ وَجهه وكُسِرت رَبَاعِيَتُه ،   وقُتِل عَمه ،   اغتممتموه لأجله ،   والنبيُّ صلى الله عليه وسلم لما رآهم قَد عَصَوْا رَبَّهُم لأجل الغنيمة - ثم بَقَوْا محرومينَ من الغنيمةِ - وقتِلَ أقاربُهم ،   اغتم لأجلهم .  <br />
الثاني : أن تكون الباء للمصاحبة ،   أي : غماً مصاحِباً لغم ،   ويكون الغمَّان للصحابة ،   بمعنى غَمًّا مع غم أو غماً على غم ،   فالغم الأولُ : الهزيمة والقتل ،   والثاني إشراف خالد بخيل الكفار ،   أو بإرجافهم : قتل الرسول صلى الله عليه وسلم فعلى الأول تتعلق الباء ب { فَأَثَابَكُمْ } .  <br />
قال أبو البقاء وقيل : المعنى بسبب غم ،   فيكون مفعولاً به .  <br />
وعلى الثاني يتعلقُ بمحذوفٍ ؛ لأنه صفة لِ  "  غَمّ  "  أي : غماً مصاحباً لغم ،   أو ملتبساً بغَمٍّ ،   وأجاز أبو البقاء أن تكون الباءُ بمعنى  "  بعد  "  أو بمعنى  "  بدل  "  وجعلها - في هذين الوجهين - صفة ل  "  غماً  "  .   وكونها بمعنى  "  بعد  "  و  "  بدل  "  بعيدٌ ،   وكأنه يريدُ تفسير المعنى ،   وكذا قَالَ الزمخشريُّ غماً بَعْدَ غَمٍّ .  <br />
واعلم أن الغموم هناك كانت كثيرة :<br />
أولاً : غَمُّهم بما نالهم من العدوِّ في الأنفس والأموال{[6089]} .  <br />
ثانياً : غمُّهم بما لَحِق المسلمين من ذلك{[6090]} .  <br />
ثالثاً : غمُّهم بما وصل إلى الرسول صلى الله عليه وسلم{[6091]} .  <br />
رابعاً : غمُّهم بما وقع منهم من المعصية وخوف عقابها{[6092]} .  <br />
خامساً : غمُّهم بسبب التوبة التي صارت واجبةً عليهم ؛ لأنهم إذا تابوا عن تلك المعصية لم تتم توبتهم إلا بتركِ الهزيمةِ والعَوْدِ إلى المحاربة بعد الانهزامِ ،   وذلك من أشق الأشياء ؛ لأن الإنسانَ بعد انهزامه - يَضعُف قلبُه ويجبن ،   فإذا أمِرَ بالمعاودةِ ،   فإن فعل خاف القتلَ ،   وإنْ لم يفعلُ خافَ الكُفْرَ وعِقَابَ الآخِرَةِ - وهذا الغَمُّ أعظمها{[6093]} .  <br />
سادسها : غمُّهم حين سمعوا أن محمداً قُتِلَ {[6094]} .  <br />
سابعها : غمُّهم حين أشرف خالد بن الوليد عليهم بخَيْل المشركين{[6095]} .  <br />
ثامنها : غمُّهم حين أشرف أبو سفيان{[6096]} ،   وذلك أن رسول الله صلى الله عليه وسلم انطلق يومئذٍ يدعو الناسَ حتى انتهى إلى أصحاب الصخرةِ ،   فلما رَأوْهُ وضع رجلٌ سَهْماً في قوسه ،   وأراد أن يَرْمِيَه ،   فقال : أنا رسولُ اللهِ ،   ففرحوا حين وجدوا رسولَ الله صلى الله عليه وسلم وفرح رسول الله صلى الله عليه وسلم حين رأى مَنْ يمتنع به ،   فأقبلوا على المشركين ،   يذكرون الفتح وما فاتهم منه ،   ويذكرون أصحابهم الذين قُتِلوا ،   فأقبل أبو سفيان وأصحابه ،   حتى وقفوا بباب الشِّعْب ،   فلما نظر المسلمون إليهم همَّهم ذلك ،   وظنوا أنهم يميلون عليهم ،   فيقتلونهم ،   فأنساهم هذا ما نالهم ،   فقال رسول الله صلى الله عليه وسلم ليس لهم أن يَعْلونا ،   اللهم إن تُقْتَل هذه العصابةُ لا تُعْبَد في الأرض ،   ثم بدأ أصحابه ،   فرمَوْهم بالحجارة حتَّى أنزلوهم .  <br />
وإذا عرفت هذا فكلُّ واحدٍ من المفسّرين فسَّر هذين الغمين بغمين من هذه الغموم وقال القفّال : وعندنا أن الله - تعالى - ما أراد بقوله : { غُمّاً بِغَمٍّ } اثنين ،   وإنما أراد مواصلة الغموم وطولها ،   أي : أن اللهَ عاقبكم بغموم كثيرة ،   مثل قتل إخوانكم وأقاربكم ،   ونزول المشركين من فوق الجبلِ عليكم ،   بحيث لم تأمنوا أن يهلك أكثركم ،   ومثل إقدامكم على المعصيةِ ،   فكأنه - تعالى - قال : أثابكم هذه الغمومَ المتعاقبةَ ؛ ليصير ذلك زاجراً لكم عن الإقدام على المعصية ،   والاشتغال بما يخالف أمرَ الله تعالى .  <br />
والغَمّ : التغطية ،   يقال : يوم غَمٌّ ،   وليلة غَمَّةٌ - إذا كانا مُظْلِمَيْن - ومنه : غُمَّ الهلال - إذا لَمْ يُرَ ،   وغَمَّ الأمرُ ،   يَغْمَى - إذا لم يُتَبَيَّنُ .  <br />
قوله : { لِّكَيْلاَ تَحْزَنُواْ } هذه لام  "  كي  "  وهي لام جرٍّ ،   والنصب - هنا - ب  "  كي  "  لئلاَّ يلزم دخول حرفِ جرٍّ على مثله ،   وفي متعلَّق هذه اللام قولانِ :<br />
أحدهما : أنه { فَأَثَابَكُمْ } وفي  "  لا  "  على هذا وجهانِ :<br />
الأول : أنها زائدةٌ ؛ لأنه لا يترتب على الاغتمام انتفاء الحُزْن ،   والمعنى : أنه غمهم ليُحْزِنهم ؛ عقوبةً لهم على تركهم مواقفهم ،   قاله أبو البقاء .  <br />
الثاني : أنها ليست زائدة ،   فقال الزمخشريُّ : معنى { لِّكَيْلاَ تَحْزَنُواْ } لتتمرَّنوا على تجرُّع الغمومِ ،   وتتضرروا باحتمال الشدائد ،   فلا تحزنوا فيما بعدُ على فائتٍ من المنافع ،   ولا على مُصِيبٍ من المضارّ .  <br />
وقال ابن عطية :  "  المعنى : لتعلَمُوا أن ما وقع بكم إنما هو بجنايتكم ،   فأنتم ورَّطتم أنفسكم ،   وعادة البشر أن جاني الذنب يصبر للعقوبة ،   وأكثر قلق المعاقب وحزنه إذا ظن البراءة من نفسه "  .  <br />
ثانيهما : أن اللام تتعلق ب  "  عَفَا  "  لأن عَفْوه يُذْهِب كُلَّ حُزْن ،   وفيه بُعْدٌ ؛ لطول الفَصْلِ .  <br />
ثم قال : { وَاللَّهُ خَبِيرٌ بِمَا تَعْمَلُونَ } أي : عالم بجميع أعمالكم وقصودكم ودواعيكم ،   قادرٌ على مجازاتها ،   وهذا زَجْرٌ للبُعْدِ عَنِ الإقْدَامِ عَلَى الْمَعْصِيَةِ .
