قوله : { وَإِنِّي خِفْتُ الموالي } : العامَّةُ على  "  خِفْتُ  "  بكسر الخاء ،   وسكون الفاء ،   وهو ماضٍ مسندٌ لتاءِ المتكلِّم ،   و  "  الموالِي  "  مفعولٌ به ؛ بمعنى : أنَّ مواليهُ كانُوا شِرَارَ بني إسرائيل ،   فخافهم على الدِّين ،   قاله الزمخشريٌُّ .  <br />
قال أبو البقاء{[21394]} :  "  لا بُدَّ من حذفِ مضافٍ ،   أي : عدم الموالي ،   أو جَوْرَ الموالي  "  .  <br />
وقال الزهريُّ كذلك ،   إلا انه سكَّن ياء{[21395]}  "  المَوالِي  "  وقد تقدّر الفتحةُ في الياء ،   والواو ،   وعليه قراءة زيد بن عليٍّ { تُطْعِمُونَ أَهالِيكُمْ } [ المائدة : 89 ] .  <br />
وتقدَّم إيضاحُ هذا .  <br />
وقرأ عثمانُ بنُ عفَّان ،   وزيدُ بن ثابت ،   وابن عبَّاس ،   وسعيد بن جبير ،   وسعيد بن العاص ،   ويحيى بنُ يعمر ،   وعليُّ بنُ الحسين في آخرين :  "  خَفَّتِ  "  بفتحِ الخاءِ ،   والفاءِ مشددةً ،   وتاء تأنيثٍ ،   كُسرتْ ؛ لالتقاءِ السَّاكنين ،   و  "  المَوالِي  "  فاعلٌ به ؛ بمعنى : دَرَجُوا ،   وانقرضُوا بالموتِ .  <br />
قوله : { مِنْ وَرَائِي } هذا متعلِّقٌ في قراءة الجمهور بما تضمَّنَهُ الموالي من معنى الفعلِ ،   أي : الذين يلون الأمْرَ بعدي ،   ولا يتعلَّق ب  " خِفْتُ "  لفساد المعنى ،   وهذا يدل على أن يُراد ب " ورائي "  معنى : خَلْفي ،   وبعدي ،   وأما في قراءة  "  خَفَّتِ  "  بالتشديد{[21396]} ،   فيتعلَّق الظرف بنفس الفعل ،   ويكون  "  وَرَائِي  "  بمعنى قُدَّامي ،   والمعنى : أنهم خفُّوا قدَّامهُ ،   ودرجُوا ،   ولم يبق منهم من به تقوٍّ واعتضادٌ ،   ذكر هذين المعنيين الزمخشريُّ .  <br />
والمَوالِي : بنُو العمِّ يدلُّ على ذلك تفسيرُ الشَّاعر لهم بذلك في قوله : [ البسيط ]<br />
مَهْلاً ،   بَنِي عمِّنَا ؛ مَهْلاً موالينَا *** لا تَنْبُشُوا بَيْننا ما كان مَدْفُوناً{[21397]}<br />
وقال آخر : [ الوافر ]<br />
ومَوْلى قد دفعتُ الضَّيْمَ عَنْهُ *** وقَدْ أمْسَى بمنزلةِ المَضِيمِ{[21398]}<br />
وهو قولُ الأصمِّ .  <br />
وقال مجاهدٌ : العَصَبَة{[21399]} .  <br />
وقال أبو صالحٍ : الكلالة{[21400]} .  <br />
وقال ابنُ عبَّاس والحسن والكلبيُّ{[21401]} : الورثة .  <br />
وعن أبي مسلمٍ : المولى يرادُ به النَّاصرُ ،   وابن العمِّ ،   والممالك ،   والصَّاحب ،   وهو هنا من يقُوم بميراثه مقام الولد ،   والمختار ،   أنَّ المراد من الموالي الذين يخلفُون بعده ،   إما في السِّياسة ،   أو في المال ،   أو في القيام بأمر الدِّين ؛ وهو يدلُّ على معنى القُرْبِ والدُّنوِّ ،   ويقالُ : وليته أليه ولْياً ،   أي : دَنَوتُ منهُ ،   وأوليْتُهُ إيَّاهُ ،   وكُل ممَّا يليكَ ،   وجلستُ ممَّا يليهِ ،   ومنهُ الوَلْيُ ،   وهو المطرُ الذي يلي الوسميَّ ،   والوليَّة : البرذعةُ [ الَّتي ]{[21402]} تلي ظهْر الدَّابَّة ،   ووليُّ اليتيم ،   ووليُّ القتيل ؛ لأنَّ من تولَّى أمراً ،   فقد قرُبَ منه .  <br />
وقوله تعالى : { فَوَلِّ وَجْهَكَ شَطْرَ المسجد الحرام } [ البقرة : 144 ] من قولهم : ولاه بركنه ،   أي جعله ما يليه ،   وأما ولَّى عنِّي ،   إذا أدبر ،   فهو من باب تثقيل الحشو للسلْب ،   وقولهم : فلان أولى من فلان ،   أي : أحق ؛ أفعل التفضيل من الوالي أو الولي ،   كالأدنى ،   والأقرب من الدَّاني ،   والقريب ،   وفيه معنى القرب أيضاً ؛ لأنَّ من كان أحقَّ بالشيء ،   كان أقرب إليه ،   والمولى : اسمٌ لموضعِ الولي ،   كالمرمى والمبنى : اسمٌ لموضعِ الرَّمي والبناءِ .  <br />
والجمهورُ على  "  وَرَائِي  "  بالمدِّ ،   وقرأ ابنُ كثير{[21403]} -في رواية عنه-  "  وَرَايَ "  بالقصر ،   ولا يبعدُ ذلك عنه ،   فإنه قد قصر { شُرَكَايَ } [ النحل : 27 ] في النَّحل ؛ كما تقدَّم ،   وسيأتي أنَّه قرأ { أَن رَّآهُ استغنى } في العلق [ الآية : 7 ]  ؛ كأنه كان يُؤْثِرُ القصْر على المدِّ ؛ لخفَّته ،   ولكنَّه عند البصريين لا يجوزُ سعةً .  <br />
قوله : { وَكَانَتِ امرأتي عَاقِراً } أي : لا تَلِدُ ،   والعَقْرُ في البدن : الجُرح ،   وعقرتُ الفرس بالسَّيف ،   ضربتُ قوائِمهُ .  <br />
قوله { مِنْ لدُنْكَ } يجوز أن يتعلق ب  "  هَب  "  ويجوزُ أن يتعلَّق بمحذوفٍ على أنَّه حال من  "  وليًّا  "  لأنه في الأصل صفةٌ للنكِرةِ ،   فقُدِّم عليها .  <br />
{ يَرِثُنِي وَيَرِثُ } : قرأ أبو عمرو{[21404]} ،   والكسائيُّ بجزم الفعلين على أنَّهما جوابٌ للأمر ؛ إذ تقديره : إن يهبْ ،   يرث ،   والباقون برفعهما ؛ على انَّهما صفةٌ ل  " وليًّا  "  .  <br />
وقرأ عليٌّ -رضي الله عنه- وابنُ عبَّاسٍ ،   والحسن{[21405]} ،   ويحيى بن يعمر ،   والجحدريُّ ،   وقتادةُ في آخرين :  "  يَرِثُنِي  "  بياء الغيبة ،   والرَّفع ،   وأرثُ مسنداً لضمير المتكلِّم .  <br />
فصل فيما قرئ به من قوله : { يَرِثُنِي وَيَرِثُ }<br />
قال صاحب  "  اللَّوامح  "  :  "  في الكلام تقديم وتأخيرٌ ؛ والتقدير : يرثُ نُبُوَّتِي ،   إنْ منُّ قبلهُ وأرِثُ مالهُ ،   إن مات قبلي  "  .   ونُقِلَ هذا عن الحسن .  <br />
وقرأ عليٌّ أيضاً ،   وابنُ عبَّاس ،   والجحدريُّ{[21406]}  "  يَرِثُني وارثٌ  "  جعلوه اسم فاعلٍٍ ،   أي : يَرثُنِي به وارثٌ ،   ويُسَمَّى هذا  "  التجريد  "  في علم البيان .  <br />
وقرأ{[21407]} مجاهدٌ  "  أوَيْرِثٌ  "  وهو تصغيرُ  "  وارِثٍ  "  والأصل :  "  وُوَيْرِثٌ  "  بواوين ،   وجب قلبُ أولاهما همزة ؛ لاجتماعهما متحركين أول كلمةٍ ،   ونحو  " أوَيْصِلٍ  "  تصغير  "  واصلٍ  "  والواوُ الثانيةُ بدلٌ عن ألفِ  "  فاعلٍ  "  و  " أوَيْرِثٌ "  مصروفٌ ؛ لا يقال : ينبغي أن يكون غير مصروفٍ ؛ لأنَّ فيه علتين : الوصفيَّة ،   ووزن الفعل ،   فإنه بزنة  "  أبَيْطِر  "  مضارع  "  بَيْطَرَ  "  وهذا ممَّا يكون الاسم فيه منصرفاً في التكبير ممتنعاً في التصغير ،   لا يقالُ ذلك لأنه غلطٌ بيِّنٌ ؛ لأنَّ  "  أوَيْرِثاً  "  وزنه فُويْعِلٌ ،   لا أفَيْعلٌ ؛ بخلاف  "  أحَيْمِر  "  تصغير  " أحْمَرَ "  .  <br />
وقرأ الزهريُّ{[21408]} :  "  وارثٌ  "  بكسر الواو ،   ويعنون بها الإمالة .  <br />
قوله : { رَضِيًّا } مفعولٌ ثانٍ ،   وهو فعيلٌ بمعنى فاعلٍ ،   وأصله  "  رَضِيوٌ  "  لأنه من الرِّضوان .  <br />
فصل<br />
معنى قوله : { فَهَبْ لِي مِن لَّدُنْكَ وَلِيّاً } أعطني من أبناء .  <br />
واعلم أنَّ زكريَّا -عليه السلام- قدَّم السؤال ؛ لأمور ثلاثة :<br />
الأول : كونه ضعيفاً .  <br />
والثاني : أن الله تعالى ما ردَّ دعاءه .  <br />
والثالث : كونُ المطلُوب سبباً للمنفعة في الدِّين ،   ثم بعد ذلك صرَّح بالسُّؤال .  <br />
أمَّا كونه ضعيفاً ،   فالضَّعيف : إمَّا أن يكون في الباطن ،   أو في الظَّاهر ،   والضَّعْف في الباطن أقوى من ضعف الظَّاهر ،   فلهذا ابتدأ ببيان ضعف الباطن ،   فقال : { وَهَنَ العظم مِنِّي } وذلك لأنَّ العظم أصلبُ أعضاء البدن ،   وجعل كذلك لمنتفعين :<br />
الأولى : ليكون أساساً وعمداً يعتمد عليها بقيَّة الأعضاء ؛ لأنَّها موضوعة على العظام ،   والحاملُ يجبُ أن يكون أقوى من المحمول عليه .  <br />
الثاني : أنَّها في بعض المواضع وقاية لغيرها .  <br />
واحتج أصحابُ القول الأوَّل أنَّه إذا .   .   .  {[21409]} أوّلاً ،   ثم ردَّ بأنها تكونُ كغيرها من الأعضاء كعظام الصَّلف وقحف الرأس ،   وما كان كذلك ،   فيجبُ أن يكون صلباً ؛ ليصبر على ملاقاة الآفاتِ ،   ومتى كان العظم صلبا ،   فمتى وصل الضعف إليه ،   كان ضعف ما عداه مع رخاوته أولى ؛ ولأنّ العظم حاملٌ لسائر الأعضاء ،   فوصولُ الضعف إلى الحامل موجبٌ لوصوله إلى المحمول ،   فلهذا خصَّ العظم بالوهْنِ من بين سائر الأعضاء .  <br />
وأما ضعف الظاهر ،   فلاستيلاء ضعف الباطن عليه ،   وذلك ممَّا يزيدُ الدُّعاء تأكيداً ؛ لما فيه من الارتكان على حول الله وقوته .  <br />
وأما كونهُ غير مردُود الدُّعاءِ ،   فوجه توسله به من وجهين :<br />
الأول : أنَّه إذا قبله أوَّلاً ،   فلو ردَّه ثانياً ،   لكان الردُّ محبطاً للإنعام الأول ،   والمنعم لا يسعى في إحباط إنعامه .  <br />
والثاني : أنَّ مخالفة العادةِ تشقُّ على النَّفس ،   فإذا تعوَّد الإنسانُ إجابة الدُّعاء ،   فلو ردَّ بعد ذلك ،   لكان ذلك في غاية المشقَّة ،   والجفاء ممن يتوقع منه الإنعام يكون أشقَّ ،   فكأنَّ زكريَّا -عليه السلام- قال : إنك إن رددَتَّنِي بعدما عودتَّني القُبول مع نهايةِ ضعفي ،   كان ذلك بالغاً إلى النِّهاية القصوى في [ ألم ]{[21410]} القلب ،   فقال : { وَلَمْ أَكُنْ بِدُعَآئِكَ رَبِّ شَقِيّاً } .  <br />
تقولُ العربُ : سعد فلانٌ بحاجته : إذا ظَفِر بها ،   وشَقِي بها : إذا خَابَ ،   ولم [ يَبْلُغْهَا ]{[21411]} .  <br />
وأمَّا كون المطلُوب منتفعاً به في الدِّين ،   فهو قوله : { وَإِنِّي خِفْتُ الموالي مِن وَرَآئِي } .  <br />
فصل في اختلافهم في المراد من قوله : { خِفْتُ الموالي }<br />
قال ابن عباس والحسن : الموالي : الورثة{[21412]} وقد تقدم .  <br />
واختلفوا في خوفه من الموالي{[21413]} ،   فقيل : خافهم على إفساد الدِّين .  <br />
وقيل : خاف أن ينتهي أمرُه إليهم بعد موته في مالٍ ،   وغيره ،   مع أنَّه عرف من حالهم قصورهم في العلم والقدرة عن القيام ببعضه .  <br />
وقيل : يحتمل أن يكون الله قد أعلمه أنَّه لم يبقَ من أنبياء إسرائيل نبيٌّ له أبٌ إلاَّ نبيٌّ واحدٌ ،   فخاف أن يكون ذلك الواحدُ من بَني عمِّه ،   إذا لم يكن له ولدٌ ،   فسأل الله أن يهب له ولداً ،   يكونُ هو ذلك النبيِّ ،   والظاهرُ يقتضي أن يكون خائفاً في أمر يهتمُّ بمثله الأنبياء ولا يمتنع أن يكون زكريَّا كان إليه مع النبوة الربانيَّة من جهة الملك ؛ فخاف منهم بعده على أحدهما أو عليهما .  <br />
وقوله : { خِفْتُ } خرج على لفظ أصل الماضي ،   لكنه يفيد أنه في المستقبل أيضاً ؛ كقول الرجل : قد خفتُ أن يكون كذا ،   أي :  "  أنَا خَائِفٌ  "  لا يريدُ أنه قد زال الخوف عنه .  <br />
قوله : { وَكَانَتِ امرأتي عَاقِراً } أي : أنَّها عاقرٌ في الحال ؛ لأنَّ العاقر لا يجوزُ [ أن تحبل في العادة ]{[21414]} ،   ففي الإخبار عنه بلفظ الماضي إعلامٌ بتقادُمِ العهد في ذلكَ ،   والغرضُ من هذا بيانُ استبعاد حصول الولدِ ،   فكان إيرادهُ بلفظ الماضِي أقوى ،   وأيضاً : فقد يوضعُ الماضي ،   أي : مكان المستقبل ،   وبالعكس ؛ قال الله تعالى : { وَإِذْ قَالَ الله يا عيسى ابن مَرْيَمَ أَأَنتَ قُلتَ لِلنَّاسِ اتخذوني وَأُمِّيَ إلهين } [ المائدة : 116 ] .  <br />
وقوله : { مِن وَرَآئِي } قال أبو عبيدة : من قُدَّامي ،   وبين يديَّ .  <br />
وقال آخرون : بعد موتي .  <br />
فإن قيل : كيف علم حالهم من بعده ،   وكيف علم أنَّهُمْ يقون بعده ،   فضلاً عن أن يخاف شرَّهم ؟ .  <br />
فالجوابُ{[21415]} : أنه قد يعرفُ ذلك بالأمارات والظنّ في حصول الخوف ،   وربما عرف ببعض الأمارات استمرارهم على عادتهم في الفساد .  <br />
وقوله : { فَهَبْ لِي مِن لَّدُنْكَ وَلِيّاً } الأكثر على أنه طلب الولد ،   وقيل : بل طلب من يقُوم مقامه ،   ولداً كان ،   أو غيره .  <br />
والأول أقربُ ؛ لقوله تعالى في سورة آل عمران ؛ حكاية عنه { رَبِّ هَبْ لِي مِن لَّدُنْكَ ذُرِّيَّةً طَيِّبَةً } [ آل عمران : 38 ] .  <br />
وأيضاً : فقوله ها هنا  "  يَرُثُنِي  "  يؤيِّده .  <br />
وأيضاً : يؤيِّده قوله تعالى في سورة الأنبياء : { وَزَكَرِيَّآ إِذْ نادى رَبَّهُ رَبِّ لاَ تَذَرْنِي فَرْداً } [ الأنبياء : 89 ] فدلَّ على أنَّه سأل الولد ؛ لأنَّه أخبر ها هنا أنَّ له مواليَ ،   وأنَّه غيرُ منفردٍ عن الورثةِ ،   وهذا إن أمكن حمله على وارثٍ يصلح أن يقوم مقامه ،   لكنَّ حمله على الولد اظهرُ .  <br />
واحتجَّ أصحابُ القول [ الثالث ]{[21416]} بأنَّه لما بشِّر بالولد ،   استعظمه على سبيل التعجُّب ؛ وقال  "  أنَّى يكونُ لِي غلامٌ  "  ولو كان دعاؤُه لطلب الولد ،   ما استعظم ذلك .  <br />
وأجيبَ بأنَّه -عليه السلام- سأل عمَّا يوهب له ،   أيوهبُ له هو وامرأتهُ على هيئتهما ؟ أو يوهبُ له بأن يُحَوَّلا شابَّيْن ،   يُولد لمثلهما ؟ ! وهذا يُحْكَى عن الحسن .  <br />
وقيل : إنَّ قول زكريَّا -عليه السلام- في الدُّعاء  "  وكَانتِ امْرَاتِي عاقراً  "  إنما سأل ولداً من غيرها أو منها ؛ بأن يصلحها الله تعالى للولدِ ،   فكأنَّه -عليه السلام- قال : أيسْتُ أن يكون لي منها ولدٌ ،   فهبْ لي من لدنك وليًّا ،   كيف شئت : إمَّا بأن تصلحها للولادةِ ،   وإمَّا أن تهبهُ لي من غيرها ،   فلمَّا بُشِّر بالغلام ،   سأل أن يرزق منها ،   أو من غيرها ،   فأخبر بأنه يرزقه منها .  <br />
