قال {  وَإِنْ خِفْتُمْ أَلاَّ تُقْسِطُواْ فِي الْيَتَامَى } ( 3 ) لأنه من  " أَقْسَطَ "   " يُقْسِطُ "  .  و " الإِقْساطُ "  :  العَدْل .  وأما  " قَسَطَ "  فإنَّه  " جَار "  قال {  وَأَمَّا الْقَاسِطُونَ فَكَانُواْ لِجَهَنَّمَ حَطَباً } ف " أَقْسَطَ "  :  عَدلَ و " قَسَطَ "  :  جارَ .  قال {  وَأَقْسِطُواْ إِنَّ اللَّهَ يُحِبُّ الْمُقْسِطِينَ } . <br />
وقال {  مَثْنَى وَثُلاَثَ وَرُبَاعَ فَإِنْ خِفْتُمْ أَلاَّ تَعْدِلُواْ [ 92ب ] فَوَاحِدَةً } ( 3 ) يقول :   " فانكِحوا واحدة {  أَوْ مَا مَلَكَتْ أَيْمَانُكُمْ } .  أي :  انكحوا ما ملكت أيمانكم .  وأما ترك الصرف في {  مَثْنَى وَثُلاَثَ وَرُبَاعَ } ( 3 ) فانه عدل عن  " اثنين "  و " ثَلاثٍ "  و " أَربعٍ "  كما انه من عدل  " عُمَر "  عن  " عامِر "  لم يصرف .  وقال تعالى {  أُوْلِي أَجْنِحَةٍ مَّثْنَى وَثُلاَثَ وَرُبَاعَ } فنصب .  وقال {  أَن تَقُومُواْ لِلَّهِ مَثْنَى وَفُرَادَى } فهو معدول كذلك ،  ولو سميت به صرفت لأنه إذا كان اسما فليس في معنى  " اثنين "  و " ثلاثة "  و " أربعة "  .  كما قال  " نَزَالِ "  حين كان في معنى  " انزِلوا "  وإذا سميت به رفعته . <br />
قال الشاعر :  [ من الوافر وهو الشاهد الثاني والستون بعد المائة ] : <br />
أَحَمَّ اللّهُ ذلِكَ مِنْ لِقاءٍ *** أُحادَ أُسادَ في شَهْرٍ حَلال<br />
وقال [ من الطويل وهو الشاهد الثالث والستون بعد المائة ] : <br />
وَلَكِنَّما أَهْلِي بِوادٍ أَنِيسُه *** ذِئابٌ تَبَغَّى الناسَ مَثْنَى وَمْوحَدا<br />
وقال تعالى {  فَانكِحُواْ مَا طَابَ لَكُمْ مِّنَ النِّسَاءِ } ( 3 ) يقول :   " لِيَنْكِحَ كُلٌّ واحدٍ منكُمْ كلَّ واحدةٍ من هذهِ العِدَّة "  كما قال تعالى {  فَاجْلِدُوهُمْ ثَمَانِينَ جَلْدَةً } يقول :   " فاجْلُدُوا كلَّ واحدٍ منهم "  .
