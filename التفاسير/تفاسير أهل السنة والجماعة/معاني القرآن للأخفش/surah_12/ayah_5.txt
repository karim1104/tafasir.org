وقال :  {  فَيَكِيدُواْ لَكَ كَيْداً } ( 5 ) ،  أي :  فيتخذوا لك كيدا .  وليست مثل :  {  إِن كُنتُمْ لِلرُّؤْيَا تَعْبُرُونَ } .  أراد أن يوصل الفعل [ 138 ء ] إليها باللام ،  كما يوصل ب " إلى "  ،  كما تقول :   " قَدَّمْتُ لَهُ طَعاماً "  ،  تريد :   " قَدَّمْتُ إِلَيْهِ "  .  وقال :  {  يَأْكُلْنَ مَا قَدَّمْتُمْ لَهُنَّ } ،  ومثلُه :  {  قُلِ اللَّهُ يَهْدِي لِلْحَقِّ } ،  وإِنْ شِئْتَ كان :  {  فَيَكِيدُواْ لَكَ كَيْداً } ،  في معنى :   " فَيَكِيدوك "  ،  وتجعل اللام مثل :  {  لِرَبِّهِمْ يَرْهَبُونَ } .  وقوله :  {  لِرَبِّهِمْ يَرْهَبُونَ } ،  إنِّما هو :   " لِمَكانِ رَبِّهِمْ يَرْهَبُون "  .
