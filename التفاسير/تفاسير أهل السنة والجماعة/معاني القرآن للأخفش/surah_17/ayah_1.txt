[ 145 ء ] قال :  {  سُبْحَانَ الَّذِي أَسْرَى } ،  ( 1 ) لأنك تقول  " أَسْرَيْتُ "  و " سَرَيْتُ "  . <br />
وقال :  {  إِنَّهُ هُوَ السَّمِيعُ البَصِيرُ } ،  ( 1 ) فهو فيما ذكروا - و الله أعلم - :  قُلْ يا مُحَمَّد ،   " سُبحانَ الذي أسْرى بِعَبْدِهِ "  ،  وقل :   " إِنَّهُ هُوَ السَّمِيعُ البَصِير "  .
