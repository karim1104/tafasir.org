وقال {  ولكنَّ الْبِرَّ مَنْ آمَنَ بِاللَّهِ وَالْيَوْمِ الآخِرِ وَالْمَلائِكَةِ وَالْكِتَابِ وَالنَّبِيِّينَ } ( 177 ) ثم قال {  وَآتَى الْمَالَ عَلَى حُبِّهِ } {  وَأَقَامَ الصَّلاةَ وَآتَى الزَّكَاةَ } ( 177 ) فهو على أول الكلام  " ولكنَّ البرَّ برُّ مَنْ آمَنَ باللّهِ وأقامَ الصلاةَ وآتى الزكاةَ "  ثم قال {  وَالْمُوفُونَ بِعَهْدِهِمْ إِذَا عَاهَدُواْ وَالصَّابِرِينَ } ( 177 ) ف " {  المُوفُونَ } رفع على  " ولكنَّ الموفين " * يريد  " بِرَّ الموفين "  فَلَما لم يذكر  " البرَّ "  أقام {  الموفونَ } مقام البرّ كما قال {  وَاسْألِ الْقَرْيَةَ } فنصبها على {  اسأل } وهو يريد  " أهلَ القرية "  ،  ثم نصب {  الصَّابِرِينَ } على فعل مضمر كما قال {  لكِنِ الرَّاسِخُونَ فِي الْعِلْمِ مِنْهُمْ وَالْمُؤْمِنُونَ } ثم قال {  وَالْمُقِيمِينَ } فنصب على فعل مضمر ثم قال {  وَالْمُؤْتُونَ الزَّكَاةَ } فيكون رفعا على الابتداء أو بعطفه على  " الراسخين "  .  قال الشاعر :  [ من الكامل وهو الشاهد السابع والستون ] : <br />
لا يَبْعُدَنْ قَوْمِي الذينَ هُمُ *** سُمُّ العُداةِ وآفَةُ الجَزْرِ<br />
النَازِلينَ بِكُلِّ مُعْتَرَكٍ *** والطَّيِّبُونَ معاقدَ الأُزْر<br />
ومنهم من يقول  " النازلون "  و " الطيبين "  .  ومنهم من يرفعهما جميعا وينصبهما جميعا [ 68ب ] كما فسرت لك .  ويكون {  الصَّابِرِينَ } معطوفا على ذَوِي الْقُرْبَى } {  وَآتَى الصَّابِرِينَ } . <br />
وقال {  فِي الْبَأْسَاءِ والضَّرَّاءِ } ( 177 ) فبناه على  " فَعْلاء "  وليس له  " أَفْعَلُ "  لأنه اسم ،  كما قد جاء  " أَفْعَل "  في الأسماء ليس معه  " فَعْلاء "  نحو  " أَحْمَدُ "  .  وقد قالوا  " أَفْعَلُ "  في الصفة ولم يجئ له  " فَعْلاءُ "  ،  قالوا :   " أَنْتَ مِنْ ذاكَ أَوْجلُ "  و " أَوْجَرُ "  ولم يقولوا :   " وَجْلاءُ "  ولا  " وَجْراءُ "  وهما من الخوف .  و[ منه ]  " رجلٌ أَوْجَلُ "  و " أَوْجَرُ "  .
