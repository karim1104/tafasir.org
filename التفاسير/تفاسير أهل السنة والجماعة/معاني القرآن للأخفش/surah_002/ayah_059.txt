وقال {  فَأَنزَلْنَا عَلَى الَّذِينَ ظَلَمُواْ رِجْزاً مِّنَ السَّمَاءِ } ( 59 ) وقال {  وَالرُّجْزَ فَاهْجُرْ } وقال بعضهم {  والرِجْزَ }* .  وذكروا أن  " الرُجْزَ "  :  صنم كانوا يعبدونه فأما  " الرِجْزَ "  فهو :   " الرِجْسُ .  [ والرِجْسُ :  النَجَس ] وقال {  إِنَّمَا الْمُشْرِكُونَ نَجَسٌ } و " النَجَسُ "  :  القَذَر .
