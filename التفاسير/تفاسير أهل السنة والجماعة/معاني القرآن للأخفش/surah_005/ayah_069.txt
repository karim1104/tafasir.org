وقال :  {  وَالصَّابِئُونَ وَالنَّصَارَى } ( 69 ) وقال في موضع آخر {  والصّابِئِينَ } والنصب القياس على العطف على ما بعد {  إِنّ } فأما هذه فرفعها على وجهين كأن قوله {  إِنَّ الَّذِينَ آمَنُواْ } ( 69 ) في موضع رفع في المعنى لأنه كلام مبتدأ لأَنَّ قَوْلَهُ :   " إنَّ زَيْداً مُنْطَلِقٌ "  و " زَيْدٌ مُنْطَلِقٌ "  من غير أن يكون فيه  " إنّ "  في المعنى سواء [ 105 ء ] ،  فإن شئت إذا عطفت عليه شيئا جعلته على المعنى .  كما قلت :   " إنَّ زيداً مُنْطَلِقٌ وعمرٌو "  .  ولكنه إذا جعل بعد الخبر فهو أحسن وأكثر .  وقال بعضهم :   " لما كان قبله فعل شبه في اللفظ بما يجري على ما قبله ،  وليس معناه في الفعل الذي قبله وهو {  الَّذِينَ هَادُواْ } ( 69 ) أجراه عليه فرفعه به وإن كان ليس عليه في المعنى ذلك أنه تجيء أشياء في اللفظ لا تكون في المعاني ،  منها قولهم :   " هذا جُحْرُ ضَبٍّ خَرِبٍ "  وقولهم  " كَذَبَ عَلَيْكُمْ الحَجُّ "  يرفعون  " الحَجَّ "  ب " كَذَبَ "  ،  وإنما معناه  " عَلَيْكُم الحَجَّ "  نصب بأمرهم . <br />
وتقول :   " هذا حَبُّ رُمّانِي "  فتضيف  " الرُّمَانَ "  إلَيكَ وإِنَّما لَكَ  " الحَبُّ "  وليس لك  " الرُّمَانُ "  .  فقد يجوز أشباه هذا والمعنى على خلافه .
