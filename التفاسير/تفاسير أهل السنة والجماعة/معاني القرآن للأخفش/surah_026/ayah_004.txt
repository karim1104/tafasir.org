قال {  فَظَلَّتْ أَعْنَاقُهُمْ لَهَا خَاضِعِينَ } ( 4 ) يزعمون أنها على الجماعات نحو  " هذا عُنُقٌ من الناس "  يعنُون  " الكثير "  أو ذكّركما يذكر بعض المؤنث لما أضافه إلى مذكّر .  وقال الشاعر :  [ من الطويل وهو الشاهد السادس والخمسون بعد المئتين ] : <br />
باكَرْتُهَا والدِّيكُ يَدْعُو صباحَهُ *** إِذَا مَا بَنُو نَعْشٍ دَنَوْا فَتَصَوَّبُوا<br />
فجماعات هذا  " أَعْناقٌ "  أَوْ يكون ذكّره لإضافته إلى المذكّر كما يؤنّث لإضافته إلى المؤنث نحو قوله :  [ من الطويل وهو الشاهد السابع والخمسون بعد المئتين ] : <br />
وَتشْرَقُ بالقَوْلِ الذي قَدْ أَذَعْتَهُ *** كَما شَرِقَتْ صَدْرُ القَنَاةِ مِنَ الدَّمِ<br />
وقال آخر :  [ من الرجز وهو الشاهد الثامن والخمسون بعد المئتين ] : <br />
لَمَا رَأَى مَتْنَ السَّمَاءِ انْقَدَّتِ *** .  .  .  .  .  .  .  . <br />
وقال :  [ من الطويل وهو الشاهد التاسع والخمسون بعد المئتين ] : <br />
إِذَا القُنْبُضَاتُ طَوَّفْنَ بالضُّحى *** رَقَدْنَ عَلَيْهِنَّ الحِجَالُ المُسَجَّفُ<br />
[ 155 ب ] و " القُنْبُضُ "  :  القصير .  وقال آخر :  [ من الطويل وهو الشاهد الستون بعد المئتين ] : <br />
وإِنَّ امْرَءاً أَهْدَى إِلَيْكِ وَدُونَهُ *** من الأَرْضِ مَوْمَاةٌ وبَيْداَءُ خَيْفَقُ<br />
لمََحْقُوقَةٌ أَنْ تَسْتَجيِبي لِصَوْتِهِ *** وَأَنْ تَعْلَمِي أَنَّ المُعَانَ مُوَفَّقُ<br />
فأنَّث .  والمحقوق هو المرء .  وإنما أنث لقوله  " أَنْ تَسْتَجِيبِي لِصَوْتِهِ "  ويقولون :   " بَنَاتُ عُرْسٍ "  و " بَنَاتُ نَعْشٍ "  و " بَنُو نَعْشٍ "  وقالت امرأة من العرب  " أَنَا امْرُؤُ لا أُحِبُ الشَرَّ "  .  وذكر لرؤبة رجل فقال  " كانَ أَحَدَ بناتِ مَساجِدِ اللهِ "  كأنه جعله حصاة .
