وقال {  يُوصِيكُمُ اللَّهُ فِي أَوْلاَدِكُمْ لِلذَّكَرِ مِثْلُ حَظِّ الأُنْثَيَيْنِ } ( 11 ) فالمثل مرفوع على الابتداء وإنما هو تفسير الوصية كما قال {  وَعَدَ اللَّهُ الَّذِينَ آمَنُواْ وَعَمِلُواْ الصَّالِحَاتِ لَهُم مَّغْفِرَةٌ وَأَجْرٌ عَظِيمٌ } فسر الوعد يقول :   " هكَذا وَعَدَهُمْ "  أي :  قال  " لَهُمْ مَغْفِرَةٌ "  .  قال الشاعر :  [ من الطويل وهو الشاهد الخامس والستون بعد المائة ] : <br />
عَشِيَّةَ ما وَدَّ ابْنُ غَرَّاءَ أْمُّهُ *** لَها مِنْ سِوانا إذْ دَعَا أَبَوانِ<br />
قال {  فَإِن كُنَّ نِسَاءً } ( 11 ) فترك الكلام الأول وقال  " إذا كان المتروكات نساءً "  نصب وكذلك {  وَإِن كَانَتْ وَاحِدَةً } ( 11 ) . <br />
وقال {  وَلأَبَوَيْهِ لِكُلِّ وَاحِدٍ مِّنْهُمَا السُّدُسُ } ( 11 ) فهذه الهاء التي في  " أبويه "  ضمير الميت لأنه لما قال {  يُوصِيكُمُ اللَّهُ فِي أَوْلاَدِكُمْ } ( 11 ) كان المعنى :  يوصي اللّهُ الميتَ قبلَ موتِهِ بأنَّ عَليْه لأَبويهِ كذا ولوَلَدِه كذا .  أي :  فلا يأخُذَنَّ إلاّ ماله . <br />
وقال {  فَإِن كَانَ لَهُ إِخْوَةٌ } ( 11 ) فيذكرون أن الأخوة اثنان ومثله  " إنّا فَعَلْنا "  وأنتما اثنان ،  وقد يشبه ما كان من شيئين وليس مثله ،  ولكن اثنين قد جعل جماعة [ في ] قول الله عز وجل {  إِن تَتُوبَا إِلَى اللَّهِ فَقَدْ صَغَتْ قُلُوبُكُمَا } وقال {  وَالسَّارِقُ وَالسَّارِقَةُ فَاقْطَعُواْ أَيْدِيَهُمَا } وذلك أن في كلام العرب إن كل شيئين من شيئين فهو جماعة وقد يكون اثنين في الشعر [ قال الشاعر ] : <br />
[ من الطويل وهو الشاهد السادس والستون بعد المائة ] : <br />
[ 94ء ] بِما فِي فُؤادَيْنا من الشَوْقِ والهَوى *** فَيُجْبَرُ مُنْهاضُ الفُؤَادِ المُشَعَّفُ<br />
وقال الفرزدق :  [ من الطويل وهو الشاهد السابع والستون بعد المائة ] : <br />
هُما نَفَثا فِي فِيَّ مِنْ فَمَوَيْهِما *** على النّابِحِ العاوِي أَشَدّ لِجامِ<br />
وقد يجعل هذا في الشعر واحدا .  قال :  [ من الرجز وهو الشاهد الثامن والستون بعد المائة ] : <br />
لا نُنْكِرُ القَتْلَ وقد سُبِينا *** في حَلْقِكُمْ عَظْمٌ وَقَدْ شُجِينا<br />
وقال الآخر :  [ من الوافر وهو الشاهد التاسع والستون بعد المائة ] : <br />
كُلُوا في بَعْضِ بَطْنِكُمُ تَعَفُّوا *** فإِنَّ زَمانَخكُمْ زَمَنَ خَمِيصُ<br />
ونظير هذا قوله :   " تِسْعُ مئة "  وإنما هو  " تِسْعُ مئات "  أو " مِئيِن "  فجعله واحدا ،  وذلك أن ما بين العشرة إلى الثلاثة يكون جماعة نحو :   " ثلاثة رجال "  و " عشرةُ رجال "  ثم جعلوه في  " المِئينَ "  واحدا . <br />
وقال {  مِن بَعْدِ وَصِيَّةٍ يُوصِي بِهَا } ( 11 ) لأنه ذكر الرجل حين قال {  وَوَرِثَهُ أَبَواهُ } ( 11 ) وقال بعضهم {  يُوصِى } وكلٌّ حسن .  ونظير {  يُوصِي } بالياء .
