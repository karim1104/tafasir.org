وقال {  لَيْسُواْ سَوَاءً مِّنْ أَهْلِ الْكِتَابِ } ( 113 ) لأنه قد ذكرهم ثم فسره فقال :  {  مِّنْ أَهْلِ الْكِتَابِ أُمَّةٌ قَآئِمَةٌ يَتْلُونَ آيَاتِ اللَّهِ } ( 113 ) ولم يقل  " وَأُمَّةٌ على خلافِ هذهِ الأُمَّةِ "  لأنه قد ذكر كل هذا قبل .  وقال تعالى {  مِّنْ أَهْلِ الْكِتَابِ } فهذا قد دل على أمة خلاف هذه . <br />
وقال تعالى {  آنَاءَ اللَّيْلِ } ( 113 ) وواحد  " الآناءِ "  مقصور  " إنَى "  فاعلم وقال بعضهم :   " إِنْي "  كما ترى و " إنْوٌ "  وهو ساعاتُ اللَيْل .  قال الشاعر :  [ من البسيط وهو الشاهد الثامن والخمسون بعد المائة ] : <br />
السَّالِكُ الثَّغْرَ مَخْشِيّاً مَوارِدُهُ *** فِي كُلِّ إنيٍ قَضاهُ اللَّيلُ يَنْتَعِلُ<br />
قال :  وَسِمْعُته  " يَخْتَعِلُ " * .
