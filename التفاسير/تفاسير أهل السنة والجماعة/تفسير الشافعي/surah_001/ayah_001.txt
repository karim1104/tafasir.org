قوله عز وجل { بِسْمِ اِللَّهِ اِلرَّحْمانِ اِلرَّحِيمِ { 1 } } : <br />
1- قال الشافعي :  يتعوذ{[1]} فيقول :  أعوذ بالله من الشيطان الرجيم ،  ثم يقرا مرتلا بأم القرآن ويبتدئها ب { بِسْمِ اِللَّهِ اِلرَّحْمانِ اِلرَّحِيمِ } ،  لأن النبي صلى الله عليه وسلم قرا بأم القرآن ؛  وعدها آية{[2]}  . (  مختصر المزني :  14  .  )<br />
2- قال الشافعي :  { بِسْمِ اِللَّهِ اِلرَّحْمانِ اِلرَّحِيمِ } الآية السابعة ،  فإن تركها ،  أو بعضها ،  لم تجزه الركعة التي تركها فيها  . <br />
قال الشافعي :  وبلغني أن ابن عباس رضي الله عنهما كان يقول :  إن رسول الله صلى الله عليه وسلم كان يفتتح القراءة ب { بِسْمِ اِللَّهِ اِلرَّحْمانِ اِلرَّحِيمِ } {[3]}  . ( الأم : 1/107  .  ون معرفة السنن والآثار :  1/510 و 1/515  .  )<br />
قال الشافعي :  أخبرنا عبد المجيد بن عبد العزيز{[4]} ،  عن ابن جُريج{[5]} قال :  أخبرني أَبي ،  عن سعيد ابنِ جُبير{[6]} { وَلَقَدَ ـاتَيْنَاكَ سَبْعًا مِّنَ اَلْمَثَانِى }{[7]} قال :  هي أم القرآن  .  قال أبي :  وقرأها عَلَيَّ سعيد بن جبير حتى ختمها ،  ثم قال :  { بِسْمِ اِللَّهِ اِلرَّحْمانِ اِلرَّحِيمِ } الآية السابعة  .  قال سعيد :  فقرأها عَلَيَّ ابن عباس كما قرأتها عليك ثم قال :  { بِسْمِ اِللَّهِ اِلرَّحْمانِ اِلرَّحِيمِ } الآية السابعة  .  قال ابن عباس :  فَذَخَرَهَا{[8]} لكم ،  فما أخرجها لأحد قبلكم{[9]}  .
