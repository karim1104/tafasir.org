{  أَنْ تُبْسَلَ نَفْسٌ } أي :  تُرْتَهن وتسلم ،  قال عَوْف ابن الأحْوَص بن جعفر : <br />
وَإبْسَالِي بَنِيَّ بِغَيْرِ جُرْمٍ *** بعوْنَاهُ وَلاَ بِدَمٍ مُرَاقِ<br />
بعوناه ،  أي :  جنيناه ،  وكان حَمل عن غنيٍّ لبنى قُشَيْرٍ دمَ ابنى السَّجَفِيّة ،  فقالوا :  لا نرضى بك ،  فرهنهم بنيه ،  قال النابغة الْجَعْدِيُّ : <br />
وَنحنُ رَهَنَّا بالاُفَاقَةَ عَامِراً *** بما كان في الدَّرْدَاء رَهْناً فأُبْسِلاَ<br />
وقال الشَّنْفَرَي : <br />
هنَالك لا أَرْجو حياةً تَسُرُّنيِ *** سَمِيرَ الليالي مُبْسَلا بالجزائرِ<br />
أي أبد الليالي .  وكذلك في آية أخرى :  {  أُوَلئِكَ الّذِينَ أُبْسِلُوا } {  70 "  . <br />
{  وَإنْ تَعْدِلْ كُلَّ عَدْلٍ لاَ يُؤْخَذْ مِنْهَا } مجازه :  إنْ تقسط كل قِسْطٍ لا يُقْبَل منها .  لأنّما التوبةُ في الحياة .
