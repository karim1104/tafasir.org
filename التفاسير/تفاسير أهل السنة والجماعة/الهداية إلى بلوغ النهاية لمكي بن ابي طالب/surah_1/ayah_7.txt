قوله :  ( صِرَاطَ الذِينَ أَنْعَمْتَ عَلَيْهِمْ ) [ 6 ] . <br />
صراط بدل(   {[410]} ) من الأول(   {[411]} ) . <br />
والذين أنعم عليهم هم الأنبياء صلوات الله عليهم والصدِّيقون والصالحون بدلالة قوله :  ( فَأُوْلَئِكَ مَعَ الذِينَ أَنْعَمَ اللَّهُ عَلَيْهِم مِّنَ النَّبِيئينَ(   {[412]} ) وَالصِّدِّيقِينَ وَالشُّهَدَاءِ وَالصَّالِحِينَ )(   {[413]} )(   {[414]} ) . <br />
وقيل :  هم أصحاب النبي [ عليه السلام ](   {[415]} ) ،  قاله الحسن . <br />
وقيل(   {[416]} ) :  هم المؤمنون من بني إسرائيل الذين لم يغيروا ولا بدلوا ،  بدليل قوله :  ( يَا بَنِي إِسْرَائِيلَ اذْكُرُوا نِعْمَتِيَ التِّي أَنْعَمْتُ عَلَيْكُمْ )(   {[417]} ) .  فلذلك قال هنا :  ( صِرَاطَ الذِينَ أَنْعَمْتَ عَلَيْهِمْ ) . <br />
وقيل :  هم المسلمون(   {[418]} ) . <br />
وقال أبو العالية :   " هم محمد [ عليه السلام ](   {[419]} ) وأبو بكر وعمر(   {[420]} ) "  . <br />
وقال قتادة(   {[421]} ) :   " هم الأنبياء خاصة(   {[422]} ) "  . <br />
وقال ابن عباس :   " هم أصحاب موسى قبل أن يبدلوا(   {[423]} ) "  . <br />
وهذا دعاء أمر الله عز وجل رسول الله صلى الله عليه وسلم أن يدعوا به وألا يكونوا(   {[424]} ) مثل المغضوب/ عليهم –وهم اليهود- ،  ولا مثل الضالين –وهم النصارى- ،  ولا على صراطهم(   {[425]} ) . <br />
ودخلت  " لا "  في قوله :  ( وَلاَ الضَّالِّينَ ) لئلا يتوهم أن ( الضَّالِّينَ ) عطف على ( الذِينَ ) في قوله :  ( صِرَاطَ الذِينَ )(   {[426]} ) .  فبدخول  " لا " (   {[427]} ) امتنع أن يتوهم متوهم ذلك إذ لا تقع  " لا "  إلا بعد نفي أو ما هو في معنى النفي . <br />
وقيل :   " لا "  زائدة(   {[428]} ) . <br />
وقيل :  هي تأكيد بمعنى(   {[429]} ) :   " غير(   {[430]} ) "  . <br />
ولم يجمع ( الْمَغْضُوبِ ) ،  لأنه في معنى الذين غضب عليهم/ فلا ضمير فيه إذ لا يتعدى(   {[431]} ) إلا بحرف جر .  فلو قدرت فيه ضميراً ،  كنت قد عديته إلى مفعولين أحدهما بحرف جر/ وهذا ليس يحسن فيه .  إنما تقول :   " غضبت على زيد "  و " غضب على زيد "  .  فالمخفوض يقوم مقام الفاعل .  وكذلك ( عَلَيْهِمْ ) ،  في موضع رفع يقوم مقام المفعول الذي لم يسم فاعله .  والهاء والميم(   {[432]} ) يعودان(   {[433]} ) على الألف واللام . <br />
والغضب من الله البعد من رحمته .  والضلال الحيرة . <br />
 " و( غَيْرِ المَغْضُوبِ ) "  خفض على النعت  " للذين " (   {[434]} ) من قوله ( صِرَاطَ الذِينَ ) ،  وحَسَنٌ ذلك لأنه شائع لا يراد به جمع بعينه فصار كالنكرة ،  فجاز نعته  " بغير "  ،  و " غير "  نكرة وإن أضيفت(   {[435]} ) إلى معرفة .  ويجوز أن تخفض  " غير "  على البدل(   {[436]} ) من [ الذين .  وقد قرئ ](   {[437]} ) بالنصب على الحال أو على الاستثناء ،  وقد شرحت هذا في كتاب :   " مشكل الإعراب(   {[438]} ) "  بأشبع من هذا . <br />
ويقول المأموم إذا سمع ( وَلاَ الضَّالِّينَ ) :  آمين .  ويقولها(   {[439]} ) وحده .  واختلف في قول الإمام إياها عن مالك(   {[440]} ) . <br />
و( آمين ) ،  قيل :  هو اسم من أسماء الله تعالى(   {[441]} ) . <br />
وقيل :  هو دعاء بمعنى :   " اللّهم استجب(   {[442]} ) "  . <br />
وقال ابن عباس والحسن :   " معنى  " آمين "  :  كذلك يكون "  .  وهي تمد وتقصر(   {[443]} ) لغتان(   {[444]} ) . <br />
والمؤمن داع .  فقد قال الله لموسى وهارون :  ( قَدُ اجِيبَت دَّعْوَتُكُمَا )(   {[445]} ) .  وموسى كان(   {[446]} ) هو الداعي ،  وهارون يؤمن ،  والمؤمن إذا قال :   " اللهم استجب "  فهو داع بالإجابة ،  وهو مبني(   {[447]} ) لوقوع موقع الدعاء/ وبني على حركة لالتقاء الساكنين وكان/ الفتح أولى به(   {[448]} ) لأن قبل آخره ياء(   {[449]} ) . <br />
وروى أبو هريرة(   {[450]} ) عن النبي [ عليه السلام ](   {[451]} ) قال :   " إِذَا قَالَ الإمَامُ ( غَيْرِ(   {[452]} ) المَغْضُوبِ عَلَيْهِمْ(   {[453]} ) وَلاَ الضَّالِّينَ ) ،  فَقُولُوا :  آمِينَ ،  فإنَّهُ مَنْ وَافَقَ قَوْلُهُ قَوْلَ الْمَلاَئِكَةِ غُفِرَ لَهُ ما تَقَدَّمَ مِنْ ذَنْبِهِ " (   {[454]} ) ،  أي من وافقه في الإجابة .
