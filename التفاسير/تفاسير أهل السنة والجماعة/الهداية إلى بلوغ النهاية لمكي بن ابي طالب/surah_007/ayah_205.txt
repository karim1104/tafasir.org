قوله :  {  واذكر ربك في نفسك }[ 205 ] ،  إلى آخرها . <br />
قوله :  {  تضرعا وخيفة{[26626]} }[ 205 ] :  مصدران .  {  والآصال }[ 205 ] :  جمع  " أُصُل "  ،  ك :   " طنب "  و " أطناب " {[26627]} .  وقال الفراء :  هو جمع  " أصيل "  ك :   " يمين "  و " أيمان " {[26628]} . <br />
وقيل :  [ الأصل ]{[26629]} جمع  " أصيل " {[26630]} . <br />
{  والآصال } :  جمع  " الأصل "  ،  وقد تجعل العرب  " الأصل "  واحدا ،  فيقولون :   " قد دنا الأصل " {[26631]} . <br />
ومعنى {  واذكر ربك } :  الدعاء{[26632]} ،  وهو أمر للمستمع للقرآن بأن يذكر الله في نفسه بالدعاء ،  ويعتبر بما يسمع ويتعظ . <br />
{  تضرعا } ،  أي :  تخشعا وتواضعا{[26633]} . <br />
{  وخيفة } ،  أي :  خوفا من الله{[26634]} . <br />
{  ودون الجهر } ،  أي :  واذكره دون الجهر ذكرا{[26635]} خفيا باللسان .  قال ذلك ابن زيد وغيره{[26636]} . <br />
قال الحسن :  كانوا يتكلمون في الصلاة حتى نزلت :  {  واذكر ربك في نفسك تضرعا وخيفة }{[26637]} .  أي :  مخافة منه{[26638]} . <br />
{  ودون الجهر من القول ( بالغدو{[26639]} ) والآصال }[ 205 ] . <br />
ما بين المغرب إلى العصر{[26640]} . <br />
وقيل :  هي العشي{[26641]} . <br />
{  ولا تكن من الغافلين }[ 205 ] . <br />
أي :  من اللاهين{[26642]} . <br />
قال مجاهد :  {  بالغدو } :  آخر الفجر ،  صلاة الصبح ،  {  والآصال } :  آخر العشي ،  صلاة العصر{[26643]} . <br />
وهذا إنما كان{[26644]} إذا كانت الفريضة ركعتين  " غدوة "  ،  وركعتين/ " عشية "  ،  قبل أن تفرض الصلوات الخمس{[26645]} .
