قوله :  {  ولله الاسماء الحسنى فادعوه بها }[ 180 ] الآية . <br />
 " الإلحاد "  في اللغة :  الجور والميل عن القصد{[26212]} . <br />
قال الكسائي :  يقال :   " ألحد "  :  عدل عن القصد .  و " لحد "  :  ركن إلى الشيء .  وعلى ذلك قرأ {  يلحدون{[26213]} } في  " النحل " {[26214]} ، [ يعني ] :  يركنون . <br />
واللغة الفصيحة ،   " ألحد "  الرجل في دينه :  إذا مال وجار .  و " ألحد "  القبر .  وقد تدخل [ كل ]{[26215]} واحدة منهما على الأخرى{[26216]} . <br />
ومعنى :  {  وذروا الذين يلحدون في أسمائه }[ 180 ] . <br />
قال بعض العلماء :  هو نهي من الله ،  ( عز وجل ) ،  أن يُدعى بما لا يجوز أن يوصف به ،  وذلك أنهم عدلوا بأسمائه فسموا ببعض اشتقاقها وبعض حروفها آلهتهم .  قالوا :   " اللات "  مشتق من  " الله "  .  وسموا ب :   " العزى "  ،  أخذوه من  " العزيز " {[26217]} . <br />
قال مجاهد :  أخذوا  " العزى "  من  " العزة " {[26218]} . <br />
قال ابن عباس :  {  يلحدون } :  يكذبون{[26219]} . <br />
وقال قتادة :  يشركون{[26220]} . <br />
وقال ابن زيد :  هذا منسوخ نسخه القتال{[26221]} . <br />
وقيل :  إن هذا محكم ،  وإنما هو تهديد{[26222]} ووعيد من الله ( عز وجل{[26223]} ) ،  لا أنه ( تعالى ){[26224]} ،  أمر نبيه ( عليه السلام{[26225]} ) ،  أن يتركهم يلحدون في آيات الله ( عز وجل{[26226]} ) ،  وهو مثل :  {  ذرهم{[26227]} ياكلوا ويتمتعوا }{[26228]} . <br />
