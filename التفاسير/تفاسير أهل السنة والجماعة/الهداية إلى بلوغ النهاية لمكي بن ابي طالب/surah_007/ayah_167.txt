ثم قال تعالى :  {  وإذ تأذن ربك ليبعثن عليهم إلى يوم القيامة }[ 167 ] . <br />
روى الأصبهاني عن أصحابه عن ورش :  ( تأذن ) ،  بتسهيل الهمزة{[25853]} . <br />
والمعنى :  واذكر ،  يا محمد ،  إذ أعلم ربك{[25854]} . <br />
فمعنى {  تأذن } :  أَعْلَمَ ،  والعرب تقول :   " تعلم "  بمعنى  " أعلم " {[25855]} . <br />
وقال مجاهد {  تأذن } :  قال{[25856]} . <br />
وقال قتادة {  تأذن ربك } :  أمر ربك{[25857]} . <br />
ومعنى :  {  ليبعثن عليهم }[ 167 ] . <br />
أي :  ليبعثن على اليهود {  من يسومهم سوء العذاب }[ 167 ] ،  وهو قتلهم إن لم يؤدوا الجزية ،  وذلتهم إن ودوها{[25858]} . <br />
قال ابن عباس :  هي الجزية ،  والذين يسومونهم{[25859]} :  محمد صلى الله عليه وسلم ،  وأمته ،  إلى يوم القيامة{[25860]} . <br />
قال ابن المسيب :  يستحب أن يبعث الأنباط في الجزية{[25861]} . <br />
{  إن ربك لسريع العقاب }[ 167 ] . <br />
أي :  لمن استوجب منهم العقوبة . <br />
{  وإنه لغفور رحيم }[ 167 ] . <br />
أي :  لساتر{[25862]} ذنوب من تاب ،  متعطف عليه{[25863]} .
