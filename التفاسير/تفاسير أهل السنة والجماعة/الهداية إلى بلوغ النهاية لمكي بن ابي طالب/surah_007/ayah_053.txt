قوله :  {  أو نرد }[ 53 ] ،  معطوف على [ قوله{[23895]} ] :  {  من شفعاء }[ 53 ] ،  أي :  أو هل{[23896]} نرد{[23897]} . <br />
وقرأ ابن أبي إسحاق :   " أو نرد "  ،  بالنصب{[23898]} ،  على معنى :  إلا أن نرد ،  كما قال{[23899]} : <br />
 .  .  .  .  . أو نموت فنعذرا{[23900]}<br />
وقوله {  فنعمل }[ 53 ] ،  جواب{[23901]} لقوله :  {  أو نرد } ،  أو عطف عليه ،  على قراءة من نصب (  " نرد "  ){[23902]} . <br />
وقرأ الحسن :   "  أو نرد فنعمل "  ،  بالرفع فيهما{[23903]} ،  على ( لفظ{[23904]} ) العطف على  " نُرد " {[23905]} .  ورفع  " نرد "  على الاستفهام كما ذكرنا{[23906]} . <br />
وقوله :  {  هل ينظرون إلا تاويله }[ 53 ] . <br />
/أي :  إلا ما وعدوا{[23907]} به في القرآن من العذاب . <br />
و{  ينظرون } ،  بمعنى :  ينتظرون{[23908]} . <br />
قال قتادة :  {  تاويله } ،  عاقبته{[23909]} . <br />
وقال مجاهد :  جزاءه{[23910]} . <br />
{  يوم ياتي تاويله }[ 53 ] ،  أي :  جزاؤه ،  {  يقول{[23911]} الذين نسوه من قبل }[ 53 ] ،  أي :  تركوه في الدنيا ،  {  قد جاءت رسل ربنا بالحق }{[23912]} . <br />
وقال ابن زيد :  {  تاويله } حقيقته{[23913]} ،  أي :  حقيقة القرآن فيما أوعدهم من العقاب . <br />
قال السدي :  {  الذين نسوه } ،  تركوه في الدنيا ،  لما رأوا ما أوعدهم أنبياؤهم ،  استيقنوا بالهلاك{[23914]} ،  وطلبوا الشفعاء والرجعة إلى الدنيا{[23915]} . <br />
{  قد خسروا أنفسهم }[ 53 ] . <br />
أي :  غَبَنوا أنفسهم حظوظها ببيعهم{[23916]} ما ذكر لهم من نعيم الآخرة الدائم بالخسيس من عرض الدنيا الزائل{[23917]} . <br />
{  وضل عنهم ما كانوا يفترون }[ 53 ] . <br />
أي :  أولياؤهم في الدنيا{[23918]} . <br />
وقال بعض أهل اللغة{[23919]} معناه{[23920]} :  هل ينظرون إلى ما يؤول إليه أمرهم من البعث ،  وعلى هذا تأولوا قول الله :  {  وما يعلم تاويله إلا الله }[ آل عمران :  7 ] أي :  لا يعلم وقت البعث إلا الله ،  ثم قال تعالى :  {  والراسخون في العلم يقولون آمنا به }{[23921]} . <br />
و " النسيان "  في هذا{[23922]} الموضع{[23923]} على معنيين : <br />
-يجوز أن يكون معناه :  فلما أعرضوا عنه صاروا بمنزلة من نسي{[23924]} الشيء . <br />
-والثاني :  أن يكون بمعنى الترك{[23925]} .
