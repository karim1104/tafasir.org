قوله :  ( لَّيْسَ بِأَمَانِيّكُمْ وَلاَ أَمَانِيِّ أَهْلِ الْكِتَابِ مَنْ يَّعْمَلْ سُوءاً يُجْزَ بِهِ ) الآية [ 123 ] . <br />
قال مسروق :  تفاخر النصارى والمسلمون ،  فقال هؤلاء :  نحن أفضل ،  وقال هؤلاء :  نحن أفضل منكم . <br />
فأنزل الله عز وجل :  ( لَّيْسَ بِأَمَانِيّكُمْ وَلاَ أَمَانِيِّ أَهْلِ الْكِتَابِ )(   {[13612]} ) الآية(   {[13613]} ) . <br />
/ت*<br />
ثم أفلح الله تعالى حجة المؤمنين فقال :  ( وَمَنْ يَّعْمَلْ مِنَ الصَّالِحَاتِ مِن ذَكَرٍ اَوُ انْثَى وَهُوَ مُومِنٌ فَأُوْلَائِكَ يَدْخُلُونَ الْجَنَّةَ )<br />
*ت/<br />
ثم زاد في الفضل فقال :  ( وَمَنْ اَحْسَنُ دِيناً مِمَّنَ اَسْلَمَ وَجْهَهُ لِلهِ وَهُوَ مُحْسِنٌ ) الآية . <br />
وقال قتادة :  تفاخر المسلمون وأهل الكتاب ،  فقال أهل الكتاب :  نبينا قبل نبيكم ،  وكتابنا قبل كتابكم ،  ونحن أحق بالله منكم ،  وقال المسلمون :  نبينا خاتم النبيين ،  وكتابنا يقضي على الكتب التي كانت قبله ،  فأنزل الله ( لَّيْسَ بِأَمَانِيّكُمْ ) الآية(   {[13614]} ) . <br />
أي :  ليس ذلك الذي قلتم بأمانيكم ففي ليس اسمها في جميع هذه الأقوال(   {[13615]} ) . <br />
( مَنْ يَّعْمَلْ سُوءاً ) ابتداء شرط ،  وجوابه خبره وهو :  ( يُجْزَ بِهِ ) . <br />
وقال الضحاك :  تخاصم أهل الأديان :  اليهود والنصارى والمسلمون فأنزل الله ( لَّيْسَ بِأَمَانِيِّكُمْ ) الآية(   {[13616]} ) . <br />
وقال مجاهد :  عنى بذلك أهل الشرك من عبدة الأوثان ،  قالوا :  لن نبعث ولن نعذب(   {[13617]} ) ،  وقالت اليهود والنصارى ( لَنْ يَّدْخُلَ الْجَنَّةَ إِلاَّ مَن كَانَ هُوداً اَوْ نَصَارَى )(   {[13618]} )(   {[13619]} ) . <br />
فأنزل الله ( لَّيْسَ بِأَمَانِيِّكُمْ ) يعني المشركين في قولهم :  لن نبعث ولن نعذب(   {[13620]} ) . <br />
( وَلاَ أَمَانِيِّ أَهْلِ الْكِتَابِ ) يعني قولهم لن يدخل الجنة إلا من كان هوداً أو نصارى ( مَنْ يَّعْمَلْ سُوءاً يُجْزَ بِهِ ) خلافاً لمن أدعى(   {[13621]} ) الجميع . <br />
والمعنى :  ليس الكائن من أمركم بما تتمنون يا أهل الشرك ولا بما يتمنى أهل الكتاب ،  وهم اليهود والنصارى بل ( مَنْ يَّعْمَلْ سُوءاً يُجْزَ بِهِ ) . <br />
وقيل :  التقدير :  ليس ثواب الله بأمانيكم لأنه(   {[13622]} ) قد جرى ذكر ذلك في قوله :  ( وَالذِينَ ءَامَنُوا وَعَمِلُوا الصَّالِحَاتِ سَنُدْخِلُهُمْ ) الآية(   {[13623]} ) ( وَلاَ يُظْلَمُونَ نَفِيراً ) أي :  مقدار النقير وهو النقطة في ظهر النواة وهي منبت النخلة . <br />
وقالت عائشة رضي الله عنها في قوله :  ( مَنْ يَّعْمَلْ سُوءاً يُجْزَ بِهِ ) ذلك ما يصيبهم في الدنيا ،  وقاله مجاهد وغيره(   {[13624]} ) . <br />
وقال الحسن :  ( مَنْ يَّعْمَلْ سُوءاً يُجْزَ بِهِ ) الكافر وقرأ ( وَهَلْ يُجَازَى إِلاَّ الْكَفُورُ )(   {[13625]} ) ،  وقال في قوله ( لِيَجْزِيَ الذِينَ أَسَاؤُوا بِمَا عَمِلُوا )(   {[13626]} ) .   " هِمُ الكُفَّارُ " (   {[13627]} ) ( وَيَجْزِيَ الذِينَ أَحْسَنُوا بِالْحُسْنَى )(   {[13628]} ) قال :  كانت والله لهم ذنوب ،  ولكنه غفرها لهم ،  ولم يجازهم(   {[13629]} ) بها(   {[13630]} ) . <br />
وقال ابن زيد :  يعني المشركين يريد بالآية قال :  وعد الله المؤمنين أن يكفر عنهم سيئاتهم(   {[13631]} ) . <br />
وقال الضحاك : (   {[13632]} ) ( مَنْ يَّعْمَلْ سُوءاً يُجْزَ بِهِ ) يعني بذلك اليهود والنصارى ،  والمجوس وكفار العرب(   {[13633]} ) . <br />
قوله :  ( وَلاَ يَجِدْ لَهُ(   {[13634]} ) مِن دُونِ اللَّهِ وَلِيّاً وَلاَ نَصِيراً ) . <br />
وروي أن هذه الآية لما نزلت قال أبو بكر الصديق رضي الله عنه :  يا رسول الله :  وإن لمجزون بأعمالنا ؟  قال رسول الله صلى الله عليه وسلم :   " أما المؤمن فيجزى بها في الدنيا ،  وأما الكافر فيجزى بها يوم القيامة " (   {[13635]} ) . <br />
وقال الحسن وابن(   {[13636]} ) أبي كثير(   {[13637]} ) ( مَنْ يَّعْمَلْ سُوءاً يُجْزَى بِهِ ) أي يعمل شركاً يجز به بدلالة قوله :  ( وَهَلْ يُجَازَى إِلاَّ الْكَفُورُ ) وتلاها الحسن مع هذه ( الآية(   {[13638]} ) ) استشهاداً بها(   {[13639]} ) . <br />
وروى عن ابن عباس أنه قال :  السوء هنا :  الشرك ،  ومعنى(   {[13640]} ) من يشرك(   {[13641]} ) :  يجز به(   {[13642]} ) . <br />
وقال ابن جبير :  ( مَنْ يَّعْمَلْ سُوءاً يُجْزَ بِهِ ) أي :  من يشرك(   {[13643]} ) . <br />
قالت عائشة وأبي بن كعب :  إن المعنى :  من عمل سوءاً من مؤمن ،  أو كافر جوزي به ،  وهو اختيار الطبري(   {[13644]} ) ،  واحتج بما روى أبو هريرة قال :  لما نزلت هذه الآية شقت على المسلمين منهم ما شاء الله أن تبلغ ،  فشكوا ذلك إلى النبي صلى الله عليه وسلم فقال :  قاربوا وسددوا ،  ففي كل ما يصاب به المسلم كفارة حتى النكبة ينكبها ،  أو الشوكة يشاكها(   {[13645]} ) . <br />
وروت عائشة عن أبي بكر رضي الله عنه أنه قال :  لما نزلت هذه الآية :  يا رسول الله ( مَنْ يَّعْمَلْ سُوءاً يُجْزَ بِهِ ) فقال له النبي صلى الله عليه وسلم :   " أوليس يصيبك كذا ( ويصيبك كذا(   {[13646]} ) ) فهو كفارة " (   {[13647]} ) . <br />
وقال ابن عمر :  سمعت أبا بكر يقول :  سمعت النبي صلى الله عليه وسلم يقول :  ( مَنْ يَّعْمَلْ سُوءاً يُجْزَ بِهِ ) في الدنيا(   {[13648]} ) . <br />
وروى عن ابن أبي بكر رضي الله عنه أنه قال :  يا نبي الله :  كيف الصلاح بعد هذه الآية ؟  فقال النبي صلى الله عليه وسلم  " غفر الله لك يا أبا بكر ،  ألست تمرض ؟  ألست تحزن ؟  ألست تصيبك اللاواء(   {[13649]} ) ؟  فهو ما تجزون به " (   {[13650]} ) . <br />
وروي أن أبا بكر قال :  لما نزلت هذه الآية ،  جاءت قاصمة الظهر(   {[13651]} ) ،  فقال رسول الله صلى الله عليه وسلم :   " إنما هي المضائق في الدنيا(   {[13652]} ) .
