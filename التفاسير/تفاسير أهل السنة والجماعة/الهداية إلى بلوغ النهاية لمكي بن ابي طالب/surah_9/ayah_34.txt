قوله{[28571]} :  {  يا أيها الذين آمنوا إن كثيرا من الاحبار والرهبان } ،  إلى [ قوله ] :  {  ما كنتم تكنزون }[ 34 ،  35 ] . <br />
قوله :  {  والذين يكنزون }[ 34 ] . <br />
{  والذين } :  في موضع رفع عطف على الضمير{[28572]} في :   " يأكلون "  ،  فيكون التقدير :  {  لياكلوا أموال الناس بالباطل } ،  ويأكلها معهم الذين يكنزون الذهب{[28573]} . <br />
وقيل {  الذين } :  في موضع رفع بالابتداء{[28574]} . <br />
ومعنى الآية :  يا أيها الذين صدقوا بمحمد صلى الله عليه وسلم ،  وبما جاء به ،  إن كثيرا من أحبار اليهود والنصارى ورهبانهم ،  وهم :  علماؤهم وعبادهم .  وقيل {  الاحبار } :  القراء :  {  لياكلون أموال الناس بالباطل } ،  ويأكلها معهم {  الذين يكنزون الذهب } ،  وذلك الرُّشى{[28575]} في الحكم ،  وفي تحريف كتاب الله عز وجل ،  يكتبون بأيديهم كتبا ،  ويقولون :  هذا من عبد الله ،  يأخذون بها ثمنا قليلا ،  {  ويصدون عن سبيل الله }[ 34 ] ،  أي :  يمنعون من أراد الدخول في الإسلام{[28576]} . <br />
و " الكنز " {[28577]} :  كل مال وجبت فيه الزكاة ،  فلم تؤد{[28578]} زكاته{[28579]} . <br />
وقوله :  {  ولا ينفقونها }[ 34 ] . <br />
أي :  لا يؤدون زكاتها{[28580]} . <br />
قال ابن عمر :  كل ما مال أديت زكاته ليس بكنز ،  وإن كان مدفونا ،  وكل مال لم تؤد زكاته ،  فهو كنز يكوى [ به ]{[28581]} صاحبه ،  وإن لم يكن مدفونا{[28582]} . <br />
وروي عن علي رضي الله عنه :  أربعة آلاف درهم فما دونها  " نفقة " /فإن زادت فهو  " كنز " {[28583]} ،  أديت زكاته أو لم تؤد{[28584]} . <br />
قال ابن عباس :  هي خاصة للمسلمين لمن لم يؤد زكاته منهم ،  وهي عامة في أهل الكتاب ،  من أدى الزكاة ومن لم يؤد ؛  لأنهم لا تقبل منهم نفقاتهم وإن أنفقوا{[28585]} . <br />
وقال عمر بن عبد العزيز :  أراها منسوخة بقوله :  {  خذ من أموالهم صدقة تطهرهم وتزكيهم بها }{[28586]}[ 104 ] . <br />
و " الكنز "  في كلام العرب :  كل شيء جُمع بعضه إلى بعض{[28587]} . <br />
قوله :  {  ولا ينفقونها }{[28588]} . <br />
ولم يقل :   " ينفقونهما "  ،  إنما ذلك لأن الضمير رجع على الكنوز ،  والكنوز تشتمل على الذهب والفضة{[28589]} . <br />
وقيل :  إن الضمير يرجع على :   " الأموال " {[28590]} التي تقدم ذكرها أنها تؤكل بالباطل .  وقيل :  الضمير يعود على :   " الفضة "  ،  وحذف العائد على الذهب لدلالة الكلام عليه{[28591]} ،  كأنه{[28592]} قال :  {  ولا ينفقونها }{[28593]} و " ينفقونه "  ،  ثم حذف كما قال{[28594]} : <br />
نحن{[28595]} بما عندنا وأنت بما *** عندك راض .  .  . {[28596]} . <br />
وقيل الضمير :   " للذهب "  ،  وضمير{[28597]}  " الفضة "  محذوف ،  تقديره :  {  ولا ينفقونها }{[28598]} . <br />
و " ينفقونها "  ،  والعرب تقول :   " هي الذهب [ الحمراء ]{[28599]} "  ،  فتؤنث{[28600]} . <br />
وقال معاوية :  هذه الآية في أهل الكتاب خاصة{[28601]} . <br />
وقوله :  {  فبشرهم بعذاب أليم }[ 34 ] . <br />
أي :  اجعل موضع البشارة لهم عذابا أليما ،  أي :  مؤلما ،  بمعنى موجع{[28602]} .
