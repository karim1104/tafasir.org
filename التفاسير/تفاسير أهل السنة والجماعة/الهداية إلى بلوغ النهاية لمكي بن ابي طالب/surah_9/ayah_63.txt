قوله :  {  فأن له }[ 63 ] . <br />
 " أن "  بدل من الأولى عند الخليل وسيبويه{[29157]} . <br />
وقال المبرد والجرمي{[29158]} :   " أن "  الثانية مكررة للتوكيد{[29159]} . <br />
وقال الأخفش{[29160]} :   " أن "  في موضع رفع بالابتداء ،  والمعنى :  فوجوب النار{[29161]} له . <br />
وأنكر ذلك أبو العباس ؛  لأن{[29162]}  " أن "  المشددة المفتوحة لا يبتدأ بها ،  ويضمر الخبر{[29163]} . <br />
وقال علي بن سليمان :   " أن "  في موضع رفع على إضمار مبتدأ ،  والمعنى{[29164]} :  فالواجب أن له النار{[29165]} . <br />
وكلهم أجاز كسر{[29166]}  " أن "  ،  واستحسنه سيبويه{[29167]} . <br />
ومعنى الآية :  ألا يعلم هؤلاء المنافقون {  أنه من يحادد الله } ،  أي :  يجانبه ويعاديه وحقيقته :  أنه يقال{[29168]} :  حادّ فلان فلانا ،  أي :  صار في حد غيره حده{[29169]} ،  {  فأن له نار جهنم خالدا ( فيها ) }{[29170]} ،  أي :  لابثا أبدا ،  {  ذلك الخزي العظيم }[ 63 ] ،  أي :  الهوان{[29171]} والذل .
