فقوله :  {  للفقراء والمساكين } . <br />
قال مجاهد ،  وعكرمة ،  والزهري ،  وجابر بن زيد :  / " الفقير "  :  الذي لا يسأل و " المسكين "  :  الذي يسأل{[29026]} . <br />
وقال ابن عباس :   " المسكين "  :  الطواف ،  و " الفقير "  :  فقير المسلمين{[29027]} . <br />
وقال قتادة  " الفقير "  :  المحتاج الذي به زمانة{[29028]} ،  و " المسكين "  :  الصحيح المحتاج{[29029]} . <br />
وقال الضحاك  " الفقراء "  :  فقراء المجاهدين ،  و " المساكين "  :  [ الذين ]{[29030]} لم يهاجروا{[29031]} . <br />
وذكر ابن وهب ،  عنه ،  أن  " الفقراء "  :  من المهاجرين ،  و " المساكين "  ){[29032]} :  من الأعراب .  قال :  وكان ابن عباس يقول :   " الفقراء "  من المسلمين ،  و " المساكين "  من أهل الذمة . <br />
وقال الشافعي  " الفقراء "  :  الذين لا مال لهم ولا حرفة تغنيهم ،  و " المساكين "  الذين لهم مال ،  أو{[29033]} حرفة لا تغنيهم{[29034]} . <br />
وقال{[29035]} أبو ثور  " الفقير "  :  الذي لا شيء له ،  و " المسكين "  :  الذي لا يكسب من كسبه ما يقوته . <br />
وقال عبيد الله{[29036]} بن الحسن  " المسكين "  الذي يخشع ويستكين{[29037]} ،  بأن{[29038]} لم يسأل و " الفقير "  ،  الذي يتحمل ،  ويقبل الشيء سرا ،  ولا يخشع{[29039]} . <br />
وقال محمد بن مسلمة{[29040]}  " الفقير "  :  الذي له مسكن يسكنه والخادم إلى ما هو أسفل من ذلك ،  و " المسكين "  :  الذي لا مال له{[29041]} . <br />
وقال أهل اللغة :   " المسكين "  :  الذي لا شيء له ،  و " الفقير "  :  الذي له شيء يكفيه{[29042]} . <br />
قال يونس :  قلت لأعرابي :  أفقير أنت ؟  قال :  لا ،  بل مسكين{[29043]} . <br />
وقال عكرمة  " الفقراء "  :  من اليهود والنصارى ،  و " المساكين "  :  من المسلمين{[29044]} . <br />
واختار الطبري ،  وغيره أن يكون  " الفقير "  :  الذي يعطي بفقره فقط ،  و " المسكين "  :  الذي يكون عليه مع فقره خضوع وذل السؤال{[29045]} . <br />
وأنشد أهل اللغة قول الراعي{[29046]} : <br />
أما الفقير الذي كانت حلوبته *** وفق العيال فلم يُترك له سبد{[29047]} . <br />
فجعل للفقير حلوبة{[29048]} ،  مقدار ما يكفي العيال . <br />
ف :  " المسكين "  أشد حاجة من  " الفقير "  فكل مسكين فقير ،  وليس كل فقير مسكينا . <br />
ف :   " الفقير "  :  الذي لا غنى له فوق قوت يومه ،  وهو فعيل بمعنى مفعول ،  كأنه مفقور الظهر ،  وهو الذي نزعت فقره [ من فِقر ]{[29049]} ظهره فانقطع ظهره من شدة الفقر{[29050]} ،  وهذا الاشتقاق يدل على أن  " الفقير "  أشد حاجة من  " المسكين "  ،  وقد قال تعالى :  {  فكانت{[29051]} لمساكين{[29052]} يعملون في البحر }{[29053]} ،  فسماهم :  مساكين ،  ولهم سفينة .  ولا حجة في قراءة من قرأ :   " مسّاكين " {[29054]} ،  بالتشديد{[29055]} ؛  لأن الجماعة على التخفيف . <br />

وقال الزهري :  {  المؤلفة قلوبهم } :  من أسلم من يهودي أو نصراني ،  غنيا كان أو فقيرا{[29061]} . <br />

وقال الشعبي :  كانت  " المؤلفة "  ،  على عهد النبي صلى الله عليه وسلم ،  فلما ولي أبو{[29064]} بكر انقطع ذلك{[29065]} . <br />
وهو قول مالك ،  قال :  يرجع سهم المؤلف إلى أهل السهام الباقية{[29066]} . <br />
وقال الشافعي  " المؤلفة "  :  من دخل في الإسلام{[29067]} . <br />
وقال ابن حنبل/وغيره :  {  المؤلفة قلوبهم } ،  في كل زمان{[29068]} . <br />
