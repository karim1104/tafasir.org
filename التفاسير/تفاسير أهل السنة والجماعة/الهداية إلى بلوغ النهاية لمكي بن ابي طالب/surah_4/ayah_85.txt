قوله :  ( مَّنْ يَّشْفَعْ حَسَنَةً يَكُن لَّهُ نَصِيبٌ مِّنْهَا ) الآية [ 85 ] . <br />
المعنى :  من يشفع لأخيه شفاعة حسنة يكن له نصيب منها في(   {[13037]} ) الآخرة ،  أي :  حظ ( وَمَنْ يَّشْْفَعْ شَفَاعَةً سَيِّئَةً يَكُن لَّهُ كِفْلٌ مِنْهَا ) أي :  إثم منها في الآخرة . <br />
وقال مجاهد :  هو شفاعة الناس بعضهم(   {[13038]} ) لبعض ،  وقاله الحسن وابن زيد وغيرهم(   {[13039]} ) . <br />
وقال الطبري :   " المعنى :  من يشفع وتراً(   {[13040]} ) لأصحابك يا محمد في جهاد عدوهم يكن له نصيب منها :  حظاً في الآخرة ،  ومن يشفع وتراً لكافرين يكن له كفل منها أي :  وزر وإثم في الآخرة(   {[13041]} ) . <br />
وإنما اختار ذلك لأنه في سياق الآية التي حض الله عز وجل النبي صلى الله عليه وسلم على القتال فيها ،  وحضه على أن يحرض المؤمنين على القتال معه ،  فصارت هذه الآية وعداً لمن أجاب تحريض رسول الله صلى الله عليه وسلم على القتال ،  وكان ذلك أشبه عنده من شفاعة الناس بعضهم لبعض إذ لم يجر له ذكر قبل ولا بعد . <br />
والكفل والنصيب عند أهل اللغة سواء ،  قال تعالى :  ( يُوتِكُمْ كِفْلَيْنِ مِن رَّحْمَتِهِ ) [ الحديد :  27 ] أي :  نصيبين ،  والنصيب قد يكون خيراً أو شراً . <br />
وقال الحسن :  الشفاعة الحسنة ما يجوز في الدين ،  والشفاعة السيئة ما لا يجوز في الدين . <br />
وقال الحسن :  من(   {[13042]} ) يشفع شفاعة حسنة كان له أجرها وإن لم يُشَفَّع(   {[13043]} ) . <br />
وروي أن قوله :  ( وَمَنْ يَّشْفَعْ شَفَاعَةً سَيِّئَةً يَكُن لَّهُ كِفْلٌ مِّنْهَا ) نزل في اليهود كانوا يدعون على المؤمنين في الغيبة بالهلاك ،  ويقولون لهم في الحضور :  السام عليكم ،  وهو دعاء أيضاً ،  ثم أتْبع ذلك بقوله :  ( وَإِذَا حُيِّيتُم بِتَحِيَّةٍ فَحَيُّوا بِأَحْسَنَ مِنْهَا ) وهو :  السلام(   {[13044]} ) . <br />
والمقيت :  الحفيظ عند ابن عباس(   {[13045]} ) . <br />
وقيل :  الشهيد ،  قاله مجاهد(   {[13046]} ) . <br />
وقال السدي :  المقيت :  القدير وهو اختيار الطبري(   {[13047]} ) . <br />
وقال الكسائي :  مقيتاً مقتدراً(   {[13048]} ) . <br />
وقال أبو عبيدة :  المقيت :  الحافظ المحيط(   {[13049]} ) . <br />
وقال ابن جريج :  المقيت القائم على كل شيء رواه عن ابن كثير(   {[13050]} ) . <br />
وقال بعضهم(   {[13051]} ) : <br />
وَذِي ضَغَنٍ كَفَفْتُ النَّفْسَ عنه	وَكُنْتُ عَلَى مُسَاءَتِه مَقِيتاً(   {[13052]} )
