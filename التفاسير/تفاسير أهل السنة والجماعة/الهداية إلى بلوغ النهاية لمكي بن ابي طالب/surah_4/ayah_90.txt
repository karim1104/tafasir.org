قوله :  ( اِلاَّ الذِينَ يَصِلُونَ إِلَى قَوْمٍ بَيْنَكُمْ وَبَيْنَهُم مِّيثَاقٌ ) الآية [ 90 ] . <br />
المعنى اقتلوا من وجدتم من المنافقين الذين اختلفتم فيهم إن لم يهاجروا إلا أن يتصل قوم منهم بمن بينكم وبينهم(   {[13109]} ) عهد فيدخلون(   {[13110]} ) فيما دخلوا فيه ،  ويرضون بما رضوا ،  فلا يقتل من كانت هذه حاله منهم فإن لهم حكمهم . <br />
قال السدي :  المعنى إذا أظهروا كفرهم ،  فاقتلوهم حيث وجدتموهم إلا أن يكون أحد منهم دخل في قوم بينكم وبينهم ميثاق ،  فأجروا عليه مثلما تجرون على القوم واحكموا في الجميع بحكم واحد(   {[13111]} ) . <br />
ومعنى :  ( يَصِلُونَ ) يتصلون . <br />
وقال أبو عبيدة :  معنى :  ( يَصِلُونَ إِلَى قَوْمٍ ) :  ينتسبون إليهم(   {[13112]} ) .  وهو بعيد(   {[13113]} ) ،  لأن النبي صلى الله عليه وسلم قد قاتل من ينتسب إلى من بينهم وبينه عهد ،  وليس النسب مما يمنع قتال الكفار لعهد بيننا وبين قرابتهم . <br />
وروي عن ابن عباس أنها منسوخة(   {[13114]} ) بقوله :  ( فَاقْتُلُوا المُشْرِكِينَ حَيْثُ وَجَدتُّمُوهُمْ )(   {[13115]} ) وقال قتادة :  ( اِلاَّ الذِينَ يَصِلُونَ إِلَى قَوْمٍ بَيْنَكُمْ وَبَيْنَهُمْ [ مِّيثَاقٌ ] )(   {[13116]} ) نسخ بعد ذلك ،  فنبذ إلى كل ذي عهد عهده ،  ثم أمرنا بالقتال في براءة . <br />
وقال ابن زيد :  نسخها الجهاد(   {[13117]} ) . <br />
قوله :  ( اَوْ جَاءُوكُمْ حَصِرَتْ صُدُورُهُمُ ) الآية [ 90 ] . <br />
المعنى :  إلا الذين جاؤوكم قد ضاقت صدورهم عن أن يقاتلوكم أو يقاتلوا قومهم ،  فدخلوا فيكم(   {[13118]} ) ،  فلا تقتلوهم ،  ( وَلَوْ شَاءَ اللَّهُ لَسَلَّطَهُمْ عَلَيْكُمْ ) أي :  لسلط عليكم هؤلاء الذين يتصلون بقوم بينكم وبينهم ميثاق ،  والذين يجيئونكم قد ( حَصِرَتْ صُدُورُهُمُ ) أي :  ضاقت عن قتالكم(   {[13119]} ) ،  وقتال قومهم ،  فيقاتلوكم مع أعدائكم من المشركين ،  ولكن الله كفهم عنكم(   {[13120]} ) . <br />
وقوله :  ( فَلَقَاتَلُوكُمْ ) ليست(   {[13121]} ) اللام بجواب للقسم [ كاللام في ( لَسَلَّطَهُمْ ) ،  وإنما دخلت للمجاءاة(   {[13122]} ) لا للقسم ](   {[13123]} ) ،  ومثله قوله :  ( أَوْ لَيَاتِيَنِّي بِسُلْطَانٍ مُّبِينٍ )(   {[13124]} ) .  ليست اللام بجواب للقسم وإنما دخلت للمحاذاة(   {[13125]} ) للامين اللتين قبلها ،  اللتين هما جواب قسم سليمان في قوله ( لَأُعَذِّبَنَّهُ عَذَاباً شَدِيداً اَوْ لَأَذْبَحَنَّهُ )(   {[13126]} ) ولهذا نظائر ستراها . <br />
قوله :  ( فَإِنِ اعْتَزَلُوكُمْ ) أي :  اعتزلكم هؤلاء الذين أمرتكم بالكف عن قتالهم(   {[13127]} ) فلم يقاتلوكم ( وَأَلْقَوِا إِلَيْكُمُ السَّلَمَ ) .  أي :  صالحوكم ،  وقيل المعنى :  استسلموا(   {[13128]} ) إليكم(   {[13129]} ) . <br />
( فَمَا جَعَلَ اللَّهُ لَكُمْ عَلَيْهِمْ سَبِيلاً ) أي :  ليس لكم إليهم طريق فتستحلونهم بما في أنفسهم وأموالهم وذراريهم(   {[13130]} ) ،  وهذا كله منسوخ بقوله :  ( فَإِذَا انْسَلَخَ الاَشْهُرُ الْحُرُمُ فَاقْتُلُوا الْمُشْرِكِينَ حَيْثُ وَجَدتُّمُوهُمْ وَخُذُوهُمْ وَاحْصُرُوهُمْ وَاقْعُدُوا لَهُمْ كُلّ مرْصَدٍ )(   {[13131]} ) . <br />
وقال الحسن وعكرمة :  قوله :  ( اِلاَّ الذِينَ يَصِلُونَ إِلَى قَوْمٍ ) إلى قوله ( سُلْطَاناً مُّبِيناً )(   {[13132]} ) وقوله :  ( لاَّ يَنْهَاكُمُ اللَّهُ عَنِ الذِينَ لَمْ يُقَاتِلُوكُمْ فِي الدِّينِ ) إلى ( يُحِبُّ المُقْسِطِينَ )(   {[13133]} ) قالا :  بنسخ ذلك في براءة ،  فجعل لهم أن يسيحوا في الأرض أربعة أشهر(   {[13134]} ) . <br />
قال قتادة :  ( اِلاَّ الذِينَ يَصِلُونَ ) إلى ( مُّبِيناً ) منسوخ ببراءة(   {[13135]} ) ،  وقال ابن زيد :  نسخ هذا كله ،  نسخه الجهاد(   {[13136]} ) وضرب لهم أجل(   {[13137]} ) أربعة أشهر إما أن يسلموا وإما أن يكون الجهاد . <br />
واختلف في ( حَصِرَتْ صُدُورُهُمُ )(   {[13138]} ) ،  فقال المبرد المعنى :  الدعاء(   {[13139]} ) ،  لأنه قال :  ( اَوْ جَاءُوكُمْ ) أحصر الله صدوركم أي :  ضيقها عن قتالكم ،  وقتال قومهم .  وقال الزجاج :  يجوز أن يكون خبراً بعد خبر(   {[13140]} ) ،  فالمعنى أو جاءكم ثم خبر بعد فقال :  ( حَصِرَتْ صُدُورُهُمُ ) ،  وأكثر النحويين على أنه حال ،  وقد مضمرة والتقدير :  أو جاؤوكم قد حصرت أي :  ضاقت صدورهم عن قتالكم وقتال قومهم .  أي :  جاؤوكم في هذه الحال فلا تقاتلوهم . <br />
وقال الطبري :  المعنى :  أو جاؤوكم قد حصرت(   {[13141]} ) . <br />
وقرأ الحسن حصرةً بالتنوين والنصب على الحال(   {[13142]} ) أي :  ضيقت صدورهم ،  واستحسن هذا المبرد(   {[13143]} ) ،  ويجوز على قراءة الحسن الخفض على النعت ،  والرفع على الابتداء . <br />
وقرأ أبي بن كعب :  ( بِيْنَكُمْ وَبَيْنَهُم مِّيثَاقٌ )(   {[13144]} ) ،  وحصرت صدورهم بإسقاط أو ( جَاءُوكُمْ )(   {[13145]} ) ،  ولا يقرأ به الآن .
