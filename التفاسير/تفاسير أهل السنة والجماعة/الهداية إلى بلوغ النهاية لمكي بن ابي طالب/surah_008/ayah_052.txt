قوله :  {  كدأب آل فرعون والذين من قبلهم كفروا بآيات الله } إلى قوله :  {  سميع عليم }[ 52 ،  53 ] . <br />
 " الدأب "  :  العادة{[27619]} ،  وأصله من قولهم :  {  فلان يدأب على الشيء "  : <br />
أي :  يدوم عليه ويلزمه{[27620]} . <br />
و " الكاف "  متعلقة بقوله :  {  ذلك بما قدمت أيديكم } . <br />
من المعاصي كعادة آل فرعون والذين من قبلهم{[27621]} . <br />
و " الكاف "  من :  {  كدأب } ،  في موضع رفع على إضمار مبتدأ{[27622]} . <br />
أي :  العادة في تعذيبكم عند قبض الأرواح وفي القبور مثل العادة في آل فرعون{[27623]} . <br />
ويجوز أن تكون  " الكاف "  متعلقة ب : {  ذوقوا{[27624]} عذاب الحريق } ،  فتكون في موضع نصب على معنى :  ذوقوا مثل عادة آل فرعون في إذاقتنا إياهم العذاب . <br />
فالمعنى :  فعل هؤلاء المشركون كما فعل آل فرعون ،  أو فعلنا بهم كفعلنا بآل فرعون ،  فإذا رددت التشبيه إلى فعل المشركين وفعل آل فرعون جاز ،  وإذا رددته إلى فعل الله عز وجل بهؤلاء كفعله بهؤلاء جاز ،  ويتمكن في كلا الوجهين في :   " الكاف "  الرفع والنصب . <br />
قال مجاهد المعنى :  كسنن{[27625]} آل فرعون{[27626]} . <br />
وقوله :  {  فأخذهم الله بذنوبهم }[ 52 ] . <br />
أي :  فعاقبهم الله بتكذيبهم{[27627]} . <br />
{  إن الله قوي }[ 52 ] . <br />
أي :  لا يغلبه غالب{[27628]} . <br />
{  شديد العقاب }[ 52 ] . <br />
لمن كفر به{[27629]} .
