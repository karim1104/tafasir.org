قوله :  {  ولا تحسبن الذين كفروا{[27672]} سبقوا } ،  إلى قوله :  {  وأنتم لا تظلمون }[ 59 ،  60 ] . <br />
خطّأ أبو{[27673]} حاتم من قرأ ب :   " الياء " {[27674]} ،  ووجهها{[27675]} عند غيره ظاهر حسن{[27676]} . <br />
والمعنى في  " الياء "  :  ولا تحسبن من خلفهم الذين كفروا سبقوا ،  فيكون ضمير الفاعل في {  تحسبن } ،  يعود على {  من خلفهم }[ 58 ] ،  و{  الذين كفروا } :  مفعول أول ،  و{  سبقوا } في موضع الثاني{[27677]} . <br />
وقال الفراء التقدير :  أن سبقوا{[27678]} . <br />
وقال :  وفي حرف عبد الله{[27679]} :  /{  ولا يحسبن الذين كفروا سبقوا إنهم لا يعجزون }{[27680]} . <br />
وروي عنه :   " ولا يَحْسَبَ "  ،  بفتح الباء{[27681]} ،  على إرادة النون [ الخفيفة ]{[27682]} . <br />
ومن فتح :  {  إنهم }{[27683]} ،  وقرأ ب  " الياء "  أوب :   " التاء "  فمعناه :  لأنهم{[27684]} ،  ولا يجوز أن يكون مفعولا ثانيا ب :   " حسب "  ،  كما لا يجوز :  حسبت زيدا أنه قائم ؛  لأن زيدا غير قيامه ،  ولو قلت :  حسبت أمرك أنك قائم ،  جاز فتح  " أن "  لأن أمرك هو قيامك{[27685]} . <br />
ومعنى ذلك :  لا يحسب من ظفر بالخلاص من هذه الوقعة سبق . <br />
ثم قال :  {  إنهم لا يعجزون }[ 59 ] ،  لا يفوتون{[27686]} . <br />
ومن قرأ ب :  " الياء "  فمعناه :  لا يحسب من خلفهم الذين كفروا سبقوا{[27687]} ،  أي :  بالخلاص ،  ثم قال :  {  إنهم لا يعجزون } ،  أي :  لا يفوتون .
