ثم قال تعالى :  {  وقرءنا فرقناه لتقرأه على الناس على مكث } [ 106 ] . <br />
قال ابن عباس :   " فرقناه "  فصلناه{[42036]} .  وقيل :  معنى [ فرقناه{[42037]} ] فرقنا به بين الحق والباطل والمؤمن والكافر{[42038]} . <br />
وقرأ{[42039]} :  ابن عباس وعكرمة والشعبي وقتادة  " فرقناه "  بالتشديد على معنى أنزل به آية بعد آية{[42040]} . <br />
قال ابن عباس :  أنزل القرآن جملة واحدة إلى السماء الدنيا في ليلة القدر ثم أنزل بعد ذلك في عشرين سنة .  وهو قوله :  {  وقرءانا فرقناه لتقرأه على الناس على مكث ونزلناه تنزيلا } [ 105 ]{[42041]} . <br />
ونصب  " قرآنا "  عند البصريين بإضمار فعل يفسره ما بعده{[42042]} .  ونصبه عند غير البصريين{[42043]} على العطف  " مبشرا ونذيرا " {[42044]} . <br />
وقرآنا يتأول بمعنى :  رحمة{[42045]} .  لأن القرآن رحمة .  فلا يحسن الوقف على هذا التقدير على نذيرا .  ويقف عليه على القول الأول{[42046]} . <br />
و{[42047]}/معنى {  لتقرأه على الناس على مكث } [ 106 ] على تؤدة ،  وترتله{[42048]} وتبينه{[42049]} فلا تعجل{[42050]} في تلاوته فلا يفهم عنك{[42051]} .  قال ابن عباس :   " على مكث "  على تأن{[42052]} .  وقال مجاهد على ترسل{[42053]} . <br />
ومعنى {  ونزلناه تنزيلا } أي نزلناه شيئا [ بعد{[42054]} ] شيء{[42055]} .  وقال الحسن نزل القرآن قبل أن يهاجر رسول الله صلى الله عليه وسلم إلى المدينة بثماني سنين .  وبالمدينة عشر سنين{[42056]} .
