ثم قال تعالى ذكره{[50910]} :  {  إن هؤلاء لشرذمة قليلون } أي :  قال فرعون لمن جمع من الناس :  إن موسى{[50911]} ومن معه لشرذمة قليلون .  والشرذمة{[50912]} الطائفة ،  وشرذمة{[50913]} كل شيء بقيته القليلة . <br />
قال أبو عبيدة{[50914]} :  كانوا ست{[50915]} مائة ألف وسبعين{[50916]} ألفا فوصفهم بالقلة . <br />
قال{[50917]} عبد الله بن شداد بن الهاد{[50918]} :  اجتمع يعقوب ،  وولده إلى يوسف وهم اثنان وسبعون وخرجوا{[50919]} مع موسى وهم ست{[50920]} مائة{[50921]} ألف غير من{[50922]} مات ،  وخرج فرعون على فرس أدهم حصان{[50923]} في عسكره ثمان مائة ألف{[50924]} . <br />
قال{[50925]} الزجاج{[50926]} :  كانت مقدمة فرعون :  سبع مائة ألف ،  كل رجل منهم على حصان على رأسه بيضة{[50927]} . <br />
وقال قيس بن عباد : {[50928]} كانت مقدمة فرعون ست{[50929]} مائة ألف ،  كل رجل منهم على حصان على رأسه بيضة ،  في يده حربة وهو خلفهم في الدهم{[50930]} ،  فلما انتهى موسى ببني إسرائيل البحر ،  قالت بنو إسرائيل :  يا موسى أين ما وعدتنا ؟  هذا البحر بين أيدينا ،  وهذا فرعون وجنوده قد دهمنا{[50931]} من خلفنا ،  فقال موسى للبحر :  إنفلق يا أبا خالد{[50932]} ،  فقال :  لا أنفلق يا موسى ،  أنا{[50933]} أقدم منك خلقا ،  فنودي موسى {  أن اضرب بعصاك البحر } فضربه{[50934]} فانفلق البحر ،  وكانوا اثني عشر سبطا{[50935]} ،  فسار لكل سبط طريق ،  فلما انتهى أول جنود فرعون إلى البحر ،  هابت الخيل البحر{[50936]} ،  ومثل الحصان{[50937]} منها وديق{[50938]} فرس ،  فوجد ريحها فاشتد ،  واتبعته{[50939]} الخيل فلما{[50940]} تتام{[50941]} آخر جنوده{[50942]} في البحر ،  وخرج آخر{[50943]} بني إسرائيل ،  أمر{[50944]} البحر فانطبق عليهم ،  فقالت بنو إسرائيل : {[50945]} ما مات فرعون وما كان ليموت أبدا ،  فسمع الله تكذيبهم لنبيه ،  فرمى{[50946]} به على الساحل{[50947]} كأنه ثور أحمر فرآه بنو إسرائيل{[50948]} . <br />
قال مالك :  خرج مع موسى رجلان من التجار إلى البحر{[50949]} ،  فلما أتيا إليه قالا له :  ماذا أمرك ربك به ؟ {[50950]} قال :  أمرني ربي أن أضرب البحر بعصاي{[50951]} هذه{[50952]} فيجف{[50953]} ،  فقالا / له :  إفعل ما أمرك به{[50954]} ربك فلن يخلفك ،  قال ثم ابتدرا{[50955]} إلى البحر ،  فألقيا أنفسهما فيه تصديقا به{[50956]} . <br />
قال ابن جريج{[50957]} :  أوحى الله جل ذكره إلى موسى قبل أن يسري{[50958]} بهم أن اجمع بني{[50959]} إسرائيل كل أربعة أبيات في بيت ،  ثم اذبحوا أولاد الضأن فاضربوا بدمائها على الأبواب فإني سآمر الملائكة ألا تدخل{[50960]} بيتا على بابه دم{[50961]} ،  وسآمرهم بقتل أبكار آل فرعون ثم اخبزوا خبزا{[50962]} فطيرا ،  فإنه أسرع لكم ،  ثم أسر بعبادي حتى تنتهي{[50963]} البحر ،  فيأتيك أمري ،  ففعل فلما أصبحوا ،  قال فرعون :  هذا عمل موسى وقومه ،  قتلوا أبكارنا ،  فأرسل في أثرهم ألف ألف وخمس مائة ألف وخمس{[50964]} مائة ملك مصور ،  مع كل ملك ألف رجل ،  وخرج فرعون في الكرسي{[50965]} العظيم ،  وقال :  {  إن هؤلاء لشرذمة قليلون }[ 45 ] ،  وكانوا ست{[50966]} مائة ألف ،  منهم أبناء عشرين سنة إلى الأربعين{[50967]} . <br />
وقال ابن عباس{[50968]} :  كان مع فرعون{[50969]} يومئذ ألف{[50970]} جبار كلهم عليه تاج ،  وكلهم أمير{[50971]} على خيل . <br />
قال ابن جريج : {[50972]} كان{[50973]} ثلاثون ألفا يعني من الملائكة ساقة{[50974]} ،  خلف فرعون يحسبون أنهم معهم ،  وجبريل عليه السلام{[50975]} أمامهم يرد أوائل الخيل على آخرها{[50976]} ،  فأتبعهم حتى انتهوا{[50977]} إلى البحر .
