ثم قال :   " عز وجل(   {[4253]} ) ( أَمْ كُنتُمْ شُهَدَاءَ إِذْ حَضَرَ يَعْقُوبَ المَوْتُ ) [ 132 ] . <br />
( أَمْ ) بمعنى الألف ،  أي :  أكنتم حاضرين(   {[4254]} ) يا معشر اليهود والنصارى المكذبين بمحمد صلى الله عليه وسلم/ إذ نزل بيعقوب(   {[4255]} ) الموت حين قال لبنيه :  ما تعبدون من بعد موتي ؟  قالوا :  نعبد إلهك وإله آبائك إبراهيم وهو جده(   {[4256]} ) –وإسماعيل(   {[4257]} )- وهو عمه(   {[4258]} ) –وإسحاق-وهو أبوه- صلوات اله عليهم [ و ](   {[4259]} ) على محمد(   {[4260]} ) . <br />
وقدم(   {[4261]} ) إسماعيل لأنه أكبر من إسحاق(   {[4262]} ) . <br />
( إِلَهاً وَاحِداً ) ؛  أي :  معبوداً واحداً ،  لا نشرك(   {[4263]} ) به شيئاً . <br />
( وَنَحْنُ لَهُ مُسْلِمُونَ ) :  أي :  خاضعون متذللون بالعبادة له . <br />
وروي أنه لم يقبض الله نبياً قط حتى يخيّره بين الموت والحياة ،  فلما خيّر(   {[4264]} ) يعقوب عليه السلام قال :  أنظرني(   {[4265]} )/حتى أسأل ولدي ،  وأوصيهم ففعل الله ذلك .  فجمع يعقوب ولده وهم إثنا عشر –وهم الأسباط- ،  وجمع أولادهم فقال لهم :  إنه(   {[4266]} ) قد حضرتْ وفاتي ،  وأنا أريد أن أسألكم وأوصيكم :  فما تعبدون من بعدي ؟  فأجابوه بما حكى الله تعالى عنهم ،  فدعا لهم ثم قبضه الله صلوات الله عليه وعليهم أجمعين وعلى(   {[4267]} ) محمد(   {[4268]} ) . <br />
فمعنى الكلام :  إنكم يا أهل الكتابين(   {[4269]} ) لم تحضروا ذلك –ولا شاهدتموه فكفرتم بغير علم ولا يقين فادَّعَيْتُمْ على أنبياء الله الأباطيل ونحلتموهم إلى اليهودية والنصرانية ،  وإنما بعثهم الله [ بالحنيفية المسلمة ،  وبذلك وصّوا بنيهم(   {[4270]} ) ] فلو حضرتم ذلك وسمعتموه لعلمتم أنهم على غير ما تنحلونهم من الدين . <br />
وهذه الآيات(   {[4271]} ) نزلت تكذيباً من الله لليهود والنصارى في دعواهم إبراهيم/ ويعقوب أنهما كانا(   {[4272]} ) على ملتهم . <br />
وقرأ يحيى بن يعمر(   {[4273]} ) والحسن وأبو رجاء(   {[4274]} ) والجحذري(   {[4275]} )  " وإله أبيك "  بلفظ التوحيد(   {[4276]} ) .  فيحتمل أن يكون جمعاً مسلماً ،  فيكون كالقراءة التي عليها الجماعة .  ويحتمل أن يكون موحداً وإبراهيم بدل(   {[4277]} ) منه وإسماعيل وإسحاق عطف على الأب وهما في القول الأول بدل الجمع الذي قبلهما . <br />
وجمع  " إبراهيم "  و " إسماعيل "  عند سيبويه والخليل :   " بَرَاهيمُ "  و " سَمَاعِيلُ(   {[4278]} ) " (   {[4279]} ) . <br />
وحكى الكوفيون  " بَرَاهِمَةُ "  و " سَمَاعِلَةٌ "  ،  فالهاء بدل من الياء كزنادقة وزناديق(   {[4280]} ) . <br />
وجمعهما(   {[4281]} ) عند المبرد :   " أَبَارِهٌ "  و " أَسَامِعٌ "  و " أَبَارِيهٌ "  و " أَسَامِيعٌ "  .  قال :  لأن الهمزة ليس هذا موضوع زيادتها(   {[4282]} ) . <br />
وأجاز(   {[4283]} ) أحمد بن يحيى :   " بِرَاهٌ " / كما يجوز(   {[4284]} ) في التصغير  " بُرَيْهٌ " (   {[4285]} ) . <br />
وجمع إسحاق أَسَاحِيقٌ .  وحكى الكوفيون أَسَاحِقَةٌ وأَسَاحِيقٌ وَيَعْقُوبٌ وَيَعَاقِيبٌ ،  وَيَعَاقِبَةٌ وَيَعَاقِبٌ " (   {[4286]} ) . <br />
ولا يجوز عند أحد حذف الهمزة من  " إسرائيل "  ،  ويقال(   {[4287]} ) في جمعه :   " أساريل " (   {[4288]} ) . <br />
وحكى الكوفيون  " أَسَارِلَةٌ "  و " أَسَارِلٌ "  ،  وجمعه كله مسلماً أحسن(   {[4289]} ) . <br />
