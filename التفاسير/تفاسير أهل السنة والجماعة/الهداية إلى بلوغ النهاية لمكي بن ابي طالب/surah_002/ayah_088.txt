قوله ( قُلُوبُنَا غُلْفٌ ) [ 87 ] . <br />
هو جمع أغلف كأنها في غلاف وغطاء .  يقال :   " سيف(   {[2914]} ) أغْلَفُ إذا كان في غلافه " (   {[2915]} ) . <br />
وقال قتادة :   " معناه قلوبنا لا تفقه " (   {[2916]} ) . <br />
وقال ابن عباس وغيره :   " معناه :  قلوبنا في غطاء وغلاف فليس نفهم(   {[2917]} ) ما تقول كما قال :  ( قُلُوبُنَا فِي أَكِنَّةٍ مِّمَّا تَدْعُونَا )(   {[2918]} )(   {[2919]} ) . <br />
ويجوز أن يكون  " غُلْفٌ "  جمع  " غلاف "  ،  لكن(   {[2920]} ) أسكن تخفيفاً .  ومعناه :  قلوبنا أوعية للعلم لا تحتاج إلى/ علم محمد صلى الله عليه وسلم . <br />
وعلى ذلك قراءة من قرأ بِضَمِّ اللام ،  وهي قراءة الأعرجِ(   {[2921]} )/ وابن محيصن ورويت عن أبي عمرو وابن عباس(   {[2922]} ) . <br />
قوله :  ( بَلْ لَّعَنَهُمُ اللَّهُ ) [ 87 ] . <br />
أي أبعدهم الله وطردهم وأخزاهم(   {[2923]} ) .  وأصل اللعن الطرد والإبعاد والإقصاء(   {[2924]} ) . <br />
( فَقَلِيلاً مَّا يُومِنُونَ ) [ 87 ] نصبه على أنه نعت لمصدر(   {[2925]} ) محذوف أو لظرف محذوف(   {[2926]} ) . <br />
وقيل :  هو منصوب ب " يؤمِنُونَ "  ،  و " ما "  زائدة(   {[2927]} ) . <br />
ومعناه أنهم يقرون بالله ويوحدونه ،  ويكفرون بالنبي [ عليه السلام ](   {[2928]} ) كما قال ( وَمَا يُومِنُ أَكْثَرُهُم بِاللَّهِ إِلاَّ وَهُم مُّشْرِكُونَ )(   {[2929]} )(   {[2930]} ) . <br />
ويجوز أن يكون المعنى أنهم لم يؤمنوا البتة ،  تقول العرب  " قَلَّ الشَّيء "  إذا لم يوجد(   {[2931]} ) . <br />
ويُقال :   " قَلَّما رَأَيْتُ مِثْلَ هَذَا "  أي :  ما رأيت مثله(   {[2932]} ) . <br />
وحكى الكسائي عن العرب :   " مَرَرْتُ بِبِلادٍ قَلَّ مَا تُنْبِتُ(   {[2933]} ) إلاَّ الكُرَّاثَ(   {[2934]} ) وَالبَصَلَ "  أي :  ما تنبت سواهما(   {[2935]} ) . <br />
وحكى سيبويه :   " قلّ رجلٌ يقُولُ ذَلكَ إلاّ زَيْداً " (   {[2936]} ) . <br />
وقال أهل التفسير :   " معناه :  فقليلاً(   {[2937]} ) منهم من يؤمن ،  لأن الذين آمنوا من المشركين أكثر كثيراً ممن آمن من اليهود "  .  وهذا مروي عن قتادة(   {[2938]} ) ويلزم منه رفع  " قليل "  . <br />
وقيل(   {[2939]} ) :  المعنى :  ليس يؤمنون مما في أيديهم إلا بقليل " (   {[2940]} ) . <br />
والاختيار عند أكثرهم قول من قال :   " إنهم قليلوا(   {[2941]} ) الإيمان بما أنزل على النبي(   {[2942]} ) [ عليه السلام ](   {[2943]} ) .
