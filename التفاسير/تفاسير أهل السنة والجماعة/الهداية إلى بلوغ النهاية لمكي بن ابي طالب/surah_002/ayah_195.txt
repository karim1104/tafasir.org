قوله :  ( وَأَنْفِقُوا فِي سَبِيلِ اللَّهِ وَلاَ تَلْقُوا بِأَيْدِيكُمُ إِلَى التَّهْلُكَةِ ) [ 194 ] . <br />
قال النبي [ عليه السلام ](   {[6233]} ) :  مَنْ أَنْفَقَ نَفَقَةً(   {[6234]} ) فِي سَبِيلِ الله كُتِبَتْ لَهُ بِسَبْعِمائة(   {[6235]} ) ضِعْفٍ(   {[6236]} ) " (   {[6237]} ) . <br />
و " إِلَى "  متعلقة ب( تُلْقُوا ) .  والباء متعلقة بالمصدر عند المبرد(   {[6238]} ) ،  وهي زائدة(   {[6239]} ) عند الأخفش(   {[6240]} ) . <br />
والتهلكة :  الهلاك .  حضّ الله المسلمين على النفقة في سبيله والجهاد لئلا يقوى العدو ،  فتصير عاقبة أمرهم إلى الهلاك . <br />
والتهلكة عند سفيان(   {[6241]} ) :  ترك النفقة في سبيل الله عز وجل(   {[6242]} ) . <br />
وقال ابن عباس :   " التهلكة الإمساك عن النفقة في سبيل الله تعالى " (   {[6243]} ) وقال ابن زيد وغيره :   " معناه :  لا تخرجوا إلى الغزو بغير نفقة ،  أمروا(   {[6244]} ) أن ينفقوا في سبيل الله وأن لا يخرجوا بغير نفقة ،  فيهلكوا أنفسهم " (   {[6245]} ) . <br />
وقال زيد بن أسلم :   " كان رجال يخرجون إلى البعوث(   {[6246]} ) بغير نفقة ،  فإما أن يقطع بهم ،  وإما أن يكونوا عالة على الناس ،  فأمروا ألا يخرجوا على تلك الحال " (   {[6247]} ) . <br />
وقال البراء بن عازب :   " ( وَلاَ تَلْقُوا(   {[6248]} ) بِأَيْدِيكُمُ إِلَى التَّهْلُكَةِ ) :  هو الرجل يصيب الذَّنْبَ فيلقي بيده إلى التهلكة ،  يقول :   " لا توبة لي "  .  فأمروا ألا ييأسوا من رحمة الله عز وجل " (   {[6249]} ) . <br />
وقال أبو قلابة(   {[6250]} ) :  هو الرجل يصيب الذنوب ،  فيقول :   " لا توبة لي "  ،  فينهمك في المعاصي ،  [ فأمر/ألا ييأس ] (   {[6251]} ) من رحمة الله سبحانه " (   {[6252]} ) .  وقال أبو أيوب الأنصاري(   {[6253]} ) :   " فينا(   {[6254]} ) نزلت هذه الآية ،  وذلك أنا(   {[6255]} )/معشر الأنصار لما أعز الله دينه قلنا سراً :  إن أموالنا قد ضاعت ،  فلو أقمنا فيها نصلحها "  .  فأنزل الله عز وجل يرد علينا ما(   {[6256]} ) قد هممنا به من التخلف عن الجهاد " (   {[6257]} ) . <br />
فمعناه :  ( وَلاَ تُلْقُوا بِأَيْدِيكُمُ إِلَى التَّهْلُكَةِ ) أي :  لا تتأخروا عما هو أنفع لكم وهو الغزو(   {[6258]} ) .  والعرب تقول :   " ألْقَى فلان بيديه " (   {[6259]} ) إذا استسلم(   {[6260]} ) . <br />
قوله :  ( وَأَحْسِنُوا ) [ 194 ] . <br />
قيل :  معناه :  أحسنوا الظن بالله عز وجل في المغفرة لمن تاب .  هذا على قول من قال :  ( وَلاَ تَلْقُوا بِأَيْدِيكُمُ إِلَى التَّهْلُكَةِ ) .  معناه في الذنوب ،  وألا(   {[6261]} ) ييأس من رحمة الله عز وجل(   {[6262]} ) . <br />
وقيل معناه(   {[6263]} ) :  أحسنوا الإنفاق(   {[6264]} ) . <br />
وقيل :  معناه :  أداء(   {[6265]} ) الفرائض(   {[6266]} ) . <br />
وقيل :  معناه :  أحسنوا(   {[6267]} ) الظن بالله تعالى أنه يضاعف(   {[6268]} ) الحسنات ويخلف النفقة(   {[6269]} ) . <br />
قوله :  ( فَمَا اسْتَيْسَرَ مِنَ الهَدْيِ ) [ 195 ] . <br />
 " ما "  في موضع رفع ،  أي :  فعليه ذلك(   {[6270]} ) . <br />
وقيل :  فوجب عليه(   {[6271]} ) ما استيسر .  والمعنى واحد . <br />
وقيل :  هي في موضع نصب تقديره :  فَليُهدِ ما استيسر من الهدي(   {[6272]} ) . <br />
قال أبو عمرو(   {[6273]} ) :   " الهدي(   {[6274]} ) جمع .  واحده(   {[6275]} ) :  هَدْيَةٌ ،  كَتَمْرَةٍ وتَمْرٍ " (   {[6276]} ) . <br />
وقيل :  هو مصدر لا واحد له كرجال صَومٍ .  فهو يقع للواحد والجمع والتأنيث كأنه مصدر  " هدى إلى البيت هدياً "  وبنو تميم يُثقلون ياء الهدي(   {[6277]} ) . <br />
وقال الفراء :   " لا واحد له " (   {[6278]} ) . <br />
قوله :  ( فَفِدْيَةٌ )(   {[6279]} ) [ 195 ] أي فعليه فدية . <br />
ويجوز النصب على(   {[6280]} ) معنى :  فليُفد فديةً ،  وفليأت فدية . <br />
