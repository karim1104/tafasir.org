قوله :  ( وَإِن كُنتُمْ عَلَى سَفَرٍ وَلَمْ تَجِدُوا كَاتِباً ) الآية [ 282 ] . <br />
قوله :  ( فَرِهَانٌ ) [ 281 ] هو جمع رهن ،  كبَغْلٍ وبِغَالٍ وكَبْشٍ وكِبَاشٍ(   {[9165]} ) . <br />
ومن قرأ :   " فَرُهُنٌ " (   {[9166]} ) فهو جمع الجمع .  هو(   {[9167]} ) جمع  " رِهَانٍ "  ك " كِتَابٍ "  :  و " كُتُبٍ "  ، / و " حِمَارٍ "  و " حُمُرٍ " (   {[9168]} ) . <br />
وقيل :  هو جمع  " رَهْنٍ "  ،  ك " سَقْفٍ "  و " سُقُفٍ " (   {[9169]} ) . <br />
ومن قرأ  " فَرُهْنٌ "  بالإسكان(   {[9170]} )/ فإنما أسكن(   {[9171]} ) الضمة(   {[9172]} ) لثقلها . <br />
وقرأ ابن عباس  " كِتَاباً " (   {[9173]} ) ،  وقال :   " قد لا توجد(   {[9174]} ) الصحيفة " (   {[9175]} ) ،  وكذلك قرأ أبو العالية وعكرمة والضحاك ومجاهد(   {[9176]} ) .  وهو(   {[9177]} ) واحد الكتب . <br />
وقيل :  هو جمع  " كَاتِبٍ "  ،  كما يقال :   " قَائِمٌ "  و " قِيَامٌ " (   {[9178]} ) . <br />
وروي أيضاً عن ابن عباس :  ( وَلَمْ تَجِدُوا )(   {[9179]} )  " كُتَّاباً "  ،  على وزن  " فُعَّالٍ "  وهو جمع  " كاتب "  ،   " كضَارِبٍ "  و " ضُرَّابٍ "  . <br />
وهذه الآية أرخص الله فيها في قبض الرهان عند عدم الكاتب ،  والرهن لا يكون رهناً حتى يقبض من مالكه بقوله :  ( مَّقْبُوضَةٌ ) ،  سواء قبضه المرتهن عنده أو جعله على يدي عدل(   {[9180]} ) عند مالك(   {[9181]} ) . <br />
قوله : / ( فإِنَ اَمِنَ بَعْضُكُم بَعْضاً فَلْيُؤَدِّ الذِي اؤتُمِنَ )(   {[9182]} ) [ 282 ] . <br />
أي إن(   {[9183]} ) ترك صاحب الدين ،  أخذ الرهن ،  وأمن الذي عليه الدين ،  فليؤد ما عليه لأنها(   {[9184]} ) أمانة(   {[9185]} ) ،  وليتق الله ربه فيما قد اؤتمن به(   {[9186]} ) . <br />
وقال الضحاك :   " هذه الأمانة التي فسح فيها(   {[9187]} ) إنما في(   {[9188]} ) السفر ،  دون الحضر " (   {[9189]} ) . <br />
قوله :  ( وَلاَ تَكْتُمُوا الشَّهَادَةَ ) الآية [ 282 ] .  هو(   {[9190]} ) نهي للشهداء وتحذير لهم . <br />
( فَإِنَّهُ ءَاثِمٌ قَلْبُهُ ) [ 282 ] .  أي فاجر قلبه(   {[9191]} ) . <br />
( وَاللَّهُ بِمَا تَعْمَلُونَ عَلِيمٌ ) [ 282 ] . <br />
أي يعلم ما تصنعون في شهاداتكم(   {[9192]} ) من إحالتها ،  والإتيان بها على وجهها فيحصي ذلك عليكم/ ويجازيكم به(   {[9193]} ) .
