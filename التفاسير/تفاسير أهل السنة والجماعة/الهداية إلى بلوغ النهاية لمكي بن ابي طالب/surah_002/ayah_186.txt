قوله :  ( وَإِذَا سَأَلَكَ عِبَادِي عَنِّي ) [ 184 ] إلى قوله :  ( وَأَنْتُمْ تَعْلَمُونَ ) [ 187 ] . <br />
قال الحسن :   " هذه الآية نزلت في سائل سأل النبي [ عليه السلام ](   {[5889]} ) فقال :  أي ربنا ؟  فأنزل الله عز وجل(   {[5890]} ) :  ( وَإِذَا سَأَلَكَ عِبَادِي عَنِّي ) الآية " (   {[5891]} ) . <br />
وروي أن/ سائلاً سأل [ النبي عليه السلام ](   {[5892]} ) فقال :  يا(   {[5893]} ) محمد :  أقريب ربنا فنناجيه أم بعيد فنناديه ؟  فأنزل الله [ جل ذكره ](   {[5894]} ) :  ( وَإِذَا سَأَلَكَ عِبَادِي عَنِّي ) .  الآية(   {[5895]} ) . <br />
وروي أن المشركين قالوا :  كيف يكون الله قريباً(   {[5896]} ) وبيننا وبينه سبع سماوات غلاظ ،  كل سماء مسيرة خمسمائة عام(   {[5897]} ) ،  وبين كل سماءين كذلك ؟  فأنزل الله سبحانه :  ( وَإِذَا سَأَلَكَ عِبَادِي عَنِّي ) .  الآية(   {[5898]} ) . <br />
وقال عطاء :   " لما نزلت :  ( وَقَالَ رَبُّكُمُ ادْعُونِي أَسْتَجِبْ لَكُمُ ) [ غافر :  80 ] ،  قالوا :  يا رسول الله ،  في أي ساعة ؟  قال :  فنزلت :  ( وَإِذَا سَأَلَكَ عِبَادِي عَنِّي ) .  الآية(   {[5899]} ) . <br />
قال السدي :   " ليس من عبد مؤمن يدعو الله إلا استجاب له ،  فإن كان الذي يدعو به رَزقه في الدنيا ،  وإن لم يكن له رزقاً في الدنيا ،  ادُّخِر له إلى يوم القيامة أو دفع [ به عنه ](   {[5900]} ) مكروه(   {[5901]} ) .  وكذلك قال ابن عباس . <br />
وعن النبي عليه السلام(   {[5902]} ) أنه قال :   " ما أُعْطِيَ أَحَدٌ الدُّعَاء فَمُنِع الإِجَابَةَ لأنَّ اللَّهَ يَقُولُ :  ( ادْعُونِي أَسْتَجِبْ لَكُمُ ) " (   {[5903]} ) . <br />
والمعنى عند الطبري :   " فإني(   {[5904]} ) قريب في كل/وقت أجيب دعوة الداعي إذا(   {[5905]} ) دعان " (   {[5906]} ) . <br />
وقال مجاهد :   " لما نزلت :  ( ادْعُونِي أَسْتَجِبْ لَكُمُ ) ،  قالوا :  إلى أين ؟  فنزلت(   {[5907]} ) :  ( فَأَيْنَمَا تُوَلُّوا فَثَمَّ وَجْهُ اللَّهِ )(   {[5908]} ) .  " <br />
وقال قتادة :   " لما نزلت :  ( ادْعُونِي أَسْتَجِبْ لَكُمُ ) ،  قال قوم :  كيف ندعو يا رسول الله ؟  فنزلت ( وَإِذَا سَأَلَكَ ) "  الآية(   {[5909]} ) . <br />
وقوله(   {[5910]} ) :  ( أُجِيبُ دَعْوَةَ الدَّاعِ إِذَا دَعَانِ )(   {[5911]} ) [ 185 ] . <br />
فمعناه :  إذا شئت كما قال :  ( فَيَكْشِفُ مَا تَدْعُونَ إِلَيْهِ إِنْ شَاءَ )(   {[5912]} ) . <br />
وقوله ( فَلْيَسْتَجِيبُوا لِي وَلْيُومِنُوا بِيَ ) [ 185 ] . <br />
أي فليستجيبوا إلى طاعتي ،  يقال :   " استَجبْتُ لَهُ(   {[5913]} ) واسْتَجَبْتُهُ "  بمعنى أجبته(   {[5914]} ) . <br />
وقال أبو(   {[5915]} ) عبيدة :   " معناه :  فليجيبوني " (   {[5916]} ) . <br />
وتحقيق اللفظ عند/أهل العلم :  فليستدعوا الإجابة ، /كما يقال :   " استنصر "  ،  إذا استدعى النصر . <br />
وعن أبي رجاء الخراساني أنه قال :   " ( فَلْيَسْتَجِيبُوا لِي )(   {[5917]} ) فليدعوني " (   {[5918]} ) .  وقيل :  هو التلبية . <br />
وقوله :  ( وَلْيُومِنُوا بِيَ ) [ 185 ] :  أي وليصدقوا بي إذا هم استجابوا لي بالطاعة أني لهم من وراء طاعتهم لي في الثواب عليها ،  وإجزال(   {[5919]} ) الكرامة عليها . <br />
وقال أبو رجاء :  ( وَلْيُومِنُوا بِيَ ) :  معناه :   " وليصدقوا(   {[5920]} ) بي ) "  أني أستجيب لهم(   {[5921]} ) . <br />
وقوله :  ( لَعَلَّهُمْ يَرْشُدُونَ ) [ 185 ] .  معناه :  لعلهم يهتدون .  و " لعل "  من الله واجبة . <br />
وقيل :  معنى الإجابة هنا ،  هو الإجابة بالثواب على الأعمال [ والطاعات ](   {[5922]} ) ،  فمعنى الدعاء هنا مسألة العبد ربه ،  إتمام ما وعده إياه من الجزاء على الطاعة(   {[5923]} ) . <br />
وروي عن النبي [ عليه السلام ](   {[5924]} ) .  أنه قال :   " الدُّعَاءُ هُوَ العبادَةُ "  ،  ثم قرأ ( وَقَالَ رَبُّكُمُ ادْعُونِي أَسْتَجِبْ لَكُمُ إِنَّ الذِينَ يَسْتَكْبِرُونَ عَنْ عِبَادَتِي )(   {[5925]} )(   {[5926]} ) "  . <br />
وقيل :  معناه :  أجيب دعوة الداعي إن شئت(   {[5927]} ) . <br />
وقال الحسن في قوله تعالى :  ( ادْعُونِي أَسْتَجِبْ لَكُمُ ) :   " اعملوا وأبشروا فإنه حق على الله أن يستجيب للذين آمنوا وعملوا الصالحات ويزيدهم من فضله " (   {[5928]} ) . <br />
وقوله :  ( فَإِنِّي قَرِيبٌ ) [ 185 ] .  وقْف عند يعقوب(   {[5929]} ) .  وقال نصير(   {[5930]} ) :  الوقف  " دعانِ "  و " يَرْشُدُونَ " (   {[5931]} ) . <br />
و(   {[5932]} )الفعل من  " يرْشُدونَ "  :  رَشَدَ يَرْشُدُ رُشْداً ،  ويقال :  رَشَدَ يَرْشَدُ(   {[5933]} ) رَشَداً ،  فيقال :  الرُّشْدُ وَالرَّشَدُ(   {[5934]} ) ،  كما يقال(   {[5935]} ) :  البُخْلُ والبَخَلُ ،  والشُّغْلُ والشَّغَلُ ،  وَالسُّقْمُ وَالسَّقَمُ ،  وَالعُدْمُ والعَدَمُ وَالحُزْنُ ،  والحَزَنُ ،  وَالسُّخْطُ وَالسَّخَطُ ،  وَالخُبْرُ وَالخَبَرُ ،  وَالعُرْبُ وَالعَرَبُ ،  وَالعُجْمُ وَالعَجَمُ .
