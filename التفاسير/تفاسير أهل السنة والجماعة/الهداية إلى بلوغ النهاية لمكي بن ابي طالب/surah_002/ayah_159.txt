ثم قال(   {[4985]} ) :  ( اِنَّ الذِينَ يَكْتُمُونَ مَا أَنْزَلْنَا مِنَ البَيِّنَاتِ وَالهُدَى ) [ 158 ] . <br />
يعني أحبار(   {[4986]} ) اليهود وعلماء النصارى كتموا أمر محمد صلى الله عليه وسلم ونبوته وتركوا اتباعه مع كونه مذكوراً عندهم في التوراة والإنجيل موصوفاً(   {[4987]} ) . <br />
وذكر عن ابن(   {[4988]} ) عباس أن معاذ بن(   {[4989]} ) جبل سال بعض أحبار(   {[4990]} ) اليهود عن ما في التوراة من ذكر النبي عليه السلام فكتموه إياه ،  فأنزل الله عز وجل ( اِنَّ الذِينَ يَكْتُمُونَ )(   {[4991]} ) .  إلى :  ( اللاَّعِنُونَ )(   {[4992]} ) .  وهو قول مجاهد(   {[4993]} ) . <br />
وقال قتادة :   " هم أهل الكتاب كتموا الإسلام ،  وهو دين الله سبحانه وكتموا محمداً وهم يجدونه مكتوباً عندهم في التوراة والإنجيل " (   {[4994]} ) . <br />
/وقوله :  ( مِنْ بَعْدِ مَا بَيَّنَّاهُ لِلنَّاسِ فِي الكِتَابِ )(   {[4995]} ) [ 158 ] . <br />
يعني بالكتاب هنا :  التوراة والإنجيل .  وهذه الآية وإن كانت خاصة لبعض الناس فإنها عامة لكل من سئل عن علم يعلمه فكتمه . <br />
وقال النبي [ صلى الله عليه وسلم ](   {[4996]} )  " مَنْ سُئِلَ عَنْ عِلْمٍ يَعْلَمُهُ فَكَتَمَهُ أُلْجِمَ يَوْمَ القِيَامَةِ بِلِجَامٍ مِنَ النَّارِ(   {[4997]} ) " (   {[4998]} ) . <br />
ثم قال :  ( أُوْلاَئِكَ يَلْعَنُهُمُ اللَّهُ وَيَلْعَنُهُمُ اللاَّعِنُونَ ) [ 158 ] . <br />
أي :  أولئك الذين كتموا أمر محمد صلى الله عليه وسلم يعلنهم الله عز وجل بكتمانهم ،  أي :  يبعدهم(   {[4999]} ) الله من رحمته ويطردهم . <br />
وقوله(   {[5000]} ) :  ( وَيَلْعَنُهُمُ اللاَّعِنُونَ ) [ 158 ] . <br />
قيل :  معناه :  ويسأل ربَّهم اللاعنون أن(   {[5001]} ) يلعنهم(   {[5002]} ) لأن كل لاعن إنما يقول :   " اللهم العن هذا "  . <br />
وقيل(   {[5003]} ) :  اللاعنون البهائم ؛  إذا أسْنَتَتِ(   {[5004]} ) السَّنَةُ ،  قالت البهائم :   " هذا من أجل(   {[5005]} ) عصاة بني آدم ،  لعن الله عصاة بني آدم " (   {[5006]} ) . <br />
وقال عكرمة :   " يلعنهم كل شيء حتى الخنافس والعقارب/ يقولون :  منعنا القطر بذنوب بني آدم " (   {[5007]} ) . <br />
وإنما جُمِعَ جَمْعَ السلامة(   {[5008]} ) على هذا القول ،  وهو ما لا يعقل لأنه لما أخبر عنهم بمثل ما يخبر عن بني آدم جمعهم كما يجمع بني آدم(   {[5009]} ) . <br />
وقال قتادة :   " اللاعنون هنا :  الملائكة والمؤمنون " (   {[5010]} ) .  وقاله الربيع(   {[5011]} ) . <br />
قال مقاتل :   " اللاعنون كل ما على وجه الأرض إلا الثقلين :  الجن والإنس ،  وذلك أن الكافر إذا أدخل(   {[5012]} ) قبره ضربته/الملائكة بمقمعة حين تقول له :  من ربك ؟  فيقول :  لا أدري ،  فيصيح صيحة يسمعها كل ما على وجه الأرض من غير الجن والإنس ،  فلا تَقِر تلك الصيحة في سمع شيء إلا لعنه "  . <br />
قال مجاهد :   " اللاعنون البهائم " (   {[5013]} ) . <br />
ولما وصفت باللعنة :  جاز جمعها(   {[5014]} ) بالواو والنون ،  وإن كانت لا تعقل .  وله نظائر كثيرة(   {[5015]} ) . <br />
وقيل : / اللاعنون الملائكة الذين يسوقون أهل الكفر إلى النار .  والهاء في ( خَالِدِينَ فِيهَا ) تعود على اللعنة(   {[5016]} ) . <br />
واللعن أصله الطرد والبعد من الرحمة(   {[5017]} ) . <br />
قال مقاتل :   " اللعنة النار "  .  وحسن ذلك عنده لما كانت عاقبة اللعنة المصير إلى النار . <br />
وقال السدي :   " اللاعنون ما عدا بني آدم والجن " (   {[5018]} ) . <br />
وروي عن البراء بن(   {[5019]} ) عازب :   " أن الكافر إذا وضع في قبره أتته دابة كأن عينيها(   {[5020]} ) قدران(   {[5021]} ) من نحاس ،  معها عمود من حديد ،  فتضربه ضربة بين كتفيه فيصيح ،  فلا يسمع أحد صوته إلا لعنه ،  ولا(   {[5022]} ) يبقى شيء إلا سمع صوته إلا الثقلين :  الجن والإنس " (   {[5023]} ) . / وهو قول الضحاك(   {[5024]} ) . <br />
ويروى عن ابن مسعود أنه قال :   " اللاعنون :  الاثنان إذا تلاعنا ألحقت اللعنة مستحقها منهما ،  فإن لم يستحقها واحد منهما(   {[5025]} ) رجعت على اليهود " (   {[5026]} ) . <br />
