ثم قال تعالى :  ( وَأَوْصَى بِهَا إِبْرَاهِيمُ بَنِيهِ وَيَعْقُوبُ ) [ 131 ] . <br />
أي :  وأوصى بقوله :  ( أَسْلَمْتُ لِرَبِّ العَالَمِينَ ) إبراهيم –صلى الله عليه وسلم [ و ](   {[4228]} ) على محمد(   {[4229]} )- بنيه ،  وأوصى بها يعقوب صلى الله عليه وسلم [ و ](   {[4230]} ) على محمد [ بنيه ](   {[4231]} ) . <br />
والهاء في  " بها " (   {[4232]} ) تعود على كلمة الإسلام وهي قوله :  ( أَسْلَمْتُ )(   {[4233]} ) . <br />
وقيل :  تعود على الملة(   {[4234]} ) ،  وكلمة الإسلام أقرب إليها(   {[4235]} ) . <br />
[ وقيل :  بل ](   {[4236]} ) أوصاهم باتباع الملة ،  ف " يعقوب "  على هذا معطوف على  " إبراهيم " (   {[4237]} ) . <br />
وقيل :  إن يعقوب مرفوع بإضمار(   {[4238]} ) فعل .  والتقدير  " وقال يعقوب :  يا بني إن الله " (   {[4239]} ) . <br />
والمعنى في  " أَوْصَى "  عهد إليهم بذلك ،  وأمرهم به . <br />
قال ابن عباس :   " وصّاهم بالإسلام " (   {[4240]} ) . <br />
وفي التشديد في  " وصَّى "  معنى تكرير الوصية . <br />
وقوله :  ( إِنَّ اللَّهَ اصطَفَى لَكُمُ الدِّينَ ) [ 131 ] . <br />
معناه :  اختاره لكم .  ودخلت الألف واللام في  " الدين "  لتقدم علمهم(   {[4241]} ) به وتكرير الوصية عليهم . <br />
ثم قال :  ( فَلاَ تَمُوتُنَّ إِلاَّ وأَنْتُمْ مُّسْلِمُونَ ) [ 132 ] . <br />
أي :  فاتقوا الله أن تموتوا إلا على الإسلام . <br />
والمعنى :  لا تفارقن(   {[4242]} ) هذا الدين أيام حياتكم لأن أحداً لا يدري متى تأتيه منيته ،  فلذلك قال لهم :  لا تموتن إلا وأنتم مسلمون ،  لأنكم لا تدرون متى يأتيكم الموت ،  ولم ينههم عن الموت/ لأن ذلك ليس إليهم . <br />
وقيل :  المعنى :  الزموا الإسلام ،  فإذا أدرككم الموت صادفكم(   {[4243]} ) مسلمين(   {[4244]} ) . <br />
وعرف المعنى كما عرف في قول العرب/  " لا أَرَيَنَّكَ(   {[4245]} ) هَا هُنَا "  .  فالنهي في اللفظ للمتكلم ،  وفي المعنى للمتكلم أي :  لا تكن ها هنا ،  فإنه من يكن ها هنا أراه(   {[4246]} ) . <br />
قال الأخفش :   " ( بَنِيهِ ) ،  قطع ،  ثم يبتدئ :  ( وَيَعْقُوبُ يَا بَنِيَّ ) ،  أي :  وقال يعقوب :  يا بني " (   {[4247]} ) . <br />
وقال أبو حاتم وغيره :   " الوقف ( وَيَعْقُوبُ ) ،  ثم يبتدئ(   {[4248]} ) ( يَا بَنِيَّ ) " (   {[4249]} ) .  أي : (   {[4250]} ) وقال كل واحد منهما :  ( يَا بَنِيَّ إِنَّ اللَّهَ اصْطَفَى(   {[4251]} ) لَكُمُ(   {[4252]} ) الدِّينَ ) .
