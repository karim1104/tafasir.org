قوله :  ( اللَّهُ لاَ إِلَهَ إِلاَّ هُوَ الحَيُّ القَيُّومُ ) الآية [ 253 ] . <br />
أجاز النحاس :  لا إله إلا إياه على الاستثناء(   {[8265]} ) . <br />
وروي عن أبي ذر أنه سأل النبي [ عليه السلام ](   {[8266]} ) فقال له :  أي ما أنزل عليك من القرآن أعظم "  .  فقال النبي صلى الله عليه وسلم :  ( اللَّهُ لاَ إِلَهَ إِلاَّ هُوَ الحَيُّ القَيُّومُ )(   {[8267]} ) . <br />
وقال ابن عباس :   " أشرف آية في القرآن آية الكرسي "  .  نبه الله تعالى ذكره عباده بهذه الآية ألا يعبد(   {[8268]} ) غيره ،  وأن يحذر(   {[8269]} ) مما وقع فيه من تقدم ذكره ،  في قوله :  ( فَمِنْهُم مَّنَ ءَامَنَ وَمِنْهُم مَّن كَفَرَ ) واختلفوا فاقتتلوا(   {[8270]} ) وشبهه . <br />
وروى مالك عن يحيى بن سعيد(   {[8271]} ) عن ابن المسيب أنه قال :  قال رسول الله صلى الله عليه وسلم :   " مَنْ قَرَأَ آيَةَ الكُرْسِيِّ إِذَا نَامَ ،  لَمْ يَزَلْ فِي أَمَانِ اللَّهِ حَتَّى يَنْتَبِهَ(   {[8272]} ) .  وَمَنْ قَرَأَهَا إِذَا انْتَبَهَ ،  لَمْ يَزَلْ فِي أَمَانِ اللَّهِ حَتَّى يَعُودَ ،  وَمَنْ قَرَأَهَا إِذَا خَرَجَ(   {[8273]} ) مِنْ مَنْزِلِهِ لَمْ يَزَلْ فِي أَمَانِ(   {[8274]} ) اللَّهِ عز وجل(   {[8275]} ) حَتَّى(   {[8276]} ) يَعُودَ .  وَمَنْ قَرَأَهَا عِنْدَ حِجَامَةٍ كَانَتْ لَهُ مَنْفَعَتَانِ :  مَنْفَعَةٌ لِلْحِجَامَةِ الَّتِي هُوَ فِيهَا ،  وَمَنْفَعَةٌ لِلْحِجَامَةِ الَّتِي تَكُونُ(   {[8277]} ) بَعْدَهَا .  وَمَنْ قَرَأَهَا دُبُرَ كُلِّ صَلاَةٍ ،  أَدْخَلَهُ اللَّهُ(   {[8278]} ) الجَنَّةَ(   {[8279]} ) "  . <br />
وروى أبو هريرة أن النبي [ عليه السلام ](   {[8280]} ) قال :   " لِكُلِّ شَيْءٍ سَنَامٌ ،  وَسَنَامُ القُرْآنِ سُورَةُ البَقَرَةِ ،  مِنْهَا آيَةٌ لاَ تُقْرَأُ فِي بَيْتٍ فِيهِ شَيْطَانٌ إِلاَّ خَرَجَ مِنْهُ ،  وَهِيَ آيَةُ الكُرْسِيِّ " (   {[8281]} ) . <br />
وقوله :  ( الْحَيُّ القَيُّومُ ) [ 253 ] .  الحي الذي لا يموت ،  والقيوم الذي لا يزول(   {[8282]} ) . <br />

وقوله :  ( القَيُّومُ ) [ 253 ] . <br />
قال ابن عباس :   " معناه :  الذي لا يزول " (   {[8287]} ) . <br />
وقال مجاهد :   " معناه القائم على كل شيء(   {[8288]} ) "  . <br />
وهو  " فَيْعُولٌ "  من  " قَامَ "  ،  ولا يحسن أن يكون  " فَعُولاً "  لأنه يلزم منه أن يقال :   " قَوُوُمٌ " (   {[8289]} ) . <br />
وقد قال ابن كيسان :   " ليس في كلام العرب ،  فعول من الواو "  . <br />
وروي عن عمر أنه/قرأ  " الْقَيَّامُ "  ووزنه  " فَيْعَالٌ "  من  " قَامَ " (   {[8290]} ) . <br />
وقرأ علقمة  " الْقَيِّمُ(   {[8291]} ) " (   {[8292]} ) ،  ووزنه عند البصريين  " فعيل "  ،  ثم أدغم فكان أصله قيوماً ،  وأصله عند الكوفيين  " قويم "  مثل فعيل ،  ويلزمهم ألا يعل كما لم(   {[8293]} ) يعل  " طويل "  وشبهه . <br />
وصفات/الله مطلقة في غاية الكمال والتمام ،  لا يجوز عليها حوالة ولا تغيير(   {[8294]} ) ،  بخلاف صفات المخلوقين . <br />

وقال(   {[8300]} ) الربيع :   " السنة هو الذي يكون به الإنسان بين النائم واليقظان ،  وهو الوسنان ،  والنوم الاستثقال(   {[8301]} ) "  . <br />

وقد روى أبو هريرة أن النبي صلى الله عليه وسلم(   {[8302]} ) قال :   " وَقَعَ فِي نَفْسِ مُوسَى صلى الله عليه وسلم(   {[8303]} ) :  هَلْ يَنَامُ اللهُ جَلَّ ثَنَاؤُهُ ؟  فَأَرْسَلَ اللَّهُ(   {[8304]} ) إِلَيْهِ مَلَكاً فَأَرَّقَهُ ثَلاَثاً ثُمَّ أَعْطَاهُ قَارُورَتَيْنِ ،  فِي كُلِّ يَدٍ قَارُورَةٌ ،  وَأَمَرَهُ أَنْ يَتَحَفَّظَ بِهِمَا .  فَجَعَلَ يَنَامُ وَتَكَادُ يَدَاهُ/تَلْتَقِيَانِ(   {[8305]} ) ثُمَّ يَسْتَيْقِظُ فَيَحْبِسُ إِحْدَاهُمَا عَنْ الأُخْرَى ،  حَتَّى نَامَ نَوْمَهُ فَاصْطَفَقَتْ يَدَاهُ فَانْكَسَرَتْ القَارُورَتَانِ ،  فَجَعَلَ اللَّهُ لَهُ(   {[8306]} ) ذَلِكَ مَثَلاً(   {[8307]} ) "  ،  أي أن الله لو كان ينام(   {[8308]} ) لم تمتسك(   {[8309]} ) السموات والأرضون كما لم تمتسك(   {[8310]} ) القارورتان في يدي موسى عليه السلام . <br />
قوله :  ( مَن ذَا الذِي يَشْفَعُ عِندَهُ إِلاَّ بِإِذْنِهِ ) [ 254 ] . <br />
أي من يشفع لمن أراد الله عقوبته إلا بأمره لهم بالشفاعة .  وهذا يدل على جواز الشفاعة بإذنه لمن شاء من رسله وأوليائه .  وقيل :  معناه(   {[8311]} ) :  من ذا الذي يذكر الله بقلبه حتى يأذن له ،  لا إله إلا هو . <br />
قوله :  ( يَعْلَمُ مَا بَيْنَ أَيْدِيهِمْ وَمَا خَلْفَهُمْ(   {[8312]} ) ) [ 254 ] . <br />
قال ابن جريج :   " يعلم ما مضى أمامهم من الدنيا ( وَمَا خَلْفَهُمْ ) ما يكون بعدهم من أمر الدنيا والآخرة " (   {[8313]} ) . <br />

قال ابن جبير عن ابن عباس :   " كرسيه :  علمه "  ،  ودل على ذلك قوله :  ( وَلاَ يَئُودُهُ حِفْظُهُمَا )(   {[8314]} ) . <br />
قال أبو هريرة :   " الكرسي بين يدي العرش " (   {[8315]} ) . <br />
قال مجاهد :   " ما السموات والأرض في الكرسي إلا كحلقة بأرض فلاة " (   {[8316]} ) . <br />
قال السدي :  السموات والأرض في جوف الكرسي بين يدي العرش وهو موضع قدميه(   {[8317]} ) "  . <br />
قال ابن زيد :   " حدثني أبي ،  قال :  قال رسول الله صلى الله عليه وسلم/ :   " مَا السَّمَوَات السَّبْع فِي الكُرْسِيِّ إِلاَّ كدراهِمَ سَبْعَةٍ أُلْقِيَتْ فِي تُرْسٍ " (   {[8318]} ) . <br />
وقال أبو ذر :  سمعت النبي [ عليه السلام ](   {[8319]} ) يقول :   " مَا الكُرْسِيُّ فِي العَرْشِ إِلاَّ كَحَلَقَةٍ مِنْ حَدِيدٍ أُلْقِيَتْ بَيْنَ ظَهْرَيْ فَلاَةٍ مِنَ الأَرْضِ(   {[8320]} ) "  . <br />

وروى الأعمش عن مجاهد أنه قال :   " مثل السموات تحت الكرسي كحلقة ملقاة في الفلاة "  . <br />
وروى ابن جبير عن ابن عباس أنه قال :   " الكرسي الذي وسع السموات والأرض(   {[8321]} ) موضعه من العرش موضعه من السرير(   {[8322]} ) ،  ولا يقدر قدر العرش إلا الذي خلقه " (   {[8323]} ) . <br />

( العَظِيمُ ) [ 254 ] . <br />
أي لا شيء أعظم منه جلالة وهيبة وسلطانا(   {[8340]} ) .  ولا يحسن أن يكون بمعنى العلو في المسافة والارتفاع من مكان إلى مكان تعالى الله عن ذلك- إنما هو علو قدرة وجلالة وهيبة وسلطان ،  لا علو ارتفاع من مكان إلى مكان ،  ليس كمثله شيء .  لا يجوز عليه الحركة ولا الانتقال ولا التغير من حال إلى حال ،  فافهمه(   {[8341]} ) . <br />
وقيل :  معنى ( العَلِيُّ ) :  العلي عن النظراء والأشباه ،  لا علو مكان(   {[8342]} ) . <br />
وقيل :  إن ( العَظِيمُ ) هنا(   {[8343]} ) بمعنى المعظم ،  بمعنى الذي يعظمه خلقه لم يزل على ذلك(   {[8344]} ) .  ولا يحسن أن يتأول أن تعظيمه محدث . /بل لم يزل معظماً قبل كون الخلق كما لم يزل قادراً وعالماً قبل كون(   {[8345]} ) المعلوم والمقدور ،  ولا يزال معظماً بعد فناء الخلق(   {[8346]} ) . <br />
وقد طعن/في هذا القول(   {[8347]} ) . <br />
وقيل :  يلزم ألا يكون معظماً قبل الخلق ولا بعدهم ،  إذ لا معظم له(   {[8348]} ) .  فالجواب عن ذلك ما ذكرنا أنه لم يزل ولا يزال كذلك كالعلم والقدرة وشبههما .  تقول العرب :   " هذه خمر عتيقة(   {[8349]} ) "  ،  بمعنى معتقة(   {[8350]} ) .  وقيل :  معنى  " العظيم "  :  أن له عظمة هي صفة له/لا تكيف(   {[8351]} ) .
