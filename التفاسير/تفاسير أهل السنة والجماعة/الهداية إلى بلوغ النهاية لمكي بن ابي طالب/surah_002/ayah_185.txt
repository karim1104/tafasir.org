قوله :  ( شَهْرُ رَمَضَانَ الذِي أُنْزِلَ فِيهِ القُرْءَانُ ) إلى قوله :  ( وَلَعَلَّكُمْ(   {[5772]} ) تَشْكُرُونَ ) [ 184 ] . <br />
أي :  المفترض عليكم شهر رمضان الذي أنزل فيه القرآن إلى سماء الدنيا وذلك ليلة أنزل الله جل ذكره القرآن من اللوح المحفوظ جميلة إلى سماء الدنيا ،  ثم نزل(   {[5773]} ) بعد ذلك نجوماً(   {[5774]} ) على ما شاء الله .  كذلك أتت الرواية عن النبي [ عليه السلام(   {[5775]} ) ] . <br />
قال ابن عباس :   " أنزل الله القرآن جملة من الذكر في ليلة أربع(   {[5776]} ) وعشرين من شهر(   {[5777]} ) رمضان فجعل في بيت العزة(   {[5778]} ) "  . <br />
وروى واثلة(   {[5779]} ) عن النبي صلى الله عليه وسلم أنه قال :   " نَزَلَتْ صُحُفُ(   {[5780]} ) إبْرَاهِيمَ(   {[5781]} ) أَوَّلَ لَيْلَةٍ مِنْ رَمَضَانَ ،  وَنَزَلَتْ التَّوْرَاةُ لِسِتٍ(   {[5782]} ) مَضَيْنَ مِنْ رَمَضَانَ .  [ وَنَزَلَ الإِنْجِيلُ لِثَلاَثِ عَشْرَةَ لَيْلَةً خَلَتْ مِنْهُ .  وَنَزَلَ الزَّبُورُ لِثَمَانِي عَشْرَةَ لَيْلَةً خَلَتْ مِنْ رَمَضَانَ .  وَنَزَلَ القُرْآنُ لِأَرْبَعٍ وَعِشْرِينَ مِنْ رَمَضَانَ(   {[5783]} ) ] .  وَكَانَ بَيْنَ نَزُولِ أَوَّلِ القُرْآنِ(   {[5784]} ) وَآخِرِهِ عِشْرُونَ سَنَةً(   {[5785]} ) "  . <br />
وقيل :  معناه شهر رمضان الذي أنزل فيه القرآن بفرضه على الناس .  فأما إعرابه على هذا المعنى ،  فيكون فيه معدّى(   {[5786]} ) إليه الفعل بحرف جر(   {[5787]} ) ،  لا ظرفاً .  وعلى القول الأول ،  يكون فيه ظرفاً للنزول(   {[5788]} ) . <br />
وروى جابر بن عبد الله أن النبي [ عليه السلام(   {[5789]} ) ] قال :   " أَنْزَلَ الله أَوَّلَ لَيْلَةٍ مِنْ رَمَضَانَ صُحُفُ [ إِبْرَاهِيمَ ،  وأَنْزَلَ التَّوْرَاةَ عَلَى مُوسَى(   {[5790]} ) ] لِسِتٍّ خَلَوْنَ مِنْ رَمَضَانَ(   {[5791]} ) ،  وَأَنْزَلَ الزَّبُورَ عَلَى داوُدَ(   {[5792]} ) لاثْنَتَيْ عَشْرَةَ لَيْلَةٍ خَلَتْ مِنْ رَمَضَانَ(   {[5793]} ) ،  وَأَنْزَلَ الإِنجِيلَ عَلَى عِيسَى(   {[5794]} ) لِثَمَاني عَشْرَةَ ليلةً خَلَتْ مِنْ رَمَضَانَ(   {[5795]} ) ،  وأنزل القرآن(   {[5796]} ) لأربع وعشرين [ ليلة خلت من رمضان(   {[5797]} ) "  . <br />
/وأكثر(   {[5798]} ) الناس على أن القرآن أنزل(   {[5799]} ) ] ليلة القدر من رمضان ،  والله أعلم أي ليلة كانت ،  وذلك كله إلى سماء الدنيا ،  ثم نزل متفرقاً(   {[5800]} ) على ما ذكرنا(   {[5801]} ) . <br />
فأما إعراب ( شَهْرُ رَمَضَانَ ) ؛  فيجوز أن يكون  " شَهْرُ(   {[5802]} ) "  رفع بالابتداء ،  و( الَّذِي أُنْزِلَ ) الخبر(   {[5803]} ) . <br />
ويجوز أن يكون التقدير :  الأيام التي تصام شهر رمضان وشبهه(   {[5804]} ) . <br />
وقرأ مجاهد وشهر بن حوشب :   " شَهْرَ(   {[5805]} ) :  "  بالنصب .  ورويت عن عاصم ونصبه عند البصريين على الإغراء ،  وعند الكوفيين بالصيام ،  وهو قبيح للتفرقة بين الصلة والموصول(   {[5806]} ) . <br />
وإنما سمي الشهر شهراً(   {[5807]} ) لشهرته ودخوله وخروجه ،  ومنه  " شَهَرَ فُلانٌ سَيْفَهُ "  إذا أخرجه من غمده(   {[5808]} ) . <br />
فمن قال :   " شَهْرُ رمضان "  قال في التثنية  " شهرَا(   {[5809]} ) رمضان " /وفي الجمع  " أَشْهُرَ رمضان "  ،  و " شَهْرَات رمضان "  . <br />
ومن قال  " رمضان "  بغير شهر قال في الجميع  " رَمَضَاناتٍ "  . <br />
وحكى الكوفيون  " رماضين "  ،  وحكوا  " أَرمِضَة "  ،  وحكي  " رُمَاضٌ "  .  ولم يتصرف لأن فيه ألفا ونوناً زائدتان ،  وهو معرفة(   {[5810]} ) . <br />
قال قطرب :   " سمي رمضان رمضان لأنهم كانوا يصومونه في الحر .  فهو مشتق من الرمضاء ،  والرمضاء الرمل الحامي من الشمس(   {[5811]} ) "  . <br />
وكره مجاهد(   {[5812]} ) أن يقال رمضان للشهر(   {[5813]} ) ،  ولا يقال إلا  " شهر رمضان "  ،  كما قال الله .  وقال :   " لعل(   {[5814]} ) رمضان اسم من أسماء الله(   {[5815]} ) "  . /وقاله عطاء . /وقد أتت الآثار عن النبي عليه السلام(   {[5816]} ) بذكر(   {[5817]} ) رمضان من غير لفظ شهر(   {[5818]} ) . <br />
روى(   {[5819]} ) مالك أن ابن عباس قال :   " إن رسول الله ذكر(   {[5820]} ) رمضان ،  فقال :   " لا تَصُومُوا حَتَى تَرَوُا الهِلالَ " (   {[5821]} ) . <br />
وقال أنس بن(   {[5822]} ) مالك :   " سافرنا مع رسول الله [ عليه السلام ](   {[5823]} ) في رمضان كثيراً " (   {[5824]} ) . <br />
وإنما سمي القرآن قرآناً لأنه يجمع السور الكثيرة من قولهم :   " قَرَأْتُ الماءَ في الإناءِ "  ،  أي جمعته وضممته(   {[5825]} ) . <br />
وقيل :  إنما سمي بذلك لأنهم يقولون :   " قَرَأتِ المَرْأَةُ "  ،  و " قرأت " (   {[5826]} ) إذا حاضت(   {[5827]} ) وإذا ولدت ،  فكأنها/تظهر شيئاً كان مستوراً .  والقارئ إذا أظهر شيئاً وبيَّنه ،  فهو من إظهار الشي وتبيينه(   {[5828]} ) . <br />
وقوله :  ( هُدىً لِلنَّاسِ ) [ 184 ] .  أي رشاد للناس إلى طريق النجاة . <br />
وقوله :  ( وَبَيِّنَاتٍ مِّنَ الهُدَى وَالفُرْقَانِ ) .  [ 184 ] . <br />
أي :  والقرآن أيضاً آيات واضحات(   {[5829]} ) من الرشاد ،  والفرق بين الحق والباطل . <br />
ثم قال :  ( فَمَن شَهِدَ مِنكُمُ الشَّهْرَ فَلْيَصُمْهُ ) [ 184 ] . <br />
أي فمن شهد منكم الشهر في المصر وهو صحيح فليصمه .  و " شَهِدَ "  بمعنى  " حضر "  .  ومن كان مريضاً في المصر أو على سفر ،  فليقطر إن شاء ،  وعليه عدة من أيام أخر . <br />
وقيل :  المعنى :  فمن دخل عليه الشهر وهو مقيم في المصر لزمه الصوم [ سافر بعد ذلك أو قام ](   {[5830]} ) .  رواه الضحاك عن ابن عباس قال :   " إذا شهدت أوله في المصر(   {[5831]} ) فصم وإن سافرت " (   {[5832]} ) . <br />
وكذلك قال السدي .  ورواه أيضاً قتادة عن علي رضي الله عنه ،  وقاله عبيدة(   {[5833]} ) .  وروي أيضاً/عن عائشة رضي الله عنها(   {[5834]} ) . <br />
وعلى القول الأول كل العلماء :  إن للمسافر(   {[5835]} ) الإفطار ،  وإن أخذه أول الشهر في المصر ،  ولا يجزي صيام إلا بتبييت(   {[5836]} ) قبل الفجر(   {[5837]} ) . <br />
ومذهب [ مالك أنه إذا بيت الصيام ](   {[5838]} ) في أول الشهر أجزأه عن(   {[5839]} ) أن يبيته في كل ليلة من بقية الشهر . <br />
وقال الشافعي وأحمد بن حنبل :   " لابد من تبييت(   {[5840]} ) الصوم في كل ليلة " (   {[5841]} ) وثبت عن/حفصة(   {[5842]} ) أنها قالت :   " لا صِيَامَ لِمَنْ لَمْ يُجْمِع الصِّيامَ قَبْلَ الفَجْرِ "  وأسندته إلى النبي [ عليه السلام ](   {[5843]} ) . <br />
ثم قال تعالى :  ( [ وَمَنْ كَانَ ](   {[5844]} ) مَرِيضاً ) [ 184 ] . <br />
أي مريضاً بمرض يشق عليه الصوم ويشتد عليه(   {[5845]} ) أفطر ،  وكذلك المسافر ،  لهما أن يفطرا ويقضيا جميعاً ،  ولا إطعام عليهما . <br />
ومن أكل أو شرب ناسياً في رمضان فعليه القضاء ولا كفارة عليه(   {[5846]} ) .  وهو قول مالك وربيعة بن عبد الرحمن وأهل المدينة(   {[5847]} ) . <br />
وعن علي وأبي هريرة وابن(   {[5848]} ) عمر أنه :   " لا قضاء عليه " (   {[5849]} ) . <br />
و[ به ](   {[5850]} ) قال عطاء وطاوس والنخعي والثوري(   {[5851]} ) والأوزاعي وابن أبي ذيب(   {[5852]} ) والشافعي(   {[5853]} ) وأحمد بن حنبل وأبو ثور وأصحاب الرأي(   {[5854]} ) . <br />
ومن وطئ نهاراً(   {[5855]} ) في رمضان ناسياً فعليه القضاء عند مالك ولا كفارة عليه ،  وهو قول عطاء والليث بن(   {[5856]} ) سعد والأوزاعي(   {[5857]} ) . <br />
وقال مجاهد والحسن والثوري والشافعي(   {[5858]} ) وإسحاق وأبو ثور وأصحاب الرأي :   " لا قضاء عليه ولا كفارة " (   {[5859]} ) . <br />
وقال أحمد بن حنبل :   " عليه القضاء والكفارة " (   {[5860]} ) . <br />
وأجمعوا على أن من أكل ناسياً فظن أن ذلك قد فطره(   {[5861]} ) فجامع عامداً أن عليه القضاء ولا كفارة عليه(   {[5862]} ) . <br />
ومن أكل أو شرب عامداً في نهار رمضان فعليه القضاء والكفارة كالمجامع عامداً . <br />
وهو قول مالك والزهري ،  وبه قال الحسن وعطاء .  وهو قول الثوري والأوزاعي وأبي(   {[5863]} ) ثور وإسحاق وأصحاب الرأي(   {[5864]} ) . <br />
وقال ابن المسيب :   " عليه صوم شهر "  . <br />
وقال ربيعة بن(   {[5865]} ) أبي عبد الرحمن :   " عليه صوم اثني(   {[5866]} ) عشر يوماً على عدة الشهور "  . <br />
وعن عطاء أنه  " لا قضاء عليه ،  وعليه تحرير رقبة ،  فإن لم يجد فبدنة ،  أو بقرة أو عشرون صاعاً طعاماً للمساكين "  . <br />
وعن النخعي أن  " عليه ثلاث آلاف(   {[5867]} ) يوم "  . <br />
وعن ابن عباس/أن  " عليه عتق رقبة ،  أو صوم شهر ،  أو إطعام ثلاثين مسكيناً "  . <br />
و(   {[5868]} )روي عن علي رضي الله عنه أنه قال :   " مَن أفطر يوماً في رمضان متعمداً لم يقضه أبداً طول الدهر "  .  وروي ذلك عن ابن مسعود(   {[5869]} ) . <br />
وعن الزهري أيضاً أنه قال :   " إن كان فعل ذلك ابتداعاً لدين غير الإسلام ،  ضربت عنقه ،  وإن كان فعل ذلك فسقاً جلد "  . <br />
ومن استقاء(   {[5870]} ) فقاء ،  فعليه القضاء ولا كفارة عليه . <br />
وهو قول علي بن أبي طالب(   {[5871]} ) وزيد بن أرقم(   {[5872]} ) ،  وابن(   {[5873]} ) عمر وعلقمة ،  وهو قول الزهري ومالك والشافعي(   {[5874]} ) وأحمد(   {[5875]} ) وأصحاب الرأي . <br />
وقال عطاء وأبو ثور :   " عليه الكفارة مع القضاء "  .  فإن دَرعه(   {[5876]} ) القيء فقاء فلا شيء عليه عند الجميع . <br />
ثم قال(   {[5877]} ) :  ( يُرِيدُ اللَّهُ بِكُمُ اليُسْرَ وَلاَ يُرِيدُ بِكُمُ العُسْرَ ) [ 184 ] . <br />
أي يرخص الله عليكم إرادة التيسير ،  ولا يريد بكم العسر(   {[5878]} ) في دينكم . <br />
ثم قال :  ( وَلِتُكْمِلُوا(   {[5879]} ) العِدَّةَ ) [ 184 ] :  أي تكملوا عدة ما أفطرتم/ فتقضوه في أيام أخر(   {[5880]} ) . <br />
ثم قال :  ( وَلِتُكَبِّرُوا اللَّهَ عَلَى مَا هَدَاكُمْ ) [ 184 ] . <br />
قال ابن عباس :   " حق على المسلمين أن يكبروا إذا نظروا إلى هلال شوال " (   {[5881]} ) . <br />
وقيل :  هو التكبير في العيد عند الغدو إلى المصلى .  قاله علي بن(   {[5882]} ) أبي طالب(   {[5883]} ) ،  وزيد بن(   {[5884]} ) أسلم(   {[5885]} ) . <br />
وكذلك كان النبي [ عليه السلام ](   {[5886]} ) يفعل إذا خرج إلى المصلى .  فهو سنة عند الزهري وغيره .  يقول الرجل :  الله أكبر ،  الله أكبر ،  الله أكبر(   {[5887]} ) . <br />
