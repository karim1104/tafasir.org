قوله :  ( اللَّهُ يَسْتَهْزِئُ بِهِمْ )[ 14 ] . <br />
معناه :  الله يجازيهم على قولهم .  والعرب تسمي(   {[934]} ) جزاء الذنب باسمه .  قال الله جل ذكره :  ( وَجَزَاؤُا سَيِّئَةٍ سَيِّئَةُ(   {[935]} ) مِّثْلُهَا )(   {[936]} ) .  وقال :  ( فَمَنِ اعْتَدَى عَلَيْكُمْ فَاعْتَدُوا عَلَيْهِ بِمِثْلِ مَا اعْتَدَى عَلَيْكُمْ )(   {[937]} ) . <br />
فالأول من هذا استهزاء وسيئة ،  وعدوان ،  والثاني :  جزاء عليه ،  فسمي باسمه اتساعاً لأن المعنى قد علم(   {[938]} ) . <br />
وقيل :  معنى(   {[939]} ) ( اللَّهُ يَسْتَهْزِئُ بِهِمْ ) :  أي يقطع عنهم نورهم يوم القيامة إذا أخلوا(   {[940]} ) على الصراط [ ويديم نور(   {[941]} ) ] المؤمنين وهو قوله :  ( فَالْتَمِسُوا نُوراً فَضُرِبَ بَيْنَهُم بِسُورٍ )(   {[942]} ) ،  فهو يعطيهم يوم القيامة نوراً لا يتم لهم ،  ولا ينتفعون به/ لانقطاعه عنهم . <br />
وقال الحسن :   " إن جهنم تجمد كما تجمد الإهالة(   {[943]} ) في القدر ،  فيقال لهم :  هذا طريق ،  فيمضون فيه فيخسف بهم(   {[944]} ) إلى الدرك الأسفل من النار(   {[945]} ) . <br />
وروى أبو صالح عن ابن عباس أنه قال :   " يقال لأهل النار يوم القيامة :  أخرجوا من النار .  وتفتح لهم أبواب النار فإذا رأوا الأبواب قد فتحت أقبلوا إليها ،  يريدون الخروج منها ،  والمؤمنون ينظرون إليهم من الجنة –وهم على الأرائك- فإذا انتهى أهل النار إلى أبوابها يريدون الخروج منها(   {[946]} ) غلقت [ أبوابها دونهم(   {[947]} ) ] ،  فذلك قوله(   {[948]} ) :  ( اللَّهُ يَسْتَهْزِئُ بِهِمْ ) ،  قال :  ويضحك المؤمنون(   {[949]} ) عند ذلك ،  وهو قوله :  ( فَاليَوْمَ الذِينَ ءَامَنُوا مِنَ الكُفَّارِ يَضْحَكُونَ )(   {[950]} )(   {[951]} ) . <br />
وقيل :  معنى :  ( اللَّهُ يَسْتَهْزِئُ بِهِمْ ) أي يظهر لهم من أحكامه في الدنيا في حقن دمائهم وسلامة أموالهم خلاف ما يظهر لهم من عذابه يوم القيامة جزاء على إظهارهم للمؤمنين في الدنيا خلاف ما يبطنون . <br />
وقيل :  معناه :  يُمْلي لهم ،  كما قال :  ( سَنَسْتَدْرِجُهُم مِّنْ حَيْثُ لاَ يَعْلَمُونَ )(   {[952]} ) . <br />
/ قوله :  ( وَيَمُدُّهُمْ فِي طُغْيَانِهِمْ يَعْمَهُونَ )[ 14 ] . <br />
أي يطيل لهم في الأجل المكتوب لهم ،  وهم في طغيانهم يتحيرون(   {[953]} ) .  والطغيان والعتو والعلو بغير الحق ،  والعمه(   {[954]} ) التحير(   {[955]} ) . <br />
/ وقيل :  معنى ( يَعْمَهُونَ ) يركبون رؤوسهم ،  فلا يبصرون رشدهم(   {[956]} ) كما قال :  ( اَفَمَنْ يَّمْشِي مُكِبّاً عَلَى وَجْهِهِ )(   {[957]} ) وهذا كله من صفات المنافقين عند أكثر المفسرين . <br />
وقال الضحاك :   " هو في اليهود "  .
