قوله :  ( لاَ إِكْرَاهَ فِي الدِّينِ قَد تَّبَيَّنَ الرُّشْدُ مِنَ الغَيِّ ) [ 255 ] . <br />
أكثر الناس على أن هذه الآية مخصوصة نزلت في أهل الكتاب ألا يكرهوا على الدين(   {[8352]} ) ،  إذا أدوا الجزية .  فأما أهل الأوثان فلا تؤخذ منهم الجزية ويكرهون على الدين(   {[8353]} ) .  قاله ابن عباس ؛  قال :   " كانت المرأة تجعل على نفسها إن عاش لها ولد أن تُهَوِّدَهُ ،  فلما أجليت بنو النضير ،  كان فيهم من أبناء الأنصار ،  فقالت(   {[8354]} ) الأنصار :  لا ندع أبناءنا ،  فأنزل الله :  ( لاَ إِكْرَاهَ فِي الدِّينِ ) الآية ،  فمن شاء لحق ،  ومن شاء لم يلحق(   {[8355]} ) "  . <br />
وقد قيل :  إن الآية منسوخة منسخها :  ( يَاأَيُّهَا النَّبِيءُ(   {[8356]} ) جَاهِدِ الكُفَّارَ وَالمُنَافِقِينَ )(   {[8357]} )(   {[8358]} ) . <br />
وأكثر الناس على أن(   {[8359]} ) هذه الآية نزلت في غير عبدة الأوثان ،  ومن [ لا كتاب له ](   {[8360]} ) ،  ومن لا يؤدي الجزية من أهل الكتاب . <br />
والألف واللام في  " الدِّين "  عوض من ضمير يعود على الله .  والمعنى  " وهو العلي العظيم لا إكراه في دينه " (   {[8361]} ) . <br />
وقيل :  هما للتعريف(   {[8362]} ) .  والدين :  الإسلام(   {[8363]} ) . <br />
قوله :  ( قَد تَّبَيَّنَ الرُّشْدُ مِنَ الغَيِّ ) [ 255 ] . <br />
قرأ أبو عبد الرحمن  " الرَّشَدَ "  بفتحتين ،  وهما لغتان(   {[8364]} ) ،  كالبُخْل(   {[8365]} ) والبَخَل ،  والشُّغْل والشَّغَل ،  والسُّقْم والسَّقَم ،  والعُدْم والعَدَم/ والعُرْب والعَرَب ،  والعُجْم والعَجَم ،  والسُّخْط والسَّخَط ،  والحُزْن والحَزَن ،  والوُلْد والوَلَد(   {[8366]} ) . <br />
والرشد إصابة الحق ،  والغي ضده .  وهو مصدر غَوِيَ يَغْوِي غَيّاً(   {[8367]} ) ،  وأصله  " غَوْياً " (   {[8368]} ) .  وبعضهم يقول :  غَوَى ،  يَغْوَى بالفتح فيهما(   {[8369]} ) . <br />
وقال بعضهم :  غَوَى يَغْوِي/إذا عدا الحق فضل ،  فمعناه :  استبان الإيمان من الكفر . <br />
قوله :  ( فَمَنْ يَّكْفُرْ بِالطَّاغُوتِ ) [ 255 ] أي بالشيطان ،  والجبت(   {[8370]} ) :  السحر . <br />
وقيل :  الكاهن(   {[8371]} ) . <br />
وقال مجاهد في قوله ( يُرِيدُونَ(   {[8372]} ) أَنْ يَّتَحَاكَمُوا إِلَى الطَّاغُوتِ )(   {[8373]} ) ،  هو كعب بن(   {[8374]} ) الأشرف ،  وهو مشتق من  " طغى "  مقلوب وأصله طَغَوُوتٌ مثل جَبَرُوتٌ(   {[8375]} ) . <br />
وقيل أصله طَيغُوتٌ لأنه يقال :  طَغَوْتُ وَطَغَيْتُ(   {[8376]} ) . <br />
وقيل :  هو في معنى الطغيان(   {[8377]} ) وليس بمشتق منه ،  إنما يؤدي عن معناه ،  كما قيل :  رجل لآلٌ من اللؤلؤ ،  يؤدي عن معناه ،  وليس بمشتق منه(   {[8378]} ) . <br />
قوله :  ( بِالْعُرْوَةِ الوُثْقَى ) [ 255 ] . <br />
أي تمسك بأوثق ما يستوثق به ويتمسك به . <br />
وقيل :  الجبت والطاغوت كل ما يعبد من دون الله(   {[8379]} ) . <br />
قال سيبويه :   " الطاغوت واحد مؤنث يقع على الجميع " (   {[8380]} ) . <br />
وقال المبرد :   " هو جماعة(   {[8381]} ) "  ،  يريد به الشياطين(   {[8382]} ) . <br />
