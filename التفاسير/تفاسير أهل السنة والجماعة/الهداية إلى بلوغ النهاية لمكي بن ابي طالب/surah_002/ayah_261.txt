قوله :  ( مَّثَلُ الذِينَ يُنفِقُونَ أَمْوَالَهُمْ فِي سَبِيلِ اللَّهِ ) [ 260 ] . <br />
قال الطبري :   " هذه الآية مردودة إلى قوله :  ( مَّن ذَا الذِي يُقْرِضُ اللَّهَ قَرْضاً حَسَناً فَيُضَاعِفهُ لَهُ أَضْعَافاً كَثِيرَةً ) وإلى الآيات التي بعدها(   {[8666]} ) "  . <br />
قال(   {[8667]} ) السدي :   " نزلت في الذي ينفق(   {[8668]} ) على نفسه في سبيل الله عز وجل ويخرج " (   {[8669]} ) .  والمثل في هذه الآية إنما هو للنفقة(   {[8670]} ) لا للمنفق(   {[8671]} ) ،  وفي الكلام حذف ،  والتقدير :   " مثل نفقة الذين ينفقون "  ،  ودل  " ينفقون "  على النفقة فحسن حذفها(   {[8672]} ) . <br />
وروى نافع عن ابن عمر أنه قال :   " لما نزلت :  ( مَّثَلُ الذِينَ يُنفِقُونَ أَمْوَالَهُمْ فِي سَبِيلِ اللَّهِ كَمَثَلِ حَبَّةٍ ) الآية ،  قال النبي عليه السلام :   " اللَّهُمَّ زِدْ أُمَّتِي "  ،  فنزلت : <br />
( مَّنْ ذَا الذِي يُقْرِضُ اللَّهَ قَرْضاً حَسَناً فَيُضَاعِفُهُ لَهُ أَضْعَافاً كَثِيرَةً ) .  فقال رسول الله صلى الله عليه وسلم :   " رَبِّ زِدْ أُمَّتِي :  فنزلت :  ( اِنَّمَا يُوَفَّى الصَّابِرُونَ أَجْرَهُم بِغَيْرِ حِسَابٍ ) " (   {[8673]} ) . <br />
قال مالك في قوله تعالى :  ( اِنَّمَا يُوَفَّى الصَّابِرُونَ أَجْرَهُم بِغَيْرِ حِسَابٍ )(   {[8674]} )  " هو الصبر على فجائع الدنيا وأحزانها " (   {[8675]} ) . <br />
قال مالك :   " وبلغني أن الصبر من الإيمان كالرأس من الجسد "  . <br />
قلت :  والصبر على طاعة الله عز وجل وعن محارم الله تعالى أفضل من(   {[8676]} ) الصبر على المصائب والفجائع .  كذا ،  قال عمر وغيره . <br />
ثم قال تعالى :  ( وَاللَّهُ يُضَاعِفُ لِمَنْ يَّشَاءُ ) [ 260 ] . <br />
أي ممن أنفق في غير سبيل الله ،  يضاعف أيضاً إن شاء(   {[8677]} ) . <br />
وقيل :  ( لِمَن يَّشَاءُ ) :  هو المنفق في سبيل الله عز وجل ،  يزيد على سبعمائة ضعف إلى ألفي(   {[8678]} ) ألف ضعف إن شاء .  روي ذلك/عن ابن عباس(   {[8679]} ) . <br />
قوله :  ( وَاللَّهُ وَاسِعٌ )(   {[8680]} ) [ 260 ] . <br />
أي يزيد(   {[8681]} ) من يشاء من خلقه على السبعمائة ما شاء(   {[8682]} ) . <br />
( عَلِيمٌ ) .  أي عليم بما ينفق المنفقون(   {[8683]} ) في سبيله . <br />
وقيل(   {[8684]} ) :  عليم بمن يزيده على السبعمائة ضعف ومن لا يزيده(   {[8685]} ) .
