قوله :  ( وَيَسْئَلُونَكَ(   {[7032]} ) عَنِ اليَتَامَى(   {[7033]} ) قُلِ اِصْلاَحٌ لَّهُمْ خَيْرٌ ) [ 218 ] . <br />
كان(   {[7034]} ) سبب نزول هذه الآية أنه لما نزل :  ( اِنَّ الذِينَ يَاكُلُونَ أَمْوَالَ(   {[7035]} ) اليَتَامَى ظُلْماً )(   {[7036]} ) ،  انطلق من كان معه يتيم فعزل طعامه من طعامه ،  وشرابه من شرابه ،  فكان يفضل الشيء من طعام اليتيم فيحبس له حتى يفسد فاشتد عليهم ذلك ،  فذكروا ذلك للنبي [ عليه السلام ](   {[7037]} ) ،  فأنزل الله تعالى :  ( وَيَسْئَلُونَكَ عَنِ اليَتَامَى قُلِ اِصْلاَحٌ لَّهُمْ خَيْرٌ وَإِن تُخَالِطُوهُمْ فَإِخْوَانُكُمْ ) .  قاله ابن عباس وغيره(   {[7038]} ) . <br />
وقيل :  إنه لما نزل :  ( وَلاَ تَقْرَبُوا مَالَ اليَتِيمِ إِلاَّ بِالتِي هِيَ أَحْسَنُ )(   {[7039]} ) ،  اجتنب الناس مخالطتهم فنزلت :  ( وَإِن تُخَالِطُوهُمْ فَإِخْوَانُكُمْ ) ،  فخالطوهم(   {[7040]} ) . <br />
( وَاللَّهُ يَعْلَمُ المُفْسِدَ مِنَ المُصْلِحِ ) [ 218 ] .  أي يعلم حين تخلط مالك بماله(   {[7041]} ) ،  أتريد بذلك إصلاح ماله أو إفساده يريد الجنسين(   {[7042]} ) . <br />
قوله :  ( وَلَوْ شَاءَ اللَّهُ لاَعْنَتَكُمُ ) [ 218 ] .  أي :  لحرم(   {[7043]} ) عليكم مخالطتهم فتتعبون ،  ويشق(   {[7044]} ) ذلك عليكم(   {[7045]} ) . <br />
وقيل :  معناه :  لأَوْبقَكُمْ فأهلككم(   {[7046]} ) بما قد أصبتم من أموالهم(   {[7047]} ) . <br />
قال أبو إسحاق :   " معناه :  لكلفكم(   {[7048]} ) ما يشتد عليكم ،  فتعنتون " (   {[7049]} ) . <br />
وأصله من :   " عَنِتَ البَعِيرُ "  إذا حدث في رجله كسر بعد جبر(   {[7050]} ) . <br />
قوله(   {[7051]} ) :  ( إِنَّ اللَّهَ عَزِيزٌ حَكِيمٌ ) [ 208 ] . <br />
أي :  عزيز في سلطانه ،  حكيم في فعله وأحكامه وتدبيره(   {[7052]} ) .
