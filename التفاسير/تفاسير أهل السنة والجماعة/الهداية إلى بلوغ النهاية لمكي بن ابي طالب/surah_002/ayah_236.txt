قوله :  ( لاَّ جُنَاحَ عَلَيْكُمُ إِن طَلّقْتُمُ النِّسَاءَ مَا لَمْ تَمَسُّوهُنَّ ) الآية [ 234 ] . <br />
رفع الله الحرج عمن طلق المفروض(   {[7733]} ) لها الصداق قبل(   {[7734]} ) أن يدخل بها(   {[7735]} ) . <br />
ومعنى ( تَمَسُّوهُنَّ ) :  تجامعوهن .  قاله ابن عباس وغيره(   {[7736]} ) . <br />
ومعنى :  ( أَوْ تَفْرِضُوا لَهُنَّ ) [ 234 ] :  رفع الجناح أيضاً عمن طلق التي لم يفرض لها قبل الفرض ،  فمعناه :  أو توجبوا لهن فريضة . <br />
قوله :  ( وَمَتِّعُوهُنَّ ) [ 234 ] . <br />
قال ابن عباس وغيره :   " هو واجب للتي لم يفرض لها(   {[7737]} ) ؛  يمتع الموسر(   {[7738]} ) بخادم ،  والوسط بالورق(   {[7739]} ) ،  ودون ذلك بالكسوة والنفقة " (   {[7740]} ) .  وقاله قتادة(   {[7741]} ) . <br />
وقال أبو حنيفة :   " يمتع التي لم يفرض لها ،  إذا طلق قبل الدخول بنصف مقدار صداقها " (   {[7742]} ) .  وبه قال الشافعي(   {[7743]} ) ،  ولم يجد/نصفاً من غيره . <br />
وأوجب علي بن أبي طالب رضي الله عنه المتعة لكل مطلقة(   {[7744]} ) . <br />
وبه قال الحسين فرض لها أو لم يفرض دخل بها أو لم يدخل(   {[7745]} ) . <br />
وروى مالك(   {[7746]} ) عن نافع عن ابن عمر أنه قال :   " لكل مطلقة متعة إلا التي سمى لها صداقاً ولم تمس(   {[7747]} ) ،  فحسبها نصف ما فرض لها " (   {[7748]} ) . <br />
ومذهب مالك أنه لا يجبر على المتعة أحد(   {[7749]} ) من المطلقين إنما هو ندب(   {[7750]} ) ،  وبه قال شريح(   {[7751]} ) . <br />
وقال ابن المسيب : / " كانت المتعة واجبة بالآية التي في الأحزاب لقوله :  ( فَمَتِّعُوهُنَّ وَسَرِّحُوهُنَّ/سَرَاحاً جَمِيلاً )(   {[7752]} ) ،  قال :   " ثم نسختها التي في البقرة لقوله :  ( حَقّاً عَلَى المُحْسِنِينَ ) ولم يقل :   " حقاً عليكم "  ولا  " واجباً(   {[7753]} ) عليكم " (   {[7754]} ) . <br />
وقد أجمعوا/أن المطلقة قبل الدخول لا تضرب بالمتعة مع الغرباء كان قد فرض لها أم لم يفرض ،  فدل على أنها غير واجبة ،  فصارت المتعة في البقرة ندباً لمن أحسن واتقى لا(   {[7755]} ) فرضاً(   {[7756]} ) . <br />
وروي عن ابن عباس أنه قال :  ( عَلَى المُوسِعِ قَدْرُهُ ) أي على قدر يسره ،  ( وَعَلَى المُقْتِرِ قَدْرُهُ ) :  أي على قدر عسره للتي لم يسم لها صداقاً ،  ولم يدخل بها خاصة " (   {[7757]} ) . <br />
ومعنى :  ( لاَّ جُنَاحَ عَلَيْكُمُ ) ،  أي الجناح مرفوع/عنكم في الطلاق قبل المسيس لأنه يجوز أن يقع بعد المسيس الجناح على المطلق ،  وذلك الذي يتزوج للذوق .  قال النبي صلى الله عليه وسلم :   " إِنَّ اللَّهَ لاَ يُحِبُّ الذَّوَّاقِينَ وَلاَ الذَّوَّاقَاتِ " (   {[7758]} ) .  فَرَفْعُ الجناح في الطلاق قبل المس يدل على أنه يقع في الطلاق بعد المس وهو ما ذكرنا(   {[7759]} ) . <br />
وقيل :  إنما رفع الجناح عن طلاق التي لم يدخل بها ،  لأن الرجل يطلق متى شاء [ حائضاً كانت أو طاهراً ](   {[7760]} ) ،  ولا عدة ولا سنة في طلاقهن ،  وليس ذلك في المدخول بها ،  لأنه إن طلق وهي حائض وجب عليه مراجعتها ،  ولحقه ضيق وإثم إن تعمد(   {[7761]} ) مخالفة السنة ،  ولزمت العدة .
