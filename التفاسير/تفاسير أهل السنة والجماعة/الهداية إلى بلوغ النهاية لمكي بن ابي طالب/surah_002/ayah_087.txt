قوله :  ( وَلَقَدَ آَتَيْنَا مُوسَى الكِتَابَ ) [ 86 ] . <br />
لام ( لَقَدْ ) حيث وقعت لام تأكيد ،  وقد تقع جواباً للقسم ،  وعيسى(   {[2873]} ) لا ينصرف لأنه أعجمي معرفة(   {[2874]} ) . <br />
وقيل :  هو عربي من(   {[2875]} )  " عَاسَهُ يَعُوسُهُ "  إذا ساسه وقام عليه ولا ينصرف على هذا للتعريف والتأنيث . <br />
( القُدُسُ ) أصله الطهر .  وفيه لغة نادرة(   {[2876]} ) وهي فتح القاف والدال(   {[2877]} ) .  والكتاب هو التوراة . <br />
( وَقَفَّيْنَا ) :  أردفنا وأتبعنا بعضهم بعضاً على منهاج(   {[2878]} ) واحد ،  وشريعة واحدة ،  لأن كل من بعث بعد موسى صلى الله عليه وسلم إلى زمان(   {[2879]} ) عيسى صلى الله عليه وسلم فإنما يأمر بني إسرائيل بلزوم التوراة والعمل بما فيها ،  فلذلك قال :  ( مِنْ بَعْدِهِ بِالرُّسُلِ ) .  أي :  على منهاجه وطريقته . <br />
ثم قال :  ( وَءَاتَيْنَا عِيسَى ابنَ مَرْيَمَ البَيِّنَاتِ ) [ 86 ] . <br />
ابتداء كلام آخر . <br />
وأصل  " قَفَوْت "  من القَفَا "  يقال :   " قَفَوْتُ فُلاناً(   {[2880]} ) "  إذا صرت خلف قفاه(   {[2881]} ) .   " وَدَبَرْتُهُ "  إذا صرت خلف دبره(   {[2882]} ) . <br />
والبينات التي أوتيها(   {[2883]} ) عيسى عليه السلام هي إحياء الموتى وإبراء الأكمه وخلق الطير ،  وغيره من الحجج والبينات(   {[2884]} ) . <br />
( وَأَيَّدْنَاهُ ) :  قويناه وأعناه ونصرناه [ والأيد والأد ](   {[2885]} ) القوة(   {[2886]} ) . <br />
وقرأ ابن(   {[2887]} ) محيصن :   " وأيدناه "  بالمد(   {[2888]} ) . <br />
و " روح القدس "  :  جبريل(   {[2889]} ) .  قاله قتادة والسدي والضحاك والربيع بن أنس(   {[2890]} ) .  وروي ذلك عن النبي [ عليه السلام ](   {[2891]} ) .  وقاله ابن عباس(   {[2892]} ) . <br />
وقال ابن زيد :   " هو الإنجيل سمي روحاً كما سمي القرآن روحاً ،  فقال تعالى :  ( أَوْحَيْنَا إِلَيْكَ رُوحاً مِنَ اَمْرِنَا )(   {[2893]} )(   {[2894]} ) . <br />
ويرد هذا القولَ قولُه تعالى :  ( إِذَاَيَّدتُّكَ بِرُوحِ القُدُسِ )(   {[2895]} ) ،  ثم قال تعالى :  ( وَإِذْ عَلَّمْتُكَ الكِتَابَ وَالحِكْمَةَ وَالتَّوْرَاةَ وَالاِنْجِيلَ )(   {[2896]} ) .  فدل هذا [ على ](   {[2897]} ) أن روح القدس غير الإنجيل/ .  فإن حُمل على أنه أعيد للتأكيد كما قال :  ( فِيهِمَا فَاكِهَةٌ وَنَخْلٌ(   {[2898]} ) وَرُمَّانٌ )(   {[2899]} ) ،  وقال :  ( مَنْ كَانَ عَدُوّاً لِلَّهِ وَمَلاَئِكَتِهِ وَرُسُلِهِ وَجِبْرِيلَ وَمِيكَائِيلَ )(   {[2900]} ) ،  فهو وجه . <br />
وروى الضحاك عن ابن عباس أنه قال :   " روح القدس هو الاسم الذي كان يحيي به عيسى صلى الله عليه وسلم الموتى " (   {[2901]} ) . <br />
وقال مجاهد :   " القدس :  الله جل ذكره ،  وسمي جبريل روحاً لأنه كان بتكوين الله له(   {[2902]} ) من غير ولادة كما سمي عيسى صلى الله عليه وسلم/ روحاً ،  فقال :  ( وَرُوحٌ(   {[2903]} ) مِّنْهُ(   {[2904]} ) )(   {[2905]} )(   {[2906]} ) . <br />
وقال السدي :   " القدس هنا البركة " (   {[2907]} ) . <br />
وقال الربيع :   " القدس هو الله(   {[2908]} ) ،  ويدل عليه قوله :  ( المَلِكُ القُدُّوسُ )(   {[2909]} )(   {[2910]} ) .  والقدوس والقدس واحد " (   {[2911]} ) .  ورواه ابن وهب عن مجاهد أيضاً(   {[2912]} ) . <br />
