ثم قال :  ( أُوْلاَئِكَ عَلَيْهِمْ صَلَوَاتٌ مِّن رَّبِّهِمْ وَرَحْمَةٌ ) [ 156 ] . <br />
أي :  مغفرة من ربهم . <br />
وقيل :  ترحم من ربهم ورحمة(   {[4894]} ) .  ( وَأُوْلاَئِكَ هُمُ المُهْتَدُونَ ) . <br />
قال الليث(   {[4895]} ) في قوله :  ( إِنَّا لِلهِ ) الآية  " معناها :  نحن والذي أصبنا به لله ونحن وإياه إلى الله راجعون "  . <br />
/قال ابن جبير :   " لم يعط هذه الآية أحد من الأمم قبلنا(   {[4896]} ) ولا نبي قبل نبينا .  ولو علمها يعقوب عليه السلام لم يقل :  ( يَا أَسَفَى عَلَى يُوسُفَ )(   {[4897]} ) " (   {[4898]} ) . <br />
وقال عكرمة :   " انطفأ مصباح/النبي [ عليه السلام ](   {[4899]} ) ليلة فقال :  ( إِنَّا لِلهِ وإِنَّا إِلَيْهِ رَاجِعُونَ ) .  فَقِيلَ(   {[4900]} ) :  يَا نَبِيَّ الله :  أَمُصِيبَةٌ هَذِهِ ؟  فَقَالَ(   {[4901]} ) :  نَعَمْ ،  كُلُّ شَيْءٍ آذَى المُؤْمِنَ فَهُوَ لَهُ مُصِيبَةٌ ، /وَلَهُ فِيهِ أَجْرُ المُصِيبَةِ " (   {[4902]} ) . <br />
وروي أن النبي [ صلى الله عليه وسلم ](   {[4903]} ) قال(   {[4904]} ) :   " مَا مِنْ أَحَدٍ أُصِيبَ بِمُصِيبَةٍ فَاسْتَرْجَعَ إِلاَّ اسْتَوْجَبَ مِنَ الله ثَلاَثَ(   {[4905]} ) خِصَالٍ ؛  كُلُّ خَصْلَةٍ خَيْرٌ مِنَ الدُّنْيَا وَمَا فِيهَا " (   {[4906]} ) . <br />
قال أبو عبيد(   {[4907]} ) :  يعني قوله :  ( أُوْلاَئِكَ عَلَيْهِمْ صَلَوَاتٌ مِن رَّبِّهِمْ وَرَحْمَةٌ وَأُوْلاَئِكَ هُمُ المُهْتَدُونَ ) .
