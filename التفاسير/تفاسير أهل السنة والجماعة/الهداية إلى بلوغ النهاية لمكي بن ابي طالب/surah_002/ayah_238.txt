قوله :  ( حَافِظُوا عَلَى الصَّلَوَاتِ وَالصَّلاَةِ الوُسْطَى ) [ 236 ] . <br />
قال ابن عباس وأبو هريرة وابن عمر وأبو سعيد الخدري وعائشة وسعيد بن جبير والضحاك ومجاهد وغيرهم :   " الوسطى صلاة(   {[7815]} ) العصر " (   {[7816]} ) .  وروي ذلك عن النبي [ صلى الله عليه وسلم ](   {[7817]} ) . <br />
وقال زيد بن ثابت وابن أبي ذئب :   " هي الظهر " (   {[7818]} ) .  وروي ذلك عن ابن عمر(   {[7819]} ) . <br />
وروي(   {[7820]} ) أن النبي عليه السلام كان يصلي في الهاجرة والناس في هاجرتهم ،  فلا يجتمع إليه أحد ،  فتكلم في ذلك فأنزل الله :  ( وَالصَّلاَةُ الوُسْطَى ) ،  يريد الظهر(   {[7821]} ) . <br />
وقال قبيصة(   {[7822]} ) بن(   {[7823]} ) ذئيب :   " هي المغرب لكونها بين الليل والنهار " (   {[7824]} ) . <br />
وقال جابر بن عبد الله وعطاء وعكرمة :   " هي الصبح/لكونها أيضاً بين الليل والنهار " (   {[7825]} ) ،  ولقوله :  ( وَقُومُوا لِلَّهِ قَانِتِينَ )[ 236 ] .  والقنوت إنما يكون في الصبح(   {[7826]} ) .  وهو [ أيضاً مروي ](   {[7827]} ) عن ابن عباس [ وعلي بن أبي طالب ](   {[7828]} )/وهو قول الربيع(   {[7829]} ) وعبد الله بن شداد بن الهاد ،  وروي ذلك عن مجاهد(   {[7830]} ) ،  وهو قول مالك(   {[7831]} ) . <br />
وهو قول أبي أمامة الباهلي(   {[7832]} ) وزيد بن أسلم وعبد الله بن عمر(   {[7833]} ) . <br />
وقد تظاهرت الأخبار عن النبي [ عليه السلام ](   {[7834]} ) أنها العصر(   {[7835]} ) . <br />
وروى ابن(   {[7836]} ) وهب وابن القاسم عن مالك :  أن الصلاة الوسطى صلاة الصبح(   {[7837]} ) . <br />
قال مالك :   " الظهر والعصر في النهار ،  والمغرب والعشاء في الليل ،  والصبح فيما بين ذلك " (   {[7838]} ) . <br />
قال مالك :   " والصبح لا تجمع إلى غيرها ،  وقد يجمع بين الظهر والعصر وبين المغرب والعشاء " (   {[7839]} ) . <br />
قال مالك :   " وهي كثيراً ما تفوت الناس ويتلهون عنها(   {[7840]} ) " (   {[7841]} ) . <br />
قلت(   {[7842]} ) :  وصلاة الصبح أفضل الصلوات ،  ولذلك أكد الله في المحافظة عليها ،  يدل على ذلك قول النبي عليه السلام :   " مَنْ شَهِدَ الصُّبْحَ فَكَأَنَّمَا قَامَ لَيْلَةً ،  وَمَنْ شَهِدَ العِشَاءَ فَكَأَنَّمَا قَامَ نِصْفَ لَيْلَةٍ " (   {[7843]} ) . <br />
وقال(   {[7844]} ) :   " بَيْنَنَا وَبَيْنَ المُنَافِقِينَ شُهُودُ العَتْمَةِ وَالصُّبْحِ ،  لاَ يَسْتَطِيعُونَهُمَا " (   {[7845]} ) . <br />
وقال :   " لَوْ يَعْلَمُونَ مَا فِي العَتْمَةِ وَالصُّبْحِ لأَتَوْهُمَا وَلَوْ حَبْواً " (   {[7846]} ) . <br />
ففضل العتمة والصبح على سائر الصلوات ،  ثم فضل(   {[7847]} ) الصبح على العتمة فدل على أنها أفضل الصلوات ،  فهي الوسطى . <br />
وقد قال عمر :   " لأن أشهد صلاة أحب إلي من أن أقوم ليلة " (   {[7848]} ) . <br />
وقد قرأ(   {[7849]} ) الرؤاسي(   {[7850]} ) :  ( وَالصَّلاَةِ الوُسْطَى ) بالنصب ،  بمعنى :  وألزموا الصلاة ،  وقد روي عن عائشة(   {[7851]} ) رضي الله عنها أنها أثبتت في المصحف :   " والصلاة الوسطى(   {[7852]} ) ،  وصلاة(   {[7853]} ) العصر "  ،  بالواو(   {[7854]} ) . <br />
وكذلك روى نافع أن حفصة أمرت أن يكتب ذلك في مصحفها ،  وقالت سمعت رسول الله صلى الله عليه وسلم يقرأها كذلك " (   {[7855]} ) . <br />
وليست هذه الزيادة توجب أن تكون الوسطى/غير العصر ،  لأن سيبويه قد حكى :   " مررت بأخيك وصاحبك "  والصاحب هو الأخ ،  فكذلك الوسطى هي العصر ،  وإن عطفت(   {[7856]} ) بالواو(   {[7857]} ) . <br />
الصلاة الوسطى هي أفضل الصلوات إعادتها(   {[7858]} ) بلفظها بعد أن دخلت في جملة الصلوات كما قال :  ( وَمَلاَئِكَتِهِ )(   {[7859]} ) .  ثم قال :  ( وَجِبْرِيلَ وَمِيكَائِيلَ )(   {[7860]} ) ،  وكما قال :  ( فِيهِمَا فَاكِهَةٌ وَنَخْلٌ/وَرُمَّانٌ )(   {[7861]} ) .  فأعاد  " جبريل وميكائيل " (   {[7862]} ) وقد دخلا في جملة الملائكة ،  وأعاد  " النخل والرمان "  ،  وقد دخلا في جملة الفاكهة ،  وذلك لفضل فيهما(   {[7863]} ) ،  فأعيدا للتنبيه على الفضل ،  فكذلك الصلاة الوسطى أعيدت لأنها أفضل الصلوات . <br />
وقد بينا أن صلاة(   {[7864]} ) الصبح أفضل الصلوات فهي هي بغير شك . <br />
وقرأ ابن عباس :   " والصلاة الوسطى صلاة العصر "  على التقدير(   {[7865]} ) . <br />
وذكر ابن(   {[7866]} ) حبيب(   {[7867]} ) عن بعضهم أنها صلاة الجمعة ،  وهو قول شاذ(   {[7868]} ) . <br />
وأصل القنوت/الطاعة ،  وهو(   {[7869]} ) أيضاً طول القيام(   {[7870]} ) . <br />
وقيل :  هو هنا السكوت لأنهم كانوا يتكلمون في الصلاة(   {[7871]} ) . <br />
قال زيد بن أرقم :   " كنا نتكلم في الصلاة حتى نزلت :  ( وَقُومُوا لِلهِ قَانِتِينَ ) ،  فأمرنا بالسكوت(   {[7872]} ) . <br />
وقيل :  القنوت هنا الدعاء(   {[7873]} ) . <br />
وقيل :  هو الركوع/والخشوع في الصلاة(   {[7874]} ) . <br />
وقال مجاهد :   " هو غض البصر/في الصلاة وضم الجناح وطول الركوع(   {[7875]} ) وأصله كله يرجع إلى الطاعة " (   {[7876]} ) .
