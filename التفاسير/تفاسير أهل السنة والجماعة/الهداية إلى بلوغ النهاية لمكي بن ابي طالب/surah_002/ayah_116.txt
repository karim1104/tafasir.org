ثم قال :  ( وَقَالُوا اتَّخَذَ اللَّهُ وَلَداً ) [ 115 ] . <br />
أي : (   {[3714]} ) وقال الذين منعوا الذكر في مساجد الله وسعوا في خرابها :  اتخذ الله ولداً . <br />
( سُبْحَانَهُ ) :  أي :  براءة له من ذلك وتنزيهاً له . <br />
قال أبو إسحاق :   " يريد به النصارى واليهود والمشركين من العرب ،  لأنهم قالوا :  الملائكة بنات الله ،  وقالت اليهود :  عزير ابن(   {[3715]} ) الله ،  وقالت النصارى :  المسيح ابن(   {[3716]} ) الله " (   {[3717]} ) . <br />
وروي عن(   {[3718]} ) ابن(   {[3719]} ) عباس أن النبي صلى الله عليه وسلم قال(   {[3720]} ) :  اللهُ تَعَالَى :  كَذَّبَنِي ابْنُ(   {[3721]} ) آدَمَ وَلَمْ يَكُنْ لَهُ ذَلِكَ .  وَشَتَمَنِي(   {[3722]} ) وَلَمْ يَكُنْ لَهُ ذَلِكَ .  أَمّا تَكْذِيبُهُ إِيَّايَ فَيَزْعُمُ أَنِّي لاَ أَقْدِرُ أَنْ أُعِيدَهُ(   {[3723]} ) كَمَا كَانَ .  وَأَمَّا شَتْمُهُ إِيَّايَ فَقَوْلُهُ لِي وَلَدٌ ،  [ فَسُبْحَانِي أَنْ أَتَّخِذَ صَاحِبَةً(   {[3724]} ) أَوْ ] وَلَداً " (   {[3725]} ) . <br />
قوله :  ( كُلٌّ لَهُ قَانِتُونَ ) [ 115 ] .  أي :  مطيعون . <br />
وقيل(   {[3726]} ) :  مطيعون(   {[3727]} ) يوم القيامة(   {[3728]} ) . <br />
وقيل :  مقرون بالعبودية(   {[3729]} ) . <br />
وقال الفراء :   " هو خصوص يراد به أهل الطاعة " (   {[3730]} ) . <br />
وأصل القنوت في اللغة الطاعة ،  والقنوت القيام الطويل(   {[3731]} ) . <br />
وقال الحسن :   " يعني اليهود والنصارى ومشركي العرب ؛  كل [ له قائم ](   {[3732]} ) بالشهادة بأنه(   {[3733]} ) عبد له " (   {[3734]} ) . <br />
قال يحيى :   " إنما خص الحسنُ اليهود والنصارى ومشركي العرب ،  لأنهم هم الذين كانوا بحضرة النبي [ عليه السلام ](   {[3735]} ) يومئذ "  . <br />
وقال في آية أخرى : (   {[3736]} ) ( وَلَئن سَأَلْتَهُم مَّنْ خَلَقَهُمْ لَيَقُولنَّ اللَّهُ(   {[3737]} ) )(   {[3738]} ) .  من تفسير ابن(   {[3739]} ) سلام .
