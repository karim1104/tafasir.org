 " أن "  :  في موضع نصب ،  على تقدير :   " وأجدر بأن لا " {[29516]} . <br />
تقول :   " هو جدير{[29517]} بأن يفعل "  ،  و " خليق بأن يفعل "  ،  وإن شئت حذفت  " الباء "  ،  ولا يحسن حذف  " الباء "  إلا مع  " أن "  ،  لو قلت :  هو جدير بالفعل ،  لم يكن بُد{[29518]} من  " الباء " {[29519]} . <br />
والمعنى :  الأعراب أشد جحودا لتوحيد الله سبحانه ،  ونفاقا على رسوله عليه السلام ،  من أهل الحضر والأمصار ،  وذلك لجفائهم{[29520]} ،  وقسوة قلوبهم{[29521]} . <br />
{  وأجدر ألا يعلموا }[ 97 ] . <br />
