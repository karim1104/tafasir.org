ثم قال تعالى :  {  وإخوانهم يمدونهم في الغي }[ 202 ] . <br />
أي :  وإخوان الشياطين{[26578]} تمدهم{[26579]} الشياطين في الغي{[26580]} . <br />
{  ثم لا يقصرون }[ 202 ] . <br />
أي :  لا يقصرون عما أقصر عنه الذين اتقوا {  إذا مسهم طائف من الشيطان }{[26581]} . <br />
وهذا خبر من الله ( عز وجل ){[26582]} عن حال المؤمنين وحال الكفار ،  أن المؤمن إذا أصاب الذنب تذكر العقوبة فتاب ورجع وأبصر{[26583]} رشده ،  والكافر يمد له{[26584]} إخوانه من الشياطين في الغي ،  ثم لا يقصر عن غيه{[26585]} ،  ولا يرجع كما فعل المؤمن{[26586]} . <br />
و " المد "  :  الزيادة{[26587]} ،  ف " الهاء "  و " الميم "  في :  {  وإخوانهم } تعود على الشياطين .  ودل  " الشيطان "  في قوله {  طائف من الشيطان } على الشياطين .  و " الإخوان "  كناية عن الكفار{[26588]} . <br />
والضمير المرفوع في :   " يُمِدُّون "  يعود على  " الشياطين "  . <br />
و " الهاء "  و " الميم "  في {  يمدونهم } تعود على  " الكفار "  ،  وهو الإخوان{[26589]} . <br />
وقيل المعنى :  ثم لا يقصر الشياطين في مدهم في الغي{[26590]} للكفار .  قاله :  قتادة . <br />
وقال{[26591]} :  {  لا يقصرون } عنهم ولا يرحمونهم{[26592]}/ فالضمير في {  يقصرون } للشياطين .  وعلى القول الأول للمشركين ،  وهو الأكثر . <br />
و " الغي "  :  الجهل والوقوع في الهلكة{[26593]} . <br />
وهذا الكلام عند أبي إسحاق مُقدم متصل في النية بقوله :  {  ولا أنفسهم ينصرون{[26594]} }[ 197 ] . <br />
ثم قال :  {  وإخوانهم } ،  أي :  وإخوانهم{[26595]} يعني :  الشياطين ،  {  يمدونهم } ،  يعني :  الكفار{[26596]} . <br />
يقال :   " قصَّر "  عن الشيء و " أقصر " {[26597]} . <br />
وقد أنكر أبو حاتم وأبو عبيد{[26598]} قراءة نافع ،  ب " ضم الياء "  في :  {  يمدونهم }{[26599]} ،  وهي مشهورة{[26600]} . <br />
حكى المبرد{[26601]} :   " مددت له في كذا "  :   " زينته له ،  واستدعيته{[26602]} أن يفعله ،  و " أمددته في كذا "  أي :  أعنته{[26603]} برأيي وغير ذلك{[26604]} . <br />
وحكى غير المبرد :   " مدَّه{[26605]} "  و " أمده "  بمعنى{[26606]} . <br />
وقيل {  في الغي } :  متعلق ب " الإخوان " {[26607]} ،  التقدير :  وإخوانهم في الغي يمدونهم ،  والأول{[26608]} أحسن ،  كما قال :  {  ويمدهم في طغيانهم{[26609]} }{[26610]} .
