قوله :  {  وبينهما حجاب [ وعلى الاعراف رجال يعرفون كلا بسيماهم{[23738]} ] }[ 46 ] ( الآية ){[23739]} . <br />
والمعنى :  وبين الجنة والنار {  حجاب } ،  أي :  حاجز ،  وهو السور الذي ذكره الله[ عز وجل ]{[23740]} فقال :  {  فضرب بينهم بسور{[23741]} } ،  وهو :  الأعراف{[23742]} . <br />
قال مجاهد :   " الأعراف "  ،  حجاب بين الجنة والنار{[23743]} . <br />
قال السدي :  الحجاب ،  وهو السور ،  وهو :  الأعراف{[23744]} . <br />
و " الأعراف "  :  جمع ،  واحدها :  عُرْف ،  وكل مرتفع من الأرض فهو :  عُرْفٌ{[23745]} . <br />
وقيل لعرف الديك :  عرف لارتفاعه{[23746]} . <br />
وقال السدي :  إنما سمي  " الأعراف "  أعرافا{[23747]} ؛  لأن أصحابه يعرفون الناس{[23748]} . <br />
وقال ابن عباس :  هو جسر بين الجنة والنار ،  عليه ناس من أهل الذنوب بين الجنة والنار{[23749]} . <br />
وذكر الشعبي{[23750]}عن حذيفة{[23751]} :  أن  " أصحاب الأعراف "  قوم تجاوزت بهم حسناتهم النار ،  [ و{[23752]} ] قصرت سيئاتهم عن الجنة ،  {  وإذا صرفت أبصارهم تلقاء أصحاب النار/قالوا ربنا لا تجعلنا مع القوم الظالمين } ،  فبيناهم كذلك إذا طلع عليهم ربك فقال ( لهم{[23753]} ) :  اذهبوا فادخلوا الجنة فإني قد غفرت لكم{[23754]} . <br />
وروي عنه أنه قال :  يوقفون هنالك على السور حتى يقضي الله بينهم{[23755]} . <br />
وعن ابن عباس :   " الأعراف "  ،  السور الذي بين الجنة والنار{[23756]} ،  و " الرجال "  :  قوم استوت حسناتهم وسيئاتهم{[23757]} ،  فوقفوا على{  الأعراف } ،  يعرفون أهل الجنة بسيماهم ،  وأهل النار بسيماهم{[23758]} . <br />
وعنه{[23759]}  أنه قال أيضا :   " الأعراف "  الشيء المشرف{[23760]} . <br />
وعنه أنه قال :  [ هو{[23761]} ] كعرف الديك{[23762]} . <br />
وعنه أنه قال :  هم رجال كانت لهم ذنوب عظام وكان حسم{[23763]} أمرهم لله ،  فوقفوا على السور{[23764]} . <br />
وقال ابن مسعود :  من كانت حسناته أكثر من سيئاته بواحدة دخل الجنة ،  ومن كانت سيئاته أكثر من حسناته بواحدة دخل النار .  ومن استوت حسناته وسيئاته كان من  " أصحاب الأعراف "  فوقفوا على الصراط{[23765]} ،  ثم عرفوا أهل الجنة وأهل النار ،  فإذا نظروا إلى أهل الجنة نادوا :  {  سلام عليكم }[ 46 ] ،
