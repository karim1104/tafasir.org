ثم قال تعالى :  {  إن الذين اتقوا إذا مسهم طائف }[ 201 ] . <br />
و " الطائف "  و " الطيف "  عند جماعة من البصريين ،  سواء وهو ما كان كالخيال ،  والشيء يُلِمُّ بك{[26562]} . <br />
وقيل :  إن  " طيفا "  مخفف من :   " طيف "  ،  مثل :  مَيْتٌ ومَيِّتٌ{[26563]} . <br />
وقال بعض الكوفيين :   " الطائف "  :  ما طاف به من وسوسة الشيطان .  و " الطيف "  :  من اللم والمس{[26564]} . <br />
وقال الكسائي :   " الطيف "  :  اللهو ،  و " الطائف "  :  كل ما طاف حول الإنسان{[26565]} . <br />
وقرأ ابن جبير :   " طَيِّفٌ "  ،  مشدودا{[26566]} . <br />
و " الطيف "  عند أهل العربية :  مصدر طاف{[26567]} . <br />
وقال الكسائي :  هو مخفف من :   " طيف " {[26568]} . <br />
وقال الكسائي :   " طاف "  من :  الواو{[26569]} . <br />
وقال الأحمر{[26570]} سمعت :   " طفت أطيف " {[26571]} . <br />
وحكى البصريون :   " طاف بطيف "  و " طفت أطيف " {[26572]} . <br />
قال ابن جبير ومجاهد  " الطيف "  :  الغضب{[26573]} . <br />
وقال ابن عباس  " طائف "  :  لمة من الشيطان{[26574]} . <br />
وقال السدي :  {  تذكروا }[ 201 ] أي :  تذكروا عقاب الله ( عز وجل ){[26575]} ،  فتابوا ،  أي :  تابوا إذا زلوا{[26576]} . <br />
قال ابن عباس :  {  فإذا هم مبصرون }[ 201 ] ،  أي :  منتهون عن المعصية{[26577]} .
