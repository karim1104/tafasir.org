ثم قال تعالى :  {  وأملي لهم إن كيدي متين }[ 183 ] . <br />
أي :  وأؤخرهم مدة من الدهر . <br />
و " المَلاَوة "  :  القطعة من الدهر ،  يقال بضم  " الميم "  وفتحها وكسرها ،  لغات فيها{[26244]} . <br />
{  إن كيدي }{[26245]} ،  أي :  إن عذابي{[26246]} . <br />
{  متين } أي :  شديد{[26247]} . <br />
وقيل :   " الكيد "  هنا :  هو أخذهم من حيث لا يشعرون{[26248]} . <br />
وأصل  " الكيد "  :  المكر{[26249]} . <br />
وقرأ ابن عباس :   " أن كيدي "  ،  بفتح الهمزة{[26250]} ،  جعل  " أنَّ "  :  مفعولا من أجله ،  أي :  من أجل أن الكيد متين وقع الإملاء{[26251]} . <br />
{  وأملي{[26252]} لهم }[ 183 ] ،  وقف{[26253]} . <br />
{  لا يعلمون }[ 182 ] ،  وقف{[26254]} . <br />
إن جعلت {  وأملي } مستأنفا{[26255]} .
