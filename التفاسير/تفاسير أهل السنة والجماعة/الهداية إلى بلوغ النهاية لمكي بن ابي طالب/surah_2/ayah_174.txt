قوله :  ( اِنَّ الذِينَ يَكْتُمُونَ مَا أَنْزَلَ اللَّهُ مِنَ الكِتَابِ ) [ 173 ] .  [ إلى قوله :  ( المُتَّقُونَ ) [ 176 ] . <br />
هذه الآية عند قتادة وغيره نزلت في أهل الكتاب(   {[5251]} ) .  كتموا ما أنزل الله عز وجل في كتابهم من أمر محمد صلى الله عليه وسلم(   {[5252]} ) . <br />
قال ابن عباس :   " هم اليهود كتموا اسم محمد صلى الله عليه وسلم وأخذوا(   {[5253]} ) عليه طمعاً(   {[5254]} ) قليلاً(   {[5255]} ) "  .  وهو قول السدي والربيع(   {[5256]} ) . <br />
وقال عكرمة في هذه الآية وفي قوله :  ( إِنَّ الذِينَ يَشْتَرُونَ بِعَهْدِ اللَّهِ وَأَيْمَانِهِمْ ثَمَناً قَلِيلاً )(   {[5257]} ) :  نزلت جميعاً في يهود " (   {[5258]} ) . <br />
قال ابن مسعود : / " قال النبي صلى الله عليه وسلم :   " مَن حَلَفَ عَلَى يَمِينٍ يَقْطَعُ بِها مَالاً(   {[5259]} ) لَقِيَ اللَّهَ ،  وَهُوَ عَلَيْهِ غَضْبَانٌ " (   {[5260]} ) .  وتصديقه في كتاب الله عز وجل :  ( إِنَّ الذِينَ يَشْتَرُونَ بِعَهْدِ اللَّهِ وَأَيْمَانِهِمْ ثَمَناً قَلِيلاً )(   {[5261]} ) . <br />
ومعنى ( يَشْتَرُونَ )(   {[5262]} ) يبتاعون به(   {[5263]} ) . <br />
والهاء في  " به "  تعود على الكتمان ،  أي :  وابتاعوا بكتمانهم ما أنزل الله عز وجل في كتابهم(   {[5264]} )/ من ذكر محمد صلى الله عليه وسلم . <br />
( ثَمَناً قَلِيلاً ) :  أي :  أخذوا عليه طمعاً(   {[5265]} ) قليلاً ،  أي :  أخذوا الرشوة وكتموا ما أنزل الله عز وجل وبدلوه وحرفوه . <br />
ثم قال تعالى :  ( أُوْلَئِكَ مَا يَاكُلُونَ فِي بُطُونِهِمُ إِلاَّ النَّارَ ) [ 173 ] . <br />
أي :  ما يأكلون في بطونهم من الرشا إلا(   {[5266]} ) ما يؤديهم إلى النار ،  ومثله :  ( اِنَّمَا يَاكُلُونَ فِي بُطُونِهِمْ نَاراً وَسَيَصْلَوْنَ سَعِيراً )(   {[5267]} ) أي :  ما يوردهم(   {[5268]} ) النار .  فاستغنى في(   {[5269]} ) الآيتين بذكر النار لفهم السامعين المعنى(   {[5270]} ) لأنه لما كان ما(   {[5271]} ) يأكلون من الطيبات بالرشا يوردهم النار كانوا كأنهم يأكلون النار .  وإنما قال ( فِي بُطُونِهِمُ ) ،  وقد علم أن الأكل لا يكون إلا في البطن لأن العرب تقول :   " جُعْتُ فِي غَيْرِ بَطْنِي "  و " شَبِعْتُ فِي غَيْرِ بَطْنِي "  .  فقيل في الآية :  ( فِي بُطُونِهِمُ )/للفرق والتأكيد(   {[5272]} ) . <br />
وقوله :  ( وَلاَ يُكَلِّمُهُمُ اللَّهُ يَوْمَ القِيَامَةِ ) [ 173 ] . <br />
أي :  لا يكلمهم بما يحبون ولا بما يشتهون ،  ويكلمهم بما يكرهون لأنه قد أخبر بأنه يقول لهم ( قَالَ اخْسَئُوا فِيهَا وَلاَ تُكَلِّمُونِ )(   {[5273]} ) وقيل :  معنى ( وَلاَ يُكَلِّمُهُمُ ) :  يغضب(   {[5274]} ) عليهم .  يقال :   " فُلانٌ لا يُكَلِّمُ فُلاَناً "  إذا غضب عليه(   {[5275]} ) . <br />
وقيل :  المعنى :  لا يسمعهم كلامه لأن الأبرار يسمعون كلامه . <br />
وقيل :  معناه :  لا يرسل لهم الملائكة بالتحية(   {[5276]} ) . <br />
وقوله :  ( وَلاَ يُزَكِّيهِمْ ) [ 173 ] :  أي لا يطهرهم من دنس ذنوبهم وكفرهم . <br />
( وَلَهُمْ عَذَابٌ اَلِيمٌ ) [ 173 ] :  أي(   {[5277]} ) موجع(   {[5278]} ) .
