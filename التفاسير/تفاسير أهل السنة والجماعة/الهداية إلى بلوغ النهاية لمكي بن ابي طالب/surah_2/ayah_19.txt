قوله :  ( أَوْ كَصَيِّبٍ مِّنَ السَّمَاءِ ) الآية(   {[1013]} ) [ 18 ] . <br />
الصَّيِّب :  المطر(   {[1014]} ) ،  وأصله  " صَيْبُوبٌ "   " عند البصريين(   {[1015]} ) من  " صابَ "   " يَصُوبُ "  ،  والصَّوْبُ :  نزول المطر .  يقال(   {[1016]} ) :   " صَابَ المطر(   {[1017]} ) "  إذا نزل(   {[1018]} ) . <br />
وقال الكوفيون :   " أصله صَوِيب على فعيلٍ كَرَغِيفٍ " (   {[1019]} ) ،  ويلزمهم(   {[1020]} ) ألا يعلوه كما لا يعلوا  " طويلاً "  .  واعتل عند البصريين لأن الياء(   {[1021]} ) إذا أسكنت(   {[1022]} ) وأتت بعدها واو ،  قلب من الواو ياء وأدغمت الأولى في الثانية  " كَمَيِّتٍ "  و " هَيِّنٍ "  . <br />
وقيل :  الصَّيِبُ :  السحاب الذي فيه المطر ،  لا المطر(   {[1023]} ) ،  روي عن ابن عباس(   {[1024]} ) . <br />
ومعنى :  صاب :  نزل وقصد .  والمعنى أن الله جل ذكره أباح للمؤمنين أن يمثلوا المنافقين بالذي استوقد ناراً أو(   {[1025]} ) بالصيب .  و " أو "  للإباحة(   {[1026]} ) .  وجمع  " صيب "  :  صيائب . <br />
والرعد :  مختلف فيه ،  فقال مجاهد :   " هو ملك يزجر السحاب بصوته ،  فالمسموع صوته(   {[1027]} ) "  . <br />
وقال شهر بن حوشب(   {[1028]} ) :   " الرعد ملك يتوكل بالسحاب/ يسوقه كما يسوق الحادي الإبل يسبح كلما خالفت سحابة صاح بها ،  فإذا اشتد غضبه ،  طارت النار من فيه فهي الصواعق(   {[1029]} ) "  . <br />
وقال ابن عباس :   " الرعد ريح " (   {[1030]} ) ،  قيل :  إنها تختنق تحت(   {[1031]} ) السحاب ،  فتتصاعد فيكون منه ذلك الصوت . <br />
وروى مجاهد عن ابن عباس أنه قال :   " الرعد اسم ملك ،  وصوته هذا تسبيحه ،  فإذا اشتد زجره بالسحاب(   {[1032]} ) ،  اضطرم السحاب من خوفه فيحتك ،  فتخرج الصواعق من بينه(   {[1033]} ) " (   {[1034]} ) . <br />
ومعنى الآية عند ابن عباس أنه مثل ضربه الله في المنافقين ،  فالظلمات(   {[1035]} ) ماهم فيه من الكفر والحذر من القتل على ما أبطنوا . <br />
وقوله :  ( مُحِيطٌ بِالْكَافِرِينَ )[ 18 ] أي جامعهم بقوة(   {[1036]} ) .
