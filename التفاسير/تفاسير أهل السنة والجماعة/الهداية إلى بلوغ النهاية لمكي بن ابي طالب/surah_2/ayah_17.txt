قوله :  ( مَثَلُهُمْ كَمَثَلِ الذِي اسْتَوْقَدَ نَاراً ) الآية(   {[974]} ) [ 16 ] . <br />
معناه(   {[975]} ) مثل هؤلاء المنافقين في حقنهم دماءهم بما أظهروا من الإيمان وَسِتْرِهم على غير(   {[976]} ) ذلك ،  كمثل الذي استوقد ناراً(   {[977]} ) ،  فلما أضاءت ما حوله ،  ذهب الله بنورهم .  فضياء ما حولهم هو ما حقن إقرارهم من دمائهم ومنع من(   {[978]} ) أموالهم في الدنيا . <br />
وقوله :  ( ذَهَبَ اللَّهُ بِنُورِهِمْ )[ 16 ] . <br />
هو ما يجدون يوم القيامة من عدم نورهم لأنهم لا ينتفعون بما أظهروا من الإيمان(   {[979]} ) إذ كان(   {[980]} ) باطنهم خلاف [ ما أظهروا ](   {[981]} ) . <br />
وقوله :  ( اسْتَوْقَدَ نَاراً )[ 16 ] .  أي استوقدها من غيره . <br />
وقيل :  معناه :  أوقد ،  أي أوقدها هو . <br />
واستفعل في كلام العرب يأتي على وجهين : <br />
- يكون بمعنى استدعى الفعل من غيره نحو قوله : <br />
( وَإِذِ اسْتَسْقَى مُوسَى لِقَوْمِهِ(   {[982]} ) )(   {[983]} ) .  أي استدعى أن يسقوا(   {[984]} ) . <br />
- ويكون بمعنى فعل(   {[985]} ) ،  نحو قوله :  ( وَاسْتَغْنَى اللَّهُ )(   {[986]} ) فلم يستدع غني من واحد ،  وبعده :  وَاللَّهُ غَنِيٌّ )(   {[987]} ) يدل على ذلك . <br />
و " ما "  في قوله :  ( أَضَاءَتْ مَا حَوْلَهُ ) في موضع نصب بِ " أَضَاءَتْ "  . <br />
وقيل :  هي زائدة/ لا موضع لها من الإعراب(   {[988]} ) .  والمعنى :   " فلما أضاءت النار(   {[989]} ) ،  الموضع الذي بحوله .   " فما "  غير زائدة(   {[990]} ) . <br />
قال قتادة :   " هي لا إله إلا الله ،  قالوها بأفواههم فَسَلِمُوا بها من القتل حتى إذا ماتوا ذهب الله بنورهم(   {[991]} ) "  . <br />
قال مجاهد :   " هو إقبالهم على المؤمنين والهدى .  وذهاب النور ،  هو(   {[992]} ) إقبالهم على(   {[993]} ) الكفار(   {[994]} ) . <br />
وقيل :  ذهب الله بنورهم ،  أي أظهر المؤمنين على ما أبطنوا من الكفر والنفاق . <br />
فبعد أن كان لهم عند المؤمنين نور بما أظهروا من الإيمان صاروا لا نور لهم عندهم ،  لما أعلمهم(   {[995]} ) به من سوء/ ما أبطنوا . <br />
والقول الأول عليه أكثر المفسرين ؛  أن ذهاب نورهم إنما يكون يوم القيامة وهو الذي ذكره الله في قوله :  ( يَوْمَ يَقُولُ المُنَافِقُونَ وَالمُنَافِقَاتُ لِلذِينَ ءَامَنُوا انظُرُونَا نَقْتَبِسْ مِن نُّورِكُمْ )(   {[996]} )(   {[997]} ) . <br />
والهاء والميم تعود على  " الذي "  لأنه بمعنى  " الذين كما قال :  ( وَالذِي جَاءَ بِالصِّدْقِ وَصَدَّقَ بِهِ أُوْلاَئِكَ هُمُ المُتَّقُونَ )(   {[998]} ) .  وفي هذا اختلاف ستراه في موضعه إن شاء الله(   {[999]} ) . <br />
ومنه قول الشاعر : <br />
إنَّ الَّذي حَانَتْ(   {[1000]} ) بِفَلْجٍ دِمَاؤُهُمْ       هُمُ الْقَوْمُ كُلَّ الْقَوْمِ يَا وأُمَّ خَالِدِ(   {[1001]} ) . <br />
وقيل :  تعود على المنافقين المتقدمي الذكر . <br />
وجواب  " لما "  محذوف تقديره :  فلما أضاءت ما حوله طفئت ،  ذهب الله بنورهم . <br />
قوله(   {[1002]} ) :  ( وَتَرَكَهُمْ فِي ظُلُمَاتٍ لاَّ يُبْصِرُونَ )(   {[1003]} ) [ 16 ] . <br />
هذا تمثيل للكفر الذي هم فيه يسيرون . <br />
وقيل :  هو شيء يكون في الآخرة وهو قوله :  ( فَضُرِبَ بَيْنَهُمْ بِسُورٍ )(   {[1004]} )(   {[1005]} ) . <br />
فظلمات جمع ظلمة ،  ويجوز إسكان(   {[1006]} ) اللام من  " ظلمات "  استخفافاً ،  ومن العرب من يبدل من الضمة فتحة ،  فيقول ظُلُمَاتٌ(   {[1007]} ) . <br />
وقال الكسائي :   " من قال :  ظُلُمَات(   {[1008]} ) بفتح اللام فهو جمع  " ظُلَم "  ،  و " ظُلَم "  جمع  " ظُلْمَة " (   {[1009]} ) .  ولا يجوز الفتح في مثل هذا الحرف الثاني مما لامه واو(   {[1010]} ) ونحو خطوات .  إنما/ يجوز الإسكان لا غير .  فإن كان اللام ياء لم يجز فيه إلا الإسكان نحو [ كُلْيَةٍ وَكُلْيَاتٍ(   {[1011]} ) ] .
