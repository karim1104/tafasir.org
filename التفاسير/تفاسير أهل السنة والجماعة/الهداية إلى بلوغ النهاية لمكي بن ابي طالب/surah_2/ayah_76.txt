/قوله :  ( قَالُوا ءَامَنَّا ) [ 75 ] . <br />
أي :  بأن(   {[2601]} ) صاحبكم نبي إليكم خاصة . <br />
وروي عن ابن عباس :   " أي : (   {[2602]} ) إذا لقوا محمداً(   {[2603]} ) .  قالوا :  آمنا ،  وإذا خلوا(   {[2604]} ) كفروا ،  وهم المنافقون من اليهود " (   {[2605]} ) . <br />
قوله(   {[2606]} ) :  ( بِمَا فَتَحَ اللَّهُ عَلَيْكُمْ ) [ 75 ] . <br />
كانوا يستفتحون بمحمد/ صلى الله عليه وسلم ،  فقالوا :  لا تقروا بأنه نبي ،  وقد كنتم تستفتحون به(   {[2607]} ) ،  أي :  تنظرون(   {[2608]} ) إذ(   {[2609]} ) سألتم الله به نصركم على عدوكم فقد علمتم أنه نبي ،  فإذا أقررتم لهم(   {[2610]} ) بنبوته حاجوكم بذلك عند ربكم . <br />
وقال(   {[2611]} ) أبو العالية :  ( بِمَا فَتَحَ اللَّهُ عَلَيْكُمْ ) يعني ما أنزل عليكم في التوراة من ذكر محمد صلى الله عليه وسلم(   {[2612]} ) . <br />
وقال قتادة :   " بما مَنَّ الله عليكم في التوراة من ذكر محمد صلى الله عليه وسلم(   {[2613]} ) فيحتجون عليكم بذلك " (   {[2614]} ) . <br />
وروي أن النبي [ عليه السلام قال لهم :  يا إخْوَةَ القِرَدَةِ(   {[2615]} ) ] والخَنَازِيرِ ،  فقالوا :  مَنْ أَخْبَرَ بِهَذَا مُحَمَّداً(   {[2616]} ) ؟  مَا(   {[2617]} ) جَرَى هَذَا إلاّ مِنْكُمْ .  أَفَتُحَدِّثُونَهُم بِمَا فَتَحَ الله عَلَيْكُمْ ؟  " (   {[2618]} ) . <br />
وقال السدي :   " كان ناس من اليهود آمنوا ثم نافقوا ،  وكانوا يحدثون المؤمنين بما مر على أسلافهم من العذاب ،  فقال بعضهم لبعض :  أتحدثونهم بما فتح الله عليكم من العذاب ليحاجوكم به .  أي :  يقولون لكم :  نحن أحب إلى الله منكم وأكرم منكم ؟  " (   {[2619]} ) . <br />
وعن ابن(   {[2620]} ) زيد قال :   " كانوا إذا قيل لهم :  أتعلمون أن في التوراة كذا وكذا ؟  قالوا :  نعم .  فيقول لهم رؤساؤهم :  لا تخبروهم بالذي أنزل عليكم ،  فيحاجوكم به عند ربكم .  وقال النبي [ عليه السلام ](   {[2621]} ) :  لا يَدْخُلُ عَلَيْنا(   {[2622]} ) قَصَبَةَ المَدِينَةِ إِلاّ مُؤْمِنٌ .  فقال رؤساؤهم :  اذهبوا فقولوا(   {[2623]} ) :  آمنا وادخلوا .  فإذا رجعتم اكفروا ،  وهو قوله :  ( وَقَالَتْ طَّائِفَةٌ مِّنَ اَهْلِ الكِتَابِ ءَامِنُوا بِالذِي أُنْزِلَ عَلَى الذِينَ ءَامَنُوا وَجْهَ النَّهَارِ )(   {[2624]} ) الآية .  فكانوا يؤمنون بالبكرة ويكفرون بالعشي " (   {[2625]} ) .
