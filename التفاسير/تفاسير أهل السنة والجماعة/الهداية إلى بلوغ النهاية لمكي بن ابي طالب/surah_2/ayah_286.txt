قوله :  ( اِلاَّ وُسْعَهَا ) [ 285 ] :  أي طاقتها فيما تعبدها(   {[9238]} ) به . <br />
فهذا توسيع(   {[9239]} ) ورخصة من الله(   {[9240]} ) وهو/ مثل قوله :  ( وَمَا جَعَلَ عَلَيْكُمْ فِي الدِّينِ مِنْ حَرَجٍ )(   {[9241]} ) ،  ومثل :  ( يُرِيدُ اللَّهُ بِكُمُ اليُسْرَ )(   {[9242]} ) ،  ومثل :  ( فَاتَّقُوا اللَّهَ مَا اسْتَطَعْتُمْ )(   {[9243]} ) . <br />
قوله :  ( إِن نَّسِينَا ) [ 285 ] .  أي :  نسينا فرضاً فرضته علينا ،  فلم نفعله(   {[9244]} ) . <br />
( أَوَاَخْطَأْنَا ) [ 285 ] . <br />
أي :  في فعل شيء نهيتنا(   {[9245]} ) عنه ،  ففعلناه على غير قصد إلى معصيتك(   {[9246]} ) . <br />
قال النبي صلى الله عليه وسلم(   {[9247]} ) :   " تَجَاوَزَ اللَّهُ لِهَذِهِ الأُمَّةِ عَنْ نِسْيَانِهَا وَمَا حَدَّثَتْ بِهِ أَنْفُسَهَا " (   {[9248]} ) . <br />
وكان النحاس يقول :  النسيان هنا الترك لأن الله تعالى لا يوصف بأن يعاقب على النسيان فيسأل في العفو عنه ،  لأنه(   {[9249]} ) ليس من تعمد العبد .  إنما هي آفة تدخل عليه .  وهو قول(   {[9250]} ) قطرب . <br />
قوله :  ( إِصْراً ) [ 285 ] . <br />
أي :  عهداً .  قاله ابن عباس ومجاهد والسدي وغيرهم(   {[9251]} ) . <br />
وقال غيره :   " لا تحمل علينا ذنوبنا ،  فتعاقبنا بمسخ أو عذاب كما كان من قبلنا(   {[9252]} ) "  . <br />
وقال الضحاك :  في قوله :  ( وَإِن تُبْدُوا مَا فِي أَنْفُسِكُمُ أَوْ تُخْفُوهُ يُحَاسِبْكُم بِهِ اللَّهُ ) ،  قال :   " إذا دعي الناس ليوم الحساب أخبرهم الله بما كانوا يسرون في أنفسهم ،  فيقول :   " إِنَّهُ لَمْ يَغِبْ عَنِّي شَيْءٌ وَإِنَّ كُتَّابِي مِنَ المَلاَئِكَةِ لَمْ يَكُونُوا يَطَّلِعُونَ عَلَى مَا تُسِرُّونَ فِي أَنْفُسِكُمْ ،  وَإِنِّي لاَ أُحَاسِبْكُمْ بِهِ اليَوْمَ "  . <br />
قال الضحاك :  هذا قول ابن عباس(   {[9253]} ) . <br />
وعن ابن عباس أن الله جل ذكره نسخ هذه الآية بقوله :  ( لاَ يُكَلِّفُ اللَّهُ نَفْساً اِلاَّ وُسْعَهَا ) ،  قال :  لما نزلت :  ( وَإِن تُبْدُوا مَا فِي أَنفُسِكُمُ ) الآية ،  وجدوا في أنفسهم منها وجداً شديداً(   {[9254]} ) فنسخ الله ذلك وأجارهم . (   {[9255]} ) منها ،  ثم علمهم أن يقولوا(   {[9256]} ) إذا عملوا سيئة :  ( رَبَّنَا لاَ تُؤَاخِذْنَا إِن نَّسِينَا أَوَاَخْطَأْنَا ) إلى قوله :  ( مِن قَبْلِنَا ) . <br />
قال :   " وكان الذين(   {[9257]} ) من قبلهم ،  إذا عملوا سيئة حرم الله عليهم طيبات أحلت لهم ،  وذلك قوله : ( فَبِظُلْمِ مِّنَ الذِينَ هَادُوا/ حَرَّمْنَا عَلَيْهِمْ طَيِّبَاتٍ(   {[9258]} ) اُحِلَّتْ لَهُمْ )(   {[9259]} ) . <br />
وقال للمؤمنين :  قولوا :  ( رَبَّنَا وَلاَ تُحَمِّلْنَا مَا لاَ طَاقَةَ لَنَا بِهِ ) إلى قوله :  ( الكَافِرِينَ ) . <br />
قال :   " فهذا شيء أعطاه الله أمة محمد صلى الله عليه وسلم لم(   {[9260]} ) يُعْطِهِ(   {[9261]} ) أحداً ممن كان قبلهم من الأمم ألا يؤاخذوا(   {[9262]} ) بنسيان(   {[9263]} ) ولا خطأ غيرَهم "  .  وقال الحسن :   " قال النبي صلى الله عليه وسلم :   " تَجَاوَزَ اللهُ لاِبْنِ آدَمَ عَمَّا نَسِيَ وَعَمَّا أَخْطَأَ(   {[9264]} ) وعَمَّا أُكْرِهَ عَلَيْهِ وَعَمَّا غُلِبَ عَلَيْهِ " (   {[9265]} ) . <br />
وروى عبد الله بن أبي أوفى(   {[9266]} ) أن النبي عليه السلام(   {[9267]} ) قال :   " إِنَّ اللَّهَ تَجَاوَزَ لأُمَّتِي عَمَّا حَدَّثَتْ(   {[9268]} ) بِهِ أنْفُسَهَا مَا لَمْ تَعْمَلْ بِهِ أَوْ تَكَلَّمْ بِهِ " (   {[9269]} ) . <br />
وروي أن ابن عمر قرأ هذه الآية وبكى بكاءً شديداً ،  ثم قال :   " والله لئن آخذنا اللهُ بهذا لنهلكن "  ،  فبلغ ذلك ابن عباس ،  فقال :   " يرحم(   {[9270]} ) الله أبا عبد الرحمن ؛  لقد(   {[9271]} ) وجد المسلمون منها مثل ما وجد حتى أنزل الله بعدها :  ( لَهَا مَا كَسَبَتْ وَعَلَيْهَا مَا اكْتَسَبَتْ )(   {[9272]} ) . <br />
قال ابن جبير :   " لما نزل :  ( وَإِن تُبْدُوا مَا فِي أَنفُسِكُمُ أَوْ تُخْفُوهُ يُحَاسِبْكُم بِهِ )(   {[9273]} ) ،  شق ذلك على الناس حتى نزلت بعدها :  ( لَهَا مَا كَسَبَتْ وَعَلَيْهَا مَا اكْتَسَبَتْ ) " (   {[9274]} ) . <br />
قال مجاهد :   " معنى(   {[9275]} ) :  ( وَإِن تُبْدُوا مَا فِي أَنفُسِكُمُ أَوْ تُخْفُوهُ ) :  يعني من الشك واليقين " (   {[9276]} ) . <br />
قال ابن جبير :   " نسخت ( لاَ يُكَلِّفُ اللَّهُ نَفْساً اِلاَّ وُسْعَهَا ) قوله :  ( وَإِن تُبْدُوا مَا فِي أَنفُسِكُمُ ) أو تخفوه(   {[9277]} ) الآية " (   {[9278]} ) .  وقاله إبراهيم/(   {[9279]} ) والشعبي(   {[9280]} ) . <br />
قال الضحاك :   " لما نزلت ( امَنَ الرَّسُولُ بِمَا أُنزِلَ إِلَيْهِ مِن رَّبِّهِ )(   {[9281]} ) ،  إلى آخر/ السورة ،  قال الله جل ذكره :   " قَدْ فَعَلْتُ "  . <br />
وروي عن الحسن والضحاك –أو عن أحدهما- أنه قال في قوله تعالى :  ( وَقَالُوا سَمِعْنَا وَأَطَعْنَا ) :   " معناه :  سمعنا القرآن(   {[9282]} ) أنه جاء من عند الله وأطعنا "  .  يقول :   " أقروا على أنفسهم بالطاعة لله فيما أمرهم به ونهاهم عنه "  . <br />
وروى حذيفة أن النبي عليه السلام(   {[9283]} ) قال :   " أُوتيتُ هَؤُلاَءِ الآيَاتِ مِنْ [ آخِرٍ سُورَةِ ](   {[9284]} ) البَقَرَةِ مِنْ بَيْتٍ كَنْزٍ تَحْتَ العَرْشِ ،  لَمْ يُعْطَ أَحَدٌ مِنْهُ قَبْلِي ،  وَلاَ يُعْطَى أَحَدٌ(   {[9285]} ) مِنْهُ بعْدِي " (   {[9286]} ) . <br />
وروى النعمان بن بشير(   {[9287]} ) أن النبي عليه السلام(   {[9288]} ) قال :   " إِنَّ اللَّهَ تَعَالَى كَتَبَ كِتَاباً قَبْلَ أَنْ يَخْلُقَ السَّمَوَاتِ وَالأَرْضَ بِأَلْفَيْ(   {[9289]} ) عَامٍ ،  فَأَنْزَلَ اللَّهُ مِنْهُ آيَتَيْنْ خَتَمَ بِهِمَا سُورَةَ البَقَرَةِ وَلاَ تُقْرَآنِ فِي دَارِ ثَلاَثَ لَيَالٍ فَيَقْرَبَهَا شَيْطَانٌ(   {[9290]} ) " (   {[9291]} ) . <br />
وقال علي بن أبي طالب رضي الله عنه(   {[9292]} ) :   " إن(   {[9293]} ) خواتم سورة البقرة وفواتحها من كنز تحت العرش " (   {[9294]} ) . <br />
وروي أنها لما نزلت قال النبي صلى الله عليه وسلم :   " وَيَحِقُّ لَهُ أَنْ يُؤْمِنَ " (   {[9295]} ) يعني نفسه . <br />
وذكر ابن(   {[9296]} ) الأنباري في هذه الآية ثلاثة أقوال :  قال :  إن(   {[9297]} ) الله تعالى يعاقب الذي يحدث/ نفسه بالمعصية ،  ولا يعملها ،  بِهَمٍّ أو حزن(   {[9298]} ) وبشبهه ،  ثم لا يحاسبه على ذلك يوم القيامة وهو معنى قول عائشة رضي الله عنها(   {[9299]} ) . <br />
/ والقول الثاني :  إن الله يُقبل على العبد يوم القيامة فيخبره بما حدث به نفسه من خير وشر ،  ثم لا يجزيه بما لم يظهر منه من عمل ،  وهو معنى قول الضحاك . <br />
والقول الثالث :  إنه منسوخ بقوله :  ( لاَ يُكَلِّفُ اللَّهُ نَفْساً اِلاَّ وُسْعَهَا )(   {[9300]} ) .  فالوسوسة وحديث النفس لا يملك الإنسان صرفه ،  ولا قدرة له على دفعه . <br />
قوله :  ( لاَ تُؤَاخِذْنَا إِن نَّسِينَا أَوَاَخْطَأْنَا )[ 285 ] . <br />
روى مالك عن نافع عن ابن(   {[9301]} ) عمر أنه قال :  قال النبي صلى الله عليه وسلم(   {[9302]} ) : /  " أَتَانِي جِبْرِيلُ ،  فَقَالَ لِي :  يَا مُحَمَّدُ ،  اللَّهُ يُقْرِئُكَ السَّلاَمَ وَيَقُولُ لَكَ :  إِنِّي قَدْ تَجَاوَزْتُ لَكَ عَنْ أُمَّتِكَ الخَطَأَ وَالنِّسْيَانَ وَمَا اسْتُكْرِهُوا عَلَيْهِ " (   {[9303]} ) . <br />
وقال(   {[9304]} ) ابن زيد :   " ( وَلاَ تَحْمِلْ عَلَيْنَا إِصْراً ) ،  أي ذنباً لا توبة منه ولا كفارة فيه " (   {[9305]} ) . <br />
وقال ابن وهب عن مالك :   " الإصر :  الأمر الغليظ " (   {[9306]} ) . <br />
وقال أهل اللغة :   " الإصر :  الثقل " (   {[9307]} ) . <br />
وقيل :  معناه(   {[9308]} ) :  لا تحمل علينا فرضاً يصعب علينا أداؤه ،  كما حملت على بني إسرائيل بعضهم يقتل(   {[9309]} ) بعضاً .  وشبهه(   {[9310]} ) . <br />
قوله :  ( وَلاَ تُحَمِّلْنَا مَا لاَ طَاقَةَ لَنَا بِهِ )[ 285 ] :  أي لا تكلفنا من الأعمال ما لا نطيق . <br />
وقال قتادة :  معناه :  لا تشدد علينا كما [ شددت على ](   {[9311]} ) من كان قبلنا(   {[9312]} ) . <br />
ومعنى :  ( مَالاَ طَاقَةَ لَنَا بِهِ ) أي :  ما لا نستطيعه إلا بمشقة شديدة وكلفة عظيمة .  فإنما سألوا دفع ما في طاقتهم لو كلفوه ،  ولكن له مشقة وكلفة .  ولم يسألوا دفع ما لا يطيقونه لو كلفوه ،  لأن ذلك لا يوصف به الله عز وجل فيجوز أن يسألوا في دفعه عنهم(   {[9313]} ) . <br />
قوله :  ( وَاعْفُ عَنَّا ) [ 285 ] :  أي امح ذنوبنا .  والعافي الدارس(   {[9314]} ) . <br />
( وَاغْفِرْ لَنَا ) [ 285 ] :  أي حط عنا(   {[9315]} ) ذنوبنا . <br />
( أَنْتَ مَوْلاَنَا ) [ 285 ] :  أي(   {[9316]} ) ولينا . <br />
وروت أم سلمة أن النبي عليه السلام قال :   " إِنَّ اللَّهَ تَجَاوَزَ لَكُمْ عَنْ ثَلاَثٍ :  عَنِ الخَطَأ والنِّسْيَانِ وَالاسْتِكْرَاهِ " (   {[9317]} ) . <br />
وعن أبي هريرة(   {[9318]} ) أن النبي عليه السلام(   {[9319]} ) قال :   " إِنَّ اللَّهَ تَجَاوَزَ لأُمَّتِي عَنْ كُلِّ شَيْءٍ حَدَّثَتْ بِهِ أَنْفُسُهَا مَا لَمْ تَعْمَلْ بِهِ " (   {[9320]} ) .
