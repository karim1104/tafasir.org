قوله :  ( لاَ يُؤَاخِذُكُمُ اللَّهُ بِاللَّغْوِ فِي أَيْمَانِكُمْ ) [ 223 ] . <br />
قال ابن عباس :   " هو ما سبق(   {[7257]} ) به اللسان على عجلة كقولك :   " لا والله ،  بلى والله " (   {[7258]} ) .  وكذلك قالت عائشة رضي الله عنها(   {[7259]} ) . <br />
وقال مجاهد :   " هما الرجلان يتبايعان فيقول أحدهما :  والله لا أبيعك بكذا .  ويقول الآخر :  والله(   {[7260]} ) لا أشتريه بكذا وكذا ،  فهو اللغو " (   {[7261]} ) . <br />
وقال أبو هريرة :   " لغو اليمين أن يحلف الرجل على الشيء(   {[7262]} ) يظن(   {[7263]} ) أنه هو بيقين منه ،  ثم يظهر له خلاف ظنه " (   {[7264]} ) .  وهذا القول أحسن الأقوال في لغو اليمين المعفو/عنها .  وروي مثله عن ابن عباس(   {[7265]} ) . <br />
وروي عنه أيضاً أنه قال :   " هو الرجل يحلف على الشيء(   {[7266]} ) فيرى الذي هو خير منه فأمره الله أن يكفر عن يمينه ويفعله/ ،  وأعلمه أنه لا يؤاخذه على ذلك " (   {[7267]} ) .  فهذه ثلاثة أقوال عن ابن عباس . <br />
وقال عطاء بن يسار(   {[7268]} ) :   " هو الخطأ في اليمين "  . <br />
وقال الحسن وغيره :   " هو الرجل يحلف على الشيء ،  وهو يظن أنه صادق ،  ثم يظهر له خلاف ذلك ،  فلا كفارة عليه ولا إثم " (   {[7269]} ) ،  وهو قول أبي هريرة المختار . <br />
وقال طاوس :   " هو الرجل يحلف(   {[7270]} ) في الغضب ،  فلا كفارة عليه ولا إثم " (   {[7271]} ) ،  وذكر قول النبي [ عليه السلام ](   {[7272]} ) :   " لاَ يَمِينَ فِي غَضَبٍ " (   {[7273]} ) . <br />
وقال/ابن جبير :   " هو الرجل يحلف أن يفعل ما نهى الله عنه أو يترك ما أمر الله به ،  فيمينه لغو ،  ولا كفارة عليه " (   {[7274]} ) . <br />
وقد قال ابن المسيب وابن الزبير :   " لا كفارة(   {[7275]} ) في معصية " (   {[7276]} ) .  وكذلك قال ابن عباس . <br />
وقال الشعبي :   " كفارة من حلف على المعصية أن يتوب منها "  . <br />
وقال :  [ زيد بن ](   {[7277]} ) أسلم :   " هو الرجل يقول :  أعمى الله بصري إن لم أفعل كذا وكذا ،  وأخرجني الله من مالي(   {[7278]} ) إن لم أصنع كذا ،  وشبهه مما يدعو به على نفسه .  فهو لغو ولا كفارة/فيه(   {[7279]} ) " (   {[7280]} ) . <br />
وقال ابن زيد :   " هو قول الرجل :  أنا كافر إن فعلت كذا ،  وجعلت مع الله إلاهاً إن صنعت كذا ،  وشبهه ،  فلا كفارة فيه(   {[7281]} ) " (   {[7282]} ) . <br />
وقال الضحاك  " اللغو من الأيمان ،  هي اليمين المكفرة يحلف ألا يفعل فيكفر ويفعل(   {[7283]} ) ،  ولا يؤاخذه الله بذلك ،  ولكن يؤاخذه بما يحلف عليه وقلبه يتيقن أن يمينه كذبة(   {[7284]} ) .  فتلك اليمين لا كفارة فيها ،  وهي اليمين الغموس ،  وهي أعظم من أن تكفر " (   {[7285]} ) . <br />
وقيل :  لغو اليمين الحنث ،  لا يؤاخذ الله من حنث في يمينه وكفر ،  لأن التكفير يسقط الإثم " (   {[7286]} ) . <br />
وقال إبراهيم :   " هو الرجل يحلف ألا يفعل الشيء ،  ثم ينسى فيفعله ،  فيمينه لغو " (   {[7287]} ) . <br />
واللغو في كلام العرب ما لا يحتاج إليه(   {[7288]} ) . <br />
قال(   {[7289]} ) نفطويه :   " اللغو في نفس اللغة الشيء المطرح(   {[7290]} ) يقال :  ألغيت هذا ،  أي طرحته " (   {[7291]} ) .  واللغو في الأيمان ما لم تكن النية معتقدة له ،  إنما جرى في عرض(   {[7292]} ) الكلام .  وهو معنى قول ابن عباس وعائشة(   {[7293]} ) . <br />
والسهو(   {[7294]} ) داخل في هذه الآية بغير نية ،  ويدخل فيه أيضاً ما ليس بيمين نحو قول الرجل :   " لا وحياتك "  وشبهه ،  وهذا يرجع إلى قول أبي هريرة وقول الحسن المتقدمي(   {[7295]} ) الذكر . <br />
وكان ابن عباس لا يرى الكفارة(   {[7296]} ) إلا في الأيمان التي تكون(   {[7297]} ) لغواً .  فأما ما كسبت(   {[7298]} ) القلوب ،  وعقدت اليمين فيه وهي تعلم أنها كاذبة(   {[7299]} ) ،  فلا كفارة فيه ،  والله يؤاخذ على ذلك بما شاء إن شاء(   {[7300]} ) . <br />
وتقدير الآية التي في المائدة(   {[7301]} ) على هذا القول :   " لا يؤاخذكم الله باللغو في إيمانكم فكفارته إطعام عشرة/ مساكين أو كسوتهم أو تحرير رقبة ،  فمن لم يجد فصيام ثلاثة أيام ،  ذلك كفارة أيمانكم إذا حلفتم ،  ولكن يؤاخذكم بما عقدتم الأيمان "  .  أي حلفتم وأنتم تعلمون أنكم كاذبون(   {[7302]} ) . <br />
[ كذلك التقدير ](   {[7303]} ) عند ابن عباس وابن جبير والضحاك وغيرهم(   {[7304]} ) . <br />
وقال قتادة :   " المؤاخذة هنا في الأيمان الفاجرة إنما هي إلزام الرجل الكفارة على يمينه ؛  فمن حلف وهو يعلم أنه كاذب(   {[7305]} ) وجبت(   {[7306]} ) عليه الكفارة ،  وهي المؤاخذة التي ذكر الله .  ومن حلف وهو يظن أنه/صادق ،  فهو لغو يمين ولا كفارة فيه ،  ولا إثم " (   {[7307]} ) . <br />
وروي عن الربيع مثله(   {[7308]} ) .  وهو قول عطاء والحكم(   {[7309]} ) . <br />
وقال السدي مثل ذلك إلا أنه قال :   " يؤاخذه في الدنيا بالكفارة على يمينه الفاجرة ويؤاخذه في الآخرة إن شاء الله عليها إذ(   {[7310]} ) حلف وهو يعلم أنه كاذب "  . <br />
والأيمان/عند أكثر الفقهاء ثلاثة : <br />
- اللغو ؛  وهو(   {[7311]} ) قوله :   " لا(   {[7312]} ) والله وبلى والله فلا شيء فيها "  . <br />
- والثانية(   {[7313]} ) :  العمد(   {[7314]} ) "  ؛  وهو أن يحلف متعمداً ألا يفعل الشيء ،  ثم يريد(   {[7315]} ) أن يفعله ويرى(   {[7316]} ) أن ذلك خير فيكفر ويفعل(   {[7317]} ) ولا شيء عليه .  وهي التي في قوله :  ( بِمَا عَقَّدتُّمُ الاَيْمَانَ فَكَفَّارَتُهُ )(   {[7318]} )(   {[7319]} ) . <br />
- والثالثة :  الغموس ؛  وهو أن يحلف على الشيء وهو يعلم أنه كاذب فلا كفارة فيها لعظمها ،  والله يفعل بفاعلها ما شاء .  وهي التي في قوله :  ( وَلَكِنْ يُّؤَاخِذُكُم بِمَا كَسَبَتْ قُلُوبُكُمْ )(   {[7320]} ) . <br />
وقال زيد بن(   {[7321]} ) أسلم :  ( وَلَكِنْ يُّؤَاخِذُكُم بِمَا كَسَبَتْ قُلُوبُكُمْ )(   {[7322]} ) .  قال :   " هو الشرك بالله والكفر لا اليمين "  .  يريد قول الرجل :  أنا كافر بالله(   {[7323]} ) إن فعلت وأنا مشرك إن فعلت ،  يؤاخذ عليه إن اعتقده بقلبه .  فإن لم يعتقده بقلبه ،  فهو اللغو الذي قال فيه :  ( لاَّ يُؤَاخِذُكُمُ اللَّهُ بِاللَّغْوِ فِي أَيْمَانِكُمْ )(   {[7324]} ) . <br />
ومن قال :   " أقسمت ألا أفعل "  ،  فإن أراد بالله وفعل(   {[7325]} ) كفر ،  وإن لم يرد ذلك فلا شيء عليه(   {[7326]} ) ،  وكل/أسماء الله يجب فيها الكفارة ،  وكذلك صفاته(   {[7327]} ) . <br />
وإذا حلف بالقرآن وحنث ،  فقال ابن مسعود :   " عليه لكل آية كفارة "  ،  وبه قال الحسن البصري . <br />
وأكثر الفقهاء على أن :  فيه كفارة يمين ،  ومنهم من قال :   " لا كفارة(   {[7328]} ) فيه لعظمه ،  وجلالة قدره(   {[7329]} ) "  . <br />
والعهد والميثاق والكفالة إذا أضيف ذلك إلى الله جلّ وحلف به ،  ففيه كفارة يمين عند مالك وغيره(   {[7330]} ) . <br />
ومن قال :   " حلفت "  ولم يحلف ،  فإن أراد اليمين كفر .  وإن أراد الكذب(   {[7331]} ) لم يكن عليه شيء . <br />
هذا(   {[7332]} ) قول مالك وغيره(   {[7333]} ) . <br />
ومن حلف بصدقة ماله أو بهديه(   {[7334]} ) أن يجعله في سبيل الله ؛  فقال عطاء والشعبي وغيرهم :   " لا شيء عليه "  . <br />
وروي عن عمر رضي الله عنه ،  وعائشة أن عليه كفارة يمين وهو قول جماعة من التابعين . <br />
وقال مالك(   {[7335]} ) :   " يخرج ثلث ماله " (   {[7336]} ) وهو قول الزهري وغيره .  وروي عن ابن عمر وابن عباس أنه :  يتصدق من ماله بمقدار الزكاة .  وقد قيل :  يعني بما جعل على نفسه .  روي ذلك عن ابن عمر .  وقال قتادة :   " يهدي بدنة "  . <br />
وقال/جابر بن زيد :   " إن كان ماله كثيراً(   {[7337]} ) نحو ألفين(   {[7338]} ) فعشره ،  وإن كان وسطاً نحو ألف ،  فسبعه ،  وإن كان قليلاً نحو خمسمائة فخمسه . <br />
ومن حلف بالمشي إلى بيت الله فحنث فلا شيء عليه(   {[7339]} ) عند ابن(   {[7340]} ) المسيب والقاسم بن محمد(   {[7341]} ) . <br />
وقال الحسن وجابر بن زيد(   {[7342]} ) وطاوس والنخعي وعطاء وقتادة/وغيرهم/ " عليه كفارة يمين(   {[7343]} ) "  ،  وهو قول الشافعي . <br />
وقال الشعبي ومالك وأبو حنيفة :   " يمشي كما حلف " (   {[7344]} ) . <br />
وقال ابن شبرمة :   " يحرم من يومه "  . <br />
وقال مالك رضي الله عنه(   {[7345]} ) :   " إن حنث في غير البلد الذي حلف فيه فعليه أن يأتي إلى ذلك البلد ،  فيمشي(   {[7346]} ) منه "  . <br />
ومن حلف بعتق رقبة فحنث ،  فأكثر الناس [ على أن عليه ](   {[7347]} ) كفارة يمين ،  وهو قول ابن عمر وابن عباس وأبي هريرة وعائشة وأم سلمة ،  وحفصة وقاله الحسن . <br />
