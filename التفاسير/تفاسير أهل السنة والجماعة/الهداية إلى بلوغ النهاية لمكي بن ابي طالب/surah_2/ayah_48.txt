قوله :  ( لاَّ تَجْزِي ) [ 47 ] . <br />
أي لا تقضي ،   " جزى عني الشيء "  قضى ،  و " أجزأني "  ،  كفاني ،  مهموز(   {[1892]} ) . <br />
وقيل :  هما بمعنى واحد(   {[1893]} ) .  وأصل الجزاء القضاء والتعويض . <br />
قوله :  ( نَفْسٌ عَن نَّفْسٍ ) [ 47 ] . <br />
أي لا تقضي ولا تغني .  وهي خاصة لقول النبي [ عليه السلام ](   {[1894]} ) :   " شَفاعَتي(   {[1895]} ) لأَهْلِ الكَبائِرِ(   {[1896]} ) مِنْ أُمَّتِي(   {[1897]} ) "  . <br />
ولقوله :   " لَيْسَ مِنْ نَبِيّ ،  إلاّ وَقَدْ أُعْطِيَ دَعْوَةً ،  وإِنِّي اخْتَبَأْتُ دَعْوَتِي شَفَاعَةً لٍأُمَّتِي(   {[1898]} ) وَهِيَ نَائِلَة(   {[1899]} ) مِنْهُم مَنْ لا يُشْرِكُ بِالله شَيْئاً " (   {[1900]} ) . <br />
فألفاظ(   {[1901]} ) الآية عامة ،  ومعناها الخصوص ،  هي في الكفار خاصة ،  وفي هذه الآية رد على اليهود لأنهم زعموا أنهم لا يعذبون يوم القيامة لأنهم أبناء الأنبياء ،  وأن آباءهم يشفعون لهم عند الله ،  فرد الله ذلك عليهم في هذه الآية . <br />
قوله :  ( مِنْهَا عَدْلٌ ) [ 47 ] .  أي :  فداء . <br />
وعن ابن عباس :   " عدل :  بدل " (   {[1902]} ) . <br />
وعن النبي صلى الله عليه وسلم :   " العَدْلُ :  الفِدْيَةُ " (   {[1903]} ) . <br />
وقولهم :   " لا يقبل منه(   {[1904]} ) ،  صرف ولا عدل " (   {[1905]} ) . <br />
قيل(   {[1906]} ) :  العدل :  الفدية ،  والصرف :  الحيلة .  قاله ابن(   {[1907]} ) السكيت(   {[1908]} ) . <br />
وقال المازني(   {[1909]} ) :   " العدل :  الفريضة ،  والصرف :  النافلة " (   {[1910]} ) . <br />
وقيل :  للفدية(   {[1911]} ) :  عدل ،  لأنها مثل الشيء ،  وأصل  " عدل(   {[1912]} ) الشيء "  مثله .  والعِدل –بكسر العين- ما حُمل على الظهر .  يقال :   " عِنْدي غلامٌ عِدْلُ غُلامِكَ ،  وَشاةٌ عِدْلُ شَاتِكَ "  ،  بِكسر العين ،  إذا كان أحدهما يعدل الآخر .  وكذلك يفعل في كل شيء يماثل الشيء من جنسه فإن أردت أن عندك/ قيمته من غير جنسه فتحت العين(   {[1913]} ) فقلت :   " عِنْدِي عَدْلُ غُلامِكَ وعَدْلُ شاتِكَ "  .  أي قيمتها بفتح العين(   {[1914]} ) . <br />
وروي في  " العدل "  الذي بمعنى الفدية كسر العين(   {[1915]} ) لغة(   {[1916]} ) . <br />
والضمير في  " ولاَّهُمْ "  يعود على الكفار لأن النَّفْسَين(   {[1917]} ) [ المذكورتين تدلان(   {[1918]} ) ] على ذلك . <br />
وقيل :  تعود(   {[1919]} ) على النفسين(   {[1920]} ) لأنهما بمعنى(   {[1921]} ) الجمع لم يقصد بهما قصد نفسين بأعيانهما ولأن التثنية أول(   {[1922]} ) الجمع ،  فهي جمع .
