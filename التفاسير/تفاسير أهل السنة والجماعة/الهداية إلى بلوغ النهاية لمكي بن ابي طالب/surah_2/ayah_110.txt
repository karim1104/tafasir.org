وقوله(   {[3555]} ) :  ( وَأَقِيمُوا الصَّلاَةَ وَءَاتُوا الزَّكَاةَ ) [ 109 ] . <br />
إقامة الصلاة هو أداؤها بفروضها لوقتها وهي الخمس الصلوات المفروضة(   {[3556]} ) . <br />
قال أنس بن(   {[3557]} ) مالك :   " فرضت الصلاة على النبي صلى الله عليه وسلم ليلة الإسراء خمسين ،  ثم نقصت حتى جعلت خمساً تخفيفاً من الله ،  ثم نودي يا مُحَمَّدُ :  إِنَّهُ لاَ يُبَدَّلُ القَوْلُ لَدَيَّ ،  وَإِنَّ لَكَ بِهَذه الخَمْسِ خَمْسِين " (   {[3558]} ) . <br />
ويجمع أوقاتها(   {[3559]} ) قوله(   {[3560]} ) :  ( فَسُبْحَانَ اللَّهِ حِينَ تُمْسُونَ )(   {[3561]} )/ .  يريد المغرب والعشاء الآخر/ ( وَحِينَ تُصْبِحُونَ )(   {[3562]} ) يريد الصبح .  ( وَحِينَ تُظْهِرُونَ )(   {[3563]} ) .  يريد الظهر ( وَعَشِيّاً )(   {[3564]} ) .  العصر(   {[3565]} ) .  فأما قوله تعالى :  ( اَقِمِ الصَّلاَةَ لِدُلُوكِ الشَّمْسِ )(   {[3566]} ) .  فقال علي بن أبي طالب(   {[3567]} )  " دلوكها غروبها "  .  وهو قول ابن(   {[3568]} ) مسعود(   {[3569]} ) . <br />
وروي عن ابن عباس :   " دلوكها زوالها "  .  وقاله ابن عمر وأبو هريرة .  ( وَقُرْءَانَ الفَجْرِ )(   {[3570]} ) صلاة الصبح . <br />
وقال قتادة :   " ( وَسَبِّحْ بِحَمْدِ رَبِّكَ قَبْلَ طُلُوعِ الشَّمْسِ )(   {[3571]} ) :  هي صلاة الفجر ( وَقَبْلَ(   {[3572]} ) غُرُوبِهَا )(   {[3573]} ) صلاة العصر .  ( وَمِنَ انَاءِ اللَّيْلِ )(   {[3574]} ) صلاة المغرب والعشاء الآخرة . <br />
وأجمع أهل العلم على أن أول وقت الظهر الزوال . <br />
وقال مالك :   " آخر وقتها أن يصير ظل كل شيء مثله بعد الزوال "  . <br />
وبه قال الثوري والشافعي(   {[3575]} ) وأبو ثور(   {[3576]} ) . <br />
وقال عطاء :   " لا تفريط في الظهر حتى(   {[3577]} ) تصفرّ الشمس "  . <br />
وقال طاوس(   {[3578]} ) :   " لا تفوت حتى الليل "  . <br />
وقال النعمان :   " آخر وقتها ما لم يصر الظل قامتين "  وأول وقت العصر إذا صار ظل كل شيء مثله " (   {[3579]} ) .  هذا قول مالك والثوري والشافعي(   {[3580]} ) وإسحاق(   {[3581]} ) وأحمد بن حنبل(   {[3582]} ) وأبي/ ثور . <br />
وقال النعمان :   " أول وقت العصر أن يصير(   {[3583]} ) الظل قامتين بعد الزوال ،  ولا تجزئ الصلاة قبل(   {[3584]} ) ذلك "  . <br />
وآخر وقتها أن يصير ظلك مثليه . <br />
وقال أحمد وأبو ثور :   " آخر وقتها ما لم تصفرّ الشمس على وجه الأرض " (   {[3585]} ) . <br />
وقال إسحاق :   " آخر وقتها أن يصلي منها ركعة قبل غروب الشمس لقول النبي عليه السلام "  مَنْ أَدْرَكَ مِنَ العَصْرِ رَكْعَةً قَبْلَ غُرُوبِ الشَّمْسِ(   {[3586]} ) ،  فَقَدْ أَدْرَكَهَا " (   {[3587]} ) . <br />
وهذا عند الشافعي(   {[3588]} ) وغيره إنما هو لأهل العذر(   {[3589]} ) . <br />
وروي عن ابن عباس :   " أن آخر وقتها غروب الشمس "  .  ووقت المغرب غروب الشمس وقتاً واحداً "  ،  وهو قول مالك والشافعي(   {[3590]} ) والأوزاعي(   {[3591]} ) . <br />
وقال الثوري وأحمد وإسحاق :   " وقتها إلى أن يغيب الشفق "  . <br />
ووقت العشاء مغيب الشفق .  وهو الحُمرة في قول ابن(   {[3592]} ) عمر وابن عباس ومالك وسفيان وابن أبي ليلى والشافعي(   {[3593]} ) . <br />
وروي عن أبي هريرة  " أنه البياض "  ،  وهو قول زفر(   {[3594]} ) .  وآخر وقتها عند النخعي(   {[3595]} ) ربع الليل . <br />
وروي عن عمر رضي الله عنه وأبي هريرة ،  وعمر بن عبد العزيز(   {[3596]} )  " إلى ثلث الليل "  وهو قول مالك .  وقال الثوري وابن المبارك(   {[3597]} ) وإسحاق  " إلى نصف الليل "  .  وروي ذلك عن عمر أيضاً .  ووقت صلاة الصبح انصداع الفجر إلى طلوع الشمس ،  ووقت الجمعة بعد الزوال ،  ومن(   {[3598]} ) صلى الجمعة قبل الزوال لم تجزه عند الجميع إلا أحمد بن(   {[3599]} ) حنبل فإنه(   {[3600]} ) أجاز قبل الزوال . <br />
ومعنى قولهم في التشهد :   " التحيات "  ،  قال أبو عبيد(   {[3601]} )  " التحيات(   {[3602]} ) الملك ،  والصلوات هي الصلوات(   {[3603]} ) الخمس لله ،  والطيبات(   {[3604]} ) هي الأعمال الزكيات لله "  .
