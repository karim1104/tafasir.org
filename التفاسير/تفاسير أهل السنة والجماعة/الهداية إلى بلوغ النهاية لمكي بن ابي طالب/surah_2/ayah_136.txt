قوله ( قُولُوا ءَامَنَّا بِاللَّهِ وَمَا أُنْزِلَ إِلَيْنَا ) [ 135 ] إلى قوله ( عَمَّا كَانُوا يَعْمَلُونَ ) [ 140 ] معناه :  قولوا أيها المؤمنون لهؤلاء اليهود والنصارى الذين قالوا لكم كونوا هوداً أو نصارى تهتدوا :   " آمنا بالله "  ،  أي :  صدقنا به ،  وصدقنا بما أنزل إلينا وهو القرآن ،  وبما أنزل إلى إبراهيم وإلى(   {[4326]} ) إسماعيل وإلى(   {[4327]} ) إسحاق وإلى يعقوب وإلى الأسباط ،  وهم اثنا عشر ولداً ليعقوب أنبياء كلهم ،  ولد كل واحد منهم أمة من الناس فسموا الأسباط(   {[4328]} ) . <br />
والسبط في اللغة الشجرة .  أي :  هم في الكثرة مثل الشجر(   {[4329]} ) . <br />
قال ابن عباس :   " كل الأنبياء من بني إسرائيل إلا عشرة :  نوح وهود وصالح ولوط وإبراهيم وإسحاق ويعقوب وإسماعيل(   {[4330]} ) ومحمد عليه السلام وعيسى صلى الله(   {[4331]} ) عليهم(   {[4332]} )/أجمعين(   {[4333]} ) "  . <br />
ثم قال :  ( وَمَا أُوتِيَ مُوسَى وَعِيسَى ) [ 135 ] . <br />
أي :  وصدقنا بما أوتي موسى وعيسى ،  يعني :  التوراة والإنجيل ،  وصدقنا بما أوتي النبيون من ربهم يعني :  من [ الكتب ،  كل ذلك حق(   {[4334]} ) ] ( لاَ نُفَرِّقُ بَيْنَ أَحَدٍ مِّنْهُمْ ) أي :  لا نفرق بين أحد من النبيين فنؤمن ببعض ونكفر ببعض كما فعلت اليهود والنصارى ،  بل نؤمن بالكل وبما جاء به الكل . <br />
وقال ابن عباس ووهب بن(   {[4335]} ) منبه :   " الأنبياء كلهم مائة وأربعة وعشرون ألف(   {[4336]} )/نبي(   {[4337]} ) كلهم من بني إسرائيل إلا عشرين نبياً .  وإسرائيل هو يعقوب بن إسحاق بن إبراهيم الخليل صلوات الله عليهم أجمعين(   {[4338]} ) .  وعدد الرسل ثلاثمائة(   {[4339]} ) وثلاثة عشر كلهم(   {[4340]} ) من بني يعقوب إلا عشرين رسولاً "  . <br />
وقال غير ابن عباس مثل قوله وزاد ،  فقال :   " عدد الأنبياء صلوات الله عليهم مائة وأربعة وعشرون ألف(   {[4341]} ) نبي ،  أولهم آدم وآخرهم محمد صلى الله عليه وسلم وعليهم أجمعين والرسل منهم ثلاثمائة وثلاثة عشر رسولاً ،  ذكر الله منهم في القرآن ستة وعشرين وهم(   {[4342]} ) :  آدم وإدريس ونوح وهود وصالح وإبراهيم ولوط وشعيب وإسماعيل وإسحاق ويعقوب ويوسف وموسى وهارون والأسباط واليسع وإلياس ويونس/وأيوب وداود وسليمان وزكرياء وعزير ويحيى وعيسى ومحمد صلى الله عليه وسلم(   {[4343]} ) وعليهم أجمعين .  ومنهم خمسة لم تذكر أسماؤهم وذكروا بغير أسمائهم وهم : <br />
الذي مرّ على القرية قيل هو أرميا(   {[4344]} ) ،  وصاحب موسى وهو الخضر(   {[4345]} ) ،  وقيل إنه ليس بنبي .  وثلاثة ذكرهم الله في سورة  " يس(   {[4346]} ) "  ولم يذكر أسماءهم ،  فأما ذو القرنين ،  فأكثر الناس على أنه ليس بنبي .  وكذلك اختلف في ذي الكفل . <br />
ثم قال ( وَنَحْنُ لَهُ مُسْلِمُونَ ) [ 135 ] . <br />
أي :  خاضعون بالطاعة لله(   {[4347]} ) . <br />
وروى ابن عباس أن النبي [ عليه السلام(   {[4348]} ) ] سأله نفر من اليهود :  بمن(   {[4349]} ) تؤمن من الرسل ؟  فقرأ عليهم الآية .  فلما ذكر عيسى جحدوا نبوّته ،  وقالوا :  لا نؤمن بعيسى ولا بمن آمن بعيسى ،  فأنزل الله جل ذكره :  ( قُلْ يَا أَهْلَ الكِتَابِ هَلْ تَنقِمُونَ مِنَّا إِلاَّ أَنَ امَنَّا بِاللَّهِ )(   {[4350]} ) إلى ( وَأَنَّ(   {[4351]} ) أَكْثَرَكُمْ فَاسِقُونَ )(   {[4352]} )(   {[4353]} ) .
