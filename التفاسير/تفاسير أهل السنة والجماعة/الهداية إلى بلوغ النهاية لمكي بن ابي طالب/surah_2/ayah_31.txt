قوله :  ( وَعَلَّمَ ءَادَمَ الاَسْمَاءَ(   {[1542]} ) )[ 30 ] الآية . <br />
اختلف في اشتقاق آدم ؛  فقال فيه ابن عباس :   " سمي آدم لأنه خلق من أديم الأرض " (   {[1543]} ) . <br />
وقال(   {[1544]} ) قطرب :   " آدم أفعل من الأدمة " (   {[1545]} ) .  وقيل هو أفعل من  " أدمت بين الشيئين "  أي خلطتهما(   {[1546]} ) . <br />
وقال الطبري :   " هو فعل ماض رباعي سمي به الشخص "  . <br />
وقال قطرب :   " من قال هو من أديم الأرض ،  ويلزمه صرفه لأنه فاعل "  . <br />
وذكر النحاس(   {[1547]} ) أنه أفعل من أديم الأرض وأدمتها ،  وهو ظاهر وجهها ،  ومنه سمي الإدام لأنه وجه الطعام وأعلاه والعرب تسمي الجلد الظاهر أدمة(   {[1548]} ) ،  والباطن بشرة . <br />
وحكي عن الأصمعي(   {[1549]} ) أن باطنه(   {[1550]} ) الأدمة وظاهره(   {[1551]} ) البشرة ،  وهو أولى من الأولى .  ويجمع آدم إذا كان صفة كحُمْر ،  وأوادم إذا كان اسماً  " كأحامدٍ " (   {[1552]} ) . <br />
قوله :  ( الاَسْمَاءَ )[ 30 ] .  قيل :   " وعلمه أسماء(   {[1553]} ) كل شيء حتى القصعة والقسوة "  .  قاله قتادة(   {[1554]} ) . <br />
وقيل :   " علمه أسماء الملائكة خاصة " (   {[1555]} ) قاله(   {[1556]} ) الربيع بن خثيم(   {[1557]} ) . <br />
قال مجاهد :   " علمه الله اسم كل شيء :  هذا جبل ،  هذا بحر ،  هذا كذا ،  هذا كذا ،  لكل الأشياء " (   {[1558]} ) . <br />
قال ابن جبير :   " علمه اسم كل شيء حتى البعير والبقرة والشاة " (   {[1559]} ) . <br />
قال عكرمة :   " علمه اسم الغراب والحمامة وكل شيء " (   {[1560]} ) . <br />
وقال غيرهم :   " علمه أسماء الأجناس والأنواع " (   {[1561]} ) . <br />
وقال ابن زيد(   {[1562]} ) :   " علمه أسماء ذريته كلهم " (   {[1563]} ) . <br />
واختار الطبري أن يكون علمه أسماء ذريته والملائكة لقوله :  ( ثُمَّ عَرَضَهُمْ ) ولم يقل  " عرضها(   {[1564]} ) "  ولا  " عرضهن " / الذي هو لما لا يعقل(   {[1565]} ) . <br />
وقيل :  علمه اسم كل شيء ومنفعته ولماذا يصلح(   {[1566]} ) . <br />
وقال القتبي(   {[1567]} ) :   " علمه أسماء ما خلق في الأرض " (   {[1568]} ) . <br />
وفي قراءة أُبي :   " ثُمَّ عَرَضَها " (   {[1569]} ) ،   " يريد عرض الأسماء " (   {[1570]} ) . <br />
وقوله  " عَرَضَها "  ولم يقل  " عرضهم "  يدل على أن الاسم هو المسمى ،  وهو مذهب أهل السنة .  وفي قراءة عبد الله  " ثُمَّ عَرَضَهُنَّ "  على التأنيث لما لا يعقل من الموات والأجناس . <br />
/ وقال ابن عباس :   " إنما عرض الأسماء على الملائكة " (   {[1571]} ) . <br />
وعن ابن(   {[1572]} ) مسعود :   " أنه إنما عرض الخلق " (   {[1573]} ) . <br />
فعلى الأول يكون  " عرضها "  .  وعلى الثاني يكون  " عرضهم " (   {[1574]} ) . <br />
قال مجاهد :   " عرض أصحاب الأسماء على الملائكة " (   {[1575]} ) . <br />
وقال ابن زيد :   " عرض أسماء(   {[1576]} ) ذريته كلها ،  أخذهم من ظهره ،  ثم عرضهم على الملائكة " (   {[1577]} ) . <br />
وقال ابن الأنباري(   {[1578]} ) :  الهاء في  " كلها "  تعود على/ الأسماء ،  والهاء في  " عَرَضَهم "  تعود على الأشخاص .  والهاء ف  " أنبِئهم "  وفي  " بأسمائهم " (   {[1579]} ) ،  وفي  " أنبأهم "  وفي  " بأسمائهم " (   {[1580]} ) كلها تعود على الملائكة على قول من قال :  إن الله تعالى علمه أسماء الملائكة ،  ويعود على الأشخاص على القول الآخر . <br />
قوله :  ( إِنْ كُنْتُمْ صَادِقِينَ )[ 30 ] . <br />
جوابه عند المبرد محذوف ،  معناه :  إن كنتم صادقين(   {[1581]} ) أن بني آدم يفسدون في الأرض ويسفكون الدماء فأنبئوني(   {[1582]} ) .
