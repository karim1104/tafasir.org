قوله :  ( وَلاَ جُنَاحَ عَلَيْكُمْ فِيمَا عَرَّضْتُم بِهِ مِنْ خِطْبَةِ النِّسَاءِ ) [ 233 ] . <br />
هو أن يعرض لها في العدة ،  فيقول :   " إنك لجميلة ،  وإن النساء من حاجتي(   {[7706]} ) ،  وإن فيك لراغب حريص ،  ولأحسن(   {[7707]} ) إليك "  ونحوه(   {[7708]} ) . <br />
قوله :  ( أَوَ اَكْنَنتُمْ فِي أَنفُسِكُمْ ) [ 233 ] . <br />
أي أخفيتم الخطبة ولم تبدوها ،  لا حرج في جميع ذلك(   {[7709]} ) . <br />
قال جابر بن زيد :   " ( لاَّ تُوَاعِدُوهُنَّ سِرّاً ) :  قال :  هو الزنا " (   {[7710]} ) . <br />
وقاله الحسن وقتادة والضحاك(   {[7711]} ) .  وقال ابن جبير :   " سراً نكاحاً " (   {[7712]} ) . <br />
وأصل السر الغشيان من غير وجهه(   {[7713]} ) . <br />
وقال ابن عباس :  لا تواعدوهن سراً ،  ألا(   {[7714]} ) ينكحن غيركم ،  ولا تعاهدوهن(   {[7715]} ) على ذلك(   {[7716]} ) . <br />
وقال مجاهد وعكرمة :   " لا تعاهدها على النكاح و[ تأخذ ميثاقها ألا ](   {[7717]} ) [ تتزوج غيرك ،  تفعل ذلك معها ](   {[7718]} ) سراً " (   {[7719]} ) . <br />
وقال ابن زيد :   " ( لاَّ تُوَاعِدُوهُنَّ سِرّاً ) :  لا تنكحوهن وتخفوا النكاح ،  فإذا خرجت من العدة ،  أظهرتموه " (   {[7720]} ) . <br />
واختار(   {[7721]} ) الطبري أن يكون السر الزنا(   {[7722]} ) . <br />
قوله :  ( اِلاَّ أَن تَقُولُوا قَوْلاً مَّعْرُوفاً ) [ 233 ] ،  هو التعريض المذكور . <br />
قال ابن زيد :   " نسخ هذا كله بقوله :  ( وَلاَ تَعْزِمُوا عُقْدَةَ النِّكَاحِ حَتَّى يَبْلُغَ الكِتَابُ أَجَلَهُ )[ 233 ](   {[7723]} ) . <br />
وأكثر الناس على أنه محكم ،  وأنه كله نهي عن عقد النكاح في العدة(   {[7724]} ) ،  ثم أرخص في التعريض الذي [ ليس هو ](   {[7725]} ) بوعد ولا عقد(   {[7726]} ) . <br />
ومعنى ( يَبْلُغَ الكِتَابُ أَجَلَهُ ) [ 233 ] :  تنقضي العدة(   {[7727]} ) . <br />
وقال السدّي :   " حتى تنقضي أربعة أشهر وعشر(   {[7728]} ) وهو الكتاب الذي ينقضي(   {[7729]} ) " (   {[7730]} ) .  وقاله قتادة(   {[7731]} ) ،  وهو قول ابن عباس والضحاك(   {[7732]} ) .
