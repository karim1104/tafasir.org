قوله :  ( وَيَسْئَلُونَكَ عَنِ المَحِيضِ )(   {[7098]} ) [ 220 ] . <br />
كان أصحاب النبي [ عليه السلام ](   {[7099]} ) في بدء الإسلام لا يساكنون النساء في المحيض ولا يواكلونهن(   {[7100]} ) ،  فسألوا النبي [ عليه السلام ](   {[7101]} ) عن ذلك ،  فعرفهم الله تعالى في الآية أن [ الذي يجتنب ](   {[7102]} ) من الحائض هو جماعها حتى تطهر(   {[7103]} ) ،  وأن ما سواء ذلك حلال(   {[7104]} ) . <br />
ثم قال :  ( فَإِذَا تَطَهَّرْنَ فَاتُوهُنَّ مِنْ حَيْثُ أَمَرَكُمُ اللَّهُ ) [ 220 ] . <br />
أي : (   {[7105]} ) في الفرج خاصة .  فهذا يدل على منع إتيانهن في الأدبار . <br />
وقيل :  معنى ( مِنْ حَيْثُ أَمَرَكُمُ اللَّهُ ) طهراً غير حيض . <br />
ومعنى :  ( قُلْ هُوَ أَذًى ) [ 220 ] . <br />
قال السدي :   " قل(   {[7106]} ) يا محمد :  قل(   {[7107]} ) هو قذر(   {[7108]} )/ " (   {[7109]} ) ،  وكذلك قال(   {[7110]} ) قتادة(   {[7111]} ) . <br />
وقال مجاهد :   " ( قُلْ هُوَ أَذًى ) قل هو دم " (   {[7112]} ) . <br />
والآية ناسخة لما كان عليه بنو إسرائيل من شريعتهم لأنهم كانوا لا يجتمعون مع الحائض في بيت ولا يواكلونها ،  فنسخت الآية ذلك .  فقالت اليهود عند نزولها :   " ما يدع محمد شيئاً من أمرنا إلا خالفنا فيه " (   {[7113]} ) . <br />
فللرجل أن يستمتع [ من الحائض ](   {[7114]} ) بما دون الفرج غير الدبر .  وهو قول عائشة وأم سلمة وابن عباس والحسن وعطاء والشعبي والنخعي والثوري وغيرهم(   {[7115]} ) .  وهو قول الشافعي الصحيح(   {[7116]} ) . <br />
ويروى(   {[7117]} ) عن ميمونة(   {[7118]} ) وسعيد بن المسيب أنها تعتزل فيما بين السرة والركبة ،  ويستمتع بها فيما دون ذلك(   {[7119]} ) ،  وهو قول مالك(   {[7120]} ) وأبي حنيفة(   {[7121]} ) . <br />
وقال عكرمة الشعبي :   " لا بأس بإتيانها دون الفرج " (   {[7122]} ) [ يريدان الفخذ ](   {[7123]} ) . <br />
وقال الثوري :   " لا بأس أن يباشرها(   {[7124]} ) إذا اتقى موضع الدم " (   {[7125]} ) . <br />
فإن أتاها وهي حائض ؛  فقال ابن عباس :   " يتصدق بدينار أو(   {[7126]} ) بنصف " (   {[7127]} ) . <br />
وقيل :  إن كان في فور الدم وقوته يتصدَّق بدينار ،  وإن كان في آخره وضعفه تصدق بنصف(   {[7128]} ) .  قاله النخعي وغيره(   {[7129]} ) . <br />
وقال الأوزاعي :   " إن كان وطئها في الدم تصدق بدينار ،  وإن وطئها بعد انقطاع الدم وقبل(   {[7130]} ) الطهر بالماء ،  تصدق بنصف دينار " (   {[7131]} ) . <br />
وقال سعيد بن جبير :   " عليه عتق رقبة "  . <br />
وقال الحسن :   " عليه مثل الذي على من وطئ في رمضان "  .  وجماعة/الفقهاء التابعين يقولون :  لا شيء عليه ويستغفر الله من ذلك ولا يعد ،  وقد أخطأ . <br />
قاله مالك والشافعي وأبو حنيفة .  وهو قول الشعبي والزهري وربيعة وأبي الزناد والليث بن سعد والثوري(   {[7132]} ) . <br />
وقال مالك والشافعي وابن حنبل وغيرهم :   " لا يطأها حتى تغتسل بالماء ،  فإن فعل قبل ذلك ،  وقد انقطع الدم ،  لم يكن عليه شيء ،  وقد أخطأ ويستغفر الله " (   {[7133]} ) . <br />
وهو قول سالم/بن عبد الله(   {[7134]} ) وسليمان بن يسار(   {[7135]} ) والزهري وربيعة والثوري(   {[7136]} ) . <br />
وقال عطاء وطاوس ومجاهد :   " إذا احتاج إلى وطئها قبل أن تغتسل ،  أمرها أن تتوضأ ،  ثم أصاب منها ما شاء " (   {[7137]} ) . <br />
وهو معنى قراءة من قرأ :   " حتى يَطْهُرْنَ "  مخففاً(   {[7138]} ) ،  أي ينقطع عنهن الدم . <br />
وفي مصحف أبي وابن مسعود :   " حتى يتطهرن " (   {[7139]} ) بالتاء(   {[7140]} ) أي بالماء(   {[7141]} ) ( فَإِذَا تَطَهَّرْنَ ) أي اغتسلن .  هذا قول الجماعة(   {[7142]} ) . <br />
وقال مجاهد وطاوس :   " إذا تطهرن للصلاة " (   {[7143]} ) . <br />
فليس يحب للقارئ أن يقف على  " يطهرن "  في(   {[7144]} ) قراءة من خففه لئلا يبيح(   {[7145]} ) وطء الحائض إذا انقطع عنها الدم ولم تتطهر(   {[7146]} ) بالماء(   {[7147]} ) .  فأما(   {[7148]} ) من قرأه بالتشديد(   {[7149]} ) ،  فالوقف عليه حسن لأن معناه(   {[7150]} ) :   " يتطهرن بالماء "  وقربها بعد التطهر بالماء إجماع . <br />
قوله :  ( مِّن حَيْثُ أَمَرَكُمُ اللَّهُ ) أمروا أن يأتوهن من حيث نهوا عنه " (   {[7151]} ) .  يعني يأتونهن(   {[7152]} ) بعد التطهر في الموضع الذي أمروا أن يعتزلوه في الحيض وهو الفرج . <br />
فهذا نص من الله على إتيان النساء في طهورهن(   {[7153]} ) في/الفرج دون غيره . <br />
وقيل :  معناه(   {[7154]} ) :  من قبل طهرهن ،  لا من قبل حيضهن(   {[7155]} ) . <br />
وقيل :  معناه إيتوهن من قبل النكاح الذي أمرتم به ،  وحل لكم لا من قبل الزنا الذي نهيتم عنه ،  وحرم عليكم(   {[7156]} ) . <br />
قوله :  ( يُحِبُّ التَّوَّابِينَ ) [ 220 ] . <br />
أي الراجعين/عن الذنوب ،  والمتطهرين بالماء للصلاة .  وهو/ظاهر(   {[7157]} ) اللفظ ،  وعليه أكثر أهل التأويل(   {[7158]} ) . <br />
وقال مجاهد :   " ( وَيُحِبُّ المُتَطَهِّرِينَ ) :  أي(   {[7159]} ) الذين لا(   {[7160]} ) يأتون النساء في أدبارهن " (   {[7161]} ) . <br />
وقيل :  معناه :  ويحب المتطهرين من الذنوب أن يعودوا(   {[7162]} ) بعد التوبة(   {[7163]} ) .  و( المُتَطَهِّرِينَ ) يعني به النساء والرجال ،  غلب المذكر على المؤنث ،  ولم يقل المتطهرات ،  لأنه يخص النساء خاصة إذ لا يغلب المؤنث على المذكر(   {[7164]} ) . <br />
وقيل :  عني بالمتطهرين اللواتي يتطهرن من الحيضة بالماء ،  وهذا يدل على أن الحائض لا توطأ إلا بعد التطهر بالماء/لأن(   {[7165]} ) من وطئها(   {[7166]} ) قبل التطهر بالماء ،  فقد وطئ من لا يحبه الله ،  وذلك ممنوع . <br />
ومن وطئ بعد التطهر بالماء ،  فقد وطئ من يحب(   {[7167]} ) الله .  وذلك حسن لأن الله إنما أحبهن على فعلهن وهو التطهر بالماء ،  ولم يحبهن على غير فعلهن ،  وهو انقطاع دم الحيض(   {[7168]} ) ،  فشكر الله لهن تطهرهن بالماء . <br />
وأتى  " بالمتطهرين " (   {[7169]} ) بلفظ التذكير لأنه يكون من الرجال والنساء ،  فغلب المذكر . <br />
وقوله :  ( أَنَّى شِئْتُمْ ) [ 221 ] .  أي مقبلة(   {[7170]} ) ومدبرة في الفرج(   {[7171]} ) .
