ثم قال تعالى(   {[4730]} ) :  ( وَلِكُلٍّ وِجْهَةٌ هُوَ مُوَلِّيهَا ) [ 147 ] . <br />
 " هو "  يرجع إلى  " كُلٍّ "  .  والهاء في  " مُوَلِّيهَا "  ترجع إلى القبلة(   {[4731]} ) .  وقيل :   " هُوَ "  يرجع إلى الله جل ذكره(   {[4732]} ) . <br />
فأما من قرأ  " هو مُوَلاَّها " (   {[4733]} ) ،  فهو يرجع إلى  " كُلٍّ "  لا غير . <br />
قال مجاهد :   " معناه(   {[4734]} ) :  ولكل صاحب(   {[4735]} ) ملة قبلة " (   {[4736]} ) .  يعني لليهود قبلة وللنصارى قبلة . <br />
قال ابن عباس :   " يعني بذلك أهل الأديان ؛  لكل أهل دين قبلة يرضونها .  ووجهة(   {[4737]} ) الله عز وجل حيث توجه المؤمنون " (   {[4738]} ) . <br />
وقال الضحاك :   " معناه ولكل صاحب ملة قبلة ،  وصاحب القبلة يولِّيها وجهه " (   {[4739]} ) . <br />
وقال السدي :   " المعنى ولكل قوم قبلة قد ولّوها " (   {[4740]} ) . <br />
والمعنى عند أهل العربية ،  هو موليها نفسه/أو وجهه(   {[4741]} ) . <br />
فأما من قرأ :   " مُوَلاَّهَا "  فالضمير على هذه القراءة لواحد ،  أي :  ولكل(   {[4742]} ) واحد من الناس قبلة ،  الواحد مولاها ،  أي :  مصروف إليها(   {[4743]} ) . <br />
وقال الأخفش :   " المعنى(   {[4744]} ) :  موليها الله إياه على ما يزعمون " /(   {[4745]} )يريد(   {[4746]} ) على قراءة موليها . <br />
وقال علي بن سليمان :   " المعنى هو متوليها ،  والوجهة والجهة والوجه واحد " (   {[4747]} ) . <br />
وعن قتادة في قوله :  ( هُوَ مُوَلِّيهَا )(   {[4748]} ) ،  قال :   " هي صلاتهم إلى بيت المقدس وصلاتهم إلى الكعبة " (   {[4749]} ) . <br />
فيكون التقدير على هذا :  ولكل ناحية وَجَّهَكَ إليها(   {[4750]} ) ربُّك يا محمد قبلةُ الله مولّيها عباده " (   {[4751]} ) .  وهو(   {[4752]} ) قول الأخفش الذي تقدم . <br />
ومعنى  " مولِّيها "  مول وجهه إليها ومستقبلها . <br />
وقال الطبري :   " التولية في الآية للكل ،  ووُجدت(   {[4753]} ) للفظ(   {[4754]} )  " كل "  ،  قال :   " فمعنى الكلام :  ولكل أهل ملة وجهة ،  الكل موليها وجوههم .  قال :  وأما قراءة ابن(   {[4755]} ) عامر فمعناه(   {[4756]} ) :  هو موجَّه نحوها ،  ويكون الكل حينئذ غير مسمى فاعله ،  ولو سمي فاعله لكان الكلام :  ولكل ذي ملة وجهة(   {[4757]} ) ،  الله موليها إياه بمعنى موجهه إليها . <br />
ورويت قراءة شاذة بإضافة  " كل "  إلى  " وجهة " (   {[4758]} ) ،  وهي قراءة/لا تجوز لأنه لا فائدة في الكلام إذا لم يتم الخبر " (   {[4759]} ) . <br />
ولو ثَنيتَ(   {[4760]} ) على قراءة الجماعة لقلت :   " هُمَا مُوَلِّياهَا "  ،  وفي الجمع [ هُمْ مُوَلُّوهَا ] وعلى قراءة ابن عباس/في التثنية  " هُمَا مُوَلَياهَا "  .  وفي الجمع [ هُمْ مُوَلَّوْهَا ](   {[4761]} ) .  فإن جئت بالمفعول الثاني في قراءة الجماعة ،  قلت في التثنية :   " هُمَا [ مُوَلِّياهَا هُمَا ](   {[4762]} ) وفي الجمع :   " هُمْ(   {[4763]} ) مُوَلُّوهَا هُمْ " (   {[4764]} ) . <br />
