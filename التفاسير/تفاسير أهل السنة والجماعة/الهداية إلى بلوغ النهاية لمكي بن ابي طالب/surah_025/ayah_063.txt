ثم قال تعالى{[50336]} :  {  وعباد الرحمن الذين يمشون على الأرض }[ 63 ] ،  عباد رفع بالابتداء ،  والخبر محذوف عند الأخفش{[50337]} . <br />
وقال الزجاج وغيره :  الخبر {  أولئك يجزون الغرفة{[50338]} }[ 75 ] . <br />
وقيل{[50339]} :  الخبر {  الذين يمشون{[50340]} }[ 63 ] ،  {  سلاما }[ 53 ] ،  منصوب{[50341]} على المصدر ،  وإن شئت  " تعالوا "  . <br />
وقوله :  {  قالوا سلاما }[ 63 ] ،  منسوخ{[50342]} بالأمر{[50343]} بالقتال :  إنما كان هذا قبل أن يؤمروا{[50344]} بالقتال ولم يتكلم سيبويه في شيء من الناسخ والمنسوخ إلا في هذه الآية ،  وهو من التسلم{[50345]} ،  لا من التسليم تقول :   " سلاما{[50346]} منك "  أي :  تسلما منك . <br />
قال سيبويه في الآية :  ولم يؤمر{[50347]} المسلمون يومئذ أن يسلموا على المشركين ،  ولكنه{[50348]} على قولك لا خير بيننا{[50349]} ولا شر ،  وقد ردت{[50350]} على سيبويه هذه العبارة .  إنما كان حسبه أن يقول :  ولم يؤمر المسلمون يومئذ أن يحاربوا{[50351]} المشركين .  ومعنى{[50352]} قول سيبويه على الصحة ،  ولم يؤمر المسلمون يومئذ أن يُسلّموا على المشركين ،  ولكن أمروا أن يَسْلموا منهم ويتبرءوا{[50353]} ثم نسخ ذلك الأمر بالحرب .  ومعنى الآية :  وعباد الرحمن الذين يرضاهم لنفسه عبادا :  هم الذين يمشون على الأرض في سكون ،  وتواضع ،  وخشوع ،  واستكانة .  هذا هو ضد{[50354]} مشي{[50355]} المختال الفخور{[50356]} المرح الذي هو مذموم الحال . <br />
ومعنى{[50357]} {  وإذا خاطبهم الجاهلون قالوا سلاما }{[50358]}[ 63 ] ،  أي : {[50359]} إذا خاطبهم أهل الجهل من العصاة{[50360]} والكفار بالقبيح صانوا أنفسهم عن مساواتهم في القبيح ،  قالوا قولا حسنا يسلمون{[50361]} به من مساواتهم في القبيح . <br />
وهم{[50362]} الذين{[50363]} يبيتون لربهم :  يصلون ويقولون كذا وكذا ،  ما حكى الله عنهم . <br />
قال أبو هريرة :  هم الذين لا يتجبرون ولا يتكبرون . <br />
وقيل معنى :  {  يمشون على الأرض هونا }[ 63 ] ،  أي{[50364]} بالسكينة والوقار{[50365]} ،  وغير مستكبرين ،  ولا متجبرين ،  ولا ساعين بفساد . <br />
قال ابن عباس :  يمشون بالطاعة والعفاف والتواضع{[50366]} . <br />
وقال مجاهد :  يمشون بالسكينة والوقار والحلم{[50367]} . <br />
وقال زيد بن أسلم{[50368]} :  التمست تفسير هذه الآية فلم أجدها عند أحد ،  فأتيت في النوم فقيل لي :  هم{[50369]} الذين لا يريدون يفسدون في الأرض . <br />
وقال ابن زيد{[50370]} هم الذين لا يتكبرون في الأرض ،  ولا يتجبرون ولا يفسدون ،  وهو قوله تعالى :  {  تلك الدار الآخرة نجعلها للذين{[50371]} لا يريدون{[50372]} علوا في الأرض ولا فسادا }{[50373]} . <br />
وقال الحسن{[50374]} يمشون حلماء ،  علماء ،  لا يجهلون ،  وإن جهل عليهم لم يجهلوا . <br />
{  وإذا خاطبهم الجاهلون قالوا سلاما }{[50375]}[ 63 ] ،  إذا{[50376]} خاطبهم الجاهلون بالله بما / يكرهون{[50377]} من القول أجابوهم بالمعروف والسداد من الخطاب ،  فقالوا :  تسلما{[50378]} منكم وبراءة بيننا وبينكم . <br />
قال الحسن{[50379]} :  إن المؤمنين قوم ذلل ،  ذلت والله منهم الأسماع ،  والأبصار ،  والجوارح ،  حتى يحسبهم الجاهل{[50380]} مرضى ،  وما بالقوم من مرض وإنهم لأصحاب القلوب ،  ولكن دخلهم من الخوف ما لم يدخل غيرهم ،  ومنعهم من الدنيا علمهم بالآخرة ،  فلما وصلوا إلى بغيتهم قالوا :  {  الحمد لله الذي أذهب عنا الحزن }{[50381]} ،  والله ما حزنهم حزن الدنيا ،  ولا تعاظم في أنفسهم بما طلبوا به الجنة :  أبكاهم الخوف من النار ،  وإنه من لا يتعزى بعزاء{[50382]} الله تقطع نفسه على الدنيا حسرات ومن لم ير لله عليه{[50383]} نعمة إلا في مطعم أو مشرب{[50384]} فقد قل عمله وحضر عذابه .
