{  وَمَا أَرْسَلْنَا مِن قَبْلِكَ مِن رَّسُولٍ وَلَا نَبِيٍّ إِلَّا إِذَا تَمَنَّى أَلْقَى الشَّيْطَانُ فِي أُمْنِيَّتِهِ فَيَنسَخُ اللَّهُ مَا يُلْقِي الشَّيْطَانُ ثُمَّ يُحْكِمُ اللَّهُ آيَاتِهِ وَاللَّهُ عَلِيمٌ حَكِيمٌ ( 52 ) لِيَجْعَلَ مَا يُلْقِي الشَّيْطَانُ فِتْنَةً لِّلَّذِينَ فِي قُلُوبِهِم مَّرَضٌ وَالْقَاسِيَةِ قُلُوبُهُمْ وَإِنَّ الظَّالِمِينَ لَفِي شِقَاقٍ بَعِيدٍ ( 53 ) وَلِيَعْلَمَ الَّذِينَ أُوتُوا الْعِلْمَ أَنَّهُ الْحَقُّ مِن رَّبِّكَ فَيُؤْمِنُوا بِهِ فَتُخْبِتَ لَهُ قُلُوبُهُمْ وَإِنَّ اللَّهَ لَهَادِ الَّذِينَ آمَنُوا إِلَى صِرَاطٍ مُّسْتَقِيمٍ ( 54 ) وَلَا يَزَالُ الَّذِينَ كَفَرُوا فِي مِرْيَةٍ مِّنْهُ حَتَّى تَأْتِيَهُمُ السَّاعَةُ بَغْتَةً أَوْ يَأْتِيَهُمْ عَذَابُ يَوْمٍ عَقِيمٍ ( 55 ) الْمُلْكُ يَوْمَئِذٍ لِّلَّهِ يَحْكُمُ بَيْنَهُمْ فَالَّذِينَ آمَنُوا وَعَمِلُوا الصَّالِحَاتِ فِي جَنَّاتِ النَّعِيمِ ( 56 ) وَالَّذِينَ كَفَرُوا وَكَذَّبُوا بِآيَاتِنَا فَأُوْلَئِكَ لَهُمْ عَذَابٌ مُّهِينٌ ( 57 ) } . <br />

التفسير : <br />
52 - وَمَا أَرْسَلْنَا مِن قَبْلِكَ مِن رَّسُولٍ وَلَا نَبِيٍّ إِلَّا إِذَا تَمَنَّى أَلْقَى الشَّيْطَانُ فِي أُمْنِيَّتِهِ فَيَنسَخُ اللَّهُ مَا يُلْقِي الشَّيْطَانُ ثُمَّ يُحْكِمُ اللَّهُ آيَاتِهِ وَاللَّهُ عَلِيمٌ حَكِيمٌ . <br />
الرسول :  من جاء بشرع جديد أنزله الله عليه ،  وأيده بمعجزة تؤيد رسالته . <br />
النبي :  صاحب معجزة تؤيد نبوته ،  وقد أمره الله أن يدعو الناس إلى شريعة من قبله ،  ولم ينزل الله عليه كتابا بشرع جديد ،  فالرسول صاحب شرع ،  والنبي حافظ شرع . <br />
تمنى :  لها عدة معان ،  منها :  أراد ،  وقرأ . <br />
ينسخ :  يزيل ويبطل . <br />
يحكم :  يحفظها من التأثر بوساوس الشيطان ،  ويجعلها محكمة مثبتة لا تقبل الرد بحال . <br />
كان عليه الصلاة والسلام حريصا على هداية قومه ،  حزينا من إعراضهم عن الرسالة والدعوة ،  وكان إذا قرأ القرآن الكريم ،  تصدى المنافقون ،  وشياطين الإنس يؤولونه تأويلا بعيدا ،  ويلصقون التهم بالقرآن ،  فيقولون إنه سحر وشعر وكهانة ،  وأساطير الأولين ،  والرسول بشر يتمنى لو اتسع نطاق الدعوة ،  وكثر المؤمنون ،  وربما دخل عليه الشيطان من هذا الباب . <br />
روى البخاري عن ابن عباس في ذلك :  إذا حدث – أي :  النبي – ألقى الشيطان في حديثه ،  فيبطل الله ما يلقي الشيطان . <br />
والمعنى :  أن النبي صلى الله عليه وآله وسلم كان إذا حدث نفسه ،  ألقى الشيطان في حديثه على جهة الحيلة ،  فيقول :  لو سألت الله عز وجل أن يغنمك ليتسع المسلمون .  ويعلم الله عز وجل أن الصلاح في غير ذلك ،  فيبطل ما يلقي الشيطان ،  أي :  أن المراد حديث النفس . <br />
قال النحاس :  وهذا من أحسن ما قيل في الآية وأعلاه وأجله . <br />
رأيان في تفسير الآية<br />
الأول : <br />
تأويل تمني بمعنى :  قرأ ،  وأمنيته :  بمعنى قراءته ،  ويدل على استعمال التمني بمعنى القراءة ،  قول حسان بن ثابت في عثمان بن عفان بعد قتله : <br />
تمنى كتاب أول ليلة *** وآخرها لاقى حمام المقادر<br />
أي :  قرأ كتاب الله في أول الليلة ،  وفي آخرها قتله الثائرون عليه . <br />
ويكون معنى الآية : <br />
وما أرسلنا قبلك يا محمد – رسولا ولا نبيا – إلا إذا قرأ وتلا كلام الله ،  ألقى الشيطان في قراءته وتلاوته ،  بعض الأقاويل والأباطيل ،  والشبه والتخيلات على أوليائه ،  ليجادلوه بالباطل ويردوا ما جاء به ،  تعجيزا لمسيرة دعوته ،  قال تعالى :  وإن الشياطين ليوحون إلى أوليائهم ليجادلوكم .  ( الأنعام :  121 ) . <br />
فَيَنسَخُ اللَّهُ مَا يُلْقِي الشَّيْطَانُ ثُمَّ يُحْكِمُ اللَّهُ آيَاتِهِ . <br />
فيزيل الله ما وسوس به الشيطان ،  من الكلمات والخرافات والأباطيل ،  التي تعلق بها بعض الكفار ،  ثم يجعل آياته محكمة محصنة مثبتة ،  لا تقبل التشويه والتزييف ،  أو الزيادة والنقصان . <br />
وَاللَّهُ عَلِيمٌ حَكِيمٌ . <br />
وهو عليم بكل شيء ،  له الحكمة التامة والحجة البالغة ،  فيجازي المفتري بافترائه ،  ويظهر الحق للمؤمنين . <br />
ومن حكمة الله تمكين الشيطان من إلقاء الشبهات ،  ليحاج أولياؤه بها ،  فيتمكن المؤمنون من ردها ،  ودحض المفتريات التي يتشدقون بها ،  ويرجع الحق إلى نصابه ،  فتظهر الحقيقة ناصعة بيضاء من بين تلك الظلماتxxxvi . <br />
الرأي الثاني : <br />
تفسير التمني بمعنى الرغبة والإرادة . <br />
ويكون معنى الآية : <br />
وما أرسلنا قبلك يا محمد من رسول ولا نبي ،  إلا إذا تمنى ،  وأراد هداية قومه إلى الحق ،  ألقى الشيطان فيما تمناه ،  الشبه في نفوس قومه ،  ليصدهم عن سبيله ،  ثم يبطل الله ما يلقيه الشيطان من الشبه ،  في نفوس الناس ،  ثم يحكم الله آياته ،  ويحميها ويمنعها من أباطيل المبطلين . <br />
وخلاصة المعنى : <br />
إن الصراع بين الحق والباطل أمر قديم ،  عرف الأنبياء والمرسلون قبلك يا محمد ،  وإن الأمر ينتهي بنصر الحق على الباطل ،  بتدبير الله وحكمته ،  فلا تجزع يا محمد مما يأتي به شياطين قومك من السعي بالباطل ،  في آيات الله معاجزين ،  بتسويل الشيطان الرجيم ،  أولئك أصحاب الجحيم وأباطيلهم إلى زوالxxxvii . <br />
من تفسير المراغي<br />
قال الشيخ أحمد مصطفى المراغي في تفسير المراغي : <br />
هذا وقد دس بعض الزنادقة في تفسير هذه الآية أحاديث مكذوبة لم ترد في كتاب من كتب السنة الصحيحة ،  وأصول الدين تكذبها ،  والعقل السليم يرشد إلى بطلانها ،  وأنها ليست من الحق في شيء .  1 ه . <br />
وخلاصة هذه الآراء المفتراة : <br />
أن النبي صلى الله عليه وآله وسلم عندما قرأ سورة النجم وفيها هذه الآيات :  أفرأيتم اللات والعزى * ومنواة الثالثة الأخرى *ألكم الذكر وله الأنثى .  ( النجم :  19 – 21 ) . <br />
ألقى الشيطان عندها كلمات ،  فقال :  وإنهن لهن الغرانيق العلى ،  وإن شفاعتهن لهي التي ترجى .  وكان ذلك من سجع الشيطان وفتنته ،  فوقعت هاتان الكلمتان في قلب كل مشرك بمكة ،  وانطلقت بها ألسنتهم ،  واستبشروا بها ،  ثم نسخ الله ما ألقى الشيطان ،  وأحكم الله آياته ،  وحفظ القرآن من الفرية . <br />
وقال :  وَمَا أَرْسَلْنَا مِن قَبْلِكَ مِن رَّسُولٍ وَلَا نَبِيٍّ إِلَّا إِذَا تَمَنَّى أَلْقَى الشَّيْطَانُ فِي أُمْنِيَّتِهِ فَيَنسَخُ اللَّهُ مَا يُلْقِي الشَّيْطَانُ ثُمَّ يُحْكِمُ اللَّهُ آيَاتِهِ وَاللَّهُ عَلِيمٌ حَكِيمٌ . <br />
وقد تصدى المفسرون والعلماء لبيان بطلان قصة الغرانيق ،  واستدلوا على بطلانها بالقرآن والسنة والمعقول . <br />
فالقرآن حافل بالتنديد بعبادة الأصنام ،  وبيان أنها لا تنفع ولا تضر ولا تسمع ولا تجيب ،  وليس لها شفاعة ،  بل هي حصب جهنم ،  وعبادتها احتقار للعقل ،  وتقليد أحمق للآباء والأجداد . <br />
وقد تكفل الله بحفظ كتابه من كل تحريف أو باطل ،  قال تعالى :  لَا يَأْتِيهِ الْبَاطِلُ مِن بَيْنِ يَدَيْهِ وَلَا مِنْ خَلْفِهِ تَنزِيلٌ مِّنْ حَكِيمٍ حَمِيدٍ .  ( فصلت :  42 ) . <br />
وقال سبحانه :  إِنَّا نَحْنُ نَزَّلْنَا الذِّكْرَ وَإِنَّا لَهُ لَحَافِظُونَ .  ( الحجر :  9 ) . <br />
من تفسير الفخر الرازي<br />
نقل فخر الدين الرازي كلاما طويلا ،  يفيد أن قصة الغرانيق باطلة عند أهل التحقيق ،  فحديث الغرانيق واهي الأصل ،  لم يخرجه أحد من أهل الصحة ،  ولا رواه بسند سليم متصل ثقة . <br />
وقال أبو بكر البزار : <br />
هذا الحديث لا نعلمه يروى عن النبي صلى الله عليه وآله وسلم بإسناد متصل يجوز ذكره ،  وهو من ناحية موضوعه يصادم أصلا من أصول العقيدة ،  وهو عصمة النبي صلى الله عليه وآله وسلم من أن يدس عليه الشيطان شيئا في تبليغ رسالتهxxxviii . <br />
وقد قال الله تعالى :  وما ينطق عن الهوى * إن هو إلا وحي يوحى .  ( النجم :  3 ،  4 ) . <br />
وقال سبحانه :  قُلْ مَا يَكُونُ لِي أَنْ أُبَدِّلَهُ مِن تِلْقَاء نَفْسِي إِنْ أَتَّبِعُ إِلاَّ مَا يُوحَى إِلَيَّ .  .  .  ( يونس :  15 ) . <br />
وقد روى البخاري في صحيحه أن النبي صلى الله عليه وآله وسلم قرأ سورة النجم ،  فلما بلغ آية السجدة سجد معه المسلمون والمشركون والإنس والجن ،  وليس فيه حديث الغرانيق . <br />
وروي حديث البخاري من طرق كثيرة ،  وليس فيها البتة حديث الغرانيق . <br />
وقد كان أعظم سعي النبي صلى الله عليه وآله وسلم في نفي الأوثان ،  وضلال عبادتها . <br />
ولو جوزنا زيارة شيء على القرآن لارتفع الأمان عن الشرع ،  فإنه لا فرق في العقل بين النقصان من الوحي وبين الزيادةxxxix . <br />
وقال سبحانه :  وَلَوْ تَقَوَّلَ عَلَيْنَا بَعْضَ الْأَقَاوِيلِ * لَأَخَذْنَا مِنْهُ بِالْيَمِينِ * ثُمَّ لَقَطَعْنَا مِنْهُ الْوَتِينَ * فَمَا مِنكُم مِّنْ أَحَدٍ عَنْهُ حَاجِزِينَ .  ( الحاقة :  44 – 47 ) . <br />
وقد أولع المستشرقون والطاعنون في الإسلام بقصة الغرانيق ،  والأمر لا يثبت للمناقشة ،  فالنص القرآني يقرر أن هذه قاعدة عامة ،  في الرسالات كلها مع الرسل كلهم ،  من حيث تعرضها للجحود والكنود ،  والإغراء والوعد والوعيد ،  قال تعالى :  وَمَا أَرْسَلْنَا مِن قَبْلِكَ مِن رَّسُولٍ وَلَا نَبِيٍّ إِلَّا إِذَا تَمَنَّى أَلْقَى الشَّيْطَانُ فِي أُمْنِيَّتِهِ فَيَنسَخُ اللَّهُ مَا يُلْقِي الشَّيْطَانُ ثُمَّ يُحْكِمُ اللَّهُ آيَاتِهِ وَاللَّهُ عَلِيمٌ حَكِيمٌ . <br />
ومن تأمل سيرة الرسل الكرام ،  وسيرة الرسول الأمين محمد صلى الله عليه وآله وسلم وجد التفسير واضحا ،  فالعقبات في وجه الدعوة كثيرة ،  والرسول بشر ،  فربما حاول أن يذلل العقبات من طريق الدعوة . <br />
وربما فكر في عمل مجلس للفقراء ،  ومجلس لللأغنياء تلبية لرغبات المشركين فتنزل آيات القرآن توجب أن تكون الجلسة عامة ،  وأن تفتح الأبواب للناس جميعا على السواء . <br />
قال تعالى :  وَاتْلُ مَا أُوحِيَ إِلَيْكَ مِن كِتَابِ رَبِّكَ لَا مُبَدِّلَ لِكَلِمَاتِهِ وَلَن تَجِدَ مِن دُونِهِ مُلْتَحَدًا * وَاصْبِرْ نَفْسَكَ مَعَ الَّذِينَ يَدْعُونَ رَبَّهُم بِالْغَدَاةِ وَالْعَشِيِّ يُرِيدُونَ وَجْهَهُ وَلَا تَعْدُ عَيْنَاكَ عَنْهُمْ تُرِيدُ زِينَةَ الْحَيَاةِ الدُّنْيَا وَلَا تُطِعْ مَنْ أَغْفَلْنَا قَلْبَهُ عَن ذِكْرِنَا وَاتَّبَعَ هَوَاهُ وَكَانَ أَمْرُهُ فُرُطًا * وَقُلِ الْحَقُّ مِن رَّبِّكُمْ فَمَن شَاء فَلْيُؤْمِن وَمَن شَاء فَلْيَكْفُرْ .  .  .  ( الكهف :  27 – 29 ) . <br />
وكذلك في قصة عبد الله ابن أم مكتوم حين أعرض عنه النبي صلى الله عليه وآله وسلم ،  وانشغل بأغنياء مكة رجاء أن يؤمنوا ،  وأن يؤمن بعدهم الأتباع والأعوان ،  فنزلت سورة عبس وتولى .  توضح توجيه الله العلي الكبير ،  في عرض الهدى على الناس أجمعين ،  فليس على الرسول إلا البلاغ ،  أما الإيمان فأمره موكول إلى الله تعالى . <br />
قال تعالى :  عَبَسَ وَتَوَلَّى * أَن جَاءهُ الْأَعْمَى * وَمَا يُدْرِيكَ لَعَلَّهُ يَزَّكَّى * أَوْ يَذَّكَّرُ فَتَنفَعَهُ الذِّكْرَى * أَمَّا مَنِ اسْتَغْنَى * فَأَنتَ لَهُ تَصَدَّى * وَمَا عَلَيْكَ أَلَّا يَزَّكَّى * وَأَمَّا مَن جَاءكَ يَسْعَى * وَهُوَ يَخْشَى * فَأَنتَ عَنْهُ تَلَهَّى * كَلَّا إِنَّهَا تَذْكِرَةٌ * فَمَن شَاء ذَكَرَهُ * فِي صُحُفٍ مُّكَرَّمَةٍ * مَّرْفُوعَةٍ مُّطَهَّرَةٍ * بِأَيْدِي سَفَرَةٍ * كِرَامٍ بَرَرَةٍ .  ( عبس :  1 – 16 ) .
