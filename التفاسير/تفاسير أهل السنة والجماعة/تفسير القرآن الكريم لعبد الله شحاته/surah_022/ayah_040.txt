40 - الَّذِينَ أُخْرِجُوا مِن دِيَارِهِمْ بِغَيْرِ حَقٍّ إِلَّا أَن يَقُولُوا رَبُّنَا اللَّهُ وَلَوْلَا دَفْعُ اللَّهِ النَّاسَ بَعْضَهُم بِبَعْضٍ لَّهُدِّمَتْ صَوَامِعُ وَبِيَعٌ وَصَلَوَاتٌ وَمَسَاجِدُ . <br />
الذين أخرجوا من ديارهم :  يعني مكة . <br />
بغير حق :  بغير موجب في الإخراج . <br />
إلا أن يقولوا :  أي بقولهم . <br />
ربنا الله :  وحده .  وهذا القول حق ،  فالإخراج به إخراج بغير حق . <br />
الصوامع :  جمع صومعة ،  وهي معبد خاص برهبان النصارى في الصحراء . <br />
الدير والبيع :  جمع بيعة بزنة حرفة ،  وهي متعبد النصارى عامة . <br />
وصلوات :  جمع صلاة ،  وهي كنيسة اليهود ،  وأطلق عليها صلاة ،  لأنهم يصلون فيها ،  وصلوتا بالعبرية :  معبد اليهود . <br />
مساجد :  واحدها نسجد وهو معبد المسلمين . <br />
قال ابن عباس : <br />
الَّذِينَ أُخْرِجُوا مِن دِيَارِهِمْ بِغَيْرِ حَقٍّ .  أخرجوا من مكة إلى المدينة بغير حق ،  يعني :  محمدا وأصحابه . <br />
إِلَّا أَن يَقُولُوا رَبُّنَا اللَّهُ . <br />
قال ابن كثير : <br />
ما كان لهم إساءة ولا ذنب ،  إلا أنهم وحدوا الله وعبدوه لا شريك له ،  كما قال تعالى :  يُخْرِجُونَ الرَّسُولَ وَإِيَّاكُمْ أَن تُؤْمِنُوا بِاللَّهِ رَبِّكُمْ .  ( الممتحنة :  1 ) . <br />
وقال تعالى في قصة أصحاب الأخدود :  وما نقموا منهم إلا أن يؤمنوا بالله العزيز الحميد .  ( البروج :  8 ) . <br />
وَلَوْلَا دَفْعُ اللَّهِ النَّاسَ بَعْضَهُم بِبَعْضٍ لَّهُدِّمَتْ صَوَامِعُ وَبِيَعٌ وَصَلَوَاتٌ وَمَسَاجِدُ يُذْكَرُ فِيهَا اسْمُ اللَّهِ كَثِيرًا . <br />
خلق الله الإنسان بيده ونفخ فيه من روحه ،  وأسجد له الملائكة وزوجه حواء ،  وأودع في الإنسان الصفات والإمكانيات ،  والاختيار والإرادة ،  ما يجعله سيد قراره ،  فهو يختار الهدى بإرادته ،  أو يختار الضلال بإرادته ،  وشاء الله أن تكون لهذه الحياة الدنيا نواميس وسنن كونية ،  ومن هذه النواميس صراع قوى الخير مع قوى الشر ،  وقد يتغلب الشر حينا ،  لكن العاقبة للمتقين ،  من أجل ذلك أنزل الله الكتب ،  وأرسل الرسل ،  وشرع الجهاد والنضال ،  لإيقاف البغي والعدوان ،  وكأن القرآن يحث المؤمنين على الجهاد والنضال ،  فقد جرت العادة أنه لا يدفع الشر إلا بمثله ،  والبادئ أظلم ،  يقول شوقي : <br />
والشر إن تلقه بالخير ضقت به ***  ذرعا وإن تلقه بالشر ينسجم<br />
ويقول الآخر : <br />
ولا خير في حلم إذا لم تكن له *** بوادر تحمي صفوه أن يكدرا<br />
فوضع الندى في موضع السيف بالعلا ***   مضر كوضع السيف في موضع الندى<br />
فلولا ما شرعه الله للأنبياء والمؤمنين ،  من قتال الأعداء ،  لهدمت معابدهم ،  واستبيحت حرماتهم ،  ومن هذه المعابد ما يأتي : <br />
الصوامع :  جمع صومعة ،  وكانت قبل الإسلام مختصة برهبان النصارى ،  وعباد الصابئة ،  والمراد بها هنا :  متعبد الرهبان . <br />
البيع :  جمع بيعة بوزن كسرة ،  وهي مصلى النصارى جميعا . <br />
الصلوات :  جمع صلاة ،  وهي كنيسة اليهود . <br />
المساجد :  جمع مسجد ،  وأكثر ما يطلق على مصلى المسلمين . <br />
وقيل :  المعنى لولا هذا الدفع لهدمت في زمن موسى الكنائس ،  وفي زمن عيسى الصوامع والبيع ،  وفي زمن محمد صلى الله عليه وآله وسلم المساجد . <br />
يُذْكَرُ فِيهَا اسْمُ اللَّهِ كَثِيرًا . <br />
قيل :  الضمير راجع إلى المساجد لأنها أقرب المذكورات . <br />
وقال الضحاك :  الجميع يذكر فيها الله كثيرا . <br />
وقال الطبري : <br />
الصواب لهدمت صوامع الرهبان ،  وبيع النصارى ،  وصلوات اليهود – وهي كنائسهم – ومساجد المسلمين ،  التي يذكر فيها اسم الله كثيرا ؛  لأن هذا هو المستعمل المعروف في كلام العرب . <br />
وَلَيَنصُرَنَّ اللَّهُ مَن يَنصُرُهُ إِنَّ اللَّهَ لَقَوِيٌّ عَزِيزٌ . <br />
إن الله تعالى تكفل بالنصر لمن نصر دينه وشريعته . <br />
قال تعالى :  إِن تَنصُرُوا اللَّهَ يَنصُرْكُمْ وَيُثَبِّتْ أَقْدَامَكُمْ .  ( محمد :  7 ) . <br />
إِنَّ اللَّهَ لَقَوِيٌّ عَزِيزٌ . <br />
فبقوته خلق كل شيء فقدره تقديرا ،  وبعزته لا يقهره قاهر ولا يغلبه غالب ،  بل كل شيء ذليل لديه فقير إليه ،  ومن كان القوي العزيز ناصره فهو المنصور ،  وعدوه هو المقهور . <br />
قال الله تعالى :  وَلَقَدْ سَبَقَتْ كَلِمَتُنَا لِعِبَادِنَا الْمُرْسَلِينَ * إِنَّهُمْ لَهُمُ الْمَنصُورُونَ * وَإِنَّ جُندَنَا لَهُمُ الْغَالِبُونَ .  ( الصافات :  171 – 173 ) . <br />
جاء في تفسير المراغي : <br />
وَلَيَنصُرَنَّ اللَّهُ مَن يَنصُرُهُ<br />
أي :  وليعينن من يقاتل في سبيله ،  لتكون كلمة الله هي العليا ،  وكلمة الذين كفروا السفلى ،  ولقد أنجز الله وعده ونصر المسلمين على صناديد قريش ،  وأكاسرة العجم ،  وقياصرة الروم ،  وأورثهم أرضهم وديارهم . <br />
ونحو الآية قوله تعالى :  يَا أَيُّهَا الَّذِينَ آمَنُوا إِن تَنصُرُوا اللَّهَ يَنصُرْكُمْ وَيُثَبِّتْ أَقْدَامَكُمْ * وَالَّذِينَ كَفَرُوا فَتَعْسًا لَّهُمْ وَأَضَلَّ أَعْمَالَهُمْ .  ( محمد :  7 ،  8 ) .
