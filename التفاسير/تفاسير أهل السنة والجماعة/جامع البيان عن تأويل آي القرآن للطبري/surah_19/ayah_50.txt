وَوَهَبْنا لَهُمْ مِنْ رَحَمتِنا يقول جلّ ثناؤه :  ورزقنا جميعهم ،   يعني إبراهيم وإسحاق ويعقوب من رحمتنا ،   وكان الذي وهب لهم من رحمته ،   ما بسط لهم في عاجل الدنيا من سعة رزقه ،   وأغناهم بفضله .  <br />
وقوله وَجَعَلْنا لَهُمْ لِسَانَ صِدْقٍ عَلِيّا يقول تعالى ذكره :  ورزقناهم الثناء الحسن ،   والذكر الجميل من الناس ،   كما : <br />
حدثني عليّ ،   قال :  حدثنا عبد الله ،   قال :  ثني معاوية ،   عن عليّ ،   عن ابن عباس ،   قوله :  وَجَعَلْنا لَهُمْ لِسانَ صِدْقِ عَلِيّا يقول :  الثناء الحسن .  <br />
وإنما وصف جلّ ثناؤه اللسان الذي جعل لهم بالعلوّ ،   لأن جميع أهل الملل تحسن الثناء عليهم ،   والعرب تقول :  قد جاءني لسان فلان ،   تعني ثناءه أو ذمه ومنه قول عامر بن الحارث : <br />
إنّي أتَتْنِي لِسانٌ لا أُسَرّ بِهَا *** مِنْ عَلْوَ لا عَجَبٌ مِنْها وَلا سَخَرُ<br />
ويُروى :  لا كذب فيها ولا سَخَر .  <br />
جاءَتْ مُرجَمّةً قد كُنْتُ أحْذَرُها *** لَوْ كانَ يَنْفَعُنِي الإشْفاقُ والحَذَرُ<br />
مرجمّة :  يظنّ بها .  <br />
