قوله :  {  لا أبرح حتى أبلغ مجمع البحرين } آية 60<br />
عَنِ ابْنِ زَيْدٍ ،   فِي قَوْلِهِ :  "  {  لا أَبْرَحُ حَتَّى أَبْلُغَ مَجْمَعَ الْبَحْرَيْنِ } ،   قَالَ :  حتى أنتهي "  .  <br />
عَنْ قَتَادَة ،   فِي قَوْلِهِ :  "  {  مَجْمَعَ الْبَحْرَيْنِ } ،   قَالَ :  بحر فارس ،   والروم ،   هما بحر المشرق ،   والمغرب "  .  <br />
عَنْ أَبِي بْنِ كَعْبٍ ،   فِي قَوْلِهِ :  "  {  مَجْمَعَ الْبَحْرَيْنِ } ،   قَالَ :  أفريقية "  .  <br />
عَنْ مُحَمَّد بن كَعْبٍ ،   فِي قَوْلِهِ :  "  {  مَجْمَعَ الْبَحْرَيْنِ } ،   قَالَ :  طنجة "  .  <br />
عَنِ السُّدِّيِّ ،   فِي قَوْلِهِ :  "  {  مَجْمَعَ الْبَحْرَيْنِ } ،   قَالَ :  الكر والرس ،   حيث يصبان في البحر "  .   قوله :  {  أو أمضي حقبا } آية 60<br />
عَنْ ابْنِ عَبَّاسٍ ،   فِي قَوْلِهِ :  "  {  أَوْ أَمْضِيَ حُقُبًا } ،   قَالَ :  دهرا "  .  <br />
عَنْ مُجَاهِدٍ ،   فِي قَوْلِهِ :  "  {  أَوْ أَمْضِيَ حُقُبًا } ،   قَالَ :  سبعين خريفاً .   وفي قوله : {  فلما بلغا مجمع بينهما } قال :  بين البحرين {  نسيا حوتهما } قال :  أضلاه في البحر {  فاتخذ سبيله في البر عجبا } قل :  موسى يعجب من أثر الحوت ودوراته التي غاب فيها {  فارتدا على آثارهما قصصا } قال :  اتباع موسى وفتاه أثر الحوت حيث يشق البحر راجعين .
