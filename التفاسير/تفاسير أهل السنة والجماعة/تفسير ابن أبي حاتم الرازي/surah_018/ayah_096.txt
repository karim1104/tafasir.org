قوله :  {  زُبَرَ الْحَدِيدِ } آية 96<br />
عَنْ ابْنِ عَبَّاسٍ ،   فِي قَوْلِهِ :  "  {  زُبَرَ الْحَدِيدِ } ،   قَالَ :  قطع الحديد "  .  <br />
قوله :  {  بَيْنَ الصَّدَفَيْنِ }<br />
عَنْ ابْنِ عَبَّاسٍ ،   فِي قَوْلِهِ :  "  {  بَيْنَ الصَّدَفَيْنِ } ،   قَالَ :  الجبلين "  .  <br />
عَنْ مُجَاهِدٍ ،   فِي قَوْلِهِ :  "  {  بَيْنَ الصَّدَفَيْنِ } ،   قَالَ :  رأس الجبلين "  .  <br />
قوله :  {  قِطْرًا }<br />
عَنْ ابْنِ عَبَّاسٍ ،   فِي قَوْلِهِ :  "  {  قِطْرًا } ،   قَالَ :  النحاس "  .  <br />
عَنْ عِكْرِمَةَ ،   فِي قَوْلِهِ :  "  {  آتُونِي أُفْرِغْ عَلَيْهِ قِطْرًا } ،   قَالَ :  نحاساً "  .  <br />
عَنْ عِكْرِمَةَ ،   فِي قَوْلِهِ :  "  {  آتُونِي أُفْرِغْ عَلَيْهِ قِطْرًا } ،   قَالَ :  نحاساً ليلزم بعضه بعضاً "  .
