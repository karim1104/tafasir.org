قَوْلهُ تَعَالَى :  {  كُلا نُمِدُّ هَؤُلاءِ وَهَؤُلاءِ مِنْ عَطَاءِ رَبِّكَ }<br />
عَنِ الضَّحَّاكِ رَضِيَ اللهُ عَنْهُ ،   فِي قَوْلِهِ :  "  {  كُلا نُمِدُّ هَؤُلاءِ وَهَؤُلاءِ مِنْ عَطَاءِ رَبِّكَ } ،   أي :  أنَّ الله قسم الدُّنْيَا بين البر والفاجر ،   والآخرة خصوصاً عند ربك للمتقين "  .  <br />
قوله تَعَالَى :  {  كُلا نُمِدُّ }<br />
عَنِ الحَسَنِ رَضِيَ اللهُ عَنْهُ ،   فِي قَوْلِهِ :  "  {  كُلا نُمِدُّ } قَالَ :  كلا نرزق في الدُّنْيَا البر ،   والفاجر "  .  <br />
عَنِ السُّدِّىِّ رَضِيَ اللهُ عَنْهُ ،   فِي قَوْلِهِ :  "  {  كُلا نُمِدُّ هَؤُلاءِ وَهَؤُلاءِ } ،   يَقُولُ :  نمد الكفار والمؤمنين {  مِنْ عَطَاءِ رَبِّكَ } يَقُولُ :  مِنَ الرزق "  .  <br />
عَنِ ابْنِ زَيْدٍ رَضِيَ اللهُ عَنْهُ ،   فِي قَوْلِهِ :  "  {  كُلا نُمِدُّ هَؤُلاءِ وَهَؤُلاءِ } هؤلاء أَهْل الدُّنْيَا ،   وهؤلاء أَهْل الآخرة {  وَمَا كَانَ عَطَاءُ رَبِّكَ مَحْظُورًا } قَالَ :  ممنوعاً "  .  <br />
عَنِ الضَّحَّاكِ رَضِيَ اللهُ عَنْهُ فِي قَوْلِهِ :  "  {  مَحْظُورًا } قَالَ :  ممنوعاً "  .  <br />
