قَوْلهُ تَعَالَى :  {  فَاسْلُكِي سُبُلَ رَبِّكِ ذُلُلا } آية 69<br />
عَنْ مُجَاهِدٍ فِي قَوْلِهِ :  "  {  فَاسْلُكِي سُبُلَ رَبِّكِ ذُلُلا } ،   قَالَ :  طرقا لا يتوعر عليها مكان سلكته "  .  <br />
قوله تَعَالَى :  {  وَذَلَّلْنَاهَا لَهُمْ }<br />
عَن ابن زَيْدٍ في الآية ،   قَالَ :   " الذلول الّذِي يقاد ويذهب به حيث أراد صاحبه ،   قَالَ :  فهم يخرجون بالنحل وينتجعون بها ،   ويذهبون وهي تتبعهم ،   وقرأ :  {  أَوَلَمْ يَرَوْا أَنَّا خَلَقْنَا لَهُمْ مِمَّا عَمِلَتْ أَيْدِينَا أَنْعَامًا فَهُمْ لَهَا مَالِكُونَ وَذَلَّلْنَاهَا لَهُمْ } "  .  <br />
عَن السُّدِّيِّ رَضِيَ اللهُ عَنْهُ ،   فِي قَوْلِهِ :  "  {  فَاسْلُكِي سُبُلَ رَبِّكِ ذُلُلا } ،   قَالَ :  ذليلة لذلك ،   وفي قوله :  {  يَخْرُجُ مِنْ بُطُونِهَا شَرَابٌ مُخْتَلِفٌ أَلْوَانُهُ } ،   قَالَ :  هَذَا العسل {  فِيهِ شِفَاءٌ لِلنَّاسِ } ،   يَقُولُ :  فيه شفاء الأوجاع التي شفاؤها فيه "  .  <br />
عَنْ مُجَاهِدٍ 6 رَضِيَ اللهُ عَنْهُ ،   فِي قَوْلِهِ :  "  {  شَرَابٌ مُخْتَلِفٌ أَلْوَانُهُ فِيهِ شِفَاءٌ لِلنَّاسِ } ،   قَالَ :  هُوَ العسل فيه الشفاء ،   وفي القرآن "  .  <br />
عَن ابْنِ مَسْعُودٍ ،   قَالَ :  " عليكم بالشفاءين :  العسل والقرآن "  .
