قوله تَعَالَى :  {  فَاخْلَعْ نَعْلَيْكَ } آية 12<br />
عَنْ عَلِيٍّ رَضِيَ اللهُ عَنْهُ ،   فِي قَوْلِهِ :  "  {  فَاخْلَعْ نَعْلَيْكَ } ،   قَالَ :  كانتا مِنْ جلد حمار ميت ،   فقيل لَهُ :  اخلعها "  .  <br />
عَنِ الزهري ،   فِي قَوْلِهِ :  "  {  فَاخْلَعْ نَعْلَيْكَ } ،   قَالَ :  كانتا مِنْ جلد حمار أهلي "  .  <br />
عَنْ مُجَاهِدٍ رَضِيَ اللهُ عَنْهُ ،   قَالَ :  " كانت نعلا موسى التي قِيلَ لَهُ :  اخلعها مِنْ جلد الخنزير "  .  <br />
عَنْ عِكْرِمَةَ رَضِيَ اللهُ عَنْهُ ،   فِي قَوْلِهِ :  "  {  فَاخْلَعْ نَعْلَيْكَ } ،   قَالَ :  كي تمس راحة قدميك الأَرْض الطيبة "  .  <br />
قَوْلهُ تَعَالَى :  {  إِنَّكَ بِالْوَادِ الْمُقَدَّسِ }<br />
عَنْ ابْنِ عَبَّاسٍ رَضِيَ اللهُ عَنْهُمَا ،   فِي قَوْلِهِ :  "  {  إِنَّكَ بِالْوَادِ الْمُقَدَّسِ } ،   قَالَ المبارك {  طُوًى } ،   قَالَ :  اسم الوادي "  .  <br />
عَنْ عِكْرِمَةَ رَضِيَ اللهُ عَنْهُ ،   فِي قَوْلِهِ :  "  {  بِالْوَادِ الْمُقَدَّسِ } ،   قَالَ :  الطاهر "  .  <br />
عَنِ الحَسَنِ رَضِيَ اللهُ عَنْهُ ،   فِي قَوْلِهِ :  "  {  بِالْوَادِ الْمُقَدَّسِ } ،   قَالَ :  واد بفلسطين قدس مرتين "  .  <br />
قَوْلهُ تَعَالَى :  {  طُوًى }<br />
عَنْ ابْنِ عَبَّاسٍ رَضِيَ اللهُ عَنْهُمَا ،   فِي قَوْلِهِ :  "  {  بِالْوَادِ الْمُقَدَّسِ طُوًى } ،   يَعْنِي الأَرْض المقدسة ،   وذلك أنه مر بواديها ليلا فطوي ،   يقال طويت وادي كذا وكذا ،   والطاوي مِنَ الليل وارتفع إلي أعلى الوادي :  وذلك نَبِيّ الله موسى عَلَيْهِ السَّلامُ "  .  <br />
عَنِ ابْنِ أَبِي نجيج رَضِيَ اللهُ عَنْهُ ،   فِي قَوْلِهِ :  "  {  طُوًى } ،   قَالَ :  طاالأرض حافيا كما تدخل الكعبة حافيا ،   يَقُولُ :  مِنْ بركة الوادي هَذَا قول سَعِيدُ بْنُ جُبَيْرٍ يَقُولُ :  {  طُوًى } :  اسم الوادي "  .
