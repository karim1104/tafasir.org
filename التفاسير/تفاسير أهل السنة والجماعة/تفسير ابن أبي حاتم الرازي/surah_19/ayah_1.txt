قوله :  {  كهيعص } آية 1<br />
عَنْ ابْنِ عَبَّاسٍ ،   فِي قَوْلِهِ :  "  {  كهيعص } ،   قَالَ :  كبير ،   هاد ،   أمين ،   عزيز صادق "  ،   وفي لفظ :  كاف بدل كبير .  <br />
عَنْ ابْنِ عَبَّاسٍ ،   {  كهيعص } ،   قَالَ :  كاف مِنْ كريم ،   وهاء مِنْ هاد ،   وياء مِنْ حكيم ،   وعين مِنْ عليم ،   وصاد مِنْ صادق "  .  <br />
عَنْ ابْنِ مَسْعُودٍ ،   وناس مِنَ الصحابة  "  {  كهيعص } هُوَ الهجاء المقطع :  الكاف مِنَ الملك ،   والهاء مِنَ الله ،   والياء والعين مِنَ العزيز ،   والصاد مِنَ المصور "  .  <br />
عَنْ ابْنِ عَبَّاسٍ ،   فِي قَوْلِهِ :  "  {  كهيعص } قسم أقسم الله به ،   وهو مِنَ أسماء الله "  .  <br />
عَنْ مُحَمَّد بن كَعْبٍ ،   فِي قَوْلِهِ :  "  {  كهيعص } ،   قَالَ :  الكاف مِنَ الملك ،   والهاء مِنَ الله ،   والعين مِنَ العزيز ،   والصاد مِنَ الصمد "  .  <br />
عَنِ الرَّبِيعِ بن أنس ،   فِي قَوْلِهِ :  "  {  كهيعص } ،   قَالَ :  يا مِنْ يجير ،   ولا يجار عليه "  .
