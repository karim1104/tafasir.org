أي اسْتَغْرِقْ جميعَ الأوقاتِ بالعبادات ،   فإنَّ إخلالَكِ لحظةً من الزمان بِفَرضٍ تؤديه ،   أو نَفْلِ تأتيه حَسْرَةٌ عظيمةٌ وخُسرَانٌ مبينٌ .  <br />
قوله { إِنَّ الْحَسَنَاتِ يُذْهِبْنَ الْسَّيِئَاتِ } الحسنات ما يجود بها الحق ،   والسيئات ما يذنبها العبد ،   فإذا دخلت حسناتُه على قبائح العبد مَحَتْها وأَبْطَلَتْها .  <br />
ويقال حسناتُ القُربة تَذْهَبُ بسيئات الزَّلَّة .  <br />
ويقال حسناتُ الندم تَذْهَبُ بسيئات الجُرْم .  <br />
ويقال ( انسكاب ) العَبْرَةَ تُذْهِبَ سيئاتِ العَثْرَة .  <br />
ويقال حسنات الاستغفار تُذْهِبُ سيئات العصيان .  <br />
ويقال حسنات الاستغفار تُذْهِبُ سيئات الإصرار .  <br />
ويقال حسناتُ العناية تذهب سيئات الجناية .  <br />
ويقال حسنات العفو عن الإخوان تذْهِبُ الحقدَ عليهم .  <br />
ويقال حسنات الكَرَمَ تُذْهِب سيئات الخَدَم .  <br />
ويقال حسنُ الظنِّ يُذْهَبُ سوأتهم بكم .  <br />
ويقال حسنات الفضل من الله تُذْهِبُ سيئاتِ حسبان الطاعة من أنفسكم .  <br />
ويقال حسناتُ الصدق تَذْهَبُ بسيئات الإعجاب .  <br />
ويقال حسناتُ الإخلاص تَذْهَبُ بسيئات الرياء<br />
