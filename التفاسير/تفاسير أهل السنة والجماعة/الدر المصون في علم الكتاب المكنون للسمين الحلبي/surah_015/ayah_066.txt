قوله تعالى : { وَقَضَيْنَآ إِلَيْهِ } ضَمَّن القضاءَ معنى الإِيحاء ،   فلذلك تَعَدَّى تعديتَه ب  "  إلى  "  ،   ومثلُه : { وَقَضَيْنَآ إِلَى بَنِي إِسْرَائِيلَ } [ الإِسراء : 4 ] .  <br />
قوله : { ذَلِكَ الأَمْرَ }  "  ذلك  "  مفعولُ القضاء ،   والإِشارةُ به إلى ما وَعَدَ من إهلاكِ قومِه ،   و  "  الأمرَ  "  : إمَّا بدلٌ منه أو عطفُ بيانٍ له .  <br />
قوله : { أَنَّ دَابِرَ } العامَّةُ على فتحِ  "  أنَّ  "  وفيها أوجهٌ ،   أحدُها : أنها بدلٌ مِنْ  "  ذلك  "  إذا قلنا :  "  الأمر  "  عطفُ بيان .   الثاني : أنها بدل من  "  الأمر  "  سواء قلنا : إنها بيانٌ أو بدل ممَّا قبلَه .   والثالث : أنه على حَذْفِ الجارِّ ،   أي : بأنَّ دابِرَ ،   ففيه الخلافُ المشهور .  <br />
وقرأ زيدُ بن علي بكسرِها ؛ لأنه بمعنى القولِ ،   أو على إضمار القول .   وعَلَّله الشيخُ بأنَّه لمَّا عَلَّق ما هو بمعنى العلم كُسِر .   وفيه النظرُ المتقدم .  <br />
قوله : { مُّصْبِحِينَ } حالٌ من الضمير المستتر في  "  مقطوعٌ  "  وإنما جُمِع حَمْلاً على المعنى ،   وجعله الفراء وأبو عبيد خبراً ل  "  كان  "  مضمرة ،   قالا :  "  وتقديره : إذا كانوا مُصْبِحين نحو :  "  أنتَ ماشياً أحسنُ منك راكباً  "  .   وهو تكلُّفٌ .   و  "  مُصْبِحين  "  داخلين في الصَّباح فهي تامَّةٌ .
