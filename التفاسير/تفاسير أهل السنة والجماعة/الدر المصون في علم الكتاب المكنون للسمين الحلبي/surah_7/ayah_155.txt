قوله تعالى : { وَاخْتَارَ مُوسَى } :  "  اختار  "  يتعدَّى لاثنين إلى أوَّلهما بنفسه وإلى ثانيهما بحرف الجر ،   ويجوز حَذْفُه ،   تقول :  "  اخترت زيداً من الرجال  "  ،   ثم تَتَّسِعُ فتحذف  "  مِنْ  "  فتقول :  "  زيداً الرجال  "  قال :<br />
اخْتَرْتُكَ الناسَ إذ رثَّتْ خلائِقُهُمْ *** واعتلَّ مَنْ كان يُرجى عنده السُّؤلُ<br />
وقال الراعي :<br />
فقلْتُ له اخترها قَلوصاً سمينة *** ونابٌ علينا مثل نابِكَ في الحيا<br />
وقال الفرزدق :<br />
منا الذي اختير الرجالَ سماحةً *** وجوداً إذا هَبَّ الرياح الزعازعُ<br />
وهذا النوعُ مقصورٌ على السماع ،   حَصَره النحاة في ألفاظ ،   وهي : اختار وأَمَر كقوله :<br />
أمرتك الخيرَ فافعلْ ما أُمِرْتَ به *** فقد تَرَكْتُك ذا مالٍ وذا نَشَبِ<br />
واستغفر كقوله :<br />
أستغفرُ اللهَ ذنباً لستُ محصيَه *** ربَّ العبادِ إليه الوجهُ والعملُ<br />
وسمَّى [ نحو : ] سَمَّيْتُ ابني بزيد ،   وإن شئت : زيداً .   و  "  دعا  "  بمعناه قال :<br />
دَعَتْني أمُّ عمرو أخاها ولم أكن *** أخاها ولم أَرْضَعْ لها بلَبانِ<br />
و  "  كَنَى  "  تقول : كَنَيْته بفلان ،   وإن شئت فلاناً .   و  "  صَدَق  "  قال تعالى :<br />
{ وَلَقَدْ صَدَقَكُمُ اللَّهُ وَعْدَهُ } [ آل عمران : 152 ] .   و  "  زوَّج  "  قال تعالى :<br />
{ زَوَّجْنَاكَهَا } [ الأحزاب : 37 ] .   ولم يزد الشيخ عليها .   ومنها أيضاً  "  حدَّث وأنبأ ونبّأ وأخبر وخبَّر  "  إذا لم تُضَمَّن معنى أَعْلَمَ .   قال تعالى : { مَنْ أَنبَأَكَ هَذَا } [ التحريم : 3 ] وقال : { فَلَمَّا نَبَّأَهَا بِهِ } [ التحريم : 3 ] .   وتقول : حَدَّثْتك بكذا ،   وإن شئت : كذا ،   قال :<br />
لَئِنْ كان ما حُدِّثْتُه اليومَ صادقاً *** أَصُمْ في نهارِ القيظِ للشمس باديا<br />
و  "  قومه  "  مفعولٌ ثانٍ على أوَّلهما ،   والتقدير : واختار موسى سبعين رجلاً من قومه .   ونقل أبو البقاء عن بعضهم أن  "  قومَه  "  مفعول أول و  "  سبعين  "  بدل ،   أي : بدل بعض من كل ،   ثم قال :  "  وأرى أن البدلَ جائزٌ على ضعف وأن التقدير : سبعين رجلاً منهم  "  .   قلت : إنما كان ممتنعاً أو ضعيفاً لأنَّ فيه حَذْفَ شيئين : أحدُهما المختار منه ،   فإنه لا بد للاختيار من مختارٍ ومختار منه ،   وعلى البدل إنما ذُكِر المختارُ دونَ المختار منه .   والثاني : أنه لا بد من رابط بين البدل والمبدل منه وهو  "  منهم  "  كما قدَّره أبو البقاء ،   وأيضاً فإن البدلَ في نية الطَّرْح .  <br />
وأصل اختار : اختَيَرَ افتعل من لفظ الخير كاصطفى من الصفوة .   و  "  لميقاتنا  "  متعلقٌ به أي : لأجل ميقاتنا .   ويجوز أن يكونَ معناها الاختصاصَ ،   أي : اختارهم مخصصاً بهم الميقات كقولك : اختير لك هذا .  <br />
قوله : { لَوْ شِئْتَ } مفعولُ المشيئة محذوف أي : لو شِئْتَ إهلاكنا ،   و  "  أهلكتهم  "  جواب لو ،   والأكثر الإِتيانُ باللام في هذا النحو ،   ولذلك لم يَأْتِ مجرداً منها إلا هنا ،   وفي قوله { أَن لَّوْ نَشَآءُ أَصَبْنَاهُمْ } [ الأعراف : 100 ] وفي قوله : { لَوْ نَشَآءُ جَعَلْنَاهُ أُجَاجاً } [ الواقعة : 70 ] .   ومعنى  "  من قبل  "  أي : قبل الاختيار وأَخْذ الرجفة .  <br />
قوله : { وَإِيَّايَ } قد يتعلَّق به مَنْ يرى جواز انفصال الضمير مع القدرة على اتصاله ،   إذ كان يمكنُ أن يُقال : أهلكتنا ،   وهو تعلُّقٌ واهٍ جداً لأن مقصودَه صلى الله عليه وسلم التنصيص على هلاك كلٍ على حِدَتِه تعظيماً للأمر ،   وأيضاً فإن موسى لم يتعاطَ ما يقتضي إهلاكَه بخلاف قومه ،   وإنما قال ذلك تسليماً منه لربِّه ،   فعطَف ضميرَه تنبيهاً على ذلك ،   وقد تقدم لك  فهذه من هذا  في قوله { وَلَقَدْ وَصَّيْنَا الَّذِينَ أُوتُواْ الْكِتَابَ مِن قَبْلِكُمْ وَإِيَّاكُمْ } [ النساء : 131 ] وقوله { يُخْرِجُونَ الرَّسُولَ وَإِيَّاكُمْ }<br />
[ الممتحنة : 1 ] .  <br />
قوله : { أَتُهْلِكُنَا } يجوز فيه أن يكون على بابه ،   أي : أتَعُمُّنا بالإِهلاك أم تخصُّ به السفهاءَ منا ؟/ ويجوز أن يكونَ بمعنى النفي ،   أي : ما تُهْلِك مَنْ لم يُذنب بذنب غيره ،   قاله أبو بكر بن الأنباري ،   قال :  "  وهو كقولك : أتُهين مَنْ يكرمك ؟ وعن المبرد : هو سؤالُ استعطاف .   و  "  منا  "  في محل نصب على الحال من  "  السفهاء  "  ويجوز أن تكون للبيان .  <br />
قوله : { تُضِلُّ بِهَا } يجوز فيها وجهان ،   أحدهما : أن تكون مستأنفةً فلا محلَّ لها .   والثاني : أن تكون حالاً من  "  فتنتك  "  أي : حالَ كونها مُضِلاً بها .   ويجوز أن تكون حالاً من الكاف لأنها مرفوعةٌ تقديراً بالفاعلية ،   ومنعه أبو البقاء قال :  "  لعدم العامل فيها  "  وقد تقدَّم البحث معه فيه غير مرة .
