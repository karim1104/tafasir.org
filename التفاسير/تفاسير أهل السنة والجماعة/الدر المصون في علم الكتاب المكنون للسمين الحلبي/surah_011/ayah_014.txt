قوله تعالى : { أَنَّمَآ أُنزِلِ } :  "  ما  "  يجوز أن تكون كافةً مهيِّئة .   وفي  "  أُنْزِل  "  ضميرٌ يعود على ما يوحَى إليك ،   و  "  بعلمٍ  "  حال أي : ملتبساً بعلمِه ،   ويجوزُ أن تكونَ موصولةً اسميةً أو حرفية اسماً ل  "  إنَّ  "  فالخبرُ الجارُّ تقديرُه : فاعلموا أن تنزيلَه ،   أو أنَّ الذي أُنْزِل ملتبسٌ بعلمٍ .  <br />
وقرأ زيد بن علي  "  نَزَّل  "  بفتح النون والزاي المشددة ،   وفاعل  "  نَزَّل  "  ضميرُ اللَّه تعالى ،   و { وَأَن لاَّ إِلَهَ إِلاَّ هُوَ } نسقٌ على  "  أنَّ  "  قبلها ،   ولكن هذه مخففةٌ فاسمُها محذوفٌ ،   وجملةُ النفي خبرُها .  <br />
قوله : { نُوَفِّ } الجمهورُ على  "  نُوَفِّ  "  بنون العظمة وتشديد الفاء مِنْ وَفَّى يُوَفِّي ،   وطلحة وميمون بياء الغيبة ،   وزيد بن علي كذلك إلا أنه خَفَّفَ الفاءَ مِنْ أوفى يوفي ،   والفاعلُ في هاتَيْن القراءتين ضميرُ اللَّه تعالى .   وقرىء  "  تُوَفَّ  "  بضم التاء وفتح الفاء مشددةً مِنْ وفّى يُوَفِّى مبنياً للمفعول .    "  أعمالُهم  "  بالرفع قائماً مقام الفاعل .   وانجزم  "  نُوَفِّ  "  على هذه القراءاتِ لكونِه جواباً للشرط ،   كما في قوله تعالى  { مَن كَانَ يُرِيدُ حَرْثَ الآخِرَةِ نَزِدْ لَهُ [ فِي حَرْثِهِ ] وَمَن كَانَ يُرِيدُ حَرْثَ الدُّنْيَا نُؤْتِهِ مِنهَا }<br />
[ الشورى : 20 ] .  <br />
وزعم الفراء أن  "  كان  "  زائدة قال :  "  ولذلك جَزَم جوابَه  "  ولعلَّ هذا لا يصح إذ لو كانت زائدةً لكان  "  يريد  "  هو الشرط ،   ولو كان شرطاً لانجزم ،   فكان يُقال : مَنْ كان يُرِدْ  "  .  <br />
وزعم بعضُهم أنه لا يُؤْتى بفعل الشرط ماضياً والجزاء مضارعاً إلا مع  "  كان  "  خاصة ،   ولهذا لم يَجِىءْ في القرآن إلا كذلك ،   وهذا ليس بصحيحٍ لوروده في غير  "  كان  "  قال زهير :<br />
2639  ومَنْ هاب أسبابَ المنايا يَنَلْنَه *** ولو رام أسبابَ السماء بسُلَّمِ<br />
وأما القرآن فجاء من باب الاتفاق أنه كذلك .  <br />
وقرأ الحسن البصري  "  نوفِي  "  بتخفيف الفاء/ وثبوتِ الياء مِنْ أوفى ،   ثم هذه القراءةُ محتمِلَةٌ : لأن يكون الفعل مجزوماً ،   وقُدِّر جزمُه بحذفِ الحركة المقَّدرة كقوله :<br />
2640  ألم يَأْتيك والأنباءُ تُنْمي *** بما لاقَتْ لَبُونُ بني زياد<br />
على أن ذلك قد يأتي في السَّعَةِ نحو : { إِنَّهُ مَن يَتَّقِ } [ يوسف : 90 ] ،   وسيأتي محرَّراً في سُورته ،   ولأن يكون الفعلُ مرفوعاً لوقوع الشرط ماضياً كقوله :<br />
2641  وإنْ شُلَّ رَيْعانُ الجميعِ مخافةً *** نقولُ جِهاراً ويَلْكُمْ لا تُنَفِّروا<br />
وكقول زهير :<br />
2642  وإنْ أتاه خليلٌ يومَ مَسْألةٍ *** يقولُ لا غائبٌ مالي ولا حَرِمُ<br />
وهل الرفعُ لأنه على نيةِ التقديمِ وهو مذهبُ سيبويه أو على نية الفاءِ ،   كما هو مذهب المبرد ؟ خلافٌ مشهور .
