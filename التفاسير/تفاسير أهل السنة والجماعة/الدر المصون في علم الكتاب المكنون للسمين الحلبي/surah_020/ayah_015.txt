قوله : { أَكَادُ أُخْفِيهَا } العامةُ على ضمِّ الهمزةِ مِنْ  "  أُخْفيها  "  .   وفيها تأويلاتٌ ،   أحدُها : أن الهمزةَ في  "  أُخْفيها  "  للسَّلْبِ والإِزالةِ أي : أُزيل خفاءَها نحو : أعجمتُ الكتابَ أي : أزلْتُ عُجْمَتَه .   ثم في ذلك معنيان ،   أحدهما : أنَّ الخفاءَ بمعنى السَّتْر ،   ومتى أزال سَتْرَها فقد أظهرَها .   والمعنى : أنها لتحقُّق وقوعِها وقُرْبِها أكادُ أُظْهِرُها لولا ما تَقْتضيه الحكمةُ من التأخير .   والثاني : أنَّ الخفاءَ هو الظهورُ كما سيأتي .   والمعنى : أزيلُ ظهورَها ،   وإذا أزالَ ظهورَها فقد استترَتْ .   والمعنى : أني لِشدَّةِ أبهامها أكاد أُخْفيها فلا أُظْهِرُها/ البتةَ ،   وإن كان لا بد من إظهارِها ؛ ولذلك يوجدُ في بعض المصاحف كمصحف أُبَيّ : أكاد أُخْفيها مِنْ نفسي فكيف أُظْهِرُكُمْ عليها ؟ وهو على عادةِ العرب في المبالغة في الإِخفاء قال :<br />
أيامَ تَصْحَبُني هندٌ وأُخْبِرُها ***  ما كِدْت أكتُمُه عني من الخبرِ<br />
وكيف يُتَصَوَّرُ كِتْمانُه مِنْ نفسه ؟<br />
والتأويلُ الثاني : أنَّ  "  كاد  "  زائدةٌ .   قاله ابنُ جُبَيْر .   وأنشدَ غيرُه شاهداً عليه قولَ زيدِ الخيل :<br />
سريعٌ إلى الهَيْجاء شاكٍ سلاحُه ***  فما إنْ يكادُ قِرْنُه يتنفَّسُ<br />
وقال آخر :<br />
وألاَّ ألومَ النفسَ فيما أصابني ***  وألا أكادَ بالذي نِلْتُ أَبْجَحُ<br />
ولا حُجَّةَ في شيءٍ منه .  <br />
والتأويل الثالث : أنَّ الكَيْدُوْدَةَ بمعنى الإِرادة ونُسِبت للأخفش وجماعةٍ ،   ولا ينفعُ فيما قصدوه .  <br />
والتأويل الرابع : أنَّ خبرَها محذوفٌ تقديره : أكاد آتي به لقُرْبها .   وأنشدوا قول ضابىء البرجمي :<br />
هَمَمْتُ ولم أفْعَلْ وكِدْتُ وليتني ***  تَرَكْتُ على عثمانَ تَبْكي حلائِلُهْ<br />
أي : وكِدْتُ أفعلُ ،   فالوقفُ على  "  أكادُ  "  ،   والابتداء ب  "  أُخْفيها  "  ،   واستحسنه أبو جعفر .  <br />
وقرأ أبو الدرداء وابنُ جبير والحسنُ ومجاهدٌ وحميدٌ  "  أَخْفيها  "  بفتح الهمزة .   والمعنى : أُظْهرها ،   بالتأويل المتقدم يقال : خَفَيْتُ الشيءَ : أظهَرْتُه ،   وأَخْفَيْتُه : سترته ،   هذا هو المشهور .   وقد نُقِل عن أبي الخطاب أنَّ خَفَيْتُ وأَخْفَيْتُ بمعنىً .   وحُكي عن أبي عبيد أنَّ  "  أَخْفى  "  من الأضدادِ يكون بمعنى أظهر وبمعنى سَتَر ،   وعلى هذا تَتَّحد القراءتان .   ومِنْ مجيءِ خَفَيْتُ بمعنى أظهَرْت قولُ امرىء القيس :<br />
خَفَاهُنَّ مِنْ أَنْفاقِهِنَّ كأنما ***  خفاهُنَّ وَدْقٌ مِنْ عَشِيٍّ مُجَلِّبِ<br />
وقول الآخر :<br />
فإنْ تَدْفِنوا الداءَ لا نَخْفِهِ ***  وإنْ تُوْقِدُوْا الحربَ لا نَقْعُدِ<br />
قوله : { لِتُجْزَى } هذه لامُ كي ،   وليسَتْ بمعنى القسمِ أي : لَتُجْزَيَنَّ كما نقله أبو البقاء عَنْ بعضهم .   وتتعلَّق هذه اللامُ ب  "  اُخْفيها  "  .   وجعلها بعضُهم متعلقةً ب  "  آتيةٌ  "  وهذا لا يَتِمُّ إلاَّ إذا قَدَّرْتَ أنَّ  "  أكاد أُخْفيها  "  معترضةٌ بين المتعلَّقِ والمتعلَّقِ به ،   أمَّا إذا جعلتَها صفةً لآتِيَةٌ فلا يتجه على مذهب البصريين ؛ لأن اسمَ الفاعلِ متى وُصِفَ لم يعملْ ،   فإنْ عَمِل ثم وُصِف جاز .  <br />
وقال أبو البقاء :  "  وقيل ب  "  آتِيَةٌ  "  ،   ولذلك وَقَفَ بعضُهم عليه وَقْفَةً يسيرةً إيذاناً بانفصالِها عن أخفيها  "  .  <br />
قوله : { بِمَا تَسْعَى } متعلقٌ ب  "  تُجْزَى  "  .   و  "  ما  "  يجوز أَنْ تكونَ مصدريةً أو موصولةً اسميةً ،   ولا بدَّ من مضاف أي : تُجْزى بعقابِ سَعْيها أو بعقابِ ما سَعَتْه .  <br />
