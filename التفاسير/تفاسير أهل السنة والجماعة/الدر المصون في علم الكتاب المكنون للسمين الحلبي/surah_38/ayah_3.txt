وأمَّا كسرُ تاءِ  "  لات  "  فعلى أصلِ التقاءِ الساكنين ك جَيْرِ ،   إلاَّ أنه لا تُعْرف تاءُ تأنيثٍ إلاَّ مفتوحةً .  <br />
وقرأ عيسى أيضاً بكسرِ التاءِ فقط ،   ونصبِ  "  حين  "  كالعامَّةِ .   وقرأ أيضاً  "  ولات حينُ  "  بالرفعِ ،    "  مناصَ  "  بالفتح .   وهذه قراءةٌ مشكلةٌ جداً لا تَبْعُدُ عن الغلطِ مِنْ راويها عن عيسى فإنه بمكانةٍ مِنْ العلمِ المانعِ له من مثلِ هذه القراءةِ .   وقد خَرَّجها أبو الفضلِ الرازيُّ في  "  لوامحه  "  على التقديمِ والتأخيرِ ،   وأنَّ  "  حين  "  أُجْرِي مُجْرى قبل وبعد في بنائِه على الضمِّ عند قَطْعِه عن الإِضافة بجامع ما بينه وبينهما مِن الظرفيةِ الزمانيةِ .   و  "  مَناصَ  "  اسمُها مبنيٌّ على الفتح فُصِل بينَه وبينها ب  "  حين  "  المقطوعِ عن الإِضافة .  / والأصلُ : ولاتَ مناص حين كذا ،   ثم حُذِفَ المضافُ إليه  "  حين  "  ،   وبُني على الضم وقَدَّم فاصلاً بين  "  لات  "  واسمِها .   قال :  "  وقد يجوزُ أَنْ يكونَ لذلك معنًى لا أَعْرِفُه  "  .   وقد رُوِي في تاءِ  "  لاتَ  "  الفتحُ والكسرُ والضمُّ .  <br />
وقوله :  "  فنادَوْا  "  لا مفعولَ له ؛ لأنَّ القصدَ : فَعَلوا النداءَ ،   مِنْ غيرِ قصدِ منادى .   وقال الكلبيُّ :  "  كانوا إذا قاتلوا فاضْطُرُّوا نادى بعضُهم لبعضٍ : مناص أي : عليكم بالفرارِ ،   فلَمَّا أتاهم العذابُ قالوا : مناص  "  .   فقال اللَّهُ تعالى لهم : ولات حينَ مناصٍ  "  .   قال القشيريُّ :  "  فعلى هذا يكونُ التقديرُ : فنادَوْا مناص ،   فحُذِف لدلالةِ ما بعده عليه  "  .   قلت : فيكون قد حَذَفَ المنادى وهو بعضاً وما ينادُوْن به ،   وهو مناص ،   أي : نادَوْا بعضَهم بهذا اللفظِ .   وقال الجرجانيُّ :  "  أي : فنادَوْا حين لا مناص أي : ساعةَ لا مَنْجَى ولا فَوْتَ ،   فلمَّا قَدَّم  "  لا  "  وأَخَّر  "  حين  "  اقتضى ذلك الواوَ كما تقتضي الحالُ إذا جُعِل ابتداءً وخبراً مثلَ ما تقول :  "  جاء زيدٌ راكباً  "  ثم تقول : جاء وهو راكبٌ .   ف  "  حين  "  ظرفٌ لقولِه  "  فنادَوْا  "  .   قال الشيخ :  "  وكونُ أصلِ هذه الجملةِ فنادَوْا : حين لا مناص ،   وأنَّ  "  حين  "  ظرفٌ لقولِه :  "  فنادَوْا  "  دعوى أعجميةٌ في نَظْمِ القرآن ،   والمعنى على نظمِه في غايةِ الوضوح  "  .   قلت : الجرجانيُّ لا يَعْني أنَّ حين ظرفٌ ل  "  نادَوْا  "  في التركيبِ الذي عليه القرآن الآن ،   إنما يعني بذلك في أصلِ المعنى والتركيب ،   كما شَبَّه ذلك بقولِك  "  جاء زيدٌ راكباً  "  ثم ب  "  جاء زيدٌ وهو راكبٌ  "  ف  "  راكباً  "  في التركيبِ الأولِ حالٌ ،   وفي الثاني خبرُ مبتدأ ،   كذلك  "  حين  "  كان في الأصل ظرفاً للنداء ،   ثم صار خبرَ  "  لات  "  أو اسمَها على حسبِ الخلافِ المتقدِّم .  <br />
والمناصُ : مَفْعَل مِنْ ناص يَنُوص أي : هَرَبَ فهو مصدرٌ يقال : نَاصه يَنُوصه إذا فاته فهذا متعدٍّ ،   وناصَ يَنُوص أي : تأخَّر .   ومنه ناص عن قِرْنِه أي : تأخَّر عنه جُبْناً .   قاله الفراء ،   وأنشد قولَ امرئ القيس :<br />
                         أمِنْ ذِكْرِ سَلْمى أَنْ نَأَتْكَ تَنُوْصُ *** فتَقْصُرُ عنها حِقْبةً وتَبُوْصُ<br />
قال أبو جعفر النحاس :  "  ناصَ يَنُوص أي : تقدَّم فيكون من الأضداد  "  .   واستناص طلب المَناص .   قال حارثة بن زيد :<br />
                              غَمْرُ الجِراءِ إذا قَصَرْتُ عِنانَه *** بيديْ اسْتَناصَ ورام جَرْيَ المِسْحَلِ<br />
ويقال : ناص إلى كذا ينوص نَوْصاً أي : التجأ إليه .
