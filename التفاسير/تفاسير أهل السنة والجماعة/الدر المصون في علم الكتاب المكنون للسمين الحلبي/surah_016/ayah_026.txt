قوله تعالى : { مِّنَ الْقَوَاعِدِ } :  "  مِنْ  "  لابتداءِ الغاية ،   أي : من ناحيةِ القواعدِ ،   أي : أتى أمرُ الله وعذابُه .  <br />
قوله : { مِن فَوْقِهِمْ } يجوز أن يتعلَّقَ ب  "  خَرَّ  "  وتكون  "  مِنْ  "  لابتداء الغاية ،   ويجوز/ أَنْ يتعلق بمحذوفٍ على أنها حالٌ من  "  السقف  "  وهي حالٌ مؤكِّدة ؛ إذ السقفُ لا يكون تحتهم .   وقال جماعة : ليس قوله  "  مِنْ تحتِهم  "  تأكيداً ؛ لأنَّ العرب تقول :  "  خَرَّ علينا سَقْفٌ ،   ووقع علينا حائط  "  إذا كان يملكه وإنْ لم يَقعْ عليه ،   فجاء بقوله  "  من فوقهم  "  ليُخْرج هذا الذي في كلام العرب ،   أي : عليهم وَقَعَ وكانوا تحته فهلكوا .   وهذا معنىً غيرُ طائلٍ ،   والقولُ بالتأكيد أَنْصَعُ منه .  <br />
والعامَّةُ على  "  بُنْيانِهم  "  .   وفرقة :  "  بِنْيَتَهُمْ  "  .   وفرقة - منهم أبو جعفر-  "  بَيْتهم  "  .   والضحاك  "  بُيوتهم  "  .  <br />
والعامَّةُ أيضاً :  "  السَّقْفُ  "  مفرداً .   وفرقةٌ بفتحِ السين وضمِّ القاف بزنةِ عَضُد ،   وهي لغةٌ في السَّقْف ،   ولعلها مخففةٌ من المضموم ،   وكَثُرَ استعمالُ الفرعِ لخفَّتِه كقول تميم :  "  رَجْل  "  ،   ولا يقولون :  "  رَجُل  "  .   وقرأ الأعرج  "  السُّقُف  "  بضمتين .   وزيدٌ بن علي بضم السين وسكونِ القاف ،   وقد تقدَّم مثلُ ذلك في قراءةِ { وَبِالنُّجْمِ هُمْ يَهْتَدُونَ } [ النحل : 16 ] .
