قوله تعالى : { أَيّاً مَّا تَدْعُواْ } :  "  أيَّاً  "  منصوب ب  "  تَدْعُوا  "  على المفعول به ،   والمضافُ إليه محذوفٌ ،   أي : أيَّ الاسمين .   و  "  تَدْعوا  "  مجزوم بها فهي عاملةٌ معمولةٌ ،   وكذلك الفعلُ ،   والجوابُ الجملةُ الاسمية مِنْ قوله { فَلَهُ الأَسْمَآءَ الْحُسْنَى } .   وقيل : هو محذوفٌ تقديرُه : جاز ،   ثم استأنفَ فقال : فله الأسماءُ الحسنى  "  .   وليس بشيءٍ .  <br />
والتنوين في  "  أَيَّاً  "  عوضٌ من المضافِ إليه .   وفي  "  ما  "  قولان ،   أحدهما : أنها مزيدةٌ للتاكيد .   والثاني : أنها شرطيةٌ جُمِعَ بينهما تأكيداً كما جُمِع بين حَرْفَيْ الجرِّ للتاكيد ،   وحَسَّنه اختلافُ اللفظ كقوله :<br />
فَأَصْبَحْنَ لا يَسْأَلْنني عن بما به *** .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .  <br />

يا شاةَ مَنْ قنَصَ لِمَنْ حَلَّتْ له *** .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .  <br />
واحتُمِل أن تكونَ شرطيةً ،   وجُمِع بينهما تأكيداً لِما تقدم .   و  "  تَدْعُوا  "  هنا يحتمل أن يَكونَ من الدعاء وهو النداء فيتعدَّى لواحدٍ ،   وأن يَكونَ بمعنى التسمية فيتعدَّى لاثنين ،   إلى الأولِ بنفسه ،   وإلى الثاني بحرفِ الجر ،   ثم يُتَّسَعُ في الجارِّ فيُحذف كقوله :<br />
دَعَتْني أخاها أمُّ عمروٍ .   .   .   .   .   .   .   .   .   *** .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .   .  <br />
والتقدير : قل : ادعُوا معبودَكم بالله أو بالرحمن/ بأيِّ الاسمين سَمَّيتموه .   وممَّن ذهب إلى كونها بمعنى  "  سَمَّى  "  الزمخشري .  <br />
ووقف الأخوان على  "  أيّا  "  بإبدال التنوين ألفاً ،   ولم يقفا على  "  ما  "  تبييناً لانفصالِ ،    "  أيَّ  "  مِنْ  "  ما  "  .   ووقف غيرُهما على  "  ما  "  لامتزاجها ب  "  أيّ  "  ،   ولهذا فُصِل بها بين  "  أيّ  "  وبين ما أُضيفت إليه في قوله تعالى<br />
{ أَيَّمَا الأَجَلَيْنِ } [ القصص : 28 ] .   وقيل :  "  ما  "  شرطيةٌ عند مَنْ وقف على  "  أياً  "  وجعل المعنى : أيَّ الاسمينِ دَعَوْتموه به جاز ثم استأنف { مَّا تَدْعُواْ فَلَهُ الأَسْمَآءَ الْحُسْنَى } يعني أنَّ  "  ما  "  شرطٌ ثانٍ ،   و  "  فله الأسماءُ  "  جوابُه ،   وجوابُ الأول مقدِّرٌ .   وهذا مردودٌ بأنَّ  "  ما  "  لا تُطْلق على آحاد أولي العلم ،   وبأنَّ الشرطَ يقتضي عموماً ،   ولا يَصِحُّ هنا ،   وبأن فيه حَذْفَ الشرط والجزاء معاً .
