قوله تعالى : { وَيَسْئَلُونَكَ عَنِ المحيض قُلْ هُوَ أَذًى }[ البقرة :222 ]<br />
قال الطبريُّ عن السُّدِّيِّ : إنَّ السائلَ ثابتُ بْنُ الدَّحْدَاحِ ،   وقال قتادةُ وغيره : إِنما سألوه ،   لأنَّ العرب في المدينةِ وما والاها ،   كانُوا قد استنوا بسُنَّة بني إِسرائيل في تجنُّب مواكلة الحائِضِ ،   ومساكَنَتِها ،   فنزلَتِ الآية .  <br />
وقوله تعالى : { فاعتزلوا النساء فِي المحيض }[ البقرة :222 ]<br />
يريدُ : جماعَهُنَّ ،   بما فَسَّر من ذلك رسولُ اللَّه صلى الله عليه وسلم مِنْ أنْ تشدَّ الحائِضُ إِزارها ،   ثُمَّ شأنُه بأعلاها .  <br />
قال أحمدُ بن نَصْرِ الداوديّ : روي أنَّ رَسُولَ اللَّه صلى الله عليه وسلم قَالَ : ( اتقوا النِّسَاءَ فِي المَحِيضِ ،   فَإِنَّ الجُذَامَ يَكُونُ مِنْ أَوْلاَدِ المَحِيضِ ) ،   انتهى .  <br />
وقوله تعالى : { وَلاَ تَقْرَبُوهُنَّ حتى يَطْهُرْنَ }[ البقرة :222 ] .  <br />
وقرأ حمزة وغيره { يَطَّهَّرْنَ } ،   بتشديد الطاء والهاء ،   وفتحهما ،   وكلُّ واحدة من القراءَتَيْنِ يحتملُ أنْ يراد بها الاِغتسالُ بالماء ،   وأن يراد بها انقطاعُ الدمِ وزوالُ أذاه ،   قال ابنُ العربيُّ في  " أحكامه "  : سمعْتُ أبا بَكْرٍ الشَّاشِيَّ يقولُ : إِذا قيل : لا تَقْرَبْ ،   بفتح الراء ،   كان معناه : لا تَلْتَبِسْ بالفعلِ ،   وإِذا كان بضم الراء ،   كان معناه لا تَدْن منه ،   انتهى .  <br />
وجمهورُ العلماء على أنَّ وطأها في الدَّمِ ذنْبٌ عظيمٌ يتاب منْه ،   ولا كفَّارة فيه بمالٍ ،   وجمهُورهم على أن الطُّهْر الذي يُحِلُّ جماعَ الحائِض ،   هو بالماءِ ،   كطهر الجُنُب ،   ولا يجزئ من ذلك تَيَمُّمٌ ولا غيره .  <br />
وقوله تعالى : { فَإِذَا تَطَهَّرْنَ .   .   .   } [ البقرة :222 ] .  <br />
الخلافُ فيها كما تقدَّم ،   وقال مجاهدٌ وجماعةٌ : { تَطَهَّرْنَ } ،   أي : اغتسلن بالماء بقرينةِ الأمر بالإِتيان ،   لأنَّ صيغة الأمْرِ من اللَّهِ تعالى لا تقعُ إِلا على الوَجْه الأكمل ،   و{ فَأْتُوهُنَّ } : أمر بعد الحَظْر ،   يقتضي الإِباحة ،   والمعنى : من حيثُ أمركم اللَّه باعتزالهن وهو الفَرْج ،   أو من السُّرَّة إِلى الرُّكْبة ،   على الخلاف في ذلك ،   وقال ابن عبَّاس : المعنى : من قِبَلِ الطُّهْرِ ،   لا من قِبَلِ الحَيض ،   وقيل : المعنى مِنْ قِبَلِ حالِ الإِباحة ،   لا صائماتٍ ولا مُحْرِماتٍ ،   ولا غيرَ ذلك ،   و{ التَّوَّابُون } : الرجَّاعون ،   وعُرْفُهُ من الشَّرِّ إِلى الخير ،   و{ المُتَطَهِّرْونَ } قال عطاءٌ وغيره : المعنى بالماء ،   وقال مجاهد وغيره : المعنى من الذنوب .
