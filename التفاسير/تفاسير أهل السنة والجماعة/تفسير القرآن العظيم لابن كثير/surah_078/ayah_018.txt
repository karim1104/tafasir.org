{ يَوْمَ يُنْفَخُ فِي الصُّورِ فَتَأْتُونَ أَفْوَاجًا } قال مجاهد : زُمَرًا{[29651]} .   قال ابن جرير : يعني تأتي كل أمة مع رسولها ،   كقوله : { يَوْمَ نَدْعُوا كُلَّ أُنَاسٍ بِإِمَامِهِمْ } [ الإسراء : 31 ] {[29652]} .  <br />
وقال البخاري : { يَوْمَ يُنْفَخُ فِي الصُّورِ فَتَأْتُونَ أَفْوَاجًا } حدثنا محمد ،   حدثنا أبو معاوية ،   عن الأعمش ،   عن أبي صالح ،   عن أبي هريرة قال : قال رسول الله صلى الله عليه وسلم :  " ما بين النفختين أربعون "  .  <br />
قالوا : أربعون يومًا ؟ قال :  " أبيتُ "  .   قالوا : أربعون شهرًا ؟ قال :  " أبيت "  .   قالوا : أربعون سنة ؟ قال :  " أبيت "  .   قال :  " ثم يُنزلُ الله من السماء ماء فينبتُونَ كما ينبتُ البقلُ ،   ليس من الإنسان شيءٌ إلا يَبلَى ،   إلا عظمًا واحدا ،   وهو عَجْبُ الذنَب ،   ومنه يُرَكَّبُ الخَلْقُ يومَ القيامة " {[29653]} .
