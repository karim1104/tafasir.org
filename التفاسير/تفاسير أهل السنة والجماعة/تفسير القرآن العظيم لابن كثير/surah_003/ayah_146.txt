ثم قال تعالى - مسليًا للمسلمين{[5800]} عما كان وقع في نفوسهم يوم أُحُد - : { وَكَأَيِّنْ مِنْ نَبِيٍّ قَاتَلَ مَعَهُ رِبِّيُّونَ كَثِيرٌ } قيل : معناه : كم من نبي قُتِل وقتل معه ربيون من أصحابه كثير .   وهذا القول هو اختيار ابن جرير ،   فإنه قال : وأما الذين قرؤوا : { قُتِلَ مَعَهُ رِبِّيُّونَ كَثِيرٌ } فإنهم قالوا : إنما عنى بالقتل النبي وبعض من معه من الربيين دون جميعهم ،   وإنما نفي الوهن والضعف عمن بقي من الربيين ممن لم يقتل .  <br />
قال : ومن قرأ { قَاتَلَ } فإنه اختار ذلك لأنه قال : لو قتلوا{[5801]} لم يكن لقوله : { فَمَا وَهَنُوا } وجه معروف ؛ لأنهم يستحيل أن يُوصَفوا بأنهم لم يهنوا ولم يضعفوا بعد ما قتلوا .  <br />
ثم اختار قراءة من قرأ { قُتِلَ مَعَهُ رِبِّيُّونَ كَثِيرٌ } ؛ لأن الله [ تعالى ] {[5802]} عاتب بهذه الآيات والتي{[5803]} قبلها من انهزم يوم أحد ،   وتركوا القتال أو سمعوا الصائح يصيح :  " إن{[5804]} محمدا قد قتل "  .   فعذلهم الله على فرارهم وترْكِهم القتال فقال لهم : { أَفَإِنْ مَاتَ أَوْ قُتِلَ } أيها المؤمنون ارتددتم عن دينكم وانقلبتم على أعقابكم ؟ .  <br />
وقيل : وكم من نبي قتل بين يديه من أصحابه ربيون كثير{[5805]} .  <br />
وكلام ابن إسحاق في السيرة يقتضي قولا آخر ،   [ فإنه ]{[5806]} قال : أي وكأين من نبي أصابه القتل ،   ومعه ربيون ،   أي : جماعات فما وهنوا بعد نبيهم ،   وما ضعفوا عن عدوهم ،   وما استكانوا لما أصابهم في الجهاد عن الله وعن دينهم ،   وذلك الصبر ،   { وَاللَّهُ يُحِبُّ الصَّابِرِينَ } .  <br />
فجعل قوله : { مَعَهُ رِبِّيُّونَ كَثِيرٌ } حالا وقد نصر هذا القول السهيلي وبالغ فيه ،   وله اتجاه لقوله : { فَمَا وَهَنُوا لِمَا أَصَابَهُمْ } الآية ،   وكذلك حكاه الأموي في مغازيه ،   عن كتاب محمد بن إبراهيم ،   ولم يقل {[5807]} غيره .  <br />
وقرأ بعضهم : { قَاتَلَ مَعَهُ رِبِّيُّونَ كَثِيرٌ } قال سفيان الثوري ،   عن عاصم ،   عن زرّ ،   عن ابن<br />
مسعود { رِبِّيُّونَ كَثِيرٌ } أي : ألوف .  <br />
وقال ابن عباس ،   ومجاهد وسعيد بن جُبَير ،   وعِكْرِمة ،   والحسن ،   وقتادة ،   والسُّدِّي ،   والرَّبِيع ،   وعطاء الخراساني : الربيون : الجموع الكثيرة .  <br />
وقال عبد الرزاق ،   عن مَعْمَر عن الحسن : { رِبِّيُّونَ كَثِيرٌ } أي : علماء كثير ،   وعنه أيضًا : علماء صبر أبرار أتقياء .  <br />
وحكى ابن جرير ،   عن بعض نحاة البصرة : أن الربيين هم الذين يعبدون الرب ،   عز وجل ،   قال : ورد بعضهم عليه قال : لو كان كذلك لقيل رَبيون ،   بفتح الراء .  <br />
وقال ابن زيد :  " الربيون : الأتباع ،   والرعية ،   والربابيون :{[5808]} الولاة .  <br />
{ فَمَا وَهَنُوا لِمَا أَصَابَهُمْ فِي سَبِيلِ اللَّهِ وَمَا ضَعُفُوا وَمَا اسْتَكَانُوا } قال قتادة والربيع بن أنس : { وَمَا ضَعُفُوا } بقتل نبيهم { وَمَا اسْتَكَانُوا } يقول : فما ارتدوا عن نصرتهم ولا عن دينهم ،   أنْ قاتلوا على ما قاتل عليه نبي الله حتى لحقوا بالله .  <br />
وقال ابن عباس { وَمَا اسْتَكَانُوا } تَخَشَّعوا .   وقال السُّدِّي وابن زيد : وما ذلوا لعدوهم .  <br />
وقال محمد بن إسحاق ،   وقتادة والسدي : أي ما أصابهم ذلك حين قُتِل نبيهم .   { وَاللَّهُ يُحِبُّ الصَّابِرِينَ }
