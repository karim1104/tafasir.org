قال الله : { ففهمناها سليمان } ( 79 ) . <br />
و( في ){[2636]} تفسير الكلبي أن ( أصحاب ){[2637]} الحرث استعدوا على أصحاب الغنم فنظر داود ثمن الحرث فإذا هو ( قريب ){[2638]} من ثمن الغنم ،  فقضى بالغنم ( الصاحب ){[2639]} الحرث ،  فمروا ( بسليمان ){[2640]} فقال : كيف قضى{[2641]} فيكم نبي الله ؟ فأخبروه فقال : نعم ما قضى ،  وغيره كان أرفق بالفريقين كليهما ،  فدخل أصحاب الغنم على داود فأخبروه ،  فأرسل إلى سليمان فدخل عليه ،  فعزم عليه داود بحث النبوة و( بحق ){[2642]} الملك ،  وحق الوالد لما حدثتني كيف رأيت فيما قضيت ،  فقال سليمان{[2643]} : قد عدل النبي وأحسن ،  وغيره كان ( أوفق ){[2644]} قال : ما هو ؟ قال : تدفع الغنم إلى أهل الحرث فينتفعون بسمنها ،  ولبنها ،  وأصوافها ،  وأولادها عامهم هذا وعلى أهل الغنم أن يزرعوا لأهل الحرث مثل الذي أفسدت غنمهم ،  فإذا كان مثله حين ( أفسدوه ){[2645]} قبضوا غنمهم .  ( قال ){[2646]} له داود : نعم ما قضيت . <br />
[ ا ]{[2647]} سفيان عن أبي إسحاق عن مسروق قال : كان عنبا . <br />
و ]{[2648]} قال الكلبي : وكان الحرث عنبا . <br />
وتفسير مجاهد أن داود أعطى ( صاحب ){[2649]} الحرث [ رقاب ]{[2650]} الغنم بأكلها الحرث .  وحكم سليمان بجزة الغنم وألبانها لأهل الحرث ،  وعلى أهل الحرث رعيتها ،  ويحرث لهم أهل الغنم حتى يكون كهيئته يوم أكل ،  [ ثم ]{[2651]} يدفعونه إلى أهله ويأخذون غنمهم ){[2652]} . <br />
قوله : { وكنا لحكمهم شاهدين } ( 78 ){[2653]} يعني داود وسليمان ،  لقضائهم شاهدين . <br />
{ ففهمناها سليمان } ( 79 )<br />
قال يحيى : كان هذا القضاء يومئذ ،  وقد تكون لأمة شريعة ولأمة أخرى شريعة غيرها ،  وقضاء غير قضاء الأمة الأخرى . <br />
[ وحدثني ]{[2654]} بحر السقاء عن الزهري عن سعيد بن المسيب [ وحرام بن محيصه ]{[2655]} عن البراء بن عازب أن ناقة له وقعت في حائط قوم ( فأفسدت فيه ){[2656]} فاختصموا إلى رسول الله صلى الله عليه وسلم فقال : "ما أجد لكم ( إلا قضاء ){[2657]} سليمان بن داود .  إنه قضى على أهل المواشي حفظ مواشيهم بالليل و( قضى ){[2658]} على أهل الحوائط حفظ حوائطهم بالنهار" . <br />
قال يحيى : إنما في هذا الحديث أنه يضمن ما يكون من الماشية بالليل ،  وليس فيه كيف القضاء في ذلك الفساد ( اليوم ){[2659]} .  وإنما القضاء اليوم في ذلك الفساد ما بلغ ذلك الفساد من النقصان . <br />
[ وحدثني ]{[2660]} عاصم بن حكيم عن إسماعيل بن أبي خالد عن الشعبي أن شاة أكلت غزل حائك قال : فأتوا شريحا : ( قال ){[2661]} فقرأ شريح هذه الآية : { وداود وسليمان إذ يحكمان في الحرث إذ نفشت فيه غنم القوم } [ و ]{[2662]} قال : ( و ){[2663]} النفش لا يكون إلا بالليل ،  إن كان ليلا ضمن ،  وإن كان نهارا لم يضمن . <br />
[ قال : و ]{[2664]} [ حدثني ]{[2665]} حماد بن سلمة عن محمد بن زياد [ وحدثني عثمان عن نعيم بن عبد الله ]{[2666]} عن أبي هريرة [ كلاهما ]{[2667]} قال : قال رسول الله صلى الله عليه سلم : "الدابة العجماء جبار ،  والبئر جبار ،  والمعدن جبار ،  وفي الركاز الخمس" . <br />
قال يحيى : ( هي ){[2668]} عندنا في حديث النبي ( عليه السلام ){[2669]} في ناقة البراء ابن عازب أنه بالنهار ،  وأما إذا أفسدت بالليل فصاحبها ضامن ،  والله أعلم . <br />
قوله : { وكلا آيتنا ( حكما وعلما } ( 79 ) يعني ){[2670]} ( أعطينا حكما وعلما ){[2671]} [ يعني ]{[2672]} و( عقلا ){[2673]}  تفسير السدي ،  يعني ( بذلك ){[2674]} : داود وسليمان . <br />
قال : { وسخرنا مع داود الجبال يسبحن والطير } ( 79 ) كانت جميع الجبال وجميع الطير تسبح مع داود بالغداة والعشي ويفقه تسبيحها . <br />
[ ا ]{[2675]} سعيد عن قتادة في قوله : { يسبحن } قال : يصلين{[2676]} ،  [ يفقه ذلك داود ] . {[2677]}<br />
قوله : { وكنا فاعلين } ( 79 ) أي : قد فعلنا ذلك بداود .
