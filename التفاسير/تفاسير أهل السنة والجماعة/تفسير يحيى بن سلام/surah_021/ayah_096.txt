قوله : { حتى إذا فتحت يأجوج ومأجوج } ( 96 ) [ يعني فلما فتحت يأجوج ومأجوج .  تفسير السدي ]{[2778]} : يموجون في الأرض فيفسدون فيها . <br />
[ ا يونس بن أبي إسحاق عن سعيد بن عمرو بن جعدة عن الزهري قال : قالت أم سلمة : كان رسول الله صلى الله عليه وسلم نائما في بيته فاستقيظ محمرة عيناه{[2779]} ،  فقال : "لا إله إلا الله ثلاثا ،  ويل للعرب من أمر قد اقترب ،  قد فتح اليوم من يأجوج ومأجوج مثل هذا" ،  وعقد يونس بيده تسعين مفرجة شيئا . <br />
حدثني ]{[2780]} أبو أمية عن حميد بن هلال عن أبي الضيف عن كعب ( الأحبار ){[2781]} قال : إن يأجوج ومأجوج ينقرون كل يوم بمنقارهم في السد فيسرعون فيه ،  فإذا أمسوا قالوا : نرجع غدا فنفرغ منه ،  فيصبحون وقد عاد كما كان ،  فإذا أراد الله ( تبارك وتعالى ){[2782]} خروجهم ،  قذف على ألسن بعضهم الاستثناء ( فقالوا ){[2783]} : نرجع غدا إن شاء الله فنفرغ منه ،  فيصبحون وهو كما تركوه ،  ( فينقرونه ) ، {[2784]} فيخرجون على الناس ،  فلا يأتون على شيء إلا أفسدوه ،  فيمر أولهم على البحيرة [ 37 أ ] فيشربون ماءها ،  ويمرّ أوسطهم فيلحسون طينها/ ويمر آخرهم فيقول : قد كان هاهنا ( ماء مرة ){[2785]} فيقهرون الناس ،  ويفر الناس منهم في البرية والجبال ،  فيقولون : قد قهرنا أهل الأرض فهلموا إلى أهل السماء ،  فيرمون ( نبالهم ){[2786]} إلى السماء فترجع تقطر دما ،  فيقولون : قد فرغنا من أهل الأرض وأهل السماء ،  فيبعث الله عليه أضعف خلقه : النغف ( وهي ){[2787]} ( دود ){[2788]} تأخذهم في رقابهم فتقتلهم ،  حتى تنتن الأرض من جيفهم ،  ويرسل الله الطير فتنقل جيفهم إلى البحر ،  ثم يرسل الله ( تبارك وتعالى ){[2789]} السماء فتطهر الأرض . <br />
وفي حديث عبد الرحمان بن يزيد عن عطاء بن يزيد : ويستوقد المسلمون من قسيهم ،  وجعابهم ،  ونشابهم ،  و( أترستهم ){[2790]} سبع سنين . <br />
قال كعب : وتخرج الأرض زهرتها وبركتها ،  ويتراجع الناس ،  حتى إن الرمانة لتشبع السكن ،  قيل وما السكن ؟ قال : أهل البيت .  ( قال ){[2791]} وتكون سلوة من عيش . <br />
فبينما الناس كذلك إذ جاءهم خبر أن ذا السويقتين صاحب الجيش قد غزا البيت ،  فيبعث المسلمون جيشا ،  فلا يصلون إليهم ولا يرجعون إلى أصحابهم حتى يبعث الله ( تبارك وتعالى ){[2792]} ريحا طيبة يمانية من تحت العرش ،  فتكفت روح كل مؤمن ،  ثم لا أجد مثل الساعة إلا كرجل أنتج مهرا فهو ينتظر متى يركبه ،  فمن تكلف من أمر الساعة ما وراء هذا فهو متكلف . <br />
[ ا ]{[2793]} سعيد عن قتادة عن أبي رافع عن أبي هريرة أن رسول الله صلى الله عليه وسلم قال : " إن يأجوج ومأجوج ( يخرقونه ){[2794]} كل يوم حتى إذا كادوا يرون شعاع الشمس قال ( الذين ){[2795]} عليهم ارجعوا ،  ( فستخرقونه ){[2796]} غدا ،  فيعيده الله كأشد ما كان ،  حتى إذا بلغت مدتهم وأراد الله أن يبعثهم على الناس حفروا ،  حتى إذا كادوا يرون شعاع الشمس قال ( الذين ){[2797]} عليهم ارجعوا ( فستخرقونه ){[2798]} إن شاء الله غدا ،  فيعودون إليه وهو كهيئة حين تركوه ( فيخرقونه ){[2799]} فيخرجون على الناس ،  فينشفون المياه ويتحصن الناس منهم في حصونهم ،  فيرمون ( نشابهم ){[2800]}  إلى السماء ( فيرجع ){[2801]} فيها كهيئة ( الدم ){[2802]} فيقولون قهرنا أهل الأرض وعلونا أهل السماء ،  فيبعث الله عليهم نغفا في ( أقفائهم ){[2803]} فيقتلهم بها ،  فقال رسول الله [ صلى الله عليه وسلم ]{[2804]} : والذي نفسي بيده إن دواب الأرض لتسمن و( تشكر ){[2805]} من لحومهم [ شكرا" ]{[2806]} . <br />
[ ا ]{[2807]} سعيد عن قتادة أن أبا سعيد الخدري قال : إن الناس يحجون ،  ويعتمرون ويغرسون بعد خروج يأجوج ومأجوج . <br />
قوله : { وهم من كل حدب ينسلون } ( 96 ) . <br />
[ ا ]{[2808]} سعيد عن قتادة قال : من كل أكمة{[2809]} ومن كل ( نجو ){[2810]} ينسلون يخرجون . <br />
[ ا عاصم بن حكيم أن مجاهدا قال : جمع الناس من كل مكان جاءوا منه يوم القيامة فهو حدب . {[2811]}<br />
ا ]{[2812]} سعيد عن قتادة عن نوف البكالي عن عبد الله بن عمرو قال : إن الله ( تبارك وتعالى ){[2813]} خلق الملائكة ،  والجن ،  والإنس فجزأهم عشرة أجزاء ،  تسعة أجزاء منها الملائكة ،  وجزء واحد الجن والإنس .  وجزأ الملائكة عشرة أجزاء ،  تسعة أجزاء منهم الكروبيون الذين يسبحون الليل والنهار لا يفترون ،  وجزء منهم واحد لرسالته ،  ولخزائنه ،  وما يشاء من أمره{[2814]} .  وجزأ ( الجن والإنس ){[2815]} عشرة أجزاء تسعة أجزاء منهم الجن ،  والإنس جزء واحد ،  فلا يولد من الإنس مولود إلا ولد من الجن تسعة .  وجزأ الإنسان عشرة أجزاء ،  تسعة أجزاء منهم يأجوج ومأجوج ،  وسائرهم بنو آدم . <br />
قال يحيى : يعني ما سوى يأجوج ومأجوج من ولد آدم . <br />
وكان الحسن يقول : الإنس كلهم من عند آخرهم ولد آدم ،  و الجن كلهم من عند آخرهم ولد إبليس . <br />
[ ا ]{[2816]} الحسن بن دينار عن عبد الله بن محمد بن عروة ( عن ){[2817]} رجل من آل مسعود الثقفي قال : حدثني [ أخي أو ]{[2818]} ( ابن أخي أو ){[2819]} ابن عمي  قال : قلت لعبد الله بن عمرو : يأجوج ومأجوج الأذرع هم أم الأشبار ؟ فقال : يا ابن أخي ما أجد من ولد آدم بأعظم منهم ولا أطول ،  ولا يموت الميت منهم حتى يولد له ألف فصاعدا ،  فقلت ما طعامهم ؟ قال : هم في ماء ما شربوا ،  وفي شجر ما هضموا وفي نساء ما نكحوا . <br />
( حدثني ){[2820]} يونس بن أبي إسحاق عن أبيه قال : بلغني أن هؤلاء الترك مما سقط من دون الردم من ولد يأجوج ومأجوج .
