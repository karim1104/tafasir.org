( قوله عز وجل ){[7140]} : { لتنذر قوما }( 6 ) [ يعني ]{[7141]} قريشا . <br />
{ ما أنذر ( آباؤهم ) }{[7142]} ( 6 )<br />
( ا ){[7143]} سعيد عن قتادة قال : قال ( بعضهم ){[7144]} : لتنذر قوما لم ينذر ( آباؤهم ){[7145]} .  وقال بعضهم : ( بالذي ){[7146]} انذر آباؤهم{[7147]} . <br />
قال يحيى [ من قال ]{[7148]} : لم ينذر آباؤهم يعني مثل قوله : { ما أتاهم من نذير من قبلك }{[7149]} يعني قريشا .  ومن قال : ( مثل ){[7150]} الذي انذر آباؤهم فيأخذها من هذه الآية :{ أفلم يتدبروا القول أم جاءهم ما لم يأت آباءهم الأولين }{[7151]}/ يعني من كانوا قبل قريش . <br />
وتفسير السدي : { لتنذر قوما } [ يعني ]{[7152]} لتحذر قوما ما في القرآن من الوعيد ( { ما أنذر آباؤهم } ){[7153]} كما انذر آباؤهم ،  يعني كما حذر آباؤهم .  قال : { فهم غافلون }( 6 ) عما جاءهم به النبي ( صلى الله عليه وسلم ){[7154]} في غفلة من  البعث .
