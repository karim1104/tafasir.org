قوله [ عز وجل ]{[4245]} : { ويوم ينفخ في الصور }( 87 ) والصور قرن . <br />
[ قال ]{[4246]} ابن مجاهد عن أبيه : [ قرن ]{[4247]} كهيئة البوق{[4248]} . <br />
{ ففزع من في السماوات ومن في الأرض إلا من شاء الله } ( 87 ) [ قال ]{[4249]} : وهذه النفخة الأولى . <br />
وتفسير الحسن في قوله { إلا من شاء الله } قال : استثنى [ الله ]{[4250]} طوائف من أهل السماء يموتون بين النفختين . <br />
[ وعن ]{[4251]} [ خالد عن عبد الرحمن بن زياد عن عمارة بن غراب قال : قال رسول الله [ صلى الله عليه وسلم ]{[4252]} : { إلا من شاء الله } ،  الشهداء ،  يقولون : ما أحسن هذا الصوت . <br />
[ { ثم ( نفخ ){[4253]} فيه أخرى فإذا هم قيام ينظرون }{[4254]} قال : يقولون : سبحان الله ما أحسن هذا الصوت ،  كأنه الأذان في الدنيا .  فلم يفزعوا ولم يموتوا إلا الموتة الأولى ]{[4255]} . <br />
[ وعن ]{[4256]} المبارك [ بن فضالة ]{[4257]} عن الحسن قال : قال رسول الله صلى الله عليه [ وسلم ]{[4258]} : "أنا سيد ولد آدم يوم القيامة ،  ( وأنا ){[4259]} أول من تنشق عنه الأرض فأجد موسى متعلقا بالعرش ،  فلا أدري أصعق فيمن صعق أم أجزته الصعقة الأولى" . <br />
[ وعن ]{[4260]} عثمان عن نعيم بن عبد الله عن أبي هريرة قال : قال رسول الله [ صلى الله عليه ]{[4261]} [ وسلم ]{[4262]} : أنا أول من تنشق عنه{[4263]} الأرض ،  فأجد موسى متعلقا بالعرش ،  فلا أدري أصعق فيمن صعق ( أو ){[4264]} أجزته الصعقة الأولى" . <br />
[ وعن ]{[4265]} ( أبي ){[4266]} الأشهب عن الحسن قال : قال رسول الله [ صلى الله عليه وسلم ]{[4267]} "أنا سيد ولد آدم [ يوم القيامة ]{[4268]} ،  أنا أول شافع وأول من تنشق عنه الأرض ،  فأجد ){[4269]} موسى متعلقا بالعرش / فلا أدري أحوسب بالصعقة ( الأولى ){[4270]} أم خرج قبلي" . <br />
قوله : { وكل أتوه داخرين }( 87 ) صاغرين ،  تفسير سعيد عن قتادة{[4271]} .  يعني النفخة الآخرة . <br />
[ وعن ]{[4272]} المبارك [ بن فضالة ]{[4273]} عن الحسن قال : لطم رجل من المسلمين عين رجل من اليهود ،  فشكاه اليهودي إلى{[4274]} ( النبي ){[4275]} [ صلى الله عليه وسلم ]{[4276]} فقال : لم لطمته ؟ قال : قلت : والذي اصطفى محمدا على البشر ،  فقال : ما اصطفى الله محمدا إنما اصطفى [ الله ] موسى .  أثنى رسول الله [ صلى الله عليه وسلم ]{[4277]} على موسى ثم قال : "غير أني سيد ولد آدم يوم القيامة ،  وأول من تنشق عنه الأرض ،  فأخذ موسى متعلقا بالعرش ،  فلا أدري أصعق فيمن صعق ( أو ){[4278]} أجزته الصعقة الأولى" . <br />
[ قوله [ عز وجل ]{[4279]} : { وكل أتوه داخرين }( 87 ) .  قال قتادة : صاغرين  يعني النفخة الآخرة ]{[4280]} . <br />
[ عن أبيه عن ]{[4281]} المبارك [ بن فضالة ]{[4282]} عن الحسن قال : قال رسول الله [ صلى الله عليه وسلم ]{[4283]} : "بين النفختين أربعون{[4284]} .  الأولى يميت الله بها كل حي ،  والأخرى يحيي الله بها كل ميت" . <br />
الحسن بن دينار عن أبي مسعود ( الجزري ){[4285]} عن عكرمة قال : النفخة الأولى من الدنيا ،  والثانية من الآخرة . <br />
[ وعن ]{[4286]} عاصم بن حكيم عن سليمان التيمي عن أسلم العجلي عن ( مراية ){[4287]} العجلي عن عبد الله بن عمرو وقال : النافخان في السماء  ( الدنيا ){[4288]}  ( اثنان ){[4289]} [ رأس ]{[4290]} أحدهما بالمشرق ورجلاه ( في المغرب ){[4291]} ،  ورأس أحدهما بالمغرب ورجلاه بالمشرق . <br />
( قال يحيى ){[4292]} : وبلغني عن الشعبي أنه بلغه أن رجلا ( كان ){[4293]} يقول : إن لله صورين .  فقال : كذب ،  قال الله : { ونفخ في الصور فصعق من في السماوات ومن في الأرض إلا من شاء الله ثم نفخ فيه أخرى } {[4294]} إنما هو صور واحد . <br />
( قال يحيى وبلغني عن ابن مسعود ){[4295]} قال : يقوم ملك بين السماء والأرض فينفخ فيه . <br />
وتفسير سعيد عن قتادة أن المنادي ،  وهو صاحب{[4296]} الصور ،  ينادي من الصخرة من بيت المقدس .
