قوله [ عز وجل ]{[4180]} : { وإذا وقع القول عليهم }( 82 )<br />
قال قتادة : أي ( حق ){[4181]} القول عليهم{[4182]} والقول ،  الغضب . <br />
{ أخرجنا لهم دابة من الأرض تكلمهم }( 82 ) [ وفي بعض القراءة{[4183]} : تحدثهم ]{[4184]} . <br />
{ ( أن ){[4185]} الناس كانوا بآياتنا لا يوقنون }( 82 )<br />
[ عن أبيه عن ]{[4186]} سعيد عن قتادة أن ابن عباس كان يقول : أنها دابة ذات زغب{[4187]} وريش ،  لها أربع قوائم ،  تخرج من بعض أودية تهامة{[4188]} . <br />
[ وعن ]{[4189]} عاصم بن حكيم عن داود بن يزيد الاودي عن الشعبي قال : دابة الأرض ( زباء ){[4190]} ذات وبر ،  ( رباء ){[4191]} تناغي السماء . <br />
[ وعن ]{[4192]} إبراهيم بن محمد ]{[4193]} عن صالح مولى التوأمة قال : سمعت عبد الله بن عمرو يقول : تخرج الدابة من مكة من صخرة  بشعب اجياد . <br />
[ عن  أبيه ]{[4194]} ( قال ){[4195]} ( و ){[4196]} حدثني حماد بن سلمة عن طلحة بن عبد الله بن كريز عن عبد الله بن عمرو قال : إذا خرجت الدابة فزع الناس إلى الصلاة ،  فتأتي الرجل وهو يصلي فتقول : طول ما أنت مطول فوالله لأخطمنك{[4197]} .  قال حماد : ( يومئذ ){[4198]} يعرف المنافق من / المؤمن .  قال عبد الله بن عمرو : ولو أشاء أن أضع قدمي على مكانها الذي تخرج منه لفعلت{[4199]} . <br />
[ وعن ]{[4200]} عاصم بن حكيم عن هشام عن الحسن أن موسى سال ربه أن يريه دابة الأرض .  قال : فخرجت ثلاثة أيام ولياليها لا يرى واحد من طرفيها ،  أو لا يرى طرفاها .  قال : فرأى منظرا كريها فقال : ربّ ردها ،  فرجعت . <br />
[ وعن ]{[4201]} عاصم بن حكيم عن هشام عن قيس بن سعد عن أبي الكفيل قال : كنا جلوسا  عند حذيفة ،  فذكروا الدابة ،  فقال حذيفة : أنها تخرج ثلاث خرجات : مرة  في بعض ( الوادي ){[4202]} ،  ثم ( تكمن ){[4203]} ،  ثم تخرج في بعض القرى حتى تذكر ( ويهريق فيها ){[4204]} الأمراء الدماء .  فبينما{[4205]} الناس على أعظم المساجد ،  وأفضلها ،  وأشرفها يعني المسجد الحرام ،  إذ ترفع الأرض ،  ( فيهرب ){[4206]} الناس وتبقى عصابة من المؤمنين يقولون : ( انه ){[4207]} ( لن ينجنا ){[4208]} من أمر الله شيء فتخرج ،  فتجلو وجوههم فتجعلها على اضوأ كوكب دري في السماء ،  وتتتبع الناس ،  فتجلو( وجه المؤمن ){[4209]} وتخطم الكافر ،  لا يدركها طالب ولا ينجو منها هارب ،  قالوا : وما الناس  يومئذ يا حذيفة ؟ قال جيران في الرباع ،  شركاء في الأموال ،  أصحاب في الأسفار{[4210]} . <br />
[ وعن ]{[4211]} عاصم بن حكيم عن الوليد بن عبد الله الزهري عن{[4212]} عبد الملك ابن المغيرة الثقفي أنه سمع عبد الرحمن بن ( البيلماني قال ){[4213]} : سمعت عبد الله بن عمرو يقول : ( يبيت ){[4214]} الناس يسيرون إلى جمع ،  وتبيت دابة الأرض ( تسْرى ){[4215]} إليهم ،  فيصبحون قد جعلتهم بين رأسها وأذنيها ،  فما من مؤمن إلا تمسحه ولا كافر لا ولا منافق إلا تخطمه{[4216]} ،  وان التوبة لمفتوحة . <br />
[ ا ]{[4217]} سعيد عن قتادة عن العلاء بن زياد أن عبد الله بن عمرو قال : لا تقوم الساعة حتى يجتمع أهل البيت على لإناء الواحد ،  ( يعرفون ){[4218]} ( مؤمنيهم ){[4219]} من ( كفارهم ){[4220]} .  قالوا كيف ذلك ؟ قال : ( تخرج دابة الأرض ){[4221]} فتمسح كل إنسان على مسجده ،  فأما المؤمن فتكون نكتة بيضاء فتفشو في وجهه حتى يبيض لها وجهه ،  وأما الكافر نكتة سوداء فتفشو في وجهه حتى يسود لها وجهه حتى إنهم ليتبايعون في أسواقهم ،  يقول هذا : كيف تبيع هذا يا مؤمن ؟ ويقول هذا : كيف ( تأخذ ){[4222]} هذا يا كافر ؟ فما يرد بعضهم على بعض{[4223]} . <br />
قوله [ عز وجل ]{[4224]} : { تكلمهم أن الناس }( 82 ) [ قال السدي : يعني أهل مكة خاصة ]{[4225]} .  قال يحيى [ وهم ]{[4226]} مشركو أهل مكة .  { كانوا بآياتنا لا يوقنون }( 82 ) تفسير الحسن : تكلمهم بهذا الكلام . <br />
وقال بعضهم : ( تقول ){[4227]} : أن الناس كانوا ( بي ){[4228]} لا يوقنون . <br />
[ وعن ]{[4229]} سعيد عن قتادة قال : في بعض القراءة : تحدثهم { أن الناس كانوا{[4230]} بآياتنا لا يوقنون } لا يؤمنون{[4231]} . <br />
وبعضهم يقرأها : { تكلمهم } أي تسمهم{[4232]} .
