جواب سؤالٍ مقدّرٍ كأنّه قيل: كلّهم مؤمنهم وكافرهم يأتيه فرداً، فقال: انّ المؤمنين يكونون بوصف الحبّ او مع محبّيهم غير منقطعى النّسبة عن اخلاّئهم فانّ كلّ نسبة وخلّة الاّ النّسبة والخلّة فى الله وقد تعدّد الاخبار بأنّ الرّسول (ص) قال لعلىٍّ (ع) يا علىّ، قل اللّهمّ اجعل لى فى قلوب المؤمنين ودّاً، فقال علىّ (ع) ذلك ونزلت الآية، وفى بعض الاخبار ولاية امير المؤمنين (ع) هى الودّ الّذى قال الله تعالى؛ والودّ بتثليث الواو مصدر ودّ من باب علم ومنع او وصف منه والمناسب هو معناه الوصفىّ فانّ المقصود انّا سنجعل لهم محبّاً هو محبوبهم عند الرّجوع الينا، فانّ نورهم يعنى امامهم يسعى حينئذٍ بين ايديهم وبايمانهم وان كان المراد به معناه المصدرىّ فالمقصود هو هذا المعنى، فانّ الحبّ الحقيقىّ هو ملكوت الامام الّذى يظهر على صدر السّالك وهذا يشير الى ما قاله الصّوفيّة من الفكر والحضر والسّكينة وهو ظهور الامام بملكوته على السّالك وانّ السّالك ينبغى ان يكون تمام اهتمامه بظهور الشّيخ عليه وانّه البغية القصوى والقنية العظمى.
