قوله تعالى: { ٱلْيَوْمَ أُحِلَّ لَكُمُ ٱلطَّيّبَـٰتُ وَطَعَامُ ٱلَّذِينَ أُوتُواْ ٱلْكِتَـٰبَ حِلٌّ لَّكُمْ وَطَعَامُكُمْ حِلٌّ لَّهُمْ وَٱلْمُحْصَنَـٰتُ مِنَ ٱلْمُؤْمِنَـٰتِ وَٱلْمُحْصَنَـٰتُ مِنَ ٱلَّذِينَ أُوتُواْ ٱلْكِتَـٰبَ مِن قَبْلِكُمْ إِذَا ءاتَيْتُمُوهُنَّ أُجُورَهُنَّ مُحْصِنِينَ غَيْرَ مُسَٰفِحِينَ وَلاَ مُتَّخِذِيۤ أَخْدَانٍ } [المائدة: 5].  - محمد بن يعقوب: عن محمد بن يحيى، عن أحمد بن محمد، عن محمد بن سنان، عن أبي الجارود، قال سألت أبا جعفر (عليه السلام) عن قول الله عز و جل: { وَطَعَامُ ٱلَّذِينَ أُوتُواْ ٱلْكِتَـٰبَ حِلٌّ لَّكُمْ وَطَعَامُكُمْ حِلٌّ لَّهُمْ } ، فقال (عليه السلام): " الحبوب و البقول ".  - و عنه: عن أبي علي الأشعري، عن محمد بن عبد الجبار، عن محمد بن إسماعيل، عن علي بن النعمان، عن ابن مسكان، عن قتيبة الأعشى، قال: سأل رجل أبا عبد الله (عليه السلام) و أنا عنده فقال له: الغنم يرسل فيها اليهودي و النصراني فتعرض فيها العارضة، فيذبح، أ نأكل ذبيحته؟ فقال أبو عبد الله (عليه السلام): " لا تدخل ثمنها في مالك، و لا تأكلها، فإنما هو الاسم و لا يؤمن عليه إلا مسلم ".  فقال له الرجل: قال الله تعالى: { ٱلْيَوْمَ أُحِلَّ لَكُمُ ٱلطَّيّبَـٰتُ وَطَعَامُ ٱلَّذِينَ أُوتُواْ ٱلْكِتَـٰبَ حِلٌّ لَّكُمْ }؟ فقال له أبو عبد الله (عليه السلام): " كان أبي (صلوات الله عليه) يقول: إنما هي الحبوب و أشباهها ".  و روى هذا الحديث الشيخ في (التهذيب) بإسناده عن الحسين بن سعيد، عن علي بن النعمان، عن ابن مسكان، عن قتيبة، قال: سأل رجل أبا عبد الله (عليه السلام)، مثله.  - و عنه: عن عدة من أصحابنا، عن أحمد بن محمد بن خالد، عن عثمان بن عيسى، عن سماعة، عن أبي عبد الله (عليه السلام)، قال: سألته عن طعام أهل الكتاب و ما يحل منه، قال: " الحبوب ".  - و عنه: عن محمد بن يحيى، عن أحمد بن محمد بن عيسى، عن محمد بن سنان، عن عمار بن مروان عن سماعة، قال: سألت أبا عبد الله (عليه السلام) عن طعام أهل الكتاب و ما يحل منه، فقال: " الحبوب ".  - الشيخ: بإسناده عن الحسين بن سعيد، عن محمد بن خالد، عن ابن أبي عمير، عن هشام بن سالم، عن أبي عبد الله (عليه السلام)، في قول الله تعالى: { وَطَعَامُ ٱلَّذِينَ أُوتُواْ ٱلْكِتَـٰبَ حِلٌّ لَّكُمْ } ، فقال: " العدس و الحمص و غير ذلك ".  - و عنه: بإسناده عن أحمد بن محمد بن عيسى، عن محمد بن عيسى، عن محمد بن سنان، عن عمار بن مروان، عن سماعة، قال: سألت أبا عبد الله (عليه السلام) عن طعام أهل الكتاب ما يحل منه، قال: " الحبوب ". - محمد بن يعقوب: عن علي بن إبراهيم، عن أبيه، عن ابن محبوب، عن علي بن رئاب، عن زرارة ابن أعين، قال سألت أبا جعفر (عليه السلام) عن قول الله عز و جل: { وَٱلْمُحْصَنَـٰتُ مِنَ ٱلَّذِينَ أُوتُواْ ٱلْكِتَـٰبَ مِن قَبْلِكُمْ } ، فقال: «هذه منسوخة بقوله: [الممتحنة: 10] ".  - و عنه: عن محمد بن يحيى، عن أحمد بن محمد، عن ابن فضال، عن الحسن بن الجهم، قال: قال لي أبو الحسن الرضا (عليه السلام): " يا أبا محمد، ما تقول في رجل تزوج نصرانية على مسلمة؟ " قلت: جعلت فداك، و ما قولي بين يديك؟ قال: " لتقولن، فإن ذلك تعلم به قولي ". قلت: لا يجوز تزويج النصرانية على مسلمة، و لا غير مسلمة. قال: " و لم؟ " قلت: لقول الله عز و جل: [البقرة: 221] قال: " فما تقول في هذه الآية: { وَٱلْمُحْصَنَـٰتُ مِنَ ٱلَّذِينَ أُوتُواْ ٱلْكِتَـٰبَ مِن قَبْلِكُمْ }؟ قلت: فقوله: [البقرة: 221] نسخت هذه الآية. فتبسم، ثم سكت.  - العياشي: عن قتيبة الأعشى، قال: سأل الحسن بن المنذر أبا عبد الله (عليه السلام): إن الرجل يبعث في غنمه رجلا أمينا يكون فيها، نصرانيا أو يهوديا، فتقع العارضة فيذبحها و يبيعها؟ فقال أبو عبد الله (عليه السلام): " لا تأكلها، و لا تدخلها في مالك، فإنما هو الاسم، و لا يؤمن عليه إلا المسلم ".  فقال رجل لأبي عبد الله (عليه السلام) و أنا أسمع: فأين قول الله { وَطَعَامُ ٱلَّذِينَ أُوتُواْ ٱلْكِتَـٰبَ حِلٌّ لَّكُمْ }؟ فقال أبو عبد الله (عليه السلام): " كان أبي يقول: إنما ذلك الحبوب و أشباهه ".  - عن هشام بن سالم، عن أبي عبد الله (عليه السلام)، في قول الله تبارك و تعالى: { وَطَعَامُ ٱلَّذِينَ أُوتُواْ ٱلْكِتَـٰبَ حِلٌّ لَّكُمْ } ، قال: «العدس و الحبوب و أشباه ذلك» يعني أهل الكتاب.  - عن ابن سنان، عن أبي عبد الله (عليه السلام)، قال: { وَٱلْمُحْصَنَـٰتُ مِنَ ٱلْمُؤْمِنَـٰتِ }. قال: " هن المسلمات ".  - عن مسعدة بن صدقة، قال: سئل أبو جعفر (عليه السلام) عن قول الله: { وَٱلْمُحْصَنَـٰتُ مِنَ ٱلَّذِينَ أُوتُواْ ٱلْكِتَـٰبَ مِن قَبْلِكُمْ } ، قال: " نسختها [الممتحنة: 10] ".  - عن أبي جميلة، عن أبي عبد الله (عليه السلام) في: { وَٱلْمُحْصَنَـٰتُ مِنَ ٱلَّذِينَ أُوتُواْ ٱلْكِتَـٰبَ مِن قَبْلِكُمْ } ، قال: " هن العفائف ".  - عن العبد الصالح (عليه السلام)، قال: سألناه عن قوله تعالى: { وَٱلْمُحْصَنَـٰتُ مِنَ ٱلَّذِينَ أُوتُواْ ٱلْكِتَـٰبَ مِن قَبْلِكُمْ }  ما هن، و ما معنى إحصانهن؟ قال: " هن العفائف من نسائهم ". قوله تعالى: { وَمَن يَكْفُرْ بِٱلإِيمَٰنِ فَقَدْ حَبِطَ عَمَلُهُ وَهُوَ فِي ٱلآخِرَةِ مِنَ ٱلْخَٰسِرِينَ } [المائدة: 5].  - محمد بن الحسن الصفار: عن عبد الله بن عامر، عن أبي عبد الله البرقي، عن الحسين بن عثمان، عن محمد بن الفضيل، عن أبي حمزة، قال: سألت أبا جعفر (عليه السلام) عن قول الله تبارك و تعالى: { وَمَن يَكْفُرْ بِٱلإِيمَٰنِ فَقَدْ حَبِطَ عَمَلُهُ وَهُوَ فِي ٱلآخِرَةِ مِنَ ٱلْخَٰسِرِينَ } ، قال: " تفسيرها في بطن القرآن: و من يكفر بولاية علي، و علي هو الإيمان ".  - ابن شهر آشوب في (المناقب): عن الباقر (عليه السلام)، و عن زيد بن علي، و ابن الفارسي في (الروضة) عن زيد بن علي في قوله تعالى: { وَمَن يَكْفُرْ بِٱلإِيمَٰنِ فَقَدْ حَبِطَ عَمَلُهُ } ، قال: بولاية علي (عليه السلام).  - العياشي، عن جابر، عن أبي جعفر (عليه السلام)، قال: سألته عن تفسير هذه الآية { وَمَن يَكْفُرْ بِٱلإِيمَٰنِ فَقَدْ حَبِطَ عَمَلُهُ }: " يعني بولاية علي (عليه السلام) { وَهُوَ فِي ٱلآخِرَةِ مِنَ ٱلْخَٰسِرِينَ } ».  - محمد بن يعقوب: عن الحسين بن محمد، عن معلى بن محمد، عن الحسن بن علي، عن حماد ابن عثمان، عن عبيد بن زرارة، قال: سألت أبا عبد الله (عليه السلام) عن قول الله عز و جل: { وَمَن يَكْفُرْ بِٱلإِيمَٰنِ فَقَدْ حَبِطَ عَمَلُهُ } ، قال: " ترك العمل الذي أقر به، [من ذلك] أن يترك الصلاة من غير سقم و لا شغل ".  - و عنه: عن محمد بن يحيى، عن أحمد بن محمد، عن ابن فضال، عن ابن بكير، عن عبيد بن زرارة، قال: سألت أبا عبد الله (عليه السلام)، عن قول الله: { وَمَن يَكْفُرْ بِٱلإِيمَٰنِ فَقَدْ حَبِطَ عَمَلُهُ } ، فقال: " من ترك العمل الذي أقر به ".  قلت: فما موضع ترك العمل حتى يدعه أجمع؟ قال: " منه الذي يدع الصلاة متعمدا، و لا من سكر، و لا من علة ".  - العياشي: عن عبيد بن زرارة، قال: سألت أبا عبد الله (عليه السلام)، عن قول الله عز و جل: { وَمَن يَكْفُرْ بِٱلإِيمَٰنِ فَقَدْ حَبِطَ عَمَلُهُ } ، قال: " ترك العمل الذي أقر به، من ذلك أن يترك الصلاة من غير سقم و لا شغل ".  قال: قلت له: الكبائر من أعظم الذنوب؟ قال: فقال: " نعم ".  قلت: هي أعظم من ترك الصلاة؟ قال: " إذا ترك الصلاة تركا ليس من أمره كان داخلا في واحدة من السبعة ".  - عن أبان بن عبد الرحمن، قال: سمعت أبا عبد الله (عليه السلام) يقول: " أدنى ما يخرج به الرجل من الإسلام أن يرى الرأي بخلاف الحق فيقيم عليه ". قال: { وَمَن يَكْفُرْ بِٱلإِيمَٰنِ فَقَدْ حَبِطَ عَمَلُهُ }. و قال: " الذي يكفر بالإيمان: الذي لا يعمل بما أمر الله به، و لا يرضى به ".  - عن محمد بن مسلم، عن أحدهما (عليهما السلام)، في قول الله: { وَمَن يَكْفُرْ بِٱلإِيمَٰنِ فَقَدْ حَبِطَ عَمَلُهُ }. قال: " هو ترك العمل حتى يدعه أجمع- قال- منه الذي يدع الصلاة متعمدا، لا من شغل، و لا من سكر " يعني: النوم.  - عن هارون بن خارجة، قال: سألت أبا عبد الله (عليه السلام)، عن قول الله: { وَمَن يَكْفُرْ بِٱلإِيمَٰنِ فَقَدْ حَبِطَ عَمَلُهُ وَهُوَ فِي ٱلآخِرَةِ مِنَ ٱلْخَٰسِرِينَ } ، قال: فقال: " من ذلك ما اشتق فيه ".  - علي بن إبراهيم، قال: من آمن ثم أطاع أهل الشرك فقد حبط عمله و كفر بالإيمان.
