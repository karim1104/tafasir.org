قوله : { إِنَّا لاَ نُضِيعُ } : يجوزُ أَنْ يكونَ خبرَ { إِنَّ الَّذِينَ } والرابطُ : إمَّا تَكَرُّرُ الظاهرِ بمعناه ،   وهو قولُ الأخفش .   ومثلُه في الصلة / جائزٌ .   ويجوز ان يكونَ الرابطُ محذوفاً ،   أي : منهم ،   ويجوز أن يكونَ الرابطُ العمومَ ،   ويجوز أن يكونَ الخبرُ قولَه : { أُوْلَئِكَ لَهُمْ جَنَّاتُ } ،   ويكونَ قولُه : { إِنَّا لاَ نُضِيعُ } اعتراضاً .   قال ابن عطية : ونحوُه في الاعتراض قولُه : إنَّ الخليفةَ إنَّ اللهَ أَلْبَسَه *** سِرْبالَ مُلْكٍ به تُزْجى الخواتِيمُ

قال الشيخ :  "  ولا يتعيَّنُ أن يكونَ  "  إنَّ اللهَ ألبسَه  "  اعتراضاً لجوازِ أَنْ يكونَ خبراً عن  "  إنَّ الخليفة  "  .   قلت : وابن عطيةَ لم يَجْعَلْ ذلك متعيِّناً بذلك هو نحوه في أحد الجائزين فيه .   ويجوز أن تكون الجملتان - أعني قولَه { إِنَّا لاَ نُضِيعُ } وقولَه { أُوْلَئِكَ لَهُمْ جَنَّاتُ } - خَبَريْن ل  "  إنَّ  "  عند مَنْ يرى جوازَ ذلك ،   أعني تعدُّدَ الخبر ،   وإنْ لم يكونا في معنى خبرٍ واحد .

وقرأ الثقفيُّ  "  لا نُضَيِّع  "  بالتشديدِ ،   عَدَّاه بالتشديد كما عَدَّاه الجمهورُ بالهمزة .
