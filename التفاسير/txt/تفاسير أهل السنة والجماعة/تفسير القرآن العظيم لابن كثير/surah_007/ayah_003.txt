ثم قال تعالى مخاطبًا للعالم : { اتَّبِعُوا مَا أُنزلَ إِلَيْكُمْ مِنْ رَبِّكُمْ } أي : اقتفوا آثار النبي الأمي الذي جاءكم بكتاب أنزل إليكم من رب كلّ شيء ومليكه ،   { وَلا تَتَّبِعُوا مِنْ دُونِهِ أَوْلِيَاءَ } أي : لا تخرجوا عما جاءكم به الرسول إلى غيره ،   فتكونوا قد عدلتم عن حكم الله إلى حكم غيره .

{ قَلِيلا مَا تَذَكَّرُونَ } كقوله : { وَمَا أَكْثَرُ النَّاسِ وَلَوْ حَرَصْتَ بِمُؤْمِنِينَ } [ يوسف : 103 ] .   وقوله : { وَإِنْ تُطِعْ أَكْثَرَ مَنْ فِي الأرْضِ يُضِلُّوكَ عَنْ سَبِيلِ اللَّهِ } [ الأنعام : 116 ] وقوله : { وَمَا يُؤْمِنُ أَكْثَرُهُمْ بِاللَّهِ إِلا وَهُمْ مُشْرِكُونَ } [ يوسف : 106 ] .
