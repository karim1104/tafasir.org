وقوله : { وَإِخْوَانِهِمْ } أ : ي وإخوان الشياطين من الإنس ،   كقوله : { إِنَّ الْمُبَذِّرِينَ كَانُوا إِخْوَانَ الشَّيَاطِينِ } [ الإسراء : 27 ] وهم أتباعهم والمستمعون لهم القابلون لأوامرهم { يَمُدُّونَهُمْ فِي الْغَيِّ } أي : تساعدهم الشياطين على [ فعل ] المعاصي ،   وتسهلها عليهم وتحسنها لهم .

وقال ابن كثير : المد : الزيادة .   يعني : يزيدونهم في الغي ،   يعني : الجهل والسفه .

{ ثُمَّ لا يُقْصِرُونَ } قيل : معناه إن الشياطين تمد ،   والإنس لا تقصر في أعمالهم بذلك .   كما قال علي بن أبي طلحة ،   عن ابن عباس في قوله : { وَإِخْوَانُهُمْ يَمُدُّونَهُمْ فِي الْغَيِّ ثُمَّ لا يُقْصِرُونَ } قال : لا الإنس يقصرون عما يعملون من السيئات ،   ولا الشياطين تمسك عنهم .

قيل : معناه كما رواه العوفي ،   عن ابن عباس في قوله : { يَمُدُّونَهُمْ فِي الْغَيِّ ثُمَّ لا يُقْصِرُونَ } قال : هم الجن ،   يوحون إلى أوليائهم من الإنس { ثُمَّ لا يُقْصِرُونَ } يقول : لا يسأمون .

وكذا قال السُّدِّي وغيره : يعني إن الشياطين يمدون أولياءهم من الإنس ولا تسأم من إمدادهم في الشر ؛ لأن ذلك طبيعة لهم وسَجِيَّة ،   لا تفتر فيه ولا تبطل عنه ،   كما قال تعالى : { أَلَمْ تَرَ أَنَّا أَرْسَلْنَا الشَّيَاطِينَ عَلَى الْكَافِرِينَ تَؤُزُّهُمْ أَزًّا } [ مريم : 83 ] قال ابن عباس وغيره : تزعجهم إلى المعاصي إزعاجا .
