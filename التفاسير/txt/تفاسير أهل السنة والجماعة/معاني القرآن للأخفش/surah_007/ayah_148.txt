وقال {  مِنْ حُلِيِّهِمْ } ( 148 ) وقال بعضهم {  حِلِيِّهِم } و{  حَلْيِهِم } {  عِجْلاً جَسَداً لَّهُ خُوَارٌ } ( 148 ) وقال بعضهم {  جُوْارٌ } وكلٌّ من لغات العرب .

وأمّا قوله {  مِنْ حُلِيِّهِمْ } بضم الحاء فانه  " فُعول "  وهي جماعة  " الحَلْيِ "  ومن قال {  حِلِيِّهِم } في اللغة الأخرى [ ف ]* لمكان الياء كما قالوا :   " قِسِيّ "  و " عِصِيّ "  .
