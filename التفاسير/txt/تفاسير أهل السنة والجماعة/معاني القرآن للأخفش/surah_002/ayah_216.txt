وقال {  كُتِبَ عَلَيْكُمُ الْقِتَالُ وَهُوَ كُرْهٌ لَّكُمْ } ( 216 ) وقال بعضهم {  حَمَلَتْهُ أُمه كَرْها } وقال بعضهم :  {  كُرْها } وهما لغتان مثل  " الغُسْل "  و " الغَسْل "  ،  و " الضُعْف "  و " الضَّعْف "  إلا أنه قد قال بعضهم انه إذا كان في موضع المصدر كان  " كُرْهاً "  كما تقول :   " لا تقوم إلا كَرْهاً "  وتقول :   " لا تقوم إلا على كُرْهٍ "  وهما سواء مثل  " الرُّهْبِ "  و " الرَّهْبِ "  وقال بعضهم :   " الرَّهَب "  كما قالوا :   " البُخْل "  و " البَخْل "  و " البَخَل "  .  وإنما قال {  كُرْهٌ لَكُم } أي :  ذُو كُرْهٍ وحذف  " ذو "  كما قال {  وَسْألِ الْقَرْيَةَ } .
