وقال {  وَمِنَ النَّاسِ مَن يَشْرِي نَفْسَهُ ابْتِغَاءَ مَرْضَاتِ اللَّهِ } ( 207 ) يقول :   " يَبيعُها "  كما تقول :   " شَرَيْتُ هذا المتاعَ "  أي :  بِعْتُهُ [ 72ء ] و " شَرَيْتُهُ "  "  :  اشْتَرَيتهُ أيضاً ،  يجوز في المعنيين جميعا ،  كما تقول :   " إنَّ الجِلَّ لأَفْضَلُ المَتاعِ "  ،  و " إنَّ  " الجِلَّ لأَرْدَؤُهُ "  ،  وعلى ذلك يجوز مع كثير مثله .  وكذلك  " الجَلَلُ "  يكون العظيمَ ويكون الصغيرَ .  وكذلك  " السَّدَفُ "  يكون الظُلْمةَ والضَوْءَ .  وقال الشاعر :

[ من الرمل وهو الشاهد الثامن والثلاثون بعد المائة ] :

وأَرى أَربْدَ قد فارَقَني *** ومن الآرْزاءِ رُزْء ذُو جَلَل

أي :  عظيم .  وقال الآخر :  [ من الطويل وهو الشاهد التاسع والثلاثون بعد المائة ] :

ألا إنَّما أَبْكي لِيَوْمٍ لقِيتُه *** بِجُرْثُمِ صادٍ كلُّ ما بَعْدَهُ جَلَلَ

أي :  صغير .

وأما قوله {  ابْتِغَاءَ مَرْضَاتِ اللَّهِ } ( 207 ) فان انتصابه على الفعل وهو على  " يَشْرِي "  كأنه قال  " لابتِغاءِ مَرْضاةِ الله "  فلما نزع اللام عمل الفعل .  ومثله {  حَذَرَ الْمَوْتِ } وأشباه هذا كثير .  قال الشاعر :  [ من الطويل وهو الشاهد الأربعون بعد المائة ] :

واغْفِرُ عَوْراءَ الكرِيم ادّخارَه *** وَأُعْرِضُ عَنْ شَتِم اللئيمِ تَكَرُّما

لما حذف اللامَ عمل فيه الفعل .
