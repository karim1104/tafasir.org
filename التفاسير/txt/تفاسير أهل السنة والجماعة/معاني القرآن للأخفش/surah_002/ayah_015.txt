وأما قوله {  وَيَمُدُّهُمْ فِي طُغْيَانِهِمْ يَعْمَهُونَ } ( 15 ) فهو في معنى  " ويَمُدُّ لَهُم "  كما قالت العرب :   " الغلام يلعب الكِعاب "  تريد :   " يلعب ** بالكِعاب "  وذلك أَنهم يقولون  " قد مَدَدْتُ له "  و " أَمْدَدْتُه "  في غير هذا المعنى وهو قوله جل ثناؤه {  وَأَمْدَدْنَاهُم بِفَاكِهَةٍ } وقال {  وَلَوْ جِئْنَا بِمِثْلِهِ مَدَداً } .  وقال بعضهم [ 23ء ] ( مِدادا ) و( مَدّا ) من  " أَمْدَدْناهُم "  وتقول  " مَدّ النهرُ فهو مادّ "  و " أَمَدّ الجُرُح فهو مُمِدّ "  .  وقال يونس :   " ما كان من الشَرّ فهو  " مدَدْت "  وما كان من الخير فهو  " أَمْدَدْت "  .  [ ف ] تقول كما فسرت له فإذا أردت أنك تركته قلت :   " مَدَدْتُ له "  وإذا أردت أنك أعطيته قلت :   " أَمْدَدْتُه "  .
