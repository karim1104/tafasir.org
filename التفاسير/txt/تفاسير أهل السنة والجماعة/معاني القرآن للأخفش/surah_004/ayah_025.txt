وقال {  وَمَن لَّمْ يَسْتَطِعْ مِنكُمْ طَوْلاً أَن يَنكِحَ الْمُحْصَنَاتِ } ( 25 ) على  " ومن لم يجد طولا أن ينكح "  يقول  " إلى أنْ ينكِحَ "  لأن حرف الجر يضمر مع  " أن "  .

وقال {  وَاللَّهُ أَعْلَمُ بِإِيمَانِكُمْ بَعْضُكُمْ مِّن بَعْضٍ } ( 25 ) فرفع {  بَعْضُكُمْ } على الابتداء .

وقال {  بِإِذْنِ أَهْلِهِنَّ } ( 25 ) لأن :   " الأَهْلَ "  جماعة ولكنه قد يجمع فيقال :   " أَهْلُونَ "  كما تقول :   " قَوْمٌ "  و " أقوامٌ "  فتجمع الجماعة [ 95ء ] وقال {  شَغَلَتْنَا أَمْوَالُنَا وَأَهْلُونَا } فجمع .  وقال {  قُواْ أَنفُسَكُمْ وَأَهْلِيكُمْ نَاراً } فهذه الياء ياء جماعة فلذلك سكنت وهكذا نصبها وجرها بإسكان الياء وذهبت النون للإضافة .

وقال {  وَأَن تَصْبِرُواْ خَيْرٌ لَّكُمْ } ( 25 ) يقول :   " والصَبْر خَيْرٌ لكم "  .

وقال {  والله أعلم بإيمانكم بعضكم من بعض } ( 25 ) أي :  الله أعلم بإيمان بعضكم من بعض .
