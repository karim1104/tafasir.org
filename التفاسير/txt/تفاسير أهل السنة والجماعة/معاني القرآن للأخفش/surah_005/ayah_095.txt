وقال :  {  فَجَزَاءٌ مِّثْلُ مَا قَتَلَ مِنَ النَّعَمِ } ( 95 ) أي :  فعليه جزاء مثل ما قتل من النعم .

[ و ] قال {  يَحْكُمُ بِهِ ذَوَا عَدْلٍ مِّنْكُمْ هَدْياً } ( 95 ) انتصب على الحال ،  {  بَالِغَ الْكَعْبَةِ } ( 95 ) من صفته ،  وليس قولك [ 106 ء ] {  بَالِغَ الْكَعْبَةِ } بمعرفة ،  لأن فيه معنى التنوين ،  لأنه إذا قال  " هذا ضاربُ زَيْدٍ "  في لغة من حذف النون ولم يفعل بعد ،  فهو نكرة .  ومثل ذلك {  هذا عَارِضٌ مُّمْطِرُنَا } ففيه بعض التنوين غير أنه لا يوصل إليه من أجل الاسم المضمر .

ثم قال {  أَوْ كَفَّارَةٌ طَعَامُ مَسَاكِينَ } ( 95 ) أيْ :  أوْ عليه كفارةٌ .  رفعٌ منون ،  ثم فسر فقال  " هِيَ طعامُ مساكين "   ،  وقال بعضهم {  كَفّارَةُ طَعَامِ مَساكِينَ } بإضافة الكفارة إليه .

[ و ] قال {  أَو عَدْلُ ذلك صِيَاماً } ( 95 ) يريد :  أَوْ عَلَيْهِ مثلُ ذلك َ من الصيام .  كما تقول :   " عَليْها مثلُها زُبْداً "  .  وقال بعضهم {  أَوْ عِدلُ ذلك صياما } ،  فكسر ،  وهو الوجه ،  لأن  " العِدْلَ "  :  المِثْل .  وأَمَّا  " العَدْل "  فهو المصدر ،  تقول :   " عَدَلْتُ هذا بهذا عَدْلاً حَسَنا "  ،  و " العَدْل "  أَيْضاَ :  المِثْلُ .  وقال {  وَلاَ يُقْبَلُ مِنْهَا عَدْلٌ } أي :  مِثْلٌ  ،  ففرقوا بين ذا وبين  " عدل المتاع "   ،  كما تقول :   " امرأَةٌ رَزانٌ "  و " حَجَرٌ رَزِينٌ "  .
