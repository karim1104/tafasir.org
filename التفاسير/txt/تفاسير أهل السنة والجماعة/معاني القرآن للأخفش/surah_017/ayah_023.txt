وقال {  فَلاَ تَقُل لَّهُمَا أُفٍّ } ( 23 ) قد قرئت {  أُفِّ } و{  أُفّاً } لغة جعلوها مثل {  تَعْساً } وَقرأ بعضهم {  أُفَّ } وذلك أن بعض العرب يقول  " أُفَّ لَكَ "  على الحكاية :  أي لا تَقُلْ لهما هذا القول ،  والرفعُ قبيح لأنَّه لم يجيء بعده باللام ،  والذين قالوا {  أُفِّ } فسكروا كثير وهو أجود .  وكسر بعضهم ونّون .  وقال بعضهم {  أُفِّي } كأنه أضاف هذا القول إلى نفسه فقال :   " أُفّي هذا لكما "  والمكسور هنا منون ،  وغير منون على انه اسم متمكن نحو  " أَمْسِ "  وما أشبهه .  والمفتوح بغير نون كذلك .

وقال {  وَلاَ تَنْهَرْهُمَا } ( 23 ) لأنه يقول :   " نَهَرَه "   " يَنْهَرَه "  واِنْتَهَرُه "   " يَنْتَهِرُهُ "  .
