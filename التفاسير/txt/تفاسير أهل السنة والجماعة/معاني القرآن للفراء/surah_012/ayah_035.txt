وقوله :  {  ثُمَّ بَدَا لَهُمْ مِّن بَعْدِ ما رَأَوُاْ الآيَاتِ 35 } ،  آيات البراءة قَدّ القميص من دبر ،  {  لَيَسْجُنُنَّهُ حَتَّى حِينٍ } ،  فهذه اللام في اليمين وفي كل ما ضارع القول .  وقد ذكرناه .  أَلا ترى قوله :  {  وَظَنُّوا ما لَهُمْ مِنْ مَحِيصٍ } ،  {  وَلَقَدْ عَلِمُوا لَمن اشْتَرَاهُ } ،  دخلت هذه اللام و ( ما ) مع الظنّ ( والعلم ) ؛  لأنهما في معنى القول واليمين .

حدّثنا الفراء قال :  حدّثنا ابن الغَسيل الأنصارىّ ،  عن عِكرمة ،  قال :  الْحِين حينان :  حين لا يدرك ،  وهو قوله عزّ وجلّ :  {  هَلْ أَتَى على الإنْسَانِ حِينٌ مِنَ الدَّهْرِ } ،  ( قال الفرّاء :  فهذا يقلّ ويكثر ) ،  ليست له غاية .  قال عكرمة :  وحينٌ يدرَك ،  وهو قوله :  {  تُؤْتِى أُكُلَها كُلَّ حِينٍ } ،  يعنى :  ستّة أشهر .
