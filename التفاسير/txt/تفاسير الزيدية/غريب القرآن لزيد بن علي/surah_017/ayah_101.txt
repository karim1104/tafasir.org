وقوله تعالى :  {  وَلَقَدْ آتَيْنَا مُوسَى تِسْعَ آيَاتٍ بَيِّنَاتٍ } قال :  الإِمامُ أَبو الحُسين زيد بن علي عليهما السَّلامُ :  وهي الطُّوفانُ ،  والمَوتُ والجَرادُ ،  والقمَّلُ ،  والضَّفَادعُ ،  والدَّمُ ،  ولِسَانُهُ وعَصاهُ والبَحرُ .  ويقالُ الطُّوفانُ ،  والجَرادُ ،  والقُمّلُ ،  والضَّفادعُ ،  والدَّمُ ،  والعَصا ،  والسِنينُ ونَقصٍ منَ الثَّمراتِ ،  وَيَدُّهُ .  ويقالُ [ الطُّوفان ] والجَرادُ ،  والقُمّلُ ،  والضَّفادعُ ،  والدَّمُ ،  وعَصَاه ،  ويَدُّه ،  قال الإِمام زيد بن علي عليهما السلام :  وكانتْ عَصا مُوسى عَليهِ السَّلامُ مِن عَوسجٍ ،  وَلَم تُسَخرْ لأحدٍ بَعدَهُ .  وكان اسمُها مَاسا .
