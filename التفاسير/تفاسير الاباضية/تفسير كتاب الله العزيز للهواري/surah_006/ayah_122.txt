قوله :  {  أَوَمَن كَانَ مَيْتاً } أي كافراً {  فَأَحْيَيْنَاهُ } بالإِسلام في تفسير الحسن .  <br />
 وقال مجاهد :  ضالاًّ فهديناه .   وهو واحد .  <br />
 {  وَجَعَلْنَا لَهُ نُوراً يَمْشِي بِهِ فِي النَّاسِ }  ،  <br />
قال مجاهد :  يعني الهدى {  كَمَن مَّثَلُهُ فِي الظُّلُمَاتِ } أي ظلمات الكفر في تفسير الحسن .  <br />
 وقال مجاهد :  في ظلمات الضلالة ،   وهو واحد .   <br />
{  لَيْسَ بِخَارِجٍ مِّنْهَا }  ،  أي :  هو مقيم في ظلمات الكفر والضلالة .  <br />
وقال بعضهم :  {  فَأَحْيَيْنَاهُ }  ،  هذا المؤمن معه من الله بيّنة ،   عليها يعمل ،   وبها يأخذ ،   [ وإليها ينتهي ]{[1100]} .   قال :  {  كَمَن مَّثَلُهُ فِي الظُّلُمَاتِ لَيْسَ بِخَارِجٍ مِّنْهَا } ؛  هذا الكافر في الضلالة متحيّر فيها .   <br />
قال  : هل يستويان مثلاً ،   أي هل يستويان هذان ؟  على الاستفهام ،   أي  : إنهما لا يستويان .  <br />
[ قال بعضهم ]{[1101]} :  بلغنا أَنَّها نزلت في عمر بن الخطاب وأبي جهل بن هشام ،   ثم هي عامة بعدُ .  <br />
ذكر الحسن قال :  قال رسول الله صلى الله عليه وسلم :   "  اللهم أيد الإِسلام{[1102]} بعمر بن الخطاب أو بأبي جهل بن هشام{[1103]}  "  فأيده بعمر بن الخطاب ،   وأحسبه قال :  وأهلك أبا جهل بن هشام أو كما قال .  <br />
قال :  {  كَذَلِكَ زُيِّنَ لِلْكَافِرِينَ مَا كَانُوا يَعْمَلُونَ } .
