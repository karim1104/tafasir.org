قال تعالى :  {  وَلَقَدْ زَيَّنَّا السَّمَآءَ الدُّنْيَا بِمَصَابِيحَ } وهي الكواكب {  وَجَعَلْنَاهَا } أي :  الكواكب {  رُجُوماً لِلشَّيَاطِينِ } يعني ما جعل منها رجوماً .   وكان الوقت الذي جعلت فيه رجوماً حين{[4037]} بعث النبي عليه السلام .  <br />
ذكروا عن أبي رجاء العطاردي قال :  كنا قبل أن يبعث محمد صلى الله عليه وسلم ما نرى نجماً يرمى به ،   فبينما نحن ذات ليلة إذا النجوم قد رمي بها .   فقلنا ما هذا ؟  إن هذا إلا أمر حدث .   فجاءنا أن النبي عليه السلام قد بعث ،   فأنزل الله في هذه الآية ،   وفي سورة الجن : {  وَأَنَّا كُنَّا نَقْعُدُ مِنْهَا مَقَاعِدَ لِلسَّمْعِ فَمَن يَسْتَمِعِ الآنَ يَجِدْ لَهُ شِهَاباً رَّصَداً } [ الجن : 9 ] .  <br />
ذكروا عن حسان بن أبي بلال قال :  من قال في النجوم سوى هذه الأشياء الثلاثة فهو كاذب ،   ثم مفتن مبتدع .   قال تعالى :  {  وَلَقَدْ زَيَّنَّا السَّمَآءَ الدُّنْيَا بِمَصَابِيحَ وَجَعَلْنَاهَا رُجُوماً لِّلشَّيَاطِينِ } .   وقال : {  وَهُوَ الَّذِي جَعَلَ لَكُمُ النُّجُومَ لِتَهْتَدُواْ بِهَا فِي ظُلُمَاتِ الْبَرِّ وَالْبَحْرِ } [ الأنعام : 97 ] فهي مصابيح ورجوم ويهتدى بها{[4038]} .  <br />
ذكروا عن سعيد بن جبير عن ابن عباس قال :  إذا رأيتم الكواكب قد رمي بها فتواروا فإنها تحرق ولا تقتل .   وفي تفسير الحسن :  إنه يقتلهم في أسرع من طرف{[4039]} .  <br />
قال تعالى :  {  وَأَعْتَدْنَا لَهُمْ } أي :  وأعددنا لهم {  عَذَابَ السَّعِيرِ } أي :  في الآخرة ،   أي :  للذين يرجمون من الشياطين ولجماعة الشياطين .   تفسير الحسن :  الذين هم يسترقون السمع ،   يسترق أحدهم السمع وهو يعلم أنه محترق وأن له في الآخرة عذاب السعير .   والكلبي يقول :  هم شرار إبليس .   وقال الحسن :  الشيطان والعفريت والمارد لا يكون إلا الكافر من الجن .
