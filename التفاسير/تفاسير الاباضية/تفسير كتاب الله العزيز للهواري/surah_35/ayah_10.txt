قوله تعالى :  {  مَن كَانَ يُرِيدُ العِزَّةَ فَلِلَّهِ العِزَّةُ جَمِيعاً } قال بعضهم :  من كان يريد العزَّة فليتعزَّر بطاعة الله .  <br />
وتفسير الحسن :  أن المشركين عبدوا الأوثان لتعزّهم ،   كقوله : {  وَاتَّخَذُوا مِن دُونِ اللهِ آلِهَةً لِّيَكُونُوا لَهُمْ عِزّاً }<br />
[ مريم :  81 ] قال :  من كان يريد العزة فليعبد الله حتى يعزّه .  <br />
قوله تعالى :  {  إِلَيْهِ يَصْعَدُ الْكَلِمُ الطَّيِّبُ } هو التوحيد {  وَالْعَمَلُ الصَّالِحُ يَرْفَعُهُ } أي :  يرفعه التوحيد .   أي :  لا يرتفع العمل الصالح إلا بالتوحيد ،   ولا التوحيد إلا بالعمل الصالح ؛  كقوله :  {  وَسَعَى لَهَا سَعْيَهَا وَهُوَ مُؤْمِنٌ } والإِيمان قول وعمل ،   لا ينفع القول دون العمل{[3089]} .   {  فَأُوْلَئِكَ } أي :  الذين هذه صفتهم {  كَانَ سَعْيُهُم مَّشْكُورًا } [ الإِسراء :  19 ] أي :  كان عملهم مقبولاً ،   يعني من وحد الله وعمل بفرائضه .  <br />
قال :  {  وَالَّذِينَ يَمْكُرُونَ السَّيِّئَاتِ } أي :  يعملون الشرك والنفاق {  لَهُمْ عَذَابٌ شَدِيدٌ } أي :  عذاب جهنم .   {  وَمَكْرُ أُوْلَئِكَ هُوَ يَبُورُ } أي :  وعمل أولئك هو يبور ؛  أي :  هو يفسد عند الله ،   أي :  لا يقبل الله الشرك والنفاق ،   ولا يقبل العمل من المشرك والمنافق ،   أي :  لا يقبل العمل إلا من المؤمنين ،   وقال في آية أخرى :  {  إِنَّمَا يَتَقَبَّلُ اللهُ مِنَ المُتَّقِينَ } [ المائدة :  27 ] .
