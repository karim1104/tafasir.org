قوله :  {  وَالْوَالِدَاتُ } [ يعني المطلقات في تفسير مجاهد ]{[380]} {  يُرْضِعْنَ أَوْلاَدَهُنَّ حَوْلَيْنِ كَامِلَيْنِ لِمَنْ أَرَادَ أَن يُتِمَّ الرَّضَاعَةَ } .   ذكر بعض المفسرين قال :  أنزل الله في أول هذه الآية :  {  حَوْلَيْنِ كَامِلَيْنِ } ثم أنزل اليسر والتخفيف فقال :  {  لِمَنْ أَرَادَ أَن يُتِمَّ الرَّضَاعَةَ } .  <br />
ذكروا عن ابن عمر وابن عباس أنهما كانا لا يريان الرضاع بعد الحولين شيئاً .  <br />
قوله :  {  وَعَلَى المَوْلُودِ لَهُ } يعني الأب {  رِزْقُهُنَّ وَكِسْوَتُهُنَّ بِالْمَعْرُوفِ } أي كلٌّ على قدر ميسرته .   {  لاَ تُكَلِّفُ نَفْسٌ إِلاَّ وُسْعَهَا } .   كقوله :  {  لاَ يُكَلِّفُ اللهُ نَفْساً إِلاَّ مَا ءَاتاهَا } [ الطلاق : 7 ] أي ما أعطاها .   وكقوله :  {  لِيُنفِقْ ذُو سَعَةٍ مِّن سَعَتِهِ }[ الطلاق : 7 ] .  <br />
قوله :  {  لاَ تُضَارَّ وَالِدَةٌ بِوَلَدِهَا وَلاَ مَوْلُودٌ لَهُ بِوَلَدِهِ } ذكر بعض المفسّرين قال :  نهى الله الوالد عن الضرار :  أن ينزعه من أمه إذا رضيت أن ترضعه بما كان مسترضعاً به غيرَها ويدفعه إلى غيرها .   ونهيت الوالدة أن تضارّ بولدها فتدفعه إلى زوجها إضراراً ،   إذا أعطاها ما كان مسترضعاً به غيرَها .  <br />
قوله :  {  وَعَلَى الوَارِثِ مِثْلُ ذَلِكَ } [ تفسير قتادة .   قال :  على وارث المولود إن كان المولود لا مال له {  مِثْلُ ذَلِكَ } أي :  مثل الذي كان على والده لو كان حيّاً من أجر الرضاع ]{[381]} .   وقال الحسن :  على الرجال دون النساء .   والوارث :  وارث الصبي إذا مات والد الصبي{[382]} وبقي وارثه ،   فعليه يكون وليس على الأم منه شيء ،   ولا على الأخوة من الأم ،   وذلك في النفقة والضرار{[383]} .   [ وقال ابن عباس :  {  وَعَلَى الوَارِثِ مِثْلُ ذَلِكَ } هو في الضرار ] .  <br />
قوله :  {  فَإِنْ أَرَادَا فِصَالاً } [ يعني فِطاماً ]{[384]} {  عَن تَرَاضٍ مِّنْهُمَا وَتَشَاوُرٍ } قبل انقضاء الحولين ،   بعد أن يستطيع الطعام ولا تدخل عليه ضرورة فيه .  <br />
قال بعض المفسّرين :  {  عَن تَرَاضٍ مِّنْهُمَا وَتَشَاوُرٍ } إذا كان ذلك عن رضى منهما ومشورة .   {  فَلاَ جُنَاحَ عَلَيْهِمَا } .  <br />
قال :  {  وَإِنْ أَرَدتُمْ أَن تَسْتَرْضِعُوا أَوْلاَدَكُمْ } أي لأولادكم {  فَلاَ جُنَاحَ عَلَيْكُمْ إِذَا سَلَّمْتُم مَّا ءَاتَيْتُم بِالمَعْرُوفِ وَاتَّقُوا اللهَ وَاعْلَمُوا أَنَّ اللهَ بِمَا تَعْمَلُونَ بَصِيرٌ } يقول :  إن تراضيا إن يسترضعاه {  فَلاَ جُنَاحَ عَلَيْكُمْ إِذَا سَلَّمْتُم مَّا ءَاتَيْتُم بِالمَعْرُوفِ } [ تفسير مجاهد :  حساب ما رضع الصبي إذا تراضيا أن يسترضعا له ،   إذا خافا الضيعة عليه{[385]} ] .
