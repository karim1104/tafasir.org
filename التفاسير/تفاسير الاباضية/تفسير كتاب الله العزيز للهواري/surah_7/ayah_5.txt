قال :  {  فَمَا كَانَ دَعْوَاهُمْ } أي :  [ قولهم ]{[1147]} إذ جاءهم بأسنا ،   أي عذابنا {  إِلاَّ أَن قَالُوا إِنَّا كُنَّا ظَالِمِينَ } .   قال بعضهم :  لم يكن لهم هجّيرى{[1148]} حين جاءهم العذاب إلا أن قالوا إنا كنا ظالمين .   كقوله في سورة الأنبياء :  {  وَكَمْ قَصَمْنَا مِن قَرْيَةٍ كَانَتْ ظَالِمَةً } ،   أي مشركة ،   يعني أهلها ،   إلى قوله .   .   .   :  {  يَا وَيْلَنَا إِنَّا كُنَّا ظَالِمِينَ فَمَا زَالَت تِّلْكَ دَعْوَاهُمْ }<br />
[ سورة الأنبياء : 10-15 ] أي لم يكن لهم هجِّيري ( إِلاَّ أَن قَالُوا :  يَا وَيْلَنَا إِنَّا كُنَّا ظَالِمِينَ ) .   قال :  الهجّيري :  الذي يلزمه الرجل يردده .  <br />
ذكروا عن الحسن أنه كانت هجيراه سبحان الله العظيم ،   سبحان الله وبحمده .   وكانت هجّيري بعضهم :  {  أَلاَ إِلَى اللهِ تَصِيرُ الأُمُورُ } .   يعني بهجِّيراهم أنه لم يكن لهم قول حين جاءهم العذاب {  إِلاَّ أَن قَالُوا إِنَّا كُنَّا ظَالِمِينَ } .  <br />
قال الحسن :  فَمَا كَان دَعْوَايهُمْ ،   أي ما دعوا به حين جاءهم العذاب إلا أن قالوا :  يا ويلنا إنا كنا ظالمين .   فاقرّوا بالظلم على أنفسهم ونادوا بالتوبة حين لم تنفعهم التوبة .   قال :  فهي مثل قوله :  {  فَنَادَوْا وَلاَتَ حِينَ مَنَاصٍ } [ سورة ص : 3 ] أي ليس حين نزو{[1149]} ولا فرار .   وكقوله :  {  وَقَالُوا ءَامَنَّا بِهِ وَأَنَّى لَهُمُ التَّنَاوُشُ مِن مَّكَانٍ بَعِيدٍ وَقَدْ كَفَرُوا بِهِ مِن قَبْلُ } [ سورة سبأ : 52-53 ] ،   وأنّى لهم التَّناوُش ،   أي وأني لهم الرّدّ إلى الدنيا ،   أي فيؤمنوا ،   فهذا تفسير ابن عباس .   وقال الحسن :  التَّناوُش :  الإِيمان ،   أي :  وكيف لهم بالإِيمان وقد جاءهم العذاب .
