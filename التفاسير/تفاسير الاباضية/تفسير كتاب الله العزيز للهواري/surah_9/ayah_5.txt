قوله :  {  فَإِذَا انسَلَخَ الأَشْهُرُ الحُرُمُ فَاقْتُلُوا المُشْرِكِينَ حَيْثُ وَجَدتُّمُوهُمْ } ذكروا عن بعضهم قال :  إنه ذكر في أول السورة أهل العهد فقال :  {  فَسِيحُوا فِي الأَرْضِ } من يوم النحر خمسين ليلة إلى انسلاخ المحرم لمن لا عهد له .   فأمر الله نبيه إذا مضى هذا الأجل في المشركين ممن لم يكن له عهد فقال :  {  فَاقْتُلُوا المُشْرِكِينَ حَيْثُ وَجَدتُّمُوهُمْ } .  <br />
قال :  {  وَخُذُوهُمْ وَاحْصُرُوهُمْ وَاقْعُدُوا لَهُمْ كُلَّ مَرْصَدٍ } فأمر بقتالهم في الحلّ والحرم وعند البيت حتى يشهدوا أن لا إله إلا الله وأن محمداً رسول الله .   وأمره في أهل العهد أن يتم لهم عهدهم أربعة أشهر بعد يوم النحر إلى عشر يمضين من ربيع الآخر ،   ثم يُقتَلون حيث وَجَدَهم .  <br />
قال الحسن :  {  فَإِذَا انسَلَخَ الأَشْهُرُ الحُرُمُ } الأشهر الحرم في هذا الموضع الأشهر التي أُجِّلُوا فيها والتي كانوا يَحْرُمون فيها على المسلمين لأنهم في عهد منها{[1394]} ،   آخرها عشر ليالٍ يمضين من شهر ربيع الآخر ؛  وسمّاها حرماً لأنه نهى عن قتالهم فيها وحرّمه .   وقول الكلبي مثل القول الأول ،   لهم خمسون ليلة إلى انسلاخ المحرّم ثم يقتلون حيث وُجِدوا .  <br />
قال :  {  فَإِن تَابُوا } أي من الشرك {  وَأَقَامُوا الصَّلاَةَ وَءَاتَوُا الزَّكَاةَ } أي وأقروا بالزكاة ،   لأن من المسلمين من لا تجب عليهم الزكاة {  فَخَلُّوا سَبِيلَهُمْ إِنَّ اللهَ غَفُورٌ رَّحِيمٌ } أي يغفر لهم الكفر إذا آمنوا .   كقوله :  {  قُل لِّلَّذِينَ كَفَرُوا إِن يَنتَهُوا يُغْفَرْ لَهُم مَّا قَدْ سَلَفَ } [ الأنفال : 38 ] .  <br />
وقال بعضهم في قوله :  {  إِلاَّ الَّذِينَ عَاهَدتُّم مِّنَ المُشْرِكِينَ } .   .   إلى آخر الآية ،   قال :  هم مشركو قريش الذين عاهدهم نبي الله زمان الحديبية ،   وكان عهدهم أن لا إغلال ولا أسلال{[1395]} .   فغلّوا نبي الله ونكثوا العهد وظاهروا المشركين على المسلمين .
