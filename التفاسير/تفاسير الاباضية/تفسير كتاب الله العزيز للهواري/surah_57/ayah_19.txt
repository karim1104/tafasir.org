قال تعالى :  {  وَالذِينَ آمَنُواْ بِاللهِ وَرُسُلِهِ أُوْلَئِكَ هُمُ الصِّدِّيقُونَ } أي :  صدِّقوا بما جاء من عند الله ،   وعملوا بما صدّقوا به {  وَالشُّهَدَآءُ عِندَ رَبِّهِمْ } أي :  الذين يقاتلون في سبيل الله .  <br />
وتفسير الشهداء :  يشهدون كرامة الله ،   في تفسير الحسن .   وقد ترجى الشهادة لأقوام لم يُقتلوا في سبيل الله سيُلحقهم الله بالشهداء .  <br />
ذكروا عن رسول الله صلى الله عليه وسلم أنه قال لأصحابه يوماً :   "  ما الشهيد عندكم ؟  قالوا :  القتيل في سبيل الله .   فقال :  إن شهداء أمتي إذاً لقليل .   ثم قال :  القتيل في سبيل الله شهادة ،   والبطن شهادة ،   والطاعون شهادة ،   والغرق والحرق شهادة ،   النِّفاس شهادة ،   والسّلّ شهادة  " {[3817]} .  <br />
ذكروا عن الحسن قال :  قال رسول الله صلى الله عليه وسلم :   "  ما بين حياة الشهيد في الدنيا وحياته في الآخرة إلا كمضغ تمرة  " {[3818]} .  <br />
ذكروا أن رسول الله صلى الله عليه وسلم قال :   "  إن الشهيد لا يجد ألم القتل إلا كما يجد القرصة  " {[3819]} .  <br />
ذكروا عن الحسن قال قال رسول الله صلى الله عليه وسلم :   "  من سأل الشهادة صادقاً من قِبَل نفسه فله أجر الشهيد وإن مات على فراشه " {[3820]} .  <br />
ذكروا عن الحسن أنه قرأ هذه الآية :  {  الَّذِينَ آمَنُوا بِاللهِ وَرُسُلِهِ أُوْلَئِكَ هُمُ الصِّذِّيقُونَ وَالشُّهَدَآءُ عِندَ رَبِّهِمْ } فقال :  كل مُّؤمن شهيد وإن مات على فراشه .   [ وتفسير مجاهد في قوله :  ( وَالشُّهَدَاءُ ) :  يشهدون على أنفسهم بالإِيمان ]{[3821]} .  <br />
قال تعالى :  {  لَهُمْ أَجْرُهُمْ وَنُورُهُمْ } قال :  {  وَالَّذِينَ كَفَرُواْ وَكَذَّبُواْ بآيَاتِنَآ أُوْلَئِكَ أَصْحَابُ الْجَحِيمِ } .
